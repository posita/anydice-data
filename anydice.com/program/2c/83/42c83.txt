set "maximum function depth" to 5

function: minimum of R:d{
 result:1@ {R}
}

function: myexploder ADIE:d into highest TAKEN:n from POOL:n of BDIE:d
{
MINA: [minimum of ADIE]
MAXA: [maximum of ADIE]
MINB: [minimum of BDIE]
MAXB: [maximum of BDIE]
result: [bhelper ADIE BDIE]
}

function: bhelper A:n B:d
{
result : A -  (  [ highest TAKEN of POOLd{MINB..MAXB}]/(A^2)  ) 
}


output d20


output [ myexploder d20 into highest 1 from 1 of d{0,31}  ] named "longfoot"

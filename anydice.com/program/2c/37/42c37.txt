\ (F = ROLL) counts how many dice show face F \

function: one pair ROLL:s {
  MAXSAME: 0
  NPAIR: 0
  loop F over {1..6} {
    C: (F = ROLL)
    if C > MAXSAME { MAXSAME: C }
    if C >= 2 { NPAIR: NPAIR + 1 }
  }
  RUN: 0
  CUR: 0
  loop F over {1..6} {
    CUR: (CUR + 1) * ((F = ROLL) >= 1)
    if CUR > RUN { RUN: CUR }
  }
  if (MAXSAME = 2) & (NPAIR = 1) & (RUN < 4) {
    if (1 = ROLL) >= 2 { result: -1 }
    result: 1
  }
  result: 0
}

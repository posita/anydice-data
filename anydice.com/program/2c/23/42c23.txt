  if RESULT <= 4 { result: 1d4 }
  if RESULT <= 6 { result: 1d6 }
  if RESULT <= 8 { result: 1d8 }
  if RESULT <= 10 { result: 1d10 }
  if RESULT <= 12 { result: 1d12 }
  result: 1d20
}

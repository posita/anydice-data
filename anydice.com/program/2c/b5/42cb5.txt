function: dpercent TENS:n ONES:n {
  if TENS = 0 & ONES = 0 { result: 100 }
  result: TENS * 10 + ONES
}

\ Map the 1-100 roll into your four specific numbers and ratios \
function: map ROLL:n to FIRST:n SECOND:n THIRD:n FOURTH:n {
  if ROLL <= 25 { result: FIRST }            \ 25% \ 
  if ROLL <= 67 { result: SECOND }           \ 42% \
  if ROLL <= 90 { result: THIRD }            \ 23% \
  result: FOURTH                             \ 10% \
}

ASTRAGALUSRAREW: {4:10, 1:40, 3:40, 6:10}


function: firstmap ROLL:n {
    if ROLL <= 8 { result: 1 } \ 40% chance \
    if ROLL <= 16 { result: 3 } \ 40% chance \
    if ROLL <= 18 { result: 4 } \ 10% chance \
    result: 6 \ 10% chance \
}


function: secondmap TENS:n ONES:n {
  VAL: (TENS - 1) * 10 + (ONES - 1)
  if VAL < 40 { result: 1 }  \ 40% \
  if VAL < 80 { result: 3 }  \ 40% \
  if VAL < 90 { result: 4 }  \ 10% \
  result: 6                  \ 10% \
}


CUSTOM_TWE: [firstmap d20]
CUSTOM_TEN: [secondmap 1d10 1d10]

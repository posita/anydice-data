function: fifth edition advantage attack ROLL:n {
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*DMG_DICE + DMG_MOD}
if ROLL+TO_HIT >= AC {result: DMG_DICE + DMG_MOD}
result: 0}

DMG_DICE: 1d8+1d6
DMG_MOD: 4
TO_HIT: 7
CRIT: 20

loop AC over {20} {
 output [fifth edition advantage attack [highest 1 of 2d20]] named "[AC] AC"
}

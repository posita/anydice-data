\ Function to check for any matches (Outputs 1 for True, 0 for False) \
function: has doubles A:n B:n C:n {
  result: A = B | A = C | B = C
}

\ Function to see exactly how many dice match (1 = Unique, 2 = Double, 3 = Triple) \
function: match count A:n B:n C:n {
  if A = B & B = C { result: 3 } 
  if A = B | A = C | B = C { result: 2 }
  result: 1 
}

output [has doubles 1d10 1d10 1d10] named "Odds of Any Doubles (1 = Yes, 0 = No)"
output [match count 1d10 1d10 1d10] named "Match Breakdown"

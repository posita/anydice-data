function: ataquedragona TIPO:n {
    if TIPO = -1 {result: [lowest 1 of 2d20] + 11}
    if TIPO = 0 {result: d20 + 11}
    if TIPO = 1 {result: [highest 1 of 2d20] + 11}
}

function: impacto TIRADA:n contra AC:n {
    if TIRADA >= AC {result: 1}
    result: 0
}

function: daniodragona TIRADA:n contra AC:n {
    if TIRADA > AC {
        result: 2d6 + 1d8 + 6
    }
    result: 0
}

function: CANTIDAD:n ataques TIPO:n contra AC:n {
          TOTAL: 0
          loop N over {1..CANTIDAD}{
                    TOTAL: TOTAL + [impacto [ataquedragona TIPO] contra AC]
          }
          result: TOTAL
}

function: CANTIDAD:n danios TIPO:n contra AC:n {
          TOTAL: 0
          loop N over {1..CANTIDAD}{
                    TOTAL: TOTAL + [daniodragona [ataquedragona TIPO] contra AC]
          }
          result: TOTAL
}

CANTIDAD: 3

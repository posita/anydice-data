function: city AROLL:s v BROLL:s {
 ATOTAL: 0
 BTOTAL: 0

 AHITS: 0
 BHITS: 0

 DIFFERENCE: [absolute (#AROLL - #BROLL)]

 loop I over {1..#AROLL} {
  ATOTAL: ATOTAL + I@AROLL
 }
 loop J over {1..#BROLL} {
  BTOTAL: BTOTAL + J@BROLL
 }

 if #AROLL >= #BROLL {
  loop K over {(DIFFERENCE+1)..#BROLL} {
   if K@AROLL > (K-DIFFERENCE)@BROLL {AHITS: AHITS + 1}
   if K@AROLL < (K-DIFFERENCE)@BROLL {BHITS: BHITS + 1}
  }
 } if #AROLL < #BROLL {
  loop K over {(DIFFERENCE+1)..#AROLL} {
   if (K-DIFFERENCE)@AROLL > K@BROLL {AHITS: AHITS + 1}
   if (K-DIFFERENCE)@AROLL < K@BROLL {BHITS: BHITS + 1}
  }
 }

 if ATOTAL > BTOTAL {
  result: #AROLL - BHITS
 }
 if ATOTAL < BTOTAL {
  result: -1 * (#BROLL - AHITS)
 }
 result: 0
}

output [city 4d6 v 3d6]

function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}
function: mean of ROLL:s {
  result: ROLL / 2
}

function: explode N:n {
 if N = 1 {
  result: 1d6
 }
 result: N
}

set "maximum function depth" to 2
function: highestt N:n of A:s B:s C:s D:s{
    result: {1..N}@[sort {A, B, C, D}]
}

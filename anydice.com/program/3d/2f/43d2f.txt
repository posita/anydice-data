\ 1. Funkcja z poprzedniego kroku: podmienia 7 na 1k6 \
function: rzut bazowy WYNIK:n {
  if WYNIK = 7 { result: 1d6 }
  result: WYNIK
}

\ 2. Funkcja grupująca wyniki w kategorie (lokacje) \
\ Możesz dowolnie zmieniać liczby w nawiasach { }, aby pasowały do Twojej tabeli \
function: sprawdz lokacje WYNIK:n {
  if WYNIK = 12 { result: 1 }               \ 1 = Głowa \
  if WYNIK = {3, 4, 10, 11} { result: 2 }   \ 2 = Ręce (Prawa i Lewa) \
  if WYNIK = {5, 9} { result: 3 }           \ 3 = Nogi (Prawa i Lewa) \
  if WYNIK = {6, 8} { result: 4 }           \ 4 = Boczne Korpusy (Prawy i Lewy) \
  if WYNIK = 2 { result: 5 }                \ 5 = Centralny Korpus \
  
  \ Wynik 1 to efekt wyrzucenia jedynki na dodatkowej kości po zastąpieniu siódemki \
  if WYNIK = 1 { result: 6 }                \ 6 = Inne / Wynik z 1k6 \
  
  result: 0 \ Awaryjne zero, gdyby wpisano złą liczbę \
}

\ 3. Wywołanie całości z legendą \
output [sprawdz lokacje [rzut bazowy 2d6]] named "1:Głowa, 2:Ręce, 3:Nogi, 4:Boki, 5:CT, 6:Wynik 1"



function: rmwe DICE:n RANK:n {
result: (DICE + RANK * 4)
}

function: roll DICE:s blunder BB:n at BV:n fail FB:n at FV:n success SB:n at SV:n critical CB:n at CV:n {
result: 0
+ (DICE <= BB) * (BV - FV)
+ (DICE <= FB) * (FV - 0)
+ (DICE >= SB) * (SV - 0)
+ (DICE >= CB) * (CV - SV)
}

function: roll DICE:s pattern PAT:n {
if (PAT = PATTERN_FOUR)   { result: [roll DICE blunder 1 at -1 fail 2  at 0 success 3  at 1 critical 4  at 2] }
if (PAT = PATTERN_SIX)    { result: [roll DICE blunder 1 at -1 fail 3  at 0 success 4  at 1 critical 6  at 2] }
if (PAT = PATTERN_EIGHT)  { result: [roll DICE blunder 1 at -1 fail 4  at 0 success 5  at 1 critical 8  at 2] }
if (PAT = PATTERN_TEN)    { result: [roll DICE blunder 1 at -1 fail 5  at 0 success 6  at 1 critical 10 at 2] }
if (PAT = PATTERN_TWELVE) { result: [roll DICE blunder 1 at -1 fail 6  at 0 success 7  at 1 critical 12 at 2] }
if (PAT = PATTERN_TWENTY) { result: [roll DICE blunder 1 at -1 fail 10 at 0 success 11 at 1 critical 20 at 2] }
if (PAT = SPECIAL_ONE)    { result: [roll DICE blunder 0 at -0 fail 9  at 0 success 10 at 1 critical 13 at 0] }
result: DICE
}

PATTERN_FOUR: 100
PATTERN_SIX: 200
PATTERN_EIGHT: 300
PATTERN_TEN: 400
PATTERN_TWELVE: 500
PATTERN_TWENTY: 600

SPECIAL_ONE: 1

output [roll d20 pattern PATTERN_TWENTY]

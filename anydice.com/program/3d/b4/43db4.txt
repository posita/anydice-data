function: city AROLL:s v BROLL:s {
 ATOTAL: 0
 BTOTAL: 0

 AHITS: 0
 BHITS: 0

 DIFFERENCE: [absolute (#AROLL - #BROLL)]

 loop I over {1..#AROLL} {
  ATOTAL: ATOTAL + I@AROLL
 }
 loop J over {1..#BROLL} {
  BTOTAL: BTOTAL + J@BROLL
 }

 loop K over {DIFFERENCE..[highest of #AROLL and #BROLL]} {
  if K@AROLL > K@BROLL {AHITS: AHITS + 1}
  if K@AROLL < K@BROLL {BHITS: BHITS + 1}
 }

 if ATOTAL > BTOTAL {
  result: #AROLL - BHITS
 }
 if ATOTAL < BTOTAL {
  result: -1 * (#BROLL - AHITS)
 }
 result: 0
}

output [city 4d6 v 3d6]

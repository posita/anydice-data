DC: 16
DIE: d20 - 1d4

MINUSFIVE: DC <= DIE - 5
MINUSFOUR: DC <= DIE - 4
MINUSTHREE: DC <= DIE - 3
MINUSTWO: DC <= DIE - 2
MINUSONE: DC <= DIE - 1
PLUSZERO: DC <= DIE + 0
PLUSONE: DC <= DIE + 1
PLUSTWO: DC <= DIE + 2
PLUSTHREE: DC <= DIE + 3
PLUSFOUR: DC <= DIE + 4
PLUSFIVE: DC <= DIE + 5

function: roll ROLL:s until X:n successes or Y:n failures die DICE:d {
    S: ROLL = 1           \ count successes in current roll \
    F: ROLL = 0           \ count failures in current roll \
    if S >= X | F >= Y { 
        result: S 
    } else if {S, F, X, Y} = {1, 0, 3, 3} {
        result: 3
    } else { 
        result: S + [roll DICE until X-S successes or Y-F failures die DICE] 
    }
}

set "maximum function depth" to 99
output [roll MINUSFIVE until 3 successes or 3 failures die MINUSFIVE] named "MINUS 5"
output [roll MINUSFOUR until 3 successes or 3 failures die MINUSFOUR] named "MINUS 4"
output [roll MINUSTHREE until 3 successes or 3 failures die MINUSTHREE] named "MINUS 3"
output [roll MINUSTWO until 3 successes or 3 failures die MINUSTWO] named "MINUS 2"
output [roll MINUSONE until 3 successes or 3 failures die MINUSONE] named "MINUS 1"
output [roll PLUSZERO until 3 successes or 3 failures die PLUSZERO] named "PLUS 0"
output [roll PLUSONE until 3 successes or 3 failures die PLUSONE] named "PLUS 1"

output [roll PLUSTHREE until 3 successes or 3 failures die PLUSTHREE] named "PLUS 3"
output [roll PLUSFOUR until 3 successes or 3 failures die PLUSFOUR] named "PLUS 4"
output [roll PLUSFIVE until 3 successes or 3 failures die PLUSFIVE] named "PLUS 5"

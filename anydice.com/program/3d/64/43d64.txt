AT:7 ATT:7
DE:9 DEF:9
CR:13 CRI:10
DI:3 DIC:1
SI:5 SID:11
AR:3 ARM:3


function: noneg A:n {
 if A<=0{result:-500}
 else {result:A}
}
function: nonegg A:n {
 if A<=0{result:0}
 else {result:A}
}

function: crit A:n med B:n st C:n{
 if A<0{result:0}
 result: (B+A)dC
}

S: 1d20-1d20+AT-DE
ST: 1d20-1d20+ATT-DEF
S: [noneg S]/CR
ST: [noneg ST]/CRI

output [nonegg[crit S med DI st SI]-AR]
output [nonegg[crit ST med DIC st SID]-ARM]
output S
output ST

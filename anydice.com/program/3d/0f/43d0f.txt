# Funzione che identifica le 10 caselle "vuote" (0, 30, prigione, imprevisti)
function: is_question POS:n {
  if POS=4 | POS=7 | POS=8 | POS=12 | POS=16 | POS=20 | POS=24 | POS=28 | POS=30 | POS=0 | POS>30 { result: 0 }
  result: 1
}

# Funzione del turno per 2 giocatori
function: step STATE:n {
  # Estrazione delle variabili dallo stato numerico (es: 30213021)
  P1: STATE / 1000000
  Q1: (STATE / 10000) - (P1 * 100)
  P2: (STATE / 100) - (P1 * 10000) - (Q1 * 100)
  Q2: STATE - (P1 * 1000000) - (Q1 * 10000) - (P2 * 100)
  
  # Se uno dei due ha già vinto, lo stato si congela
  if P1 >= 30 | P2 >= 30 { result: STATE }
  
  # Turno Giocatore 1
  N_P1: P1 + 1d3
  if N_P1 > 30 { N_P1: 30 }
  N_Q1: Q1 + [is_question N_P1]
  
  # Se il Giocatore 1 arriva a 30, il Giocatore 2 non tira
  if N_P1 = 30 {
     result: (N_P1 * 1000000) + (N_Q1 * 10000) + (P2 * 100) + Q2
  }
  
  # Turno Giocatore 2
  N_P2: P2 + 1d3
  if N_P2 > 30 { N_P2: 30 }
  N_Q2: Q2 + [is_question N_P2]
  
  # Ritorna il nuovo stato aggiornato
  result: (N_P1 * 1000000) + (N_Q1 * 10000) + (N_P2 * 100) + N_Q2
}

FINAL_P2_POS: (S30 / 100) - (FINAL_P1_POS * 10000) - (FINAL_P1_Q * 100)
FINAL_P2_Q: S30 - (FINAL_P1_POS * 1000000) - (FINAL_P1_Q * 10000) - (FINAL_P2_POS * 100)

TOTAL_QUESTIONS: FINAL_P1_Q + FINAL_P2_Q
output TOTAL_QUESTIONS named "Totale carte domanda (2 Giocatori)"

function: ip T:n {
  if T >= 0 {
    result: (T+2)/3
  }
  else {
    result: T/3
  }
}

function: bp R:s at T:n { 
  \ this relies on the fact that sequences are sorted \ 
  if (1@R = 2@R) | (3@R = 2@R) {
    BPQ: [ip T] - 2@R

    if 2@R = 0 & BPQ < 1 {
      BPQ: 1
    }

    if 2@R = 9 & BPQ > 0 {
      BPQ: 0
    }

    result: BPQ
  }
}

loop TARGET over {-1..28} {
  output [bp 3d{0..9} at TARGET] named "FTV: [TARGET], BP"
}

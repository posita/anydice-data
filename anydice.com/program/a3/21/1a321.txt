function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE
 }
 result: DAMAGE
}

AC: d{14..20}
STRENGTH: 3

ATTACK: 8
DAMAGE: 3d6 + STRENGTH
THREAT: 20
CRITICAL: 3
output [d20 vs AC] named "Shortbow"

ATTACK: 3
DAMAGE: 3d6 + STRENGTH + 10
THREAT: 20
CRITICAL: 2
output [d20 vs AC] named "Sharpshooter"

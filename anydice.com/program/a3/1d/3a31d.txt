DIE: {0,0,1,1,2,3}


function: roll ROLL:s {
  ROLLEDSUM: 0
  DICETOROLL: 5
  loop P over {1..5} {
    if P@ROLL = 3 {
      ROLLEDSUM: ROLLEDSUM+3
      DICETOROLL: DICETOROLL-1
    }
    
  }
  result: ROLLEDSUM+DICETOROLLdDIE
}

output 5dDIE named "throw 5d DIE"

output [roll 5dDIE] named "throw 5d DIE one reroll"


\output [roll[roll 5dDIE]] named "throw 5d DIE two reroll"\

\TWO: [roll ONE]
output TWO

THREE: [roll TWO]
output THREE\

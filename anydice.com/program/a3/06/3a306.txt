DIE: {0,0,1,1,2,3}

function: roll ROLL:n with DICETOROLL:n dice{
  ROLLEDSUM: 0
  DICETOROLL: 5
  loop P over {1..5} {
    if P@ROLL = 3 {
      \result: 3+4dDIE\
      ROLLEDSUM: ROLLEDSUM+3
      DICETOROLL: DICETOROLL-1
    }
    
  }
  result: ROLLEDSUM+DICETOROLLdDIE
}

ROLLEDSUM: 0
DICETOROLL: 5

SIMPLE: 5dDIE
output SIMPLE named "throw 5d DIE"

output [roll 5dDIE with 5 dice] named "throw 5d DIE one reroll"


\output [roll[roll 5dDIE]] named "throw 5d DIE two reroll"\

LVL : 14
TICK : 3 \max number of ticks\
DC : 11
ENEMIES : 1
WEAPON : 3d8+8+1d6

Y : (LVL+1)/2
X : Y-1
Z : (X-1)/2

function: two if A:n above B:n {
 if A > B { result: 2 } 
 result: 1
}

\output 3d8+8+1d6\

function: hit A:n B:n{
 if A >= 2*B { result: 2 }
 if A = 20 {result : 2}
 if A >= B {result: 1}
 result: 0
}


if ENEMIES >1{
 output [hit d20 DC]*(1d8+(2*X)d8 + 5+WEAPON)+[hit d20 DC]*(1d8+(2*X)d8 + 5+WEAPON) named "2 targets Amp'd Psy Weapon Lvl [Y]"
}
else{
 output (1d8+(2*X)d8 + 5 +WEAPON) named "Amp'd Psy Weapon Lvl [Y]"
}
output (1d8+(1*X)d8 + 5+WEAPON) named "Psy Weapon (Basic)"
output 1d12+Xd12 +(1d4+X)*dTICK+WEAPON named "Withering Grasp ([TICK] ticks)"
output 2d12 + Xd12 +(1d4+X)*dTICK+WEAPON named "Shocking Grasp vs Metal ([TICK]t)"
\output 12d6 + ([highest of Y-5 and 0]*2)d6 named "Blood Feast"\
\output 3d8+(Z*2)d8 + ((1+Z)d6)*dTICK named "Acid H+[Z]"\
\output (12d10 + ([highest of Y-6 and 0]*2)d10)/[two if d20 above DC] named "Disintegrate"\
if ENEMIES >3{
 output [two if d20 above DC]*(4d6 + ((Y-2)*2)d6) named "Two target scorching"
}
if LVL>14 {
 output 10d8+ ([highest of Y-8 and 0]*2)d8 + LVL*2 +WEAPON named "Polar Ray (2*[LVL])"
}

function: reroll R:n under N:n {
   if R < N { result: 1d20 } else {result: R}
}

function: rerolladv ONE:n and TWO:n under N:n {
   if ONE < N {
result: [highest of 1d20 and TWO]
   } 
else if TWO < N {
result: [highest of ONE and 1d20]
}
else {
 result: [highest of ONE and TWO]
}
}

function: rerolldis ONE:n and TWO:n under N:n {
   if ONE < N {
result: [lowest of 1d20 and TWO]
   } 
else if TWO < N {
result: [lowest of ONE and 1d20]
}
else {
 result: [lowest of ONE and TWO]
}
}

LUCK: [reroll 1d20 under 2]
LUCKY_ADV: [rerolladv 1d20 and 1d20 under 2]
LUCKY_DIS: [rerolldis 1d20 and 1d20 under 2]

function: truncate A:n to two {
 if A > 2 { result: 2 }
 result: A
}

output [truncate 4d(LUCK>=18) to two] named "4 Wave attacks Halfling Champion criticals"
output [truncate 8d(LUCK>=18) to two] named "8 Wave attacks Halfling Champion criticals"

output [truncate 2d(1@3d20>=19) to two] named "2 Wave attacks Elven Accuracy Hexblade Foresight criticals"

output [truncate 4d(LUCKY_ADV>=18) to two] named "4 Wave ADV attacks Halfling Champion criticals"
output [truncate 8d(LUCKY_ADV>=18) to two] named "8 Wave ADV attacks Halfling Champion criticals"

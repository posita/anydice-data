\Tests vs Armor\
AR: 0
DIR: -1         \to chose highest or lower\
function: rollinjury MODIFIED:n for NUMBER:n of DICE:d{
 if MODIFIED=1 { result: {1..NUMBER}@DICE }
 if MODIFIED=-1 { result: {(#DICE - NUMBER + 1)..#DICE}@DICE }
 result: 0
}

output [highest 2 of 2d6]+AR named "No Mods"
output [rollinjury DIR for 2 of 3d6]+AR named "-/+1DICES"
output [rollinjury DIR for 2 of 4d6]+AR named "-/+2DICES"
output [rollinjury DIR for 2 of 5d6]+AR named "-/+3DICES"
output [rollinjury DIR for 2 of 6d6]+AR named "-/+4DICES"

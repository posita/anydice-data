function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

\output 5d{1,1,1,1,10,10}\
output 3d[reroll {10} on d{1,1,1,0,10,10}] named "Resist: 3d"
output 4d[reroll {10} on d{1,1,1,0,10,10}] named "Resist: 4d"
output 5d[reroll {10} on d{1,1,1,0,10,10}] named "Resist: 5d"

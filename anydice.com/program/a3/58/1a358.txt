function: reroll R:n under N:n {
   if R < N { result: 1d20 } else {result: R}
}
function: rerolladv ONE:n and TWO:n under N:n {
   if ONE < N {
       result: [highest of 1d20 and TWO]
   } 
   else if TWO < N {
       result: [highest of ONE and 1d20]
   }
   else {
       result: [highest of ONE and TWO]
   }
}

}

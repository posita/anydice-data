SKILL: 7

function: abs N:n {
  if N < 0 { result: -N }
  else { result: N }
}

SIDES: 10
KEEP: 3
function: k S:n {
  if S < 0 {
    result: [lowest KEEP of (KEEP + [abs S])dSIDES]
  } else {
    result: [highest KEEP of (KEEP + S)dSIDES]
  } 
}

loop N over {-SKILL..SKILL}{
  \M: -3
  output [kM] >= [kN] named "[M]>=[N]"\
  output [kN] named "[N]"
}

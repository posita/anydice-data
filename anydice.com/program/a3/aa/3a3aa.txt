  COUNT: 0
  loop DIE over ROLL {
    if DIE >= 6 { 
      if DIE < 10 { COUNT: COUNT + 1 }
      if DIE >= 10 { COUNT: COUNT + 2 }
    }
  }
  result: COUNT
}

SWORD: 1d8 + 1d6 + 4
GUN: 1d10 + 2 + 4
SNEAK: 4d6

TARGET_AC: 20

SWORD_HIT: 6
GUN_HIT: 11

SWORD_HIT_ODDS: (21 - (TARGET_AC - SWORD_HIT))
GUN_HIT_ODDS: (21 - (GUN_HIT - TARGET_AC))

SWORD_ONE_DAMAGE: ((SWORD + SNEAK) * SWORD_HIT_ODDS / 20)
SWORD_TWO_DAMAGE: ((SWORD + SNEAK) * SWORD_HIT_ODDS / 20)
GUN_ONE_DAMAGE: ((GUN + SNEAK) * GUN_HIT_ODDS / 20)
GUN_TWO_DAMAGE: ((GUN + SNEAK) * GUN_HIT_ODDS / 20)
GUN_THREE_DAMAGE: ((GUN + SNEAK) * (GUN_HIT_ODDS - 5) / 20)

output ((SWORD_ONE_DAMAGE + SWORD_TWO_DAMAGE) + (GUN_ONE_DAMAGE + GUN_TWO_DAMAGE + GUN_THREE_DAMAGE)) named "Average Damage w/ Sneak Attack"


SWORD_ONE_DAMAGE_UNDEAD: ((SWORD + (SNEAK / 2)) * SWORD_HIT_ODDS / 20)

GUN_THREE_DAMAGE_UNDEAD: ((GUN + (SNEAK / 2)) * (GUN_HIT_ODDS - 5) / 20)

output ((SWORD_ONE_DAMAGE_UNDEAD + SWORD_TWO_DAMAGE_UNDEAD) + (GUN_ONE_DAMAGE_UNDEAD + GUN_TWO_DAMAGE_UNDEAD + GUN_THREE_DAMAGE_UNDEAD)) named "Average Damage w/ Half Sneak Attack"

output ((SWORD_ONE_DAMAGE_FIRST + SWORD_TWO_DAMAGE_FIRST) + (GUN_ONE_DAMAGE_FIRST + GUN_TWO_DAMAGE_FIRST + GUN_THREE_DAMAGE_FIRST)) named "Average Damage w/ Sneak Attack & Hit Bonus on First Round"


SWORD_HIT_ODDS_BONUS: (21 - (TARGET_AC - SWORD_HIT) + 5)
GUN_HIT_ODDS_BONUS: (21 - (GUN_HIT - TARGET_AC) + 5)

SWORD_ONE_DAMAGE_BONUS: ((SWORD + SNEAK) * SWORD_HIT_ODDS_BONUS / 20)

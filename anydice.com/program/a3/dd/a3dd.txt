function: empower COUNT:n of DIE:d by NUMBER:n {
 SEQUENCE : {}
 loop P over {1..COUNT} {
   SEQUENCE: [sort {DIE, SEQUENCE}]
 }
 loop R over {1..NUMBER} {
   AVERAGE: [maximum of DIE]/2
   if #SEQUENCE@SEQUENCE < AVERAGE {
     SEQUENCE: [set element #SEQUENCE in SEQUENCE to [1d6]]
     SEQUENCE: [sort SEQUENCE]
   }
 }
 result: SEQUENCE
}

function: set element I:n in SEQ:s to N:n {
  NEW: {}
  loop J over {1..#SEQ} {
    if I = J { NEW: {NEW, N} }
    else { NEW: {NEW, J@SEQ} }
  }
  result: NEW
}

output [empower 8 of 1d6 by 4]

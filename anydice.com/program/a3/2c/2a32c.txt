FAIL: 0
PASS: 1

function: ip T:n {
  if T >= 0 {
    result: (T+2)/3
  }
  else {
    result: T/3
  }
}

function: bp R:s at T:n { 
  \ this relies on the fact that sequences are sorted \
  S: 1@R + 2@R + 3@R

  if S < T {
    result: PASS
  }
  else {
    result: FAIL
  }
}

loop TARGET over {-1..28} {
  output [bp 3d{0..9} at TARGET] named "FTV: [TARGET], BPQ"
}

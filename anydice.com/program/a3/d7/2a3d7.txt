Z : (X-1)/2

 if A > B { result: 2 }
 result: 1
}

 output [two if d20 above DC]*(1d8+(2*X)d8 + 5) named "2 targets Amp'd Psy Weapon Lvl [Y]"
}
else{
 output (1d8+(2*X)d8 + 5) named "Amp'd Psy Weapon Lvl [Y]"
}
output (1d8+(1*X)d8 + 5) named "Imaginary Weapon"
output 1d12+Xd12 +(1d4+X)*TICK named "Withering Grasp [TICK] ticks"
output 2d12 + Xd12 +(1d4+X)*TICK named "Shocking Grasp vs Metal"
output 12d6 + ([highest of Y-5 and 0]*2)d6 named "Feast"
output 3d8+(Z*2)d8 + ((1+Z)d6)*TICK named "Acid H+[Z]"
output (12d10 + ([highest of Y-6 and 0]*2)d10)/[two if d20 above DC] named "Disintegrate"

 output [two if d20 above DC]*(4d6 + ((Y-2)*2)d6) named "Two target scorching"
}

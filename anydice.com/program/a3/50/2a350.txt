function: vtmresult ROLL:s DIFFICULTY SPECIALIZED   {
 ONES: (ROLL = 1)
 TENS: (ROLL = 10)
 SUCCESSES: (ROLL >= DIFFICULTY) + (SPECIALIZED * TENS)
 NETSUCCESSES: (ROLL >= DIFFICULTY) - (ROLL = 1) + (SPECIALIZED * (ROLL = 10))

 if (ONES > 0) & (SUCCESSES < 1) { result: -1}
 if NETSUCCESSES <= 0 {result: 0}
 if NETSUCCESSES >= 5 {result: 5}
 result: NETSUCCESSES
}

output [vtmresult 7d10 7 0]
output [vtmresult 9d10 7 0]

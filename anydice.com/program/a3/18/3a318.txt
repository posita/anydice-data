set "explode depth" to 3
function: evaluate ROLL:s {
  if [count 6 in ROLL] >= 1 {
    result: ROLL + [explode d6] -1
  } else {
    result: ROLL
  }
}

\ In the example below a commander (-3 damage) with 8 HP but wearing a shroud (+2hp) and +2 undying (+2hp) is hit with murdering winter (8 damage) in cold 3 (+6 damage) but with +5 CR (-6 damage) \
output ([evaluate 2d6]+8+2+2)-([evaluate 2d6]+8+6-3-6)

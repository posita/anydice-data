function: roll A:s damage D:d against AC:n with bonus B:n {
  \ crit \
  if ((A = 20) & A+B >= AC) | A+B >= AC+10 {
    result: D * 2
  }
  \ hit \
  if (A + B) >= AC { 
    result: D
  }
  \ miss \
  result: 0
}

output [roll 1d20 damage 1d6 + 4 against 10 with bonus 7] + [roll 1d20 damage 1d6 + 4 against 10 with bonus 3] named "Agile"

output [roll 1d20 damage 1d10 + 4 against 10 with bonus 7] + [roll 1d20 damage 1d10 + 4 against 10 with bonus 2] named "Strong"

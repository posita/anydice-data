function: reroll R:n under N:n {
   if R < N { result: 1d20 } else {result: R}
}

function: rerolladv ONE:n and TWO:n under N:n {
   if ONE < N {
result: [highest of 1d20 and TWO]
   } 
else if TWO < N {
result: [highest of ONE and 1d20]
}
else {
 result: [highest of ONE and TWO]
}
}
LUCK: [reroll 1d20 under 2]
LUCKY_ADV: [rerolladv 1d20 and 1d20 under 2]

output [lowest of 4d(LUCK>=18) and 2] named "4 Wave attacks Halfling Champion criticals"
output [lowest of 8d(LUCK>=18) and 2] named "8 Wave attacks Halfling Champion criticals"

output [lowest of ( (7d(1@3d20>=20)) + (2d20>=20) ) and 2] named "9 Wave attacks Samurai Fighting Spirit Elven Accuracy"
output [lowest of 2d(1@3d20>=19) and 2] named "2 Wave attacks Elven Accuracy Hexblade Foresight criticals"

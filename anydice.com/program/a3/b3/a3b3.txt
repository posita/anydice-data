function: X miscast SEQUENCE:s {
if (6 = SEQUENCE)>1{result:(d3+1)*X}
else{result:{}}}
function: X amnesia SEQUENCE:s {
if (6 = SEQUENCE)>1{result:((d3+1)*X >9)*(d2-1)}
else{result:{}}}
function: X catastrophicdetonation SEQUENCE:s {
if (6 = SEQUENCE)>1{result:((d3+1)*X >14)*(d2-1)}
else{result:{}}}
function: X breachintheveil SEQUENCE:s {
if (6 = SEQUENCE)>1{result:((d3+1)*X >19)*(d2-1)}
else{result:{}}
}
loop DICE over {2..5} {
 output [DICE miscast DICE d6] named "Miscast Roll for [DICE]d6"
 output [DICE amnesia DICE d6] named "Amnesia for [DICE]d6"
 output [DICE catastrophicdetonation DICE d6] named "Catastrophic Detonation for [DICE]d6"
 output [DICE breachintheveil DICE d6] named "Breach in the Veil for [DICE]d6"
}

function: attack ROLL:n {
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*DMG_DICE + DMG_MOD}
if ROLL+TO_HIT >= AC {result: DMG_DICE + DMG_MOD}
result: 0}

DMG_DICE: 5d6
DMG_MOD: 8
TO_HIT: 10
CRIT: 20

loop AC over {10..25} {
 output [attack d20] named "psionic shot - [AC] AC"
}

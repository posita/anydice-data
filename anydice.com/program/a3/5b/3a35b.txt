function: roll TIMES:n die SIDES:n count SUCC:n {
  SUCCESSES: 0
  DIE: 1dSIDES
  loop N over {1..TIMES} {
    SUCCESSES: SUCCESSES + [DIE success on SUCC]
  }
  result: SUCCESSES
}

function: N:n success on SUCC:n {
  SUCCESSES: 0
  if N >= SUCC { SUCCESSES: 1 }
  result: SUCCESSES
}

output [roll 4 die 6 count 5] named "roll nd6 count 5+"

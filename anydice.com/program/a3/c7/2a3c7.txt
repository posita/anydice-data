
function: burst DC save SVROLL:s ag BONUS doing DMG:s plus DMGBON {
    DAMAGE: DMG + DMGBON   
    if (SVROLL+BONUS) < DC {
    result: DAMAGE
    }
    else
    { result: (DAMAGE)/2 }
}

function: rapid DC save SVROLL:s ag BONUS doing DMG:s plus DMGBON {
    DAMAGE: DMG + DMGBON   
    if (SVROLL+BONUS) < DC {
    result: DAMAGE*2
    }
    else
    { result: 0 }
}

function: attack ROLL:s w ATBON ag AC doing DMG:s plus DMGBON {
    if AC < (ROLL+ATBON) {
        if ROLL = 20 {
            result: DMG+DMG+DMGBON
        }
        result: DMG+DMGBON
    }
    else 
    { result: 0 }
}

function: blades ROLL:s { result: (1@ROLL>=4) + (1@ROLL=6) + (2@ROLL=6) }
function: ROLL:s dc N:n {
  loop X over ROLL {
   if X >= N & (ROLL = X) > 3 { result: 3 }
   if X >= N & (ROLL = X) > 2 { result: 2 }
   if X >= N & (ROLL = X) > 1 { result: 1 }
  }
  result: 0
}

output [blades 2@2d6] named "blades 0d"
loop P over {1..7} { output [blades Pd6] named "blades [P]d" }
loop P over {2..14}{
 output [Pd6 dc 1] named "ogre [P]d"
}

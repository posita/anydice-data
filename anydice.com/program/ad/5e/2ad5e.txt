function: attack A:n {
  result: 1d20 + 2 + ((A < 0) * (-A) + (A > 0))@Ad6
}

function: mod N:n by D:n {
  result: N - D * (N / D)
}

function: handcannon A:n with DH:n dice K:n {
  D: DH / 2
  H: DH - D * 2
  result: D * 2 + [handdam [attack A] with H dice K]
}

function: handdam R:n with H:n dice K:n {
  if (R < EVASION) {
    result: H
  } else if (R < 20) {
    result: [highest of Kd6 - ARMOR and 0] * 2 + 1
  } else {
    result: [highest of {1..K}@(2*K)d6 - ARMOR and 0] * 2 + 1
  }
}

function: kinhammerlast A:n with DH:n {
  D: DH / 2
  H: DH - D * 2
  result: 2 * D + [kindamlast [attack (A + H)]]
}

function: kindamlast R:n {
  if (R < EVASION) {
    result: 0
  } else if (R < 20) {
    result: [highest of 2d6 + 2 - ARMOR and 0] * 2
  } else {
    result: [highest of {1,2}@4d6 + 2 - ARMOR and 0] * 2 + 1
  }
}

function: kinhammerfirst R:n {
  if (R < EVASION) {
    result: 0
  } else if (R < 20) {
    result: [highest of 2d6 + 2 - ARMOR and 0] * 2
  } else {
    result: [highest of {1,2}@4d6 + 2 - ARMOR and 0] * 2 + 1
  }
}

function: handlast DC:n {
  D: DC / 2
  C: DC - D * 2
  if C {
    HCF: [handcannon 2 with 0   dice 2]
    HCS: [handcannon 0 with HCF dice 1]
    result: 2 * D + 2 * (HCS / 2)
  } else {
    result: 2 * D + 1
  }
}

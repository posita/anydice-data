function: adv A:n and B:n and C:n {
  if (A>B) {
    result: A+[highest of B and C]
  } else {
    result: B+[highest of A and C]
  }
}
function: dis A:n and B:n and C:n {
  if (A>B) {
    result: B+[lowest of A and C]
  } else {
    result: A+[lowest of B and C]
  }
}

output 2d10
output d10+d12
output d8+d10
output [adv d12 and d10 and d10]
output [dis d8 and d10 and d10]

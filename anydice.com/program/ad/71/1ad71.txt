function: twentysworn X:n over Y:s mod Z{
  if X > ((3 * 1@Y) - (3*Z)) { result: 2 } \ strong \
  if X > ((3 * 2@Y) - (3*Z)) { result: 1 } \ weak \
  result: 0 \ miss \
}
function: ironsworn X:n over Y:s mod Z:n{
  if X+Z > (1@Y) { result: 2 }
  if X+Z > (2@Y) { result: 1 }
  result: 0
}
loop Z over {1..5} {
  output [twentysworn 1d20 over 2d10 mod Z] named "twentysworn -[Z]"
}
loop Z over {1..5} {
  output [ironsworn 1d6 over 2d10 mod Z] named "ironsworn +[Z]"
}

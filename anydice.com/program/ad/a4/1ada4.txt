\ The Dark and the Peculiar \
\ Anydice Code \

function: darkroll ROLL:s { \ pass me a sequence (of dice), which will then run me for all possible results of that pool \
  BEST_COUNT: 0
  loop PICK over {1..6} { \ for each possible pick \
    \ count the result of the pick \
    COUNT: [count PICK in ROLL]
    BEST_COUNT: [highest of COUNT and BEST_COUNT]
  }
  result: BEST_COUNT
}

D: 6
loop P over {1..9} {
  output [darkroll PdD] named "[P]d[D]"
}

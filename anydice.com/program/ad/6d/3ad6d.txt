function: dc ROLL:n threshold THRESHOLD:n {
 if ROLL <= 1 { result: 1 + [dc d6 threshold THRESHOLD] }
 result: ROLL <= THRESHOLD
}

DCH: [dc d6 threshold 3]
DCM: [dc d6 threshold 2]
DCL: [dc d6 threshold 1]

output 4dDCH named "4 dice threshold 3"
output 3dDCM named "3 dice threshold 2"
output 2dDCL named "2 dice threshold 1"

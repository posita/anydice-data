DIE: d100
SCALE: 10
ATTRIBUTE: 5 * d{0..10}
SKILL: d{0..5}

function: check ROLL:n against A:n with S:n {
 if ROLL > A {
  STEPS_OVER: (ROLL - A + SCALE - 1) / SCALE
  result: [highest of S - STEPS_OVER and -1]
 } else {
  result: S
 }
}

function: roll A:n with S:n {
 result: [check DIE against A with S]
}

output [roll ATTRIBUTE with SKILL]

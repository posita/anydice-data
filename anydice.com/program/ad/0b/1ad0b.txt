\ Quantum Shielding \

\ N is the damage value, Q is the modifier (-1) if Quantum Deflection is active \

Q:-1

function: qs DIE:s T:s {
	if DIE + Q < T {result: 0}
	if DIE + Q >= T {result: T}
}

loop N over {6, 5, 4, 3, 2, 1} {
output [qs 1d6 N] named "[N]"
}

output [qs 1d6 1d6] named "1d6"

output [qs 1d6 1d3] named "1d3"

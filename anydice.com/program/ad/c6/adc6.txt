function: reroll gs damage of D:s {
if 1@D <= 2 {result: 1d6 + 2@D} 
else if 2@D <= 2 {result: 1@D + 1d6} 
else if {1,2}@D <= 4 {result: 2d6} 
 
}

function: reroll ga damage of D:s {
if 1@D <= 2 {result: 1d12 }
else {result: D } 
 
}

function: reroll pa damage of D:s {
if 1@D <= 2 {result: 1d10 }
else {result: D } 
}


function: reroll pam bonus damage of D:s {
if 1@D <= 2 {result: 1d4 }
else {result: D } 
}

loop STR over {1..5} { 

GREATSWORD: [reroll gs damage of 2d6] + STR
GREATAXE: [reroll ga damage of 1d12] + STR
POLEARM: [reroll pa damage of 1d10] + STR
POLEARMMASTER: [reroll pam bonus damage of 1d4] + STR

output 2d6 named "2d6 + [STR]"
output GREATSWORD named "gwm GS + [STR]"
output GREATAXE named "gwm GA + [STR]"
output POLEARM named "gwm PA + [STR]"
output POLEARMMASTER named "gwm PA master + [STR]"
output POLEARMMASTER + POLEARM named "PA w/ bonus attack + [STR]"

}

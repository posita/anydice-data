function: target TARGET:n reroll REROLL:n VALUE:n {
  if VALUE = REROLL {
    result: 1 + [target TARGET reroll REROLL BASEDIE]
  }
  if VALUE  < TARGET {
    result: 1
  }
  result: 0
}

loop N over {1..6}{
  DIE: [target Y reroll N BASEDIE]
  output NdDIE named "Roll [N] Keep over [Y]"
}

function: dice DICE:n skill SKILL:n spec SPEC:n  {
 if SPEC > 2 {
  DICE: DICE + 1
  SPEC: 2
 }

 ROLL: DICEd10
 S: [count {1..SKILL} in ROLL]
 if SPEC > 0 {
  S: S + [count {1..SPEC} in ROLL]
 }
 
 F: [fumbles in ROLL]
 S: S - F
 C: [crits in ROLL]
 S: S + C

 
 
 result: S
}

function: fumbles in DICE:s {
 F: [count {10} in DICE] - 1
 if F < 0 { F: 0}
 result: F
}

function: crits in DICE:s {
 C: [count {1} in DICE] - 1
 if C < 0 { C: 0}
 result: C
}
output [dice 4 skill 5 spec 0]

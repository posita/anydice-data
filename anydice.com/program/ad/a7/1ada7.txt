function: ptba ROLL:s plus MOD:n {
  TOTAL: 1@ROLL + 2@ROLL + MOD
  RESULT: -1
  if TOTAL <= 6 { \ 6- \
    RESULT: 0
  }
  if TOTAL >=7 & TOTAL <= 9 { \ 7-9 \
    RESULT: 1
  }
  if TOTAL >=10 { \ 10+ \
    RESULT: 2
  }
  result: RESULT
}

loop B over {-2..4} {
  output [ptba 2d6 plus B] named "roll+ [B]"
}

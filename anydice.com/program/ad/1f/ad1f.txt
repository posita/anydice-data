
CRITICALHIT:2
HIT:1

EBLASTFLAG:0
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 | ROLL = 19 {
   result: CRITICALHIT 
  
 }
 if ROLL+ATTACK < DEFENSE {if MISSCOUNT = 0{ if EBLASTFLAG=0{ ATTACK:ATTACK+10 MISSCOUNT:1}}}
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}


function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}


function: spattack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
   result: CRITICALHIT }
  
 }

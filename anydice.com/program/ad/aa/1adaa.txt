\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 5
DMGBONUS: 1d8
PROFICIENCY: 6
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 4d10 + DMGBONUS
CRITDAMAGE: 8d10 + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Firebolt"

DMGBONUS: 0
PROFICIENCY: 6
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 3d8 + DMGBONUS
CRITDAMAGE: 6d8 + DMGBONUS

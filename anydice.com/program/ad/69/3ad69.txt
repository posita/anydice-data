
function: attack ROLL:n check AC:n  SEC:n{
if ROLL=1 {result:0}
if ROLL>=18 {
result: 2d4+4d8+7+[sharpshooter SEC check AC 1]
}
ATK: ROLL+17
if ATK>=AC {
result: 1d4+2d8+7+[sharpshooter SEC check AC 1]
}
result:0
}




function: sharpshooter ROLL:n check AC:n  SEC:n{
if ROLL=1 {result:0}
if ROLL>=18 {
result: 2d4+4d8+7+10+[sharpshooter SEC check AC 1]
}
ATK: ROLL+17-5
if ATK>=AC {
result: 1d4+2d8+7+10+[sharpshooter SEC check AC 1]
}
result:0
}



function: comparison DIEA:n and DIEB:n vs AC:n {
result: [sharpshooter DIEA check AC DIEB] - [attack DIEA check AC DIEB]
}


output 4d[comparison 1d20 and 1d20 vs 15]

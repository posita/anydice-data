\ Attack Sequence \

\ Hit Values \
A: 20 	\ # Attacks \
H: 3 	\ Hit Roll Target \
W: 5 	\ Wound Roll Target \
S: 5 	\ Save Roll Target \
D: d3 	\ Damage \
F: 7	\ Feel No Pain Roll Target \


function: hit TO_HIT:s {
	result: TO_HIT >= H
}

function: wound TO_WOUND:s {
	result: TO_WOUND >= W
}

function: save TO_SAVE:s {
	result: d{0,1,1,1,3,3}
}

function: gate HR:s WR:s SR:s D:s {
	result: HR * WR * SR * D
}

function: fnp DICE:s{
	if DICE < F {result: 1}
	result: 0
}

output Ad[gate [hit 1d6] [wound 1d6] [save 1d6] D]d[fnp  1d6] named "[A] Attacks, [H]+ To Hit, [W]+ To Wound, [S]+ Save, [D] Damage, [F]+ FNP"

SNS: 10
SPD: 10
SNSB: ((10*(SNS-10)))/20
SPDB: ((10*(SPD-10)))/20
EVA: 10 + ((SNSB + SPDB)/2)
EVAK: 10 + ((SNSB + ((SPD-11)/2))/2)
EPAV: 10 + (((SNS-12)/2) + ((SPD-12)/2)/2)
EVAT: 10 + (((SNS-13)/2) + ((SPD-13)/2)/2)

\total damage values\
output (1d20>EVA)*(2d8) named "shieldless dmg"
output (1d20>EVA-1)*(2d8-1) named "buckler dmg"
output (1d20>EVA-2)*(2d8-2) named "targe dmg"
output (1d20-1d6>EVA-3)*(2d8) named "heater dmg"
output (1d20-[highest 1 of 2d6]>EVAK-3)*(2d8) named "kite shield dmg"
output (1d20-[highest 1 of 2d6]>EPAV-3)*(2d8-1) named "pavise dmg"
output (1d20-[highest 1 of 3d6] > EVAT-3)*(2d8-1) named "tower shield dmg"

output 0 named "-"

\hit rates\
output (1d20>EVA) named "shieldless hit rate"
output (1d20>EVA-1) named "buckler hit rate"

output (1d20-1d6>EVA-3) named "heater hit rate"
output (1d20-[highest 1 of 2d6]>EVAK-3) named "kite shield hit rate"
output (1d20-[highest 1 of 2d6]>EPAV-3) named "pavise hit rate"
output (1d20-[highest 1 of 3d6] > EVAT-3) named "tower shield hit rate"

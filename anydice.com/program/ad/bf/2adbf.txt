MISS: 0
HIT: 1

function: attacka ROLL:n vs DEFENSE:n {
 if ROLL = 2 { result: MISS }
 if ROLL + FIRSTATTACK >= DEFENSE { result: HIT }
 result: MISS
}


function: damage for ATTACKARESULT:n {
 if ATTACKARESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}
 

function: attacka {
 result: [damage for [attacka 2d6 vs DEFENSE]]
}

DEFENSE: d{7..10}

FIRSTATTACK: 2

HITDAMAGE: 2
MISSDAMAGE: 0
output [attacka] named "Standard"

FIRSTATTACK: 4
HITDAMAGE: 1
MISSDAMAGE: 0
output [attacka] named "Light"

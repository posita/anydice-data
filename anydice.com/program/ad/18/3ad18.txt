function: roll ROLL:s{
  result: ROLL >= 6
}

\
loop N over {3,4,5,6,7,8} {
  C: N + 3
  output [roll Nd(N+3)] named "[N]d[C]"
}
\

loop N over {6,7,8,9,10,11,12,13}{
  output [roll Nd6] named "[N]d6"
}

loop N over {6,7,8,9,10,11,12,13}{
  output [roll 6dN] named "6d[N]"
}

output [roll 7d7] named "7d7"

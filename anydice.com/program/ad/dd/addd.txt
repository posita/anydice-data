function: mult VALS:s {
    SUM: 0
    loop P over {1..#VALS} {
        SUM: SUM + [over P@VALS]
    }
    result: SUM
}

function: over VAL:n {
    OFF: 2d10 - VAL
    result: [signum OFF]
}

function: signum ROLL:n {
    if ROLL >= 0 {
        result: 0
    } else {
        result: ROLL
    }
}

loop I over {3..20} {
    loop J over {3..20} {
        if (I <= J) {
            output [mult {I, J}] named "[I] & [J]"
        }
    }
}

function: count R:s T:n {
  S: 0
  L: #R
  loop I over {1..L} {
    V: I@R
    if V >= T {S: S+1}
  }
  result: S
}

function: crit S:n {
  if S > 1 {result: S-1}
  result: 0
}


loop T over {4..12} {
  loop S over {1..5} {
    loop Q over {1..5} {
      D: Q*2+2
      if T<=D {
        output [count SdD T] named "[S] d [D] vs [T]"
      }
    }
  }
}

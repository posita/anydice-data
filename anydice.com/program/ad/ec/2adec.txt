AC:15
SPA:8
MOD:4
DC:16

function: success ROLL:n AGAINST:n CRIT:n {
  if ROLL >= AGAINST {
    if ROLL = CRIT {
      result:2
    }
    else {
      result:1
    }
  }
  else {
    result:0
  }
}

function: damage ROLL:n {
  if ROLL = 1 {
    result:2
  }
  else {
    result:ROLL
  }
}

function: attack HD:n DD:d {
  HIT:[success HD AC 20]
  if HIT = 0 {
    result:0
  }
  if HIT = 1 {
    result:[damage DD]
  }
  if HIT = 2 {
    result:[damage DD] + [damage DD]
  }
}

output
  [attack d20 7d8] + MOD

function: successes TN:n ROLL:s {
  SUCCESSES: {}
  loop P over {TN..12} {
    if P=(P/2)*2 {
      SUCCESSES: {SUCCESSES, P}
    }
  }

  R: {}
  loop P over {1..#ROLL} {R: {R, P@ROLL}}

  result: [count SUCCESSES in R]
}

output [successes 2 4d12] named "TN=2"
output [successes 4 4d12] named "TN=4"
output [successes 6 4d12] named "TN=6"
output [successes 8 4d12] named "TN=8"
output [successes 10 4d12] named "TN=10"
output [successes 12 4d12] named "TN=12"

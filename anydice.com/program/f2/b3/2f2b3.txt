function: compare ACTION:n with CHALLENGEDIEA:n and CHALLENGEDIEB:n {
  HIGHER_THAN_ONE: 0
  HIGHER_THAN_BOTH: 0

  if ACTION > CHALLENGEDIEA {
    HIGHER_THAN_ONE: 1
  }
  if ACTION > CHALLENGEDIEB {
    HIGHER_THAN_ONE: 1
  }
  if ACTION > CHALLENGEDIEA & ACTION > CHALLENGEDIEB {
    HIGHER_THAN_BOTH: 1
  }

  if HIGHER_THAN_BOTH {
    result: 2
  } else if HIGHER_THAN_ONE {
    result: 1
  } else {
    result: 0
  }
}


loop I over {1..10} {
  ROLL: I + d6
  THRESHOLD: d10
  output [compare ROLL with THRESHOLD and THRESHOLD] named "1d10 vs 1d6+ [I] (0 is a Miss, 1 is a Weak Hit and 2 is a Strong Hit"
}

function: boom ROLL:s N:n {
if [ROLL contains 6]{ 
 result:[count {6} in ROLL] + [boomhelp (N-[count {6} in ROLL])d6 (N-[count {6} in ROLL])]
  } else{
 if 1@ROLL>= 4 {result: 0} else {result: -1}
}
}

function: boomhelp ROLLX:s NX:n {
 if [ROLLX contains 6]{
  result:[count {6} in ROLLX] + [boomhelp (NX-[count {6} in ROLLX])d6 (NX-[count {6} in ROLLX])]
 } else {result: 0}
}

function: reroll ROLL:s N:n {
if [ROLL contains 6]{ 
 result:[count {6} in (N-[count {6} in ROLL])d6]+[count {6} in ROLL] 
  } else{
 if 1@ROLL>= 4 {result: 0} else {result: -1}
}
}

function: critic ROLL:s {
if [ROLL contains 6]{ 
 result:[count {5..6} in ROLL]
  } else{
 if 1@ROLL>= 4 {result: 0} else {result: -1}
}
}

function: blades ROLL:s {
 if [ROLL contains 6] {
  result: [count {6} in ROLL]
 }else{
 if 1@ROLL >= 4 {result: 0} else {result: -1}
}
}

output [blades 2@2d6] named "0d"
loop P over {4} {

 output [boom Pd6 P] named "[P]boom"
}

ATTACKS: 2
TOHIT: 2
SAVE: 6
DAMAGE: 4

function: save S:n H:n L:n D:n {
  if H = 0 {
    result: 0
  }
  if S > H {
    result: 0
  }
  if S = H {
    result: [lowest of D+H-1 and L]
  }
  result: [lowest of D+H and L]
}

function: compare D:n T:n {
  if D >= T {
    result: 1
  }
  result: 0
}

function: attack M:n H:n S:n D:n {
  HITS: Md[compare 1d6 H]
  result: [save Sd{0, 1} HITS S D]
}

output [attack ATTACKS TOHIT SAVE DAMAGE]
\
output [save SAVEd{0, 1} ATTACKS SAVE]
\

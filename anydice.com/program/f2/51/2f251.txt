function: damage X:n {
  if (X = 1 | X = 2) {
    result: 1d12
  } else {
    result: X
  }
}

function: sword damage X:n {
  if (X = 2) {
    result: 2d6
  } else {
    result: X
  }
}

output [damage 1d12] named "Greataxe Damage With Great Weapon Fighting"
output 1d12 named "Greataxe Damage Without Great Weapon Fighting"

output [sword damage 2d6] named "Greatsword Damage With Great Weapon Fighting"
output 2d6 named "Greatsword Damage Without Great Weapon Fighting"

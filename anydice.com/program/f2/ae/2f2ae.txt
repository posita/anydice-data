function: maxicohelper ROLL:n CONFIRM:n HIGH:n LOW:n {
  if ROLL = -1 | ROLL = 100 {
    if CONFIRM < LOW | CONFIRM > HIGH { result: 0 }
    result: 100
  }
  result: HIGH * 10 + ROLL
}
function: maxico POOL:d {
  ONES: 1d{-1,0..9,100}
  result: [maxicohelper 1d{ONES} 1d{ONES} [highest 1 of POOL] [lowest 1 of POOL]]
}

loop STAT over {1..6} {
  output [maxico (STAT)d{0..9}]
}

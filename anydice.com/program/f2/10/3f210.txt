\============\
\ Input format for anydice.com \
\============\
\ Order of characters in output \
\ - Warrior \
\ - Rogue \
\ - Mage \
\ Grouped by Level and Potion use. \
\ Potions are in this order: \
\ - none \
\ - Power \
\ - Armor \
\ - Luck \

\ Level 1 \
output 9d(1d12) named "Warrior Lv. 1"
output 7d(1d6+(1d10 < 5)*1d20) named "Rogue Lv. 1"
output 6d(2d8+1) named "Mage Lv. 1"

output 9d(1d12+1) named "Warrior Lv. 1 (Power)"
output 7d(1d6+1+(1d10 < 5)*1d20) named "Rogue Lv. 1 (Power)"
output 6d(2d8+2) named "Mage Lv. 1 (Power)"

output 10d(1d12) named "Warrior Lv. 1 (Armor)"
output 8d(1d6+(1d10 < 5)*1d20) named "Rogue Lv. 1 (Armor)"
output 7d(2d8+1) named "Mage Lv. 1 (Armor)"

output 9d(1d12+(1d10 < 2)*1d20) named "Warrior Lv. 1 (Luck)"
output 7d(1d6+(1d10 < 6)*1d20) named "Rogue Lv. 1 (Luck)"
output 6d(2d8+1+(1d10 < 2)*1d20) named "Mage Lv. 1 (Luck)"

\ Level 2 \
output 12d(1d12+(1d10 < 3)*1d20) named "Warrior Lv. 2"
output 9d(1d6+(1d10 < 9)*1d20) named "Rogue Lv. 2"
output 7d(3d8+1+(1d10 < 2)*1d20) named "Mage Lv. 2"

output 12d(1d12+1+(1d10 < 3)*1d20) named "Warrior Lv. 2 (Power)"
output 9d(1d6+1+(1d10 < 9)*1d20) named "Rogue Lv. 2 (Power)"
output 7d(3d8+2+(1d10 < 2)*1d20) named "Mage Lv. 2 (Power)"

output 13d(1d12+(1d10 < 3)*1d20) named "Warrior Lv. 2 (Armor)"
output 10d(1d6+(1d10 < 9)*1d20) named "Rogue Lv. 2 (Armor)"
output 8d(3d8+1+(1d10 < 2)*1d20) named "Mage Lv. 2 (Armor)"

output 12d(1d12+(1d10 < 4)*1d20) named "Warrior Lv. 2 (Luck)"
output 9d(1d6+(1d10 < 10)*1d20) named "Rogue Lv. 2 (Luck)"
output 7d(3d8+1+(1d10 < 3)*1d20) named "Mage Lv. 2 (Luck)"

\ Level 3 \
output 15d(2d12+(1d10 < 3)*1d20) named "Warrior Lv. 3"
output 10d(2d6+(1d10 < 8)*2d20) named "Rogue Lv. 3"
output 6d(3d20+(1d10 < 3)*2d20) named "Mage Lv. 3"

output 15d(2d12+1+(1d10 < 3)*1d20) named "Warrior Lv. 3 (Power)"
output 10d(2d6+1+(1d10 < 8)*2d20) named "Rogue Lv. 3 (Power)"
output 6d(3d20+1+(1d10 < 3)*2d20) named "Mage Lv. 3 (Power)"

output 16d(2d12+(1d10 < 3)*1d20) named "Warrior Lv. 3 (Armor)"
output 11d(2d6+(1d10 < 8)*2d20) named "Rogue Lv. 3 (Armor)"
output 7d(3d20+(1d10 < 3)*2d20) named "Mage Lv. 3 (Armor)"

output 15d(2d12+(1d10 < 4)*1d20) named "Warrior Lv. 3 (Luck)"
output 10d(2d6+(1d10 < 9)*2d20) named "Rogue Lv. 3 (Luck)"
output 6d(3d20+(1d10 < 4)*2d20) named "Mage Lv. 3 (Luck)"

\ IMPORTANT: ALWAYS CHECK "AT LEAST" FOR SUCCESS PROB, AND "AT MOST" FOR FAILURE PROB \
\ NOTE: if intended SUCCESS=0 then REROLL(risk) no die, searching just be success (obtain at least a red die) (NORMAL ROLL)\
\ NOTE: if intended SUCCESS=1 then REROLL(risk) one red d6 searching for a 6 (BONUS) \
\ NOTE: if intended SUCCESS=2 then REROLL(risk) one red d6 searching for a 6 and again, reroll this same d6 searching for 5,6 (CRITIC) \

BLACKS: 1
SUCCESS: 1
REROLLS: 2

function: R:n red vs TOT:n risk REROLLS:n ROLLS:n{  
  PROB: [count {1..R} in TOT]

  if (R < REROLLS+1 | REROLLS = 0) {
    if (SUCCESS = 0) { result: PROB }
    if (SUCCESS = 1) { result: PROB & ROLLSd6 >= 6  }
    if (SUCCESS = 2) { result: PROB & ROLLSd6 >= 11 }
  }
 
  if (PROB=1) {
    result: 1 + [R-1 red vs d(R) risk REROLLS-1 ROLLS+1]
  }

  result: 0
}

loop REDS over {2..4} {
  TOT: REDS+BLACKS
  output [REDS red vs d(TOT) risk REROLLS 0] named "[REDS]/[BLACKS]"
}


output [count {1..4} in d5] & [count {1..3} in d4] & [count {1..2} in d3] \& [count {1} in d2]\ & 2d6>=6

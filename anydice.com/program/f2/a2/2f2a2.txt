ATTACKS: 1
TOHIT: 4
SAVE: 4
DAMAGE: 3

function: save S:n H:n {
  if H = 0 {
    result: 0
  }
  if S > H {
    result: 0
  }
  if S = H {
    result: H-1
  }
  result: H
}

function: compare D:n T:n {
  if D >= T {
    result: 1
  }
  result: 0
}

function: attack M:n H:n S:n D:n {
  HITS: Md[compare 1d6 H]
  result: [save Sd{0, 1} [lowest of HITS+D and S]]
}

\
output [attack ATTACKS TOHIT SAVE DAMAGE]
output [save SAVEd{0, 1} ATTACKS SAVE]
\

loop S over {2, 4, 6} {
  loop D over {0..6} {
    if S >= D {
      loop A over {0..6} {
        if A != 0 | D != 0 | A != D {
          output [attack A TOHIT S D] named "[A] Att [D] Dam [S] Save"
        }
      }
    }
  }
}

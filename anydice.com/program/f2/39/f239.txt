BLACKS: 1

\ NOTE: if intended SUCCESS=0 then REROLL(risk) no die, searching just be success (obtain at least a red die) (NORMAL ROLL)\
\ NOTE: if intended SUCCESS=1 then REROLL(risk) one red d6 searching for a 6 (BONUS) \
\ NOTE: if intended SUCCESS=2 then REROLL(risk) one red d6 searching for a 6 and again, reroll this same d6 searching for 5,6 (CRITIC) \
SUCCESS: 0
REROLLS: 2

function: R:n red vs B:n reroll REROLLS:n {
  PROB: [count {1..R} in B]

  if (REROLLS = 0) {
    if (SUCCESS = 0) { result: PROB}
    if (SUCCESS = 1) { result: PROB & [count {6} in d6]}
    if (SUCCESS = 2) { result: PROB & [count {6} in d6] & [count {5,6} in d6] }
  }
   
  if (PROB = 1) {
    result: 1 + [R-1 red vs d(R) reroll REROLLS-1]
  }

  result: 0 
}


loop REDS over {2..10} {
  TOT: REDS+BLACKS
  output [REDS red vs d(TOT) reroll REROLLS] named "[REDS]/[BLACKS]"
}

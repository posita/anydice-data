OFFENSE: 0
EVASIVENESS: 4

MULTIATTACK: 2
BONE: 1d6
DODGE: BONE + EVASIVENESS

function: attack BONUS:n {
  result: BONE + OFFENSE + BONUS > DODGE & BONE != 1 | BONE = 6
}

output [attack 1] named "Focused Attack"
output [attack 0] named "Standard Attack"
output [attack -1] named "Aimed Attack"

function: multiattack X:n{
  TOTAL: 0
  loop N over {1..X} {
    DAMAGE: [attack 0]
    TOTAL: TOTAL + DAMAGE
  }
  result: TOTAL
}

output [multiattack MULTIATTACK] named "[MULTIATTACK] Attacks"

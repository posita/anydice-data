function: roll ROLL:s {
  \ we assume that ROLL always has exactly two dice; AnyDice sorts them in descending order \
  HI: 1@ROLL
  LO: 2@ROLL
  \ handle the special cases first \
  if LO = 6 & HI = 6 { result: 6 } \ double 10 = "crit success/fail" \
 \ mirror pair = crit success \
  if LO = HI           { result:  6 } \ equal pair = crit failure \ 
  \ at this point, we know there's no equal or mirror pair \
  if [absolute LO - 6] < [absolute HI - 6] { result: HI } else { result: LO }
}

output [roll 2d12]
output 2d6
output 1d12

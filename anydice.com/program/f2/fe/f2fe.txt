function: reroll DICE:n if lt NUM:n {
  if DICE >= NUM {
    result: DICE
  } else {
    result: 4d(d6 > 4)
  }
}

output 4d(d6 > 4) named "base pool"
output 1@2d(4d(d6 > 4)) named "pool with advantage"
output [reroll 4d(d6 > 4) if lt 1] named "reroll on failure"
output [reroll 4d(d6 > 4) if lt 2] named "reroll if 1 or fewer successes"
output [reroll 4d(d6 > 4) if lt 3] named "reroll if 2 or fewer successes"
output [reroll 4d(d6 > 4) if lt 4] named "reroll if 3 or fewer successes"
output [reroll 4d(d6 > 4) if lt 5] named "always reroll"
output 4d(1@2d(d6 > 4)) named "reroll individuals"

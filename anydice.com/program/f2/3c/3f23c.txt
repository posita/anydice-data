CURSE: 1d10
EGO: 1d10


output CURSE + EGO named "Regular output"
output CURSE > EGO named "With Ego"
output EGO > CURSE named "With Curse"
output EGO = CURSE named "With neither"

loop N over {1..10} {
  output CURSE >= (11-N) & EGO >= (11-N) & CURSE = EGO named "Black flash with a skill of [N]"
}

loop N over {1..10} {
  output CURSE >= 11-N | EGO >= 11-N named "Black flash in the zone with a skill of [N]"
}

ATTACKS_NUMBER: 12
ATTACKER_QUA: 4
TARGET_DEF: 3

function: hit roll DICE:s {
 DICE: ATTACKS_NUMBER d6
 HITS: [count {ATTACKER_QUA..6} in DICE ]
 result: [armor save HITS]
}

function: armor save HITS:n {
 DICE: HITS d6
 DEF_LIMIT : TARGET_DEF-1
 result: [count {1..DEF_LIMIT} in DICE]
}

output [hit roll ATTACKS_NUMBER]

\ Empowered Spell Calculator \

          NUM_DICE: 10
          DIE_SIZE: 8       \ Damage = 10d8 \
          CHA_MOD:  5       \ +5 Charisma  \
ELEMENTAL_AFFINITY: 0       \ boolean 1 or 0 \



DAMAGE: NUM_DICEdDIE_SIZE
TOTAL_DAMAGE: 0

function: affinity A:n {
  if !ELEMENTAL_AFFINITY { result: A }
  if A = 1 { result: 2 }
  result: A
}

function: reroll R:n {
  if R > DIE_SIZE/2 { result: R }
  result: [affinity 1dDIE_SIZE]
}

loop X over {1..NUM_DICE} {
  if X <= NUM_DICE - CHA_MOD {
    TOTAL_DAMAGE: TOTAL_DAMAGE + [affinity X @ DAMAGE]
  }
  else {
    TOTAL_DAMAGE: TOTAL_DAMAGE + [reroll X @ DAMAGE]
  }
}

output NUM_DICEdDIE_SIZE named "Normal Damage"
if !ELEMENTAL_AFFINITY {
  output TOTAL_DAMAGE named "Empowered Spell"
}
else {
  output TOTAL_DAMAGE named "Elemental Affinity + Empowered Spell"
}

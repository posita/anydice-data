ATTACKS: 4
TOHIT: 3
SAVE: 4

function: save S:n H:n L:n {
  if H = 0 {
    result: 0
  }
  if S > H {
    result: 0
  }
  if S = H {
    result: H-1
  }
  result: [lowest of H and L]
}

function: compare D:n T:n {
  if D >= T {
    result: 1
  }
  result: 0
}

function: attack M:n H:n S:n {
  HITS: Md[compare 1d6 H]
  result: [save Sd{0, 1} HITS S]
}

output [attack ATTACKS TOHIT SAVE]
\
output [save SAVEd{0, 1} ATTACKS]
\

function: isbetween R {
  result: R > 0
}

function: successprobability for X {
  A: (1d6 + 1 + X) - 1d6
  B: (1d6 + 1 + X - 1) - 1d6
  result: [isbetween A] | [isbetween B]
}

output [successprobability for 0]   named "x = 0"
output [successprobability for 1/2] named "x = 0.5"
output [successprobability for 1]   named "x = 1.0"

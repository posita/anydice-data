 HIDDEN_STRIKES: 0
 RESULT: 0
 loop I over {0..ATTACKS}{
  RESULT: RESULT + [hit [highest 1 of 2d20] over OVER with hidden strikes HIDDEN_STRIKES i I chicken CHICKEN logic]
 }
 result: RESULT
}

 if HIT>=OVER {
  if HIDDEN_STRIKES=0 {
   if HIT>=19 {

    HIDDEN_STRIKES: 1
    result: [normal hidden damage]}}
  else if HIT>=19 {
   result: [crit damage]}
  else {
   result: [normal damage]}}
 else{ result:0 }
}

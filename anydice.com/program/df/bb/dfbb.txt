HIT:   11
AC:    18
DDMG:  4d8
FLAT:  5

ROLL:  1@2d20
TOTAL: ROLL+HIT
CRIT:  ROLL=20
HITS:  TOTAL>=AC

HITZ:   11
ACZ:    18
DDMGZ:  4d8
FLATZ:  5

ROLLZ:  1@1d20
TOTALZ: ROLLZ+HITZ
CRITZ:  ROLLZ=20
HITSZ:  TOTALZ>=ACZ

    result: H * (D + F) + C * D
}

output [attack CRIT HITS DDMG FLAT]

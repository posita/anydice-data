function: TIMES:n count hits above TARGET:n {
         if (TIMES = 1) {
             result: (1d10 >= TARGET)
         } else {
             result: [(TIMES - 1) count hits above TARGET] + (1d10 >= TARGET)
         }
}

output [4 count hits above 9]

output [3 count hits above 7]

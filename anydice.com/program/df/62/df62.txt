function: SEQUENCE:s has NUMBER:n{
  loop I over SEQUENCE{
    result: (I >= NUMBER) > 0
  }
}

function: SEQUENCE:s under BIG:n SMALL:n {
  loop I over SEQUENCE{
    if I < BIG {
      if I < SMALL { result: 2 }
      else { result: 1 }
    }
    else { result: 0 }
  }
}

function: ROLL:n at least VAL:n {
  if ROLL > VAL { result: ROLL }
  result: VAL
}

function: ROLL:n at most VAL:n {
  if ROLL > VAL { result: VAL }
  result: ROLL
}

function: ROLL:n point cost {
  if ROLL <= 13 {
    result: ROLL - 8
  }
  if ROLL <= 15 {
    result: 5 + ((ROLL - 13) * 2)
  }
  if ROLL <= 17 {
    result: 9 + ((ROLL - 15) * 3)
  }
  result: 17
}

function: HIGH:n from ROLLHI:d others ROLLLOW:d {
  result: [highest HIGH of 6 d [ROLLHI point cost]] + [lowest (6 - HIGH) of 6 d [ROLLLOW point cost]]
}

function: at least HI:n of VAL:n using ROLL:d {
  result: [highest HI of 6 d [[ROLL at least VAL] point cost]] + [lowest (6 - HI) of 6 d [ROLL point cost]]
}

function: NUM:n fours minus S:n {
  result: NUM d 4 + 10 - (NUM * S)
}

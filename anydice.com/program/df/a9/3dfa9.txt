function: basic pool score DICE:s {
 BASE: [highest 1 of DICE]
 SIXES: [count {6} in DICE]
 ONES: [count {1} in DICE]
 result: BASE + SIXES - ONES
}

function: mitigation pool score DICE:s {
 SIXES: [count {6} in DICE]
 ONES: [count {1} in DICE]
 FOURS_AND_FIVES: [count {4,5} in DICE]
 
 MITIGATIONS: [lowest of ONES and FOURS_AND_FIVES]
 REMAINING_ONES: ONES - MITIGATIONS
 
 BASE: [highest 1 of DICE]
 
 result: BASE + SIXES - REMAINING_ONES
}

output [basic pool score 5d6] named "5d6 Basic"

output [mitigation pool score 5d6] named "5d6 With Mitigation"

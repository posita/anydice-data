function: roll ROLL:s TH:n {
  TOTAL: 1@ROLL + 2@ROLL
  EDGE: (TOTAL - TH)/2
  COMP: (1@ROLL <= 2) | (2@ROLL <= 2)
  if COMP & TOTAL < TH { result: -3}
  if TOTAL < TH { result: -2}
  if COMP { result: -1}
  result: EDGE
}

loop DIE_SIZE over {4,6,8,10,12}{
  loop TH over {3..15}{
   if 2*DIE_SIZE >= TH {
    output [roll 2dDIE_SIZE TH] named "2d[DIE_SIZE] TH[TH]"
  }
}}

function: reroll N:n tn TN:n {
 if N < TN { result: 0 }
 if N < 10 { result: 1 }
 result: 1 + [reroll d10 tn TN]
}

function: owod N:n tn TN:n {
 if N = 1 { result: -1 }
 if N < TN { result: 0 }
 if N < 10 { result: 1 }
 result: 1 + [reroll d10 tn TN]
}

OWOD: [owod d10 tn 4]

loop N over {8} {
  output [highest of [lowest of NdOWOD and 10] and -4] named "[N]d"
}

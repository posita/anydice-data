function: attack ATTACK:n hit HIT:n {
    result: [count {HIT..6} in ATTACKd6]
}

function: hits HITS:n wound WOUND:n {
    result: [count {WOUND..6} in HITSd6]
}

function: wounds WOUNDS:n vs SAVE:n rend REND:n for DAMAGE:n{
    result: [count {SAVE+[absolute REND]..6+[absolute REND]} in WOUNDSd6] * DAMAGE
}

function: attack ATTACK:n hit HIT:n wound WOUND:n vs SAVE:n rend REND:n for DAMAGE:n {
    result: [wounds [hits [attack ATTACK hit HIT] wound WOUND] vs SAVE rend REND for DAMAGE]
}

function: damage {
    result: [attack ATTACK hit HIT wound WOUND vs SAVES rend REND for DAMAGE]
}

function: crop SEQUENCE:n {                  \ limit the display range between [min;max] \
 if (SEQUENCE < 0){result: 0}
 result: SEQUENCE
}

function: use chart ROLL:n CHART:s{
 if (ROLL >= 4@CHART){result: 4}
 if (ROLL >= 3@CHART){result: 3}
 if (ROLL >= 2@CHART){result: 2}
 if (ROLL >= 1@CHART){result: 1}
 if (ROLL < 1@CHART){result: 0}
 result: 0
}

\********Final Functions to compare Sets of Dices (Optimized at max) **********\

function: find best sets ATT:s DEF:s {
 ATT_SET: [test first ATTdQ]
 DEF_SET: [test first DEFdQ]
 result: [compare sets ATT_SET DEF_SET]
}

function: test first ROLL:s {
  THREES: ROLL=3                          \test the value on each element of the Sequence parameter : optimized\
  TWOS:   ROLL=2
  ONES:   ROLL=1

  \compare the sets to find the more numerous one, choosing the higher if equal\
  if (THREES>0) & (THREES>=TWOS) & (THREES>=ONES) { result: 3}
  if (TWOS>0) & (TWOS>=ONES) { result: 2}
  if (ONES>0) { result: 1}

  result: 0
}

function: compare sets ATT_SET:n DEF_SET:n {
 if (ATT_SET>DEF_SET) { result: ATT_SET-DEF_SET }
 result: 0
}

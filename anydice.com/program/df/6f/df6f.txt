function: GROLL:n gwf GREROLL:d reroll SEQUENCE:s{
  if [SEQUENCE contains GROLL] {
    result: GREROLL
  }
  else {
    result: GROLL
  }
}

DIE_ONE: [d6 gwf d6 reroll {1,2}]
DIE_TWO: [d6 gwf d6 reroll {1,2}]
DIE_THREE: [d6 gwf d6 reroll {1,2}]
DIE_FOUR: [d6 gwf d6 reroll {1,2}]
DIE_FIVE: [d10 gwf d10 reroll {1, 2}]
DIE_SIX: [d10 gwf d10 reroll {1, 2}]
DIE_SEVEN: [d4 gwf d4 reroll {1, 2}]
DIE_EIGHT: [d4 gwf d4 reroll {1, 2}]

\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL = CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + {ACCBONUS} >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 4+2
DMGBONUS: 0
PROFICIENCY: 3
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: DIE_ONE+DIE_TWO + ATTACKMOD + DMGBONUS
CRITDAMAGE: DIE_ONE+DIE_TWO+DIE_THREE+DIE_FOUR + ATTACKMOD + DMGBONUS+6d8+20


function: glaiveattackroll GLAIVEROLL:n vs AC:n {
 if GLAIVEROLL = CRITROLL { result: GLAIVECRITDAMAGE } 
 if GLAIVEROLL = 1 { result: 0 }
 if GLAIVEROLL + PROFICIENCY + GLAIVEATTACKMOD + {ACCBONUS} >= AC { result: GLAIVEDAMAGE }
 result: 0
}

GLAIVEATTACKMOD: 4
GLAIVEDAMAGE: DIE_FIVE + GLAIVEATTACKMOD + DMGBONUS
GLAIVECRITDAMAGE: DIE_FIVE+DIE_SIX + GLAIVEATTACKMOD + DMGBONUS+6d8

function: buttattackroll BUTTROLL:n vs AC:n {
 if BUTTROLL = CRITROLL { result: BUTTCRITDAMAGE } 
 if BUTTROLL = 1 { result: 0 }
 if BUTTROLL + PROFICIENCY + GLAIVEATTACKMOD + {ACCBONUS} >= AC { result: BUTTDAMAGE }
 result: 0
}

BUTTDAMAGE: DIE_SEVEN + GLAIVEATTACKMOD + DMGBONUS
BUTTCRITDAMAGE: DIE_SEVEN+DIE_EIGHT + GLAIVEATTACKMOD + DMGBONUS+6d8

function: scimitarattackroll SCIMITARROLL:n vs AC:n {
 if SCIMITARROLL = CRITROLL { result: SCIMITARCRITDAMAGE } 
 if SCIMITARROLL = 1 { result: 0 }
 if SCIMITARROLL + PROFICIENCY + SCIMITARATTACKMOD + {ACCBONUS} >= AC { result: SCIMITARDAMAGE }
 result: 0
}

SCIMITARATTACKMOD: 4+2
SCIMITARDAMAGE: d6 + SCIMITARATTACKMOD + DMGBONUS
SCIMITARCRITDAMAGE: 2d6 + SCIMITARATTACKMOD + DMGBONUS+6d8

output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "greatsword +2, all attacks assume 2nd-level smite on crits"
output ATTACKTIMESd[glaiveattackroll d20 vs ENEMYAC] named "glaive"
output ATTACKTIMESd[glaiveattackroll d20 vs ENEMYAC]+[buttattackroll d20 vs ENEMYAC] named "glaive, polearm master"
output 3d[scimitarattackroll d20 vs ENEMYAC] named "scimitar of speed"

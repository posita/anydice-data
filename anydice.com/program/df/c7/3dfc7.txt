function: attack X:n bonus B:n ac A:n {
    R : 0
	
    if X+B >= 10+A {
        R: 4
    }
    else if X+B >= A {
        R: 3
    }
    else if X+B <= A-10 {
        R: 1
    }
    else if X+B < A{
        R: 2
    }
	
	if (R!=4 & X=20){
        R: R+1
    }
	if (R!=1 & X=1){
        R: R-1
    }
	
    result : R
}

function: dmgresult X:n CS:d S:d F:d CF:d {
    if X = 4 {
        result: CS
    }
    if X = 3 {
        result: S
    }
    if X = 2{
        result: F
    }
    if X = 1 {
        result: CF
    }
}

function: normalattack X:n bonus B:n ac A:n{
    X : [attack X bonus B ac A]
    result: [dmgresult X (2*(1d10+4) + 1d8) (1d10+4) (0) (0)]
}


function: vicious X:n bonus B:n ac A:n{
    X : [attack X bonus B ac A]
    result: [dmgresult X (2*(2d10+4) + 1d8) (2d10+4) (0) (0)]
}

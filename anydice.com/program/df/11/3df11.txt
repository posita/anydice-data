function: advantage X:n and Y:n and Z:n {
  if X > Y {
    if Y > Z { result: X + Y }
    result: X + Z
  } else {
    if X > Z { result: X + Y }
    result: Y + Z
  }
}

function: disadvantage X:n and Y:n and Z:n {
  if X > Y {
    if X > Z { result: Y + Z }
    result: X + Y
  } else {
    if Y > Z { result: X + Z }
    result: X + Y
  }
}

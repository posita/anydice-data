function: ATTACKS:n attacks over OVER:n chicken CHICKEN:n {
 result: [attack 1 of ATTACKS hit [highest 1 of 2d20] over OVER chicken CHICKEN with 0 and 0]
}

function: attack ATTACK:n of ATTACKS:n hit HIT:n over OVER:n chicken CHICKEN:n with HIDDEN_STRIKES:n and RESULT:n {
 if (ATTACK<=ATTACKS){
  if HIT>=OVER {
   if HIDDEN_STRIKES=0 {
    if HIT>=19 {
     HIDDEN_STRIKES: 1
     RESULT: RESULT + [crit hidden damage]}
    else if ATTACK>=CHICKEN {
     HIDDEN_STRIKES: 1
     RESULT: RESULT + [normal hidden damage]}
    else {
     RESULT: RESULT + [normal damage]}}
   else if HIT>=19 {
    RESULT: RESULT + [crit damage]}
   else {
    RESULT: RESULT + [normal damage]}}
  result: [attack ATTACK+1 of ATTACKS hit [highest 1 of 2d20] over OVER chicken CHICKEN with HIDDEN_STRIKES and RESULT]}
 result: RESULT
}

function: normal damage{ 
 result: 3d6 + 5
}
function: crit damage{ 
 result: 6d6 + 5
}
function: normal hidden damage{ 
 result: 5d6 + 5
}
function: crit hidden damage{ 
 result: 10d6 + 5
}

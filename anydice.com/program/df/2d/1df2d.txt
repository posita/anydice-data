\** 2D6 reroll Advantages **\

function: anychoice DICE:n{
 if DICE < 4 { result: 1d6 }
 result: DICE
}

function: badchoice DICE:n{
 if DICE > 3 { result: 1d6 }
 result: DICE
}

output [anychoice 1d6] + [anychoice 1d6] named "reroll one or both for better"
output 2d6 named "normal"
output [badchoice 1d6] + [badchoice 1d6] named "reroll one or both for worse"

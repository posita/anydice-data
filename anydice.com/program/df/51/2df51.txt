function: success ROLL:s {
 if ROLL <= 6 { result: 0 }
 else if (ROLL >= 7) & (ROLL <= 9) { result: 1 }
 else if ROLL >= 10 { result: 2 }
}

loop N over {-1..3} {
  output [success 2d6+N] named "2d6+[N]"
  \output 2d6+N >= 7 named "2d6+[N] any success"\
}

\output [success [highest 2 of 3d6]] named "2d6 adv"\
\output [success [lowest 2 of 3d6]] named "2d6 dis"\

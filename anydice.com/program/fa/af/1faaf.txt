function: hit TO_HIT:s {
    result: TO_HIT >= BS
}

function: wound TO_WOUND:s {
    if S > T * 2 {
        result: TO_WOUND >= 2
    }
    if S > T {
        result: TO_WOUND >= 3
    }
    if S = T {
        result: TO_WOUND >= 4
    }
    if S < T / 2 {
        result: TO_WOUND >= 6
    }
    if S < T {
        result: TO_WOUND >= 5
    }
}

function: save TO_SAVE:s {
    result: TO_SAVE + AP < AS
}

BS: 3
S: 9
T: 4
D: 3
AS: 3
AP: -2

output [save d[wound d[hit 4d6]d6]d6] named "Talos"

 if A > B { result: A }

}

function: highest of A:n and B:n and C:n {
 if A > B & A > C { result: A }
 if B > A & B > C { result: B }
result: C
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: multiattack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}

function: save ROLL:n plus BONUS:n vs DC:n for DMG:d save SAVE:d {
    if ROLL + BONUS >= DC{
        result: DMG
    }
    result:SAVE
}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}


function: playerattack PLAYERDMG:n at PLAYERHP:n vs ENEMYDMG:n at ENEMYHP:n  {
     if PLAYERDMG > ENEMYHP {
           result:1
     }    
     result: [enemyattack PLAYERDMG at PLAYERHP vs ENEMYDMG at ENEMYHP - PLAYERDMG]
}
function: enemyattack PLAYERDMG:n at PLAYERHP:n vs ENEMYDMG:n at ENEMYHP:n  {
     if ENEMYDMG > PLAYERHP {
           result:0
     }    
     result: [playerattack PLAYERDMG at PLAYERHP - ENEMYDMG vs ENEMYDMG at ENEMYHP]
}

    if PLAYERINIT >= ENEMYINIT {
         result: [playerattack PLAYERDMG at PLAYERHP vs ENEMYDMG at ENEMYHP]
    }
    result: [playerattack PLAYERDMG at PLAYERHP vs ENEMYDMG at ENEMYHP]
}

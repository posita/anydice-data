FAIL: -1
PARTSUCCESS: 0    \Fail, but less than 10\
SUCCESS: 1
HIGH: 30          \DC of arcana check\
LOW: HIGH - 10    \Fail by 10 or more\

function: save ROLL:n {
 if ROLL <= LOW { result: FAIL }
 if ROLL >= HIGH { result: SUCCESS  }
 result: PARTSUCCESS
}

loop ADJ over {0,5,10,15}{ 
\output [save [lowest 1 of 2d20] + ADJ] named "DC [HIGH] Arcana Check + [ADJ] bonus, with Disadvantage"\
output [save 1d20 + ADJ] named "DC [HIGH] Arcana Check + [ADJ] bonus" 
\output [save [highest 1 of 2d20] + ADJ] named "DC [HIGH] Arcana Check + [ADJ] bonus, with Advantage"\
}

AD: 2
AE: 4
BREAK: 1
OPENING: 1
TO_HIT: 7
RED_DIE: d{0, 1, 2, 3, 3, 4}
BASE: 1dRED_DIE
\ATTACK: d10\
\HITS: ADd10 >= TO_HIT\
\output 1dRED_DIE\
\output HITS\
function: attack ROLL:n vs EVASION:n with OPENINGS:n then HIT:d {
  if (ROLL + OPENINGS) >= EVASION | ROLL = 10 {
    result: 1
  }
  result: 0
}
HITS: ADd[attack d10 vs TO_HIT with OPENING then 1dRED_DIE]
output HITS

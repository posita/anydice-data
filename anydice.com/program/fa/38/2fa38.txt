\ *** FORMULAS *** \

function: attack ROLL:n vs AC:n {
  if ROLL <= 1 { result: 0 }
  if ROLL >= CRIT { result: 2 }
  result: ROLL >= AC
}

function: damage for ATTACKRESULT:n HITDAMAGE:d crit CRITICALDAMAGE:d {
  if ATTACKRESULT = 2 {
    result: CRITICALDAMAGE
  }
  if ATTACKRESULT = 1 {
    result: HITDAMAGE
  }
  result: 0
}

function: attack with ROLL:d for HITDMG:d crit CRITDMG:d {
  result: [damage for [attack ROLL vs AC] HITDMG crit CRITDMG]
}

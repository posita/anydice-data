if ROLL < 6 {
result: 0
} else {
if ROLL > 18 {
loop N over {1..4} {
DMG: DMG + 2d{{1..2}, {3..6}:4}
}

} else {

DMG: DMG + 2d{{1..2}, {3..6}:4}
}

}
}

}

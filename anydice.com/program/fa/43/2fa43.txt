function: check A:n by B {
if A >= B{
result: 1
}
else {
result: 0
}
result: 0
} 

function: multiply A:n by B:n {
result: A * B
}


ATTACKS: 1
HITSPERATTACK: 4
DAMAGE: 4
HITRED: 2

AC: 16
DEX: 5
HITBONUSES: 3
DAMBONUSES: 3
ADVANTAGE: 1

if ADVANTAGE = 1 {
TOHIT: [highest 1 of 2d20] + DEX + HITBONUSES
}
else {
TOHIT: 1d20 + DEX + HITBONUSES
}

loop N over {0..HITSPERATTACK-1}{
TEMP: N * 2
TEMPHIT: TOHIT - TEMP
TEMPDAMAGE: [multiply {1..DAMAGE} by [check TEMPHIT by AC]]
output TEMPDAMAGE named "Damage [N+1]"
}

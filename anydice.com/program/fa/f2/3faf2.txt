function: anycomboisnumber NUM:n A:n B:n C:n D:n {
  VALUES: [sort {A, B, C, D}]
  \ Check all 1-die combos \
  if A = NUM | B = NUM | C = NUM | D = NUM { result: 1 }
  \ Check all 2-die combos \
  if A+B = 7 | A+C = 7 | A+D = 7 | B+C = 7 | B+D = 7 | C+D = 7 { result: 1 }
  \ Check all 3-die combos \
  if A+B+C = 7 | A+B+D = 7 | A+C+D = 7 | B+C+D = 7 { result: 1 }
  \ Check 4-die combo \
  if A+B+C+D = 7 { result: 1 }
  result: 0
}

output [anycomboisseven 1@4d6 2@4d6 3@4d6 4@4d6] named "Chance any subset of 4d6 sums to 7"

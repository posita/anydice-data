AONE: d20
ATWO: d20

function: advroll ROLL:n ROLLTWO:n {
if ROLLTWO > ROLL {result: ROLLTWO}
result: ROLL
}

function: disroll ROLL:n ROLLTWO:n {
if ROLLTWO < ROLL {result: ROLLTWO}
result: ROLL
}

AC: 20
function: dam DICE:n BON:n ROLL:n  ROLLMOD:n {
if ROLL = 20 {result: DICE + DICE + BON}
if ROLL+ROLLMOD >= AC {result: DICE + BON}
result: 0
}

ENEMYSAVE: 5
function: potent DAM:n ROLL:n SPELSAV:n{
if ROLL+ENEMYSAVE < SPELSAV {result: DAM}
result: DAM/2
}

INT: 5
PROF: 6
ABON: INT + PROF
SAVE: 8 + INT + PROF
FIREBOLT: 4d10
FROSTBITE: 4d6
TOLL: 4d12

output [potent TOLL AONE SAVE] named "Potent Toll"
output [dam FIREBOLT INT AONE ABON] named "Empowered Firebolt"
output [potent FROSTBITE AONE SAVE] + INT named "Potent, Empowered Frostbite"

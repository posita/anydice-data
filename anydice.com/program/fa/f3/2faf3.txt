function: outcome ROLL:s {
  A: 0
  B: 0
  C: 0

  \ FACES \
  loop I over ROLL {  
    if I = 1 {A: A + 1    }
    if I = 2 { A: 0    }
    if I = 3 {
      B: B + 1
      C: 0
    }
    if I = 4 {B: 0    }
    if I = 5 {C: C + 1    }
    if I = 6 {C: C + 1  }
  }

  if C > B { result: -1 }
  if A > 0 { 
             if A < C { result: 200 } \ ? \  
             if A >= B { result: 1 } \ ? \  
             result: 1 \ ter caveira = falha \
           }
  if B = C { result: 0 }
  if B > A { result: 0 } \ ter mais shields que caveiras = mixed \
  if A <= 0 { result: 600 } \ nao ter caveiras = sucesso \

  result: 0                                   \ Default result \
}

function: hit AV:n ASK:n against DV:n DSK:n {
  result: (((AV <= ASK) & (DV > DSK)) | ((AV <= 4) | (AV <= (ASK - 10))))
}

function: hit AV:n ASK:n against DV:n DSK:n with damage DAM:d {
 result: [highest of ([hit AV ASK against DV DSK] * (DAM)) and 0]
}

function: hit AV:n ASK:n against DV:n DSK:n with damage DAM:d against dr DR:n {
 result: [highest of ([hit AV ASK against DV DSK] * (DAM - DR)) and 0]
}

GR: 3d6
SKILL: 14
DEFENSE: 10

loop LDF over {8..16} {
 loop LSK over {10..16} {
   output [hit GR LSK against GR LDF] named "[LSK] against [LDF]"
 }
}

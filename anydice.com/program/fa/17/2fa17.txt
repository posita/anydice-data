set "explode depth" to 50
set "maximum function depth" to 10

function: moneydie D:d keep on TN:n {
 SIZE: [maximum of D]
 if SIZE <= 4 {
  result: 1 + [explode D>=TN]
 } else {
  result: 1 + [explode D>=TN] + [moneydie d(SIZE-2) keep on TN]
 }
}

output [lowest of [moneydie d4 keep on 2] and 50] named "d4"

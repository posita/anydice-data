\**** Damage calculator ****\

function: multiattack ATTACKS:n {
 TOTAL: 0
 loop N over {1..ATTACKS}{
  TOTAL: TOTAL + [attackroll d20 vs ENEMYAC]
 }
 result: TOTAL
}

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{16..20}

ATTACKMOD: 2
DMGBONUS: 2
PROFICIENCY: 2
ACCBONUS: 0
CRITROLL: 20
DAMAGE: 1d4 + DMGBONUS
CRITDAMAGE: 2d4 + DMGBONUS
output [multiattack 100] named "100 attacks"

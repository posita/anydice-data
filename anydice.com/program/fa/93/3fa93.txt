set "maximum function depth" to 20
function: how many DIE:d to NUM:n {
  if NUM <= 0 {
    result: 0
  }
  MAX: [maximum of DIE]
  if #MAX = 0 {
    result: 1
  }
  MINIT: (NUM - 1) / MAX + 1
  result: MINIT + [how many DIE to (NUM - MINITdDIE)]
}

MINIT: 99 / 6 + 1
loop N over {MINIT.. 100} {
  output Nd6 >= 100 named "[N] rolls"
}

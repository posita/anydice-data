\**** Damage calculator ****\

function: multiattack ATTACKS:n {
 TOTAL: 0
 loop N over {1..ATTACKS}{
  TOTAL: TOTAL + [attackroll d20 vs ENEMYAC]
 }
 result: TOTAL
}

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 12-20\
ENEMYAC: d{12..20}

ATTACKMOD: 4
DMGBONUS: 2
PROFICIENCY: 3
ACCBONUS: 0
CRITROLL: 20
DAMAGE: 2d8 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 4d8 + ATTACKMOD + DMGBONUS

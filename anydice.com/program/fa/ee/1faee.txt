MISS: 0
HIT: 1

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }

 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

AC: 20
ATTACK: 10
HITDAMAGE: d6
MISSDAMAGE: 0

output 1d[damage for [attack d20 vs AC]] named "[AC]"

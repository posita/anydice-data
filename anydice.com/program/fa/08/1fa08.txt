function: critmult ATTACK:n bonus BONUS versus AC damage DAMAGE crit CRIT {
if ATTACK = 1 { result: 0 }
if ATTACK = 20 { result: CRIT }
if ATTACK + BONUS < AC { result: 0 }
result: DAMAGE
}

function: critmultfact ATTACK:n bonus BONUS versus AC damage DAMAGE crit CRIT {
if ATTACK = 1 { result: 0 }
if ATTACK = 20 { result: CRIT + [critmult d20 bonus BONUS versus AC damage DAMAGE crit CRIT] }
if ATTACK + BONUS < AC { result: 0}
result: DAMAGE + [critmult d20 bonus BONUS versus AC damage DAMAGE crit CRIT]
}

function: sneakcheck A:n {
if A = 0 { result: [critmult d20 bonus DEX_R + PROF versus TARGET_AC damage d4 + SNEAK crit 2d4 + 2dSNEAK] }
result: A + [critmult d20 bonus DEX_R + PROF versus TARGET_AC damage d4 crit 2d4]
}

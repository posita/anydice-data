

MISS: 0
HIT: 1
CRITICAL_HIT: 2

function: attack ROLL:n ATTACK vs AC:n 
{
    if ROLL = 1 
    { 
       if ROLL + ATTACK >= AC + 10 { result: HIT }
       result: MISS 
    }
    if ROLL = 20 
    {
       if ROLL + ATTACK < AC { result: HIT }
       result: CRITICAL_HIT
    }
    if ROLL + ATTACK >= AC
    { 
       result: HIT 
    }
    result: MISS
}

function: damage DMG:s for RESULT:n
{
    result: [damage DMG deadly 0 for RESULT]
}

function: damage DMG:s deadly CRIT_EXTRA:d for RESULT:n
{
    if RESULT = MISS { result: 0 }
    if RESULT = HIT { result: DMG }
    if RESULT = CRITICAL_HIT { result: DMG*2 + CRIT_EXTRA }
}

function: damage DMG:s fatal CRIT_EXTRA:d for RESULT:n
{
    if RESULT = MISS { result: 0 }
    if RESULT = HIT { result: DMG }
    if RESULT = CRITICAL_HIT { result: (#DMGdCRIT_EXTRA)*2 + CRIT_EXTRA }
}

function: size of X:d { result: [maximum of X] / #(X) }

DICE: 35
OFFSET: 10
CUTOFF: 15 \ show clash outcome chances from -cutoff to +cutoff \
SELF_AUTOS: 7
ENEMY_AUTOS: 6

CD: d10>6
function: clash vs N:n {
    TMP: (SELF_AUTOS+(N dCD))-(ENEMY_AUTOS+(DICE dCD))
    result: [highest of [lowest of TMP and CUTOFF] and -CUTOFF]
}

loop N over {DICE-OFFSET..DICE+OFFSET} {

    output [clash vs N]>0 named "clash [N]d vs [DICE]d"
}

\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
if ROLL >= CRITROLL { result: CRITDAMAGE }
if ROLL = 1 { result: MISSDAMAGE }
if ROLL + PROFICIENCY + ATTACKMOD >= AC { result: DAMAGE }
result: MISSDAMAGE
}

\looped against enemy AC from 13-18\
ENEMYAC: d{13..18}

ATTACKMOD: 4
PROFICIENCY: 3
CRITROLL: 20

ATTACKTIMES: 1
DAMAGE: 2d6 + ATTACKMOD
CRITDAMAGE: 6d6 + ATTACKMOD
MISSDAMAGE: 0
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Greatsword"
output ATTACKTIMESd[attackroll [highest 1 of 2d20] vs ENEMYAC] named "Greatsword w/ Advantage"

ATTACKTIMES: 1
DAMAGE: 1d12 + ATTACKMOD
CRITDAMAGE: 4d12 + ATTACKMOD
MISSDAMAGE: 0
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Axe"
output ATTACKTIMESd[attackroll [highest 1 of 2d20] vs ENEMYAC] named "Axe w/ Advantage"

function: soldiers A:n SO:n {
  if A = 1{
    result: SO - SO/5
  }
  result: SO
}
function: supplies A:n SU:n SO:n{
  if A = 2{
    result: SU + (2d6+1)*SO
  }
  result: SU
}

DAYS: 6
SOLDIERS: d{500}
SUPPLIES: d{0}

loop N over {1..DAYS}{
  SUPPLIES: [supplies 1d2 SUPPLIES SOLDIERS]
  SOLDIERS: [soldiers 1d2 SOLDIERS]
  output SOLDIERS named "Soldiers remaining day [N]"
  output SUPPLIES named "Supplies destroyed day [N]"
}

function: roll ROLLA:n then ROLLB:n with ATTACK:n against AC:n and damage DMG with THREAT and MULTIPLIER {
  if ROLLA+ATTACK >= AC & ROLLA-THREAT >= 0 & ROLLB+ATTACK > AC & MULTIPLIER = 2 {result: DMG + DMG}
  if ROLLA+ATTACK >= AC & ROLLA-THREAT >= 0 & ROLLB+ATTACK > AC & MULTIPLIER = 3 {result: DMG + DMG + DMG}
  if ROLLA+ATTACK >= AC & ROLLA-THREAT >= 0 & ROLLB+ATTACK > AC & MULTIPLIER = 4 {result: DMG + DMG + DMG + DMG}
  if ROLLA+ATTACK >= AC & ROLLA-THREAT >= 0 & ROLLB+ATTACK > AC & MULTIPLIER = 5 {result: DMG + DMG + DMG + DMG + DMG}
  if ROLLA+ATTACK >= AC & ROLLA-THREAT >= 0 & ROLLB+ATTACK > AC & MULTIPLIER = 6 {result: DMG + DMG + DMG + DMG + DMG}
  if ROLLA+ATTACK >= AC {result: DMG }
  result: 0
}

function: roll attack with BONUS and DMG and THREAT and CRIT against AC {
  result: [roll 1d20 then 1d20 with BONUS against AC and damage DMG with THREAT and CRIT]
}

function: average longsword BONUS vs AC:n {
  result: [roll 1d20 then 1d20 with BONUS against AC and damage 1d8+1 with 9 and 2]
}

function: modifier ROLL {
  result: (10 - ROLL)/2
}

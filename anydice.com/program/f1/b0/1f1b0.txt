function:  ROLL:n vs DEFENSE:n ATTACK:n {
 if ROLL = 1 {result: MISS}
 if ROLL = 20 {result: CRITICALHIT}
 if ROLL + ATTACK>= DEFENSE {result: HIT}
 result: MISS
}


function: save spell one ROLL:n {
 if ROLL + MONSTERSAVEONE >= CHARACTERSPELLSAVEONE {result: SAVESPELLMISSDAMAGEONE}
 result: SAVESPELLHITDAMAGEONE
}

function: save spell two ROLL:n {
 if ROLL + MONSTERSAVEONE >= CHARACTERSPELLSAVETWO {result: SAVESPELLMISSDAMAGETWO}
 result: SAVESPELLHITDAMAGETWO
}

\ Multi attacks \

function: bigmultiattack NUMBEROFATTACKS:n {
  TOTAL: 0
  loop N over {1..NUMBEROFATTACKS} {
    TOTAL: TOTAL + [damage for big [d20 vs AC BIGATTACK]]
  }
  result: TOTAL
}

function: bigmultiattackadvantage NUMBEROFATTACKS:n {
  TOTAL: 0
  loop N over {1..NUMBEROFATTACKS} {
    TOTAL: TOTAL + [damage for big [[highest 1 of 2d20] vs AC BIGATTACK]]
  }
  result: TOTAL
}

function: calc A:n size S:n {
if A = 1 {result: 0}
if A = S {result: S + [calc dS size S]}
result: A
}
set "maximum function depth" to 5
S: 6
T: 2
D: 3 + T
M: 4 + T
R: 1
C: (1 + T) * 1
output ((([calc dS size S] + M - C) / (D*R)) + (([calc [lowest 1 of 2dS] size S] + M - C) /(D*R)) + (([calc [lowest 1 of 3dS] size S] + M - C) / (D*R)) + (([calc [lowest 1 of 4dS] size S] + M - C) / (D*R))) * 2

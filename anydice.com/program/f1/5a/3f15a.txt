
function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS vs AC hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}

output
[attack d20 plus 18 vs 20
    hit  2d6 + 22 + 2d10            + [attack [highest 1 of 2d20] plus 18 vs 20 for 2d6 + 22      crit 2d6 + 2d6    on 12]
    miss 2d6 + 22 + 2d10            + [attack [highest 1 of 2d20] plus 18 vs 20 for 2d6 + 22      crit 2d6 + 2d6    on 12]
    crit 4d6 + 22 + 4d10 + 2d6      + [attack [highest 1 of 2d20] plus 18 vs 20 for 2d6 + 22      crit 2d6 + 2d6    on 12] on 12
] named "[20] Two Barbarian Attacks"

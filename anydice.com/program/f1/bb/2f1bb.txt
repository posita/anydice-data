COUNT:-1

function: countem {
result: COUNT+1
}

function: six A:n B:n SUM:n COUNT:n {
TOTAL: SUM+A+B
if COUNT=5 {result: TOTAL}
if A<6 & B<6 {result: TOTAL}
if A=6 & B=6 {result:[six 1d6 1d6 TOTAL [countem]]}
if A=6 {result:[six 1d6 B TOTAL [countem]]}
if B=6 {result:[six A 1d6 TOTAL [countem]]}
}

output [six 1d6 1d6 0 [countem]]
output 2d10

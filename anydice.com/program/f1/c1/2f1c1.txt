function: countem {
result: COUNT+1
}

function: six A:n B:n C:n SUM:n COUNT:n {
TOTAL: SUM+A+B+C
if COUNT=5 {result: TOTAL}
if A<6 & B<6 & C<6 {result: TOTAL}
if A=6 & B=6 & C=6 {result:[six 1d6 1d6 1d6 TOTAL [countem]]}

if A=6 & B=6 {result:[six 1d6 1d6 C TOTAL [countem]]}
if B=6 & C=6 {result:[six A 1d6 1d6 TOTAL [countem]]}
if C=6 & A=6 {result:[six 1d6 B 1d6 TOTAL [countem]]}

if A=6 {result:[six 1d6 B C TOTAL [countem]]}
if B=6 {result:[six A 1d6 C TOTAL [countem]]}
if C=6 {result:[six A B 1d6 TOTAL [countem]]}
}

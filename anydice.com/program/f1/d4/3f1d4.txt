function: calc A:n siz S:n depth D:n mod M:n{
if D = 5 {result:M}
if D = 0 & A = 1 {result: 0}
if A = S {result: S + [calc dS siz S depth D + 1 mod M]}
result: A + M
}
T: 1
MOD: 4 + T
RES: 5 + T
CST: 1
SIZ: 4
MAP: 1
ATT: 4
EFF: 0
A: [highest of 0 and ([calc dSIZ siz SIZ depth 0 mod MOD] - RES)/CST]
loop N over {1..ATT}{
EFF: EFF + [highest of 0 and ([calc [lowest 1 of [highest of 1 and MAP]dSIZ] siz SIZ depth 0 mod MOD] - RES)/CST]
MAP: MAP + 1
}
output EFF

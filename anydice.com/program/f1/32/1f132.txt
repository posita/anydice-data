function: ROLL:n vs AC:n {
 if ROLL >= RITON & ROLL + ATTACK >= AC {
  result: [riton d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: riton ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE - STRENGTH - 2
 }
 result: DAMAGE 
}

AC: 15
STRENGTH: 4
ATTACK: 2 + STRENGTH

DAMAGE: 1d6+4+2+1d4+1d8
RITON: 20
CRITICAL: 2
output [2d20 vs AC] named "Kobold Steel"

set "maximum function depth" to 15

COUNT:0

function: countem NUM:n {
result: COUNT+NUM
}

function: six A:n B:n C:n SUM:n COUNT:n {
TOTAL: SUM+A+B+C
if COUNT>=5 {result: TOTAL}
if A<6 & B<6 & C<6 {result: TOTAL}
if A=6 & B=6 & C=6 {result:[six 1d6 1d6 1d6 TOTAL [countem 3]]}

if A=6 & B=6 {result:[six 1d6 1d6 C TOTAL [countem 2]]}
if B=6 & C=6 {result:[six A 1d6 1d6 TOTAL [countem 2]]}
if C=6 & A=6 {result:[six 1d6 B 1d6 TOTAL [countem 2]]}

if A=6 {result:[six 1d6 B C TOTAL [countem 1]]}
if B=6 {result:[six A 1d6 C TOTAL [countem 1]]}
if C=6 {result:[six A B 1d6 TOTAL [countem 1]]}
}

output [six 1d6 0 0 0 [countem 0]] named "sorc burst lvl 1"

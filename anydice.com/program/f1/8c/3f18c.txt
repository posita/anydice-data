MY_RESTARTS: 2

\ 30%
ENEMY_ATTACK: 4d6
ENEMY_EXTRA_DAMAGE: 0
ENEMY_RESTARTS: 1
\

\ 70%
ENEMY_ATTACK: 15

\

\ 61%

ENEMY_EXTRA_DAMAGE: 0
ENEMY_RESTARTS: 0
\

\ 87%
ENEMY_ATTACK: 10

\

\ 96%
ENEMY_ATTACK: d6 + 5

\

ENEMY_ATTACK: 3d6 + 5
ENEMY_EXTRA_DAMAGE: 0
ENEMY_RESTARTS: 0

\MY_ATTACK: 3d{3..6}\ \reroll 1 and 2\

MY_FIGHT: MY_ATTACK > ENEMY_ATTACK

MY_HEART_LOSS: (ENEMY_ATTACK + ENEMY_EXTRA_DAMAGE - MY_DEFENSE + 9) / 10

WIN_AND_SURVIVE_BASE: MY_FIGHT & (MY_HEART_LOSS < MY_HEARTS)
WIN_AND_SURVIVE: WIN_AND_SURVIVE_BASE
LOSE: 1-WIN_AND_SURVIVE
LOSE_N: LOSE

loop N over {1..MY_RESTARTS}{
 LOSE_N: LOSE_N * LOSE
}
WIN_AND_SURVIVE: 1-LOSE_N

loop N over {1..ENEMY_RESTARTS}{
 WIN_AND_SURVIVE : WIN_AND_SURVIVE * WIN_AND_SURVIVE_BASE
}

\WIN_AND_SURVIVE: WIN_AND_SURVIVE * WIN_AND_SURVIVE\ \enemy restart fight once\
\WIN_AND_SURVIVE: 1-(1-WIN_AND_SURVIVE)*(1-WIN_AND_SURVIVE)\ \I restart fight once\

output WIN_AND_SURVIVE named "win and survive"


output MY_FIGHT named "win"
output MY_HEART_LOSS named "heart loss"

output MY_ATTACK named "my attack"
output ENEMY_ATTACK named "enemy attack"

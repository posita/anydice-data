function: explode N:n on SIZE:n {
 if N = SIZE & SIZE = 20 {
  result: N + [explode d(SIZE - 8) on SIZE - 8]
 } else if N = SIZE & SIZE > 4 {
  result: N + [explode d(SIZE - 2) on SIZE - 2]
 } else if N = SIZE & SIZE = 4 {
  result: N + [explode d(SIZE) on SIZE]
 }
 result: N
}

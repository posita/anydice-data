function: max A B {
  if A > B {
    result: A
  } else {
    result: B
  }
}

function: minimum choice ROLL:s {
  HIGH: 1@ROLL
  MID: 2@ROLL
  LOW: 3@ROLL

  result: (HIGH + MID) * LOW
}

function: maximum choice ROLL:s {
  HIGH: 1@ROLL
  MID: 2@ROLL
  LOW: 3@ROLL

  result: (MID + LOW) * HIGH
}

function: optimal choice ROLL:s {
  HIGH: 1@ROLL
  MID: 2@ROLL
  LOW: 3@ROLL

  CHOICE_HIGH: (MID + LOW) * HIGH
  CHOICE_MID: (HIGH + LOW) * MID
  CHOICE_LOW: (HIGH + MID) * LOW

  BEST: [max CHOICE_HIGH CHOICE_MID]
  result: [max BEST CHOICE_LOW]
}

output [minimum choice 3d6] named "Choose Minimum"
output [maximum choice 3d6] named "Choose Maximum"

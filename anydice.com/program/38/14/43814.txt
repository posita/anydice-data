DADO_A: 1d12
DADO_B: 1d12
DT: 7

function: checar critico de A:n e B:n {
  if A = B & A >= 5 {
    result: 1
  }
  result: 0
}

function: checar falha de A:n e B:n {
  if A = B & A = 1 {
    result: 1
  }
  result: 0
}

function: sucesso A:n e B:n {
  if A + B >= DT {
    result: 1
  }
  result: 0
}

function: rolar A:n e B:n {
  if A = B & A >= 6 {
    result: A + B
  }
  result: A + B
}

output [checar critico de DADO_A e DADO_B] named "Chance de Sucesso Critico (1 = SIM, 0 = NAO)"
output [checar falha de DADO_A e DADO_B] named "Chance de Falha Critica (1 = SIM, 0 = NAO)"
output [sucesso DADO_A e DADO_B] named "Chance de Sucesso (1 = Sim, 0 = NAO)"
output [rolar DADO_A e DADO_B] named "Todos os Resultados"

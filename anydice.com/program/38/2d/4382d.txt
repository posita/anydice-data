function: invisible X:n to Y:n {
 if X=1{result: 0}
 else {result: Y}
}
function: lancer X:n vs Y:n to A:n B:n {
 if X>=Y {
  if X>=20{result: B}
  else {result: A}
 }
 else {result: 0}
}
function: dnd R:n mod M:n vs AC:n to A:n B:n {
 if R+M>=AC {
  if R=20 {result: B}
  if R=1 {result: 0}
  else {result: A}
 }
 else {result: 0}
}
function: luck LCK:n {
 if LCK<=14 {result: 14}
 else {result: LCK}
}
function: ac H:n vs A:n {
 if H<A {result: 0}
 else {result: H}
}

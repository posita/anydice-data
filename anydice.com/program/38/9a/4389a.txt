\ Define shot hit chances \
CR_SHOT: d10 >= 5
DD_SHOT: d10 >= 6
MI_SHOT: d10 >= 5
FR_SHOT: d10 >= 10
DS_SHOT: d10 >= 5

\ Fleet Composition \
CRUISERS: 2
DESTROYERS: 7
MISSILE_SHIPS: 6
FREIGHTERS: 3
DEFENSE_SATS: 9

\ Calculate Total Hits \
output (CRUISERS * 1)d CR_SHOT 
     + (DESTROYERS * 1)d DD_SHOT 
     + (MISSILE_SHIPS * 1)d MI_SHOT 
     + (FREIGHTERS * 1)d FR_SHOT 
     + (DEFENSE_SATS * 1)d DS_SHOT 
     named "Enemy Fleet Total Hits"

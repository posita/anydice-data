function: crush ATTACK:n
{
  if ATTACK > 5 {
    if ATTACK = 12 {
      result: 4d6
    }
    result: 2d6
  }
  result: 0
}

function: noncrush ATTACK:n
{
  if ATTACK > 5 {
    result: 2d6
  }
  result: 0
}

function: rerollones D:n
{
  if D = 1 {
    result: d6
  }
  result: D
}

output [crush 2d6] named "normal crush"
output [crush [highest 2 of 3d6]] named "advantage crush"
output [noncrush 2d6] named "normal"
output (2d6>5)*2d[rerollones d6] named "reroll ones on damage"
output ([highest 2 of 3d6]>5)*2d[rerollones d6] named "advantage reroll ones on a damage"

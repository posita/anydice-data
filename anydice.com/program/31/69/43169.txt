function: full difficulty SEQ:s {
  if (SEQ = 1) = #SEQ { result: 1}
  if (SEQ = -1) & ((SEQ = -1) > (SEQ = 1)) { result: -1 }
  result: 0
}

function: good POS:s bad NEG:s {
SUCC: (POS = 1) + (NEG = 1)
FAIL: (NEG = -1)
DIFF: #NEG
if SUCC >= DIFF { result: 1 }
if FAIL & (FAIL > SUCC) { result: -1 }
result: 0
}

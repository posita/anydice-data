
function: noncrush ATTACK:n
{
  if ATTACK > 5 {
    result: 2d6
  }
  result: 0
}

function: rerollones D:n
{
  if D = 1 {
    result: d6
  }
  result: D
}

output 0 named "Tagless."
output [noncrush 2d6] named "normal"
output 0 named "One tag only."
output (2d6>5)*2d[rerollones d6] named "crushing"
output (2d6>4)*2d6 named "reliable"
output ([highest 2 of 3d6]>5)*2d6 named "advantage"
output 0 named "Two tags."
output ([highest 2 of 3d6]>5)*2d[rerollones d6] named "advantage crushing"
output ([highest 2 of 3d6]>4)*2d6 named "advantage reliable"
output (2d6>5)*2d[rerollones d6] named "reliable crushing"

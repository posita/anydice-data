function: seq A:s {result: A}
function: roll A:s {
 X: A
 Y: 0
 loop N over {1..8} {
  Y: Y + [count {4..6} in X]
  X: [seq [count 6 in X]d6]
 }
 result: Y
}

function: corr A:s {
 X: A
 Y: 0
 loop N over {1..8} {
  Y: Y + [count 1 in X]
  X: [seq [count 6 in X]d6]
 }
 result: Y
}

output [roll 8d6]
output [corr 8d6]

function: succeeds with N:n dice at threshold T:n {
  result: [highest 1 of N d6] >= T
}

function: count of GROUPS:s succeed at threshold T:n {
  S: 0
  loop G over GROUPS {
    S: S + [succeeds with G dice at threshold T]
  }
  result: S
}

output [count of {1,1,2,2} succeed at threshold 6] named "Players hitting 6+"

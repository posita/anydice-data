WIS: 2
EXHAUSTION: 3
START_DC: 14

function: stop with ROLL:n modifier MOD:n exhaustion EX:n against DC:n {
 if ROLL = 20 { result: 1 }
 result: ROLL + MOD + EX >= DC
}

loop FAILURES over {0..10} {
 DC: START_DC + FAILURES

 output [stop with d20 modifier WIS exhaustion EXHAUSTION against DC]
  named "Attempt [FAILURES + 1], DC [DC]"
}

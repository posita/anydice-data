function: crush D:n
{
  if D = 1 {
    result: d6
  }
  result: D
}

function: brutal A:n B:n {
  if A = B {
    result: A + B + d6
  }
  result: A + B
}
function: threebrutal A:n B:n C:n{
  if A = B {
    result: A + B + C + d6
  }
else if A = C {
    result: A + B + C + d6
  }
else if B = C {
    result: A + B + C + d6
  }
  result: A + B + C
}

function: brutalintocrush A:n B:n {
  if A = B {
    result: [crush A] + [crush B] + [crush d6]
  }
  result: [crush A] + [crush B]
}

function: A:n B:n C:n axedvantage {
  if 1@[sort {A, B, C}]=2@[sort {A, B, C}] {
    result: 1@[sort {A, B, C}] + 2@[sort {A, B, C}] + d6
  }
  result: 1@[sort {A, B, C}] + 2@[sort {A, B, C}]
}


output [brutal 1d6 1d6] named "Single Brutal"

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

output 1d20 named "Normal d20"

output
[reroll {1} on d20 to d20] named "Halfling Luck"

output [highest 1 of 2d20] named "Normal Advantage"

output
[highest 1 of 2d[reroll {1} on d20 to d20]] named "Halfling Luck (Advantage)"

output [lowest 1 of 2d20] named "Normal Disadvantage"

[lowest 1 of 2d[reroll {1} on d20 to d20]] named "Halfling Luck (Disadvantage)"

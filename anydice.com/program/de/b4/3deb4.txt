\ Combo-Strike roller

function: combo SIZE:n PREV:n {
  ROLL: dSIZE
  if ROLL >= PREV {
    result: ROLL + combo SIZE ROLL
  } else {
    result: ROLL
  }
}

function: runcombo SIZE:n { combo SIZE 0 }

output [runcombo 4]  named "d4 combo"
output [runcombo 6]  named "d6 combo"
output [runcombo 10] named "d10 combo"
output [runcombo 12] named "d12 combo"

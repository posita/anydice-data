\**** Damage calculator Dual Wield Liveblade ****\

function: attackmain ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: (ROLL + ATTACKBONUS - DEF + BONUSDMG + WPNDAMAGEA)*2} 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS - 0 >= DEF { result: ROLL + ATTACKBONUS - DEF + (BONUSDMG) + WPNDAMAGEA }
 result: 0
}

function: dualwield ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: WPNDAMAGEB*2} 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS >= DEF { result: WPNDAMAGEB}
 result: 0
}

\looped against enemy DEF\
ENEMYDEF: d{11}

ATTACKBONUS: 5
BONUSDMG: 0
CRITROLL: 12
BITEA: 1d6+4
BITEATIMES: 2
BITEB: 0
WPNDAMAGEA: 14
WPNDAMAGEB: 14
COMBOTIMES: 1
output [attackmain 2d6 vs ENEMYDEF] + ([dualwield 2d6 vs ENEMYDEF] * COMBOTIMES) + (BITEA * BITEATIMES) + BITEB named "Dual Wield Liveblade"

output 2d6

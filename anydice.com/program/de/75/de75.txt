function: great DIE:d {
  result: [reroll DIE as DIE if under 3]
}
function: good DIE:d {
  result: [reroll DIE as DIE if under 2]
}
function: reroll ROLL:n as DIE:d if under LIMIT:n {
  if ROLL < LIMIT { result: DIE }
  else { result: ROLL }
}

loop SIDES over {6} {
  GOOD: [good dSIDES]
  output 2dGOOD+2 named "2d[SIDES]+2, reroll 1s"
}

loop SIDES over {6} {
  GREAT: [great dSIDES]
  output 2dGREAT+2 named "2d[SIDES]+2, reroll 1s and 2s"
}

output [highest 2 of 3d6]+2 named "3d6+2, keep the highest two"

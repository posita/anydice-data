THRESHOLD: 13 \ USE 11 for Scientist \


\ DICE DEFINITIONS \
BLACK: {0,3,3,3,4,5}
YELLOW: {5,5,4,2,2,0}
BLUE: {0,1,2,3,6,6}
RED: {0,1,1,4,6,6}

function: N:n purple {
  result: [N purple Nd{1:5, 0} normal]
}

\ This helper squashes NORMAL to an integer \
function: N:n purple NORMAL:n normal {
  MULTIPLIES: N - NORMAL
  result: NORMALd{-1,0,2,4,5} * (2 ^ (MULTIPLIES))
}

\ OUTPUTS \
DICE: {4..7}

loop N over DICE {
  output NdRED > THRESHOLD named "[N]red"
}
loop N over DICE {
  output NdBLUE > THRESHOLD named "[N]blue"
}
loop N over DICE {
  output NdBLACK > THRESHOLD named "[N]black"
}
loop N over DICE {
  output NdYELLOW > THRESHOLD named "[N]yellow"
}
loop N over DICE {
  output [N purple] > THRESHOLD named "[N]purple"
}

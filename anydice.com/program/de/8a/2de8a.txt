function: attack X:n H:n I:n W:n O:n S:n V:n D:n{
 if X>=H {result: [wound 1d6 W O S V D]}
 if X<=I {result: [rehit 1d6 H W O S V D]}
 result: 0
}

function: rehit X:n H:n W:n O:n S:n V:n D:n{
 if X>=H {result: [wound 1d6 W O S V D]}
 result: 0
}

function: wound X:n W:n O:n S:n V:n D:n{
 if X>=W {result: [save 1d6 S V D]}
 if X<=O {result: [rewound 1d6 W S V D]}
 result: 0
}

function: rewound X:n W:n S:n V:n D:n{
 if X>=H {result: [wound 1d6 S V D]}
 result: 0
}

function: save X:n S:n V:n D:n{
 if X>=S {result: 0}
 if X<=0 {result: [resave 1d6 S D]}
 result: D
}

function: resave X:n S:n D:n{
 if X>=S {result: 0}
 result: D
}

output 4d[attack 1d6 2 0 3 0 5 0 3]
output 4d[attack 1d6 3 0 2 0 4 0 3]

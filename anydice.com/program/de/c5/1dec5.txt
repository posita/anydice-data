
function: attack ROLL:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL > 1 {
        result: DMG
    }
    result: 0
}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

function: gwf DICE:d {
    result: [reroll {1,2} on DICE]
}

function: multiattack X:n {
  TOTAL: 0
  loop N over {1..X} {
    DAMAGE: [attack d20 for 2d[gwf d6] + 5 crit 2d[gwf d6] on 18]
    TOTAL: TOTAL + DAMAGE
  }
  result: TOTAL
}

function: multiattackgwm X:n {
  TOTAL: 0
  loop N over {1..X} {
    DAMAGE: [attack d20 for 2d[gwf d6] + 5 + 10 crit 2d[gwf d6] on 18]
    TOTAL: TOTAL + DAMAGE
  }
  result: TOTAL
}

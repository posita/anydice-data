
function: attack ROLL:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL > 1 {
        result: HIT
    }
    result: MISS
}

function: attack ROLL:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}

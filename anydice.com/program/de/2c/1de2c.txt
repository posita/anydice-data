function: pbta R:n B:n F:n S:n {
 if R+B>S {result: 2}
 if R+B<F {result: 0}
 else {result: 1}
}

function: ironsworn RA:n RB:n T:n {
 if RA > T & RB > T {result: 2}
 if RA > T | RB > T {result: 1}
 else {result: 0}
}

function: bitd DICE:s {
 R: 1@DICE
 if R = 6 {result: 2}
 if R > 3 {result: 1}
 result: 0
}

function: pool S:s E:s T:s W:s {
 X: [count 6 in S]+[count {6..8} in E]+[count {6..10} in T]+[count {6..12} in W]
 if X = 0 {result: 0}
 if X = 1 {result: 1}
 result: 2
}

function: test F:s S:s E:s T:s W:s {
 HITS: [count 6 in S]+[count {6..8} in E]+[count {6..10} in T]+[count {6..12} in W]
 MISS: [count 1 in F]+[count 1 in S]+[count 1 in E]+[count 1 in T]+[count 1 in W]
 if HITS != 0 {
  if MISS = 0 {
   result: 2
  }
  result: 1
 }
 result: 0
}

function: bladepool F:s S:s E:s T:s W:s {
 FM: 1@F
 SM: 1@S
 EM: 1@E
 TM: 1@T
 WM: 1@W
 MAX: [highest of FM and [highest of SM and [highest of EM and [highest of TM and WM]]]]
 if MAX < 4 {result: 0}
 if MAX < 6 {result: 1}
 else {result: 2}
}

function: diff F:s S:s E:s T:s W:s {
 HITS: [count 6 in S]+[count {6..8} in E]+[count {6..10} in T]+[count {6..12} in W]
 MISS: [count 1 in F]+[count 1 in S]+[count 1 in E]+[count 1 in T]+[count 1 in W]
 D: HITS - MISS
 if D < 0 {result: 0}
 if D = 0 {result: 1}
 if D > 0 {result: 2}
}

function: ironpool D:s {
 RA: 1@D
 RB: 2@D
 result: [ironsworn 1dRA 1dRB 3]
}

\**********************************************************************************************\
function: crop SEQUENCE:n {                  \ limit the display range between [min;max] \
 if (SEQUENCE < 0){result: 0}
 result: SEQUENCE
}

function: use chart ROLL:n CHART:s{
 if (ROLL >= 4@CHART){result: 4}
 if (ROLL >= 3@CHART){result: 3}
 if (ROLL >= 2@CHART){result: 2}
 if (ROLL >= 1@CHART){result: 1}
 if (ROLL < 1@CHART){result: 0}
 result: 0
}

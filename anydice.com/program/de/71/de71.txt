function: great DIE:d {
  result: [reroll DIE as DIE if under 3]
}
function: good DIE:d {
  result: [reroll DIE as DIE if under 2]
}
function: reroll ROLL:n as DIE:d if under LIMIT:n {
  if ROLL < LIMIT { result: DIE }
  else { result: ROLL }
}

loop SIDES over {6} {
  GOOD: [good dSIDES]
  output 2dGOOD named "2d[SIDES], reroll 1s"
}

loop SIDES over {6} {
  GREAT: [great dSIDES]
  output 2dGREAT named "2d[SIDES], reroll 1s and 2s"
}

output [highest 2 of 3d6] named "3d6, keep the highest two"

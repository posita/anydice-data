
PLAYERDICE:3
PLAYERMAX:0

MONSTERDICE:2
MONSTERMAX:0

function: max A:n B:n{
if A > B { result: A }
else {result: B}
}

function: battle A:n vs B:n{
if B < A {result: 1}
else {result: 0}
}

loop N over {1..PLAYERDICE}{
TEMP:d6
PLAYERMAX: [max PLAYERMAX TEMP]
}

loop N over {1..MONSTERDICE}{
TEMP:d6
MONSTERMAX: [max MONSTERMAX TEMP]
}

output PLAYERMAX named "PlayerMax"
output MONSTERMAX named "MonsterMax"


output [battle PLAYERMAX vs MONSTERMAX] named "Ergebnis (1 = Spieler gewinnt (auch mit unendschieden))"

function: attack ROLL:n BONUS:n DMG:d DMG_BON:n{
  if ROLL = 20 {
    result: DMG * 150 / 100 + DMG_BON
  }
  else {
    if ROLL + BONUS >= 10 {
      result: DMG + DMG_BON
    }
    else {
      result: 0
    }
  }
}

function: attacks ATK:n times with BONUS at DMG dommage plus DMG_BONUS {
  TOT:0
  loop N over {1..ATK}{
    TOT: TOT + [attack 1d20 BONUS DMG DMG_BONUS]
  }
  result: TOT
}

output [attacks 2 times with 6 at 1d6+2 dommage plus 0] named "Vaar"
output [attacks 1 times with 1 at 2d6 dommage plus 7] named "Erden"
output [attacks 1 times with 7 at 2d6+2 dommage plus 0] named "Jinka"

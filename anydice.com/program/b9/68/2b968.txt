\ count every 6s. Count 5s too if and only if they weren't cancelled by 1s \

 A: 0
 X: [count {6} in ROLL]
 Y: [count {5} in ROLL] - [count {1} in ROLL]

if (Y < 0)
{A: X}
else
{A : X+Y}
if A <=3
{ result: A}
result: 3
}

}

    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}



function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: multiattack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}

function: check ROLL:n plus BONUS:n vs OPPROLL:n plus MOD:n hit HIT:d miss MISS:d {
    if ROLL + BONUS >= OPPROLL + MOD {
        result: HIT
    }
    result: MISS
}

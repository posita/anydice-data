function: countones ROLL:s {
  result: [count 1 in ROLL]
}

function: getresult ROLL:n NUMONES:n {
  if NUMONES > 1 {
    result: 1
  }
  if ROLL = 1 { 
    result: 2  
  }
  if ROLL = 3 | ROLL = 4 { 
    result: 3  
  }
  if ROLL >= 5 & ROLL <= 9 { 
    result: 4  
  }
  if ROLL >= 10 { 
    result: 5  
  }
}

function: rolldice N:n {
  R: Nd20
  L: [lowest 1 of R]
  O: [countones R]
  result: [getresult L O]
}


output [rolldice 7] named "[7]d20"

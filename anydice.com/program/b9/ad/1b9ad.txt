
function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS vs AC hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}


function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

function: gwf DICE:d {
    result: [reroll {1,2} on DICE]
}

ADVANTAGE: [highest 1 of 2d20]

DISADVANTAGE: [lowest 1 of 2d20]


PROF: +5
STR: +4

loop AC over {2 + PROF + STR + 1 .. 19 + PROF + STR + 1} {

    output
    [attack ADVANTAGE plus PROF + STR + 1 vs AC for 6d6 + 2d8 + STR crit 6d6 + 2d8 on 20] named "booming blade vs [AC]"

    output
    [attack ADVANTAGE plus PROF + STR + 1 - 5 vs AC for 6d6 + 2d8 + STR + 10 crit 6d6 + 2d8 on 20] named "booming blade -5/+10 vs [AC]"

    ATTACK: [attack ADVANTAGE plus PROF + STR + 1 vs AC for 6d6 + STR crit 6d6 on 20]
    output ATTACK named "attack vs [AC]"

    ATTACK_GWM: [attack ADVANTAGE plus PROF + STR + 1 - 5 vs AC for 6d6 + STR + 10 crit 6d6 on 20]
    output
    [attack ADVANTAGE plus PROF + STR + 1 - 5 vs AC for 6d6 + STR + 10 crit 6d6 + ATTACK_GWM on 20] named "attack -5/+10 vs [AC] with GWM crits"

    output
    [attack ADVANTAGE plus PROF + STR + 1 vs AC for 6d6 + STR crit 6d6 + ATTACK on 20] named "attack vs [AC] with GWM crits"
}

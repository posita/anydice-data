function: exthr R:n again N:n {
 BONUSES: 0
 if N = R { BONUSES: (N >= 7) + (N = 10) + [exthr R again d10] }
 result: (N >= 7) + (N = 10) + BONUSES
}

function: exthr R:n and L:n again N:n {
 BONUSES: 0 
 if N = R { BONUSES: (N >= 7) + (N = 10) + [exthr R and L again d10] }
 if N = L { BONUSES: (N >= 7) + (N = 10) + [exthr R and L again d10] }
 result: (N >= 7) + (N = 10) + BONUSES
}

function: exthr R:n and L:n and M:n again N:n {
 BONUSES: 0 
 if N = R { BONUSES: (N >= 7) + (N = 10) + [exthr R and L and M again d10] }
 if N = L { BONUSES: (N >= 7) + (N = 10) + [exthr R and L and M again d10] }
 if N = M { BONUSES: (N >= 7) + (N = 10) + [exthr R and L and M again d10] }
 result: (N >= 7) + (N = 10) + BONUSES
}

EXTHR: [exthr 1 again d10]

loop N over {10..10} {
 output [lowest of 50 and NdEXTHR] named "[N]d"
}

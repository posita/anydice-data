function: roll SET:s le SUCC:n {
  EXPLODED: 0
  SUCCESSES: 0
  loop N over SET {
    DIEROLL: dN
    SUCCESSES: SUCCESSES + 1d (dN <= SUCC) 
  }
  result: SUCCESSES 
}

function: helper N:n le LE:n {
  if N <= LE { result: 1 }
  result: 0
}

loop TN over {1..3} {
output [roll {20, 12, 10, 8} le TN] named "roll {20..8} <= [TN]"
output [roll {20, 12, 10, 8, 6} le TN] named "roll {20..6} <= [TN]"
output [roll {20, 12, 10, 8, 6, 6} le TN] named "roll {20..6,6} <= [TN]"
output [roll {20, 12, 10, 8, 6, 4} le TN] named "roll {20..6,4} <= [TN]"
}

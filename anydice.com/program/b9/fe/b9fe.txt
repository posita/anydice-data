NUMBER: {1,1,1,1,2,2,2,3,3,4}
SIDES: {4,6,8,10,6,8,10,8,10,8}

function: roll rank RANK:n {
  if RANK = 0 {
    result: [explode 1d20]
  }
  else {
    result: [explode 1d20] + (RANK@NUMBER)d[explode d(RANK@SIDES)]
  }
}

function: roll rank RANK:n with advantage ADV:n {
  if RANK = 0 {
    result: [highest 1 of 2d[explode d20]]
  }
  else {
    result: [explode 1d20] + [highest (RANK@NUMBER) of (RANK@NUMBER + ADV)d[explode d(RANK@SIDES)]]
  }
}

function: roll rank RANK:n with disadvantage DIS:n {
  if RANK = 0 {
    result: [lowest 1 of 2d[explode d20]]
  }
  else {
    result: [explode 1d20] + [lowest (RANK@NUMBER) of (RANK@NUMBER + DIS)d[explode d(RANK@SIDES)]]
  }
}

output [roll rank 0]

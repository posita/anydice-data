function: doubles in R:s {
 if #R = 2 {
  if 1@R = 2@R { result: 1 }
 } else {
  if [count 1@R in R] = 2 { result: 1 }
 }
 result: 0
}

function: doubles in R:s with T:n advantage {
 if T {
  C: 1
  loop I over {1..2} {
   if 3@R = I@R { C: C + 1 }
  }
 } else {
  C: 1
  loop I over {3..4} {
   if 2@R = I@R { C: C + 1 }
  }
 }
 if C = 2 { result: 1 }
 result: 0
}

output 2d6+1d12 named "2d6+12"
output 3d8 named "3d8"
output 2d6+1@2d12 named "2d6+1@2d12"
output {1,2,3}@4d8 named "{1,2,3}@4d8"
output 2d6+2@2d12 named "2d6+2@2d12"
output {2,3,4}@4d8 named "{2,3,4}@4d8"
output [doubles in 2d6] named "Din2d6"
output [doubles in 3d8] named "Din3d8"

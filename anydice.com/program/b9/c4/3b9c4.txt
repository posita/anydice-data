function: outcome of DICE:s and THORNS:s
{
  BASECOUNT: [count {6} in DICE]
  MESSY: [count {4,5} in DICE] 
  
  EXTRA: [count {6} in THORNS] 
  EXTRA_MESSY: [count {4,5} in THORNS] 

  COUNT: BASECOUNT + EXTRA
  MESSYCOUNT: MESSY + EXTRA_MESSY

  if COUNT >= 2 { BASE: 4 } 
  else if COUNT = 1 { BASE: 3 } 
  else if MESSYCOUNT > 0 { BASE: 2 } 
  else { BASE: 1 } 

  THORNCOUNT: [count {7,8} in THORNS] 

  if BASE = 4 { result: BASE }
  else { BASE: BASE - THORNCOUNT }

  if BASE < 1 { BASE: 0 } 

  result: BASE
}

loop N over {1..3} {
  loop M over {0..4} {
    output [outcome of Nd6 and Md8] named "[N]d6 + [M]d8"
  }
}

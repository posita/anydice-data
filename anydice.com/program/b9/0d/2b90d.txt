function: roll A:n plus B:n vs D:n {
if A = 0 | A + B + 10 <= D { result: 1 }
if A = 18 | A + B > D + 10  { result: 4 }
if A + B > D { result: 3 }
result: 2
}


loop Y over {10} {
output [roll 2d{0..9} plus Y vs 2d{0..9}-2] named "Stat ([Y]) vs Easy"
output [roll 2d{0..9} plus Y vs 2d{0..9}] named "Stat ([Y]) vs Medium"
output [roll 2d{0..9} plus Y vs 2d{0..9}+2] named "Stat ([Y]) vs Hard"
output [roll 2d{0..9} plus Y vs 2d{0..9}+4] named "Stat ([Y]) vs Insane"
output [roll 2d{0..9} plus Y vs 2d{0..9}+6] named "Stat ([Y]) vs Impossible"
output [roll 2d{0..9} plus Y vs 2d{0..9}+8] named "Stat ([Y]) vs Impossible"
output [roll 2d{0..9} plus Y vs 2d{0..9}+10] named "Stat ([Y]) vs Ungodly"
}

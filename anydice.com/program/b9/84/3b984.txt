LOW: 9
MID: 14
HIGH: 19
ULTRA: 24

function: evaluate ROLL:s {
 if (ROLL <= LOW) { result: 0 }
 else {
  if (ROLL <= MID) { result: 1 }
  else {
   if (ROLL <= HIGH) { result: 2 }
   else {
    if (ROLL <= ULTRA) { result: 3 }
    else {
     result: 4
    }
   }
  }
 }
}

\OUTPUT LIST\
\
loop N over {1..5} {
 output [evaluate [highest 1 of Nd4]+d6] named "[N]d4"
 output [evaluate [highest 1 of Nd6]+d6] named "[N]d6"
 output [evaluate [highest 1 of Nd8]+d6] named "[N]d8"
 output [evaluate [highest 1 of Nd10]+d6] named "[N]d10"
 output [evaluate [highest 1 of Nd12]+d6] named "[N]d12"
}
\

loop N over {0..5} {
 output [evaluate d4+d6+N] named "d4+[N]"
 output [evaluate d6+d6+N] named "d6+[N]"
 output [evaluate d8+d6+N] named "d8+[N]"
 output [evaluate d10+d6+N] named "d10+[N]"
 output [evaluate d12+d6+N] named "d12+[N]"
}

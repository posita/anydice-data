function: norerolls F:n S:n {
   result: F+S
}

function: rerollboth R:n under N:n {
   if R < N { result: 2d6 } else {result: R}
}

output [norerolls 1d6 1d6] named "no rerolls"
output [count {5..12} in [norerolls 1d6 1d6]] named "total chance to roll 5+ without rerolls"

output [rerollboth 2d6 under 5] named "reroll both if the total is not 5 or more"
output [count {5..12} in [rerollboth 2d6 under 5]] named "total chance to roll 5+ when rerolling both dice"

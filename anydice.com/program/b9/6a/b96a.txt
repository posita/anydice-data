function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  result: CRITICALHIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: advantageattack FIRSTROLL:n and SECONDROLL:n vs DEFENSE:n {
 if FIRSTROLL= 20 | SECONDROLL= 20{
 result: CRITICALHIT }
 if FIRSTROLL+ ATTACK >= DEFENSE | SECONDROLL+ ATTACK >= DEFENSE{
 result: HIT}
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}


function: attack {
 result: [damage for ([attack d20 vs DEFENSE])]
}

function: advantageattack {
 result: [damage for ([advantageattack d20 and d20 vs DEFENSE])]
}

function: reroll A:n B:n C:n{ 
 if A <= B { 
  A:[reroll 1dC B C] 
 } 
 else { 
  result: A 
 }
}

\Description ---------------------------------------------\
\This function simulates the Great Weapon Master feat     \
\Variables -----------------------------------------------\
\A = Dice value I.E. A=6 for 1d6                          \
\B = Number of damage dice I.E. B = 2 for Greatsword (2d6)\
\C = Lower limit for reroll. Default 2                    \
\---------------------------------------------------------\

function: greatweaponmaster A:n B:n C:n{
 X:0
 loop N over {1..B} {
  X: X + [reroll 1dA C A]
 }
 result: X
}

\
output [greatweaponmaster DIE NUMDIE MIN] + MOD

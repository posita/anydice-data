NUMBEROFDICE: 0
DIE: 6

function: ob NUMBER:n {
  TOTALSUM: 0
  loop LOOPING over {1..NUMBER} {
    X: 1dDIE
    X: [checkmax X]
    TOTALSUM: TOTALSUM + X
  }
  result: TOTALSUM
}

function: checkmax N:n {
  if N = DIE {
    result: [ob 2]
  }
  result: N
}

set "maximum function depth" to 5
\ output [ob NUMBEROFDICE] named "OB[NUMBEROFDICE]D[DIE]" \
loop N over {1..8}{ output [ob N] named "OB[N]D[DIE]" }
loop N over {1..8}{ output  NdDIE named "[N]D[DIE]" }

\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n with DAM:n and CRIT:n{
 if ROLL >= CRITROLL { result: CRIT } 
 if ROLL + ATTACKMOD >= AC { result: DAM }
 result: 0
}

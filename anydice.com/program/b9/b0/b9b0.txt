NUMBEROFDICE: 1
DIE: 8

function: ob NUMBER:n {
  TOTALSUM: 0
  loop LOOPING over {1..NUMBER} {
    X: 1dDIE
    X: [checkmax X]
    TOTALSUM: TOTALSUM + X
  }
  result: TOTALSUM
}

function: checkmax N:n {
  if N = 6 {
    result: [ob 2]
  }
  result: N
}

set "maximum function depth" to 5
output [ob NUMBEROFDICE] named "OB[NUMBEROFDICE]D[DIE]"

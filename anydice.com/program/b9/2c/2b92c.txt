AC:15
SAVE:3

ATK:6
ROLL:1d20
BASE:1d4+4
CRIT:1d4

DC:11
PSN:3d6

TOTAL:0

function: A:n at B:n vs C:n {
if A=1 {result:0}
if A=20 {result:BASE+[1d20 at SAVE fails DC]+CRIT}
if A+B>=C {result:BASE+[1d20 at SAVE fails DC]}

}

function: dummy A:n at B:n vs C:n {
if A=1 {result:0}
if A=20 {result:1}
if A+B>=C {result:1}
result:0
}

function: A:n at B:n fails C:n {
if A+B<C {result:PSN}
result:PSN/2
}

loop N over {10..19} {

}

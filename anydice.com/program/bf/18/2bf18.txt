
function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

output[reroll {1,2} on 1d10+5+1d8 to d10] named "1r longsword 2h current"
output 1d10+7+1d8 named "1r longsword 2h new"

output[reroll {1,2} on 1d10+5+2d8 to d10] named "2r longsword 2h current"
output 1d10+7+2d8 named "2r longsword 2h new"

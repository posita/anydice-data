\ Note. Not all of this is perfectly accurate. DPR will be accurate, but the specific chance to achieve a damage value might not be. I can't be bothered fixing it. \

function: h ROLL:n AC:n{
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE + RIDER }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE + RIDER
 }
 result: DAMAGE + RIDER
}

function: ch ROLL:n AC:n{
 if ROLL + ATTACK >= AC | ROLL = 20 { result: 1}
 result: 0
}

function: save ROLL:n BONUS:n for DAMAGE:n saved SAVED:n against DC:n {
 if ROLL + BONUS >= DC {result: SAVED}
 result: DAMAGE
}

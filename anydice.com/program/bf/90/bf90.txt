function: attack ROLL:n vs DEFENSE:n {
	if ROLL = 1 { result: MISS }
	if ROLL = 20 { result: CRITICALHIT+BONUS }
	if ROLL + ATTACK >= DEFENSE { result: HIT+BONUS }
	result: MISS
}

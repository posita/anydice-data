\ Functions \
function: calculate modifier STAT {
    if STAT >= 10 {
        result: (STAT - 10) / 2
    }
    result: (STAT - 11) / 2
}

function: attack with cantrip DAMAGE:n on ARMOR_CLASS with ATTACK_MODIFIER {
    result: (1d20 + ATTACK_MODIFIER >= ARMOR_CLASS) * DAMAGE
}

function: enemy saving throw against SAVE_DIFFICULT with MODIFIER or DAMAGE:n {
    result: (1d20 + MODIFIER < SAVE_DIFFICULT) * DAMAGE
}

function: enemy attack with DAMAGE:n on ARMOR_CLASS with ATTACK_MODIFIER {
    result: (1d20 + ATTACK_MODIFIER >= ARMOR_CLASS) * (DAMAGE + 2)
}

N: 5
D: 20

function: roll X vs Y
{
  result: Nd(dD+X>Y)
}

function: pipeline X:n into Y vs Z
{
  if X = 0 {result: -1}
  if X = 1 {result: 0}
  if X = 2 {result: [roll Y-4 vs Z]-1}
  if X = 3 {result: [roll Y vs Z]-1}
  if X = 4 {result: [roll Y+4 vs Z]-1}
  if X = 5 {result: [roll Y+8 vs Z]-1}
}

output [pipeline [roll 0 vs 15] into 0 vs 15] 
output [roll 0 vs 15] -1

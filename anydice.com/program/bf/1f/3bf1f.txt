function: rollone DIE:n difficulty DIF specialty SPECIALTY {
      if ((DIE = 10) & SPECIALTY) {result: 2}
      else if (DIE >= DIF) {result: 1}
      else if (DIE = 1) {result: -1 }
      else {result: 0}
}
function: roll DICE difficulty DIF specialty SPECIALTY {
  SUCCESSES: 0
  if ((DICE > 0) & (DIF > 1) & (DIF < 11)) {
    loop I over {1..DICE} {
      SUCCESSES: SUCCESSES + [rollone d10 difficulty DIF specialty SPECIALTY]
    }
  }
  result: SUCCESSES
}

loop N over {4..14} {
  if N >= 8 {
    output (1d20 + N > 14) * (1d12 + 5 + 1d4) + (1d20 + N - 5 > 14) * (1d12 + 5 + 1d4) named "DPR Vs Soldier Lv5+"
  }
  else if N >= 6 & N < 8  {
    output (1d20 + N > 14) * (1d12 + 4 + 1d4) + (1d20 + N - 5 > 14) * (1d12 + 4 + 1d4) named "DPR Vs Soldier Lv3 - 4"
  } else {
    output (1d20 + N > 14) * (1d12 + 3) + (1d20 + N - 5 > 14) * (1d12 + 3) named "DPR Vs Soldier Lv 1 - 2"
  }

}

function: pbta X:n {
  if X > 9 { result: 2 }
  if X > 6 { result: 1 }
  result: 0
}
loop N over {-1..3} {
  output [pbta 2d6+N] named "pbta [N]"
}
function: alt X:s under Z:n {
  if (1@X) <= Z { result: 2 }
  if (2@X) <= Z { result: 1 }
  result: 0
}
loop N over {0..10} {
  output [alt 2d10 under N] named "alt [N]"
}

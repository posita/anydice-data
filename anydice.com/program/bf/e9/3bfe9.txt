\ 
0: Abject Failure (1-3)
1: Success, but with complication (4-6)
2: Clear success, no complication (7-9)
3: Critical (10+)
\

function: outcome ROLL:s {
  if 1@ROLL = 1 { result: 0}
  if 1@ROLL = 2 { result: 0}
  if 1@ROLL = 3 { result: 0}
  if 1@ROLL = 4 { result: 1}
  if 1@ROLL = 5 { result: 1}
  if 1@ROLL = 6 { result: 1}
  if 1@ROLL = 7 { result: 2}
  if 1@ROLL = 8 { result: 2}
  if 1@ROLL = 9 { result: 2}
  if 1@ROLL = 10 { result: 3}
  if 1@ROLL = 11 { result: 3}
  if 1@ROLL = 12 { result: 3}
  result: 0
}

loop SKILL over {4, 6, 8, 10, 12} {
  loop STAT over {1..5} {
    output [outcome STATdSKILL] named "[STAT]d[SKILL]"
  }
}

function: a AT:s p PA:s f F:n {
X: AT
if X >= 0{if F > 0{
X: AT/2 + F
}else{
X: AT/2
}}
result: [cap PA-X 3d20] < 0
}

function: cap WE:n WU:s{
if 1@WU = 20{
if 2@WU > WE{
result: -100}
else if 2@WU=20{
result: -100}}

if 3@WU = 1{
if 2@WU = 1{
result: 2*WE + 20 - 1@WU}
}else if 2@WU <= WE{
result: 2*(WE - 2@WU)}
result: WE - 2@WU
}
function: capd D:d X:n{
if -X > #D*5{result: #D}
else{
result: D+X
}}

function: dod X:s PA:d F:n D:d DD:n{
if X < 0{
result: [a X p PA f F] * ([capd D X] + DD)}
else{
result: [a X p PA f F] * (D + DD)
}
}

function: do AT:d PA:d F:n D:d DD:n{
 result: [dod [cap AT-F 3d20] PA F D DD]
}
AT: 20
PA: 10
F: 0
X: [cap AT-F 3d20]
output [a X p PA f F]

N: 4
D: 20

function: roll X vs Y
{
  result: Nd(dD+X>Y)
}

function: droll X vs Y
{
  result: Nd([lowest 1 of dD+X]>Y)
}

function: aroll X vs Y
{
  result: Nd([highest 1 of dD+X]>Y)
}

function: pipeline X:n into Y vs Z
{
  if X = 0 {result: -1}
  if X = 1 {result: 0}
  if X = 2 {result: [droll Y vs Z]-1}
  if X = 3 {result: [roll Y vs Z]-1}
  if X = 4 {result: [aroll Y vs Z]-1}
}

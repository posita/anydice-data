set "maximum function depth" to 5

function: replace lowest A:s with X:n{
if #A@A < X & X >= 0 {result: {1..(#A-1)}@A + X}
result: A
}

function: X:n dos A:n{
if X >= A { result: (X - A)/10 }
result: -1
}

function: hits DOS:n{
if MODE = 0 & DOS >= 0 {result: 1}
else if MODE = 1 & DOS >= 0
{
  if ROF <= DOS/2 {result: ROF}
  else {result: DOS/2 + 1}
}
else if MODE = 2 & DOS >= 0
{
  if ROF <= DOS {result: ROF}
  else {result: DOS + 1}
}
result: 0
}


function: soak DOS:n{
if DOS > 2 & RAZORSHARP {result: [highest of AP-(PEN*2) and 0] + TB}
else {result: [highest of AP-PEN and 0] + TB}}

function: damage{
result:  [highest NUMBEROFDTENS of (NUMBEROFDTENS+1)d[explode d10]]*TEARING + NUMBEROFDTENSd[explode d10]*!TEARING + BONUS}

function: total damage{
result: [hits DOS] * [highest of ([replace lowest [damage] with DOS] - [soak DOS]) and 0]
}


NUMBEROFDTENS: 6
BONUS: 6

ROF: 3
MODE: 0
TOHIT: d{100}

AP: 10
PEN: 6
TB: 15

TEARING: 0
RAZORSHARP: 0

DOS:[TOHIT dos 1d100]
output [total damage]

\**** Attack VS Saving Throw Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: savingthrow ROLLSAVE:n vs EST:n{
 if ROLLSAVE + EST < SPELLSAVEDC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

\looped against enemy Saving Throw Bonus from 0-2\
ENEMYSAVINGTHROW: d{0..2}

ATTACKMOD: 2
DMGBONUS: 0
PROFICIENCY: 2
ACCBONUS: 0
CRITROLL: 20
DAMAGE: 1d8 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d8 + ATTACKMOD + DMGBONUS
output [attackroll d20 vs ENEMYAC] named "+2 Attack Mod"

ATTACKMOD: 3
DMGBONUS: 0
PROFICIENCY: 2
ACCBONUS: 0
CRITROLL: 20
DAMAGE: 1d8 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d8 + ATTACKMOD + DMGBONUS
output [attackroll d20 vs ENEMYAC] named "+3 Attack Mod"

SPELLSAVEDC: 13
SPELLDMGBONUS: 0
DAMAGE: 1d8 + SPELLDMGBONUS
output [savingthrow d20 vs ENEMYSAVINGTHROW] named "DC13 spell"

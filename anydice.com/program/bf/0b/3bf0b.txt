function: range on DIE:n between LOW:n and HIGH:n {

if DIE >= LOW & DIE <= HIGH { result: 1 }

result: 0

}


function: count doubles in S:s {
 DOUBLES: 0
 loop P over {1..6} {
   DOUBLES: DOUBLES + ((S = P)/2)
 }
 result: DOUBLES
}

loop DIE over {4, 6, 8, 10, 12} {

loop RANK over {0..5} {

ROLL: RANKdDIE

if RANK = 0 {
ROLL: [lowest 1 of 2dDIE]
}

output [range on ROLL between 1 and 3] named "[RANK]d[DIE] failure"
output [range on ROLL between 4 and 6] named "[RANK]d[DIE] complication"
output [range on ROLL between 7 and 9] named "[RANK]d[DIE] success"
output [range on ROLL between 10 and 12] named "[RANK]d[DIE] critical"
if RANK != 0 {
output [count doubles in ROLL ] >=1 named "[RANK]d[DIE] twist"
}
} 

}

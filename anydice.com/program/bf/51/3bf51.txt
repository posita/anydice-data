function: hit ATTACKS:n {
  DICE: ATTACKS dQ                       \freeze the hits-dices sequence\
  result: [hit roll DICE]
}

function: hit roll DICES:s {
  HITS: [count {WS..3} in DICES]
  ZEROS:  [count 0 in DICES]
  ONES:   [count 1 in DICES]
  TWOS:   [count 2 in DICES]
  THREES: [count 3 in DICES]
  SETS: {ZEROS,ONES,TWOS,THREES}
  result: [save DICES with SETS]
}

function: save HITS:n with SETS:s {
  DICES: RD dQ                              \freeze the saves-dices sequence\
  result: [damages DICES with SETS]
}

function: damages POOL:s with THRESHOLDS:s {
  if(POOL >= 4@THRESHOLDS) { result: 4}
  if(POOL >= 3@THRESHOLDS) { result: 3}
  if(POOL >= 2@THRESHOLDS) { result: 2}
  if(POOL >= 1@THRESHOLDS) { result: 1}
  result: 0
}


\************************************************************************\
function: finaldamages THRESHOLDS:s from POOL:n{
  if(POOL >= 4@THRESHOLDS) { result: 4}
  if(POOL >= 3@THRESHOLDS) { result: 3}
  if(POOL >= 2@THRESHOLDS) { result: 2}
  if(POOL >= 1@THRESHOLDS) { result: 1}
  result: 0
}
function: test PDICES:s {
  RDICES: RD dQ
  HITS: [finaldamages PD_THRESHOLDS from PDICES]
  SAVES: [finaldamages RD_THRESHOLDS from RDICES]
  result: HITS - SAVES
}
\************************************************************************\

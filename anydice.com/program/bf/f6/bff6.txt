function: gwf A:s as B:n
{
   RES: 0

   loop C over {1..#(A)}
   {
      if C@A < 3
      {
         RES: RES + 1dB
      }
      else
      {
         RES: RES + C@A
      }
   }

   result: RES
}

output [gwf 1d10 as 10]+[gwf 1d6 as 6]+4 +[gwf 1d4 as 4]+[gwf 1d6 as 6]+4
output [gwf 1d10 as 10]+[gwf 1d6 as 6]+5 +[gwf 1d4 as 4]+[gwf 1d6 as 6]+5
output ([gwf 1d10 as 10]+[gwf 1d6 as 6]+5)*2 +[gwf 1d4 as 4]+[gwf 1d6 as 6]+5
output ([gwf 1d10 as 10]+[gwf 1d6 as 6]+5+4)*2 +[gwf 1d4 as 4]+[gwf 1d6 as 6]+5+4
output ([gwf 1d10 as 10]+[gwf 1d6 as 6]+5+5)*2 +[gwf 1d4 as 4]+[gwf 1d6 as 6]+5+5

N: 4
D: 20
S: -4
I: 4

function: roll X vs Y
{
  result: Nd(dD+X>Y)
}

function: pipeline X:n into Y vs Z
{
  if X = 0 {result: -1}
  if X = 1 {result: 0}
  if X = 2 {result: [roll Y+S+(I*0) vs Z]-1}
  if X = 3 {result: [roll Y+S+(I*1) vs Z]-1}
  if X = 4 {result: [roll Y+4+(I*2) vs Z]-1}
  if X = 5 {result: [roll Y+8+(I*3) vs Z]-1}
}

output [pipeline [roll 0 vs 15] into 0 vs 15] 

output [roll 0 vs 5] -1

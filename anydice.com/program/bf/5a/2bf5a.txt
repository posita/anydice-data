DICE: 1d20

MOD: 12

DMG_K: 1d6+4
DED_K: 1d8

DMG_W: 1d4+4
DED_W: 1d8

AC: 19

function: HIT:n vs AC:n for DMG:n plus DED:n {
  if (HIT-AC) >= 10 { result: (2*(DMG)+DED) }
  else if (HIT-AC) >= 0 { result: DMG }
  else { result: 0 }
}

loop N over {-5..5} {
  MOD_AC: AC+N
  output ([(DICE + MOD) vs MOD_AC for DMG_K plus DED_K] + [(DICE + MOD) vs MOD_AC for DMG_W plus DED_W]) - ([(DICE + MOD) vs MOD_AC for DMG_K plus DED_K] + [(DICE + MOD - 2) vs MOD_AC for DMG_K plus DED_K]) named "Diff AC: [MOD_AC]"
}

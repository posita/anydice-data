function: at least V:n from D:s {
 X: 0
 loop N over {1..#(D)} {
  if N@D >= V { X: X + 1 }
 }
 result: X
}

function: attack ATK:n vs ARM:n {
 result: [at least ARM from ATKd6]
}

function: defend DEF:n vs ACC:n {
 result: [at least ACC from DEFd6]
}

function: engage ATK:n attack vs ARM:n armor and DEF:n defense vs ACC:n accuracy {
 result: [attack ATK vs ARM] - [defend DEF vs ACC]
}

output [engage 3 attack vs 3 armor and 3 defense vs 3 accuracy]

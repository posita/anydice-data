function: roll TIMES:n reroll REROLL:n {
  ROLL: (TIMES)d{1, 0:4, -1}
  if REROLL = 0 { result: [([weals ROLL]) limit TIMES]}
  result: [([weals ROLL] + [roll [[rerolls ROLL] limit REROLL] reroll (REROLL-1)]) limit TIMES]
}

function: weals N:s {
  result: N = 1
}

function: rerolls N:s {
  result: N < 1
}

function: N:n limit M:n {
 if N > M {result: M}
 result: N
}


loop ABILITY over {1..7}{
 loop SKILL over {0..5}{
  output [roll ABILITY reroll SKILL] named "roll [ABILITY] reroll [SKILL]" 
 }
}

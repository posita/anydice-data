function: stepup APPROACH:n and SELF:n {

  ROLL: APPROACH + SELF
  ROLLRESULT: ROLL >= 7
  ROLLRESULT: ROLLRESULT + (ROLL >= 10)

  SUCCESSESRESULT: 0
  if APPROACH >= SELF & APPROACH >= 4 { SUCCESSESRESULT: 2 }
  if SELF > APPROACH & SELF >= 4 { SUCCESSESRESULT: 1 }

  HIGHROLL: APPROACH
  if SELF > APPROACH { HIGHROLL: SELF }
  HIGHROLLRESULT: HIGHROLL >= 4
  HIGHROLLRESULT: HIGHROLLRESULT + (HIGHROLL >= 6)

  result: HIGHROLLRESULT
}

loop N over {4,6,8,10,12} {
  output [stepup dN and d6] named "d6 and d[N]"
}

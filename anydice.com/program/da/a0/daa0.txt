function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = MISS {result: MISSDMG}
 if SAVAGEATTACKER = 1 {
  FIRSTDMG: 0
  SECONDDMG: 0
  if ATTACKRESULT = HIT {
   if GREATWEAPONFIGHTING = 1 {
    FIRSTDMG: [gwf rerolls NUMDIE]
    SECONDDMG: [gwf rerolls NUMDIE]
    if SECONDDMG > FIRSTDMG {result: SECONDDMG + NUMXDIEdXDIE + MOD}
    } else {result: FIRSTDMG + NUMXDIEdXDIE + MOD}
   } else {
    FIRSTDMG: HITDMG
    SECONDDMG: HITDMG
    if SECONDDMG > FIRSTDMG {result: SECONDDMG + NUMXDIEdXDIE + MOD}
    } else {result: FIRSTDMG + NUMXDIEdXDIE + MOD}
   }
  }
  if ATTACKRESULT = CRIT {
   if GREATWEAPONFIGHTING = 1 {
    FIRSTDMG: [gwf rerolls NUMCRITDIE]
    SECONDDMG: [gwf rerolls NUMCRITDIE]
    if SECONDDMG > FIRSTDMG {result: SECONDDMG + (2*NUMXDIE)dXDIE + MOD}
    } else {result: FIRSTDMG + (2*NUMXDIE)dXDIE + MOD}
   } else {
    FIRSTDMG: CRITDMG
    SECONDDMG: CRITDMG
    if SECONDDMG > FIRSTDMG {result: SECONDDMG + (2*NUMXDIE)dXDIE + MOD}
     } else {result: FIRSTDMG + (2*NUMXDIE)dXDIE + MOD}
   }
  }
 } else {
  if ATTACKRESULT = HIT {
   if GREATWEAPONFIGHTING = 1 {result: [gwf rerolls NUMDIE] + NUMXDIEdXDIE + MOD
   } else {result: HITDMG}
  }
  if ATTACKRESULT = CRIT {
   if GREATWEAPONFIGHTING = 1 {result: [gwf rerolls NUMCRITDIE] + (2*NUMXDIE)dXDIE + MOD
   } else {result: CRITDMG}
  }
 }
}

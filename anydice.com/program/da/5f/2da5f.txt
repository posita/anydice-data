  result: SHOTSd(1d101-1<HITCHANCE)*[min of 1dMAXDAMAGE+1-1-ARMOR and 0]
}

function: min of ROLL:n and NUMBER:n {
  if ROLL >= NUMBER {
    result: ROLL
  } else {
    result: NUMBER
  }
}

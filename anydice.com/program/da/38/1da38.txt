
function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}


function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: precise SUP:n attack ROLL:n vs AC:n for DMG:d {
    if ROLL + SUP >= AC {
        result: DMG
    }
    result: 0
}

function: battlemaster SUP_DICE:n superiority SUP:d precise PRECISE_RANGE:s attacks ATTACKS:n roll ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:s to AFTER_CRIT:d use USE:d pass PASS:d {
    if ROLL = 1 {
        result: PASS
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG + 2dSUP + AFTER_CRIT
    }
    if [PRECISE_RANGE contains (ROLL + BONUS)] {
        result: [precise SUP attack ROLL + BONUS vs AC for DMG] + USE
    }
    if ROLL + BONUS >= AC {
        if SUP_DICE >= ATTACKS {
            result: DMG + SUP + USE
        }
        result: DMG + PASS
    }

    result: PASS
}

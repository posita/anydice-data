function: dsix PROLL:n and NROLL:n{
  if PROLL = 6 {PROLL: 2}
  if NROLL = 6 {NROLL: 2}

  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}

function: broken dsix PROLL:n and NROLL:n and NUM:n{
  \if PROLL = NUM| NROLL = NUM{result: 0}\
  if PROLL = NUM| NROLL = NUM{
    if PROLL = NUM-1| NROLL = NUM-1{result: 0 }
  }
  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}
A: [lowest 1 of 2d6]
B: [lowest 1 of 2d6]

output [dsix A and B] named "2d6 fudge"

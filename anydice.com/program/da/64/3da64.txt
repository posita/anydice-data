function: results ROLL:n {
    if ROLL >= 4 & ROLL <= 7 {
        result: 1
    }
    if ROLL >= 8 & ROLL <= 10  {
        result: 2
    }
    if ROLL >= 11 & ROLL <= 17 {
        result: 3
    }
    if ROLL >= 18 & ROLL <= 20  {
        result: 4
    }
    result: 5
}
output [results 4d6]
output 4d6

function: range on DIE:n between LOW:n and HIGH:n {
  if DIE >= LOW & DIE <= HIGH { result: 1 }
  result: 0
}
output [range on [highest 4 of 8d6] between 4 and 7]

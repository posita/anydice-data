function: basic R:n with D:d {
if R >= 7 {result: D}
result: 0
}

function: twice R:n with D:d {
if R >= 10 {result: D + D}
if R >= 7 {result: D}
result: 0
}

function: max R:n with D:d {
if R >= 12 {result: 2*[maximum of D]}
if R >= 10 {result: D + D}
if R >= 7 {result: D}
result: 0
}


loop N over {2}{
output [basic 2d6+N with d8]     named "monk MOD +[N]"
output [basic 2d6+N with d10]    named "fighter MOD +[N]"
output [twice 2d6+N with d8]     named "monk flurry of blows MOD +[N]"
output [basic 2d6+N with d8+d4]  named "monk +d4 MOD +[N]"
output [basic 2d6+N with d10+d4] named "fighter merciless MOD +[N]"
output [max   2d6+N with d8]     named "monk finish them MOD +[N]"
output [basic 2d6+N with d8+2d4] named "monk +2d4 MOD +[N]"
output [basic 2d6+N with d10+d8] named "bloodthirsty MOD +[N]"
}

\**** Damage calculator normal Vs advantage ****\

function: attackroll ROLL:n vs AC:n LUCKY:n{
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { 
   if LUCKY = 0 {result: [attackroll d20 vs ENEMYAC 1]}
   if LUCKY = 1 {result: 0 }}
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 5
DMGBONUS: 0

ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 2d8 + ATTACKMOD + DMGBONUS

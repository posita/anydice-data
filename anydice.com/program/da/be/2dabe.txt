function: fits ROLL:s between DIFF:n and STAT:n {
  if DIFF <= ROLL & ROLL <= STAT {
    result: 1
  }
  if ROLL <= STAT {
    result: 0
  }
  result: -1
}

loop STAT over {1,5,10,15,20} {
  loop DIFF over {0,5,10,15,20} {
    output [fits d20 between DIFF and STAT] named "[DIFF] <= d20 <= [STAT]"
  }
}

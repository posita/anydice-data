function: dsix PROLL:n and NROLL:n{

  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}
A: [lowest 1 of 2d6]
B: [lowest 1 of 2d6]
C: [lowest 1 of 2d6]
D: [lowest 1 of 2d6]


function: dmg with AROLL:n and ARMOR:n{
  if AROLL < 1 { result: 0}
  result: 6d(1d10> (ARMOR - ATTACK))  
}
loop M over {1..10}{
  loop N over {-5..+5}{
    ATTACK: [dsix A and B] - [dsix C and D] + N

    output [highest 1 of 4d[dmg with ATTACK and M]] named "[N]"
  }
}

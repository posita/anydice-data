\W40K-simplified\

A: 2
WS: 3
W: 4
ARMOR_SAVE: 2
D: 1

function: hit ATTACKS:n {
  result: [hit roll ATTACKSd6]
}

function: hit roll DICES:s {
  HITS: DICES >= WS                        \[count {WS+WS_MODIFIER..6} in DICES]\
  result: [wound HITSd6]
}

function: wound HITS:s {
  WOUNDS: HITS >= W
  result: [armor save WOUNDS]
}

function: armor save WOUNDS:n {
  UNSAVED: WOUNDSd(d6 < ARMOR_SAVE)
  result: UNSAVED*D
}

output [hit 1] named "one attack (fast)"
output A d [hit 1] named "[A] independent attacks (fast)"

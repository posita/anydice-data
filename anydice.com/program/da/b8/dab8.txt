DICE: d8

function: basic R:n with D:d {
if R >= 7 {result: D}
result: 0
}

function: twice R:n with D:d {
if R >= 10 {result: D + D}
if R >= 7 {result: D}
result: 0
}

function: max R:n with D:d {
if R >= 12 {result: 2*[maximum of D]}
if R >= 10 {result: D + D}
if R >= 7 {result: D}
result: 0
}


loop N over {0..3}{
output [max 2d6+N with DICE] named "MOD +[N]"
}

function: dsix PROLL:n and NROLL:n and DROP:n{
  if PROLL > DROP | NROLL > DROP{result: 0}
  else{
  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}}
A: [lowest 1 of 2d10]
B: [lowest 1 of 2d10]
C: [lowest 1 of 2d6]
D: [lowest 1 of 2d6]


function: dmg with AROLL:n and ARMOR:n{
  if AROLL < 1 { result: 0}
  result: 6d(1d10> (ARMOR - ATTACK))  
}
\loop M over {1..10}{\
\  loop N over {-4..+4}{\
    ATTACK: [dsix A and B and 8]  

\    output [highest 1 of 1d[dmg with ATTACK and M]] named "[M]"\
\  }\
\}\

output ATTACK
output [dsix C and D and 6]

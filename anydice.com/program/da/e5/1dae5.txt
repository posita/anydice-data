
function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}


output [reroll {1} on d4 to d4]
output [reroll {1} on d6 to d6]
output [reroll {1} on d8 to d8]
output [reroll {1} on d10 to d10]
output [reroll {1} on d12 to d12]

function: blades A:n {
  if A >= 6 {result: 1}
  else if A >= 4 {result: 0}
  else {result: -1}
}
function: bwrapper P:s {
  N: [count {6} in P]
  if N > 1 {result: 10}
  else {result: [blades [high 1 of P]]}
}


function: high NUMBER:n of DICE:s {
 result: {1..NUMBER}@DICE
}

function: ct VALUES:s in SEQUENCE:s max M:n{
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
 }
 COUNT: COUNT + [count {M} in SEQUENCE]
 result: COUNT
}

\  output [blades [highest 1 of DICEd6]] named "Blades [DICE]d6"\
\}\


function: ph A:s {
  if A >= 10 {result: 1}
  else if A >= 8 {result: 0}
  else {result: -1}
}
function: phwrapper P:s {
  N: [count {10} in P]
  if N > 1 {result: 10}
  else {result: [ph [high 1 of P]]}
}

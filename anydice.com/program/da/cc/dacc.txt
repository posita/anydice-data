function: explode N:n {
 if N = 8 {
  result: 1 + [explode d8]
 }
 result: N >= 4
}

set "maximum function depth" to 2
EXPLODED: [explode d8]

function: score ROLL:s {
 EXPLOSIONS: 0
 SUCCESSES: 0
 loop P over {1..3} {
  if P@ROLL >= 4 {
   SUCCESSES: SUCCESSES + 1
   if P@ROLL = 8 {
    EXPLOSIONS: EXPLOSIONS + 1
   }
  }
 }
 result: SUCCESSES + EXPLOSIONS d EXPLODED
}

output [highest 3 of 5d(d8 >= 4)] named "normal"
output [score 5d8] named "with explosions"

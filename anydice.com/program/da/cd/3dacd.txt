function: count past VALUES:n in SEQUENCE:s {
 COUNT:(VALUES <= SEQUENCE)
 result: COUNT
}

function: count below VALUES:n in SEQUENCE:s {
 COUNT:(VALUES >= SEQUENCE)
 result: COUNT
}

function: pool DICESEQ:s threshold THRESH:n crit CRIT:n maxcrit MAX:n bonus BONUS:n stat STAT:n{
 STAT: STAT/2
 COUNT: 0
 CRITCOUNT: [count past CRIT in DICESEQ]
 if CRITCOUNT = 0 | MAX = 0| THRESH =6{ 
   COUNT: [count past THRESH in DICESEQ]
 }else if CRITCOUNT >= MAX{
   COUNT: [count past THRESH in DICESEQ]
   COUNT: COUNT + MAX
 }else{
   COUNT: [count past THRESH in DICESEQ]
   COUNT: COUNT + CRITCOUNT
 }
if (BONUS >= 0){
 result: (COUNT + [count past 5 in BONUS d6])
}else{result: (COUNT - [count past 5 in (BONUS*-1) d6])}

}

function: limit SEQUENCE:s to NUMBER:n {
 NEWSEQ : {}
 loop P over {1..#SEQUENCE}{
  if (P@SEQUENCE > NUMBER){
  NEWSEQ: {NEWSEQ, NUMBER}
 }else if (P@SEQUENCE <= -1){
  NEWSEQ : {NEWSEQ, 0}
 }else{
  NEWSEQ : {NEWSEQ, P@SEQUENCE}
 }
}
result: NEWSEQ
}

function: sspread SEQ:s{
NEWSEQ: {}

loop P over {1..#SEQ}{
if P@SEQ < 0 {NEWSEQ : {0, NEWSEQ}}
else if P@SEQ < 4{NEWSEQ : {P@SEQ,NEWSEQ}}
else if P@SEQ < 6 {NEWSEQ :{4, NEWSEQ}}
else{NEWSEQ:{6, NEWSEQ}}

result: NEWSEQ
}
}

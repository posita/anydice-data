function: basic R:n with D:d {
if R >= 7 {result: D}
result: 0
}

function: twice R:n with D:d {
if R >= 10 {result: D + D}
if R >= 7 {result: D}
result: 0
}

function: max R:n with D:d {
if R >= 12 {result: 2*[maximum of D]}
if R >= 10 {result: D + D}
if R >= 7 {result: D}
result: 0
}


loop N over {2}{
output [basic 2d6+N with d6+1]      named "monk MOD +[N]"
output [basic 2d6+N with d6+2]      named "fighter MOD +[N]"
output [twice 2d6+N with d6+1]      named "monk flurry of blows MOD +[N]"
output [max   2d6+N with d6+1]      named "monk finish them MOD +[N]"
output [basic 2d6+N with 1@2d6+2+2] named "fighter merciless MOD +[N]" \the second +2 is for the damage forward\
}

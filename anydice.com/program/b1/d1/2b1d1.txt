\ 0 is a miss, 1 is a weak hit, 2 is a strong hit \

function: progress PROG:n {
  result: 2d(PROG > d10)
}

function: shift ROLL:n SHIFT:n {
  if ROLL <= SHIFT {
     result: SHIFT +1
  } else {
    result: ROLL
  }
}

function: action AROLL:n AMOD:n MOM:n SHIFT:n {
  if MOM < 0 & AROLL = -MOM {AROLL: 0}
  AROLL: [shift AROLL SHIFT]
  result: 2d([highest of [lowest of AROLL + AMOD and 10] and MOM] > d10)
}

loop X over {1..1} {
  loop N over {0..3} {
    output [action 1d6 X 0 N] named "1d6+[X]Shift [N] Momentum 0"
  }

}

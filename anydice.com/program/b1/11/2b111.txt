function: ROLL:n bonus BONUS:n crit CRIT:n armor AR:n defense DF:n {
 DIV:1
 if (ROLL>=CRIT){result:200}
 ROLL:ROLL+BONUS
 if (((ROLL<=AR)&AR>DF)|((ROLL<=DF)&DF>=AR)){result:0}
 if (ROLL<=((AR+DF)/2)){DIV:DIV*2}
 if (ROLL<=(AR+DF)){DIV:DIV*2}
 result:100/DIV
}
ROLL:2d10
BONUS:0
CRIT:18
AR:2
DF:4
output [ROLL bonus BONUS crit CRIT armor AR defense DF]

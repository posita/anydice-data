function: ROLL:n bonus BONUS:n crit CRIT:n armor AR:n defense DF:n {
 if (ROLL>=CRIT){result:200}
 ROLL:ROLL+BONUS
 \if ((ROLL<=AR)|(ROLL<=DF)){result:0}\
 if (ROLL<=((AR+DF)/2)){result:0}
 if (ROLL<=(AR+DF)){result:50}
 result:100
}
ROLL:2d10
BONUS:6
CRIT:18
AR:10
DF:6
output [ROLL bonus BONUS crit CRIT armor AR defense DF]

loop FACE over {0, 2, 4} {
  loop BURN over {0..6} {
    loop SKILL over {0..3} { 
      FACES: 6+FACE
      PART: (FACES-6)/2
      if SKILL < PART {
        if (SKILL+BURN) > PART {
          REGULAR: BURN - (PART - SKILL)
        }
        else {
          PART: SKILL + BURN
          REGULAR: 0
        }
      } else if SKILL > PART {
        REGULAR: (SKILL - PART)+BURN
      } else {
        REGULAR: BURN
      }
      
      output (PART d (d FACES >5)) + (REGULAR d (d6>5)) > 0 named "Spec:[PART]d[FACES] Reg:[REGULAR]d6"
    }
  }
}

function: test S:n and L:n and H:n vs D:n {
  T: H+D
  R: 0
  if S >= T { 
    R: 1 
    if S = 6 & L > 6 { R: R + 100 }
  }
  if L >= T { R: R + 10 }
  if S = L { R: R + 100 }
  result: R
}

D: d{0..5}
SKILL: d6
LUCK: [explode dD]
HAZARD: d{0..9}

loop DIFFICULTY over {-9..9} {
  output [test SKILL and LUCK and HAZARD vs DIFFICULTY] named "[DIFFICULTY]"
}

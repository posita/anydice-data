function: skill roll helper D:s BONUS:n {
  SORTED: [sort D]
  if (BONUS < 0) {
    SORTED: [reverse SORTED]
  }
  SIXES: (1@SORTED = 6) + (3@SORTED = 6)
  ONES: (1@SORTED = 1) + (3@SORTED = 1)
  result: 2@SORTED + SIXES - ONES
}

function: skill roll BONUS:n {
  result: [skill roll helper (3+[absolute BONUS])d6 BONUS]
}

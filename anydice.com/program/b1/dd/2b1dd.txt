loop FACE over {0, 2, 4} {
  loop BURN over {0..6} {
    loop SKILL over {0..3} { 
      FACES: 6+FACE
      PART: (FACES-6)/2
      if SKILL < PART {
        if (SKILL+BURN) > PART {
          REGULAR: BURN - (PART - SKILL)
        }
        else {
          SPECIAL: SKILL + BURN
          REGULAR: 0
        }
      } else if SKILL > PART {
        REGULAR: (SKILL - PART)+BURN
        SPECIAL: PART
      } else {
        REGULAR: BURN
        SPECIAL: PART
      }
      
      output (SPECIAL d (d FACES >5)) + (REGULAR d (d6>5)) > 0 named "Spec:[SPECIAL]d[FACES] Reg:[REGULAR]d6 P:[PART] S:[SKILL] B:[BURN]"
    }
  }
}

function: dndmod MOD:n with A:n max MAX:n {
  if A = 1 { result: 1 }
  if A = 20 { result: MAX+MOD }
  if (A + MOD) <= 0 { result: 1 }
  result: A + MOD
}

function: highish NUMBER:n of A:n and B:n {
 if A = 1 { result: 1 }
 if A = 20 { result: 20 }
 result: [highest of A and B]
}

function: prof PROF:n for A:n and B:n {
 if A = 1 { result: 1 }
 if PROF >= 0 & A <= PROF {
  result: [highest of A and B]
 }
 result: A
}
function: godprof PROF:n for A:n and B:n and C:n {
 if A <= B & A <= PROF { A: C }
 else if B <= A & B <= PROF { B: C }
 result: [highest of A and B]
}

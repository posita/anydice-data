MISS: 0
HIT: 1
CRIT: 3

function: attack ROLL:n and modifier ATTMOD:n vs ac DEFENCE:n {

if ROLL = 20 {result: CRIT}
if ROLL = 1 {result: MISS}
if ROLL +ATTMOD >= DEFENCE {result: HIT}
if ROLL +ATTMOD < DEFENCE {result: MISS}
else {result: MISS}

}

function: damage DAMAGE:d plus DAMMOD:n for ROLLRES:n {

if ROLLRES = MISS {result: 0}
if ROLLRES = HIT {result: DAMAGE + DAMMOD}
if ROLLRES = CRIT {result: [maximum of DAMAGE] + DAMAGE + DAMMOD}

}

loop AC over {5..25}{
output [damage 1d6 plus 1 for [attack [highest 1 of 1d20] and modifier 0 vs ac AC]] named "[AC]"
}

\ Set target numbers \
TA: 7
TB: 6

\ Build dice with zeros if unsuccessful and roll 2 of them \
DA: 2d{0:10-TA, 1..TA}
DB: 2d{0:10-TB, 1..TB}

\ create a 'sign' function \
function: sign A:s {
  if A > 0 {result: 1} 
  if A < 0 {result: -1}
  result: 0
}

\ multiply the larger die by 100 and take B away from A \
\if the result is less than 0 B wins, greater than 0 A wins, equal to 0 we have a tie \
output [sign (100*1@DA + 2@DA - 100*1@DB - 2@DB)]

\
Apocalypse World dice with crits on 12+)
Result -1 is Miss; result 0 is Weak Hit; result 1 is Hit; result 2 is Critical Hit
\

function: aw ROLL:n {
if ROLL >= 12 { result: 2 }
if ROLL >= 10 { result: 1 }
if ROLL <= 6 { result: -1 }
result: 0
}

loop DICE over {0} {
output [aw 2d6 + DICE] named "[DICE]"
}

function: successes in ROLL:s vs TN:n {
  result: [count {TN..6} in ROLL]
}
function: duds in ROLL:s vs TN:n {
  result: [count {1..TN-1} in ROLL]
}

function: successes in ROLL:s vs TN:n with P:n points {
  SUCC: 0
  REM: P
  loop X over {1..(#ROLL)}{
    N: X@ROLL
    if N >= TN {
      SUCC: SUCC + 1
    }
    else {
      REQ: TN - N
      if REQ <= REM {
        REM: REM - REQ
        SUCC: SUCC + 1
      }
    }
  }
  result: SUCC
}
      

TN: 6
SKILL: 6
PUSH: {1d6:(TN-1), {TN..6}}

output [successes in SKILLd6 vs TN] named "Normal"
output [successes in SKILLdPUSH vs TN] named "Pushed"
loop P over {1..3}{
  output [successes in SKILLd6 vs TN with P points] named "Normal ([P] Stamina)"
  output [successes in SKILLdPUSH vs TN with P points] named "Pushed ([P] Stamina)"
}

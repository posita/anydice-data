function: ROLL:n bonus BONUS:n crit CRIT:n armor AR:n evade EV:n {
 if (ROLL>=CRIT){result:200}
 if (ROLL<=(20-CRIT)){result:0}
 ROLL:ROLL+BONUS
 DIV:1
 if (ROLL<(AR/2) | ROLL<(EV/2)) {result:0}
 if (ROLL<AR) {DIV:DIV*2}
 if (ROLL<EV) {DIV:DIV*2}
 result:100/DIV
}
ROLL:2d10
BONUS:0
CRIT:17
AR:11
EV:0
output [ROLL bonus BONUS crit CRIT armor AR evade EV]

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: minroll REROLL:s on ROLL:n to REPLACEMENT:n {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: og great weapon master NUMDICE:n d DICEVAL:d {
    TOT: 0
    loop N over {1..NUMDICE}{
        TOT: TOT + [reroll {1,2} on DICEVAL to DICEVAL]
    }
    result: TOT
}

output [og great weapon master 2 d 6]

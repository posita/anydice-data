function: ATTACK:n vs DC:n rolling two DAMAGE:n sided dice plus CRIT:n{
 THRESH: DC/10
 if ATTACK < DC { result: 0 }
 if ATTACK > 60+4*THRESH { result: 4dDAMAGE+CRIT }
 result: 2dDAMAGE
}

output [1d100 vs 26 rolling two 6 sided dice plus 7] named "Ruby vs Red"
loop N over {2..5} {
 output Nd[1d100 vs 26 rolling two 6 sided dice plus 7] named "Ruby vs Red ([N] rounds)"
}

 output Nd[1d100 vs 16 rolling two 8 sided dice plus 7]+Nd[1d100 vs 16 rolling two 5 sided dice plus 7]+Nd[1d100 vs 16 rolling two 10 sided dice plus 7] named "WBY vs Brute ([N] rounds)"
}

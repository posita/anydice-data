STR:3
PRF:3
HIT:STR+PRF+1
RAW:STR+1
BRK:13+HIT

function: attack ROLL:n BNS:n for DIE:d DMG:n vs AC:n {
  if (ROLL = 1) { result: 0 }
  if (ROLL = 20) { result: DIE+DIE+DMG }
  if (ROLL+BNS>=AC) { result: DIE+DMG }
  result: 0
}

loop TAR over {10..20} {
    output [attack 1@2d20 HIT for 1d8 RAW vs TAR]+[attack 1@2d20 HIT for 1d8 RAW vs TAR]+[attack 1@2d20 HIT for 1d4 RAW vs TAR] named "vs [TAR]"
    output [attack 1@2d20 HIT-5 for 1d8 RAW+10 vs TAR]+[attack 1@2d20 HIT-5 for 1d8 RAW+10 vs TAR]+[attack 1@2d20 HIT-5 for 1d4 RAW+10 vs TAR] named "GWM vs [TAR]"
}

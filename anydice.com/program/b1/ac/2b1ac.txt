function: test S:n and L:n and H:n vs D:n {
  T: H+D
  R: 0

  \ Check skill vs target\
  if S >= T { R: 1 }

  \ Check luck vs target \
  if L >= T { R: R + 10 }

  \ Check for critical \
  if S = L { R: R + 100 }

  \ Allow success on 6,6,0 \
  if R = 100 & S = L & H = 0 { R: 10 }

  result: R
}

SKILL: d6
LUCK: d6
HAZARD: d{0..9}

loop DIFFICULTY over {-9..9} {
  output [test SKILL and LUCK and HAZARD vs DIFFICULTY] named "[DIFFICULTY]"
}

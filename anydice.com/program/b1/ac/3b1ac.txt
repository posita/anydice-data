\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ABLSCORE + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\Static variables\
ENEMYAC: {15..22}  \ range of target AC\
PROFICIENCY: 5
ABLSCORE: 5
ADV: 3 \Advantage, 1 = normal, 2=adv, 3=elven\
FULLCALC: 0 \Calc each AC = 1, Mean only = 0\

\ Below here, copy / paste and tweak each block for each attack variant \


\ Greatsword +1 crit 20 \
DMGBONUS: 1
ACCBONUS: 1
CRITROLL: 20
DAMAGE: 2d6 + ABLSCORE  + DMGBONUS
CRITDAMAGE: 4d6 + ABLSCORE  + DMGBONUS
if FULLCALC = 1 {
  loop AC over ENEMYAC {
   output [attackroll [highest 1 of ADVd20] vs AC] named "Greatsword +1 Crit 20 vs. AC [AC]|"
  }
}
output [attackroll [highest 1 of ADVd20] vs dENEMYAC] named "Greatsword +1 Crit 20 MEAN|"

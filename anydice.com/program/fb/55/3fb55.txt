function: stat full A:n B:n C:n
{
 if A = B & B = C { S: 2}
 else if A = B | B = C | A = C { S: 1 }
 
 if A >= B & B >= C { result: A + B + S }
 else if A >= B & C >= B { result: A + C + S }
 else { result: B + C + S }
}

function: stat small A:n B:n C:n
{
 if A = B | B = C | A = C { S: 1 }
 
 if A >= B & B >= C { result: A + B + S }
 else if A >= B & C >= B { result: A + C + S }
 else { result: B + C + S }
}

function: stat pair full A:n B:n C:n
{
 if A = B & B = C { S: 2}
 else if A = B | B = C | A = C { S: 1 }
 
 result: A + B + S
}

function: stat pair small A:n B:n C:n
{
 if A = B | B = C | A = C { S: 1 }
 
 result: A + B + S
}

output [stat full 1d4 1d4 1d4]
output [stat small 1d4 1d4 1d4]
output [stat pair full 1d4 1d4 1d4]
output [stat pair small 1d4 1d4 1d4]

FAIL: -1         \Fail the check, no information\
PARTSUCCESS: 0   \Fail, but less than 5\
SUCCESS: 1
HIGH: 15         \DC of arcana check\
LOW: HIGH - 5    \Fail by 5 or more\
BONUS: 0         \The accumulated bonus to the roll\

function: save ROLL:n {
 if ROLL <= LOW { result: FAIL }
 if ROLL >= HIGH { result: SUCCESS  }
 result: PARTSUCCESS
}

loop ADJ over {5..10}{ 
\Between the {} either enter BONUS, (in all caps), or a number,\ 
     \to loop through a range of values enter either a string of\
         \comma seperated numbers, (e.g. 0,2,4,6)\ 
         \or a range, (e.g. 5..15)\

\<-- Remove comment markers surrounding advantage or disadvantage if you want to see those results -->\
\output [save [lowest 1 of 2d20] + ADJ] named "DC [HIGH] Arcana Check + [ADJ] bonus, with Disadvantage"\
output [save 1d20 + ADJ] named "DC [HIGH] Arcana Check + [ADJ] bonus" 
\output [save [highest 1 of 2d20] + ADJ] named "DC [HIGH] Arcana Check + [ADJ] bonus, with Advantage"\
}

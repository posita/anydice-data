NATT: 1
AB: 6
ATTABILITYBONUS: 1
ATTCHARBONUS: 4
DMGDICE: 1
DMGDICESIZE: 8
DMGBONUS: 2
set "explode depth" to 1

ACN: 18
ACMAX: 22

function: attack ATTRES:n vs ACEFF:n {
if ATTRES = 1 {result: 0}
if ATTRES = 20{result: [damage 2]}
if ATTRES > ACEFF{result: [damage 1]}
else {result:0}
}

function: damage N{
result: N*[explode DMGDICEdDMGDICESIZE]+DMGBONUS
}

function: attacks NAT:n{
if NAT <= 0 {result: 0}
result: [attack 1d20 vs ACEFF] + [attacks NAT-1]
}

loop ACEF over {ACN..ACMAX} {
ACEFF: ACEF - AB - ATTABILITYBONUS - ATTCHARBONUS
 output [attacks NATT] named "[ACEF] AC"
}

function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT + HIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}

function: gwf damage with SIDES_ON_DMG_DIE:n sided die {
    GWF_DIE: {{1..2}:2, {3..SIDES_ON_DMG_DIE}:SIDES_ON_DMG_DIE+2}
    result: dGWF_DIE
}

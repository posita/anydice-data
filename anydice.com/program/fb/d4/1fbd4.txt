function: rerollten A:n if below B:n {
 if A < 3 { result: 1d10 }
 result: A
}

function: rerollsix A:n if below B:n {
 if A < 3 { result: 1d6 }
 result: A
}

function: rerollfour A:n if below B:n {
 if A < 3 { result: 1d4 }
 result: A
}

function: R:n attack D:d damagevsac A:n with bonus B:n and prof C:n {
  if R = 1  { result: 0}
  if R+B+C < A  { result: 0}
  result: D + B
}

}


loop A over {10..20}{
P:4
M:5
X: d20
Y: [lowest 1 of 2d20]
D: 2
H: M
function: Z:n crit {
if Z = 20 {result: 2d8 + M}
result: 1d8 + M
}
function: N:n nat {
if N = 1 {result: 0}
result: 1
}
output [highest of [lowest of X + M + P - A + 1 and 1] and 0] * ([X crit]) * ([X nat]) named "reg [A]"
output [highest of [lowest of Y + M + P + H - A + 1 and 1] and 0] * ([Y crit] + D) * ([Y nat]) named "dis [A]"
}

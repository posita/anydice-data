function: D:n plus R:n and P:n out of Q:n {
  if Q <= P { result: [highest 1 of (R+2)d(D)] }
  result: [highest 1 of (R+1)d(D)]
}
function: D:n plus R:n point P:n {
  result: [D plus R and P out of 1d10]
}

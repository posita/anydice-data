function: reroll lowest of ROLL:s as REROLL:d if less than MIN:n {
  LOWEST: (#ROLL)@ROLL       \ the lowest die is sorted last \
  if LOWEST >= MIN {
    result: ROLL + 0
  } else {
    REST: {1..#ROLL-1}@ROLL  \ all but the lowest die \
    result: REST + REROLL
  }
}

output 3d6
output [highest 3 of 4d6]
output [reroll lowest of 3d6 as d6 if less than 6]
output 2d6+6
output 2d10
output [reroll lowest of 2d10 as d10 if less than 5]

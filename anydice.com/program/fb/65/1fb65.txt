DICENUMBERS: {2..8}
DICE: {4,6,8,10}
SOFTCRITVALUES: {0:3,4:2,5:2,6:2,7:2,8}
SOFTCRITCANCELUPPERLIMITVALUES: {0:3,1:2,2:2,3:2,4:2,5}
SOFTCRITDIFF:4

function: crit or not A:s LOWMAX:n HIGHMIN:n DICEDIFF:n
{
 LOW: 0
 HIGH: 0

 if DICEDIFF@A < HIGHMIN
 {
  result: 0
 }

 loop X over {1..#A}
 {
  if X@A >= HIGHMIN
  {
   HIGH: HIGH+1
  }

  if X@A <= LOWMAX
  {
   LOW: LOW+1
  }
 }
  result: HIGH-LOW >= DICEDIFF
}

loop NOOFDICE over DICENUMBERS
{
  loop DIETYPE over DICE
  {
   SOFTCRITTHRESHOLD: DIETYPE@SOFTCRITVALUES

   SOFTCRITCANCELUPPERLIMIT: DIETYPE@SOFTCRITCANCELUPPERLIMITVALUES
 
   if DIETYPE <= NOOFDICE*2
   {
    output [crit or not NOOFDICEdDIETYPE SOFTCRITCANCELUPPERLIMIT SOFTCRITTHRESHOLD SOFTCRITDIFF] named "Unoptimised Chance of soft crit by rolling [SOFTCRITDIFF] [SOFTCRITTHRESHOLD]+ more than [SOFTCRITCANCELUPPERLIMIT] or lower for [NOOFDICE]d[DIETYPE]"
   }
 }
}

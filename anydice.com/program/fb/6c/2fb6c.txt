NATT: 2
AB: 5
ATTABILITYBONUS: 0
ATTCHARBONUS: 2
DMGDICE: 1
DMGDICESIZE: 12
DMGBONUS: 2

FUNBLEVAL: 0
CRITVAL: 20
CRITICAL: 2

ACN: 10
ACMAX:30
HP: 0

function: attack ATTRES:n vs ACEFF:n {
if ATTRES = FUNBLEVAL {result: 0}
if ATTRES = CRITVAL {result: [damage CRITICAL]}
if ATTRES > ACEFF {result: [damage 1]}
else {result:0}
}

function: damage MULT{
if MULT = 2 {result: DMGDICE*DMGDICESIZE + DMGDICEdDMGDICESIZE +DMGBONUS}
else {result: DMGDICEdDMGDICESIZE+DMGBONUS}
}

function: attacks NAT:n{
if NAT <= 0 {result: 0}
ATTREMAINING: NAT -1 
result: [attack 1d20 vs ACEFF] + [attacks ATTREMAINING]
}

loop ACEF over {ACN..ACMAX} {
ACEFF: ACEF - AB - ATTABILITYBONUS - ATTCHARBONUS
if HP <= 0 {output [attacks NATT] named "[ACEF] AC"}
else {output [attacks NATT]>=HP named "[HP] damage to [ACEF] AC"
}
}

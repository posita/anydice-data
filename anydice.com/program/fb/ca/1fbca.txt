function: fate ROLL:n TN:n {
  \ success with style \
  if ROLL - TN >= 3 {
    result: 2
  }

  \ success \
  if ROLL > TN {
    result: 1
  }

  \ tie \
  if ROLL = TN {
    result: 0
  }

  \ failure \
  result: -1

}

GMBASE: 0
PCBASE: 0

loop GMROLL over {-4..4} {
  TN: GMBASE + GMROLL
  output [fate PCBASE+4d{-1..1} TN] named "TN:[TN]"
}

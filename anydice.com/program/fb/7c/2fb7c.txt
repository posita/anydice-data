set "maximum function depth"  to 5

function: uexplode DIE:d {
 MAX: [maximum of DIE]
 result: [uexplode helper DIE]
}


function: uexplode helper R:s {
 N: #R
 OUT: N
 loop X over {1..N} {
  if X@R = MAX { OUT: OUT + 1 }
 }
 if OUT > N { result: [uexplode helper OUTdMAX] }
 result: R
}

output [uexplode 1d10]

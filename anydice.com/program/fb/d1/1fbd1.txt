\**** Damage calculator ****\

function: attackroll ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: ROLL + ATTACKBONUS - DEF + DMGBONUS + (WPNDAMAGE)*2 } 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS >= DEF { result: ROLL + ATTACKBONUS - DEF + DMGBONUS + WPNDAMAGE }
 result: 0
}

\looped against enemy DEF\
ENEMYDEF: d{8}

ATTACKBONUS: 3
DMGBONUS: 0
CRITROLL: 12
ATTACKTIMES: 1
WPNDAMAGE: 5
output ATTACKTIMESd[attackroll 2d6 vs ENEMYDEF] named "Longsword +5"
output 2d6

NATT: 1
AB: 6
ATTABILITYBONUS: 1
ATTCHARBONUS: 4
DMGDICE: 1
DMGDICESIZE: 8
DMGBONUS: 2
set "explode depth" to 3

ACN: 10

function: attack ATTRES:n vs ACEFF:n {
if ATTRES = 1 {result: 0}
if ATTRES > ACEFF | ATTRES = 20{
result: [damage 1]
}
else {result:0}
}

function: damage N{
result: 
[explode DMGDICEdDMGDICESIZE]+DMGBONUS
}

loop ACEF over {ACN..ACN+20} {
ACEFF: ACEF - AB - ATTABILITYBONUS - ATTCHARBONUS
 output [attack NATTd20 vs ACEFF] named "[ACEF] AC"
}

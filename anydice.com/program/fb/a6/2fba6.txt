function: emphasis A:n and B:n {
  \ absolutna odleglosc wyniku od 10.5 \
  AD: [absolute A*10 - 105]
  BD: [absolute B*10 - 105]
  if AD > BD {
    result: A
  } else if BD > AD {
    result: B
  }
  \ jesli wypadnie 10 i 11 to wybieramy 11 \
  if (A=10&B=11)|(A=11&B=10) {
    result: 11
  }
  result: d{}
}

output [emphasis d20 and d20]

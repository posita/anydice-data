DICENUMBERS: {2..7}
DICE: {12}
SOFTCRITVALUES: {0:3,4:2,5:2,6:2,7:2,8}
SOFTCRITCANCELUPPERLIMITVALUES: {0:3,1:2,2:2,3:2,4:2,5}
SOFTCRITDIFF:4

function: crit or not A:s LOWMAX:n HIGHMIN:n DICEDIFF:n
{
 if #A < DICEDIFF
 {
  result: 0
 }

 if DICEDIFF@A < HIGHMIN
 {
  result: 0
 }
 else
 {
  if #A = DICEDIFF
  {
   result: 1
  }

  if#A@A >LOWMAX
  {
   result: 1
  }

  NETHIGH: DICEDIFF

  loop X over {DICEDIFF+1..#A}
  {
   if X@A >=HIGHMIN
   {
    NETHIGH: NETHIGH+1
   }

   if X@A <=LOWMAX
   {
    NETHIGH: NETHIGH-(#A-X)-1
    if NETHIGH < DICEDIFF
    {
     result: 0
    }
    else
    {
     result: 1
    }
   }

   if X@A>LOWMAX & X@A<HIGHMIN
   {
    if #A-X>=3
    {
     loop Y over {#A..X+1}
     {
      if Y@A <=LOWMAX
      {
       NETHIGH: NETHIGH-1
       if NETHIGH < DICEDIFF
       {
        result: 0
       }
      }
      else
      {
       result: 1
      }
     }
    }
   }



   if NETHIGH-(#A-X)>=DICEDIFF
   {
    result: 1
   }

  }
 }
}

loop NOOFDICE over DICENUMBERS
{
  loop DIETYPE over DICE
  {
   SOFTCRITTHRESHOLD: DIETYPE@SOFTCRITVALUES

   SOFTCRITCANCELUPPERLIMIT: DIETYPE@SOFTCRITCANCELUPPERLIMITVALUES
 
   if DIETYPE <= NOOFDICE*2
   {
    output [crit or not NOOFDICEdDIETYPE SOFTCRITCANCELUPPERLIMIT SOFTCRITTHRESHOLD SOFTCRITDIFF] named "Optimised Chance of soft crit by rolling [SOFTCRITDIFF] [SOFTCRITTHRESHOLD]+ more than [SOFTCRITCANCELUPPERLIMIT] or lower for [NOOFDICE]d[DIETYPE]"
   }
 }
}

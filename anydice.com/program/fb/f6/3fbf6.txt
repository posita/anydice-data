DISASTER: -1
GRIM: 0
MESSY: 1
PERFECT: 2
CRITICAL: 3

function: fromhighest H:n {
  if H <= 3 { result: GRIM }
  if H < 6 { result: MESSY }
  result: PERFECT
}

function: getoutcome rolls ROLLS:s cuts CUTS:n {
  if {1,2}@ROLLS = 12 { result: CRITICAL }
  OUTCOME: [fromhighest {1}@ROLLS]
  if CUTS > OUTCOME { result: DISASTER }
  result: OUTCOME - CUTS
}

loop THORNS over {0..3} {
  loop DICE over {1..6} {
    ROLLS: DICEd6
    CUTS: [count {7,8} in THORNSd8]
    output [getoutcome rolls ROLLS cuts CUTS] named "[DICE]d, thorns [THORNS]"
  }
}

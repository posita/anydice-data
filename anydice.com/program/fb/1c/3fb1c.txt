\
 Blades in the Dark - Threat roll
 5: Success + Success (Critical), Success + Partial
 3: Success + Failure, Partial + Partial
 0: Failure + Failure
\

function: blades ROLL:s {
 if 1@ROLL = 6 {
  if 2@ROLL = 6 {result: 5}
  if 2@ROLL = {4..5} {result: 5}
  if 2@ROLL = {1..3} {result: 3}
 }
 if 1@ROLL = {4..5} {
  if 2@ROLL = {4..5} {result: 3}
  if 2@ROLL = {1..3} {result: 0}
 }
 result: 0
}

loop P over {2..7} {
 output [blades Pd6] named "[P]d"
}

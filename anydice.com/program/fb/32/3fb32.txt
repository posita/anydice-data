\ Latskap: vil skrive [e d6] i stedet for [explode d6] \
function: e DIE:d { result: [explode DIE] }

\ TOHIT: bom 0, treff 1, og raise 2 \
function: tohit DEFENSE with DIE:n {
  if DIE >= DEFENSE+4 { result: 2 } 
  if DIE >= DEFENSE { result: 1 }
  result: 0
}

\ Enkel funksjon som gir antall WOUNDS for DMG vs TN. -1 bom, 0 shaken, 1 ett wound, osv. \
function: wound TN with DMG:n {
  if DMG - TN < 0 { result: -1 }
  if ((DMG - TN) / 4) >= MAXWOUNDS { result: MAXWOUNDS }
  result: (DMG-TN)/4
}

\ Wound-funksjon som forholder seg til raise pÃ¥ angrepet \
function: wound TN with DMG:d and hits ATK:n  {
  if ATK = 0 { result: -1 }
  if ATK = 1 { result: [wound TN with DMG] }
  if ATK = 2 { result: [wound TN with DMG+RAISE] }
  result: -9 \ dette skal aldri skje \
}

\ wound-funksjon, tar inn RAISE-die \
function: wound TN with DMG:d and hits ATK:n raise R:d { 
  if ATK = 0 { result: -1 }
  if ATK = 1 { result: [wound TN with DMG] }
  if ATK = 2 { result: [wound TN with DMG+R] }
  result: -9      \ dette skal aldri skje \
}

\ Enkel angrepsfunksjon. Trenger at global DEFENSE, TOUGHNESS og RAISE er definert \
function: attack with SKILL:n and damage DMG:d {
  HITS: [tohit DEFENSE with SKILL] 
  result: [wound TOUGHNESS with DMG and hits HITS]
}


\ Angrepsfunksjon som ikke trenger DEFENSE, TOUGHNESS og RAISE globalt definert \
function: attack DEF with DIE:n raise R:d and damage TN with DMG:d {
  HITS: [tohit DEF with DIE] 
  result: [wound TN with DMG and hits HITS raise R]
}

\ lag et korrekt sammensatt resultat av flere piler/skudd/angrep \
\ vi har -1 ingen effekt, 0 shaken, og positive tall som antall wounds \
\ sett MAXWOUNDS:4 for Wild Cards, og :1 for basic extras \
function: addwounds DA:n and DB:n {
  if DB < 0 { result: DA }
  if DA < 0 { result: DB }
  \ DA og DB er begge nÃ¥ 0 eller positiv \
  \ nÃ¥ vet vi at vi har treff pÃ¥ begge. Komplikado: 2x shaken = 1 wound \
  if DB = 0 {
    if DA = 0 { result: 1 }
    result: DA
  }
  result: DA + DB
}

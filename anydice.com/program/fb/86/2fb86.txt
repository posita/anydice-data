function: gwfgs A:n B:n {
  if A<3 & B<3 {result: 2d6}
  if A<3 {result: B+1d6}
  if B<3 {result: A+1d6}
  result: A+B
}

function: turn A:n B:n C{
  R:0
  if A>C {R: R+[gwfgs 1d6 1d6]+3 }
  if B>C {R: R+[gwfgs 1d6 1d6]+3 }
  result:R
}

function: ew A:n {
  if A<3 { A: 1d4}
  result A
}

output [ew 1d4]

output [turn 1d20+7 1d20+7 17]

output [gwfgs 1d6 1d6]+3+[gwfgs 1d6 1d6]+3

\**** Damage calculator dual wield ****\

function: attackroll ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: ROLL + ATTACKBONUS - DEF + (DMGBONUS *2) + (WPNDAMAGEA*2) + WPNDAMAGEB} 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS - 2 >= DEF { result: ROLL + ATTACKBONUS - DEF + (DMGBONUS *2) + WPNDAMAGEA + WPNDAMAGEB}
 if ROLL + ATTACKBONUS >= DEF { result: ROLL + ATTACKBONUS - DEF + DMGBONUS + WPNDAMAGEA }
 result: 0
}

\looped against enemy DEF\
ENEMYDEF: d{8}

ATTACKBONUS: 2
DMGBONUS: 2
CRITROLL: 12
ATTACKTIMES: 1
WPNDAMAGEA: 4
WPNDAMAGEB: 4
output ATTACKTIMESd[attackroll 2d6 vs ENEMYDEF] named "lightsword +4"
output 2d6

DRAGON: 1d6
DICEA: 1d6
DICEB: 1d6

MODIFIER: 7

TN: 18

function: get dragon D:n after roll DA:n and DB:n with mod M:n vs tn T:n {
  if (D+DA+DB+M >= T & (DA = DB | DB = D | D = DA)) { result: D }
  result: 0
}

function: calc X:n ignore zero {
  if X {result: X}
  result: d{}
}

function: calc if X:n not zero {
  if X {result: 1}
  result: 0
}

function: get success chance after roll D:n DA:n DB:n with mod M:n vs tn T:n {
  result: (D+DA+DB+M >= T)
}

output [calc [get dragon DRAGON after roll DICEA and DICEB with mod MODIFIER vs tn TN] ignore zero] named "stunt points"
output [get success chance after roll DRAGON DICEA DICEB with mod MODIFIER vs tn TN] named "hit chance"
output [calc if [get dragon DRAGON after roll DICEA and DICEB with mod MODIFIER vs tn TN] not zero] named "stunt chance"

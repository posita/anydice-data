function: roll ROLL:s {
  \ length of the ROLL sequence \
  N: #ROLL

  \ subtract the last die in sequence (lowest) from the first (highest) \
  result: 1@ROLL - N@ROLL
}

function: casualties ROLL:s {
  \ length of the ROLL sequence \
  N: #ROLL

  \ Count the zero (skull) faces rolled \
  result: [count 0 in 1@ROLL] + [count 0 in N@ROLL]
}

V: d{0,2,3,4,5,6}
R: 3dV
output [roll R] named "Expected difference of 3dV highest and lowest dice"
output [casualties R] named "Number of skulls rolled with 3dV"

R: 2dV
output [roll R] named "Expected difference of 2dV highest and lowest dice"
output [casualties R] named "Number of skulls rolled with 2dV"

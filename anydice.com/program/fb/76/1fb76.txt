function:  ROLL:n crit on TARGET:n roll ROLL {
    if ROLL >= TARGET {
        result: ROLL
    }
    result: ROLL
}

function:  ROLL:n crit on TARGET:n add MOD {
    if ROLL >= TARGET {
        result: ROLL + MOD
    }
    result: ROLL
}

function: ROLL crit RESULT {
    result: [ROLL crit on [maximum of ROLL] roll RESULT]
}

function: ROLL crit add MOD {
    result: [ROLL crit on [maximum of ROLL] add MOD]
}

function: A vs B {
    result: [highest of A - B and 0]
}

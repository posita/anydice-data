\
 Blades in the Dark
 0: Things go bad.
 1: Success, but with undesired consequences.
 2: Clean success.
\

function: blades ROLL:s {
 if 1@ROLL = 6 {
  result: 2
 }
 result: 1@ROLL >= 4
}

output [blades 0d6] named "."
output [blades 2@2d6] named "0d"
output [blades 1d6] named "1d"
output [blades 2d6] named "2d"
output [blades 3d6] named "3d"
output [blades 4d6] named "4d"
output [blades 5d6] named "5d"
output [blades 6d6] named "6d"

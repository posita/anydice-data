function: X:n toattack AC:n againstac M:n weaponattackmod B:n weaponcritsrange{ 
\returns one on a hit, otherwise zero \

NUM:0 

if X= 1 {NUM: 0} \sad crit\

if X >=B { \crit chance\
    if X+M >=AC { NUM: 1} \crit confirmed\
    else if X=20 { NUM:1}    \crit no confirm still hits on a 20\ 
}

else {NUM:0 } \miss\

result: NUM

}

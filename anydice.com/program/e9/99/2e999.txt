
function: N:n mod M:n {
  D: N / M
  result: N - (D * M)
}

SIX: 1
FIV: 20
FOU: 400
THR: 8000
TWO: 160000
ONE: 24000000
DIE: {ONE, TWO, THR, FOU, FIV, SIX}

function: count successes in ROLL:n {
  ONES: [(ROLL / ONE) mod 10] / 1
  TWOS: [(ROLL / TWO) mod 10] / 2 * 2
  THRS: [(ROLL / THR) mod 10] / 3 * 3
  FOUS: [(ROLL / FOU) mod 10] / 4 * 4
  FIVS: [(ROLL / FIV) mod 10] / 5 * 5
  SIXS: [(ROLL / SIX) mod 10] / 6 * 6
  result: ONES + TWOS + THRS + FOUS + FIVS + SIXS
}

MIN: 1
MAX: 6

loop N over {MIN..MAX} {
  output [count successes in NdDIE] named "[N]d6"
}

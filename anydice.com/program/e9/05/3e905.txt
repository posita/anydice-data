MAX_DEPTH: 1

function: check without crit {
 result: 3d6
}

function: check with crit A:n B:n C:n DEPTH{
if A=B & B=C & DEPTH < MAX_DEPTH{
  result: [check with crit 1d6 1d6 1d6 DEPTH+1] + A + B + C
 }
else {
  result: A + B + C
 }
}

function: check with critadvantage A:n B:n C:n D:n DEPTH {
COL: [sort {A, B, C, D}]
if A=B&B=C {
 result: [check with crit 1d6 1d6 1d6 DEPTH+1] + A + B + C
} else if B=C&C=D {
 result: [check with crit 1d6 1d6 1d6 DEPTH+1] + B + C + D
} else if A=C&C=D {
 result: [check with crit 1d6 1d6 1d6 DEPTH+1] + A + C + D
} else if A=B&B=D {
 result: [check with crit 1d6 1d6 1d6 DEPTH+1] + A + B + D
} else {
 result: 1 @ COL + 2 @ COL + 3 @ COL
}

}


output [check with critadvantage 1d6 1d6 1d6 1d6 0]
output [check with crit 1d6 1d6 1d6 0]

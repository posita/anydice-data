HITS: 0

function: remove negative X:n {
    if (X) = -3{
        result: -1
    }
    if (X) < 0{
        result: 0
    }
    if (X) = 3{
        result: 2
    }
    result: 1
}

function: success with score of X:n and difficulty Y:n {
  loop N over {1..X} {
     HITS: HITS + [remove negative 1d6 - 3]
  }
  result: HITS >= Y
}

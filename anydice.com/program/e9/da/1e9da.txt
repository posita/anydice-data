function: ties DIEA:s v DIEB:s {
  ONESA: 1@{1} = DIEA
  TWOSA: 1@{2} = DIEA
  TOTA: ONESA + TWOSA
  ONESB: 1@{1} = DIEB
  TWOSB: 1@{2} = DIEB
  TOTB: ONESB + TWOSB
  result: TOTA - TOTB
}

function: plus DIEA:s v DIEB:s {
  ONESA: 1@{1} = DIEA
  TWOSA: 1@{2} = DIEA
  TOTA: ONESA + TWOSA
  ONESB: 1@{1} = DIEB
  TWOSB: 1@{2} = DIEB
  TOTB: ONESB + TWOSB
  if TOTA - TOTB = 0 {
    if TWOSA > TWOSB {
      result: 1
    }
    else {
      if TWOSA = TWOSB {
        result: 0
      }
      else {
        result: -1
      }
    }
  }
  else {
    result: TOTA - TOTB
  }
}

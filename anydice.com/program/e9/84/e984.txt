function: highest of X:s and Y:s and Z:s with RANK:n {
 SUM: 0
 if (X > Y) { 
  SUM: SUM + X 
  if (Y > Z) {
   result: (SUM + Y + RANK)
  } else {
   result: (SUM + Z + RANK)
 }} else { 
  SUM: SUM + Y
  if (X > Z) {
   result: SUM + X + RANK
  } else {
   result: SUM + Z + RANK
  }
 }
}


loop A over {2..6}{
 loop B over {A..6}{
  loop C over {B..6}{
   output [highest of 1d(2*A) and 1d(2*B) and 1d(2*C) with 0] named "highest of 1d2*[A], 1d2*[B], 1d2*[C]"
}}}

  MAX: [maximum of D]
  result: [unfailing D helper]
}

function: unfailing D:n helper {
  if D = 1 {result: MAX + [unfailing 1dMAX]}
  if D = MAX {result: MAX + [unfailing 1dMAX]}
  result: D
}

\
Kult Die Calculator
-----------------------------------------------------
A result of "1" means "Success"
A result of "0" means "Success with Complications"
A result of "-1" means "Failure"

Input Values:
MOD: Anything added to the roll to modify it.
\

\Input Values\
MOD: 21

\PROGRAM STARTS HERE\

function: cyberdie VAL:n {
result: [cyberdiehelp 1d10 VAL]
}

function: cyberdiehelp ROLL:s VAL:n {
if ROLL=10 {result: 1d10+ROLL+VAL}
else if ROLL=1 {result: VAL+ROLL-1d10}
else {result: ROLL+VAL}
}

output [cyberdie MOD] named "Roll Result"

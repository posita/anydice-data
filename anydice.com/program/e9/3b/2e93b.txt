function: ATTACK attack vs AC {
 result: ATTACK >= AC
} 

function: true or none SUCCESS DAMAGE {
 result: SUCCESSdDAMAGE
}

function: flurry AC AA DA AB DB {
 result: [true or none [AA attack vs AC] DA] + 2d[true or none [AB attack vs AC] DB]
}

function: no flurry AC AA DA AB DB {
 result: [true or none [AA attack vs AC] DA] + [true or none [AB attack vs AC] DB]
}

MACE_DAMAGE:1d6+6
MACE_ATTACK:1d20+8
GAUNTLET_ATTACK:1d20+9
GAUNTLET_DAMAGE:1d4+7
UNARMED_ATTACK:1d20+7
UNARMED_DAMAGE:1d4+5




loop AC over {15..20} {
 output [no flurry AC MACE_ATTACK MACE_DAMAGE UNARMED_ATTACK UNARMED_DAMAGE] named "Mace vs AC [AC] no flurry"
 output [no flurry AC GAUNTLET_ATTACK GAUNTLET_DAMAGE GAUNTLET_ATTACK GAUNTLET_DAMAGE] named "Gauntlets vs AC [AC] no flurry"
 output [flurry AC MACE_ATTACK MACE_DAMAGE UNARMED_ATTACK UNARMED_DAMAGE] named "Mace vs AC [AC] flurry"
 output [flurry AC GAUNTLET_ATTACK GAUNTLET_DAMAGE GAUNTLET_ATTACK GAUNTLET_DAMAGE] named "Gauntlets vs AC [AC] flurry"
}


loop AC over {15..20} {
 output [no flurry AC MACE_ATTACK MACE_DAMAGE UNARMED_ATTACK UNARMED_DAMAGE] > [no flurry AC GAUNTLET_ATTACK GAUNTLET_DAMAGE GAUNTLET_ATTACK GAUNTLET_DAMAGE] named "mace > gauntlet vs AC [AC] no flurry"
 output [flurry AC MACE_ATTACK MACE_DAMAGE UNARMED_ATTACK UNARMED_DAMAGE] > [flurry AC GAUNTLET_ATTACK GAUNTLET_DAMAGE GAUNTLET_ATTACK GAUNTLET_DAMAGE] named "mace > gauntlet vs AC [AC] flurry"
}

function: D:n plus B:n hit AC:n{
  if D>AC{
    if D=20 { result: 4d8+8 }
    result: 2d8+4
  }
  else{ result: 0 }
}

function: D:n plus B:n hitt AC:n{
  if D>AC{
    if D=20 { result: 6d8+8 }
    result: 3d8+4
  }
  else{ result: 0 }
}

loop N over {10..20}{
  A: [1d20 plus 17 hit N] + [1d20 plus 17-5 hit N] + [1d20 plus 17-10 hit N]
  B: [[highest 2 of 2d20] plus 17 hit N] + [1d20 plus 17-5 hit N]
  C: [[highest 2 of 2d20] plus 17 hitt N]
  output A named "AC [N]"
}

function: roll ROLL:n plus MOD:n vs AC:n crit on CRITS:s {
    if ROLL = 1 | ROLL+MOD < AC {
        result: 0
    }
    if [CRITS contains ROLL] | ROLL+MOD >= AC {
        result: 1
    }
}

function: roll ROLL:n plus MOD:n vs DC:n {
    if ROLL+MOD >= DC {
        result: 1
    }
    else {
        result: 0
    }
}

function: save MOD:n vs DC:n {
    result: [roll 1d20 plus MOD vs DC]
}

function: miss ROLL:n on RANGE:s from DIE:n {
    if [RANGE contains DIE] {
        result: 0
    }
    else {
        result: ROLL
    }
} 

function: negate ROLL:n on RANGE:s from DIE:n {
    if [RANGE contains DIE] {
        result: 1
    }
    else {
        result: ROLL
    }
}

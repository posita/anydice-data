\ATTACK FUNCTIONS\
function: attack ROLL:n with mod MOD:n and prof PROF:n vs AC:n plus DR:n for DMG:d normal crit on RANGE:n {
  if ROLL = 1 {result: 0}
  if ROLL >= RANGE {result: DMG + [reroll DMG] + MOD - DR}
  if ROLL + MOD + PROF >= AC {result: DMG + MOD - DR}
  else {result: 0}
}
function: attack ROLL:n with mod MOD:n and prof PROF:n vs AC:n plus DR:n for DMG:d threat on RANGE:n {
  if ROLL = 1 {result: 0}
  if ROLL >= RANGE {result: DMG + [reroll DMG] + MOD - DR + [threat helper 1d20 3 3 vs 15 for DMG crit RANGE]}
  if ROLL + MOD + PROF >= AC {result: DMG + MOD - DR}
  else {result: 0}
}
function: threat helper ROLL:n MOD:n PROF:n vs AC:n for DMG:d crit RANGE:n {
  if ROLL = 1 {result: 0}
  if ROLL + MOD + PROF >= AC {result: DMG}
  else {result: 0}
}
function: attack ROLL:n with mod MOD:n and prof PROF:n vs AC:n plus DR:n for DMG:d crit on RANGE:n bonus BONUSDMG:n {
  if ROLL = 1 {result: 0}
  if ROLL >= RANGE {result: [highest of DMG+[maximum of DMG]+BONUSDMG+MOD-DR and 0]}
  if ROLL + MOD + PROF >= AC {result: [highest of DMG+MOD-DR and 0]}
  else {result: 0}
}

  if ROLL = 1 {result: 0}
  if ROLL = 20 {result: [highest of DMG+[maximum of DMG]+MOD+[attack combo ADV with mod MOD and prof PROF vs AC plus DR for DMG crit on RANGE] and 0]}
  if ROLL + MOD + PROF >= AC {result: [highest of DMG+MOD+[attack combo 1d20 with mod MOD and prof PROF vs AC plus DR for DMG crit on RANGE] and 0]}
  else {result: 0}
}\

  else {result: 0}
}\
function: attack ROLL:n str STR:n dex DEX:n prof PROF:n vs AC:n weight WT:n plus DR:n for DMG:d slashing crit on RANGE:n {
  ATTACK: ROLL+STR+DEX+PROF
  DAMAGE: DMG+STR+DEX-DR-WT*WT
  TOTDMG: 0

  if ROLL = 1 {result: 0}
  if ROLL >= RANGE {TOTDMG: [highest of DAMAGE+[maximum of DMG] and 0]}

  if ATTACK >= AC+WT {TOTDMG: TOTDMG+DAMAGE}
  if ATTACK >= AC+WT*WT {TOTDMG: TOTDMG+DEX+PROF}
  
  result: [highest of TOTDMG and 0]
}

function: ac with armour bonus ARM:n of weight WT:n str STR:n dex DEX:n additional ADD:n with HAND:n handed F:n finesse WEAP:d and SHIELD:n shield prof PROF:n {
  AC: 5+ARM+WT+ADD+SHIELD
  WPN: HAND*([maximum of WEAP]-4)/2
    if WPN > 0 {AC: AC + PROF + WPN + !F*STR + F*DEX}
  DIF: STR-WT
    if DIF < -1 {AC: AC - DIF}
    if DIF <  0 {AC: AC + DEX/2}
    if DIF =  0 {AC: AC + DEX}
    if DIF >  0 {AC: AC + DEX + [lowest of DEX and DIF]}
  result: AC
}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d 
{
    if [REROLL contains ROLL] { result: REPLACEMENT }
    result: ROLL
}
function: reroll REROLL:s on DICE:d { result: [ reroll REROLL on DICE to DICE ] }

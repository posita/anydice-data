function: R:n or W:n {
 if R < W { result: W}
 
 result: R
}

NORMAL: 1d20
ADVANTAGE: [highest 1 of 2d20]
DISADVANTAGE: [lowest 1 of 2d20]

WILD: 1d10

WILDNORM: [NORMAL or WILD]
WILDADV: [ADVANTAGE or WILD]
WILDDIS: [DISADVANTAGE or WILD]

PENALTY: 4

TARGET: 15
BONUS: 7

function: R:n attack {
 result: R + BONUS
}

function: R:n attack with wound {
 result: [R attack] - PENALTY
}

PERCENTONLY: 0

if PERCENTONLY=1 {
 output [NORMAL attack] >= TARGET named "Normal +[BONUS] chance to hit AC [TARGET]"
 output [WILDNORM attack] >= TARGET named "Normal +[BONUS] w/ wild [WILD] chance to hit AC [TARGET]"

 output [NORMAL attack with wound] >= TARGET named "Normal +[BONUS] chance to hit AC [TARGET], -[PENALTY] wound penalty"
 output [WILDNORM attack with wound] >= TARGET named "Normal +[BONUS] w/ wild [WILD] chance to hit AC [TARGET], -[PENALTY] wound penalty"

 output [ADVANTAGE attack] >= TARGET named "Advantage +[BONUS] chance to hit AC [TARGET]"
 output [WILDADV attack] >= TARGET named "Advantage +[BONUS] w/ wild [WILD] chance to hit AC [TARGET]"

 output [ADVANTAGE attack with wound] >= TARGET named "Advantage +[BONUS] chance to hit AC [TARGET], -[PENALTY] wound penalty"
 output [WILDADV attack with wound] >= TARGET named "Advantage +[BONUS] w/ wild [WILD] chance to hit AC [TARGET], -[PENALTY] wound penalty"

 output [DISADVANTAGE attack] >= TARGET named "Disadvantage +[BONUS] chance to hit AC [TARGET]"
 output [WILDDIS attack] >= TARGET named "Disadvantage +[BONUS] w/ wild [WILD] chance to hit AC [TARGET]"

 output [DISADVANTAGE attack with wound] >= TARGET named "Disadvantage +[BONUS] chance to hit AC [TARGET], -[PENALTY] wound penalty"
 output [WILDDIS attack with wound] >= TARGET named "Disadvantage +[BONUS] w/ wild [WILD] chance to hit AC [TARGET], -[PENALTY] wound penalty"
} 
else {
output [NORMAL attack] named "Normal attack +[BONUS]"
output [WILDNORM attack] named "Normal attack +[BONUS], with [WILD] wild die"

output [NORMAL attack with wound] named "Normal attack +[BONUS], -[PENALTY] wound penalty"
output [WILDNORM attack with wound] named "Normal attack +[BONUS], with [WILD] wild die, -[PENALTY] wound penalty"

output [ADVANTAGE attack] named "Advantage attack +[BONUS]"
output [WILDADV attack] named "Advantage attack +[BONUS], with [WILD] wild die"

output [ADVANTAGE attack with wound] named "Advantage attack +[BONUS], -[PENALTY] wound penalty"
output [WILDADV attack with wound] named "Advantage attack +[BONUS], with [WILD] wild die, -[PENALTY] wound penalty"

output [DISADVANTAGE attack] named "Disadvantage attack +[BONUS]"
output [WILDDIS attack] named "Disadvantage attack +[BONUS], with [WILD] wild die"

output [DISADVANTAGE attack with wound] named "Disadvantage attack +[BONUS], -[PENALTY] wound penalty"
output [WILDDIS attack with wound] named "Disadvantage attack +[BONUS], with [WILD] wild die, -[PENALTY] wound penalty"
}

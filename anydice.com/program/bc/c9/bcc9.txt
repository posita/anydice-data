\START PARAMETERS\

MOD: 4 \Relevant Ability Score Modifier\
AC: 16 \Target's AC\
TMOD: 1 \Target's Saving Throw Modifier\
SSL: 5 \Spell Slot Level\

BBLVL: 0 \Barbarian Level\
BDLVL: 0 \Bard Level\
CLLVL: 0 \Cleric Level\
DRLVL: 0 \Druid Level\
FILVL: 5 \Fighter Level\
MOLVL: 0 \Monk Level\
PALVL: 0 \Paladin Level\
RALVL: 0 \Ranger Level\
ROLVL: 0 \Rogue Level\
SOLVL: 0 \Sorcerer Level\
WALVL: 0 \Warlock level\
WILVL: 0 \Wizard level\

\Enter 1 for true, 0 for false\
CHAMP: 0 \Champion\
COV: 0 \College of Valor\
ADV: 0 \Advantage on attacks, disadvantage to target saves\
RAGE: 0 \Rage active\
BATT: 0 \Bonus Attack\
SA: 0 \Sneak Attack active\
BLESS: 0 \Bless active\

\END OF PARAMETERS\

LVL: BBLVL + BDLVL + CLLVL + DRLVL + FILVL + MOLVL + PALVL + RALVL + ROLVL + SOLVL + WALVL + WILVL

function: prof A{
if A<5 {result: 2}
if A<9 {result: 3}
if A<13 {result: 4}
if A<17 {result: 5}
result: 6
}

SATT: BBLVL>=5 | (BDLVL>=6 & COV) | FILVL>=5 | MOLVL>=5 | PALVL>=5 | RALVL>=5
TATT: FILVL>=11
FATT: FILVL=20

\Spell Save DC\
DC: 8 + [prof LVL] + MOD
function: bc A {
if LVL<9 {result: 0}
if LVL<13 {result: 1}
if LVL<17 {result: 2}
result: 3
}
if ADV = 1{
 \Normal Hit Rate\
 HIT: (d20 + MOD + [prof LVL] + (1d4)*BLESS >= AC) | (d20 + MOD + [prof LVL] + (1d4)*BLESS >= AC)
 \Barbarian Capstone Hit Rate\
 BHIT: d20 + MOD + [prof LVL] + 2*(BBLVL = 20) + (1d4)*BLESS >= AC | d20 + MOD + [prof LVL] + 2*(BBLVL = 20) + (1d4)*BLESS >= AC
 \Archery FS Hit Rate\
 ARC: d20 + MOD + [prof LVL] + 2 + (1d4)*BLESS >= AC | d20 + MOD + [prof LVL] + 2 + (1d4)*BLESS >= AC
 \Power Attack Hit Rate\

 \Barbarian Power Attack Hit Rate\
 BPHIT: d20 + MOD + [prof LVL] - 5 + 2*(BBLVL = 20) + (1d4*BLESS) >= AC | d20 + MOD + [prof LVL] - 5 + 2*(BBLVL = 20) + (1d4*BLESS) >= AC
 \Power Attack Archery Hit Rate\
 PARC: d20 + MOD + [prof LVL] - 3 + (1d4)*BLESS >= AC | d20 + MOD + [prof LVL] - 3 + (1d4)*BLESS >= AC
 \Spell Save Hit Rate\
 MHIT: d20 + TMOD < DC & d20 + TMOD < DC
 \Normal Crits\
 X: (d20 = 20 | d20 = 20)*(SA + 1) + 1
 XGS: (d20 = 20 | d20 = 20)*(SA + 1) + 2
 \Barbarian Crits\
 BX: (d20 = 20 | d20 = 20)*(SA + [bc LVL] + 1) + 1
 \Barbarian Great Sword Crits\
 BXGS: (d20 = 20 | d20 = 20)*(SA + [bc LVL] + 1) + 2
 if LVL < 3 {
  \Champion Crits\
  CX: (d20 >= 20 | d20 >= 20)*(SA + 1) + 1
  \Champion Great Sword Crits\
  CXGS: (d20 >= 20 | d20 >= 20)*(SA + 1) + 2
 }
 else {
  if LVL < 15 {
  \Champion Crits\
   CX: (d20 >= 19 | d20 >= 19)*(SA + 1) + 1
  \Champion Great Sword Crits\
   CXGS: (d20 >= 19 | d20 >= 19)*(SA + 1) + 2
  }
  else {
  \Champion Crits\
   CX: (d20 >= 18 | d20 >= 18)*(SA + 1) + 1
  \Champion Great Sword Crits\
   CXGS: (d20 >= 18 | d20 >= 18)*(SA + 1) + 2
  }
 }
}
else if ADV = 0{
 HIT: d20 + MOD + [prof LVL] + (1d4)*BLESS >= AC
 BHIT: d20 + MOD + [prof LVL] + 2*(BBLVL = 20) + (1d4)*BLESS >= AC
 ARC: d20 + MOD + [prof LVL] + 2 + (1d4)*BLESS >= AC
 PHIT: d20 + MOD + [prof LVL] - 5 + (1d4)*BLESS >= AC
 BPHIT: d20 + MOD + [prof LVL] - 5 + 2*(BBLVL = 20) + (1d4)*BLESS >= AC
 PARC: d20 + MOD + [prof LVL] - 3 + (1d4)*BLESS >= AC
 X: (d20 = 20)*(SA + 1) + 1
 XGS: (d20 = 20)*(SA + 1) + 2
 BX: (d20 = 20)*(SA + [bc LVL] + 1) + 1
 BXGS: (d20 = 20)*(SA + [bc LVL] + 1) + 2
 MHIT: d20 + TMOD < DC
 if LVL < 3 {
  CX: (d20 >= 20)*(SA + 1) + 1
  CXGS: (d20 >= 20)*(SA + 1) + 2
 }
 else {
  if LVL < 15 {
   CX: (d20 >= 19)*(SA + 1) + 1
   CXGS: (d20 >= 19)*(SA + 1) + 2
  }
  else {
   CX: (d20 >= 18)*(SA + 1) + 1
   CXGS: (d20 >= 18)*(SA + 1) + 2
  }
 }
}
else if ADV = -1{
 HIT: (d20 + MOD + [prof LVL] + (1d4)*BLESS >= AC) & (d20 + MOD + [prof LVL] + (1d4)*BLESS >= AC)
 BHIT: d20 + MOD + [prof LVL] + 2*(BBLVL = 20) + (1d4)*BLESS >= AC & d20 + MOD + [prof LVL] + 2*(BBLVL = 20) + (1d4)*BLESS >= AC
 ARC: d20 + MOD + [prof LVL] + 2 + (1d4)*BLESS >= AC & d20 + MOD + [prof LVL] + 2 + (1d4)*BLESS >= AC
 PHIT: d20 + MOD + [prof LVL] - 5 + (1d4)*BLESS >= AC & d20 + MOD + [prof LVL] - 5 + (1d4)*BLESS >= AC
 BPHIT: d20 + MOD + [prof LVL] - 5 + 2*(BBLVL = 20) + (1d4)*BLESS >= AC & d20 + MOD + [prof LVL] - 5 + 2*(BBLVL = 20) + (1d4)*BLESS >= AC
 PARC: d20 + MOD + [prof LVL] - 3 + (1d4)*BLESS >= AC & d20 + MOD + [prof LVL] - 3 + (1d4)*BLESS >= AC
 X: (d20 = 20 & d20 = 20)*(SA + 1) + 1
 XGS: (d20 = 20 & d20 = 20)*(SA + 1) + 2
 BX: (d20 = 20 & d20 = 20)*(SA + [bc LVL] + 1) + 1
 BXGS: (d20 = 20 & d20 = 20)*(SA + [bc LVL] + 1) + 2
 MHIT: d20 + TMOD < DC | d20 + TMOD < DC
 if LVL < 3 {
  CX: (d20 >= 20 & d20 >= 20)*(SA + 1) + 1
  CXGS: (d20 >= 20 & d20 >= 20)*(SA + 1) + 2
 }
 else {
  if LVL < 15 {
   CX: (d20 >= 19 & d20 >= 19)*(SA + 1) + 1
   CXGS: (d20 >= 19 & d20 >= 19)*(SA + 1) + 2
  }
  else {
   CX: (d20 >= 18 & d20 >= 18)*(SA + 1) + 1
   CXGS: (d20 >= 18 & d20 >= 18)*(SA + 1) + 2
  }
 }
}

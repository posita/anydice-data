\ The function that'll evaluate your queries \ 
function: eval THIS_ROLL:n {
  result: (DCS <= THIS_ROLL)@POINTS
}

\ #=========#=========#=========#=========#=========#=========#	\

\ Premade dice for readability & convenience			\
EA_ADV:	[highest 1 of 3d20]	\ Elven Accuracy Advantage	\
ADV:	[highest 1 of 2d20]	\ Advantage			\
DIS:	[lowest  1 of 2d20]	\ Disadvantage			\

\ #=========#=========#=========#=========#=========#=========#	\

\ The DCs and their respective points \
DCS:	{10, 20, 30, 40,  50 }
POINTS:	{20, 30, 50, 100, 150}


\ Space for placing your rolls to evaluate. The following format is reccomended:
output	[eval	<dice roll>	] named "<name>"			       \

output	[eval	ADV + 8	+ d4	] named "Example: Biri's Nature"
output	[eval	d20 + 11	] named "Example: Biri's Scribe's Tools"

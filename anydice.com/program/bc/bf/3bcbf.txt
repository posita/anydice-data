function: count VALUES:s in SEQUENCE:s {
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
 }
 result: COUNT
}

function: max A:n B:n {
 if A > B { result: A }
 result: B
}

function: agency roll with BURNOUT:n burnout {
 result: [max ([count 3 in 6d4]-BURNOUT) 0]
}


loop B over {0..6} {
 output [agency roll with B burnout] named "[B] Burnout"
}

 if U < D {
  result: (U-1)*10+D-1
 }
 else {
  result:  (D-1)*10+U-1
 }
}

 if A < B {
  D: A
  U: B
 }
 else {
  U: A
  D: B
 }
 if C < D {
  D: C
 }
 else {
  if C < U {
   U: C
  }
 }
 result:  (D-1)*10+U-1
}

  result: A-B

   result: A

 }
}

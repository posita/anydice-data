\ rolls a dice from 0 to 2 and rerolls if 0 \
function: ROLL:n reroll zero {
  if ROLL = 0 { result: 1d{0..2} }
  else        { result: ROLL }
}

\ rolls X amount of the above dice and adds together \
function: pool SIZE:n {
  SUM: 0
  loop N over {1..SIZE} {
    SUM: SUM + [1d{0..2} reroll zero]
  }
  result: SUM
}

\ output \
output [pool 5]

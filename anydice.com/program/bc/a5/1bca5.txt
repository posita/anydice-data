function: hit {
 MHIT: [r 15 + (DEX / 2) + d{0..FSKILL} + d{0..WSKILL} + ACCURACY + SLAYING + ENCHANTMENT]
 if MONSTER {
  MHIT: [r 18 + (HD / 2) + ((!!FIGHTER + 1) * HD) + 1]
 }
 EV_ROLL: [2d{1..(2 * EV)} div 2]
 result: ((MHIT > EV_ROLL) + d{-1, 0:38, 1}) > 0
}

function: atk {
 if MONSTER { result: 0 }
 result: [dmg [hit]]
}



function: dmg HIT:n {
 if !HIT {result: 0}
 B: BASE_DAMAGE
 if UNARMED { B: BASE_DAMAGE + WSKILL }
 S: [dmg [B str mod] helper]
 F: [S fskill mod]
 W: [F wskill mod]
 result: [highest of W - d{0..AC} + d{0..(SLAYING + ENCHANTMENT)} and 0]
}
function: dmg N:n helper { result: d{1..N} }

function: N:n str mod {
 R: 39 + (d{0..STR-9} * 2)
 result: [N * R div 39]
}

function: N:n wskill mod {
 if UNARMED { result: N }
 R: 25 + d{0..WSKILL}
 result: [N * R div 25]
}

function: N:n fskill mod {
 M: 30
 if UNARMED { M: 40 }
 R: M + d{0..FSKILL}
 result: [N * R div M]
}

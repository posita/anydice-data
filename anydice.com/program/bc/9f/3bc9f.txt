function: attack ROLL:n ATTACK:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= 20 { result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE + 10 { result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT}
 result: MISS
}

function: crossbowattack ROLL:n ATTACK:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= 20 { result: CROSSBOWCRIT}
 if ROLL + ATTACK >= DEFENSE + 10 { result: CROSSBOWCRIT}
 if ROLL + ATTACK >= DEFENSE { result: CROSSBOWHIT}
 result: MISS
}


MISS: 0
HIT: 1d6
CRITICALHIT: (1d6*2)+1d10
CROSSBOWHIT: (1d10)+1
CROSSBOWCRIT: ((1d10)+1)*2

output [attack d20 10 vs 15] + [attack d20 5 vs 15]
output [attack d20 10 vs 13] + [attack d20 5 vs 13]

function: check A:n {
if A<DIF & A=DIE {result: [check A+1dDIE]}
result: A
}

function: demo A:n B:n C:n D:n {
RES:0
CR:0
if A>=DIF {RES:RES+1}
if B>=DIF {RES:RES+1}
if C>=DIF {RES:RES+1}
if D>=DIF {RES:RES+1}
if RES=0 {if 1@[sort {A,B,C,D}]<=DIF/2 {RES:-1}}
result:RES
}

loop DIF over {3..7}{
 loop DIE over {4,6,8,10,12}{
  output [demo [check 1dDIE] [check 1dDIE] [check 1dDIE] [check 1d0]] named "3d[DIE] vs [DIF]"
}}

function: dupes in DICE:s {
  D: 0
  loop X over {2..#DICE} {
    if ((X-1)@DICE = X@DICE) { D: D + 1}
  }
  result: D
}
function: roll ROLL:s vs TN {
 R:0
 A:0
 B:0
 loop I over {1..#ROLL} {
  if (I@ROLL <= TN) {
   R:(R+1)
   if ([count I@ROLL in ROLL]>1) {
    A:1
   }
  } else {
   R:(R-1)
   if ([count I@ROLL in ROLL]>1) {
    B:1
   }
  }
 }

 result: R + A - B
}


function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}

output
4d[attack d20 plus 6 + 5 vs d{15..20}
    hit  3d8 + 5 + 2
    miss 0
    crit 6d8 + 5 + 2 on 20
]

function: roll DICE:n
{
  result:DICE
}

function: opposed skill NUMBER:s dice SEQUENCE:s {
  SORTED: [sort SEQUENCE]
  R: 1@SORTED
  S: [count NUMBER in SEQUENCE]
  if S>0
  {
    S:S-1
  }
  result: R+S
}

function: attacker skill AS:s dice AD:s vs defender skill DS:s dice DD:s
{
  N:[opposed skill AS dice AD]-[opposed skill DS dice DD]
  if N<-1
  {
    result: -1
  } else
  {
    result: N
  }
}

function: unopposed skill NUMBERS:s dice SEQUENCE:s
{
  if [count NUMBERS in SEQUENCE]>0
  {
    result:1
  }else
  {
    result:0
  }
}

function: attacker skill AS:s dice AD:s damage AP:n vs defender skill DS:s dice DD:s armor DA:n
{
  N:[opposed skill AS dice AD]-[opposed skill DS dice DD]
  if N<0
  {
    result:0
  } else
  {
    if AP+N-DA=0
    {
      result: [roll 1d{0,1}]
    }else if AP+N-DA<0
    {
      result:0
    }else
    {
      result:AP+N-DA
    }
  }
}

function: attacker skill AS:s dice AD:s damage AP:n vs defender skill DS:s dice DD:s armor DA:n with agile
{
  N:[opposed skill AS dice AD]-[opposed skill DS dice DD]
  if N<=0
  {
    result:0
  } else
  {
    if AP+N-DA=0
    {
      result: [roll 1d{0,1}]
    }else if AP+N-DA<0
    {
      result:0
    }else
    {
      result:AP+N-DA
    }
  }
}

function: attacker skill AS:s dice AD:s damage AP:n fire AF:n vs defender skill DS:s dice DD:s armor DA:n
{
  N:[opposed skill AS dice AD]-[opposed skill DS dice DD]
  if N<0
  {
    result:0
  } else
  {
    if AP+N-DA=0
    {
      result: [roll 1d{0,1}]+[roll AFd{0,1}]
    }else if AP+N-DA<0
    {
      result:[roll AFd{0,1}]
    }else
    {
      result:AP+N-DA+[roll AFd{0,1}]
    }
  }
}

function: attacker skill AS:s dice AD:s damage AP:n ap APEN:n vs defender skill DS:s dice DD:s armor DA:n
{
  N:[opposed skill AS dice AD]-[opposed skill DS dice DD]
  if N<0
  {
    result:0
  } else
  {
    if N>AP+N-DA&APEN>AP+N-DA
    {
      if N<=APEN
      {
        result: N
      }else if N>APEN
      {
        result:APEN
      }
    } else
    {
      if AP+N-DA=0
      {
        result: [roll 1d{0,1}]
      }else if AP+N-DA<0
      {
        result:0
      }else
      {
        result:AP+N-DA
      }
    }
  }
}

DEFEAT: 1
FAIL: 2
BOTH: 8
SUCCESS: 14
VICTORY: 20

function: attack ROLL:n {
 if ROLL = 1 { result: DEFEAT}
 if ROLL = {2..7} {result: FAIL}
 if ROLL = {8..13} {result: BOTH}
 if ROLL = {14..19} {result: SUCCESS}
 if ROLL = 20 {result: VICTORY}
}

output [attack [lowest 1 of 2d20]] named "difficult"
output [attack d20] named "normal"
output [attack [highest 1 of 2d20]] named "easy"

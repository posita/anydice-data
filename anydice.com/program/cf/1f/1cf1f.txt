ATTACK_MODIFIER: 6
DAMAGE_MODIFIER: 5



ARMOR_CLASS: {16, 17, 18, 19, 20, 21}

loop X over ARMOR_CLASS
{
    DAMAGE_ONE: 0
    DAMAGE_TWO: 0
    DAMAGE_THREE: 0
    HIT_ONE: 1d20 + ATTACK_MODIFIER > X
    HIT_TWO: 1d20 + ATTACK_MODIFIER > X
    if (HIT_ONE) {
        DAMAGE_ONE: 1d6 + DAMAGE_MODIFIER
    }
    if (HIT_TWO) {
        DAMAGE_TWO: 1d6 + DAMAGE_MODIFIER
    }
    if (HIT_TWO * HIT_ONE){
        DAMAGE_THREE: 1d10
    }
    DAMAGE_TOTAL: DAMAGE_ONE + DAMAGE_TWO + DAMAGE_THREE
    output DAMAGE_TOTAL named "[X] AC"
}

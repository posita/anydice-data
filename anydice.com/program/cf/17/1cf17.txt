\ Hexstaff \

MISS: 0
HIT: 1
BGWF: {{1..2}, {3..10}:6}
SGWF: {{1..2}, {3..4}:3}
CRITICALHIT: 2
BIGATTACK: 2 + 2
SMALLATTACK: 2 + 2
SKELEATTACK: 4
BIGHITDAMAGE: 1d8 + 2
BIGCRITICALDAMAGE: 2d8 + 2
SMALLHITDAMAGE: 1d4 + 3
SMALLCRITICALDAMAGE: 2d4 + 3
SKELEHITDAMAGE: 1d10 + 2
SKELECRITICALDAMAGE: 2d10 + 2
MISSDAMAGE: 0
AC: 10

function:  ROLL:n vs DEFENSE:n ATTACK:n {
 if ROLL = 1 {result: MISS}
 if ROLL = 20 {result: CRITICALHIT}
 if ROLL + ATTACK>= DEFENSE {result: HIT}
 result: MISS
}

\ Damage functions \

function: damage for big ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: BIGCRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: BIGHITDAMAGE
 }
 result: MISSDAMAGE
}

function: damage for small ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: SMALLCRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: SMALLHITDAMAGE
 }
 result: MISSDAMAGE
}

function: damage for skele ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: SKELECRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: SKELEHITDAMAGE
 }
 result: MISSDAMAGE
}


\ Multi attacks \

function: bigmultiattack NUMBEROFATTACKS:n {
  TOTAL: 0
  loop N over {1..NUMBEROFATTACKS} {
    TOTAL: TOTAL + [damage for big [d20 vs AC BIGATTACK]]
  }
  result: TOTAL
}

function: bigmultiattackadvantage NUMBEROFATTACKS:n {
  TOTAL: 0
  loop N over {1..NUMBEROFATTACKS} {
    TOTAL: TOTAL + [damage for big [[highest 1 of 2d20] vs AC BIGATTACK]]
  }
  result: TOTAL
}

function: bigmultiattackdoubleadvantage NUMBEROFATTACKS:n {
  TOTAL: 0
  loop N over {1..NUMBEROFATTACKS} {
    TOTAL: TOTAL + [damage for big [[highest 1 of 3d20] vs AC BIGATTACK]]
  }
  result: TOTAL
}

function: reroll lowest N:n of ROLL:s under THRESH:n {
 R: [lowest of N and ROLL < THRESH] 
 result: {1..#ROLL - R}@ROLL + RdD
}

function: diff lowest N:n of ROLL:s under THRESH:n{
 R: [lowest of N and ROLL < THRESH] 
 result: ({1..#ROLL - R}@ROLL + RdD - ROLL)
}

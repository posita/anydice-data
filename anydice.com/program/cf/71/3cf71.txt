function: marvel X:n M:n Y:n plus B:n vs T:n {
  if M = 1 {
    if X + Y = 12 {
      result: 3
    }
    else if X + 6 + Y + B >= T {
      result: 2
    }
    else {
      result: 0
    }
  } else {
    if X + M + Y + B >= T {
      result: 1
    }
    else {
      result: -1
    }
  }
}

loop TN over {17..20} {
  loop BNS over {0..11} {
    output [marvel 1d6 1d6 1d6 plus BNS vs TN] named "+[BNS] vs [TN]"
  }
}

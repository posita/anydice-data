PROF: 4
MOD: 5
MAGIC: 2
AC: 17

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

LC    : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d12 + MOD + MAGIC          crit 1d12 on 20]
LC_PA : [attack [highest 1 of 1d20] plus MOD+     MAGIC vs AC for 1d12 + MOD + MAGIC + 2*PROF crit 1d12 on 20]
SD    : [attack [highest 1 of 1d20] plus MOD+PROF vs AC for 1d8+2d6 + PROF                  crit 1d8+2d6 on 20]


output 2dLC_PA+SD named "10th level Zenmit with Lance + Steel Defender attack with BA+Arcane Jolt"

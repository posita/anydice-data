function: attack ROLL_A:n and ROLL_B:n vs DEFENSE:n dealing DAMAGE:n {
 if ROLL_A = 1 & ROLL_B = 1 { result: 0 }
 if ROLL_A = ROLL_B & ROLL_A >= 6 { result: DAMAGE }
 if ROLL_A + ROLL_B >= DEFENSE { result: DAMAGE }
 result: 0
}

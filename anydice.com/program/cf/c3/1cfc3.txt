\ Master Summoner\

MISS: 0
HIT: 1
BGWF: {{1..2}, {3..10}:6}
SGWF: {{1..2}, {3..4}:3}
CRITICALHIT: 2
BIGATTACK: 2 + 2
SMALLATTACK: 2 + 2
SKELEATTACK: 4
MOUNTATTACK: 6
BIGHITDAMAGE: 1d8 + 2
BIGCRITICALDAMAGE: 2d8 + 2
SMALLHITDAMAGE: 1d4 + 3
SMALLCRITICALDAMAGE: 2d4 + 3
SKELEHITDAMAGE: 1d10 + 2
SKELECRITICALDAMAGE: 2d10 + 2
MOUNTBIGHITDAMAGE: 2d6 + 4
MOUNTBIGCRITICALDAMAGE: 4d6 + 4
MOUNTSMALLHITDAMAGE: 2d6 + 4
MOUNTSMALLCRITICALDAMAGE: 4d6 + 4 

SUMMONERSPELLSAVE: 13
PSHITDAMAGE: 1d12
PSMISSDAMAGE: 0
MISSDAMAGE: 0
MONSTERWISMOD: -2
AC: 10

function:  ROLL:n vs DEFENSE:n ATTACK:n {
 if ROLL = 1 {result: MISS}
 if ROLL = 20 {result: CRITICALHIT}
 if ROLL + ATTACK>= DEFENSE {result: HIT}
 result: MISS
}

function: poison spray ROLL:n {
 if ROLL + MONSTERWISMOD >= SUMMONERSPELLSAVE {result: PSMISSDAMAGE}
 result: PSHITDAMAGE
}

\ Damage functions \

function: damage for big ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: BIGCRITICALDAMAGE

\ Multi attacks \

function: bigmultiattack NUMBEROFATTACKS:n {
  TOTAL: 0
  loop N over {1..NUMBEROFATTACKS} {
    TOTAL: TOTAL + [damage for big [d20 vs AC BIGATTACK]]
  }
  result: TOTAL
}

function: bigmultiattackadvantage NUMBEROFATTACKS:n {
  TOTAL: 0
  loop N over {1..NUMBEROFATTACKS} {
    TOTAL: TOTAL + [damage for big [[highest 1 of 2d20] vs AC BIGATTACK]]
  }
  result: TOTAL
}

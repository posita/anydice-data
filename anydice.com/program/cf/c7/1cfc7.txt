function: X:n gbdamage{
if X > 9 {result: 4}
if X = 9 {result: 2}
if X = 8 {result: 2}
if X = 7 {result: 2}
if X = 6 {result: 2}
if X = 5 {result: 1}
if X = 4 {result: 1}
if X = 3 {result: 1}
if X = 2 {result: 1}
if X < 2 {result: 0}
}

loop N over {3..10} {
  output Nd[highest 1 of 2d[d10 gbdamage]] named "[N]d10, rerolled individually"
  output [highest 1 of 2d(Nd[d10 gbdamage])] named "[N]d10, rerolled collectively"
}

\
H&G  2d12 vs. 10-stat, roll HIGHER.
Critical: if 2 successes and at least one is a 12.
Stats range = 0-7 (avg=4)

 -1: Miss.
 0: Success, but with undesired consequences.
 +1: Clean success.
 +2: Critical success!
\

function: roll A:s and B:s vs S:s {
 R: -1
 if A > S {
   if B = 12 {
     R: 2
   } else {
     R: R+1
   }
 }
 if B > S {
   if R < 2 {
     if A = 12 {
       R: 2
     } else {
       R: R+1
     }
   }
 }
\
 if A = 12 {
   R: 2
 }
 if B = 12 {
   R: 2
 }
\
 result: R
}

loop P over {1..7} {
 output [roll 1d12 and 1d12 vs 10-P] named "[P]"
}

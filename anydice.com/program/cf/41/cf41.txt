POISON: 0d4
DEXBONUS: 4
WEAPONBONUS: 0
HITBONUS: DEXBONUS + WEAPONBONUS
DAMAGEBONUS: DEXBONUS+WEAPONBONUS+POISON
AC: 15

function: proficiency LEVEL {
result: ((LEVEL-1)/4) + 2
}

function: hitchelper DICE LEVEL {
result: ((DICEd20 + HITBONUS + [proficiency LEVEL]) > AC)
}

function: hitc LEVEL {
result: ((1d20 + HITBONUS + [proficiency LEVEL]) >= AC)
}

function: chitc LEVEL {
result: ((1@2d20 + HITBONUS + [proficiency LEVEL]) >= AC)
}

function: hit DICE HITS LEVEL {
result: ([hithelper DICE HITS] + DAMAGEBONUS) *[hitc LEVEL]
}

function: hithelper DICE HITS {
result: HITSd(DICE)
}

function: chit DICE HITS LEVEL {
result: (3d([hithelper DICE HITS]) + DAMAGEBONUS) * [chitc LEVEL]
}

function: sneak ATTACKS LEVEL {
result: 2d6 * [hitchelper ATTACKS LEVEL ]
}

function: csneak ATTACKS LEVEL {
result: 4d6 * [hitchelper ATTACKS*2 LEVEL]
}

function: supd FLEVEL LEVEL{
result: [supdhelper] * [hitc LEVEL]
}

function: supdhelper FLEVEL {
result:(4+((FLEVEL+1)/8))d(8+(FLEVEL/10*2))
}

function: csupd FLEVEL LEVEL{
result: 2d([supdhelper FLEVEL]) * [chitc LEVEL]
}

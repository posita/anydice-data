ATTACKER_AB: 5
ATTACKER_DAMAGE: 1d8+4

ATTACKS: 3
MAP: -5

TARGET_AC: 15


function: attack ROLL:n plus CHECK:n vs DC:n dealing DAMAGE:n {
DOS: 0
if ROLL + CHECK <= DC - 10 { DOS: -1 }
if ROLL + CHECK >= DC      { DOS: 1 }
if ROLL + CHECK >= DC + 10 { DOS: 2 }

if ROLL = 1 { DOS: DOS - 1 }
if ROLL = 20 { DOS: DOS + 1 }

if DOS < -1 { DOS: -1 }
if DOS > 2 { DOS: 2 }

DEALT: 0
if DOS > 0 { DEALT: DAMAGE * DOS }

result: DEALT
}

function: perform attacks {
DEALT: 0

if (ATTACKS >= 1) { DEALT: DEALT + [attack d20 plus ATTACKER_AB - (MAP * 0) vs TARGET_AC dealing ATTACKER_DAMAGE] }
if (ATTACKS >= 2) { DEALT: DEALT + [attack d20 plus ATTACKER_AB - (MAP * 1) vs TARGET_AC dealing ATTACKER_DAMAGE] }
if (ATTACKS >= 3) { DEALT: DEALT + [attack d20 plus ATTACKER_AB - (MAP * 2) vs TARGET_AC dealing ATTACKER_DAMAGE] }
if (ATTACKS >= 4) { DEALT: DEALT + [attack d20 plus ATTACKER_AB - (MAP * 2) vs TARGET_AC dealing ATTACKER_DAMAGE] }

result: DEALT
}

output [perform attacks]

function: pbta X:n {
   if (X <= 6) { result: -1 }
   if (X <= 9) { result: 0 }
   result: 1
}

function: alt X:n and Y:n for Z:n {
   if (X <= Z & Y <= Z) { result: 1 }
   if (X <= Z | Y <= Z) { result: 0 }
   result: -1
}

function: alt X:n for Y:n with Z:n {
   T: 20 - Y
   R: X + Z
   if (R >= T) { result: 1 }
   if (R <= 10) { result: -1 }
   result: 0
}

function: knave X:n {
   if (X >= 15) { result: 1 }
   result: -1
}

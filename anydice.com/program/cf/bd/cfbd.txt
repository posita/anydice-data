POISON: 1d4 \Basic: 1d4, Serpent Venom: 3d6, Purple Worm: 12d6\
ENLARGE: 1d4 \1d4; Spell/Potion; Actually weapon damage so critmult comes into play\
HEX: 0d6 \1d6; Earned at level 11\

DEXBONUS: 4  \ 5 Earned at level 9\
WEAPONBONUS: 1

AC:17


\CALCULATION STUFF BELOW, DO NOT MODIFY\
HITBONUS: DEXBONUS + WEAPONBONUS
DAMAGEBONUS: DEXBONUS+WEAPONBONUS+POISON+HEX
CRITMULT:3 \Half-orc trait\



function: proficiency LEVEL {
result: ((LEVEL-1)/4) + 2
}


function: hit DICE HITS LEVEL {
OUTPUT:0
loop N over {1..HITS}{
OUTPUT: OUTPUT + (DICE+DAMAGEBONUS+ENLARGE)*((1d20+HITBONUS+[proficiency LEVEL])>=AC)
}
result: OUTPUT
}

function: chit DICE HITS LEVEL {
OUTPUT:0
loop N over {1..HITS}{
OUTPUT: OUTPUT + (CRITMULTd(DICE)+DAMAGEBONUS+CRITMULTd(ENLARGE))*((1@2d20+HITBONUS+[proficiency LEVEL])>=AC)
}
result: OUTPUT
}


function: sneak ATTACKS LEVEL {
result: 2d6 * ((1@ATTACKSd20+HITBONUS+[proficiency LEVEL])>= AC) 
}

function: csneak ATTACKS LEVEL {
result: 4d6 * ((1@ATTACKSd20+HITBONUS+[proficiency LEVEL])>= AC) 
}

function: csupd ATTEMPTS FLEVEL LEVEL{
ATTACKS: ATTEMPTS
if ATTEMPTS > [supdice FLEVEL]{
ATTACKS: [supdice FLEVEL]
}

OUTPUT:0
loop N over {1..ATTACKS}{
result: OUTPUT + 1d[supdamage FLEVEL] *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)

}
result: OUTPUT
}

function: supdice FLEVEL {
result:(4+((FLEVEL+1)/8))
}

function: supdamage FLEVEL {
result:(8+(FLEVEL/10*2))
}

function: csupd ATTEMPTS FLEVEL LEVEL{
ATTACKS: ATTEMPTS
if ATTEMPTS > [supdice FLEVEL]{
ATTACKS: [supdice FLEVEL]
}

OUTPUT:0
loop N over {1..ATTACKS}{
result: OUTPUT + 2d(1d[supdamage FLEVEL]) *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)

}
result: OUTPUT
}

function: gold PERF:n mod B lvl L:n {
 if (PERF = 1) {result: 0}
 if (PERF +B <= 10) {result: (1d4)*L}
 if (PERF +B <= 14) {result: (2d4)*L}
 if (PERF +B <= 20) {result: (2d6)*L}
 if (PERF +B <= 25) {result: (4d6)*L}
 else {result: (4d8)*L}
}

function: brew POTION:n mod MOD vs DC or BONUS {
 if POTION = 1 {result: 0}
 if POTION + MOD >= BONUS {result: 2}
 if POTION + MOD >= DC {result: 1}
 else {result: 0}
}

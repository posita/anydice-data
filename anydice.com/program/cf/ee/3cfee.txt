Q: {0..3}

function: crop SEQUENCE:n {                  \ limit the display range between [min;max] \
 if (SEQUENCE < 0){result: 0}
 if (SEQUENCE > 4){result: 4}
 result: SEQUENCE
}

function: damages from ATT:n and DEF:n {
 result: [crop (ATTdQ +DEF-ATT) - DEFdQ]
}

\
output 5dQ named "5dQ"
output 5dQ +5 named "5dQ+10"
output 10dQ named "10dQ"\

output [damages from 2 and 16] named "Att 2, Def 16"
output [damages from 5 and 10] named "Att 5, Def 10"
output [damages from 8 and 8] named "Att 8, Def 8"
output [damages from 10 and 5] named "Att 10, Def 5"

\ FLEET \
INTERCEPTOR: 2
CRUISER: 0
DREADNAUGHT: 0
STARBASE: 0

\ BLUEPRINTS \


   \ INTERCEPTOR \
      \ CPU \
      INTERCEPTOR_ELECTRON: 0
      INTERCEPTOR_POSITRON: 0
      INTERCEPTOR_GLUON: 0

      \ DMG \
      INTERCEPTOR_ION: 1
      INTERCEPTOR_PLASMA: 0
      INTERCEPTOR_SOLITON: 0
      INTERCEPTOR_ANTIMATTER: 0


   \ CRUISER \
      \ CPU \
      CRUISER_ELECTRON: 0
      CRUISER_POSITRON: 0
      CRUISER_GLUON: 0

      \ DMG \
      CRUISER_ION: 0
      CRUISER_PLASMA: 0
      CRUISER_SOLITON: 0
      CRUISER_ANTIMATTER: 0


   \ DREADNAUGHT \
      \ CPU \
      DREADNAUGHT_ELECTRON: 0
      DREADNAUGHT_POSITRON: 0
      DREADNAUGHT_GLUON: 0

      \ DMG \
      DREADNAUGHT_ION: 0
      DREADNAUGHT_PLASMA: 0
      DREADNAUGHT_SOLITON: 0
      DREADNAUGHT_ANTIMATTER: 0



   \ STARBASE \
      \ CPU \
      STARBASE_ELECTRON: 0
      STARBASE_POSITRON: 0
      STARBASE_GLUON: 0

      \ DMG \
      STARBASE_ION: 0
      STARBASE_PLASMA: 0
      STARBASE_SOLITON: 0
      STARBASE_ANTIMATTER: 0


\ DMG DICE \
INTERCEPTOR_FACES: 1+ INTERCEPTOR_ELECTRON + 2*INTERCEPTOR_POSITRON + 3*INTERCEPTOR_GLUON
INTERCEPTOR_ION_DICE: {1:INTERCEPTOR_FACES, 0:6-INTERCEPTOR_FACES}
INTERCEPTOR_PLASMA_DICE: {2:INTERCEPTOR_FACES, 0:6-INTERCEPTOR_FACES}
INTERCEPTOR_SOLITON_DICE: {3:INTERCEPTOR_FACES, 0:6-INTERCEPTOR_FACES}
INTERCEPTOR_ANTIMATTER_DICE: {4:INTERCEPTOR_FACES, 0:6-INTERCEPTOR_FACES}

CRUISER_FACES: 1+ CRUISER_ELECTRON + 2*CRUISER_POSITRON + 3*CRUISER_GLUON
CRUISER_ION_DICE: {1:CRUISER_FACES, 0:6-CRUISER_FACES}
CRUISER_PLASMA_DICE: {2:CRUISER_FACES, 0:6-CRUISER_FACES}
CRUISER_SOLITON_DICE: {3:CRUISER_FACES, 0:6-CRUISER_FACES}
CRUISER_ANTIMATTER_DICE: {4:CRUISER_FACES, 0:6-CRUISER_FACES}

DREADNAUGHT_FACES: 1+ DREADNAUGHT_ELECTRON + 2*DREADNAUGHT_POSITRON + 3*DREADNAUGHT_GLUON
DREADNAUGHT_ION_DICE: {1:DREADNAUGHT_FACES, 0:6-DREADNAUGHT_FACES}
DREADNAUGHT_PLASMA_DICE: {2:DREADNAUGHT_FACES, 0:6-DREADNAUGHT_FACES}
DREADNAUGHT_SOLITON_DICE: {3:DREADNAUGHT_FACES, 0:6-DREADNAUGHT_FACES}
DREADNAUGHT_ANTIMATTER_DICE: {4:DREADNAUGHT_FACES, 0:6-DREADNAUGHT_FACES}

STARBASE_FACES: 1+ STARBASE_ELECTRON + 2*STARBASE_POSITRON + 3*STARBASE_GLUON
STARBASE_ION_DICE: {1:STARBASE_FACES, 0:6-STARBASE_FACES}
STARBASE_PLASMA_DICE: {2:STARBASE_FACES, 0:6-STARBASE_FACES}
STARBASE_SOLITON_DICE: {3:STARBASE_FACES, 0:6-STARBASE_FACES}
STARBASE_ANTIMATTER_DICE: {4:STARBASE_FACES, 0:6-STARBASE_FACES}


\ EVALUATION \
function: for INTERCEPTOR with DMG_TYPE DMG_DICE{
result: (INTERCEPTOR*DMG_TYPE)d(DMG_DICE)
}

function: for CRUISER with DMG_TYPE DMG_DICE{
result: (CRUISER*DMG_TYPE)d(DMG_DICE)
}
function: for DREADNAUGHT with DMG_TYPE DMG_DICE{
result: (DREADNAUGHT*DMG_TYPE)d(DMG_DICE)
}
function: for STARBASE with DMG_TYPE DMG_DICE{
result: (STARBASE*DMG_TYPE)d(DMG_DICE)
}

output [for INTERCEPTOR with INTERCEPTOR_ION INTERCEPTOR_ION_DICE] + [for INTERCEPTOR with INTERCEPTOR_PLASMA INTERCEPTOR_PLASMA_DICE] + [for INTERCEPTOR with INTERCEPTOR_SOLITON INTERCEPTOR_SOLITON_DICE] + [for INTERCEPTOR with INTERCEPTOR_ANTIMATTER INTERCEPTOR_ANTIMATTER_DICE] named "Expected damage of [INTERCEPTOR] Interceptor(s)"
output [for CRUISER with CRUISER_ION CRUISER_ION_DICE] + [for CRUISER with CRUISER_PLASMA CRUISER_PLASMA_DICE] + [for CRUISER with CRUISER_SOLITON CRUISER_SOLITON_DICE] + [for CRUISER with CRUISER_ANTIMATTER CRUISER_ANTIMATTER_DICE] named "Expected damage of [CRUISER] Cruiser(s)"
output [for DREADNAUGHT with DREADNAUGHT_ION DREADNAUGHT_ION_DICE] + [for DREADNAUGHT with DREADNAUGHT_PLASMA DREADNAUGHT_PLASMA_DICE] + [for DREADNAUGHT with DREADNAUGHT_SOLITON DREADNAUGHT_SOLITON_DICE] + [for DREADNAUGHT with DREADNAUGHT_ANTIMATTER DREADNAUGHT_ANTIMATTER_DICE] named "Expected damage of [DREADNAUGHT] Dreadnaught(s)"
output [for STARBASE with STARBASE_ION STARBASE_ION_DICE] + [for STARBASE with STARBASE_PLASMA STARBASE_PLASMA_DICE] + [for STARBASE with STARBASE_SOLITON STARBASE_SOLITON_DICE] + [for STARBASE with STARBASE_ANTIMATTER STARBASE_ANTIMATTER_DICE] named "Expected damage of [STARBASE] Starbase(s)"

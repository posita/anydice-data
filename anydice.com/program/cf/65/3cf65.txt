ATTACKER_AB: 11
ATTACKER_DAMAGE: 2d12+6
FLANKING_DAMAGE: 0

ATTACKS: 1
MAP: 5

TARGET_AC: 22

function: attack ROLL:n plus CHECK:n vs DC:n dealing DAMAGE:n {
DOS: 0
if ROLL + CHECK <= DC - 10 { DOS: -1 }
if ROLL + CHECK >= DC      { DOS: 1 }
if ROLL + CHECK >= DC + 10 { DOS: 2 }

if ROLL = 1 { DOS: DOS - 1 }
if ROLL = 20 { DOS: DOS + 1 }

if DOS < -1 { DOS: -1 }
if DOS > 2 { DOS: 2 }

DEALT: 0
if DOS > 0 { DEALT: DAMAGE * DOS }

result: DEALT
}

function: perform attacks with ACBONUS:n ac ABBONUS:n ab DAMAGEBONUS:n damage {
DEALT: 0

if (ATTACKS >= 1) { DEALT: DEALT + [attack d20 plus ATTACKER_AB + ABBONUS - (MAP * 0) vs TARGET_AC + ACBONUS dealing ATTACKER_DAMAGE + DAMAGEBONUS] }
if (ATTACKS >= 2) { DEALT: DEALT + [attack d20 plus ATTACKER_AB + ABBONUS - (MAP * 1) vs TARGET_AC + ACBONUS dealing ATTACKER_DAMAGE + DAMAGEBONUS] }
if (ATTACKS >= 3) { DEALT: DEALT + [attack d20 plus ATTACKER_AB + ABBONUS - (MAP * 2) vs TARGET_AC + ACBONUS dealing ATTACKER_DAMAGE + DAMAGEBONUS] }
if (ATTACKS >= 4) { DEALT: DEALT + [attack d20 plus ATTACKER_AB + ABBONUS - (MAP * 2) vs TARGET_AC + ACBONUS dealing ATTACKER_DAMAGE + DAMAGEBONUS] }

result: DEALT
}

output [perform attacks with  0 ac 0 ab 0                 damage] named "Normal"
output [perform attacks with -2 ac 0 ab (FLANKING_DAMAGE) damage] named "Flanking"
output [perform attacks with -2 ac 1 ab (FLANKING_DAMAGE) damage] named "Flanking + Courageous Anthem"
output [perform attacks with -3 ac 1 ab (FLANKING_DAMAGE) damage] named "Flanking + Courageous Anthem + Frightened 1"
output [perform attacks with -4 ac 1 ab (FLANKING_DAMAGE) damage] named "Flanking + Courageous Anthem + Frightened 2"
output [perform attacks with -5 ac 1 ab (FLANKING_DAMAGE) damage] named "Flanking + Courageous Anthem + Frightened 3"

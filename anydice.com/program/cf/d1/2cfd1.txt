function: potential A:n {
 if A = 6 { result: 4 }
 else {
  if A >= 4 { result: 3 }
  else {
   if A >= 2 { result: 2 }
   else { 
    result: 1
   }
  }
 }
}

function: challenge B:n {
 if B = 6 { result: 2 } 
 else { 
  if B >= 4 { result: 1 } 
  else { 
   result: 0 
  }
 }
}

function: apply challenge A:n vs B:n {
 if A - B < 1 { 
  result: 1 
 }
 else {
  result: A - B
 }
}

output [potential 1d6] named "1 Potential"
output [challenge 1d6] named "1 Challenge"

loop P over {1..6} {
 loop C over {1..6} {
  output [apply challenge [potential [highest 1 of Pd6]] vs [challenge [highest 1 of Cd6]]] named "[P]v[C]"
 }
}

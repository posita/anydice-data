function: dupes in A:s {
  D: 0
  loop X over {2..#A} {
    if ((X-1)@A = X@A) { result: X@A}
  }
  result: D
}
function: dupes in A:s and B:s {
  D: 0
  C: [sort {A,B}]
  loop X over {2..#C} {
    if ((X-1)@C = X@C) { result: X@C}
  }
  result: D
}
function: dupes in A:s and B:s resist R {
  D: 0
  C: [sort {A,B}]
  R: R+2
  if (R > #C) { R: #C }
  loop X over {R..#C} {
    if ((X-1)@C = X@C) { result: X@C}
  }
  result: D
}

function: dcss A:n hit vs B:n {
  C: (1dA - (2d((B*2)-1)/2)) >= 0
  D: (1d20 = 1) * (1d{-2,2})
  result: C+D > 0
}

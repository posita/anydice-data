function: if CONDITION:n then A else B {
  if CONDITION { result: A } else { result: B }
}

function: HAB:n vs DIFF:n risk RISK:n {
 CURR_PROB: d{1:H,0:D}
 if RISK=0 | HAB=1 {
  result: CURR_PROB
 } else {
  result: [if CURR_PROB=1 then [H-1 vs D risk RISK-1] else 0]
 } 
}

loop H over {11}{
loop D over {6}{
loop R over {0..6}{
output [H vs D risk R] named "Hab [H], dif[D], risk [R]"
}}}

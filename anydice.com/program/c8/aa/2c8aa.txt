function: degree with MOD {
 if MOD < 0 {
  result: d{1..6+MOD}
 }
 result: d6+MOD
}

loop MOD over {-2..3} {
 D:[degree with MOD]

 output ((D+1)/2)+2              named "mod [MOD] / ratio .5"
 output D                        named "mod [MOD] / ratio 1"
 output [highest of 2*D-4 and 0] named "mod [MOD] / ratio 2"
}

\**** damage received per enemy attack attempt ****\

function: attackroll ROLL:n vs AC:n and EXTRA:n and PEXTRA:n  {
 if ROLL >= CRITROLL { result: CRITDAMAGE  + EXTRA } 
 if ROLL = 1 { result: 0 }
 if ROLL + EXTRA + BONUSES >= AC { result: DAMAGE  + PEXTRA}
 result: 0
}


MINUS: 0
PLAYERAC: 15
BONUSES: 9 - MINUS
CRITROLL: 20
DAMAGE: 1d6 + 4 
CRITDAMAGE: 2d6 + 4 
THICC: 1d8
NAOW: 1d8


output [attackroll 1d20 vs PLAYERAC and THICC and NAOW]+[attackroll 1d20 vs PLAYERAC and THICC and NAOW]+[attackroll 1d20 vs PLAYERAC and THICC and NAOW] named "CROSSBOW"

function: phaseone ATT:s DEF:s {                \Find the environment modifiers\
  SIZE_MODS: [compare bodies 11@ATT with 11@DEF]           \compare the nbr of bodies\
  VELOCITY_MODS: [compare 10@DEF with 9@ATT]        \compare the velocity for the dodge capacity\

  ATT_DICES: (1@ATT +SIZE_MODS)                      \final number of hits-dices to roll\
  DEF_DICES: (5@DEF +VELOCITY_MODS)                  \final number of saves-dices to roll\

  \freeze the hits-dices & saves-dices sequences\
  result: [phasetwo ATT_DICESdQ DEF_DICESdQ ATT DEF]
}

function: phasetwo ATT_ROLL:s DEF_ROLL:n  ATT:s DEF:s {       \Analysis of the rolls\
  FINAL_ATT_ROLL: ATT_ROLL + ???                          \apply rolled values modifiers to hit roll\
  FINAL_DEF_ROLL: DEF_ROLL + ???                          \apply rolled values modifiers to save roll\

  ATT_SUCCESSES: [overflowing pairing FINAL_ATT_ROLL on (2@ATT+5@ATT)]      \find Hit-Successes\
  DEF_SUCCESSES: [overflowing pairing FINAL_DEF_ROLL on (7@DEF+10@DEF)]     \find Save-Successes\

  DAMAGES: ATT_SUCCESSES - DEF_SUCCESSES
  \DAMAGES: [halve ATT_SUCCESSES] - [halve DEF_SUCCESSES]\
  result: [crop DAMAGES]
}


\******** Functions to find Successes on chosen target **********\

function: overflowing pairing SEQUENCE:s on TARGET:n {
  TOTAL: 0
  \count of number in sequence\
  UN:    SEQUENCE=1
  DEUX:  SEQUENCE=2
  TROIS: SEQUENCE=3
  TEMP: {UN, DEUX, TROIS}
  \sum of chosen numbers'occurencies\
  TOTAL: {TARGET..3}@TEMP                        \final count of Successes\

  \boolean checks : 0 or 1 multiplicators\
  DT: DEUX>=TROIS
  UT: UN>=TROIS
  UD: UN>=DEUX

  if(TARGET=6){result: TROIS/2}
  if(TARGET=5){result: (TROIS+ (DT*TROIS)+(!DT*DEUX)) /2}
  if(TARGET=4){result: (TROIS+ (UT*(TROIS+DEUX)) + (!UT*(UN+DEUX))) /2}
  if(TARGET=3){result: TOTAL + ( (DEUX+ (UD*DEUX)+(!UD*UN)) /2)}
  if(TARGET=2){result: TOTAL + (UN/2)}
  if(TARGET=1){result: TOTAL}
  result: TOTAL
}

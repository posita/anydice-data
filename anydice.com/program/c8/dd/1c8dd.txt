function: hits of R on J challenge die {
SUM: 0
loop I over {1..J} {
SUM: SUM + (R > I@Jd10)
}
result: SUM
} 

loop CD over {2} {       \- number of challenge die -\
loop B over {1..5} {     \- bonus -\
ACD: 1d6+B               \- action die roll with bonus -\
output [hits of ACD on CD challenge die] named "Roll+[B] Hits on [CD] Challenge Die"
}
}

DAM:1d10

ADV:[highest 1 of 2d20]
NOR:1d20

\consider AC the effective AC (i.e., AC of target - to hit bonus)\
ACLOW:20 - 5
ACHIGH:20 - 0

CRITONE:20
CRITTWO:19

function: ROLL:n AC:n CRIT:n crit 
{
if ROLL<CRIT
   {if ROLL-AC>=0
       {result:DAM}
    else
       {result:0}
   }
else
   {result:DAM*2}
}

output [ADV ACLOW CRITONE crit] + [ADV ACLOW CRITONE crit] named "Damage: 2 attacks w/Adv - AC low - Crit 20"

output [ADV ACLOW CRITTWO crit] + [ADV ACLOW CRITTWO crit] named "Damage: 2 attacks w/Adv - AC low - Crit 19"

output [NOR ACLOW CRITONE crit] + [NOR ACLOW CRITONE crit] + [NOR ACLOW CRITONE crit] named "Damage: 3 attacks Normal - AC low - Crit 20"



output [ADV ACHIGH CRITONE crit] + [ADV ACHIGH CRITONE crit] named "Damage: 2 attacks w/Adv - AC high - Crit 20"

output [ADV ACHIGH CRITTWO crit] + [ADV ACHIGH CRITTWO crit] named "Damage: 2 attacks w/Adv - AC high - Crit 19"

output [NOR ACHIGH CRITONE crit] + [NOR ACHIGH CRITONE crit] + [NOR ACHIGH CRITONE crit] named "Damage: 3 attacks Normal - AC high - Crit 20"

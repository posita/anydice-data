function: check A:n on B:n {
  if (B < 6 & A  > 0) {
    result: [check (A-1) on 1d10]
  } else {
    result: B
  }
}

function: crit C:n at D:n on E:n {
  if (C = 1 & D+E+5 <= 10) {
    result: C-1d10
  } else if (C = 10 & D+E >= 10) {
    result: C+1d10
  } else {
    result: C
  }
}

function: phaseone ATT:s DEF:s {                \Find the environment modifiers\
  SIZE_MODS: [compare bodies 8@ATT with 8@DEF]           \compare the nbr of bodies\
  VELOCITY_MODS: [compare 7@DEF with 6@ATT]           \compare the velocity for the dodge capacity\

  ATT_BASE_DICES: [find chart level 1@ATT 1@DEF]        \base number of hit-dices\
  ATT_MODS_DICES: 2@ATT+SIZE_MODS                       \final number of added hit-dices\
  ATT_CHART: [chart used 1 1@ATT 1@DEF]                 \find the chart used for the hit roll\

  DEF_BASE_DICES: [find chart level 1@DEF 1@ATT]        \base number of save-dices\
  DEF_MODS_DICES: 4@DEF+VELOCITY_MODS                   \final number of added save-dices\
  DEF_CHART: [chart used 0 1@DEF 1@ATT]                 \find the chart used for the save roll\

  \freeze the hits roll sequences\
  result: [phasetwo [highlow roll ATT_BASE_DICES ATT_MODS_DICES] ATT_CHART DEF_BASE_DICES DEF_MODS_DICES DEF_CHART ATT DEF]
}

function: phasetwo ATT_ROLL:n CHART:s DEF_BASE_DICES:n DEF_MODS_DICES:n DEF_CHART:s ATT:s DEF:s {       \Analysis of the Hit roll\
  ATT_ROLL: ATT_ROLL-3@ATT                              \apply value modifiers on hit roll\
  if(ATT_ROLL<1@CHART){result: 0}
  else{
   if(ATT_ROLL>=4@CHART){HITS: 4}
   if(ATT_ROLL<4@CHART){HITS: 3}
   if(ATT_ROLL<3@CHART){HITS: 2}
   if(ATT_ROLL<2@CHART){HITS: 1}
   result: [phasethree [highlow roll DEF_BASE_DICES DEF_MODS_DICES] HITS DEF_CHART DEF]           \test 1\
   \result: [phasethreebis [highlow roll DEF_BASE_DICES DEF_MODS_DICES-HITS] DEF_CHART DEF]\      \test 2\
  }
  result: 0
}

function: phasethree DEF_ROLL:n HITS:n CHART:s DEF:s {       \Analysis of the Save roll\
  DEF_ROLL: DEF_ROLL-5@DEF                                   \apply value modifiers on save roll\
  if(DEF_ROLL>=4@CHART){SAVES: 4}
  if(DEF_ROLL<4@CHART) {SAVES: 3}
  if(DEF_ROLL<3@CHART) {SAVES: 2}
  if(DEF_ROLL<2@CHART) {SAVES: 1}
  if(DEF_ROLL<1@CHART) {SAVES: 0}
  DAMAGES: [crop HITS-SAVES]                   \test 1\
  result: DAMAGES
}
function: phasethreebis DEF_ROLL:n CHART:s DEF:s {       \Analysis of the Save roll\
  if(DEF_ROLL>=4@CHART){SAVES: 4}
  if(DEF_ROLL<4@CHART) {SAVES: 3}
  if(DEF_ROLL<3@CHART) {SAVES: 2}
  if(DEF_ROLL<2@CHART) {SAVES: 1}
  if(DEF_ROLL<1@CHART) {SAVES: 0}
  result: SAVES                                \test 2\
}

\test 1 : X Hits - Y Saves = Damages\
\test 2 : Hits gives Malus to Saves, then the Saves gives directly the Damages\


\******** Functions to use the charts & get the number of dices to roll *******************\

HIT_LIGHT:  {1, 4, 8, 12}
HIT_MEDIUM: {2, 5, 10, 14}
HIT_HEAVY:  {3, 6, 12, 16}
SAVE_LIGHT:  {1, 5, 7, 9}
SAVE_MEDIUM: {2, 6, 8, 11}
SAVE_HEAVY:  {3, 7, 10, 13}

function: chart used HITPHASE:n A:n B:n {          \get the chart for the hit roll or the save roll\
  if(HITPHASE){
   if(A<B){result: HIT_LIGHT}
   if(A=B){result: HIT_MEDIUM}
   result: HIT_HEAVY
  }else{
   if(A<B){result: SAVE_LIGHT}
   if(A=B){result: SAVE_MEDIUM}
   result: SAVE_HEAVY
}}

function: find chart level A:n B:n {              \compare Units Types to know which chart to use\
  if(A<B){result: 4}
  if(A=B){result: 5}
  result: 6}

function: highlow roll DICES:n MODS:n {              \switch highest or lowest X dices of the pool\
  if(MODS>=0){
   POOL: DICES+MODS
   result: [highest DICES of POOLdQ]
  }else{
    POOL: DICES-MODS                                 \make MODS positive\
    result: [lowest DICES of POOLdQ]}
}

\******** Functions to compare Sets of Dices (Optimized at max) **********\

function: find damage bonus ATT:s DEF:s {
 ATT_SET: [find best set ATT]
 DEF_SET: [find best set DEF]
 result: [compare sets ATT_SET DEF_SET]
}

function: find best set ROLL:s {
  \separate the sets in a roll\
  THREES: ROLL=3                     \test the value on each element of the Sequence parameter : optimized\
  TWOS:   ROLL=2
  ONES:   ROLL=1

  \compare the sets to find the more numerous one, choosing the higher if equal\
  if (THREES>0) & (THREES>=TWOS) & (THREES>=ONES) { result: 3}
  if (TWOS>0) & (TWOS>=ONES) { result: 2}
  if (ONES>0) { result: 1}
  result: 0
}

function: compare sets ATT_SET:n DEF_SET:n {
\if Attacker has a higher set's face than Defender, add Damage Bonus\
 if (ATT_SET>DEF_SET) { result: ATT_SET-DEF_SET }
 result: 0
}

  if(ATT_B > DEF_B) { result: ATT_B-DEF_B}
  result: 0
}
function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  result: -1       \VAL_X < VAL_Y\
}

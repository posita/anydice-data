function: attack ROLL:n TO_HIT:n DMG_DICE:d DMG_MOD:n versus AC:n  {
if ROLL = 1  {result: 0}
if ROLL >= 20 {result: 2*DMG_DICE + DMG_MOD - 3}
if ROLL+TO_HIT >= AC {result: DMG_DICE + DMG_MOD - 3}
result: 0
}

function: attack ROLL:n TO_HIT:n DMG_DICE:d DMG_MOD:n versus AC:n with splint{
if ROLL = 1  {result: 0}
if ROLL >= 20 {result: DMG_DICE / 2 + DMG_DICE + DMG_MOD - 3}
if ROLL+TO_HIT >= AC {result: DMG_DICE + DMG_MOD - 3}
result: 0
}

output [attack d20 5 1d8 2 versus 20]  named "+5 1d8+2 vs AC 20 + full plate"

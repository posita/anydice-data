CRITICALFAIL: -1
MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack DICE:n vs TN:n {
 if DICE = 2 { result: CRITICALFAIL }
 if STATSKILL + DICE = TN { result: CRITICALHIT }
 if STATSKILL - DICE = TN { result: CRITICALHIT }
 if STATSKILL + DICE > TN { result: HIT }
 result: MISS
}

STATSKILL: 24
loop DIFFICULTY over {9..36} {
 output [attack [highest 2 of 3d6] vs DIFFICULTY] named "TN [DIFFICULTY]"
\ output [attack [lowest 2 of 3d6] vs DIFFICULTY] named "TN [DIFFICULTY]"\
\ output [attack [highest 2 of 3d6] vs DIFFICULTY] named "TN [DIFFICULTY]"\
}

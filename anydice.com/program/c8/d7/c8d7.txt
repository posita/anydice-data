SPELL_MOD: 5
SPELL_LEVEL: 1

FIRST_HEAL_POT: 2d4 + 2
SECOND_HEAL_POT: 4d4 + 4
THIRD_HEAL_POT: 8d4 + 8



function: cast base HEAL:n MOD:n LEVEL:n HIGHER_LEVEL:n {
  result: HEAL + MOD + (LEVEL * HIGHER_LEVEL)
}

function: life cleric cast base HEAL:n MOD:n LEVEL:n HIGHER_LEVEL:n {
  result: [cast base HEAL MOD LEVEL HIGHER_LEVEL] + 2 + LEVEL
}

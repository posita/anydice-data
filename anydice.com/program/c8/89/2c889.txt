function: call CALL:s {
result: (CALL)}
function: darkd TARGET:n in DIE:n {
 COUNT: 0
 FAIL: 0
 VALUES: {TARGET..10,10}
 SEQUENCE: [call (DIE)d10]
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
  FAIL: FAIL + (1 = P@SEQUENCE)
 }
if COUNT > 0{
 result: COUNT}
else if FAIL >= (#SEQUENCE)/2 {
 result: -1}
else {
 result: 0}
}
output [darkd 7 in 2]
output [count {7..10,10} in 4d10]

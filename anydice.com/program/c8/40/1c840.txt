DAM:1d10

ADV:[highest 1 of 2d20]
NOR:1d20

\consider AC the effective AC (i.e., AC of target - to hit bonus)\
ACLOW:20 - 5
ACHIGH:20 - 0

CRITONE:20
CRITTWO:19


function: ROLL:n AC:n CRIT:n crit 
{
if ROLL<CRIT
   {if ROLL-AC>=0
       {result:DAM}
    else
       {result:0}
   }
else
   {result:DAM*2}
}


function: ROLL:n AC:n CRIT:n critorc 
{
if ROLL<CRIT
   {if ROLL-AC>=0
       {result:DAM}
    else
       {result:0}
   }
else
   {result:DAM*2 + DAM}
}





ATHLETICS:1d20
OPP_ATH:1d20

function: PUSH:n CONTEST:n prone
{
if PUSH<CONTEST
   {result:0}
else
   {result:1}
}


function: A:n secfunc
{
result: A + DAM
}


output [[ATHLETICS OPP_ATH prone] secfunc]

\
output [ATHLETICS OPP_ATH prone]

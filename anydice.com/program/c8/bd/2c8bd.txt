  Poisson die vs poisson distribution
\

function: rare N:n {
  if N < 120 {result: 0}
  if N = 120 {result: 1}
}

function: poisson N:n {
  if N = 8 { result: 2 + [poisson d8] }
  if N = 7 { result: 2 }
  if N > 3 { result: 1 }
  result: 0
}

output [lowest of 2d[poisson d8] and 8] named "dP"
output [lowest of 240d[rare d120] and 8] named "poisson"

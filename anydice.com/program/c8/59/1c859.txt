function: blades ROLL:s {
 if 1@ROLL = 6 {
  result: 2 + (2@ROLL = 6)
 }
 result: 1@ROLL >= 4
}

function: cost R:n {
  if R = 0 {
    result: 1
  }
  if R = 1 {
    result: 2
  }
  if R = 2 {
    result: 3
  }
  if R = 3 {
    result: 5
  }
}

output [cost [blades 2@2d6]] named "0d"
loop P over {1..7} {
 output [cost [blades Pd6]] named "[P]d"
}

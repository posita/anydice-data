THORN: 1d6

\outputs a chance 0-100 instead of 0.00 - 1.00\
function: hitchance BONUS:n vs AC:n {
  result: (21 + BONUS - AC)*5
}

function: percenthit X:n ATTACK:d with CHANCE:n {
    result: Xd{ATTACK:CHANCE, MISS:(100-CHANCE)}
}

MISS: 0

 TEMP: DIV*2
 if (DIV=TEMP){result: DIV}
 result: (SEQUENCE+1)/2
}

function: phaseone ATT:s DEF:s {                \Find the environment modifiers\
  SIZE_MODS: [compare bodies 11@ATT with 11@DEF]           \compare the nbr of bodies\
  VELOCITY_MODS: [compare 10@DEF with 9@ATT]        \compare the velocity for the dodge capacity\

  ATT_DICES: ([find chart level 1@ATT 1@DEF] +SIZE_MODS)                \final number of hits-dices to roll\
  DEF_DICES: ([find chart level 1@DEF 1@ATT] +VELOCITY_MODS)            \final number of saves-dices to roll\

  result: [phasetwo ATT_DICESdQ DEF_DICES ATT DEF]  \freeze the hits-dices & saves-dices sequences\
}

function: phasetwo ATT_ROLL:n DEF_DICES:n CHART:s ATT:s DEF:s {       \Analysis of the Hit roll\
  if(ATT_ROLL<1@CHART){result: 0}
  else{
   if(ATT_ROLL>=4@CHART){HITS: 4}
   if(ATT_ROLL<4@CHART){HITS: 3}
   if(ATT_ROLL<3@CHART){HITS: 2}
   if(ATT_ROLL<2@CHART){HITS: 1}
   result: [phasethree DEF_DICESdQ HITS ATT DEF]
  }
  result: 0
}

function: phasethree DEF_ROLL:n HITS:n CHART:s ATT:s DEF:s {       \Analysis of the Save roll\
  if(DEF_ROLL>=4@CHART){SAVES: 4}
  if(DEF_ROLL<4@CHART) {SAVES: 3}
  if(DEF_ROLL<3@CHART) {SAVES: 2}
  if(DEF_ROLL<2@CHART) {SAVES: 1}
  if(DEF_ROLL<1@CHART) {SAVES: 0}
  DAMAGES: HITS - SAVES
  result: DAMAGES
}


\test 1 : X Hits - Y Saves = Damages\
\test 2 : Hits gives Malus to Saves, then the Saves gives directly the Damages\

function: find chart level ATT:n DEF:n {              \compare Units Types to know which chart to use\
  if(ATT<DEF){result: 4}
  if(ATT=DEF){result: 5}
  result: 6}


function: check chart CHART:s ROLL:n {          \check if the roll is successfull & with which bonus\
  if(ROLL<1@CHART) {result: -1}
  if(ROLL<2@CHART) {result: 0}
  if(ROLL<3@CHART) {result: 1}
  if(ROLL<4@CHART) {result: 2}
  result: 3
}

function: highlow roll DICES:n MOD:n POOL:n {              \switch highest or lowest X dices of the pool\
  if(DICES+MOD >=DICES){result: [highest DICES of (DICES+MOD)dQ]}
  else{result: [lowest DICES of (DICES+MOD)dQ]}
}

\******** Functions to compare Sets of Dices (Optimized at max) **********\

function: find damage bonus ATT:s DEF:s {
 ATT_SET: [find best set ATT]
 DEF_SET: [find best set DEF]
 result: [compare sets ATT_SET DEF_SET]
}

function: find best set ROLL:s {
  \separate the sets in a roll\
  THREES: ROLL=3                     \test the value on each element of the Sequence parameter : optimized\
  TWOS:   ROLL=2
  ONES:   ROLL=1

  \compare the sets to find the more numerous one, choosing the higher if equal\
  if (THREES>0) & (THREES>=TWOS) & (THREES>=ONES) { result: 3}
  if (TWOS>0) & (TWOS>=ONES) { result: 2}
  if (ONES>0) { result: 1}
  result: 0
}

function: compare sets ATT_SET:n DEF_SET:n {
\if Attacker has a higher set's face than Defender, add Damage Bonus\
 if (ATT_SET>DEF_SET) { result: ATT_SET-DEF_SET }
 result: 0
}

function: compare bodies ATT_B:n with DEF_B:n {
  if(ATT_B > DEF_B) { result: ATT_B-DEF_B}
  result: 0
}
function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  result: -1       \VAL_X < VAL_Y\
}

function: ROLL:n BONUS:n AC:n CRIT:n crit 
{
if ROLL<CRIT
   {if ROLL+BONUS-AC>=0
       {result:DAM+BONUS}
    else
       {result:0}
   }
else
   {result:(DAM*2)+BONUS}
}

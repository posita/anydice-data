function: call NUM:n by DIE:n {
result: darkd NUM in (DIE)d10}
function: darkd TARGET:n in SEQUENCE:s {
 COUNT: 0
 FAIL: 0
 VALUES: {TARGET..10,10}
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
  FAIL: FAIL + (1 = P@SEQUENCE)
 }
if COUNT > 0{
 result: COUNT}
else if FAIL >= (#SEQUENCE)/2 {
 result: -1}
else {
 result: 0}
}
output [darkd 7 in 4d10]
output [call 7 by 4]

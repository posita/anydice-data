\----------------------------------------\
\--------Generic dice roll output--------\
\----------------------------------------\
\output 1d6 named "1d"\
\output 2d6 named "2d"\
\output 3d6 named "3d"\

\----------------------------------------\
\-----Odds of success for Dice Pools-----\
\----------------------------------------\
\loop DICE over {6}{
   loop DIFF over {1..3}{
    if DICE >= DIFF{
     HITS: [count {5,6} in DICEd6 ]
     output HITS >= DIFF named "[DIFF], [DICE]"
   }
  }
 }\


\----------------------------------------\
\----------contested Dice Rolls----------\
\----------------------------------------\
loop ATT over {4..6}{
 loop DEF over {1..3}{
  HITS: [count {5,6} in ATTd6]
  BLOCKS: [count {5,6} in DEFd6]
  output (HITS-BLOCKS)>1 named "a[ATT]d vs d[DEF]d"
 }  
}

\
 Blades in the Dark
 0: Critical Failure with Heavy Loss
 1: Failure with Heavy Loss
 2: Success with Heavy Loss
 3: Success with Light Loss
 4: Critical Success
\

function: blades ROLL:s {
 if 1@ROLL = 6 {
  result: 2 + (2@ROLL = 6)
 }
 result: 1@ROLL >= 4
}

output [blades 2@2d6] named "0d"
loop P over {1..20} {
 output [blades Pd6] named "[P]d"
}

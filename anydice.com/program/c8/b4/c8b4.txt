function: ph A:n {
  if (A < 2) {result: -1}
  else if (A > 2) {result: 1}
  else {result: 0}
}

function: phb A:n {
  if (A >= 10) {result: 1}
  else if (A < 7) {result: -1}
  else {result: 0}
}

loop O over {0..2} {
  loop P over {1..7} {
    N: P-O
    \output [ph [count {4..6} in Nd6]] named "[P]d6 Ob [O]  |  [N] dice"\
    output [phb [highest 1 of Nd10]] named "[P]d10 Ob [O]  |  [N] dice"
  }
}

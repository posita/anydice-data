function: ROLL:s assemble VAL:n {
 loop X over ROLL{
  if [ROLL contains VAL-X]{
   result: 1
  }
 }
 result: 0
}

function: ROLL:s interpret VAL:n {
 if [count VAL in ROLL] { result: 1 } \ success \
 if [ROLL assemble VAL] { result: 0 } \ partial \
 result: -1 \ failure \
}

loop POOL over {2..8}{
 output [POOL d8 interpret 8] named "[POOL]d"
}

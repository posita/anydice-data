function: reroll lowest of ROLL:s or both REROLL:s as of SINGLE:d if any result less than MIN:n {

   LOWEST: {#ROLL}@ROLL
   HIGHEST: {1..#ROLL}@ROLL
   LOWESTT: {#REROLL}@REROLL


   if LOWEST >= MIN {
        result: LOWEST } 

   if HIGHEST < MIN {
        result: LOWESTT }

   if LOWEST < MIN {
        result: SINGLE }
}
   
output [reroll lowest of 2d10 or both 2d10 as of 1d10 if any result less than 4]

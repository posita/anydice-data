function: advantage N:n DICE:d {
if N < 2 {
result: DICE
  }
MAX: DICE
loop I over {2..N} {
MAX: [highest of MAX and DICE]
  }
result: MAX
}

ABILITIES: 6 d [highest 3 of 4d6]
loop P over {1..6} {
 output P @ ABILITIES named "Ability [P] 4d6kh3"
}
loop N over {1..5} {
ABILITIESAD: 6 d [advantage N 3d6]
loop P over {1..6} {
 output P @ ABILITIESAD named "Ability [P] highest of [N] 3d6 set(s)"
}
}

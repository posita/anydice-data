 if ROLL = 1 {X: ROLL + MODIFIER - 10}
 else if ROLL >= CRITRANGE {X: ROLL + MODIFIER + 10}
 else {X: ROLL + MODIFIER}

 CRIT: 2
 HIT: 1
 MISS: 0

 if X >= DEFENSE  + 10 {result: CRIT}
 else if X >= DEFENSE {result: HIT}
 else {result: MISS}
}

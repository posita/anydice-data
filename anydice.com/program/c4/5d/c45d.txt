function: reroll doubles X:n Y:n {
if (X+Y>2) & (X=Y) { result: X+Y + [reroll doubles again 1d10 1d10] }
else { result: X+Y }
}

function: reroll doubles again X:n Y:n {
if (X+Y>2) & (X=Y) { result: X+Y + [reroll doubles yet again 1d10 1d10] }
else { result: X+Y }
}

function: reroll doubles yet again X:n Y:n {
if (X+Y>2) & (X=Y) { result: X+Y + 2d10 }
else { result: X+Y }
}

output [reroll doubles 1d10 1d10]

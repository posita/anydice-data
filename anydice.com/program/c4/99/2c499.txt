set "maximum function depth" to 100

DICE: {4, 6, 8, 10, 12}
RECURSE: 100

function: explodesw D:n SIDES:n ACC:n COUNT:n {
  if COUNT = 1 {
    result: D + ACC
  }
  if D = SIDES {
    COUNT: COUNT - 1
    result: [explodesw 1dSIDES SIDES D+ACC COUNT]
  }
  result: D + ACC
}

\
loop DIE over DICE {
  output [explodesw 1dDIE DIE 0 RECURSE] named "1d[DIE]"
}
\
output [explodesw 1d4 4 0 RECURSE] > [explodesw 1d6 6 0 RECURSE]

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

output d12+d12 named "d12 d12 +10"
output [reroll {1} on d12 to d12]+[reroll {1} on d12 to d12] named "d12 d12 +10 - 1 Stamina/Mana"
output [reroll {1,2} on d12 to d12]+[reroll {1,2} on d12 to d12] named "d12 d12 +10 - 2 Stamina/Mana"
output [reroll {1,2,3} on d12 to d12]+[reroll {1,2,3} on d12 to d12] named "d12 d12 +10 - 3 Stamina/Mana"
output [reroll {1,2,3,4} on d12 to d12]+[reroll {1,2,3,4} on d12 to d12] named "d12 d12 +10 - 4 Stamina/Mana"

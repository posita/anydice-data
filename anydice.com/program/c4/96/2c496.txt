function: assemble ROLL:s {
 loop VAL over ROLL{
  if [ROLL contains 8-VAL]{
   result: 1
  }
  else {
   result: 0
  }
 }
}

function: interpret ROLL:s {
 if [count 8 in ROLL] { result: 1 } \ success \
 if [assemble ROLL] { result: 0 } \ partial \
 result: -1 \ failure \
}

loop DICE over {0..8}{
 POOL: DICE+2
 output [interpret POOL d8] named "[DICE]d"
}

CRITMISS: 0
MISS: 0
HIT: 1
CRITHIT: 2

function: attack ROLL:n with BONUS:n vs DEFENSE:n {
  R: 0
  RB: ROLL + BONUS
  if RB >= DEFENSE+10 { R: CRITHIT }
  else {if RB >= DEFENSE { R: HIT }}
  if RB <= DEFENSE-10 { R: CRITMISS }
  else {if RB < DEFENSE { R: MISS }}

  if ROLL = 20 { R: R + 1 }
  if ROLL = 1 { R: R - 1 }
  R: [highest of R and CRITMISS]
  R: [lowest of R and CRITHIT]
  result: R
}

output [attack d20 with 9 vs 15] * (1d8 + 4)

function: dupesorones in DICE:s {
  D: 0
  H: 0
  loop X over {1..#DICE} {
    if ((X-1)@DICE = X@DICE) { D: D + 1}
    if (X@DICE = 1) { D: D + 1 }
    if (X@DICE >= H) { H: X@DICE }
  }
  if(D > 0){
    result: 0
  }
  result: H
}

output [dupesorones in 1d12] named "1d12"
output [dupesorones in 2d12] named "2d12"

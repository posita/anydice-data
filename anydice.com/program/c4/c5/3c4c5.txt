function: our M:s attack{
 ATTACK_STAT: (2@M)@{OPA,OSA,OSA}
 DEFENSE_STAT: (2@M)@{EPD,ESD,EPD}
 ACCURACY: 4@M*(OACM-EEVM+14)@{280:7,315,360,420,504,630,840,1120,1400,1680,1960,2240,2520:7}/840
 if 2@M > 0 {
  result:(3@M\Power\ *ATTACK_STAT *42 /(DEFENSE_STAT *50)+2) *\RNG\1d{85..100}/100 *5@M\TYPE\/16 *1d{1:ACCURACY,0:100-ACCURACY}
 }
}
function: enemy M:s attack{
 ATTACK_STAT: (2@M)@{EPA,ESA,ESA}
 DEFENSE_STAT: (2@M)@{OPD,OSD,OPD}
 ACCURACY: 4@M*(EACM-OEVM+14)@{280:7,315,360,420,504,630,840,1120,1400,1680,1960,2240,2520:7}/840
 if 2@M > 0 {
  result:(3@M\Power\ *ATTACK_STAT *42 /(DEFENSE_STAT *50)+2) *\RNG\1d{85..100}/100 *5@M\TYPE\/16 *1d{1:ACCURACY,0:100-ACCURACY}
 }
}

function: round N move{
 if N = 1 {result:OMA}
 if N = 2 {result:OMB}
 if N = 3 {result:OMC}
 if N = 4 {result:OMD}
 if N = 5 {result:OME}
 if N = 6 {result:OMF}
 if N = 7 {result:OMG}
 if N >= 8 {result:OMH}
}
function: round N counter{
 if N = 1 {result:EMA}
 if N = 2 {result:EMB}
 if N = 3 {result:EMC}
 if N = 4 {result:EMD}
 if N = 5 {result:EME}
 if N = 6 {result:EMF}
 if N = 7 {result:EMG}
 if N >= 8 {result:EMH}
}
\EDAMAGE: {}
ODAMAGE: {}
STACK: {}
CHECK:1
END:0
loop N over {1..8}{
 OUR_MOVE: [round N move]
 ENEMY_MOVE: [round N counter]
 if (1@{ENEMY_MOVE} > 1@{OUR_MOVE}) | (ESP > OSP & 1@{ENEMY_MOVE} = 1@{OUR_MOVE}){
  ODAMAGE: ODAMAGE + [enemy ENEMY_MOVE attack]
  EDAMAGE: EDAMAGE + [our OUR_MOVE attack]
  TURN: (EDAMAGE >= EHP)*(ODAMAGE < OHP)
 }
 if (1@{OUR_MOVE} > 1@{ENEMY_MOVE}) | (OSP >= ESP & 1@{OUR_MOVE} = 1@{ENEMY_MOVE}) {
  EDAMAGE: EDAMAGE + [our OUR_MOVE attack]
  TURN: (EDAMAGE >= EHP)*(ODAMAGE < OHP)
  ODAMAGE: ODAMAGE + [enemy ENEMY_MOVE attack]
 }
 STACK: STACK + TURN*CHECK
 CHECK:(EDAMAGE < EHP)*(ODAMAGE < OHP)
 if 1@{CHECK}=0 & [maximum of CHECK]>0 {output 0 < STACK named "Win Chance(1) on Turn [N]"}
 output TURN named "turn [N]"
 output CHECK named "check [N]"
 output STACK named "stack [N]"
 output EDAMAGE named "damage dealt turn [N] out of [EHP]"
 output ODAMAGE named "damage taken turn [N] out of [OHP]"
}\
OCHP: OHP
ECHP: EHP
STACK:0
END:0
loop N over {1..8}{
 OUR_MOVE: [round N move]
 ENEMY_MOVE: [round N counter]
 if (1@{ENEMY_MOVE} > 1@{OUR_MOVE}) | (ESP > OSP & 1@{ENEMY_MOVE} = 1@{OUR_MOVE}){
  OCHP: OCHP - [enemy ENEMY_MOVE attack]*([maximum of ECHP]>=0)*([maximum of OCHP]>=0)
  ECHP: ECHP - [our OUR_MOVE attack]*([maximum of OCHP]>=0)*([maximum of ECHP]>=0)
  WIN: (ECHP <=0)*(OCHP >0)
 }
 if (1@{OUR_MOVE} > 1@{ENEMY_MOVE}) | (OSP >= ESP & 1@{OUR_MOVE} = 1@{ENEMY_MOVE}) {
  ECHP: ECHP - [our OUR_MOVE attack]*([maximum of OCHP]>=0)*([maximum of ECHP]>=0)
  WIN: (ECHP <=0)*(OCHP >0)
  OCHP: OCHP - [enemy ENEMY_MOVE attack]*([maximum of ECHP]>=0)*([maximum of OCHP]>=0)
 }
 STACK: STACK + WIN
 if [maximum of ECHP]<=0 & 1@{OCHP}>0 & END=0 {
  output OCHP>0 named "Guaranteed Turn [N] Enemy KO, Final Win Chance(1)"
  END:1
 }
 if [maximum of OCHP]<=0 & END=0 {
  output STACK>0 named "Guaranteed Turn [N] Faint, Final Win Chance(1)"
  END:1
 }
 if 1@{ECHP}<=0 & [maximum of ECHP]>0 & END=0 {
  output 0 < STACK named "Possible Turn [N] Win Chance(1)"
 }

 output [highest of OCHP and 0] named "Our HP turn [N] out of [OHP]"

\------------------------ [HELPERS] ------------------------\
function: minimum of DIE:d
{
  result: 1@{DIE}
}


\------------------ [GREAT WEAPON FIGHTING] ------------------\
function: great weapon fighting DIE:d
{
  NUM_DICE: [minimum of DIE]
  DIE_TYPE: [maximum of DIE] / NUM_DICE
  FINAL_VALUE: 0
  loop N over {1..NUM_DICE}
  {
    FINAL_VALUE: FINAL_VALUE + [get roll 1dDIE_TYPE]
  }

  result: FINAL_VALUE
}
function: get roll ROLL:n
{
  if ROLL > 1
  {
    result: ROLL
  }
  result: 1dDIE_TYPE
}


\--------------------- [CALCULATIONS] ---------------------\
output [lowest 1 of 2d20] named "d20 Disadvantage"
output 2d6 named "3d6 Disadvantage"
output [great weapon fighting 2d6] named "3d6 Disadvantage reroll 1s"
output [lowest 2 of 3d6] named "Alternative 3d6 Disadvantage"
output 1d20 named "d20"
output [great weapon fighting 1d20] named "d20 reroll 1s"
output 3d6 named "3d6 reroll 1s"
output [great weapon fighting 3d6] named "3d6 reroll 1s"
output [highest 1 of 2d20] named "d20 advantage"
output 4d6 named "3d6 Advantage"
output [great weapon fighting 4d6] named "3d6 Advantage reroll 1s"
output [highest 3 of 4d6] named "Alternative 3d6 Advantage"

function: harm A:n step S:n {
  if A <= 0 {
    result: A
  }

  S: S + (3 - A)

  if S > 0 {
    result: A - [highest 1 of (S + 1)dH]
  } else {
    result: A - [lowest 1 of (-S + 1)dH]
  }
} 

function: skull C:n of A:s step S:n {
  if S > 0 {
    result: [highest C of (C + S)dA]
  }

  result: [lowest C of (C - S)dA]
}

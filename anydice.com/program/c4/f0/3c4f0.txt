function: blades ROLL:s {
 if 1@ROLL = 6 {
  result: 3 + ([count 6 in ROLL] >= 2)
 }

 if 1@ROLL >= 4 {
  result: 1 + ([count 1 in ROLL] <= 1)
 }
 
 if 1@ROLL >= 2 {
  result: 0 + ([count 1 in ROLL] <= 1)
 
 }
 
 result: 0
 
}

output [blades 1d4] named "0d"
loop P over {1..8} {
 output [blades Pd6] named "[P]d"
}

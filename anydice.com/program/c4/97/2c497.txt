set "maximum function depth" to 100

DICE: {4, 6, 8, 10, 12}
RECURSE: 100

function: explodesw D:n N:n A:n C:n {
  if C = 1 {
    result: D + A
  }
  if D = N {
    C: C - 1
    result: [explodesw 1dN N D+A C]
  }
  result: D + A
}

\
loop DIE over DICE {
  output [explodesw 1dDIE DIE 0 RECURSE] named "1d[DIE]"
}
\
output [explodesw 1d4 4 0 RECURSE] > [explodesw 1d6 6 0 RECURSE]

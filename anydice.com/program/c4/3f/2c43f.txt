function: rolling ROLL:s if lower than MIN:n reroll REROLL:s {

  INTIMACYLOW: 2@ROLL       \setting highest & lowest of initial 2d10 roll \
  INTIMACYHIGH: 1@ROLL

  if INTIMACYLOW >= MIN { result: INTIMACYLOW }  \ if both results 4 or more \

  if INTIMACYHIGH >= MIN {                                 \ if rerolling one dice, using the new lower number \
       if INTIMACYHIGH >= REROLL { result: INTIMACYHIGH }
       else { result: REROLL }
  }
  
  else { result: INTIMACYLOW }      \ both dice lower than 4 no rerolls \

  
}


output [rolling 2d10 if lower than 4 reroll 1d10]

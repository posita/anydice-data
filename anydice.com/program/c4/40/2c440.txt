function: rolling ROLL:s if lower than MIN:n reroll REROLL:s or REROLLBOTH:s {

  INTIMACYLOW: 2@ROLL
  INTIMACYHIGH: 1@ROLL
  
  REROLLLOW: 2@REROLLBOTH

  if INTIMACYLOW >= MIN { result: INTIMACYLOW }  \ both results 4 or more no rerolls \

  if INTIMACYHIGH < MIN { result: REROLLLOW }    \ both results lower than 4 rerolling both \

  if INTIMACYHIGH >= MIN { 
       if INTIMACYHIGH >= REROLL { result: INTIMACYHIGH }  \ rerolling lower result and choosing lower of the two \
       else { result: REROLL }
  }

  
}


output [rolling 2d10 if lower than 4 reroll 1d10 or 2d10]

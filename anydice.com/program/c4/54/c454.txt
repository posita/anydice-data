function: blades A:n {
  if A >= 6 {result: 1}
  else if A >= 4 {result: 0}
  else {result: -1}
}
function: shards A:n {
  if A >= 10 {result: 1}
  else if A >= 8 {result: 0}
  else {result: -1}
}

function: fringe A:n {
  if A >= 2 {result: 2}
  else if A >= 1 {result: 1}
  else {result: 0}
}

function: dominant S:n D:n C:n {
 SDB: S >= D
 SCB: S >= C
 SCOMB: SDB & SCB
 CDB: C >= D
 if (SCOMB) {result: 1}
 else if (CDB) {result: -1}
 else {result: 0}
}

function: high NUMBER:n of DICE:s {
 result: {1..NUMBER}@DICE
}

function: crisis skill S:s diss D:s diff C:s {
 SO: [count {1} in S]
 DO: [count {1} in D]
 CO: [count {1} in C]
 SUCCESS: SO + DO - CO
 HARMONY: [high 1 of S]
 DISSONANCE: [high 1 of D]
 CRISIS: [high 1 of C]
 DOMINANT: [dominant HARMONY DISSONANCE CRISIS]
 if (SUCCESS >= 2) {
  if (DOMINANT > 0) {result: 202}
  else if (DOMINANT < 0) {result: 201}
  else {result: 200}
 }
 else if (SUCCESS = 1) {
  if (DOMINANT > 0) {result: 12}
  else if (DOMINANT < 0) {result: 11}
  else {result: 10}
 }
 else {
  if (DOMINANT > 0) {result: 2}
  else if (DOMINANT < 0) {result: 1}
  else {result: 0}
 }
}

function: opposed skill S:s crisis C:s {
 SO: [count {6} in S]
 CO: [count {6} in C]
 SUCCESS: SO - CO
 if SUCCESS >= 2 {result: 2}
 else if SUCCESS = 1 {result: 1}
 else {result: 0}
}

function: posten TN:n R:s {
 COUNT: 0
 if R >= 10 {result: 2}
 else if R >= TN {result: 1}
}
function: ct VALUES:s in SEQUENCE:s max M:n{
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
 }
 COUNT: COUNT + [count {M} in SEQUENCE]
 result: COUNT
}

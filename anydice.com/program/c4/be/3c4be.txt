function: attack HIT:n with DAMAGE:n vs TOHIT:n {
 if HIT > TOHIT { result: DAMAGE }
 result: 0
}


\ Monster Setup (Elder Dragon) \
MONARMOUR: 12
TOHITMON: 15

\ Player Setup (level 4, level up: X, X, Attack, shop: 3x Attack) \
BONUSTOHIT: 6
BONUSDAMAGE: 9

\ Knight Attack \
BASEDAMAGE: 1d12+(1d12)/2+10 + BONUSDAMAGE - MONARMOUR
BASEHIT: 1d20 + 3 + BONUSTOHIT
output [attack BASEHIT with BASEDAMAGE vs TOHITMON] named "Knight"

\ Wizard Attack \
BASEDAMAGE: (1d20+6 + BONUSDAMAGE)*3/2 - MONARMOUR/3
BASEHIT: 1d20 + 0 + BONUSTOHIT
output [attack BASEHIT with BASEDAMAGE vs TOHITMON] named "Wizard"

\ Rogue Attack \
BASEDAMAGE: 2d(1d12+5 + BONUSDAMAGE) - MONARMOUR
BASEHIT: 1d20 + 0 + BONUSTOHIT
output [attack BASEHIT with BASEDAMAGE vs TOHITMON] named "Rogue"

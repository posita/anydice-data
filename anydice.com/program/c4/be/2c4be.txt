function:  strS tT svSAVE  apAP shock SHOCK shred SHRED wound TO_WOUND:s crit CRIT:n saveroll TO_SAVE:s {

 if SHRED>0 & (CRIT=6) {AP:AP-1}
 if TO_SAVE >= (SAVE-AP) {result: 0}
 if SHOCK>0 & (CRIT=6)  {result: [count {1..6} in TO_WOUND]}

    if S >= T*2  {
        result: [count {2..6} in TO_WOUND] 
    }
    if S > T {
        result: [count {3..6} in TO_WOUND]
    }
    if S = T {
        result: [count {4..6} in TO_WOUND]
    }
    if S <= T/2 {
        result: [count {6} in TO_WOUND]
    }
    if S < T {
        result: [count {5..6} in TO_WOUND]
    }

}

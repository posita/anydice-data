function: advantage DIE:n SIZE:n{
   result: [highest of DIEdSIZE and DIEdSIZE]
}

function: attack ATTACK:n against AC:n for DAMAGE:n{
   if ATTACK >= AC{
     if ATTACK = 20{
       result: 2*DAMAGE
     }
     else{
       result: DAMAGE
     }
   }
   else{
     result: 0
   }
}

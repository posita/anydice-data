function: attack DIE:n plus ATK ac AC for DMG crit CRIT {
 if (DIE = 20) {result: DMG + CRIT}
 if (DIE = 1) {result: 0}
 if (DIE + ATK >= AC) {result: DMG}
 else {result: 0}
}

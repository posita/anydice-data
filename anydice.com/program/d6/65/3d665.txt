 loop P over {-3..-1}{
  output 2d10+A+P named "2d10+[A]A+[P]P"
                     }
          }

 loop P over {-2..0}{
  output 2d10+A+P named "2d10+[A]A+[P]P"
                     }
          }

 loop P over {-1..0}{
  output 2d10+A+P named "2d10+[A]A+[P]P"
                     }
          }

 loop P over {0}{
  output 2d10+A+P named "2d10+[A]A+[P]P"
                     }
          }

 loop P over {0..1}{
  output 2d10+A+P named "2d10+[A]A+[P]P"
                     }
          }

 loop P over {0..2}{
  output 2d10+A+P named "2d10+[A]A+[P]P"
                     }
          }

  output 2d10+A+P named "2d10+[A]A+[P]P"
                     }
          }
}

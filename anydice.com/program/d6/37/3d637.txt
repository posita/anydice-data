function: attack ROLL:n vs AC:n damage DMG:n reliable R:n {
  if (ROLL >= AC) {
    result: [highest of R and DMG]
  } else {
    result: R
  }
}

function: attack initially {
  result: [attack 1d20+1+1d6 vs 10 damage 1 reliable 0]
}

function: attack again ATKONE:n {
  if (ATKONE > 0) {
    result: 4
  } else {
    result: [attack initially]
  }
}

ATKONE: [attack initially]
ATKTWO: [attack again ATKONE]
ATKTHREE: [attack again ATKONE+ATKTWO]

\
 Gild, Erfolgswahrscheinlichkeiten
 0: Misserfolg
 1: Erfolg, aber mit Konsequenzen
 2: Erfolg
 3: Kritischer Erfolg
\

function: blades ROLL:s {
 if 1@ROLL = 7 {
  result: 2 + (2@ROLL = 7)
 }
 result: 1@ROLL >= 4
}

output [blades 2@2d7] named "0d"
loop P over {1..7} {
 output [blades Pd7] named "[P]d"
}

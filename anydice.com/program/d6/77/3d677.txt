CARAC: 5
POOL: 10
THRESHOLD: 1 + CARAC  

function: issuccess N:n threshold THRESHOLD:n {
  if N = 1 { result: 2 }
  if N <= THRESHOLD { result: 1 }
  result: 0
}

function: countsuccesses ROLL:s threshold THRESHOLD:n {
  COUNT: 0
  loop I over ROLL {
    COUNT: COUNT + [issuccess I threshold THRESHOLD]
  }
  result: COUNT
}

output [countsuccesses POOL d10 threshold THRESHOLD]

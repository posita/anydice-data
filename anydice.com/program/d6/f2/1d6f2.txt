function: roll ROLL:n BONUS PARTIAL:n FULL:n CRIT:n {
 if ROLL = 1 {
  result: 0
 }
 if ROLL = CRIT { 
  result: 3
 }
 if ROLL+BONUS >= FULL {
  result: 2
 }
 if ROLL+BONUS >= PARTIAL {
  result: 1
 }
 result: 0
}

BONUS: 8
output [roll 1d20 BONUS 11 18 20] named "D20"

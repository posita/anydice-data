function: with att roll one ATROLLA:n att roll two ATROLLB:n ac AC:n tohitbonus THB:n {
  BASEDAM: 7
  SATNODICE: 2
  TOHIT: 6
  HMDICE: 1

  TOTAL: 0
  SAT: 1

  if ATROLLA = 20 {
    TOTAL: TOTAL + 2d8 + (2*SATNODICE)d6 + (2*HMDICE)d6 + BASEDAM
    SAT: 0
  } else {
    if (ATROLLA + TOHIT + THB) >= AC {
      TOTAL: TOTAL + 1d8 + SATNODICEd6 + HMDICEd6 + BASEDAM
      SAT: 0
    }
  }

  if ATROLLB = 20 {
    TOTAL: TOTAL + 2d8 + (2*HMDICE)d6 + BASEDAM
    if SAT = 1 {
      TOTAL: TOTAL + (2*SATNODICE)d6
    }
  } else {
    if (ATROLLB + TOHIT + THB) >= AC {
      TOTAL: TOTAL + 1d8 + HMDICEd6 + BASEDAM
      if SAT = 1 {
        TOTAL: TOTAL + SATNODICEd6
      }
    }
  }

  result: TOTAL
}

function: dis DICEONE:n and DICETWO:n {
  if DICEONE > DICETWO {
    result: DICETWO
  }
  result: DICEONE
}

function: adv DICEONE:n and DICETWO:n {
  if DICEONE < DICETWO {
    result: DICETWO
  }
  result: DICEONE
}

MISS:0
HIT:1
CRITICALHIT:2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

DEFENSE: d{10 .. 20}

ATTACK: 7
HITDAMAGE: 4d10 + 1
CRITICALDAMAGE: 8d10 + 1
MISSDAMAGE: 0
output 6d[attack] named "Improved Cannon Volley"

ATTACK: 6
HITDAMAGE: 4d10
CRITICALDAMAGE: 8d10
output 6d[attack] named "Cannon Volley"

function: doexplode TRY:n Y:n {
    if TRY = Y {
        result: TRY - 1 + [fixedexplode Y]
    }
    result: TRY
}

function: fixedexplode Y:n {
    TRY: 1dY
    result:[doexplode TRY Y]
}

function: deadlands X:n of Y:n {
    HIGH: 0
    loop I over {1..X}{
        TRY: [fixedexplode Y]
        HIGH: [highest of HIGH and TRY]
    }
    result:[lowest of 25 and HIGH]
}

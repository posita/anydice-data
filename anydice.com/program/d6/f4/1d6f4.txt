function: roll ROLL:n BONUS:n PARTIAL:n FULL:n CRITROLL:n CRIT:n {
 if ROLL = 1 {
  result: 0
 }
 if ROLL = CRITROLL | ROLL+BONUS>=CRIT { 
  result: 3
 }
 if ROLL+BONUS >= FULL {
  result: 2
 }
 if ROLL+BONUS >= PARTIAL {
  result: 1
 }
 result: 0
}

loop P over {-1..15} {
 output [roll 1d20 P 00 18 20 25] named "D20+[P]"
}

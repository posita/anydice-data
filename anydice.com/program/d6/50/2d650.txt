\**** Damage calculator ****\

function: attackroll ROLL:n vs REF:n {
  if ROLL + ATTACKMOD + BASEDMG - REF - DR <= 0 { result: 0 }
  if ROLL + ATTACKMOD < REF { result: 0 }
  if ROLL + ATTACKMOD >= REF { result: ROLL + ATTACKMOD + BASEDMG - REF - DR }
}

DZERO: d{-6, -5, -4, -3, -2, -1, 1, 2, 3, 4, 5, 6}
REF: 0
ATTACKMOD: 0
BASEDMG: 10
DR: 0
output [attackroll DZERO vs REF]

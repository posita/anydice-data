function: reroll R:n under N:n {
   if R < N { result: 1d6 } else {result: R}
}

function: sumrolls NDICE:n under MAX:n {
 X: 0  
  loop N over {1..NDICE}{
    X: X + [reroll 1d6 under MAX]
  }
  result: X
}

output 8d6 named "normal"
output [sumrolls 8 under 2] named "reroll 1s"
output [sumrolls 8 under 3] named "reroll 1s/2s"
output [sumrolls 8 under 4] named "reroll 1s/2s/3s"
output [sumrolls 8 under 5] named "reroll 1s/2s/3s/4s"

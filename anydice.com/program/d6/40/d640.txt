function: docpr A:n {
   HELP : A
   CPR : 0
   if HELP > 0 {
     result : HELP 
   }
   if HELP < -1 {
     result : -2
   }
   if HELP = 0{
     result : 0
   }
   if HELP = -1 {
     result : 0
   }
}

function: doritual A:n {
   HELP : A
   if HELP > 0 {
     result: HELP 
   }
   else{
     result: 0
   }
}

function: doelec A:n {
   HELP : A
   if HELP > 0 {
     result: HELP 
   }
   if HELP < -1 {
     result: -1d6 
   }
   if HELP = 0{
     result : 0
   }
   if HELP = -1 {
     result : 0
   }

}



HEAL: (1d6+5)/2
CPR: [docpr 2d6+5-10]
RITUAL: [doritual 2d6+9-13] 
ELEC: [doelec 2d6+11-17]


function: saved A:n {
    if A > 8{
     result : 1
    }
    if A < 9 {
     result : 0
    }
}

RESULT : 0
loop N over {1..10}{
   RESULT : RESULT + [saved (HEAL + CPR + RITUAL + ELEC)]
}

output RESULT

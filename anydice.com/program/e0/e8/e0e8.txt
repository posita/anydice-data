\ Attack Dice \
B: d{-1,2,2,2,1,1}
R: d{1,2,2,2,3,3}
Y: d{0,1,1,1,2,2}
G: d{1,0,0,1,1,1}

MISS: -1
function: attack B_ROLL:n and POWER:n {
 if B_ROLL = -1 { 
  result: MISS 
 } else {
  result: B_ROLL + POWER
 }
}

\ Attack Combinations \
\ 2 dice \
output [attack B and R] named "1 Blue + 1 Red"
output [attack B and Y] named "1 Blue + 1 Yellow"
output [attack B and G] named "1 Blue + 1 Green"

\ 3 dice \
output [attack B and R + R] named "1 Blue + 2 Red"
output [attack B and R + Y] named "1 Blue + 1 Red + 1 Yellow"
output [attack B and R + G] named "1 Blue + 1 Red + 1 Green"
output [attack B and Y + Y] named "1 Blue + 2 Yellow"
output [attack B and Y + G] named "1 Blue + 1 Yellow + 1 Green"
output [attack B and G + G] named "1 Blue + 2 Green"

\ 4 dice \
output [attack B and R + G + G] named "1 Blue + 1 Red + 2 Green"
output [attack B and Y + G + G] named "1 Blue + 1 Yellow + 2 Green"
output [attack B and R + R + G] named "1 Blue + 2 Red + 1 Green"
output [attack B and Y + Y + G] named "1 Blue + 2 Yellow + 1 Green"
output R + R + G + G named "2 Red + 2 Green"

\ 5 dice \
output [attack B and Y + G + G + G] named "1 Blue + 1 Yellow + 3 Green"
output [attack B and Y + Y + G + G] named "1 Blue + 2 Yellow + 2 Green"
output R + R + G + G + G named "2 Red + 3 Green"

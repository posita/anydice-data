function: attack DIE:n plus ATK ac AC for DMG crit CRIT {
 if (DIE = 20) {result: DMG + CRIT}
if (DIE = 19) {result: DMG + CRIT}
 if (DIE = 1) {result: 0}
 if (DIE + ATK >= AC) {result: DMG}
 else {result: 0}
}

CHAA: 3
PROFA: 2
CHAASI: 4
PROFB: 3
GWM: -5

\--
output 2d10
--\

output [attack 1d20 plus CHAASI+PROFB ac 15 for 3d6+CHAASI crit 3d6] named "level 5 gs with hex w/asi against 15ac"
output [attack 1d20 plus CHAA+PROFB+GWM ac 15 for 3d6+CHAA+10 crit 5d6] named "level 5 gs with hex w/ gwm against 15ac"

output [attack 1d20 plus CHAASI+PROFB ac 15 for 2d6+CHAASI+PROFB crit 2d6] named "level 5 gs with hexblades w/asi against 15ac"
output [attack 1d20 plus CHAA+PROFB+GWM ac 15 for 2d6+CHAA+PROFB+10 crit 4d6] named "level 5 gs with hexblades w/ gwm against 15ac"

output [attack 1d20 plus CHAASI+PROFB ac 15 for 3d6+CHAASI+PROFB crit 3d6] named "level 5 gs with hex and hexblades w/asi against 15ac"

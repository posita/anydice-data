\ Attack Dice \
B: d{-1,2,2,2,1,1}
R: d{1,2,2,2,3,3}
Y: d{0,1,1,1,2,2}
G: d{1,0,0,1,1,1}

\ Range Dice \
BR: d{-1,2,3,4,5,6}
RR: d{0,0,0,0,0,0}
YR: d{1,1,2,0,0,0}
GR: d{0,0,1,1,0,1}

\ Surge Dice \
BS: d{-1,1,0,0,0,1}
RS: d{0,0,0,0,0,1}
YS: d{1,0,0,1,0,1}
GS: d{0,1,1,0,1,1}

MISS: -1
function: attack B_ROLL:n and POWER:n {
 if B_ROLL = -1 { 
  result: MISS 
 } else {
  result: B_ROLL + POWER
 }
}

\ Surge Combinations \
\ 2 dice \
output [attack BS and RS] named "1 Blue + 1 Red --- Surge"
output [attack BS and YS] named "1 Blue + 1 Yellow --- Surge"
output [attack BS and GS] named "1 Blue + 1 Green --- Surge"

\ 3 dice \
output [attack BS and RS + RS] named "1 Blue + 2 Red --- Surge"
output [attack BS and RS + YS] named "1 Blue + 1 Red + 1 Yellow --- Surge"
output [attack BS and RS + GS] named "1 Blue + 1 Red + 1 Green --- Surge"
output [attack BS and YS + YS] named "1 Blue + 2 Yellow --- Surge"
output [attack BS and YS + GS] named "1 Blue + 1 Yellow + 1 Green --- Surge"
output [attack BS and GS + GS] named "1 Blue + 2 Green --- Surge"

\ 4 dice \
output [attack BS and RS + GS + GS] named "1 Blue + 1 Red + 2 Green --- Surge"
output [attack BS and YS + GS + GS] named "1 Blue + 1 Yellow + 2 Green --- Surge"
output [attack BS and RS + RS + GS] named "1 Blue + 2 Red + 1 Green --- Surge"
output [attack BS and YS + YS + GS] named "1 Blue + 2 Yellow + 1 Green --- Surge"
output RS + RS + GS + GS named "2 Red + 2 Green --- Surge"

\ 5 dice \
output [attack BS and YS + GS + GS + GS] named "1 Blue + 1 Yellow + 3 Green --- Surge"
output [attack BS and YS + YS + GS + GS] named "1 Blue + 2 Yellow + 2 Green --- Surge"
output RS + RS + GS + GS + GS named "2 Red + 3 Green --- Surge"

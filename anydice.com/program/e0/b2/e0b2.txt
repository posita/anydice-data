DICE: 30
OFFSET: 15
CUTOFF: 15 \ show clash outcome chances from -cutoff to +cutoff \
SELF_AUTOS: 6
ENEMY_AUTOS: 6

CD: d10>6
function: clash vs N:n {
    TMP: (SELF_AUTOS+(DICE dCD))-(ENEMY_AUTOS+(N dCD))
    result: [highest of [lowest of TMP and CUTOFF] and -CUTOFF]
}

loop N over {DICE-OFFSET..DICE+OFFSET} {
\    output (SELF_AUTOS+DICE d CD) > (ENEMY_AUTOS+N dCD) named "win [DICE]d vs [N]d"\
    output [clash vs N]>0 named "clash [DICE]d vs [N]d"
}

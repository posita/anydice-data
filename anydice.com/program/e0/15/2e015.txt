function: cards DRAW:s {
 result: 0 + (1@DRAW>=10) + (2@DRAW>=10)
}

function: onecard ONE:n and TWO:n {
 if ONE < 10 { 
  result: 0
 }
 if ONE >=10 {
  result: 0 + (ONE>=10) + (TWO>=10)
 }
}

output [onecard 1d13 and 1d13] named "1 Card"

loop P over {2..6} {
 output [cards Pd13] named "[P] Cards"
}

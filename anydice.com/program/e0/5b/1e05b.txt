function: images IMAGES:n imageac DEXAC:d imageroll IROLL:n roll ROLL:n plus BONUS:d vs AC:d for DMG:d crit CRIT:d on CRIT_RANGE:d use USE:d pass PASS:d {
   if IMAGES = 3 {
        if IROLL >= 6{
			if ROLL = 1{ result: PASS }
			if ROLL + BONUS >= DEXAC { result: USE }
			if ROLL + BONUS < DEXAC { results: PASS }		
		}
		if IROLL < 6{
			if ROLL = 1{ result: PASS  }
			if ROLL >= CRIT_RANGE { result: DMG + CRIT_DMG + PASS }
			if ROLL + BONUS >= AC { result: DMG + PASS}
			if ROLL + BONUS < AC { result: PASS }
		}
    }
    if IMAGES = 2 {
        if IROLL >= 8{
			if ROLL = 1{ result: PASS  }
			if ROLL + BONUS >= DEXAC { result: USE }
			if ROLL + BONUS < DEXAC { results: PASS }		
		}
		if IROLL < 11{
			if ROLL = 1{ result: PASS  }
			if ROLL >= CRIT_RANGE { result: DMG + CRIT_DMG + PASS }
			if ROLL + BONUS >= AC { result: DMG + PASS}
			if ROLL + BONUS < AC { result: PASS }	
		}
	
    }
    if IMAGES = 1 {
        if IROLL >= 11{
		    if ROLL = 1{ result: PASS  }
			if ROLL + BONUS >= DEXAC { result: USE }
			if ROLL + BONUS < DEXAC { results: PASS }
		}
		if IROLL < 11{
			if ROLL = 1{ result: PASS  }
			if ROLL >= CRIT_RANGE { result: DMG + CRIT_DMG + PASS }
			if ROLL + BONUS >= AC { result: DMG + PASS}
			if ROLL + BONUS < AC { result: PASS }
		}
    }
    if IMAGES = 0 {
        if ROLL = 1{ result: PASS  }
		if ROLL >= CRIT_RANGE { result: DMG + CRIT_DMG + PASS }
		if ROLL + BONUS >= AC { result: DMG + PASS}
		if ROLL + BONUS < AC { result: PASS }
    }
}

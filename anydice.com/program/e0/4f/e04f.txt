DV: 7
SELF_AUTOS: 6
ENEMY_AUTOS: 6

CD: d10>6
function: clash with N:n bonus B:n dvup D:d against DEF {
    TMP: [lowest of (SELF_AUTOS+(N dCD))-(ENEMY_AUTOS+(DEF dCD)) and DV+D]
    result: [abovezero TMP bonus B]
}
function: abovezero N:n bonus B:n {
    if N>0 {result: N+B}
    result: 0
}

loop D over {20,25,27,30,35,40} {
   MS: [clash with 29 bonus 1 dvup 0 against D]
   output 4dMS named "MS 29 vs [D]"
   FSS: [clash with 40 bonus 5 dvup 2 against D]
   output 1dFSS named "FSS 41 vs [D]"
}

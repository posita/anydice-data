BONUSA: 2
BONUSB: 10
BONUSC: 18

DCX: 15
DCY: 25
DCZ: 35

function: pf method success with bonus BONUS against dc DC {
 R: d20
 SUCCESS: [pf method success for roll R with bonus BONUS and dc DC]
 result: SUCCESS
}

function: pf method success for roll R:n with bonus BONUS and dc DC {
  ATTACK: BONUS + R
  if R = 1 {
    result: 0
  }
  else {
    if 20 = R {
      result: 1
    }
    else {
      if ATTACK >= DC {
        result: 1
      }
      else {
        result: 0
      }
    }
  }
}

PF_A_X: [pf method success with bonus BONUSA against dc DCX]
PF_A_Y: [pf method success with bonus BONUSA against dc DCY]
PF_A_Z: [pf method success with bonus BONUSA against dc DCZ]
PF_B_X: [pf method success with bonus BONUSB against dc DCX]
PF_B_Y: [pf method success with bonus BONUSB against dc DCY]
PF_B_Z: [pf method success with bonus BONUSB against dc DCZ]
PF_C_X: [pf method success with bonus BONUSC against dc DCX]
PF_C_Y: [pf method success with bonus BONUSC against dc DCY]
PF_C_Z: [pf method success with bonus BONUSC against dc DCZ]

output PF_A_X named "PF method with bonus [BONUSA] against DC [DCX]"
output PF_B_X named "PF method with bonus [BONUSB] against DC [DCX]"
output PF_C_X named "PF method with bonus [BONUSC] against DC [DCX]"
output PF_A_Y named "PF method with bonus [BONUSA] against DC [DCY]"
output PF_B_Y named "PF method with bonus [BONUSB] against DC [DCY]"

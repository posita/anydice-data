\ Attack Dice \
B: d{-1,2,2,2,1,1}
R: d{1,2,2,2,3,3}
Y: d{0,1,1,1,2,2}
G: d{1,0,0,1,1,1}

\ Range Dice \
BR: d{-1,2,3,4,5,6}
RR: d{0,0,0,0,0,0}
YR: d{1,1,2,0,0,0}
GR: d{0,0,1,1,0,1}

\ Surge Dice \
BS: d{-1,1,0,0,0,1}
RS: d{0,0,0,0,0,1}
YS: d{1,0,0,1,0,1}
GS: d{0,1,1,0,1,1}

MISS: -1
function: attack B_ROLL:n and POWER:n {
 if B_ROLL = -1 { 
  result: MISS 
 } else {
  result: B_ROLL + POWER
 }
}

\ Range Combinations \
\ 2 dice \
output [attack BR and RR] named "1 Blue + 1 Red --- Range"
output [attack BR and YR] named "1 Blue + 1 Yellow --- Range"
output [attack BR and GR] named "1 Blue + 1 Green --- Range"

\ 3 dice \
output [attack BR and RR + RR] named "1 Blue + 2 Red --- Range"
output [attack BR and RR + YR] named "1 Blue + 1 Red + 1 Yellow --- Range"
output [attack BR and RR + GR] named "1 Blue + 1 Red + 1 Green --- Range"
output [attack BR and YR + YR] named "1 Blue + 2 Yellow --- Range"
output [attack BR and YR + GR] named "1 Blue + 1 Yellow + 1 Green --- Range"
output [attack BR and GR + GR] named "1 Blue + 2 Green --- Range"

\ 4 dice \
output [attack BR and RR + GR + GR] named "1 Blue + 1 Red + 2 Green --- Range"
output [attack BR and YR + GR + GR] named "1 Blue + 1 Yellow + 2 Green --- Range"
output [attack BR and RR + RR + GR] named "1 Blue + 2 Red + 1 Green --- Range"
output [attack BR and YR + YR + GR] named "1 Blue + 2 Yellow + 1 Green --- Range"
output RR + RR + GR + GR named "2 Red + 2 Green --- Range"

\ 5 dice \
output [attack BR and YR + GR + GR + GR] named "1 Blue + 1 Yellow + 3 Green --- Range"
output [attack BR and YR + YR + GR + GR] named "1 Blue + 2 Yellow + 2 Green --- Range"
output RR + RR + GR + GR + GR named "2 Red + 3 Green --- Range"

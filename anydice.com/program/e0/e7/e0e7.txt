\ Attack Dice \
B: d{-1,2,2,2,1,1}
R: d{1,2,2,2,3,3}
Y: d{0,1,1,1,2,2}
G: d{1,0,0,1,1,1}

\ Range Dice \
BR: d{-1,2,3,4,5,6}
RR: d{0,0,0,0,0,0}
YR: d{1,1,2,0,0,0}
GR: d{0,0,1,1,0,1}

\ Surge Dice \
BS: d{-1,1,0,0,0,1}
RS: d{0,0,0,0,0,1}
YS: d{1,0,0,1,0,1}
GS: d{0,1,1,0,1,1}

MISS: -1
function: attack B_ROLL:n and POWER:n {
 if B_ROLL = -1 { 
  result: MISS 
 } else {
  result: B_ROLL + POWER
 }
}

\ Attack Combinations \
\ 2 dice \
output [attack B and R] named "1 Blue + 1 Red"
output [attack B and Y] named "1 Blue + 1 Yellow"
output [attack B and G] named "1 Blue + 1 Green"

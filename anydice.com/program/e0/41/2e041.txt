ONES: {1:3, 2, 0:8}
TWOS: {1:2, 0:10}

function: map A:n to X Y Z {
 if A < 0 {result: X}
 if A = 0 {result: Y}
 result: Z
}

loop N over {1..6} {
 L: [map [lowest 1 of 2 d 12] - [lowest 1 of N d 12] to 1 1 3]
 O: [map N d ONES - 2 d ONES to 1 2 3]
 T: [map N d TWOS - 2 d TWOS to 1 2 3]

 output L * 100 + O * 10 + T
}

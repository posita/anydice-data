function: is roll R:n a natural twenty {
 if R = 20 {
  result: 1
 }
 else {
  result: 0
 }
}

function: pathfinder method success with bonus BONUS against dc DC {
 R: d20
 result: ((R + BONUS) >= DC | [is roll R a natural twenty]) & R > 1
}

BONUSA: 5
BONUSB: 15
BONUSC: 25
DCX: 15
DCY: 55
DCZ: 95
METHOD_PF_A_X: [pathfinder method success with bonus BONUSA against dc DCX]
METHOD_PF_B_X: [pathfinder method success with bonus BONUSB against dc DCX]
METHOD_PF_C_X: [pathfinder method success with bonus BONUSC against dc DCX]
METHOD_PF_A_Y: [pathfinder method success with bonus BONUSA against dc DCY]
METHOD_PF_B_Y: [pathfinder method success with bonus BONUSB against dc DCY]
METHOD_PF_C_Y: [pathfinder method success with bonus BONUSC against dc DCY]
METHOD_PF_A_Z: [pathfinder method success with bonus BONUSA against dc DCZ]
METHOD_PF_B_Z: [pathfinder method success with bonus BONUSB against dc DCZ]
METHOD_PF_C_Z: [pathfinder method success with bonus BONUSC against dc DCZ]

output METHOD_PF_A_X named "Method: PF, Bonus: [BONUSA], DC: [DCX]"
output METHOD_PF_B_X named "Method: PF, Bonus: [BONUSB], DC: [DCX]"
output METHOD_PF_C_X named "Method: PF, Bonus: [BONUSC], DC: [DCX]"
output METHOD_PF_A_Y named "Method: PF, Bonus: [BONUSA], DC: [DCY]"

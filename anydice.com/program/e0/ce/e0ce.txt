function: compter X:n dans N:n{
A:0
loop I over {1..N}{
if N-X>=0{
N:N-X
A:A+1}}
result:A}

function:Luck SUCCES:n ECHECS:n{
if SUCCES>ECHECS{A:1}
else{A:0}
result:A}

function: bonus C:n ECHECS:n MISS:n MISSED:n{
if C=0{A:-ECHECS}

if C>0 & C<=(ECHECS+MISS){A:C-ECHECS}

if C> ECHECS+MISS & C<=ECHECS+MISS+MISSED+MISSED+ECHECS{BONUS:ECHECS+MISS

A:BONUS-ECHECS+[lowest of [compter 2 dans (C-BONUS)]and (MISSED+ECHECS)]}
if C>ECHECS+ECHECS+MISS+MISSED+MISSED{
BONUS:ECHECS+MISS+MISSED+MISSED+ECHECS
A:BONUS-MISSED-ECHECS+[compter 4 dans (C-BONUS)]}
result:A}

function: concentration D:n competence C:n chance L:n seuil S:n marge M:n{
POOL:(D+M)d6
SUCCES:[count {S..6} in POOL]
MISS:[count {S-1} in POOL]
ECHECS:[count {1} in POOL]
DOUBLE:[count {6} in POOL]
BIS:[count {S+1} in POOL]
MISSED:[count {S-2} in POOL]
result:SUCCES+DOUBLE+[bonus C ECHECS MISS+BIS MISSED]+L*[Luck SUCCES ECHECS]
}

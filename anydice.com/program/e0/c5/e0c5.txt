function: compter X:n dans N:n{
A:0
loop I over {1..N}{
if N-X>=0{
N:N-X
A:A+1}}
result:A}

function: bonus C:n ECHECS:n MISS:n MISSED:n{
if C=0{A:-ECHECS}
if C>0 & C<=(ECHECS+MISS){A:C-ECHECS}
if C> ECHECS+MISS & C<=ECHECS+MISS+MISSED*2{BONUS:ECHECS+MISS
A:BONUS-ECHECS+[lowest of [compter 2 dans (C-BONUS)]and MISSED]}
if C>ECHECS+MISS+MISSED*2{
BONUS:ECHECS+MISS+MISSED*2
A:BONUS-MISSED-ECHECS+[compter 4 dans (C-BONUS)]}
result:A}

function: concentration D:n competence C:n chance L:n seuil S:n{
POOL:(D+L)d6
SUCCES:[count {S..6} in POOL]
DOUBLE:[count {6} in POOL]
ECHECS:[count {1} in POOL]
MISS:[count {S-1} in POOL]
BIS:[count {S+1} in POOL]
MISSED:[count {S-2} in POOL]
result:SUCCES+DOUBLE+[bonus C ECHECS MISS+BIS MISSED]
}
output [concentration 1 competence 0 chance 0 seuil 4] named "comp 0"
output [concentration 1 competence 1 chance 0 seuil 4] named "comp 1"

output [concentration 1 competence 5 chance 0 seuil 4] named "comp 5"

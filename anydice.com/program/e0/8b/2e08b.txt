function: attack ATTACK:n hit HIT:n {
    result: [count {HIT..6} in ATTACKd6]
}

function: hits HITS:n wound WOUND:n {
    result: [count {WOUND..6} in HITSd6]
}

function: wounds WOUNDS:n vs SAVE:n rend REND:n {
    if SAVE = 6 {
        result: (WOUNDS -[count 6+[absolute REND] in WOUNDSd6])
    }
    result: (WOUNDS - [count {SAVE+[absolute REND]..6+[absolute REND]} in WOUNDSd6])
}

function: success SUCCESS:n for DAMAGE:n by MODELS:n {
    result: SUCCESS * DAMAGE * MODELS
}

function: attack ATTACK:n hit HIT:n wound WOUND:n vs SAVE:n rend REND:n for DAMAGE:n by MODELS:n {
    result: [success [wounds [hits [attack ATTACK hit HIT] wound WOUND] vs SAVE rend REND] for DAMAGE by MODELS]
}

function: damage {
    result: [attack ATTACK hit HIT wound WOUND vs SAVE rend REND for DAMAGE by MODELS]
}

function: hit {
    result: [wounds [hits [attack ATTACK hit HIT] wound WOUND] vs SAVE rend REND]
}

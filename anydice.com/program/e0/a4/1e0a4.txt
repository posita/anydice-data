function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: images IMAGES:n imageac DEXAC:n imageroll IROLL:n roll ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n use USE:d pass PASS:d {
    TARGET_IMAGE: 0
    if IMAGES = 3 {if IROLL >= 6  { TARGET_IMAGE: 1 }}
    if IMAGES = 2 {if IROLL >= 8  { TARGET_IMAGE: 1 }}
    if IMAGES = 1 {if IROLL >= 11 { TARGET_IMAGE: 1 }}
    if TARGET_IMAGE = 1 {
	if ROLL = 1 { result: PASS }
        if ROLL >= CRIT_RANGE { result: USE }
	if ROLL + BONUS >= DEXAC { result: USE }
	if ROLL + BONUS < DEXAC { result: PASS }
    }
    else {
        if ROLL = 1{ result: PASS  }
	if ROLL >= CRIT_RANGE { result: DMG + CRIT_DMG + PASS }
	if ROLL + BONUS >= AC { result: DMG + PASS}
	if ROLL + BONUS < AC { result: PASS }
    }
}

function: images IMAGES:n imageac DEXAC:d attacks ATTACKS:n roll ROLL:d plus BONUS:d vs AC:d for DMG:d crit CRIT:d on CRIT_RANGE:d {

    A: 0
    B: 0
    C: 0
    D: 0

    ATTACK: [attack ROLL plus BONUS vs AC for DMG crit CRIT on CRIT_RANGE]

    loop N over {1 .. ATTACKS}{
        D: [images 3 imageac DEXAC imageroll d20 roll ROLL plus BONUS vs AC for DMG crit CRIT on CRIT_RANGE use C pass D] 
        C: [images 2 imageac DEXAC imageroll d20 roll ROLL plus BONUS vs AC for DMG crit CRIT on CRIT_RANGE use B pass C] 
        B: [images 1 imageac DEXAC imageroll d20 roll ROLL plus BONUS vs AC for DMG crit CRIT on CRIT_RANGE use A pass B] 
        A: ATTACK + A
    }

    if IMAGES = 3 {
        result: D
    }
    if IMAGES = 2 {
        result: C
    }
    if IMAGES = 1 {
        result: B
    }
    if IMAGES = 0 {
        result: A
    }
}

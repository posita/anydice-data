function: add X:d highest to Y:d lowest {
result: [highest 1 of X] + [lowest 1 of Y]
}

function: test X:d against Y:d {
result: [add X highest to Y lowest] > 6
}

function: highest of X:d at least highest of Y:d {
result: 1@X >= 1@Y
}

loop P over {1..5} {
loop M over {1..5} {

output 1@Md6 named "highest mod die [M]d6"
output [add Pd6 highest to Md6 lowest] named "sum [P]d6 + [M]d6"
output [highest of Md6 at least highest of Pd6] named "Knockout prob with [P]d6 against [M]d6"
output 1@Pd6 > 1@Md6
output [test Pd6 against Md6] named "Player [P]d6 wins against Moderator [M]d6"
}

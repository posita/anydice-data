function: roll DIE:s {
  S: 0
  loop I over {1..6}{
    S: S + I@DIE
  }
  if S < 1 {result: 0}
  if S < 6 {result: 1}
  else {result: 6}
}

output [roll 1d{1,2,3,0,0,-1}]
output [roll 2d{1,2,3,0,0,-1}]
output [roll 3d{1,2,3,0,0,-1}]
output [roll 4d{1,2,3,0,0,-1}]
output [roll 5d{1,2,3,0,0,-1}]
output [roll 6d{1,2,3,0,0,-1}]

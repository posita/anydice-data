set "position order" to "lowest first"

function: day ROLL:s FLIP:s {
 if 1@ROLL < 3 {
  result: -1
 }
 if 1@ROLL = 3 {
  result: -2
 }
 if 1@ROLL = 6 {
  result: 1
 }
 else {
  if 1@FLIP = 1 {
   result: 1
  }
  else {
   result: 0
  }
 }
}

function: runthrough ROLLTYPE:d {
 CTR: CTR_START
 loop N over {1..CTR_START} {
  CTR: CTR + ROLLTYPE
 }
 result: CTR
}

CTR_START: 2+3+4+5+6+7-4

output [day 1d6 1d2] named "single roll"

output [day 2d6 1d2] named "good roll"

output [day 2d6 1d2] named "ridiculous roll"

output [runthrough [day 1d6 1d2]] - 6 named "no help"

output [runthrough [day 2d6 1d2]] - 6 named "play wise"

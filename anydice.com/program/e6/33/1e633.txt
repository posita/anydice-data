\
 Blades in the Dark
 0: Things go bad.
 1: Success, but with undesired consequences.
 2: Clean success.
 3: Critical success!
\

function: blades ROLL:s {
 if 1@ROLL = 6 { result: 2 }
 result: 1@ROLL >= 4
}

function: datarun ROLL:s REROLL:d {
 if 1@ROLL = 6 { result: 2 }
 result: 1@REROLL = 6
}

output [blades 2@2d6] named "b 0d"
loop P over {1..7} {
 output [blades Pd6] named "b [P]d"
}
output [datarun 2@2d6 1d6] named "d 0d"
loop P over {1..7} {
 output [datarun Pd6 (P+1)d6] named "d [P]d"
}

\ Defining Cyberpunk's Exploding D10 \
function: cyberexplode N:n {
if N = 10 { result: 10 + 1d10 }
if N = 1 { result: 1 - 1d10 }
result: N }

function: autodmg DMG_ROLL:d ROLL_RESULT:n vs TGT:n {
MARGIN: ROLL_RESULT - TGT
if MARGIN >= 4 {result: DMG_ROLL * 4}
if MARGIN = 3 {result: DMG_ROLL * 3}
if MARGIN = 2 {result: DMG_ROLL * 2}
if MARGIN = 1 {result: DMG_ROLL}
result: 0}

BASEDIE: d6

function: target TARGET:n reroll REROLL:n VALUE:n {
  if VALUE = REROLL {
    result: 1 + [target TARGET reroll REROLL BASEDIE]
  }
  if VALUE  > TARGET {
    result: 1
  }
  result: 0
}

EXPLODING: [target 4 reroll 6 BASEDIE]
NORMAL: [target 4 reroll 7 BASEDIE]

function: extended TARGET:n dicepool DICEPOOL:n {
  VALUE: DICEPOOLdNORMAL
  if VALUE < TARGET {
    result: 1 + [target TARGET-VALUE dicepool DICEPOOL-1]
  }
  result: 0
}

DP: 16
RES: 6

loop FORCE over {1..10}{
   output [highest of [lowest of FORCE and DPdNORMAL] - (RES+FORCE)dNORMAL and 0] named "Force: [FORCE]"
}

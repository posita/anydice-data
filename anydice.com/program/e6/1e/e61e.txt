function: hitcheck ROLL:n vs AC:n with TOHIT:n and HITDAMAGE:n {
  if ROLL=20 { result: HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

loop AC over {10, 12, 14, 16, 18, 20}{
  DEX: 3
  PROF: 2
  ARCHERY: 2
  SHARP: 5
  TOHITONE: DEX+PROF+ARCHERY-SHARP
  TOHITTWO: DEX+PROF+ARCHERY
  TOHITTHREE: DEX+PROF-SHARP
  TOHITFOUR: DEX+PROF
  HITDIE: 1d6+1d4
  HITDAMAGEONE: DEX
  HITDAMAGETWO: DEX+10
  output [hitcheck 1d20 vs AC with TOHITONE and HITDAMAGETWO]+[hitcheck 1d20 vs AC with TOHITONE and HITDAMAGETWO] named "Sharp Arch Cbow vs [AC] AC"
  output [hitcheck 1d20 vs AC with TOHITTWO and HITDAMAGEONE]+[hitcheck 1d20 vs AC with TOHITTWO and HITDAMAGEONE] named "Arch Cbow vs [AC] AC"
  output [hitcheck 1d20 vs AC with TOHITTHREE and HITDAMAGETWO]+[hitcheck 1d20 vs AC with TOHITTHREE and HITDAMAGETWO] named "Sharp Def Cbow vs [AC] AC"
  output [hitcheck 1d20 vs AC with TOHITFOUR and HITDAMAGEONE]+[hitcheck 1d20 vs AC with TOHITFOUR and HITDAMAGEONE] named "Def Cbow vs [AC] AC"


  HITDIE: 1d8+1d4
  output [hitcheck 1d20 vs AC with TOHITONE and HITDAMAGETWO] named "Sharp Arch Lbow vs [AC] AC"
  output [hitcheck 1d20 vs AC with TOHITTWO and HITDAMAGEONE] named "Arch Lbow vs [AC] AC"
  output [hitcheck 1d20 vs AC with TOHITTHREE and HITDAMAGETWO] named "Sharp Def Lbow vs [AC] AC"
  output [hitcheck 1d20 vs AC with TOHITFOUR and HITDAMAGEONE] named "Def Lbow vs [AC] AC"
}

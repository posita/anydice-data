\**** Profit, Revenue, Cost Function ****\

function: results {
 if REVENUE > COST { result: 2 } 
 if REVENUE = COST { result: 1 }
 if REVENUE < COST { result: 0 }
}

\Profit is in copper\

COST: 100
POTATOESCOST: 5
POTATOES: COST/POTATOESCOST
LEFTOVERP: POTATOES - 5


FLOWERS: LEFTOVERP/15
PRICE: FLOWERS*5

CUSTOMERS: 32

REVENUE:PRICE*CUSTOMERS

PROFIT: REVENUE - COST


\Time is in hours\

TIME: 2
OPENLENGTH: TIME/2

output PROFIT named "Profit"
output OPENLENGTHd[results] named "Results"

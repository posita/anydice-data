GRITTY: {0,1}
DIFFICULTY: {1..3}
PROFICIENCY: {0..3}
HELP: {0..3}

loop G over GRITTY {
loop D over DIFFICULTY {
loop P over PROFICIENCY {
loop H over HELP {
 DIE: {-1*G,0,P>2,P>1,P>0,1}
 POOL: H+3
 ROLL: POOLdDIE
 ROLL: (ROLL>=D) + (ROLL>D)
 ROLL: ROLL
 if G {
  output ROLL named "D: [D]! / P: [P] / H: [H]"
 } else {
  output ROLL named "D: [D] / P: [P] / H: [H]"
 }
}
}
}
}

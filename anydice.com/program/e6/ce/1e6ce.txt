\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE + CRITNOVA } 
 if ROLL + ACCBONUS >= AC { result: DAMAGE + NOVA }
 result: MISSDAMAGE
}


ENEMYAC: d{10..22}

ACCBONUS: 14
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 1d6 + 1d4 + 5 + 3
CRITDAMAGE: 2d6 + 2d4 + 5 + 3
MISSDAMAGE: (8d6 + 1d6 + 1d4 + 5) / 2
NOVA: 8d6
CRITNOVA: 16d6
output ATTACKTIMESd[attackroll 1d20 vs ENEMYAC] named "Consistent"

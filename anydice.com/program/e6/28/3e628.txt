\
 MAVERICK
 0: Failure with a Setback.
 1: Failure.
 2: Success.
 3: Success with a Stunt!
\

function: pathfinder ROLL:n MOD:n DC:n {

if (ROLL + MOD <= DC - 10) { RES: 0 }
else if (ROLL + MOD < DC) { RES: 1 }
else if (ROLL + MOD < DC + 10) { RES: 2 }
else { RES: 3 }

  if (ROLL = 1) { RES: RES-1 }
  if (ROLL = 20) { RES: RES+1 }
  if (RES < 0) { result: 0 }
  else if (RES > 3) { result: 3 }
  else { result: RES }

}

function: maverick ROLL:n MOD:n DC:n {

if (ROLL + MOD <= DC - 5) { result: 0 }
else if (ROLL + MOD < DC) { result: 1 }
else if (ROLL + MOD < DC + 5) { result: 2 }
else { result: 3 }

}

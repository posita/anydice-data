function: add X:d highest to Y:d lowest {
result: [highest 1 of X] + [lowest 1 of Y]
}

function: test X:d against Y:d {
result: [add X highest to Y lowest] < 7
}

function: roll P:n dice against M:n dice {
 if P = 1 | M = 1 {
  result: [test Pd6 against Md6]
 }
 else {
  NP: P-1
  NM: M-1
  result: [test Pd6 against Md6] | [roll NP dice against NM dice]
 } 
}

loop P over {1..5} {
loop M over {1..5} {
output [roll P dice against M dice] named "Player [P]d6 loses against Moderator [M]d6"
}
}

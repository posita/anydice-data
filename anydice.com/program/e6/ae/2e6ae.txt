\**** Damage calculator normal attack ****\

function: attackroll ROLL:n vs DMG:n {
 if ROLL + SAVEBONUS >= SAVETARGET { result: DMG/2} 
 if ROLL + SAVEBONUS <= SAVETARGET { result: DMG }
 result: 0
}

SPELLDMG: 8d6
SAVETARGET: 13
SAVEBONUS: 2


output [attackroll 1d20 vs SPELLDMG] named "normal attack"
output 2d6 + 6 - 5
output 1d12

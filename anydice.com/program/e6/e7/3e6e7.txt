function: strike ROLL:n MOD:n DC:n
{
if (ROLL + MOD <= DC-10) {RES: 0} else
if (ROLL + MOD < DC) {RES: 0} else
if (ROLL + MOD < DC + 10) {RES: 1} else {RES: 2}

if (ROLL = 1) {RES: RES-1 }
if (ROLL = 20) { RES: RES+1 }

if (RES < 0) { result: 0 }
else if (RES > 2) { result: 2 }
else { result: RES }
}

ROLL: d20
MOD: 9
DC: 20

output [strike ROLL MOD DC]

function: triple
{
result: [strike ROLL MOD DC]+[strike ROLL MOD-5 DC]+[strike ROLL MOD-10 DC]
}

output 2d[triple]

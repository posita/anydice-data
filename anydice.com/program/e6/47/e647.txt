


function: results CUSTOMERS:n {
 if [revenue] - COST < 0 { result: 0 } 
 if [revenue] - COST = 0 { result: 1 }
 if [revenue] - COST > 0 { result: 2}
 result: 0
}


function: revenue {
 if FLOWERS = FLOWERS { result: PRICE*CUSTOMERS }
}

CUSTOMERS: 2d20


COST: 100
POTATOESCOST: 5
POTATOES: COST/POTATOESCOST
LEFTOVERP: POTATOES - 5

FLOWERS: (32 * LEFTOVERP)/15
PRICE: 5

PROFIT: [revenue] - COST
 
output [results 2d20] named "Profit in %"
output PROFIT named "Profit"

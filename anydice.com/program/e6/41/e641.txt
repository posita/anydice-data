HIT: 8
SWRD: 1d8
DMG: 5
BM: 1d8
BMOVE: 2d8
SVDC: 15

AC: 11
DEXSV: 0

function: attack ATK:n{
    if ATK = 1 {
        result : 0}
    if ATK = 20{
        result : SWRD*2+DMG}
    if ATK+HIT>=AC {
        result : SWRD+DMG}
    else { result : 0 } }

output [attack 1d20]+[attack 1d20]+[attack 1d20]+[attack 1d20] named "four attacks"

function: fire SAV:n {
    if SAV+DEXSV>=SVDC {
        result : 0}
    else {
        result : 2d8}}

function: boomattack ATK:n {
    if ATK = 1 {
        result : [attack 1d20]+[attack 1d20]}
    if ATK = 20{
        result : SWRD*2+DMG+BM+BMOVE+[fire 1d20]}
    if ATK+HIT>=AC {
        result : SWRD+DMG+BM+BMOVE+[fire 1d20]}
    else { result : [attack 1d20]+[attack 1d20] } }

output [boomattack 1d20] named "Booming Blade Bonfire"

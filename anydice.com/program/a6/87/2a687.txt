function: ROLL:n vs AC:n {
 if ROLL >= 20 & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { 
 result: DAMAGE
 }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: 2 d DAMAGE
 }
 result: DAMAGE
}

LEVEL: 1
AC: d{13..16}
MOD: 3
PROF: 2

ATTACK: PROF + MOD
DAMAGE: 1d10 + MOD
output [d20 vs AC] named "Eldritch Blast lvl 1"

ATTACK: PROF + MOD + OVERFLOW
DAMAGE: 2d6 + MOD
output [d20 vs AC] named "Kinetic Blast lvl 1"

DAMAGE: 2d6 + MOD

LEVEL: 20

function: proficiency {
    if LEVEL >=17 {
        result: 6
    }
    if LEVEL >= 13 {
        result: 5
    }
    if LEVEL >= 9 {
        result: 4
    }
    if LEVEL >= 5 {
        result: 3
    }
    else {
        result: 2
    }
}

function: str {
    if LEVEL >= 15 {
        result: 5
    }
    else {
        result: 4
    }
}

function: dex save {
    if LEVEL >= 15 {
        result: 1@2d20+5
    }
    if LEVEL >= 7 {
        result: 1@2d20+4
    }
    if LEVEL >= 4 {
        result: 1d20+4
    }
    if LEVEL >= 3 {
        result: 1d20+3
    }
    else {
        result: 1d20-1
    }
}

function: con {
    if LEVEL >= 15 {
        result: 5
    }
    else { 
        result: 4
    }
}

function: wis save {
    if LEVEL >=16 {
        result: 1@2d20+[proficiency]+4
    }
    if LEVEL >=4 {
        result: 1d20+[proficiency]+4
    }
    else {
        result: 1d20+3
    }
}

function: attack {
    if LEVEL >= 7 {
        result: 1@2d20+[str]+[proficiency]
    }
    else {
        result: 1d20+[str]+[proficiency]
    }
}

output 1d20+[str]+[proficiency] named "Level [LEVEL] Strength Saving Throw"
output [dex save] named "Level [LEVEL] Dexterity Saving Throw"
output 1d20+[con]+[proficiency] named "Level [LEVEL] Constitution Saving Throw"
output 1d20+2 named "Level [LEVEL] Intelligence Saving Throw"
output [wis save] named "Level [LEVEL] Wisdom Saving Throw"
output 1d20 named "Level [LEVEL] Charisma Saving Throw"
output [attack] named "Level [LEVEL] To-Hit Modifier"

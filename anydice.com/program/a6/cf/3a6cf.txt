LIM: 13
function: view crop SEQUENCE:n by MAX:n{                  \ limit the display range between [MIN;MAX] \
 MIN: 7
 if (SEQUENCE < MIN){result: {}}
 if (SEQUENCE > MAX){result: MAX}
 result: SEQUENCE
}

function: high low DIR:n for NUMBER:n of DICE:d{          \ switch between highest(1) & lowest(-1) roll \
 if DIR<0 { result: {(#DICE - NUMBER + 1)..#DICE}@DICE }
 if DIR>=0 { result: {1..NUMBER}@DICE }
 result: 0
}

function: action roll from TDICES:d to RDICES:d{
 ROLL: [view crop [high low TDICES for RDICES of (RDICES+[absolute TDICES])d6] by LIM]
 if ROLL >= 12 { result: 2 } 
 if ROLL >= 7 & ROLL < 12 { result: 1 } 
 result: 0
}             \ 0: Failure, 1: Success, 2: Critical\

function: injury roll from TDICES:d to RDICES:d with RESMODS:n{             \ RESMODS=armor&ap \
 ROLL: [view crop ([high low TDICES for RDICES of (RDICES+[absolute TDICES])d6] +RESMODS) by LIM]
 if ROLL >= 9 { result: 3 } 
 if ROLL >= 7 & ROLL < 9 { result: 2 } 
 if ROLL >= 2 & ROLL < 7 { result: 1 } 
 result: 0
}             \ 0: Nothing, 1: Minor Hit, 2: Down, 3: Out of Action \

function: do attacks sequence from PARAMS:s{
 A: [action roll from 1@PARAMS to 2@PARAMS]
 if A >= 1 { result: [injury roll from 3@PARAMS to 4@PARAMS with 5@PARAMS] }
 result: 0
}

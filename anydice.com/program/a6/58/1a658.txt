
\**** Damage calculator ****\

function: attk ROLL:n vs AC:n wmod WEAPON_MOD:n add prof ADD_PROF:n does WPN_DAM:n damage {
 ADD_PROF: [highest of 0 and [lowest of 1 and ADD_PROF]]  \* add prof is either zero or one *\
 if ROLL >= 20 { result: WPN_DAM + WPN_DAM + WEAPON_MOD + (ABILITY_MODIFIER * ADD_PROF) + OTHER_BUFF_DAMAGE }
 if ROLL = 1 { result: 0 }
 if ROLL + WEAPON_MOD + ABILITY_MODIFIER + (PROFICIENCY_BONUS * ADD_PROF) + OTHER_BUFF_HIT >= AC { result: WPN_DAM + WEAPON_MOD + (ABILITY_MODIFIER * ADD_PROF) + OTHER_BUFF_DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

PROFICIENCY_BONUS: 3
ABILITY_MODIFIER: 4
WEAPON_MOD: 1
OTHER_BUFF_HIT: 0
OTHER_BUFF_DAMAGE: 2

output d[attk d20 vs d{15..20} wmod 1 add prof 1 does 1d8 damage] + d[attk d20 vs d{15..20} wmod 1 add prof 1 does 1d8 damage] named "Rapier with Dueling Bonus"

OTHER_BUFF_DAMAGE: 0
output d[attk d20 vs d{15..20} wmod 1 add prof 1 does 1d6 damage] + d[attk d20 vs d{15..20} wmod 1 add prof 1 does 1d6 damage] + d[attk d20 vs d{15..20} wmod 1 add prof 1 does 1d6 damage] named "Short Sword & Dagger"

\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE + [bonusroll d20 vs AC] } 
 if ROLL = 1 { result: [bonusroll d20 vs AC] }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE + [bonusroll d20 vs AC]}
 result: [bonusroll d20 vs AC]
}

function: bonusroll ROLL_B:n vs SAMEAC {
 if ROLL_B >= CRITROLL { result: CRITDAMAGE_B } 
 if ROLL_B = 1 { result: 0 }
 if ROLL_B + PROFICIENCY + ATTACKMOD + ACCBONUS_B >= SAMEAC { result: DAMAGE_B }
 result: 0
}

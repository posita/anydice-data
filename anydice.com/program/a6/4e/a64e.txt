AC: 8
MISSDAMAGE: 0

function: damage ROLL:n {
 if ROLL >= MAXDAMAGE {
   result: DAMAGEDICE+CRITICALDAMAGE
 }

 result: DAMAGEDICE
}

function: attack ROLL:n {
 if ROLL >= AC {
  result: [damage DAMAGEDICE]
 }
 result: MISSDAMAGE
}

MAXDAMAGE: 12
CRITICALDAMAGE: 12

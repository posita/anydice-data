function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

PROF:3
AC:10
DEX:5
ATTACK:5
STYLE:0
MAGIC:1
SHBONUS: PROF + DEX + STYLE + MAGIC - 5
BONUS: PROF + DEX + STYLE + MAGIC
loop AC over {10..25}{
output ATTACKd[attack [highest 1 of 1d20] plus BONUS vs AC for 1d12 + 4 + MAGIC + DEX crit 1d12 on 20] named "[AC] Monk dagger, dagger, dagger,dagger "
output ATTACKd[attack [highest 1 of 1d20] plus BONUS vs AC for 2d12 + 4 + MAGIC + DEX crit 2d12 on 20] named "[AC] Monk dagger smite, dagger smite, dagger smite,dagger smite"
}

\ Charges \
CHARGES: 6

\ Steam Rifle Die Sizes \
HPDIE: 8
LPDIE: 4

\ Internal Variables \
HPCHARGES: CHARGES / 2

HPROUNDS: HPCHARGES + 1
LPROUNDS: CHARGES + 1

HPBONUS: HPROUNDS * 5
LPBONUS: LPROUNDS * 5

\\
\ High Power Shot Calculations \
\\
\ Steam Rifle \
output (HPCHARGES * 2)dHPDIE + (HPCHARGES * 5) named "Steam Rifle: High Power"

\ Normal Weapons \
output HPROUNDSd6 + HPBONUS named "Hand XBow: High Power"
output HPROUNDSd8 + HPBONUS named "Light XBow: High Power"
output HPROUNDSd10 + HPBONUS named "Heavy XBow: High Power"


\\
\ Low Power Shot Calculations \
\\
\ Steam Rifle \
output (CHARGES * 2)dLPDIE + (CHARGES * 5) named "Steam Rifle: Low Power"

\ Normal Weapons \
output LPROUNDSd6 + LPBONUS named "Hand XBow: Low Power"
output LPROUNDSd8 + LPBONUS named "Light XBow: Low Power"
output LPROUNDSd10 + LPBONUS named "Heavy XBow: Low Power"

A:{0:3,1:2,2}
E:(1d6 = 6)
S:{0,1:3} \Pushing it to re-roll anything other than a shift\

function: attack with N:n to hit {
 result: 1d{0:3-N,1:2+N,2}
}

function: headshot attack with N:n to hit {
 result: 1d{0:3-N,1:2+N,3}
}

loop B over {0..3} {
 loop N over {1..20} {
  output Nd[attack with B to hit] >= 6 named "[N] attacks with +[B] bonus to hit."
 }
}

loop B over {0..3} {
 loop N over {1..20} {
  output (Nd[headshot attack with B to hit]) >= 6 named "[N] attacks with +[B] bonus to hit and Headshot."
 }
}

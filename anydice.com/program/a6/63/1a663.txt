ADV: 0

function: attack NUMBER:n {
 VALUE: [highest 1 of NUMBERd20]
 result: [check VALUE]
}

function: check NUMBER:n {
 if NUMBER = 20 {
  result: [damage 1]
 }
 
 result: [damage 0]
}

function: damage BOOL:n {
 if BOOL = 1 {
  ROLL: 0

 [reroll 1d12]
 } else {
  ROLL: [reroll 1d12]
 }

 result: ROLL
}

function: reroll ROLL:n {
 if ROLL = 1 | ROLL = 2 {
  ROLL: 1d12
 }
 
 result: ROLL
}

output [attack ADV + 1]

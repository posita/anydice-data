\ -*- mode: prog; -*- \

STRENGTH: 4
HEART: 5
WITS: 5

STN: 20 - STRENGTH
HTN: 20 - HEART
WTN: 20 - WITS

function: favoured DIE:d {
  result: [highest of DIE and DIE]
}

function: check TARGET:n SR:s FEAT:n {
  \ degree of potential success -- count T runes \
  DEGREE: 1 + [count 6 in SR]
  \ success \
  if SR + FEAT - 1 >= TARGET { result: DEGREE }
  \ saved by G rune \
  if FEAT = 12 { result: DEGREE }
  result: 0
}

function: try TARGET:n SKILL:n FEAT:n {
  result: [check TARGET (SKILL)d6 FEAT]
}

function: roll TARGET:n SKILL:n FAVOURED:n {
  if FAVOURED { result: [try TARGET SKILL [favoured 1d12]] }
  result: [try TARGET SKILL 1d12]
}

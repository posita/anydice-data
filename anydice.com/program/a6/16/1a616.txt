function: attackroll highest NUMBER:n of ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 else if ROLL = 1 { result: 0 }
 else if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

ATTACKMOD: 0
PROFICIENCY: 7
CRITROLL: 20
ATTACKTIMES: 12
ACCBONUS: 0
DMGBONUS: 5
WEAPONDIE: 2d6


loop ENEMYAC over {15..21}{
DAMAGE: WEAPONDIE + ATTACKMOD + DMGBONUS
CRITDAMAGE: WEAPONDIE + WEAPONDIE + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll highest 1 of 1d20 vs ENEMYAC] named "Gladiator vs AC [ENEMYAC]"
}

loop ENEMYAC over {15..21}{
ATTACKTIMES: 4
output ATTACKTIMESd[attackroll highest 1 of 1d20 vs ENEMYAC] named "All Slow Gladiator vs AC [ENEMYAC]"
}

loop ENEMYAC over {15..21}{
ATTACKTIMES: 8
output ATTACKTIMESd[attackroll highest 1 of 1d20 vs ENEMYAC] named "Half Slow Gladiator vs AC [ENEMYAC]"
}

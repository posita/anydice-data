function: master X with Y {
  result: [check X with Y for d12 target 7]
}

function: check X with Y for D target T {
  R: 0
  if Y > 0 {
    loop N over {1..Y} {
      R: R + [roll D target T-2]
    }
  }
  loop N over {1..(X - Y)} {
    R: R + [roll D target T]
  }
  result: R
}

function: roll Y target X {
  result: (Y >= X)
}

function: no success when X with Y{
  result: ([check X with Y] = 0)
}

output [master 1 with 0] named "1/0"
output [master 1 with 1] named "1/1"
output [master 2 with 0] named "2/0"
output [master 2 with 1] named "2/1"

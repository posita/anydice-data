function: check X with Y {
  R: 0
  if Y > 0 {
    loop N over {1..Y} {
      R: R + [roll d6 target 3]
    }
  }
  loop N over {1..(X - Y)} {
    R: R + [roll d6 target 4]
  }
  result: R
}

function: roll Y target X {
  result: (Y >= X)
}


output [check 1 with 0] named "Basic"
output [check 1 with 1] named "Basic with 1 focus"

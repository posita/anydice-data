function: roll A:n B:n C:n {
    result: 1@[sort {A, B, C}]
}

function: roll A:n B:n C:n D:n {
    result: 1@[sort {A, B, C, D}]
}

function: roll A:n B:n C:n D:n E:n{
    result: 1@[sort {A, B, C, D, E}]
}

function: roll A:n B:n C:n D:n E:n F:n{
    result: 1@[sort {A, B, C, D, E, F}]
}



function: rollorder A:n B:n C:n D:n E:n F:n {
    R: [sort {A, B, C}]
    S: [sort {D, E, F}] 
    if 1@R > 1@S{
			if 2@S = 1 {
				result: 100
			}else if 2@R > 2@S {
				result: 1@R
			}else{
				result: (1@R - 1@S) * -1
			}
    } else {
			result: 0
		}
}


function: rollrng A:n B:n C:n {
	V: [sort {A, B, C}]
	
	AA: 1@V
	BB: 2@V
	CC: 3@V

	if AA = BB & BB = CC{
		if AA = 1 {
			result: -999999
		} else {
			result: 999999
		}
	}
	
	if AA = BB {
		result: 100 * 3 * AA
	}else{
		if BB = CC & BB = 1 {
			result: -1 * AA
		} else {
			result: AA + BB
		}
	}
}


function: rolla A:n B:n C:n {
    result: A *100 + B*10 + C
}

function: rollb A:n B:n C:n {
    result: A + B + C
}

function: rollc A:n B:n C:n {
    AUX: [sort {A, B, C}]
    result: 1@AUX *100 + 2@AUX *10 + 3@AUX
}



function: rollabs A:n S:n T:n {
	if S = T & S = 1 {
		result: 1dA + S + T
	}else{
		result: A + S + T
	}
}

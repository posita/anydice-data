\ Player rolls 4d6 \
\  Player rolls can have modifiers from up to 3 sources \
\ Player keeps the highest roll \
\  For every dice that match, add +1 to the total \
\  For every die that matches the highest roll, add +2 to the total  \
	

function: distribution A:n vs CHALLENGE:s plus MOD:n { 
 result: ((A + MOD) > 1@CHALLENGE) + ((A + MOD) > 2@CHALLENGE)
}

loop M over {-4..4}{
loop X over {4, 6, 8, 10, 12} {
 
  output [distribution dX vs 2d10 plus M] named "[X]/[M]"
}}

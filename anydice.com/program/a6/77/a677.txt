AC: 8
MISSDAMAGE: 0

function: damage ROLL:n {
 if ROLL >= MAXDAMAGE {
   result: DAMAGEDICE+CRITICALDAMAGE+PROFICIENCY
 }

 result: DAMAGEDICE+PROFICIENCY
}

function: attack ROLL:n {
 if ROLL >= AC {
  result: [damage DAMAGEDICE]
 }
 result: MISSDAMAGE
}

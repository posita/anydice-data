PIFIA: -1
FALLO: 0
EXITO: 1
DIFICULTAD: 3

function: prueba de TIRADA:s de dificultad DIFILCULTAD:n {
 ROLL: TIRADA
 EXITOS: ROLL >= 5
 SEISES: ROLL = 6
 FALLOS: ROLL <4
 UNOS: ROLL = 1
 REROLLS: [lowest of SEISES and FALLOS]d6
 EXITOSREROLL: REROLLS >= 5
 TOTALEXITOS: EXITOS 
 if TOTALEXITOS = 0 & UNOS > 0 { result: PIFIA }
 if TOTALEXITOS>= DIFILCULTAD { result: EXITO }
 result: FALLO
}

GRIT: 8
loop GRIT over {6,8} {
loop DADOS over {3,4} {
loop DIFICULTAD over {1,2,3} {
output [prueba de (DADOS)d(GRIT) de dificultad DIFICULTAD] named "[DADOS]d[GRIT]/dif[DIFICULTAD]"
}}}

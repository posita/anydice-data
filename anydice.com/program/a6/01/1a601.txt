function: attack N:n plus B:n dmg D:d plus DB:n vs AC:n crit C:n smite S:d{
 if N >= C {result: 2dD + dS + DB}
 if N + B >= AC {result: dD + DB}
 result: 0
}

function: save N:n vs DC:n for half of D:n {
 if N > DC {result: D/2}
result: D
}

function: attackroll highest NUMBER:n of ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 else if ROLL = 1 { result: 0 }
 else if ROLL + PROFICIENCY + STR + ATTACKMOD >= AC { result: DAMAGE }
 result: 0
}


STR: 4
ATTACKMOD: 0
PROFICIENCY: 3
CRITROLL: 20
ATTACKTIMES: 1


loop ENEMYAC over {12..18}{
DAMAGE: 2d6 + STR
CRITDAMAGE: 2d6 + 2d6 + STR
output ATTACKTIMESd[attackroll highest 1 of 1d20 vs ENEMYAC] named "Greatsword vs AC [ENEMYAC]"
}

loop ENEMYAC over {12..18}{
ATTACKMOD: -5
DAMAGE: 2d6 + STR + 10
CRITDAMAGE: 2d6 + 2d6 + STR + 10
output ATTACKTIMESd[attackroll highest 1 of 1d20 vs ENEMYAC] named "GWM Greatsword vs AC [ENEMYAC]"
}

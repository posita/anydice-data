function: master A B C D E {
  R: 0
  T: 6
  if A > 0 {
    R: R + [check A 1d6 T]
  }
  if B > 0 {
    R: R + [check B 1d8 T]
  }
  if C > 0 {
    R: R + [check C 1d10 T]
  }
  if D > 0 {
    R: R + [check D 1d12 T]
  }
  if E > 0 {
    R: R + [check E 1d20 T]
  }
  result: R
}

function: check X ROLL T {
  S: 0
  loop N over {1..X} {
    S: S + (ROLL >= T)
  }
  result: S
}

output [master 1 0 0 0 0] named "1d6"
output [master 0 1 0 0 0] named "1d8"
output [master 2 0 0 0 0] named "2d6"
output [master 1 1 0 0 0] named "1d6/1d8"
output [master 0 2 0 0 0] named "2d8"

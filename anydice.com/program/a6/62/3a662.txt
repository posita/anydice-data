function: average of A and B {
  result: ((A+B)/2)
}

function: factorial of A {
  R: 1
  loop N over {1..A}{
    R: R * N
  }
  result: R
}

function: sine of T {
  R: 0
  loop N over {1..2}{
    R: R + ((((-1)^N)/[factorial of (2*N+1)])*(T^(2*N+1)))
  }
  result: R
}

function: hybrid with R:n {
  if R > 50 {
    result: (R*100)+(40/3)*R*[sine of ((1/50)*R-1)]*180
  }
  else {
    result: R
  }
}

output 1d100 named "1RN"
output [average of 1d100 and 1d100] named "2RN"
output [hybrid with 1d100] named "hybrid hit rate"

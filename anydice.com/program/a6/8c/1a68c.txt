\**** Damage calculator Double weapon attack Vs Booming Blade at lv 5 ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from X1-X2\
ENEMYAC: d{15}

ATTACKMOD: 4
DMGBONUS: 0
PROFICIENCY: 3
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 2d4 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 4d4 + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Weapon attack only"

ATTACKMOD: 4
DMGBONUS: 2d8
PROFICIENCY: 3
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 2d4 + ATTACKMOD + DMGBONUS + 1d8
CRITDAMAGE: 4d4 + ATTACKMOD + DMGBONUS + 1d8
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Booming Blade"

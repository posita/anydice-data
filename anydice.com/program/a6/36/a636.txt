STAIRS: 18
DEX: 0
STR: 0


function: check TRIP:n against TRIP_DC:n then FALL:n against FALL_DC:n{
  if TRIP < TRIP_DC {
    if FALL < FALL_DC{
      result: 0
    }
  }
  result: 1
}

\Going down one stair at a time\
SURVIVE: 0
loop N over {1..STAIRS}{
  TRIP_CHECK: [highest 1 of 2d20] + DEX
  FALL_CHECK: 1d20 + STR
  SURVIVE: SURVIVE + [check TRIP_CHECK against 15 then FALL_CHECK against 13]
}
output SURVIVE

\Going down two stairs at a time\
SURVIVE: 0
loop N over {1..STAIRS/2}{
  TRIP_CHECK: 1d20 + DEX
  FALL_CHECK: 1d20 + STR
  SURVIVE: SURVIVE + [check TRIP_CHECK against 15 then FALL_CHECK against 13]
}
output SURVIVE

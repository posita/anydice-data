\ Charges \
HPCHARGES: 2
LPCHARGES: 3

\ Steam Rifle Die Sizes \
DIE: 10

\ Damage Bonus \
BONUS: 3

\ Internal Variables \
CHARGES: HPCHARGES + LPCHARGES

ROUNDS: CHARGES + 1

ROUNDBONUS: ROUNDS * BONUS

\\
\ High Power Shot Calculations \
\\
\ Steam Rifle \
output ((HPCHARGES * 2)dDIE + LPCHARGESdDIE + (CHARGES * BONUS)) / ROUNDS named "Steam Rifle"

\ Normal Weapons \
output (ROUNDSd6 + ROUNDBONUS) / ROUNDS named "d6 Weapon"
output (ROUNDSd8 + ROUNDBONUS) / ROUNDS named "d8 Weapon"
output (ROUNDSd10 + ROUNDBONUS) / ROUNDS named "d10 Weapon"
output (ROUNDSd12 + ROUNDBONUS) / ROUNDS named "d12 Weapon"

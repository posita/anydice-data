AC: 8
MISSDAMAGE: 0

function: damage ROLL:n {
 if ROLL >= MAXDAMAGE {
   result: DAMAGEDICE+CRITICALDAMAGE
 }

 result: DAMAGEDICE
}

function: attack ROLL:n {
 if ROLL >= AC {
  result: [damage DAMAGEDICE]
 }
 result: MISSDAMAGE
}

DAMAGEDICE: 1d12
MAXDAMAGE: 12
CRITICALDAMAGE: 12
output [attack d20] named "Greataxe"

DAMAGEDICE: 2d6
MAXDAMAGE: 12
CRITICALDAMAGE: 6
output [attack d20] named "Greatsword"

DAMAGEDICE: 2d10
MAXDAMAGE: 20
CRITICALDAMAGE: 10
output [attack [lowest 1 of d20]] named "Ultra Greatsword"

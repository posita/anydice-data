\ Charges \

\ Steam Rifle Die Sizes \
DIE: 10

\ Internal Variables \

\\

\\

\ Normal Weapons \
output ROUNDSd6 named "Hand XBow"
output ROUNDSd8 named "Light XBow"
output ROUNDSd10 named "Heavy XBow"

function: nwod R:n again N:n count T:n{
 if T >= 10 {result: N>=R}
 if N >= 6 {
   T: T+1
   result: 1+[nwod R again d10 count T]
 }
 \if N = 1 { 
  T: T+1
  result: -1 + [nwod R again d10 count T] 
 }\
 result: N >= R
}


loop R over {4..4} {
 loop N over {1..10} {
  output [highest of -2 and [lowest of 10 and Nd[nwod R again d6 count 0]]] named "Number of success proba with [N]d10 difficulty [R]"
 }
}

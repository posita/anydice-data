function: crossb X:n ac A:n  {
    if X >= 10+A {
        result: 2d6
    }
    if X >= A {
        result: 1d6
    }
    if X < A-10 {
        result: 0
    }
    if X < A{
        result: 0
    }
}

function: doublecrossb X:n ac A:n  {
    if X >= 10+A {
        result: 4d6
    }
    if X >= A {
        result: 2d6
    }
    if X < A-10 {
        result: 0
    }
    if X < A{
        result: 0
    }
}

function: elecarc X:n dc A:n  {
    if X >= 10+A {
        result: 0
    }
    if X >= A {
        result: (1d4+4)/2
    }
    if X < A-10 {
        result: 2d4+8
    }
    if X < A{
        result: 1d4+4
    }
}

function: kinproj X:n ac A:n  {
    if X >= 10+A {
        result: 2d6+8
    }
    if X >= A {
        result: 1d6+4
    }
    if X < A-10 {
        result: 0
    }
    if X < A{
        result: 0
    }
}

output [elecarc 1d20+5  dc 17 ] named "elecarc"
output [elecarc 1d20+5  dc 17 ]+[crossb 1d20+5  ac 16 ] named "elecarc_and_crossb"
output [kinproj 1d20+7  ac 16 ] named "kinproj "

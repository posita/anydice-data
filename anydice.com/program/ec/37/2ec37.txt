function: reroll DIEA:n or DIEB:n {
  if DIEA = 1 | DIEA = 2 {
    result: DIEB
  }
  result: DIEA
}
output 2d4
output 1d4+1d6
output 3d4

output [reroll 1d6 or 1d6] +[reroll 1d4 or 1d4] named "reroll 1d4+1d6"

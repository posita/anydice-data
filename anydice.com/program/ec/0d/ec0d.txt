function: rolloff A:n against B:n {
 if A>B {result: 1} 
 if A<B {result: 2} 
}

function: includeseize A:n against B:n with a N:n on a R:n {
  if A>B {
    \ player 1 won, so check if player 2 seizes \
    if R=N { result: 2 }
    else {result:1}
  } 
  if A<B {
    \ player 2 won, so check if player 1 seizes \
    if R=N { result: 1 }
    else { result:2 }
  }
}

output [rolloff d6 against d6+1] named "raw player 1 (d6) vs player 2 (d6+1) wins"
output [includeseize d6 against d6+1 with a 6 on a d6] named "player 1 (d6) vs player 2 (d6+1) wins including seize"

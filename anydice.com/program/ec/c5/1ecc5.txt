function: success SEQ:s {
 result: {6 <= SEQ}
}

function: spiritflare SEQ:s {
  result: {4 = SEQ}
}

function: overlight V:n and S:n {
 SUC: 0
 
 loop X over {1..3} {
  SUC: SUC + [success 1dV]
 }
 loop X over {1..3} {
  SUC: SUC + [success 1dS]
 }
 
 result: SUC
}

output [overlight 6 and 6] named "both d6"
output [overlight 6 and 8] named "d6 and d8"
output [overlight 6 and 10] named "d6 and d10"
output [overlight 8 and 8] named "both d8"
output [overlight 8 and 10] named "d8 and d10"
output [overlight 10 and 10] named "both d10"

function: reroll DIEA:n or DIEB:n {
  if DIEA = 1 | DIEA = 2 {
    result: DIEB
  }
  result: DIEA
}

output [reroll 1d6 or 1d6] named "reroll 1d6"
output [reroll 1d8 or 1d8] named "reroll 1d8"
output [reroll 1d10 or 1d10] named "reroll 1d10"
output [reroll 1d12 or 1d12] named "reroll 1d12"
output [reroll 1d6 or 1d6] + [reroll 1d6 or 1d6] named "reroll 2d6"

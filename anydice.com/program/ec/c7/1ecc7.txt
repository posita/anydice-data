function: success SEQ:s {
 result: {6 <= SEQ}
}

function: spiritflare SEQ:s {
  result: {4 = SEQ}
}

function: X:n getresult S:n {
 if X = 0 {
  result: 0
 }
 if X = 1 {
  if S = 0 {
   result: 0
  }
  if S = 1 {
   result: 1
  }
 }
 if X = 2 {
  result: 1
 }
 if X = 3 {
  if S = 0 {
   result: 1
  }
  if S = 1 {
   result: 2
  }
 }
 if X = 4 {
  result: 2
 }
 if X = 5 {
  if S = 0 {
   result: 2
  }
  if S = 1 {
   result: 3
  }
 }
 if X = 6 {
  if S = 0 {
   result: 3
  }
  if S = 1 {
   result: 4
  }
 }
}

function: overlight V:n and S:n {
 SUC: 0
 FLARE: [spiritflare 1d4]

 loop X over {1..3} {
  SUC: SUC + [success 1dV]
 }
 loop X over {1..3} {
  SUC: SUC + [success 1dS]
 }
 result: [SUC getresult FLARE]
}

output [overlight 6 and 6] named "both d6"
output [overlight 6 and 8] named "d6 and d8"
output [overlight 6 and 10] named "d6 and d10"
output [overlight 8 and 8] named "both d8"
output [overlight 8 and 10] named "d8 and d10"
output [overlight 10 and 10] named "both d10"

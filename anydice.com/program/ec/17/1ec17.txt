function: ROLL:n vs AC:n {
 if ROLL >= RITON & ROLL + ATTACK >= AC {
  result: [riton d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: riton ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE - STRENGTH
 }
 result: DAMAGE 
}

AC: 16
STRENGTH: 5
ATTACK: 3 + STRENGTH

DAMAGE: 1d6 + STRENGTH
RITON: 20
CRITICAL: 2
output [2d20 vs AC] named "Gun"

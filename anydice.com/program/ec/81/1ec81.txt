function: COINS:n coinflips plus BONUS:n
{
 NUMBER_OF_POSITIVES: 0
 loop N over {1..COINS}{
  NUMBER_OF_POSITIVES: NUMBER_OF_POSITIVES + (1d8+BONUS>4)
 }
 result: NUMBER_OF_POSITIVES
}

loop N over {1..5}{
  loop O over{-1..3}{
   output [N coinflips plus O] named "rolling [N] coins plus [O]"
  }
 output 0 named "---------------------"
}

function: successes ROLL:s {
  if ROLL >= 34 {result: 7 }
  if ROLL >= 30 {result: 6 }
  if ROLL >= 26 {result: 5 }
  if ROLL >= 22 {result: 4 } 
  if ROLL >= 18 {result: 3 } 
  if ROLL >= 14 {result: 2 } 
  if ROLL >= 10 {result: 1 } 
  result: 0 \ failure \
}

loop TYPE over {0,-1,1,-2,2} {
  loop MOD over {-2,-1,+0,+1,+2,+3,+4,+5,+6,+7,+8} {
    if (TYPE =  0) { output [successes               2d{0..9} +MOD] named "[MOD]" }
    if (TYPE = -1) { output [successes [lowest  2 of 3d{0..9}]+MOD] named "[MOD] Bane" }
    if (TYPE =  1) { output [successes [highest 2 of 3d{0..9}]+MOD] named "[MOD] Boon" }
    if (TYPE = -2) { output [successes [lowest  2 of 4d{0..9}]+MOD] named "[MOD] Double Bane" }
    if (TYPE =  2) { output [successes [highest 2 of 4d{0..9}]+MOD] named "[MOD] Double Boon" } 
  }
}

function: rolloff A:n against B:n {
 if A>B {result: 1} 
 if A<B {result: 2} 
}

function: seized A:n against B:n with a N:n on a R:n {
  if A>B {
    \ player A won, so check if player 2 seizes \
    if R=N {result: 2}
    else {result:1}
  } 
  if A<B {result: 2}

}

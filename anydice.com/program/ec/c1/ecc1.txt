G:50 #number of times to run the 1st loop

H=10000 #number of times to run the 2nd loop

y=vector(mode='logical', length=G) #vector that will have all the dice averages in it

w=vector(mode='logical', length=G) #vector that will have all the standard deviations in it

for (k in 1:G) {

z=vector(mode='logical', length=H) #vector that will store the total number of successes from each loop

for (j in 1:H) {

x=vector(mode='logical', length=k) #vector that stores individual number of successes from each dice roll
for (i in 1:k) {

  S <- runif(1,0,1) #this is the actual dice simulation part

  if (S < 0.6) { #for 0 successes

    x[i] <- 0

  } else if (S >= 0.9) { #for 2 successes

    x[i] <- 2 

  } else { # for 1 success

    x[i] <- 1

  }

}

z[j] <- sum(x)

}

y[k] <- mean(z)

w[k] <- sd(z)

}

print(y)
print(w)

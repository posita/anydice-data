function: roll ROLL:s hit on TARGET:n difficulty P:n and F:n {
  FULL: F
  if FULL = 0 { FULL: P }

  HITCOUNT: ROLL >= TARGET
  if HITCOUNT >= FULL { result: 2 }
  if HITCOUNT >= P { result: 1 }
  result: 0 
}


output [roll 3d6 hit on 4 difficulty 2 and 0] named "4d6 - 0 = Failure, 1 = Partial, 2 = Success"

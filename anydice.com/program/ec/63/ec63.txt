\AVERAGE DAMAGE WITH CRIT CHANCE\

\*********FILL VALUES IN HERE***********\
\DON'T INCLUDE DAMAGE/ATTACK MODIFIERS\

DAMAGE_DICE: 1d6
ATTACK_DIE: 1d20
CRITS_ON: 20

\***************************************\


function: DMG:d crits with X:n on a Y:n {
  if X>=Y {
    result: DMG+DMG
  }
  else {
    result: DMG
  }
}

output [DAMAGE_DICE crits with ATTACK_DIE on a CRITS_ON]
output DAMAGE_DICE

CP: 1
SP: 10
EP: 50
GP: 100
PP: 1000
ROLL: 1d100
function: treasure CR:n ROLL:n
{
 if CR <= 4
 {
 if ROLL <= 30 {result: 5d6*CP}
 if ROLL <= 60 {result: 4d6*SP}
 if ROLL <= 70 {result: 3d6*EP}
 if ROLL <= 95 {result: 3d6*GP}
 if ROLL <= 100 {result: 1d6*PP}
 }
 if CR <= 10
 {
 if ROLL <= 30 {result: 4d6*CP*100 + 1d6*EP*10}
 if ROLL <= 60 {result: 6d6*SP*10 + 2d6*GP*10}
 if ROLL <= 70 {result: 3d6*EP + 2d6*GP*10}
 if ROLL <= 95 {result: 4d6*GP*10}
 if ROLL <= 100 {result: 2d6*GP*10 + 3d6*PP}
 }
 if CR <= 16
 {
 if ROLL <= 20 {result: 4d6*SP*100 + 1d6*GP*100}
 if ROLL <= 35 {result: 1d6*EP*100 + 1d6*GP*100}
 if ROLL <= 75 {result: 2d6*GP*100 + 1d6*PP*10}
 if ROLL <= 100 {result: 2d6*GP*100 + 1d6*PP*10}
 }
 if CR >= 17
 {
 if ROLL <= 15 {result: 2d6*EP*1000 + 8d6*GP*100}
 if ROLL <= 55 {result: 1d6*GP*1000 + 1d6*PP*100}
 if ROLL <= 100 {result: 1d6*GP*1000 + 2d6*PP*100}
 }
}
CR: 0
output [treasure CR ROLL] named "CP for CR [CR] or higher"
CR: 5
output [treasure CR ROLL] named "CP for CR [CR] or higher"
CR: 11
output [treasure CR ROLL] named "CP for CR [CR] or higher"
CR: 17
output [treasure CR ROLL] named "CP for CR [CR] or higher"

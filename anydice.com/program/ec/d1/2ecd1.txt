ATTD: 3
DEFD: 2
FACES: {0,0,1,1,1,3}

function: bring up to zero RES:n{
  if RES < 0 {RES: 0}
  result: RES
}

function: wounds ATT:n DEF:n {
  RES: [roll ATT DEF]
  result: [bring up to zero RES]
}

function: roll ATT:n DEF:n {
  RES: ATTdFACES - DEFdFACES
  result: RES
}

\ output [roll ATTD DEFD] named "[ATTD] Att vs. [DEFD] Def Results" \
output [wounds ATTD DEFD] named "[ATTD] Att vs. [DEFD] Def Wounds"

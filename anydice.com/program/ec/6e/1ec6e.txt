function: tres N:n {
  if N <= 14 {result: 0}
  if N <= 16 {result: 1 + [tres d16]}
}

function: dos N:n {
  if N <= 10 {result: 0}
  if N <= 14 {result: 1}
  if N <= 15 {result: 1 + [tres d16]}
  if N <= 16 {result: 2 + [tres d16]}
}

function: uno N:n {
  if N <= 6 {result: 0}
  if N <= 12 {result: 1}
  if N <= 14 {result: 2}
  if N <= 15 {result: 2+[dos d16]}
  if N <= 16 {result: 3+[dos d16]}
}

output [uno d16]

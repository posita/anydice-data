function: attackwparm ROLL:n vs DEFENSE:n bab MYBAB:n strength MYSTRENGTH:n threat MYTHREAT:n size MYSIZE:n magic MYMAGIC:n pa MYPA:n threatmod MYTHREATMOD:n other MYOTHER:n {
 MYPAPENALTY: -MYPA * (1 + MYBAB/4)
 MYATTACK: MYBAB + MYSTRENGTH + MYSIZE + MYMAGIC + MYOTHER + MYPAPENALTY
 if ROLL = 1 { result: MISS }
 if (ROLL >= MYTHREAT & ROLL + MYATTACK >= DEFENSE) | ROLL = 20 {result: [threat d20 mod MYTHREATMOD attack MYATTACK]}
 if ROLL + MYATTACK >= DEFENSE { result: HIT }
 result: MISS
}


function: threat ROLL:n mod MYMOD attack MYATTACK {
  if AUTOCRIT {result: CRITICALHIT}
  if ROLL + MYATTACK + MYMOD >= AC | ROLL = 20 {result: CRITICALHIT}
 result: HIT
}

 if ATTACKRESULT = CRITICALHIT {
  result: (MYCRITICAL d MYDAMAGE) + MYSNEAKATTACKDAMAGE + MYKILLERBONUS*MYCRITICAL
 }
 if ATTACKRESULT = HIT {
  result: MYDAMAGE + MYSNEAKATTACKDAMAGE
 }
 result: MYMISSDAMAGE
}

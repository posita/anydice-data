function: N:n modulo M:n {
 result: N - (N / M) * M
}
function: D:n crit {
  if D = 1 {
    result: 1 
  }
  if D = 20 {
    result: -1
  }
}

loop THR over {-5,0,5,10,15,20} {
ADV: [highest 1 of 2d20]
DIS: [lowest 1 of 2d20]
DIE: d20
MGN: DIE - THR
RMN: [[absolute MGN] modulo 5]
STP: (MGN / 5) + [DIE crit]
  output -STP named "Steps for Thresh of [THR]"
}

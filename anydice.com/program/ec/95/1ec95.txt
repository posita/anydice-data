function: count D:s N {
  COUNT: (D = N)
  LEN: #COUNT
  VAL: 1@COUNT
  result: COUNT
}

loop N over {1..6} {
output [count Nd{0, 1, 1, 2, 2, 3} 0] 
output [count Nd{0, 1, 1, 2, 2, 3} 1] 
output [count Nd{0, 1, 1, 2, 2, 3} 2] named "[N] 2"
output [count Nd{0, 1, 1, 2, 2, 3} 3] named "[N] 3"
}

\loop N over {1..6} {
output Nd6
}\

\function: ROLL:s count fives and above and subtract ones {
 result: (ROLL >= 5) - (ROLL = 1)
}

function: evaluate ROLL:s {
 SUCCESSES: ROLL >= 4
 if SUCCESSES > 0 { result: SUCCESSES + (ROLL = 10) }
 result: 0
}

function: evaluatee X:s {
 if X >= 12 { result: 3 }
 if X >= 10 { result: 2 }
 if X >= 7  { result: 1 }
 result: 0
}

function: dupes in DICE:s {
  D: 0
  loop X over {1..#DICE} {
    PREV: (X-1)@DICE
    CURR: X@DICE
    if (PREV = CURR | CURR = 6) { D: D + 1}
  }
  result: D
}

loop DICE over {2..2} {
 XS: DICE d6
 ROLL: [highest 2 of XS]
 output [evaluatee ROLL] named "[DICE]d"
}

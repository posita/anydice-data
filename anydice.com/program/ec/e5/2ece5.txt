DISASTER: 0 
DISADVANTAGE: 1 
MAINTAIN: 2 
LOW: 10
HIGH: 20 
DN: 1
DS: 20 

function: save ROLL:n {
 if ROLL <= LOW { result: DISASTER }
 if ROLL >= HIGH { result: MAINTAIN  }
 result: DISADVANTAGE
}

loop ADJ over {2..6}{ 
\output [save [lowest 1 of 2dDS] + ADJ] named"[DN]d[DS] + [ADJ] DisAdv"\
output [save [lowest 1 of DNdDS] + ADJ] named"[DN]d[DS] + [ADJ]" }
\output [save [highest 1 of 2dDS] + ADJ] named"[DN]d[DS] + [ADJ] Adv"\

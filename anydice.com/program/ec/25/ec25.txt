\ Attacker Diepool \   A: 6 

\ Defender Diepool \   D: 3

function: attacker DICE:s{
   COUNT: 0
 loop P over {1..6} {
  COUNT: COUNT + (P@{5,6} = DICE)
 }
 result: COUNT
}

function: defender DICE:s{
   COUNT: 0
 loop P over {1..6} {
  COUNT: COUNT + (P@{4,5,6} = DICE)
 }
 result: COUNT
}

\Attacker's Roll Result\ AR: [attacker Ad6]
\Defender's Roll Result\ DR: [defender Dd6]
\Attacker Net Hits\       X: AR - DR

output AR  named "Attacker's Hits Rolled"
output DR  named "Defender's Hits Rolled"
output X>0 named "Attacker's chance to hit"
output X   named "Attacker's net hits"

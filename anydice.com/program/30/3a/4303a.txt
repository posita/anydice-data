{
if
0>= 19 {result: 0}
else if O >= 17 {result: -1}
else if O >= 15 {result: -2}
else if O >= 13 {result: -3}
else if O >= 11 {result: -4}
else if O >= 9 {result: -5}
else if O >= 7 {result: -6}
else if O >= 5 {result: -7}
else if O >= 3 {result: -8}
else if O >= 1 {result: -9}
else {result: -99}
}

function: explodetopool N:n {
 R: 0
 if N = 6 { result: 1 + [explodetopool d6] }
 if N >= 3 {result: 1}
 result: R
}

function: legend N:n {
 R: 0
 if N >= 3 {result: 1}
 result: R 
}

LEGEND: [legend d6]
EXPLODE: [explodetopool d6]

loop N over {1..7}{
  R: N-1
  output RdLEGEND + 1dEXPLODE named "[N]D"
}

function: explodetopool N:n {
 R: 0
 if N = 1 { result: 0 }
 if N = 6 { result: 1 + [explodetopool d6] }
 if N >= 4 {result: 1}
 result: R
}

function: mythic N:n {
 R: 0
 if N >= 4 {result: 1}
 result: R 
}

MYTHIC: [mythic d6]
EXPLODE: [explodetopool d6]

loop N over {1..7}{
  loop D over {1..10}{
    R: N-1
    output (RdMYTHIC + 1dEXPLODE) >= D named "[N]D ≥ [D]"
  }
}

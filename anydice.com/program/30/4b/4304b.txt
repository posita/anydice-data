set "maximum function depth" to 80


function: chain N:n {
  if N = 1 {result: 0}
  if N = 2 {result: 0}
  if N = 3 {result: 0}
  if N = 4 {result: 1}
  if N = 5 {result: 1}
  if N = 6 {result: 1 + [chain d6]}
}

function: base N:n {
  if N = 1 {result: 0}
  if N = 2 {result: 1}
  if N = 3 {result: 1}
  if N = 4 {result: 1}
  if N = 5 {result: 1}
  if N = 6 {result: 2 + [chain d6]}
}

CHAIN_ADVANTAGE: 1
function: advantage N:n {
  if N = 1 {result: 0}
  if N = 2 {result: 0}
  if N = 3 {result: 0}
  if N = 4 {result: 1}
  if N = 5 {result: 1}
  if N = 6 & !CHAIN_ADVANTAGE {result: 1}
  if N = 6 &  CHAIN_ADVANTAGE {result: 1 + [advantage d6]}
}

loop B over {1..8}{
  loop A over {0..0}{
    output [lowest of Bd[base d6] + Ad[advantage d6] and 6+(2*B)+(A/2)] named "[B]dB + [A]dA"
  }
}

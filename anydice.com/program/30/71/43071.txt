AMOUNT: 1
DICE: 8
DROP_X_AND_LOWER: 2

function: reroll ROLL:s DROP_X_AND_LOWER DICE {
 if 1@ROLL > DROP_X_AND_LOWER {
  result: 1@ROLL
 }
 result: 1dDICE
}

X: 0
loop P over {1..AMOUNT} {
 X: [reroll 1dDICE DROP_X_AND_LOWER DICE] + X
}
output X named "[AMOUNT]d[DICE] drop [DROP_X_AND_LOWER] and below"
output AMOUNTdDICE named "[AMOUNT]d[DICE]"
output X - AMOUNTdDICE named "difference:"

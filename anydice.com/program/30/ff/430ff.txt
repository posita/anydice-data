function: score DICE:n ADV:n {
  if DICE = 0 {
    result: 0
  }

  result: [rollone d6 DICE ADV]
}

function: rollone VALUE:n DICE:n ADV:n {

  if ADV = 0 {
    if VALUE <= 4 {
      result: [score {DICE-1} ADV]
    }

    if VALUE = 5 {
      result: 1 + [score {DICE-1} ADV]
    }

    result: 2 + [score {DICE-1} ADV]
  }


  if ADV > 0 {

    if VALUE = 1 {
      result: [score {DICE-1} ADV]
    }

    if VALUE = 2 {
      result: 1 + [score {DICE-1} {ADV-3}]
    }

    if VALUE = 3 {
      result: 1 + [score {DICE-1} {ADV-2}]
    }

    if VALUE = 4 {
      result: 1 + [score {DICE-1} {ADV-1}]
    }

    if VALUE = 5 {
      result: 1 + [score {DICE-1} ADV]
    }

    result: 2 + [score {DICE-1} ADV]
  }


  if ADV < 0 {

    if VALUE = 1 {
      result: -1 + [score {DICE-1} {ADV+1}]
    }

    if VALUE <= 4 {
      result: [score {DICE-1} ADV]
    }

    if VALUE = 5 {
      result: [score {DICE-1} {ADV+1}]
    }

    result: 2 + [score {DICE-1} ADV]
  }
}


output [score 6 {-2}]

\ Re-roll  lower  than X once \

NUMBER_OF_DICE: 1
TYPE_OF_DICE: 8
REROLL_X_AND_LOWER: 2

function: reroll ROLL:s REROLL_X_AND_LOWER TYPE_OF_DICE {
 if 1@ROLL > REROLL_X_AND_LOWER {
  result: 1@ROLL
 }
 result: 1 d TYPE_OF_DICE
}

X: 0
loop P over {1..NUMBER_OF_DICE} {
 X: [reroll 1 d TYPE_OF_DICE REROLL_X_AND_LOWER TYPE_OF_DICE] + X
}
output X named "[NUMBER_OF_DICE] d [TYPE_OF_DICE] drop [REROLL_X_AND_LOWER] and below"
output NUMBER_OF_DICE d TYPE_OF_DICE named "[NUMBER_OF_DICE] d [TYPE_OF_DICE]"
output X - NUMBER_OF_DICE d TYPE_OF_DICE named "difference:"

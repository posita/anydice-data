function: clamp ROLL:n {
  if ROLL < 1 { result: 1 }
  result: ROLL
}

\ 1d6 \
output [clamp (1d6 - 2)] named "1d6 - 2 (min 1)"
output [clamp (1d6 - 1)] named "1d6 - 1 (min 1)"
loop S over {0, 1, 2} {output 1d6+S named "1d6+[S]"}

\ 2d6 \
loop S over {-1, 0, 1, 2} {output 2d6+S named "2d6+[S]"}

\ 3d6 \
loop S over {-1, 0, 1, 2} {output 3d6+S named "3d6+[S]"}

\ 4d6 \
loop S over {-1, 0, 1, 2} {output 4d6+S named "4d6+[S]"}

\ 5d6 \
loop S over {-1, 0, 1, 2} {output 5d6+S named "5d6+[S]"}

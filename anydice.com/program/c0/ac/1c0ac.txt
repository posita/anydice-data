AC:              19  \ target AC: attack hits if ATK + HIT >= AC \
ATK:           1d20  \ attack roll: use 1@2d20 for advantage or 2@2d20 for disadvantage \
ATKBONUS:        11  \ bonus to attack roll for base and TWF attacks \
CRIT:            20  \ minimum attack roll to crit \
CRITMULTI:        2  \ number of times to roll damage dice on a crit \
DMG:            1d6  \ weapon damage die rolled on successful hit for base attacks \
TWFDMG:         1d6  \ weapon damage die rolled on successful hit for TWF attacks \
DMGMODS:          5  \ integer modifiers added to damage rolls for base attacks \
TWFDMGMODS:       0  \ integer modifiers added to damage rolls for TWF attacks \
DMGBONUSDICE:     0  \ extra dice added on top of DMG to each damage roll \
TWFDMGBONUSDICE:  0  \ extra dice added on top of DMG to each damage roll for TWF attacks \
DMGOPTDICE:       0  \ extra dice added on top of DMG to first hit of base attack \
DMGOPTMODS:       0  \ integer modifiers added to first hit of base attack \
BRUTALMULTI:      0  \ number of barbarian's brutal critical dice \
BRUTALDIE:        0  \ die that is rolled for barbarian's brutal critical for base attacks \
TWFBRUTALDIE:     0  \ die that is rolled for barbarian's brutal critical for TWF attacks \
GWMASTER:         0  \ grants an extra attack on a base attack crit which overrides TWF attacks \

set "maximum function depth" to 20

function: attackroll ROLL:n {
  \ return the number of times damage is applied on an attack roll of ROLL \
  if ROLL <= 1 { result: 0 }     \ natural 1 is always a miss \
  if ROLL >= CRIT { result: CRITMULTI }  \ critical hit multiplies damage dice \
  result: ROLL + ATKBONUS >= AC
}

ATKRESULT: [attackroll ATK]  \ optimization: precalculate the miss/hit/crit probabilities \
BONUSATKRESULT: [attackroll ATK] \ allows for different attack mod for TWF attacks \

function: reroll NEWDIE:d for ORIGDIE:n { \ Great Weapon Fighting damage die rerolls on 1 or 2 \
 if ORIGDIE <= 2 { result: NEWDIE }
 result: ORIGDIE
}

function: baseattack N:n times and bonusattack M:n times {
  result: [roll ATKRESULT to baseattack N times and roll BONUSATKRESULT to bonusattack M times]
}

function: roll BASEDMGMULTI:n to baseattack N:n times and roll BADMGMULTI:n to bonusattack M:n times {
    TOTALDMG: 0
	GWMHIT: 0
    if N <= 0 & M <= 0 {
      result: 0
    } else {
		if N > 0 {
			if BASEDMGMULTI > 0 {
				TOTALDMG: TOTALDMG + BASEDMGMULTI d DMGOPTDICE + DMGOPTMODS
				if GWMASTER > 0 & BASEDMGMULTI > 1 {
					GWMASTER: 0
					result: [roll ATKRESULT to baseattack N+1 times and roll BONUSATKRESULT to bonusattack 0 times]
				}
			}
			loop I over {1..N} {
				if BASEDMGMULTI > 0 {
					if BASEDMGMULTI > 1 {
						TOTALDMG: TOTALDMG + BASEDMGMULTI d (DMG + DMGBONUSDICE) + DMGMODS + (BRUTALMULTI * BRUTALDIE)
					} else {
						TOTALDMG: TOTALDMG + BASEDMGMULTI d (DMG + DMGBONUSDICE) + DMGMODS
					}
				} else {
					result: [roll ATKRESULT to baseattack N-1 times and roll BONUSATKRESULT to bonusattack M times]
				}
			}
		}
		if M > 0 {
			loop I over {1..M} {
				if BADMGMULTI > 0 {
					if BADMGMULTI > 1 {
						TOTALDMG: TOTALDMG + BADMGMULTI d (TWFDMG + TWFDMGBONUSDICE) + TWFDMGMODS + (BRUTALMULTI * TWFBRUTALDIE)
					} else {
						TOTALDMG: TOTALDMG + BADMGMULTI d (TWFDMG + TWFDMGBONUSDICE) + TWFDMGMODS
					}
				} else {
					result: [roll ATKRESULT to baseattack N times and roll BONUSATKRESULT to bonusattack M-1 times]
				}
			}
		}
		result: TOTALDMG
	}
}

function: evaluate ROLL:s{
 SUCCESS: ROLL = 6
 FAIL: ROLL = 1
 if SUCCESS >= FAIL+9 { result: 1d6 }
 if SUCCESS >= FAIL+6 { result: 1d8 }
 if SUCCESS >= FAIL+3 { result: 1d10 }
 if SUCCESS >= FAIL { result: 1d12 }
 if SUCCESS < FAIL { result: 1d20 }
 result: 0
}

function: ROLL:n reroll less than MIN:n and more than FAIL:n as REROLL:d {
  if ROLL < MIN { if ROLL > FAIL { result: REROLL } }
  else { result: ROLL }
}

X: [d6 reroll less than 6 and more than 1 as d6]

loop DICE over {1..10} {
 output [evaluate DICE dX] named "Fate dice for [DICE]d6"
}

function: damage R:n adding N:n {
   if R+N < 0 { result: 0 } else {result: R+N}
}

function: R:n to hit on BS:n{
   if R < BS { result: 0 } else { result: 1 }
}

function: reroll X:n to hit R:n on BS:n{
   if R = X { result: [1d6 to hit on BS] } else { result: [R to hit on BS] }
   
}

function: reroll fail to hit R:n on BS:n{
   if R < BS { result: [1d6 to hit on BS] } else { result: [R to hit on BS] }
   
}

function: multihit X:n{
 TOTAL: 0
 loop N over{1..X}{
  HIT: [1d6 to hit on 3]
  TOTAL: TOTAL+HIT
 }
 result: TOTAL
}

function: multireroll X:n{
 TOTAL: 0
 loop N over{1..X}{
  HIT: [reroll 1 to hit 1d6 on 3]
  TOTAL: TOTAL+HIT
 }
 result: TOTAL
}

output [1d6 to hit on 3] named "hit"
output [reroll 1 to hit 1d6 on 3] named "hit reroll 1s"
output [reroll fail to hit 1d6 on 3] named "hit reroll fail"

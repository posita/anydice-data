\Glaive + PAM + GWF+ GWM + Hex\
function: gwf A:s as B:n
{
   RES: 0

   loop C over {1..#(A)} 
   {
      if C@A < 3
      {
         RES: RES + 1dB
      }
      else
      {
         RES: RES + C@A
      }
   }

   result: RES
}
function: crit A:s as B:n
{
   loop C over {1..#(A)}
   {
      if C@A = 20
      {
         result: B
      }
      else
      {
         result: 0
      }
   }
}

function: ROLL:n gwf REROLL:d reroll SEQUENCE:s{
  if [SEQUENCE contains ROLL] {
    result: REROLL
  }
  else {
    result: ROLL
  }
}

STR: 4
GREATSWORD_A: [d6 gwf d6 reroll {1,2}] + [d6 gwf d6 reroll {1,2}] + STR
GREATSWORD_B: [d6 gwf d6 reroll {1,2}] + [d6 gwf d6 reroll {1,2}] + STR
FLAIL_A: 1d8 + 2 + STR
FLAIL_B: 1d8 + 2 + STR
WHIP_A: 1d4 + 2 + STR
WHIP_B: 1d4 + 2 + STR
DAGGER: 1d4
HANDAXE: 1d6
FEAT_PENALTY: -1

\ Extra Attack \
output GREATSWORD_A + GREATSWORD_B named "Greatsword"
output WHIP_A + WHIP_B named "Whip"
output WHIP_A + WHIP_B + DAGGER + (2*FEAT_PENALTY) named "Whip & Dagger"
output FLAIL_A + FLAIL_B named "Flail"
output FLAIL_A + FLAIL_B + HANDAXE + (2*FEAT_PENALTY) named "Flail & Handaxe"

\ Extra Attack and Divine Favor \
output GREATSWORD_A + GREATSWORD_B + 2d4 named "Greatsword (Favor)"
output WHIP_A + WHIP_B + 2d4 named "Whip (Favor)"
output WHIP_A + WHIP_B + DAGGER + 3d4 + (2*FEAT_PENALTY) named "Whip & Dagger (Favor)"
output FLAIL_A + FLAIL_B + 2d4 named "Flail (Favor)"
output FLAIL_A + FLAIL_B + HANDAXE + 3d4 + (2*FEAT_PENALTY) named "Flail & Handaxe (Favor)"

\ Extra Attack and Improved Divine Smite \
output GREATSWORD_A + GREATSWORD_B + 2d8 named "Greatsword (Improved)"
output WHIP_A + WHIP_B + 2d8 named "Whip (Improved)"
output WHIP_A + WHIP_B + DAGGER + 3d8 + (2*FEAT_PENALTY) named "Whip & Dagger (Improved)"
output FLAIL_A + FLAIL_B + 2d8 named "Flail (Improved)"
output FLAIL_A + FLAIL_B + HANDAXE + 3d8 + (2*FEAT_PENALTY) named "Flail & Handaxe (Improved)"

\ Extra Attack, Improved Divine Smite, and Divine Favor \
output GREATSWORD_A + GREATSWORD_B + 2d8 + 2d4 named "Greatsword (Improved+Favor)"
output WHIP_A + WHIP_B + 2d8 + 2d4 named "Whip (Improved+Favor)"
output WHIP_A + WHIP_B + DAGGER + 3d8 + 3d4 + (2*FEAT_PENALTY) named "Whip & Dagger (Improved+Favor)"
output FLAIL_A + FLAIL_B + 2d8 + 2d4 named "Flail (Improved+Favor)"
output FLAIL_A + FLAIL_B + HANDAXE + 3d8 + 3d4 + (2*FEAT_PENALTY) named "Flail & Handaxe (Improved+Favor)"

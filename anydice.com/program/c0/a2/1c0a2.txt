MISS: 0
HIT: 1
CRIT: 2
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: 0 }
 if ROLL = 20 { result: CRIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}
function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRIT { result: CRITDAMAGE }
 if ATTACKRESULT = HIT { result: HITDAMAGE }
 result: MISSDAMAGE
}
DEFENCE: 20
function: attack {
 result: [damage for [attack d20 vs DEFENCE]]
}
function: aattack {
 result: [damage for [attack [highest 1 of 2d20] vs DEFENCE]]
}

set "maximum function depth" to 10

function: replace lowest A:s with X:n{
  if #A@A < X & X >= 0 {result: {1..(#A-1)}@A + X}
  result: A
  }

function: jam chance A:n TEN:n{
if MODE = 0 & RELIABLE {if  TEN = 1 & A > 96 {result:  0} result: 1}
else if MODE = 0 {result: A<96}
else if RELIABLE {if TEN = 1 & A > 94 {result:  0} result: 1}
result: A<94
}

function: scatter extra hits DOS:n{
if !SCATTER {result: 0}
result: DOS/2
}

function: X:n dos A:n Y:n{
if X >= A & Y { result: (X - A)/10 }
result: -1
}

function: hits DOS:n{
if MODE = 0 & DOS >= 0 {result: 1 + [scatter extra hits DOS]}
else if MODE = 1 & DOS >= 0
{
  if ROF <= DOS/2 {result: ROF + [scatter extra hits DOS]}
  else {result: DOS/2 + 1 + [scatter extra hits DOS]}
}
else if MODE = 2 & DOS >= 0
{
  if ROF <= DOS {result: ROF + [scatter extra hits DOS]}
  else {result: DOS + 1 + [scatter extra hits DOS]}
}
result: 0
}

function: soak DOS:n{
if DOS > 2 & RAZORSHARP {result: [highest of AP-(PEN*2) and 0] + TB}
else {result: [highest of AP-PEN and 0] + TB}}

function: ag test X:n{
result: AG < X
}

function: flamer damage{
result: NUMBEROFDTENSd[explode d10] + BONUS
}

function: damage{
WITH_TEARING: [highest NUMBEROFDTENS of (NUMBEROFDTENS+1)d[highest of [explode d10] and PROVEN]]
WITHOUT_TEARING: NUMBEROFDTENSd[highest of [explode d10] and PROVEN]
result:  (WITH_TEARING*TEARING) + (WITHOUT_TEARING*!TEARING) + BONUS}

function: flamer wounds caused AG:n{
if AG {result: [highest of [flamer damage] - [soak DOS] and 0]}
result: 0
}


function: wounds caused{
if FLAMER{result: [flamer wounds caused [ag test d100]]}
DAMAGE: [replace lowest [damage] with DOS]
result: [hits DOS] * [highest of (DAMAGE - [soak DOS]) and 0]
}


function: will it kill{
DAMAGE: [replace lowest [damage] with DOS]
WOUNDSCAUSED: [highest of DAMAGE - [soak DOS] and 0]
result: (NUMBEROFDTENS*4)@{WOUNDSCAUSED} > WOUNDS}

function: kills DOS:s{
if FLAMER{ result: [kills flamer] }
if [will it kill]{result: [hits DOS]*(WOUNDS <= [wounds caused])}                   
result: 100*[wounds caused]/WOUNDS
}

function: kills flamer{
if [maximum of [flamer wounds caused[ag test d100]]]>WOUNDS
{result: (WOUNDS <= [flamer wounds caused[ag test d100]])}
result: 100*[flamer wounds caused[ag test d100]]/WOUNDS
}

function: flamer horde kills{
QUARTER: FLAMERRANGE/4
if FLAMERRANGE != QUARTER*4 {QUARTER: QUARTER + 1}
HITS: QUARTER + 1d5
result: HITS * ([flamer wounds caused[ag test d100]]>0)
}

if FLAMER {result: [flamer horde kills]}
DAMAGE: [replace lowest [damage] with DOS]
WOUNDSCAUSED: [highest of DAMAGE - [soak DOS] and 0]
if BLAST {result: ((DOS > -1)*BLAST+ ([hits DOS]>0)*EXPLOSIVEDAMAGE)*(WOUNDSCAUSED > 0) + ([hits DOS]>0)*DEVISTATING}
result: (([hits DOS]+([hits DOS]>0)*EXPLOSIVEDAMAGE)*(WOUNDSCAUSED > 0)) + ([hits DOS]>0)*DEVISTATING}

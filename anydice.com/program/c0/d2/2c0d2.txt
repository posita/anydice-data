function: evaluate ROLL:s {
 SUCCESS: ROLL = 6
 FAIL: ROLL = 1
 if SUCCESS > FAIL { result: 1 }
 if SUCCESS = FAIL { result: 0 }
 if SUCCESS < FAIL { result: -1 }
 result: 0
}

function: ROLL:n reroll less than MIN:n and more than FAIL:n as REROLL:d {
  if ROLL < MIN { if ROLL > FAIL { result: REROLL } }
  else { result: ROLL }
}

X: [d6 reroll less than 6 and more than 1 as d6]

loop DICE over {1..10} {
 output [evaluate DICE dX] named "[DICE]d"
}

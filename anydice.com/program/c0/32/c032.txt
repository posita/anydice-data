MISS: 0
HIT: 1
TWOHITS: 3
OFFHANDHIT: 4
CRITICALHIT: 2

function: assassinattack ROLL:n and ROLLB:n vs DEFENSE:n {
 if ROLL = 1 & ROLLB = 1 { result: MISS }
 if ROLL = 1 & ROLLB + ATTACK < DEFENSE {result: MISS}
 if ROLL + ATTACK < DEFENSE & ROLLB = 1 {result: MISS}
 if ROLL < DEFENSE & ROLLB < DEFENSE {result: MISS}
 if ROLL = 1 & ROLLB + ATTACK >= DEFENSE {result: OFFHANDHIT }
 if ROLL + ATTACK >= DEFENSE & ROLLB = 1 {result: HIT}
 if ROLL + ATTACK >= DEFENSE & ROLLB + ATTACK < DEFENSE { result: HIT }
 if ROLL + ATTACK < DEFENSE & ROLLB + ATTACK >= DEFENSE { result: OFFHANDHIT }
 if ROLL + ATTACK >= DEFENSE & ROLLB + ATTACK >= DEFENSE {result: TWOHITS}
 result: MISS
}

function: assassinate for ATTACKRESULT:n {
 if ATTACKRESULT = HIT {
  result: HITDAMAGE + SNEAKDAMAGE + MODDAMAGE
 }
 if ATTACKRESULT = OFFHANDHIT {
  result: HITDAMAGE + SNEAKDAMAGE
}
 if ATTACKRESULT = TWOHITS {
  result: HITDAMAGE + HITDAMAGE + SNEAKDAMAGE
}
 result: MISSDAMAGE
}

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

TRESHOLD: 2

function: one DMG_ONE:n two DMG_TWO:n three DMG_THREE:n four DMG_FOUR:n {
if DMG_ONE <= TRESHOLD {
if DMG_TWO <= TRESHOLD {
result: DMG_THREE + DMG_FOUR
} else {
result: DMG_TWO + DMG_THREE
}
} else if DMG_TWO <= TRESHOLD {
if DMG_ONE <= TRESHOLD {
result: DMG_THREE + DMG_FOUR
} else {
result: DMG_ONE + DMG_THREE
}
} else {
result: DMG_ONE + DMG_THREE
}
}

output [one d6 two d6 three d6 four d6] named "Great Weapon Fighting"

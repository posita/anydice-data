loop LVL over {1..20} {
 BAB: 3*LVL/4
 DEX: 5+(LVL/8)
 function: skirmish {
  B: (LVL+2)/3
  if (LVL > 4) {
   B: B*2
  }
  result: Bd6
 }
 function: boosted skirmish {
  B: (LVL+1)/2
  if (LVL > 4) {
   B: B*2
  }
  result: Bd6
 }
 function: full attack {
  if (BAB > 10) {
   result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
  } else {
   if (BAB > 5) {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   } else {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   }
  }
 }
 function: single attack STEP {
  if (BAB > 5) {
   if (STEP != 0 & LVL > 4) {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   } else {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))
   }
  } else {
   if (STEP != 0 & LVL > 4) {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   } else {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))
   }
  }
 }
 output [single attack 0] named "Single Attack no Step Lv [LVL]"
 output [single attack 1] named "Single Attack w/ Step Lv [LVL]"
 output [full attack] named "Full Attack Rapid Shot Lv [LVL]"
 output [single attack 0]+[skirmish] named "Current Skirmish no Step Lv [LVL]"
 output [single attack 1]+[skirmish] named "Current Skirmish w/ Step Lv [LVL]"
 output [single attack 0]+[boosted skirmish] named "Thief Skirmish no Step Lv [LVL]"
 output [single attack 1]+[boosted skirmish] named "Thief Skirmish w/ Step Lv [LVL]"
 output 0 named ""
}

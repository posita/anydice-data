\DIETYPE = Amount and types of dice rolled, FAIL/PASS/SUCCEED/CRIT = Minimum Result to pass Threshold, AP = Action Point Addition]\

\ROLLS AND BONUS\
DIETYPE: 1d100
AP: 0

\THRESHOLDS\
FAIL: 0
PASS: 50
SUCCEED: 110
CRIT: 150

function: range on DIE:n between LOW:n and HIGH:n {
  if DIE >= LOW & DIE <= HIGH { result: 1 }
  result: 0
}

output [range on DIETYPE + AP between FAIL and PASS-1] named "FAIL"
output [range on DIETYPE + AP between PASS and SUCCEED-1] named "PASS"
output [range on DIETYPE + AP between SUCCEED and CRIT-1] named "SUCCESS"
output [range on DIETYPE + AP between CRIT and 900] named "CRIT"

loop LVL over {1..20} {
 BAB: 3*LVL/4
 DEX: 5+(LVL/8)
 function: skirmish A {
  B: (A+2)/3
  result: Bd6
 }
 function: boosted skirmish A {
  B: (A+1)/2
  result: Bd6
 }
 function: full attack A {
  if (A > 10) {
   result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
  } else {
   if (A > 5) {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   } else {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   }
  }
  result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
 }
 function: single attack A {
  if (A > 5) {
   result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
  } else {
   result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))
  }
  result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))
 }
 output [single attack BAB] named "Single Attack Lv [LVL]"
 output [full attack BAB] named "Full Attack Lv [LVL]"
 output [single attack BAB]+[skirmish LVL] named "Current Skirmish Lv [LVL]"
 output [single attack BAB]+[boosted skirmish LVL] named "Proposed Skirmish Lv [LVL]"
}

REFERENCE: 10
TUTOR: 0

function: mage with S:n arcana research level L:n spell {
  SKILL_CHECK: 1d20 + S
  BONUSES: SKILL_CHECK + TUTOR + REFERENCE
  DC: L + 15
  EFFECT: BONUSES - DC + 1
  RES: {}
  loop I over {1..#{EFFECT}}{
    T: I@{EFFECT}
    if T < 0 & T > -5 {
      RES: {RES, 0}
    } 
    else {
      RES: {RES, I@{EFFECT}}
    }
  }
  result: d{RES}
}

function: as D:d for N:n days {
  O: 0
  loop I over {1..N} {
    O: O + D
  }
  result: O
}

output 10d{[mage with 5 arcana research level 5 spell]}

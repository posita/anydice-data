function: attackroll ROLL:n a CRITROLL a CRITDAMAGE:n a PROFICIENCY a ATTACKMOD a ACCBONUS a AC:n a DAMAGE:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}


output 10d[attackroll 1d20 a 20 a 2d8 a 3 a 2 a 0 a d{10..20} a 1d8] named "test"

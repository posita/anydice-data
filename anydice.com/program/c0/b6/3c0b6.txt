\ VTM Storytelling, double 10s \

S: 0 \ # of Enemy's Dice \
TARGET: 7 \ Number needed to count as a hit \
DOUBLEON: 10 \ Numbers above which you double successes \

function: vroll ROLL:s  {
 result: (ROLL >= TARGET) + (ROLL >= DOUBLEON)
}

loop N over {2..10}{
 output [vroll Nd10] - [vroll Sd10] named "[N]d"
}

set "maximum function depth" to 3

function: roll X:n {
 if X = 1 {result: [malus d10] }
 if X = 10 {result: X+[bonus d10]}
 result: X
}

function: malus X:n {
 if X > 4 {result: 1}
 if X > 1 {result: X-5}
 result: -5 + [malus d10]
}

function: bonus X:n {
 if X < 6 {result: 0}
 if X < 10 {result: X-5}
 result: 5 + [bonus d10]
}

output [roll d10]

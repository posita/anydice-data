function: atk ATK_BONUS:n DO_FLANK_BONUS:n FLANK_BONUS:n ADV:n {
    if DO_FLANK_BONUS = 0 {
        FLANK_BONUS: 0
    }
    
    if ADV = 1 {
        result: [highest 1 of 2d20] + ATK_BONUS + FLANK_BONUS
    } else if ADV = -1 {
        result: [lowest 1 of 2d20] + ATK_BONUS + FLANK_BONUS
    } else {
        result: 1d20 + ATK_BONUS + FLANK_BONUS
    }
}

output [atk 15 0 3 0] named "Normal"
output [atk 15 0 3 1] named "With advantage"
output [atk 15 0 3 (-1)] named "With disadvantage"

output [atk 15 1 3 0] named "With flanking"
output [atk 15 1 3 1] named "With flanking and advantage"
output [atk 15 1 3 (-1)] named "With flanking and disadvantage"

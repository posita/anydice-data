function: gwf A:s as B:n
{
   RES: 0

   loop C over {1..#(A)}
   {
      if C@A < 3
      {
         RES: RES + 1dB
      }
      else
      {
         RES: RES + C@A
      }
   }

   result: RES
}
function: crit D:s as E:n
{
   loop F over {1..#(D)}
   {
     
      if F@D = 20
      {
         result: E   
      }
      else
      {
         result: 0   
      }
   }
}

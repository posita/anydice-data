\DIETYPE = Amount and types of dice rolled, FAIL/PASS/SUCCEED/CRIT = Minimum Result to pass Threshold, AP = Action Point Addition]\

DIETYPE: 1d100
FAIL: 0
PASS: 50
SUCCEED: 110
CRIT: 150
AP: 0

function: range on DIE:n between LOW:n and HIGH:n {
  if DIE >= LOW & DIE <= HIGH { result: 1 }
  result: 0
}


output [range on DIETYPE + AP between FAIL and PASS-1]
output [range on DIETYPE + AP between PASS and SUCCEED-1]
output [range on DIETYPE + AP between SUCCEED and CRIT-1]
output [range on DIETYPE + AP between CRIT and 900]

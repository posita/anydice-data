loop LVL over {1..20} {
 BAB: 3*LVL/4
 DEX: 5+(LVL/8)
 function: skirmish {
  B: (LVL+2)/3
  result: Bd6
 }
 function: boosted skirmish {
  B: (LVL+1)/2
  result: Bd6
 }
 function: full attack {
  if (BAB > 10) {
   result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
  } else {
   if (BAB > 5) {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   } else {
    result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
   }
  }
  result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
 }
 function: single attack {
  if (BAB > 5) {
   result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))+(2d4+DEX+(LVL/4)+(1+(BAB/4)))
  } else {
   result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))
  }
  result: (2d4+DEX+(LVL/4)+(1+(BAB/4)))
 }
 output [single attack] named "Single Attack Lv [LVL]"
 output [full attack] named "Full Attack Lv [LVL]"
 output [single attack]+[skirmish] named "Current Skirmish Lv [LVL]"
 output [single attack]+[boosted skirmish] named "Proposed Skirmish Lv [LVL]"
}

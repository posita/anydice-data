\ 
0: Abject Failure (1-3)
1: Success, but with complication (4-5)
2: Clear success, no complication (6-8)
3: Critical (8-10)
4: Supercritical (11-12)
\

function: roll ROLL:s cut CUT:n {  
  FINAL: 1@ROLL
  if CUT > #ROLL {CUT: #ROLL}
  if CUT = 1 { FINAL: 2@ROLL }
  if CUT = 2 { FINAL: 3@ROLL }
  if FINAL = 1 { result: 0}
  if FINAL = 2 { result: 0}
  if FINAL = 3 { result: 0}
  if FINAL = 4 { result: 1}
  if FINAL = 5 { result: 1}
  if FINAL = 6 { result: 2}
  if FINAL = 7 { result: 2}
  if FINAL = 8 { result: 2}
  if FINAL = 9 { result: 3}
  if FINAL = 10 { result: 3}
  if FINAL = 11 { result: 4}
  if FINAL = 12 { result: 4}  
  result: 0
  
}

loop SKILL over {4, 6, 8, 10, 12} {
    loop STAT over {1..5} {
         CUTS: {0,1,2}         
        if STAT = 1 {
          CUTS: {0}
        }
        if STAT = 2 {
            CUTS: {0,1}
        }
        loop CUT over CUTS {
            output [roll STATdSKILL cut CUT] named "[STAT]d[SKILL] (cut: [CUT])"
        }  
      }
}

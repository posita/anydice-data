function: crop SEQUENCE:n MIN:n MAX:n {                  \ limit the display range between [min;max] \
 if (SEQUENCE < MIN){result: MIN}
 if (SEQUENCE > MAX){result: MAX}
 result: SEQUENCE
}
function: crop min SEQUENCE:n LIM:n {                  \ limit the display range max \
 if (SEQUENCE < LIM){result: LIM}
 result: SEQUENCE
}
function: crop max SEQUENCE:n LIM:n {                  \ limit the display range min \
 if (SEQUENCE > LIM){result: LIM}
 result: SEQUENCE
}
function: halve SEQUENCE:n {                  \ divide per 2, rounded up \
 DIV: SEQUENCE/2
 TEMP: DIV*2
 if (DIV=TEMP){result: DIV}
 result: (SEQUENCE+1)/2
}

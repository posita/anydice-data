function: for armor A:n and injury I:n and base B:n {
  if I < 0 {
    result: [lowest B of (B-I)d6]+A
  }
  else if I > 0 {
    result: [highest B of (B+I)d6]+A
  }
  result: Bd6+A
}

NO_EFFECT: 2
MINOR_HIT: {2..6}
DOWN: {7,8}
OUT_OF_ACTION: 9
function: get chances for ROLL:n {
  if ROLL < NO_EFFECT { result: 0 }
  if ROLL = MINOR_HIT { result: 1 }
  if ROLL = DOWN { result: 2 }
  result: 3 \ Out of Action \
}

loop B over {2..4} {
  loop A over {-3..3} {
    loop I over {-3..3} {
      output [get chances for [for armor A and injury I and base B]] named "[A]a [I]i [B]b"
    }
  }
}

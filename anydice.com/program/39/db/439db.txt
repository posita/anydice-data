MOD: 10
DC: 16
TN: 10
CR: 7
DIE: 4

\loop P over {1..5} {
output Pd(d10 <= TN) named "[P]d12"
}\

function: failure ROLL:n {
result: (ROLL <= DC-5)
}

function: partsuccess ROLL:n {
result: (ROLL >= DC-4) & (ROLL <= DC-1)
}

function: success ROLL:n {
result: (ROLL >= DC) & (ROLL <= DC+9)
}

function: crit ROLL:n {
result: (ROLL >= DC+10)
}

output [failure 2d10+MOD] named "Fail"
output [partsuccess 2d10+MOD] named "Partial"
output [success 2d10+MOD] named "Success"
output [crit 2d10+MOD] named "Crit"

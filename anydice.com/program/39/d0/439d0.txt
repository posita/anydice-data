function: outcome X:n vs Y:n {
 result: (X > Y) - (X < Y)
}

function: contest A:n vs B:n {
 \ A = highest of 3d8
   B = highest of 2d8 \

 if B < A {
  \ B lost, so reroll the entire 2d8 pool once \
  result: [outcome [highest 1 of 2d8] vs A]
 }

 result: [outcome B vs A]
}

output [contest [highest 1 of 3d8] vs [highest 1 of 2d8]]
 named "2d8 side: -1 Lose, 0 Tie, +1 Win"


function: reroll X:n below N:n with D:d {
  if X <= N {result: D}
  result: X
}


PROF:3
GWFDFOUR: [reroll 1d4 below 2 with 1d4]
SAVDFOUR: [highest 1 of 2d4]
GWFDSIX: [reroll 1d6 below 2 with 1d6]
SAVDSIX:[highest 1 of 2d6]
GWFDEIGHT: [reroll 1d8 below 2 with 1d8]
SAVDEIGHT:[highest 1 of 2d8]


function: DICE:d on ROLL:n plus MOD:n vs AC:n {
  if ROLL = 1 { result: 0 }
  if ROLL >= 20 { result: 2dDICE+MOD }
  if ROLL + MOD+PROF >= AC { result: DICE+MOD }
  result: 0
}

function: DICE:d on ROLL:n plus MOD:n vs AC:n crit on CRIT:n {
  if ROLL = 1 { result: 0 }
  if ROLL >= CRIT { result: 2dDICE+MOD }
  if ROLL + MOD+PROF >= AC { result: DICE+MOD }
  result: 0
}

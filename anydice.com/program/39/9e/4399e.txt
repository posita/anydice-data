\ ============================================================
   SIMPLE DAMAGE CALCULATOR

   Change ONLY the four values under "USER SETTINGS".

   ATTACKS = Number of attacks
   HIT     = Required roll to hit
   WOUND   = Required roll to wound
   DAMAGE  = Damage dealt by each successful wound

   Example:
   ATTACKS: 6
   HIT:     3
   WOUND:   4
   DAMAGE:  2

   = 6 attacks, hitting on 3+, wounding on 4+, Damage 2
   ============================================================ \


\ ---------- USER SETTINGS ---------- \

ATTACKS: 6
HIT:     3
WOUND:   4
DAMAGE:  2


\ ---------- CALCULATOR ---------- \

function: single attack hitting on HIT:n wounding on WOUND:n for DAMAGE:n damage {
    result: (d6 >= HIT) * (d6 >= WOUND) * DAMAGE
}

SINGLE_ATTACK: [single attack hitting on HIT wounding on WOUND for DAMAGE damage]

TOTAL_DAMAGE: ATTACKS d SINGLE_ATTACK


\ ---------- OUTPUT ---------- \

output TOTAL_DAMAGE named "[ATTACKS] Attacks | Hit [HIT]+ | Wound [WOUND]+ | Damage [DAMAGE]"

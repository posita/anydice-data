\ SAVE FUNCTION \
function: save ROLL:n vs DEFENSE:n mod MOD:n dmg DAMAGE plus BONUS {

 DMG_TOTAL: DAMAGE + BONUS

 if ROLL + MOD <= DEFENSE - 10 { RESULT: 0 } \ crit fail \
 if ROLL + MOD < DEFENSE { RESULT: 1 } \ fail \
 if ROLL + MOD >= DEFENSE { RESULT: 2 } \ success \
 if ROLL + MOD >= DEFENSE + 10 { RESULT: 3 } \ crit success \

 if ROLL = 1 & RESULT != 0 { RESULT: RESULT - 1 } \ 1 = DoS - 1 \
 if ROLL = 20 & RESULT != 3 { RESULT: RESULT + 1 } \ 20 = DoS + 1 \

 if RESULT = 0 { result: (DMG_TOTAL) * 2 }
 if RESULT = 1 { result: (DMG_TOTAL) }
 if RESULT = 2 { result: (DMG_TOTAL) / 2 }
 if RESULT = 3 { result: 0 }
}

function: roll N:n {
if N >= 0 {result: [hr (N+1)d6]}
if N < 0 {result: [lr (-N+1)d6]}
}

function: hr S:s {
if 2@S = 6 {result: 6}
if 1@S = 6 {result: 5}
if 1@S >= 4 {result: 4}
if 1@S >= 2 {result: 3}
result: 2
}

function: lr S:s {
L: #S
if (L-1)@S = 1 {result: 1}
if L@S = 1 {result: 2}
if L@S <= 3 {result: 3}
if L@S <= 5 {result: 4}
result: 5
}

loop X over {-10..10}{
output [roll X] named "[X]"
}

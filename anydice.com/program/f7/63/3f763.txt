

SKILLONE:  1d20
SKILLTWO:  1d12
SKILLTHREE: 1d10
SKILLFOUR: 1d8
SKILLFIVE: 1d6

function: flipinner ROLL:n SIDES:n DEPTH:n {
  COIN: d2 - 1            
  if DEPTH >= 4 { 
    result: COIN 
  }
  if ROLL = SIDES {
    ROLLTWO: d(SIDES)
    NEWDEPTH: DEPTH + 1
    result: COIN + [flipinner ROLLTWO SIDES NEWDEPTH ]
  }
  result: COIN
}

function: flip DICE:d DEPTH:n {
  result: [flipinner DICE [maximum of DICE] DEPTH]
}


function: check ABILITY: n SKILL: n {
  DIE: SKILLFIVE
  if SKILL = 1 { DIE: SKILLONE }
  if SKILL = 2 { DIE: SKILLTWO }
  if SKILL = 3 { DIE: SKILLTHREE }
  if SKILL = 4 { DIE: SKILLFOUR }
  if SKILL = 5 { DIE: SKILLFIVE }

  SUCCESSES: 0
  loop N over {1..ABILITY} {
    SUCCESSES: SUCCESSES + [flip DIE 0]
  }
  result: SUCCESSES
}

output [check 1 1] named "A1, Sd20"

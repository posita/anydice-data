\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: sattackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGES } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGES }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{17}

ATTACKMOD: 5
DMGBONUS: 0
SDMGBONUS: 2
PROFICIENCY: 3
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 1d10+ ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d10 + ATTACKMOD + DMGBONUS

SATTACKTIMES: 2
DAMAGES: 1d8 + ATTACKMOD + SDMGBONUS
CRITDAMAGES: 2d8 + ATTACKMOD + SDMGBONUS

output ATTACKTIMESd[attackroll d20 vs ENEMYAC]  named "Longsword 2 handed"
output SATTACKTIMESd[sattackroll d20 vs ENEMYAC] named "longsword duelist"

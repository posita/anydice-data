FAILURE: 0
MODERATESUCCESS: 1
GREATERSUCCESS: 2
CRITICALSUCCESS: 3

function: interpret ROLL:n{
if ROLL <=14 {result: FAILURE}
if ROLL >=15 & ROLL < 21 {result: MODERATESUCCESS}
if ROLL >= 21 & ROLL < 31 {result: GREATERSUCCESS}
if ROLL >= 31 {result: CRITICALSUCCESS}
}

function: equivalence ROLL:s {
if 1@ROLL = 2@ROLL {result: 1}
result: 0
}

loop N over {0..12}{
output [interpret 2d12+N] named "Chances at [N] modifier"
output [interpret 2d12+N] + [equivalence 2d12] named "Chances at [N] modifier w/ doubles increases degree of success"
}

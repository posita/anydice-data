function: nib POOL:s TN:n {
  \ crit success \\
  if 1@POOL < TN & 2@POOL < TN & 1@POOL = 2@POOL{
    result: 2
  }\
  \ success \
  if 1@POOL < TN & 2@POOL < TN {
    result: 1
  }
  \ partial \
  if 1@POOL < TN | 2@POOL < TN  {
    result: 0
  }
  \ crit fail \\
  if 1@POOL = 2@POOL {
    result: -2
  }\
  \ fail \
  result: -1
}

DIESIZE: 10

loop N over {2..10} {
  output [nib 2dDIESIZE N] named "TN:[N]"
}

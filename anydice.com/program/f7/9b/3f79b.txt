
 
CRITMISS: -d8
MISS: 0
HEAL: 2d8
CRITHEAL:4d8

function: experttreat ROLL:n {
 if ROLL = 1 {result: CRITMISS }
 if ROLL + PROF <= 10 {result: CRITMISS}
 if ROLL = 20 {result: CRITHEAL+10}
 if ROLL + PROF >= 30{result: CRITHEAL+10}
 if ROLL + PROF >= 20{result: HEAL+10}
}

PROF: 13




function: trainedtreat ROLL:n {
 if ROLL = 1 {result: CRITMISS }
 if ROLL + PROF <= 5 {result: CRITMISS}
 if ROLL = 20 {result: CRITHEAL}
 if ROLL + PROF >= 25{result: CRITHEAL}
 if ROLL + PROF >= 15{result: HEAL}
}

output [experttreat d20] named "Expert"
# output [trainedtreat d20] named "Trained"




function: treatwounds ROLL:n with BONUS:n vs DC:n{
 if ROLL = 1 {result: CRITMISS }
 if ROLL + PROF <= DC-10 {result: CRITMISS}
 if ROLL = 20 {result: CRITHEAL + BONUS}
 if ROLL + PROF >= DC{result: CRITHEAL + BONUS}
 if ROLL + PROF >= DC{result: HEAL + BONUS}
}

output [treatwounds d20 with 10 vs 20]

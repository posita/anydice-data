function: compare ATT:s and DEF:s {

  ATT: [sort ATT]
  DEF: [sort DEF]

\ LÃ¤nge beider Arrays angleichen: mit 1er aufstocken \

if #ATT = #DEF {} else {
  if #ATT > #DEF {
    DEF: {DEF, 1:(#ATT-#DEF)}
  }
  if #ATT < #DEF {
    ATT: {ATT, 1:(#DEF-#ATT)}
  }
}

\ Loop over length of Arrays: Werte paarweise vergleichen \

  RESULT: 0

loop N over {1..#ATT} {
  if N@ATT < N@DEF {result: RESULT}
  if N@ATT > N@DEF {RESULT: RESULT + 1}
}

result: RESULT
}



function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}


output 2d4 named "2d4"

output [reroll {1,2} on d4 to d4] + [reroll {1,2} on d4 to d4] named "2d4 GWF"

output d12 named "d12"

output [reroll {1,2} on d12 to d12] named "d12 GWF"

output 2d6 named "2d6"

output [reroll {1,2} on d6 to d6] + [reroll {1,2} on d6 to d6] named "2d6 GWF"

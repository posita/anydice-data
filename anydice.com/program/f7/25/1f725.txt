function: nib POOL:s TN:n {
  \ positive memory triggered \
  if 1@POOL < TN & 2@POOL < TN & 1@POOL = 2@POOL{
    result: 1
  }
  \ negative memory triggered \
  if 1@POOL >= TN & 2@POOL >= TN & 1@POOL = 2@POOL{
    result: -1
  }
}

DIESIZE: 10
loop N over {2..10} {
  output [nib 2dDIESIZE N] named "TN:[N]"
}

function: successes in ROLL:s vs TN:n {
  result: [count {1..TN} in ROLL]
}

function: successes in ROLL:s vs TN:n pushed {
  SUCC: [successes in ROLL vs TN]
  DUDS: #ROLL - SUCC
  result: SUCC + [successes in DUDSd6 vs TN]
}

ATTRIBUTE: 3
SKILL: 1
LEVERAGE: 0
N: ATTRIBUTE + LEVERAGE
TN: SKILL
output [successes in Nd6 vs TN] named "[N]d6 <= [TN]"
output [successes in Nd6 vs TN pushed] named "[N]d6 <= [TN], Pushed"
loop TN over {2..3}{
  output [successes in Nd6 vs TN] named "[N]d6 <= [TN]"
}

function: mydice{
result: {-100,0,0,1,1,1}
}

function: red IN:n{
result: -((IN-99) / 100)
}

function: green IN:n{
result: IN + [red IN]*100
}

function: success P:n G:n{
result: [green P] >= [green G]
}

function: comp P:n G:n{
result: ([red G] > ([red P]))
}

function: true comp P:n G:n{
result: [success P G] * [comp P G]
}

loop P over {1..5} {
loop G over {1..5} {
output [true comp Pd[mydice] Gd[mydice]] named "Comp [P] vs [G]"
}
}

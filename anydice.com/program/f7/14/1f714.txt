function: hit ATTACKMOD:n vs AC:n {
    result: 1d20+ATTACKMOD>=AC
}

function: damage DAMAGE:d vs ARMOR:n {
    result: [highest of DAMAGE-[highest of ARMOR and 0] and 0]
}

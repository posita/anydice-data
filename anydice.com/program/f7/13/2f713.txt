\ Hammer of Math Angry Dice Calculator \
\ This is used to calculate the chance of getting a double with a target value. Like Double 4+. \

TARGET: 6 \ Minimum die roll needed. \
POOL: 8d6 \ Pool of dice. Normally 8. Re-rolls would add to it. \

\ Function to check. Basically a bastardized count function. \

function: check TARGET:n in POOL:s to NUMBER:n 
{
COUNT: 0
loop P over {TARGET..6}
{
COUNT: COUNT + {(P = POOL) >= NUMBER}
}
result: COUNT > 0
}

output [check TARGET in POOL to 2] named "[POOL] Chance of Double [TARGET]+"

output [check TARGET in POOL to 3] named "[POOL] Chance of Triple [TARGET]+"

output [check TARGET in POOL to 2] | [check 1 in POOL to 3] named "[POOL] Chance of Double [TARGET]+ or Triple"

output [check 3 in POOL to 2] named "Martial Excellence"

output [check 4 in POOL to 2] | [check 1 in POOL to 3] named "Total Carnage"

output [check 5 in POOL to 2] | [check 1 in POOL to 3] named "Warp Blades"

output [check 6 in POOL to 2] | [check 4 in POOL to 3] named "Unbridled Bloodlust"

output [check 6 in POOL to 3] named "Reborn in Blood"

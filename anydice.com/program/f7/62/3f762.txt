

SKILLONE:  d{0:10, 1:9, 2:1}    
SKILLTWO:  d{0:6,  1:5, 2:1}    
SKILLTHREE: d{0:5,  1:4, 2:1}   
SKILLFOUR: d{0:4,  1:3, 2:1}   
SKILLFIVE: d{0:3,  1:2, 2:1}   


function: stuntthree ROLLTHREE:n {
  if [ROLLTHREE contains 0] { result: 0 }  \ fail \
  if [ROLLTHREE contains 1] { result: 1 }  \ normal success \
  if [ROLLTHREE contains 2] { result: 1 }  \ stunt = 2 successes \
}

function: stunttwo ROLLTWO:n ROLLTHREE:n {
  if [ROLLTWO contains 0] { result: 0 }  \ fail \
  if [ROLLTWO contains 1] { result: 1 }  \ normal success \
  if [ROLLTWO contains 2] { result: 1 + [stuntthree ROLLTHREE] }  \ stunt = 2 successes \
}

function: stunt ROLL:n ROLLTWO:n ROLLTHREE:n {
  if [ROLL contains 0] { result: 0 }  \ fail \
  if [ROLL contains 1] { result: 1 }  \ normal success \
  if [ROLL contains 2] { result: 1 + [stunttwo ROLLTWO ROLLTHREE] }  \ stunt = 2 successes \
}


function: check ABILITY: n SKILL: n {
  DIE: SKILLFIVE
  if SKILL = 1 { DIE: SKILLONE }
  if SKILL = 2 { DIE: SKILLTWO }
  if SKILL = 3 { DIE: SKILLTHREE }
  if SKILL = 4 { DIE: SKILLFOUR }
  if SKILL = 5 { DIE: SKILLFIVE }

  SUCCESSES: 0
  loop N over {1..ABILITY} {
    SUCCESSES: SUCCESSES + [stunt DIE DIE DIE]
  }
  result: SUCCESSES
}

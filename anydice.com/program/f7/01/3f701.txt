function: less than VALUE:n in LIST:s {
 NUM:0
 loop N over {1..VALUE}{
  loop M over LIST {
   if M=N{
    NUM:NUM+1
   }
  }
 }
 result:NUM
}

function: MUCH:n rerolls ROLLS:d dice size DIE:n if lower than VALUE:n {
 CHECK: [lowest of MUCH and [less than VALUE in ROLLS]]
 REROLLED: (CHECK)d(DIE)
 NEW: [highest (#ROLLS-CHECK) of ROLLS] + REROLLED
 result: NEW
}


CUTOFF: 4
DICE: 15d4
SIZE: 4
REROLLS:5
output [REROLLS rerolls DICE dice size SIZE if lower than CUTOFF]

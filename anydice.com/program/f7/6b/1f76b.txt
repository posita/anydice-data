
function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}


output[attack d20 plus 5 + 4 vs 16 for (5d6)*2 crit (5d6)*2 on 20]
named "Tilak's Celestial Bane"

output[attack d20 plus 5 + 4 vs 16 for 3d8 + 9 crit 3d8 on 20]
named "Eamon's Slashing Damage"

output[attack d20 plus 5 + 4 vs 16 for (11d8)*2 crit (11d8)*2 on 20]
named "Eamon's Smite Damage"

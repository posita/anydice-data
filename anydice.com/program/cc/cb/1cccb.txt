FAIL: 0
SUCC: 1

function: improvementcheck ROLL:n vs SKILL:n {
 if ROLL > SKILL { result: FAIL }
 if ROLL > 95 { result: FAIL }
 result: SUCC
}

IMPROVEMENT: 1d10

function: improvementresult for RESULT:n and SKILL:n {
 if RESULT = FAIL { result: IMPROVEMENT + SKILL}
 result: SKILL
}

function: nextsession for CURRENT_SKILL:n {
 result: [improvementresult for [improvementcheck d100 vs CURRENT_SKILL] and CURRENT_SKILL]
}

loop SKILL over {75} {
 output [nextsession for SKILL] named "[SKILL] and Session 1"
 output [nextsession for [nextsession for SKILL]] named "[SKILL] and Session 2"
 output [nextsession for [nextsession for [nextsession for SKILL]]] named "[SKILL] and Session 3"
 output [nextsession for [nextsession for [nextsession for [nextsession for SKILL]]]] named "[SKILL] and Session 4"
 output [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for SKILL]]]]] named "[SKILL] and Session 5"
 output [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for SKILL]]]]]] named "[SKILL] and Session 7"
 output [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for SKILL]]]]]]] named "[SKILL] and Session 8"
 output [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for SKILL]]]]]]]] named "[SKILL] and Session 9"
 output [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for [nextsession for SKILL]]]]]]]]] named "[SKILL] and Session 10"
}

\ 0 is a miss, 1 is a weak hit, 2 is a strong hit \

function: action AROLL:n BROLL:n CROLL:s {
  result: (AROLL > CROLL) + (BROLL > CROLL)
}

function: thraction AROLL:n BROLL:n ZROLL:n CROLL:s {
  F: 1 @ [sort {AROLL, BROLL, ZROLL}]
  S: 2 @ [sort {AROLL, BROLL, ZROLL}]
  result: (F > CROLL) + (S > CROLL)
}

function: fraction AROLL:n BROLL:n ZROLL:n XROLL:n CROLL:s {
  F: 1 @ [sort {AROLL, BROLL, ZROLL, XROLL}]
  S: 2 @ [sort {AROLL, BROLL, ZROLL, XROLL}]
  result: (F > CROLL) + (S > CROLL)
}

function: hiaction AROLL:n BROLL:n CROLL:n DROLL:n {
  result: (AROLL > [highest of CROLL and DROLL]) + (BROLL > [highest of CROLL and DROLL])
}

output [fraction 1d6 1d6 1d6 1d6 1d6] named "best 2 of 4d6 vs d6"

output [action 1d8 1d8 1d6] named "2d8 vs d6"
output [action 1d10 1d10 1d6] named "2d10 vs d6"

output [action 1d12 1d12 1d6] named "L4:12 2d12 vs d6"

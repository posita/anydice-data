MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {result: CRITICALHIT}
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {result: CRITICALDAMAGE}
 if ATTACKRESULT = HIT { result: HITDAMAGE}
 if ATTACKRESULT = MISS {result: MISSDAMAGE}
}

DEFENSE: 15
ATTACK: 2
HITDAMAGE: 2d6 + 1
CRITICALDAMAGE: 13
MISSDAMAGE: 0


output [damage for [attack [highest 1 of 1d20] vs DEFENSE]] named "BASE DAMAGE"
output [damage for [attack [highest 1 of 2d20] vs DEFENSE]] named "1x ADV DAMAGE"
output [damage for [attack [highest 1 of 3d20] vs DEFENSE]] named "2x ADV DAMAGE"
output [damage for [attack [highest 1 of 4d20] vs DEFENSE]] named "3x ADV DAMAGE"
output [damage for [attack [highest 1 of 5d20] vs DEFENSE]] named "4x ADV DAMAGE"
output [damage for [attack [highest 1 of 6d20] vs DEFENSE]] named "5x ADV DAMAGE"
output [damage for [attack [highest 1 of 7d20] vs DEFENSE]] named "6x ADV DAMAGE"

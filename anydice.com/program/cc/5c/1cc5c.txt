SK: 80

function: penalty vs X {
  PENALTY: 1@2d100
  result: PENALTY <= X
}

function: SHOTS shots DMG each with SKILL {
  T: 0
  loop I over {1..SHOTS}{
    if SHOTS = 1 {
      T: T + (d100 <= SKILL)*DMG
    }
    else {
      T: T + [penalty vs SKILL]*DMG
    }
  }
  result: T
}

loop G over {1..3}{
  loop S over {1..3}{
    if G = 1 {
      output [S shots 1d8 each with SK] named "Total DPS for Gun [G] with [S] shot(s)"
    }
    else if G = 2 {
      output [S shots 1d10 each with SK] named "Total DPS for Gun [G] with [S] shot(s)"
    }
    else {
      output [S shots (1d10+2) each with SK] named "Total DPS for Gun [G] with [S] shot(s)"
    }
  }
}

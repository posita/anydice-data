function: damage DICE:d hit at HIT:n up {
   RES:{}
   loop N over {DICE} {
      if (N@{DICE}<HIT) {
         RES:{{RES},0}
      }
      else {
         RES:{{RES},N@{DICE}}
      }
   }
   result:dRES
}
output [damage d4 hit at 2 up] named "d4"
output [damage d6 hit at 3 up] named "d6"
output [damage d8 hit at 4 up] named "d8"

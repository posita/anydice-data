function: WLC:n wlc POWER:n power plus BOOST:n roll DICE:n dice {
  COUNT: 0
  if POWER-BOOST < 1 { POWER: 1 }
  else { POWER: POWER - BOOST }
  loop X over {1..WLC} {
    loop N over {1..DICE} {
      COUNT: COUNT + [count {POWER..6} in d6]
    }
  }
  result: COUNT
}

output [1 wlc d6 power plus 0 roll 6 dice]

ARMOR : 1

function: check X:n accuracy A:n bonus B:n target T:n {
    if X + A + B >= 20 { result : 2 }
    if X + A + B >= T { result : 1 }
    result : 0
}

function: attack X:n accuracy A:n grit G:n evasion E:n damage N:n F:n M:n {
    X : [check X accuracy A bonus G target E]
    if X > 0 {
        H : {1..N}@(X*N)dF + M
    } else { H : 0 }
    result : [highest of H - ARMOR and 0]
}

    } else { H : 0 }

}

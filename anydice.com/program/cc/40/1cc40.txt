function: roll DIE:n between DIFFICULTY:n and ASPECT:n {
    if DIE > ASPECT {result: 0}
    else if DIE < DIFFICULTY {result: 0}
    result: DIE
}

loop ASPECT over {1..9} {
    loop DIFFICULTY over {1..9} {
        output [roll d10 between DIFFICULTY and ASPECT] named "Aspect equals [ASPECT], Difficulty equals [DIFFICULTY]"
    }
}

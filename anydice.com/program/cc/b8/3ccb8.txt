F: 2
AW: 0
AR: 1
S: 2

LLA: [lowest 2 of 4d6] + F >= 2d6 + AW
LA: [lowest 2 of 3d6] + F >= 2d6 + AW
MA: 2d6 + F >= 2d6 + AW
HA: [highest 2 of 3d6] + F >= 2d6+AW


\Rapier\
output 2*(LA*(1d4 - AR)) named "Rapier L"
output (MA*(1d6 + S - AR)) named "Rapier M"

output 0 named " "

function: damage D:d DBONUS:d R:n RBONUS:n AC:n {
 DAMAGEA: 0
 DAMAGEB: 0
 if R = 1 { result: 0 }
 if R = 20 { result: D+D+DBONUS }
 if R+RBONUS >= AC { result: D+DBONUS}
 result: 0
}
function: damage critfish DA:d DB:d DBONUS:n SNEAK:d RA:n RB:n RBONUS:n AC:n {
 SNEAKA: 0
 SNEAKB: SNEAK
 if RA = 20 {
  SNEAKA: SNEAK
  SNEAKB: 0
 }
 result: [damage DA+SNEAKA DBONUS RA RBONUS AC]+[damage DB+SNEAKB DBONUS RB RBONUS AC]
}

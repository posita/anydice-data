function: compare A:s and B:s {
  result: A = B
}
function: range on DIE:n between LOW:n and HIGH:n {
  if DIE > LOW & DIE < HIGH { result: 1 }
  result: 0
}

SUCCESS: 11
FAILURE: 5

loop MOD over {-6..6} {
  output ((2d6 + MOD) > SUCCESS) & !([compare 2d6 and {6,6}]) named "[MOD] success"

  output [range on (2d6 + MOD) between FAILURE and SUCCESS] named "[MOD] mixed success"

  output ((2d6 + MOD) <= FAILURE) & !([compare 2d6 and {1,1}]) named "[MOD] fail"

  output 0 named "-------------"
}

F: 2
AW: 0
AR: 1
S: 2

LLA: [lowest 2 of 4d6] + F >= 2d6 + AW
LA: [lowest 2 of 3d6] + F >= 2d6 + AW
MA: 2d6 + F >= 2d6 + AW
HA: [highest 2 of 3d6] + F >= 2d6+AW


\Rapier\
output 6*(LA*(1d4 - AR)) named "Rapier L"
output 3*(MA*(1d6 + S - AR)) named "Rapier M"
output (HA*(2d6 + S - 2*AR) + 2*LA*(1d4 - AR)) named "Rapier H"

output 0 named " "

\Sword\
output 3*(LA*(2d6 - 2*AR)) named "Sword L"
output (MA*(2d8 + S - 2*AR)) + (LA*(2d6 - 2*AR)) named "Sword M"
output (HA*(3d8 + S - 3*AR)) named "Sword H"

output 0 named " "

\Spear\
output 3*(LLA*(2d6 - 2*AR)) named "Spear L"
output (LA*(2d8 + S - 2*AR)) + ((LLA)*(2d6 - 2*AR)) named "Spear M"
output (MA*(3d8 + S - 3*AR)) named "Spear H"

output 0 named " "

\Club\
output 3*(LA*(1d10 + S - AR)) named "Club L"
output ((MA*(1d12 + 2*S - AR))) + (LA*(1d10 + S - AR)) named "Club M"
output (([highest 2 of 3d6] + F >= 2d6 + AW)*(2d12 + 2*S - 2*AR)) named "Club H"

output 0 named " "

\Greatsword\
output (LA*(2d10 - 2*AR)) named "Gsword L"
output (MA*(2d12 + S - 2*AR)) named "Gsword M"
output (HA*(3d12 + S - 3*AR)) named "Gsword H"


output 0 named " "

\Shortbow\
output 6*(LLA*(1d4 - AR)) named "Shortbow L"
output 3*(LA*(1d6 + S - AR)) named "Shortbow M"
output (MA*(2d6 + S - 2*AR)) + 2*(LLA*(1d4 - AR)) named "Shortbow H"

output 0 named " "

\Longbow\
output 3*(LLA*(2d6 - 2*AR)) named "Longbow L"
output (LA*(2d8 + S - 2*AR)) + (LLA*(2d6 - 2*AR)) named "Longbow M"
output (MA*(3d8 + S - 3*AR)) named "Longbow H"

output 0 named " "

\Warbow\
output (LLA*(2d10 - 2*AR)) named "Warbow L"
output (LA*(2d12 + S - 2*AR)) named "Warbow M"
output (MA*(3d12 + S - 3*AR)) named "Warbow H"

output 0 named " "

\Crossbow\
output (LA*(3d8 - 3*AR)) named "XBow L"
output (MA*(3d10 + S - 3*AR)) named "XBow M"
output (HA*(4d10 + S - 4*AR)) named "XBow H"

FEAR: 1

AC: 36 - FEAR
WILL: 35 - FEAR
FLANKING: 0
HIT_BONUS: 24
DECEPTION_BONUS: 28


DICE_ONE: 1d20
HIT_ONE: DICE_ONE+HIT_BONUS
DICE_TWO: 1d20
HIT_TWO: DICE_ONE+HIT_BONUS-4
DAMAGE: 3d8+6+2d6

FF: FLANKING
if FF=0 {
  DECEPTION: 1d20+DECEPTION_BONUS
  FF: DECEPTION >= WILL
}

DAMAGE: DAMAGE + 1*FF
AC: AC - 2*FF

CRIT_ONE: 1 - (DICE_ONE != 20) * (HIT_ONE < AC+10)
CRIT_TWO: 1 - (DICE_TWO != 20) * (HIT_TWO < AC+10)

DAMAGE_ONE: (HIT_ONE>=AC)*DAMAGE + CRIT_ONE*DAMAGE
DAMAGE_TWO: (HIT_TWO>=AC)*DAMAGE + CRIT_TWO*DAMAGE

output DAMAGE_ONE+DAMAGE_TWO named "total damage"

AC: 15

MOD: 4
DMGDICEA: 2d6
DMGDICEB: 1d8
RIDERDICEA: 0
RIDERDICEB: 0

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d {
    if ROLL = 1 {
        result: 0
    }
    if ROLL = 20 {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

DMGA: [attack 1d20 plus MOD+2 vs AC for DMGDICEA+RIDERDICEA+MOD crit DMGDICEA]
DMGB: [attack 1d20 plus MOD+2 vs AC for DMGDICEB+RIDERDICEB+MOD crit DMGDICEB]

output DMGA named "Attack A"
output DMGB named "Attack B"
output DMGA + DMGB named "Multiattack"

BARBGS: [attack 1d20 plus 4 vs AC for 2d6+4 crit 2d6]

output BARBGS named "Barbarian STR 2 Rage"
output 2dBARBGS named "Barbarian STR 2 Frenzy Rage"

RANGERHM: [attack 1d20 plus 8 vs AC for 1d8+1d6+4 crit 1d8+1d6]

output RANGERHM named "Ranger DEX 4 Hunter's Mark"

ROGUESA: [attack 1d20 plus 4 vs AC for 1d8+2d6+2 crit 1d8+2d6]

output ROGUESA named "Rogue DEX 2 SA Rapier"

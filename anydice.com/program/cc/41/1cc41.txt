function: roll DIE:n equal to or greater than DIFFICULTY:n and equal to or less than ASPECT:n {
    if DIE > ASPECT {result: 0}
    else if DIE = ASPECT {result: 1}
    else if DIE < DIFFICULTY {result: 0}
    else {result: 1}
}

loop ASPECT over {1..9} {
    loop DIFFICULTY over {1..9} {
        output [roll d10 equal to or greater than DIFFICULTY and equal to or less than ASPECT] named "Aspect equals [ASPECT], Difficulty equals [DIFFICULTY]"
    }
}

function: attack R:s leadership L:n hiton H:n{
 result: R + [lowest of R = 0 and L]d(d6 >= H)
}

function: add bonus to N:n {
 if N { result: N + 1 }
 result: N
}

output [attack 5d(d6 >= 6) leadership 5 hiton 6]  named "DA"

function: movement M:s palantir P:n{
 result: M + [lowest of 1 and [lowest of M = 0 and 6]d(d3 >= 3)]
}

output [movement 6d(d6 >= 4) palantir 1] named "Expected Movement with 1 P"

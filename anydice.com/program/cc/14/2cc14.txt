\Chance for malign gateway tentacle to turn hostile before duration runs out at different spell powers\
\0 is the chance to turn hostile, 1 is the chance not to turn hostile\

function: rand N:n {
  result: 1d{1..N} - 1
}

function: rand N:n avg D:n times{
  SUM: 0
  loop I over {1..D} {
    SUM: SUM + [rand N]
  }
  result: SUM / D
}

loop I over {0..20} {
  POWER: I * 10
  output [rand POWER avg 6 times] - [rand 4]   \random2avg(pow, 6) - random2(4) turns\
    > 30 + [rand 31 avg 2 times] - 18          \30 + random2avg(31, 2) - 18\
    named "Hostility chance at [POWER] power"
}

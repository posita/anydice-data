ATTACK_DAM: 8
ATTACK_DICE: 4d8 + 6

DEFENSE_DR: 0
DEFENSE_AGI: 6
HP: 30
DEFENSE: 1d20

function: A:n is dead
{
 if A <= 0 {result: 1}
result: 0
}

function: A:n hits B:n then C:n minus D:n
{
 if A > B 
{ 
if D < 0{ result: C }
else {result: C - D }
}
 result: C
}


HP_A: [ATTACK_DICE hits (DEFENSE + DEFENSE_AGI) then HP minus (ATTACK_DAM - DEFENSE_DR)]
HP_B: [ATTACK_DICE hits (DEFENSE + DEFENSE_AGI) then HP_A minus (ATTACK_DAM - DEFENSE_DR)]
HP_C: [ATTACK_DICE hits (DEFENSE + DEFENSE_AGI) then HP_B minus (ATTACK_DAM - DEFENSE_DR)]
HP_D: [ATTACK_DICE hits (DEFENSE + DEFENSE_AGI) then HP_C minus (ATTACK_DAM - DEFENSE_DR)]
HP_E: [ATTACK_DICE hits (DEFENSE + DEFENSE_AGI) then HP_D minus (ATTACK_DAM - DEFENSE_DR)]

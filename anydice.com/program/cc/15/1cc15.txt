\ adjust threat \

THREAT: 14
THREATTWO: 10

\ set confirm bonus from Power Critical feat \

CRITICALCONFIRM: 4

\ use transposed graph \

\ leave rest as is \
MISS: 0
NORMALHIT: 1
CRITICALHITTWO: 2
CRITICALHIT: 4

ATTACK: 0

function: ROLL:n vs AC:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20] d DAMAGE
 }
 if ROLL >= THREATTWO & ROLL + ATTACK >= AC {
  result: [threattwo d20] d DAMAGE
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: NORMALHIT d DAMAGE }
 result: MISS
}

function: threat ROLL:n {
 if ROLL = 1 { result: NORMALHIT }
 if ROLL + ATTACK + CRITICALCONFIRM >= AC | ROLL = 20 {
  result: CRITICALHIT
 }
 result: NORMALHIT
}

function: threattwo ROLL:n {
 if ROLL = 1 { result: NORMALHIT }
 if ROLL + ATTACK + CRITICALCONFIRM >= AC | ROLL = 20 {
  result: CRITICALHITTWO
 }
 result: NORMALHIT
}

DAMAGE: 1d8+4

 output [d20 vs 11] named "11"
\ } \

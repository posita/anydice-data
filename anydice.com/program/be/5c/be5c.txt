ATTACK: 4
DEFENSEBONUS: 4
DEFENSE: 4 + DEFENSEBONUS
ARMOR: 0

function: floor NUM:n {
     if NUM < 0 { result: 0 }
     result: NUM
}

function: damage of DAMAGE:n PIERCING:n {
     result: [floor DAMAGE + PIERCING - ARMOR]
}

function: attack ROLL:n damage DAM:n pierce PIERCE:n {
 if ROLL + ATTACK >= DEFENSE { result: [damage of DAM PIERCE] }
 result: 0
}


function: sdamage of DAMAGE:n PIERCING:n {
     result: [floor DAMAGE + PIERCING*2 - ARMOR*2]
}

function: sattack ROLL:n damage DAM:n pierce PIERCE:n {
 if ROLL + ATTACK >= DEFENSE { result: [sdamage of DAM PIERCE] }
 result: 0
}

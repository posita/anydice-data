\**** Two Weapon Fighting modifiable Calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: secondtwfattackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: TWFCRITDAMAGE} 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: TWFDAMAGE}
 result: 0
}

function: secondtwfattackrollsneak ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: TWFCRITDAMAGE + SNEAKATTACK + SNEAKATTACK} 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: TWFDAMAGE + SNEAKATTACK}
 result: 0
}

function: twfattackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE + SNEAKATTACK + SNEAKATTACK + [secondtwfattackroll 1d20 vs ENEMYAC]} 
 if ROLL = 1 { result: 0 + [secondtwfattackrollsneak 1d20 vs ENEMYAC]}
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE + SNEAKATTACK +
 [secondtwfattackroll 1d20 vs ENEMYAC]}
 result: 0 + [secondtwfattackrollsneak 1d20 vs ENEMYAC]
}

function: sixgwfight ROLLX:n {
 if ROLLX <= 2 { result:d6 }
 result: ROLLX
}

function: twelvegwfight ROLLX:n {
 if ROLLX <= 2 { result:d12 }
 result: ROLLX
}

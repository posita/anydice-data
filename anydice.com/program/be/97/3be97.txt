PRE: 8
PUI: -4

function: mindmg VAL:s {
  if VAL >= 1 { result: VAL }
  result: 1
}
function: dmg ROLL:s {
  HIT: PRE - ROLL
  if HIT >= 0 { result: [mindmg ROLL+PUI] }
  result: 0
}

function: countd MAX:n {
  TOTAL: 0
  loop COUNT over {1..10} {
    TOTAL: TOTAL + [dmg d10]
  }
  result: TOTAL
}

output [countd 10]

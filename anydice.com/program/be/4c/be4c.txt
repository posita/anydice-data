function: cast N:n times with S:n and DC:n {
  if N > 1 {
    if S { NEW_DC: DC + 1 }
    else { NEW_DC: DC }
    result: S + [cast N - 1 times with d20 <= NEW_DC and NEW_DC]
  }
  result: S
}

DIE: d10
N: 5

loop DIE over {4,6,8,10,12,20}
{
  output [cast N times with dDIE <= 1 and 1] named "Nr of surges using a d[DIE]"
}

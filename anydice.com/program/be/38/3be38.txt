function: advantage A:n B:n DEF:n {
if A > B {X: A}
if B >= A {X:B}

if X <= DEF/2 {result: 1} 
if X > DEF/2 & X <= DEF {result: 2}
if X > DEF & X <= DEF*2 {result: 3}
if X > DEF*2 & X != 20 {result: 4}
if X = 20 {result: 5}
}

RANGE: {0, 2, 4, 6, 8, 10}

loop DEF over RANGE {
   output [advantage 1d20 1d20 DEF] named "DEF = [DEF]"
}

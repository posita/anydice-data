\ ============================== SORCERER WILDMAGIC RULES ========================== \
\ Every time a spell is cast, roll a die. If the die roll is lower than the DC, a    \
\ surge happens. The DC is equal to the amount of spells cast since the last surge.  \
\ E.g. the first spell cast has a DC of 1. If the die rolled is higher than 1, no    \
\ surge happens. The second spell cast then has a DC of 2. Once a surge has occurred \
\ the DC becomes 1 again.                                                            \
\ Optional:                                                                          \
\  - higher level spells use a smaller die, thus increasing the chance of a surge;   \
\     e.g.: lvl 0: d20, lvl 1: d12, lvl 2: d10, lvl 3: d8, lvl 4: d6, lvl 5: d4.     \
\  - as the sorcerer becomes more experienced, he can cast lower level spells        \
\    without causing a wild magic surge.                                             \
\                                                                                    \
\ TO DO: contemplate effects of Tides of Chaos                                       \

function: cast N:n times with S:n and DC:n {
  if N > 1 {
    if S { NEW_DC: 1 }
    else { NEW_DC: DC + 1 }
    result: S + [cast N - 1 times with D <= NEW_DC and NEW_DC]
  }
  result: S
}

function: cast N:n times with D:d
{
  if N < 1 { result: 0}
  result: [cast N times with D <= 1 and 1]
}

function: cast N:n cantrips
{
  result: [cast N spells at level 0]
}

function: cast N:n spells at level L:n
{
  DIE: [get die for level L]
  result: [cast N times with DIE]
}


SPELL_LEVEL_DICE: {20, 12, 10, 8, 6, 4}
function: get die for level L:n
{
  if L = 0 { result: d20 }
  D: L@SPELL_LEVEL_DICE
  result: dD
}

\ This function calculates odds according to the Rules As Written                    \
\ Note: this gives a binomial distribution                                           \
function: cast N:n times with S:n according to rules as written{

  if N > 1 {
    result: S + [cast N - 1 times with d20 <= 1 according to rules as written]
  }
  result: S
}

\ cast as many spells as there are spell slots \
\ cantrips don't count                         \

function: calculate surges for spells S:s{
  R: 0
  loop N over {1..9}{
    \ N@S gets the Nth value in S \
    R: R + [cast N@S times with [get die for level N]]
  }
  result: R
}

function: eval ROLL:s and ROLLZ:s {
  SUCCESS: ROLL >= 4
  DBLSUCC: ROLL > 5
  FAIL: ROLL <= 2
  ZFAIL: ROLLZ <= 2
  DZFAIL: ROLLZ < 2
  result: SUCCESS + DBLSUCC - FAIL - DZFAIL - ZFAIL
}

function: evil ROLL:s {
  SUCCESS: ROLL >= 4
  DBLSUCC: ROLL > 5
  FAIL: ROLL <= 2
  result: SUCCESS + DBLSUCC - FAIL
}

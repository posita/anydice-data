function: A with advantage {
 DICE: {}
 loop N over {A}{
  DICE: {DICE, N:(N*2-1)}
 }
 result: DICE
}

function: A:d with disadvantage {
 DICE: {}
 REVERSED: [reverse {A}]
 loop N over {A}{
  DICE: {DICE, N:(N@REVERSED*2-1)}
 }
 result: DICE
}
output [2d20 with advantage]
output [2d20 with disadvantage]

function: h ROLL:n AC:n{
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE + RIDER }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE + RIDER
 }
 result: DAMAGE + RIDER
}

function: ch ROLL:n AC:n{
 if ROLL + ATTACK >= AC | ROLL = 20 { result: 1}
 result: 0
}

A: d{10..30}

ATTACK: 12
DAMAGE: 3d8+1
CRITICAL: 2
THREAT: 20
RIDER: 0

output [h d20 A] named "single attack"
output 2d[h d20-2 A] named "two weapon fighting"
output [h d20 A] + [ch d20 A] * [h d20-2 A] named "rippling blow"

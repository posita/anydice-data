function: roll MARK:n on DICE:d {
 result: [count {MARK..20} in DICE] + [roll [count 20 in DICE]d20 helper]
}

function: roll D:d helper {
 result: [count {MARK..20} in D]
}

function: cut CUT:n from SUCCESSES:n {
 result: [highest of 0 and SUCCESSES-CUT]
}

M: 1
output [cut 1 from [roll M on 2d20]] < [cut 2 from [roll M on 3d20]]

function: attack ROLL:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE + RIDER }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE + RIDER
 }
 result: DAMAGE + RIDER
}

AC: d{10..30}

ATTACK: 12
DAMAGE: 3d8+1
CRITICAL: 2
THREAT: 20
RIDER: 0

output [attack d20] named "single attack"

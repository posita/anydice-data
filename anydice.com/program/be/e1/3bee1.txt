AC: 14
BASEMOD: 7

function: roll D:d vs ARMOR:n x N:n{
  R: 0
  loop A over {1..N} {
    R: R + (D>=ARMOR)
  }
  result: R
}

function: roll D:d vs ARMOR:n adv x N:n {
  R: 0
  loop A over {1..N} {
    R: R + [highest of D>=ARMOR and D>=ARMOR]
  }
  result: R
}

output (1d12+6)*[roll 1d20+BASEMOD vs AC x 4] named "w/o either"

output (1d12+6)*[roll 1d20+BASEMOD+2 vs AC x 4] named "w/ arch"
output (1d12+6)*[roll 1d20+1d4+BASEMOD vs AC x 4] named "w/ bless"

output (1d12+6)*[roll 1d20+1d4+BASEMOD+2 vs AC x 4] named "w/ arch, bless"

output (1d12+6)*[roll 1d20+BASEMOD vs AC adv x 4] named "adv w/o either"

output (1d12+6)*[roll 1d20+BASEMOD+2 vs AC adv x 4] named "adv w/ arch"
output (1d12+6)*[roll 1d20+1d4+BASEMOD vs AC adv x 4] named "adv w/ bless"

output (1d12+6)*[roll 1d20+1d4+BASEMOD+2 vs AC adv x 4] named "adv w/ arch, bless"

output (1d12+16)*[roll 1d20+1d4+BASEMOD-5 vs AC x 4] named "w/ ss, bless"

output (1d12+16)*[roll 1d20+1d4+BASEMOD-5+2 vs AC x 4] named "w/ ss, arch, bless"

\------------------------ [HELPERS] ------------------------\
function: minimum of DIE:d
{
  result: 1@{DIE}
}


\------------------ [GREAT WEAPON FIGHTING] ------------------\
function: great weapon fighting DIE:d
{
  NUM_DICE: [minimum of DIE]
  DIE_TYPE: [maximum of DIE] / NUM_DICE
  FINAL_VALUE: 0
  loop N over {1..NUM_DICE}
  {
    FINAL_VALUE: FINAL_VALUE + [get roll 1dDIE_TYPE]
  }

  result: FINAL_VALUE
}
function: get roll ROLL:n
{
  if ROLL > 2
  {
    result: ROLL
  }
  result: 1dDIE_TYPE
}


\--------------------- [CALCULATIONS] ---------------------\

output [great weapon fighting 1d6] named "1d6: Quarterstaff"
output [great weapon fighting 1d8] named "1d8: Great Club, Quarterstaff 2H, Battle Axe, Longsword, Warhammer"
output [great weapon fighting 1d10] named "1d10: Battle Axe 2H, Halberd, Longsword 2H, Pike, Warhammer 2H"
output [great weapon fighting 1d12] named "1d12: Great Axe"
output [great weapon fighting 2d6] named "2d6: Great Sword, Maul"
output [great weapon fighting 4d6] named "2d6 Crit: Great Sword, Maul"

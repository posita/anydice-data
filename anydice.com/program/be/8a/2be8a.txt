AC: 17
MOD: 5
PROF: MOD + 4

function: X:n triss {
if X = 20 {result: 3d10 + MOD}
if X = 0 {result: 0}
if X + PROF >= AC {result: d10 + MOD}
result: 0
}

output ((d20 + PROF >= AC)*(d8 + MOD + 6d6)) named "rogue lvl 12 sneak attack"
output ((1@2d20 + PROF >= AC)*(d8 + MOD + 6d6)) named "rogue lvl 12 sneak attack advantage"

output 3d((d20 + PROF >= AC)*(2d6 + MOD)) named "greatsword fighter lvl 12"
output 3d((1@2d20 + PROF >= AC)*(2d6 + MOD)) named "greatsword fighter lvl 12 advantage"
output 3d((d20 + PROF - 5 >= AC)*(2d6 + MOD + 10)) named "greatsword fighter lvl 12 gwm"
output 3d((1@2d20 + PROF >= AC)*(2d6 + MOD)) named "greatsword fighter lvl 12 adv gwm"

output 4d[d20 triss] named "triss no advantage"
output 4d[1@3d20 triss] named "triss triple advantage"

output 3d((d20 + PROF >= AC)*(d8 + MOD)) named "basic monk lvl 12"
output 4d((d20 + PROF >= AC)*(d8 + MOD)) named "basic monk lvl 12 flurry"

output 2d((d20 + PROF >= AC)*(d12 + MOD + 3) + (d20=20)*(d12)) named "barbarian lvl 12 no reckless"
output 2d((1@2d20 + PROF >= AC)*(d12 + MOD + 3) + (d20=20)*(d12)) named "barbarian lvl 12 no reckless"

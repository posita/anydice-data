function: explode N:n with S:n symbols {
 if N = 1 {
  result: 1 + [explode d6 with S symbols]
 }
 result: N <= S
}

function: ATTACK vs DEFENSE {
 \ Clamped to 0-10 hits. \
 result: [lowest of 10 and [highest of 0 and ATTACK - DEFENSE]]
}

function: N:n of D:d with R:n rerolls {
 result: [highest N of (N + R)dD]
}

M: [explode d6 with 4 symbols]
R: [explode d6 with 3 symbols]
D: [explode d6 with 2 symbols]

output [3dM vs 2dD] named "3M vs 2D"
output [[3 of M with 2 rerolls] vs 2dD] named "3Mr2 vs 2D"
output [3dM vs [2 of D with 2 rerolls]] named "3M vs 2Dr2"
output [[3 of M with 2 rerolls] vs [2 of D with 2 rerolls]] named "3Mr2 vs 2Dr2"

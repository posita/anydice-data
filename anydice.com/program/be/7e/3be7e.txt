
D: 1d8

\ Daggerheart pair multiply \
function: dhpm N:n M:n {
 if N = M {
  result: N * 3
 }
 result: N + M
}

\ Daggerheart pair extra \
function: dhpe N:n M:n O:n {
 if N = M {
  result: N * 2 + O
 }
 result: N + M
}

function: light weapon N:n {
 if N >=5 {
  result: N-5
 }
 result: N
}

function: heavy weapon N:n {
 if N >=5 {
  result: N+5
 }
 result: N
}

function: damage N:n {
 if N < 0 {
  result: 0
 }
 result: 1+N/5
}

function: M:n damage N:n {
 result: 1+N/5+M/5
}

function: dodge N:n {
 if N >= 10 {
  result: N
 }
 result: 0
}

function: brace N:n {
 result: N-5
}



output [damage [dhpm D D]] named "medium weapon"
output [damage [light weapon [dhpm D D]]] named "light weapon"
output [damage [light weapon [dhpm D D]]] + [damage [light weapon [dhpm D D]]] named "light weapon multiattack"
output [damage [heavy weapon [dhpm D D]]] named "heavy weapon"


output [damage [brace [dhpm D D]]] named "medium brace"
output [damage [dodge [dhpm D D]]] named "medium dodge"

output [damage [light weapon [brace[ dhpm D D]]]] + [damage [light weapon [brace [dhpm D D]]]] named "light weapon multiattack brace"
output [damage [light weapon [dodge[ dhpm D D]]]] + [damage [light weapon [dodge [dhpm D D]]]] named "light weapon multiattack dodge"

output [damage [heavy weapon [brace [dhpm D D]]]] named "heavy brace"
output [damage [heavy weapon [dodge [dhpm D D]]]] named "heavy dodge"

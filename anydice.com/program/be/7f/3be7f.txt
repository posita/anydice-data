function: basicsave ROLL:n MOD:n DC:n DAMAGE:n {
  if (ROLL+MOD <= DC-10) {
     RES: 3
  } else if(ROLL+MOD < DC) {
     RES: 2
  } else if(ROLL+MOD < DC + 10) {
     RES: 1
  } else {
     RES: 0
  }
  if (ROLL = 1) { RES: RES+1 }
  if (ROLL = 20) { RES: RES-1 }
  if (RES < 0) { RES: 0 }
  else if (RES > 3) { RES: 3 }
  if (RES = 1) {
    result:DAMAGE/2
  }else if (RES = 2) {
    result:DAMAGE
  }else if (RES = 3) {
    result:DAMAGE*2
  }else{
    result:0
  }

}
output[basicsave 1d20 15 21 6d8]

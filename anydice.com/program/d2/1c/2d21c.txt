FEAR: 1

AC: 36 - FEAR
WILL: 35 - FEAR
FLANKING: 0
HIT_BONUS: 27

DICE_ONE: 1d20
HIT_ONE: DICE_ONE+HIT_BONUS
DAMAGE: 3d12+8+2d6+10

FF: FLANKING

AC: AC - 2*FF

CRIT_ONE: 1 - (DICE_ONE != 20) * (HIT_ONE < AC+10)

DAMAGE_ONE: (HIT_ONE>=AC)*DAMAGE + CRIT_ONE*DAMAGE

output DAMAGE_ONE named "total damage"

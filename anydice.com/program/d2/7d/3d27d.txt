function: limits SEQUENCE:n ZERO:n MIN:n {
 if (SEQUENCE < ZERO){result: 0}
 if (SEQUENCE < MIN){result: MIN}
 result: SEQUENCE
}
function: crop SEQUENCE:n MIN:n MAX:n {
 if (SEQUENCE <= MIN){result: 0}
 if (SEQUENCE > MAX){result: MAX}
 result: SEQUENCE
}
function: crop SEQUENCE:n {
 if (SEQUENCE < 0){result: 0}
 result: SEQUENCE
}

\************************** Highest/Lowest into Target Count ****************************\

\apply highest then target\
function: highest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLL >= TARGET) }
 result: COUNT
}
\apply lowest then target\
function: lowest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 ROLLR: [reverse ROLL]
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLLR >= TARGET) }
 result: COUNT
}

\apply target then limit the max (highest or lowest doesn't matter)\
function: count from ROLL:s on TARGET:n keep NUMBER:n {
 COUNT: ROLL >= TARGET
 if(COUNT > NUMBER) { COUNT: NUMBER }
 result: COUNT
}

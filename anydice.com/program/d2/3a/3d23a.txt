GOLD: d{1,1,0,0,0,0}
BLACK: d{1,0,0,0,0,0}

function: FOCUS ROLL N:n {
  RESULT: [count {1} in GOLD]
  LOOP N-1 TIMES:
    RESULT: RESULT + [count {1} in BLACK]
  result: RESULT
}

output [FOCUS ROLL 1] named "Focus 1"
output [FOCUS ROLL 2] named "Focus 2"
output [FOCUS ROLL 3] named "Focus 3"
output [FOCUS ROLL 4] named "Focus 4"

function: ve ATTACK:s AMOD:n DAMAGE:d DMOD:n DEF:n CRIT:n{
  FAIL_DMG: 0
  HIT_DMG: (DAMAGE+DMOD)
  HIT: (ATTACK+AMOD)
  CRIT_DMG: [maximum of DAMAGE] + 1
  
  if HIT = 20 { result: CRIT_DMG }
  if HIT >= DEF { result: HIT_DMG }
  result: FAIL_DMG
}

output [ve 1d20 1 1d4 0 10 5] named "Hechicero"

output [ve [lowest 1 of 2d20] 1 1d4 0 10 5] named "Hechicero - DIS"

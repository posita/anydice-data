function: TENS:n and ONES:n higher percentage {
 if TENS = 0 {
  if ONES = 0 {
   \ Rolling 00 counts as 100. \
   result: 100
  }
  if ONES <= 5 {
   \ Rolling a 0 with a 1-5 is a fumble and stays 1-5. \
   result: ONES
  }
 \ Use higher number for the 10s and lower for the 1s. \
 }
 if ONES > TENS {
  result: ONES * 10 + TENS
 }
 result: TENS * 10 + ONES
}

output [d{0..9} and d{0..9} higher percentage]

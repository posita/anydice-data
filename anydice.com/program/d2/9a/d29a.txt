function: ROLL:s higher percentage {
 if 1@ROLL = 0 {
  \ Rolling 00 counts as 100. \
  result: 100
 }
 if 2@ROLL = 0 & 1@ROLL <= 5 {
  \ Rolling a 0 with a 1-5 is a fumble and becomes 1-5. \
  result: 1@ROLL
 }
 \ Use higher number for the 10s and lower for the 1s. \
 result: 1@ROLL * 10 + 2@ROLL
}

output [2d{0..9} higher percentage]

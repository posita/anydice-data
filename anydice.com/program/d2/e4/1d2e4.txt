function: cmp A:n to B:n {
  N: A - B
  R: 0
  if N >= 0 {
    R: 1
  }
  if N >= 3 {
    R: 2
  }
  result: R
}

function: sub A:n and B:n against X {
  N: 0
  if A > B {
    N: A - B
  }
  if B > A {
    N: B - A
  }

  R: 0
  if N <= X {
    R: R + 1
  }
  if X - N >= 2 {
    R: R + 1
  }
  result: R
}


function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

TARGET: [lowest 1 of 2d10]

output TARGET < 1d[reroll {2,3} on d{1..5,11}] + 3 named "reroll"

output TARGET < 1d{1..5,11} + 3 named "rarity"

output TARGET < 1d6 + 3 named "normal"

F: d{-1..1}

function: flip FN:n up for DICE:s {
  SORTED_DICE: [reverse [sort DICE]]
  TOTAL: 0
  loop N over {1..FN} {
    TOTAL: TOTAL + [absolute N@SORTED_DICE]
  }
  loop N over {FN + 1..#SORTED_DICE} {
    TOTAL: TOTAL + N@SORTED_DICE
  }
  result: TOTAL
}

function: flip FN:n down for DICE:s {
  SORTED_DICE: [sort DICE]
  TOTAL: 0
  loop N over {1..FN} {
    TOTAL: TOTAL - [absolute N@SORTED_DICE]
  }
  loop N over {FN + 1..#SORTED_DICE} {
    TOTAL: TOTAL + N@SORTED_DICE
  }
  result: TOTAL
}

function: ignore FN:n minuses for DICE:s {
  SORTED_DICE: [reverse [sort DICE]]
  TOTAL: 0
  loop N over {1..FN} {
    if N@SORTED_DICE > 0 {
      TOTAL: TOTAL + [absolute N@SORTED_DICE]
    }
  }
  loop N over {FN + 1..#SORTED_DICE} {
    TOTAL: TOTAL + N@SORTED_DICE
  }
  result: TOTAL
}

function: ignore FN:n pluses for DICE:s {
  SORTED_DICE: [sort DICE]
  TOTAL: 0
  loop N over {1..FN} {
    if N@SORTED_DICE < 0 {
      TOTAL: TOTAL + N@SORTED_DICE
    }
  }
  loop N over {(FN + 1)..#SORTED_DICE} {
    TOTAL: TOTAL + N@SORTED_DICE
  }
  result: TOTAL
}

output 4dF named "Regular Roll"
output 4dF + 2 named "Roll +2"
output [highest 4 of 5dF] named "Highest 4 of 5 dice"
output [flip 1 up for 4dF] named "Tachyon Squadron Flip Up"
output [flip 1 down for 4dF] named "Tachyon Squadron Flip Down"
output [ignore 2 minuses for 4dF] named "Ignore Minuses"
output [ignore 2 pluses for 4dF] named "Ignore Pluses"

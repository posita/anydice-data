function: explode VALUES:s of SEQUENCE:s {
    EXPLODES: 0
    loop P over {1..#VALUES} {
        EXPLODES: EXPLODES+(P@VALUES = SEQUENCE)
    }
    result: EXPLODES
}

function: get successes of SKILL:s with EXPLODES:s of SEQUENCE:s {
    COUNT: [count SKILL in SEQUENCE]
    EXPLODES: [explode EXPLODES of SEQUENCE]
    COUNT: COUNT + [count SKILL in EXPLODESd6]
}

output [count {6,5,4,3,2} in 1d6]

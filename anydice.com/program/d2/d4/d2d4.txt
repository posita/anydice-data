ATTRIBUTE: 2
SKILL: 2
DC: 10

FAIL: 0
SUCCEED: 1

function: check ROLL:n {
 if ROLL + ATTRIBUTE + SKILL >= DC { result: SUCCEED }
 result: FAIL
}


function: gcheck ROLL:n ROLLTWO:n {
 if ROLL > ROLLTWO { DICE: ROLL }
 else { DICE: ROLLTWO }
 if DICE + ATTRIBUTE + SKILL + 4 >= DC { result: SUCCEED }
 result: FAIL
}


output [gcheck 1d4 1d4] named "Roll with Grit"
output [check 2d4+1] named "+1 Bonus"
output [check 2d4] named "No bonus or penalty"
output [check 2d4-1] named "-1 Penalty"

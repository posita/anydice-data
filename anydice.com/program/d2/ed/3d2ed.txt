function: numberdice A:n dice B:s diceface N:n numberdepth C:n{
  if A > 1{
    if  B >= N & C > 1{
        result: B + [numberdice A-1 dice 1dN diceface N numberdepth C] + [dice 1dN diceface N numberdepth C - 1]
    }
    else{
        result: B + [numberdice A-1 dice 1dN diceface N numberdepth C]
    }
  }
  else{
    result: [dice 1dN diceface N numberdepth C]
  }
}

function: dice A:s diceface N:n numberdepth C:n{
  if  A >= N & C > 1{
    result: A + [dice 1dN diceface N numberdepth C - 1]
  }
  else{
      result: A
  }
}

function: cost A:n {
 if A < 14 {
  result: A - 8
 } else {
  if A < 17 {
   result: A * 2 - 21
  } else {
   result: A * 3 - 37
  }
 }
}

ABILITIES: 6d(2d6+5)
TOTAL: 0
loop I over {1..#ABILITIES} { 
 TOTAL: TOTAL + [cost I@ABILITIES]
}
output TOTAL named "cost 2d6+5"
\output ABILITIES named "average 2d6+5"\

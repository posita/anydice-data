 result: SEQUENCE
}

}

\************************** Highest/Lowest into Target Count ****************************\

\apply highest then target\
function: highest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLL >= TARGET) }
 result: COUNT
}
\apply lowest then target\
function: lowest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 ROLLR: [reverse ROLL]
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLLR >= TARGET) }
 result: COUNT
}

\apply target then limit the max (highest or lowest doesn't matter)\
function: count from ROLL:s on TARGET:n keep NUMBER:n {
 COUNT: ROLL >= TARGET
 if(COUNT > NUMBER) { COUNT: NUMBER }
 result: COUNT
}

\************************** Comparison Functions *********************************\
function: compare bodies ATT_B:n with DEF_B:n {
  if(ATT_B > DEF_B) { result: ATT_B-DEF_B}
  result: 0
}
function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  result: -1       \VAL_X < VAL_Y\
}

\************************** Sequence Functions ****************************\

function: phaseone ATT:s DEF:s {                     \find the environment modifiers\
  SIZE_MODS: [compare bodies 13@ATT with 13@DEF]     \compare the nbr of bodies\
  VELOCITY_MODS: [compare 12@DEF with 11@ATT]        \compare the velocity for the dodge capacity\

  ATT_DICES: (1@ATT +SIZE_MODS)                      \modifier to Hit-pool\
  DEF_DICES: (6@DEF +VELOCITY_MODS)                  \modifier to of Save-pool\


  ATT_NEWTARGET: 2@ATT+5@ATT                         \apply target-modifiers for Hit\
  DEF_NEWTARGET: 7@ATT+10@ATT                        \apply target-modifiers for Save\

  ATT_DIR: 0@ATT-0@DEF                           \find H/L-modifiers for Hit, giving the keep direction\
  DEF_DIR: 0@DEF-0@ATT                           \find H/L-modifiers for Save, giving the keep direction\

  ATT_SUCCESSES: [highlow direction ATT_DICES ATT_DIR ATT_NEWTARGET]     \find Hit-Successes\
  DEF_SUCCESSES: [highlow direction DEF_DICES DEF_DIR DEF_NEWTARGET]     \find Save-Successes\

  ATT_FINAL: ATT_SUCCESSES + (4@ATT)        \find bonus successes from value-modifiers to hit roll\
  DEF_FINAL: DEF_SUCCESSES + (9@DEF)        \find bonus successes from value-modifiers to save roll\


  \freeze the hits-dices & saves-dices sequences\
  result: (ATT_FINAL > DEF_FINAL) \[phasetwo ATT_DICES DEF_DICES ATT DEF]\
}

function: phasetwo ATT_POOL:n DEF_POOL:n  ATT:s DEF:s {       \Analysis of the rolls\




  result: (ATT_FINAL > DEF_FINAL)
}

\************************** Functions : Highest/Lowest into Target Count (& vice versa) *******************\

\apply Highest then Target Number\
function: highest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLL >= TARGET) }
 result: COUNT
}
\apply Lowest then Target Number\
function: lowest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 ROLLR: [reverse ROLL]
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLLR >= TARGET) }
 result: COUNT
}

\apply Target Number then limit the max (Highest or Lowest doesn't matter)\
function: count from ROLL:s on TARGET:n keep NUMBER:n {
 COUNT: ROLL >= TARGET
 if(COUNT > NUMBER) { COUNT: NUMBER }
 result: COUNT
}

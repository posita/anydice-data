ATTRIBUTEPC: 2
ATTRIBUTENPC: 2
SKILLPC: 2
SKILLNPC: 2

FAIL: 0
SUCCEED: 1

function: vscheck ROLLONE:n vs ROLLTWO:n {
 if ROLLONE + ATTRIBUTEPC + SKILLPC >= ROLLTWO + ATTRIBUTENPC + SKILLNPC { result: SUCCEED }
 result: FAIL
}

output [vscheck 2d4+2 vs 2d4] named "VS +2 Bonus"
output [vscheck 2d4+1 vs 2d4] named "VS +1 Bonus"
output [vscheck 2d4 vs 2d4] named "VS Base"
output [vscheck 2d4-1 vs 2d4] named "VS -1 Penalty"
output [vscheck 2d4-2 vs 2d4] named "VS -2 Penalty"

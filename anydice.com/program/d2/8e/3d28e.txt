MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= THREAT & ROLL + ATTACK >= DEFENSE | ROLL = 20 {result: CRITICALHIT}
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: rhythmattack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= THREAT & ROLL + ATTACK >= DEFENSE | ROLL = 20 {
  PAPENALTY: [highest of PAPENALTY+1 and 0]
  result: CRITICALHIT}
 if ROLL + ATTACK >= DEFENSE {
  PAPENALTY: [highest of PAPENALTY+1 and 0]
  result: HIT }
 result: MISS
}

function: secondaryattack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= THREAT & ROLL + SECONDARYATTACK >= DEFENSE | ROLL = 20 {result: CRITICALHIT}
 if ROLL + SECONDARYATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: furiousattack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= THREAT & ROLL + ATTACK + 2>= DEFENSE | ROLL = 20 {result: CRITICALHIT}
 if ROLL + ATTACK + 2>= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: (CRITICAL d DAMAGE) + SNEAKATTACKDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: DAMAGE + SNEAKATTACKDAMAGE
 }
 result: MISSDAMAGE
}

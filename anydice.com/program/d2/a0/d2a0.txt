function: cost A:n {
 if A < 14 {
  result: A - 8
 } else {
  if A < 17 {
   result: A * 2 - 21
  } else {
   result: A * 3 - 37
  }
 }
}

ABILITIES: 6d(3d6)
TOTAL: 0
loop I over {1..#ABILITIES} {
 TOTAL: TOTAL + [cost I@ABILITIES]
}

output TOTAL named "3d6+2"

ABILITIES: 6d({1,2,3}@4d6)
TOTAL: 0
loop I over {1..#ABILITIES} {
 TOTAL: TOTAL + [cost I@ABILITIES]
}

output TOTAL named "4d6 choose 3"

ABILITIES: 6d(2d6+5)
TOTAL: 0
loop I over {1..#ABILITIES} {
 COST: [cost I@ABILITIES]
 TOTAL: TOTAL + COST
}

output TOTAL named "2d6"

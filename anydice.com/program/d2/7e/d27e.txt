DICE: {4,6,8,10,12}

loop I over DICE{
 output dI named "[I]"
}

loop I over DICE{
 loop J over DICE{
  if J >= I {
   output [highest of dI and dJ] named "[I][J]"
  }
 }
}

loop I over DICE{
 loop J over DICE{
  if J >= I {
   loop K over DICE{
    if K >= J {
     output [highest of dI and [highest of dJ and dK]] named "[I][J][K]"
    }
   }
  }
 }
}

STR: 3
FIN: 3
MELEE:3
DEFENSEBONUS: 6
ARMOR: 0
ATTACK: STR + MELEE
DEFENSE: 4 + DEFENSEBONUS

function: floor NUM:n {
     if NUM < 0 { result: 0 }
     result: NUM
}

function: bound NUM:n { \Floors numbers below 0 to 0\
     if NUM > 0 { result: 0 }
     result: NUM
}

function: damage of DAMAGE:n PIERCING:n { \Generic damage calculation\
     result: [floor DAMAGE + [bound PIERCING - ARMOR]]
}

function: battack ROLL:n damage DAM:n pierce PIERCE:n{ \Blunt Weapon attack\
 if ROLL + ATTACK >= DEFENSE { result: [damage of DAM+STR PIERCE] }
 result: 0
}

function: sattack ROLL:n damage DAM:n pierce PIERCE:n{ \Sharp weapon attack\
 if ROLL + ATTACK >= DEFENSE { result: [damage of DAM+FIN PIERCE] }
 result: 0
}

function: aattack ROLL:n damage DAM:n pierce PIERCE:n { \Sneak attack with Assassin\
 if ROLL + ATTACK >= DEFENSE { result: [damage of DAM+(2*FIN) PIERCE] }
 result: 0
}

function: fattack ROLL:n damage DAM:n { \IRON FIST STRONK!\
 if ROLL + ATTACK - 2 >= DEFENSE { result: DAM+(2*STR) }
 result: 0
}

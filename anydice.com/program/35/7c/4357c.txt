function: axis A:n B:n C:n D:n E:n {
  J: [count 1 in {A,B,C,D,E}]
  M: 0
  if A = 4 { M : M+1 }
  if B = 6 { M : M +1 }
  if C = 8 { M : M +1 }
  if D = 10 { M : M +1 }
  if E = 12 { M : M +1 }
  result: J*1 + M*10 + 3@[sort {A,B,C,D,E}] * 100
}

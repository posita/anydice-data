ROLL:1d8
BONUS:4
CAP:8
GET:6

function: outcap ROLL:n BONUS:n CAP:n{
   if ROLL + BONUS > CAP {result:ROLL + BONUS - CAP}
   result:0
}

function: buff ROLL:n BONUS:n CAP:n GET:n{
   if ROLL + BONUS >= GET {result:1}
   result:0
}

output[lowest of ROLL+BONUS and CAP] named"判定结果"
output[buff ROLL BONUS CAP GET] named"特效"
output[outcap ROLL BONUS CAP] named"溢出量"

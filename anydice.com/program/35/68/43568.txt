function: tozero R:s
{
  if R=10 { result: 0}
  else { result: R }
}
function: highflip DD:s plus EE:s
{
  D: [tozero DD]
  E: [tozero EE]
  X: (D)*10+E
  Y: (E)*10+D
  if D=0 & E=0 { result: 100 }
  else if X>Y { result: X }
  else { result: Y }
}
function: lowflip DD:s plus EE:s
{
  D: [tozero DD]
  E: [tozero EE]
  X: (D)*10+E
  Y: (E)*10+D
  if D=0 & E=0 { result: 100 }
  else if X>Y { result: Y }
  else { result: X }
}
output [highflip d10 plus d10]
output [highest 1 of 2d100] 
output [lowflip d10 plus d10]
output [lowest 1 of 2d100]

\roll d4-d4\
\if result is 3 or -3, roll tiebreaker d4\
\if tiebreaker rolls 4, result becomes -4 or 4 respectively\
function: fudge emulation D:n with tiebreaker T:n {
  D_ABS: [absolute D]
  if D_ABS < 3 { result: D }
  SIGN: D/D_ABS
  if T = 4 { result: SIGN*4 }
  result: { SIGN*3 }
}
output [fudge emulation d4-d4 with tiebreaker d4] named "d4"

function: minz N:n {
if N < 0 {result: 0}
result: N
}

function: sbhit D:n {
if D = 16 {result: 16 + 1d8}
result: D
}

function: sbattack D:n {
if D > 16 {result: 0}
result: [sbhit 1d16]
}

function: rangerchain D:n {
if D > 16 {result: 0}
result: [minz ([sbhit 1d16] - 1d12)] + [rangerchain 1d20]
}

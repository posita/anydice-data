function: attack ROLL:n plus BONUS:n vs AC:n hitdice HIT_DICE:d bonus BONUS_DAMAGE:n miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 { result: MISS }
    if ROLL >= CRIT_RANGE { 
      result: (HIT_DICE + BONUS_DAMAGE + CRIT) + [attack RECKLESS_ATTACK plus BONUS vs AC hit HIT_DICE miss MISS crit CRIT on CRIT_RANGE]
    }
    if ROLL + BONUS >= AC { 
      result: HIT_DICE + BONUS_DAMAGE + (FEAT_GREAT_WEAPON_MASTER * [attack RECKLESS_ATTACK plus BONUS vs AC hit HIT_DICE miss MISS crit CRIT on CRIT_RANGE]/10)
    }
    result: MISS
}
function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 { result: MISS }
    if ROLL >= CRIT_RANGE { result: HIT + CRIT }
    if ROLL + BONUS >= AC { result: HIT }
    result: MISS
}
function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS vs AC hit DMG miss 0 crit CRIT_DMG on CRIT_RANGE]
}
function: attack ROLL:n plus ATTACK_BONUS:n vs ACS:s {
  \ return the number of times damage is applied on an attack roll of ROLL \
  if ROLL <= 1 { result: 0 }     \ natural 1 is always a miss \
  if ROLL >= CRIT { result: 2 }  \ critical hit deals damage damage twice \
  result: ROLL + ATTACK_BONUS >= ACS
}

      }

    result: DAMAGE
  } else {

    \DAMAGE: DAMAGE + (WPN_TWO_HIT d (WPN_TWO_DICE + d0)) + WPN_TWO_MAGIC + WAILS_DMG

function: cleave ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
  if ROLL + BONUS >= AC { 
    result: [attack ROLL plus BONUS vs AC hit DMG + FRENZY miss 0 crit CRIT_DMG on CRIT_RANGE]

    result: [attack ROLL plus BONUS vs AC hit (DMG - [mod STR] + FRENZY) miss 0 crit CRIT_DMG on CRIT_RANGE]
  }
}

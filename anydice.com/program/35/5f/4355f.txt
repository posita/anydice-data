function: attack ROLL:n plus ATTACK_BONUS:n vs ACS:s {
  \ return the number of times damage is applied on an attack roll of ROLL \
  if ROLL <= 1 { result: 0 }     \ natural 1 is always a miss \
  if ROLL >= CRIT { result: 2 }  \ critical hit deals damage damage twice \
  result: ROLL + ATTACK_BONUS >= ACS
}
function: downtime crime DC:n PAYOUT:n CHECKA:n CHECKB:n CHECKC:n {
  SUCCESS: 0
  if CHECKA >= DC { SUCCESS: SUCCESS + 1 }
  if CHECKB >= DC { SUCCESS: SUCCESS + 1 }
  if CHECKC >= DC { SUCCESS: SUCCESS + 1 }
  if SUCCESS >= 3 {
    result: PAYOUT 
  } else if SUCCESS >= 2 {
    result: PAYOUT / 2 
  } else if SUCCESS >= 1 {
    result: 0 
  } else {
    result: -PAYOUT
  }
}

  if CHECKB >= DC { SUCCESS: SUCCESS + 1 }
  if CHECKC >= DC { SUCCESS: SUCCESS + 1 }
  result: SUCCESS
}

DC: 10

function: ATK_ROLL:n pls ATK_MOD:n with DMG_ROLL:s pls DMG_MOD:n crit CRIT_RANGE:s {
 ATK: ((ATK_ROLL + ATK_MOD) >= DC) * (DMG_ROLL + DMG_MOD)
 if [CRIT_RANGE contains ATK_ROLL] {
  result: ATK + (ATK * ((1d20 + ATK_MOD) >= DC))
 }
 else {
  result: ATK
 }
}

output [1d20 pls 4 with 1d10 pls 3 crit {19..20}] named "Agnus"

output [1d20 pls 1 with 1d8 pls 2 crit {18..20}] + [1d20 pls 1 with 1d6 pls 1 crit {19..20}]  named "Tamen"

output [1d20 pls 6 with 1d10 pls 5 crit {19..20}] named "AgnusRaged"

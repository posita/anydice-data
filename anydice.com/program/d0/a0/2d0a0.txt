function: explode N:n maxlevels L:n {
 if N = 0 {
    if (L <= 0) {result: 6}
    result: 5 + [explode d{1,2,3,4,5,0} maxlevels L-1]
 }
 result: N
}

loop P over {0..15} {
 INCR: 4
 DICE: 1+P/INCR
 ADD: P-(P/INCR)*INCR-1
 output DICEd[explode d{1,2,3,4,5,0} maxlevels 2]+ADD named "[DICE]d add [ADD]"
}

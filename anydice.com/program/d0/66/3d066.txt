COVER: 0
SAVE: 5
WARD: 6

function: shoot hit on HIT_ON:n wound on WOUND_ON:n rend REND:n damage DAMAGE:n {
  result: [hit HIT_ON]d[wound WOUND_ON]d[save REND]d[ward DAMAGE]
}

function: hit HIT_ON:n {
  result: [count {(HIT_ON+COVER)..6} in d6]
}

function: wound WOUND_ON:n {
  result: [count {WOUND_ON..6} in d6]
}

function: save REND:n {
  result: [count {1..(SAVE-1+REND)} in d6]
}

function: ward DAMAGE:n {
  result: [count {1..(WARD-1)} in {DAMAGE}d6]
}

\ Admiral \
output 3d[shoot hit on 3 wound on 3 rend 1 damage 2] named "admiral"

\ Skywardens \
output 6d[shoot hit on 3 wound on 3 rend 1 damage 1]
+ 2d6d[shoot hit on 4 wound on 4 rend 0 damage 1] named "skywardens"

\ Arkanaut company: melee unit \
output 8d[shoot hit on 4 wound on 4 rend 0 damage 1] named "arkanaut melee"

\ Arkanaut company: ranged unit \
output 6d[shoot hit on 4 wound on 4 rend 0 damage 1]
+ 2d6d[shoot hit on 4 wound on 4 rend 0 damage 1]
+ d[shoot hit on 4 wound on 3 rend 2 damage d3] named "arkanaut ranged"

\ Frigate \
output 2d[shoot hit on 4 wound on 3 rend 2 damage d6]
+ 4d[shoot hit on 3 wound on 3 rend 1 damage 2] named "frigate"

\ Total \
output 3d[shoot hit on 3 wound on 3 rend 1 damage 2]
+ 6d[shoot hit on 3 wound on 3 rend 1 damage 1]
+ 2d6d[shoot hit on 4 wound on 4 rend 0 damage 1]
+ 8d[shoot hit on 4 wound on 4 rend 0 damage 1]
+ 6d[shoot hit on 4 wound on 4 rend 0 damage 1]
+ 2d6d[shoot hit on 4 wound on 4 rend 0 damage 1]
+ d[shoot hit on 4 wound on 3 rend 2 damage d3]
+ 2d[shoot hit on 4 wound on 3 rend 2 damage d6]
+ 4d[shoot hit on 3 wound on 3 rend 1 damage 2] named "total"

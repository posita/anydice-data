function: success DIE:n and SUCCESSFUL:n and MAX:n {
 if DIE <= SUCCESSFUL & DIE < MAX { result: 0 }
 if DIE > SUCCESSFUL & DIE < MAX { result: 1 }
 if DIE >= MAX { result: 2 }
 result: -1
}

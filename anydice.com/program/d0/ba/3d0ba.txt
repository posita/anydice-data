Q: {0..5}

function: limits SEQUENCE:n ZERO:n MIN:n {
 if (SEQUENCE < ZERO){result: 0}
 if (SEQUENCE < MIN){result: MIN}
 result: SEQUENCE
}
function: crop SEQUENCE:n MIN:n MAX:n {
 if (SEQUENCE <= MIN){result: 0}
 if (SEQUENCE > MAX){result: MAX}
 result: SEQUENCE
}
function: crop SEQUENCE:n {
 if (SEQUENCE < 0){result: 0}
 result: SEQUENCE
}

function: thresholds SEQUENCE:n ONE:n TWO:n THREE:n FOUR:n FIVE:n SIX:n {
 if (SEQUENCE >= SIX){result: SIX}
 if (SEQUENCE >= FIVE){result: FIVE}
 if (SEQUENCE >= FOUR){result: FOUR}
 if (SEQUENCE >= THREE){result: THREE}
 if (SEQUENCE >= TWO){result: TWO}
 if (SEQUENCE >= ONE){result: ONE}
 if (SEQUENCE < ONE){result: 0}
 result: SEQUENCE
}

function: damages SEQUENCE:n ONE:n TWO:n THREE:n FOUR:n FIVE:n SIX:n {
 if (SEQUENCE >= SIX){result: 6}
 if (SEQUENCE >= FIVE){result: 5}
 if (SEQUENCE >= FOUR){result: 4}
 if (SEQUENCE >= THREE){result: 3}
 if (SEQUENCE >= TWO){result: 2}
 if (SEQUENCE >= ONE){result: 1}
 if (SEQUENCE < ONE){result: 0}
 result: SEQUENCE
}

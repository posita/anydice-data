function: highest N:n of A:s B:s {
    result: {1..N}@[sort {A,B}]
}

function: lowest N:n of A:s B:s{
 result: {1..N}@[reverse [sort {A,B}]]
}

function: count N:n of A:d B:d{
  COUNT: 0
  loop P over {1..N} {
   COUNT: COUNT + [highest 1 of A B]
  }
  result: COUNT
}

function: countlow N:n of A:d B:d{
  COUNT: 0
  loop P over {1..N} {
   COUNT: COUNT + [highest 1 of A B] + [lowest 1 of A B]
  }
  result: COUNT
}

\ Low level players \
output 1d8 named "Only Physique - low"
output [count 1 of 1d4 1d6] named "Physique with Resilience - low"
output [count 1 of 2d4 2d4] named "Physique with Resilience with Adaptability - low"
output [countlow 1 of 1d4 1d4] named "Physique with Res with Adaptibility 2 - low"

WEAPONBONUS: 0  \+? Weapon \

AC:16

POISONSAVE: 19       \Basic: 10; Serpent Venom: 11 ; Wyvern Poison: 15 ; Purple Worm: 19 \

\CALCULATION STUFF BELOW, DO NOT MODIFY\
HITBONUS: DEXBONUS + WEAPONBONUS
DAMAGEBONUS: DEXBONUS+WEAPONBONUS
CRITMULT:3 \Half-orc trait\


if ENLARGE >=1{
ENLARGE: 1d4
}
if HEX >=1{
HEX: 1d6
}
if SHARPSHOOTER >=1{
HITBONUS: HITBONUS - 5
DAMAGEBONUS: DAMAGEBONUS+10
}
if ARCHERY >=1{
HITBONUS: HITBONUS + 2
}
if POISON >=1{
POISON: POISONDAMG
}


function: proficiency LEVEL {
result: ((LEVEL-1)/4) + 2
}


function: hit DICE HITS LEVEL {
OUTPUT:0
loop N over {1..HITS}{
OUTPUT: OUTPUT + (DICE+DAMAGEBONUS+ENLARGE+HEX+((1d20+ENEMYCON)>=POISONSAVE))*((1d20+HITBONUS+[proficiency LEVEL])>=AC)
}
result: OUTPUT
}

function: chit DICE HITS LEVEL {
OUTPUT:0
loop N over {1..HITS}{
OUTPUT: OUTPUT + (CRITMULTd(DICE)+DAMAGEBONUS+2d(ENLARGE)+2d(HEX)+POISON*((1d20+ENEMYCON)<POISONSAVE))*((1@2d20+HITBONUS+[proficiency LEVEL])>=AC)
}
result: OUTPUT
}


function: sneak ATTACKS LEVEL {
result: 2d6 * ((1@ATTACKSd20+HITBONUS+[proficiency LEVEL])>= AC) 
}

function: csneak ATTACKS LEVEL {
result: 4d6 * ((1@ATTACKSd20+HITBONUS+[proficiency LEVEL])>= AC) 
}


function: supdice FLEVEL {
OUTPUT: (4+((FLEVEL+1)/8))
if MARTIALFEAT >=1{
OUTPUT: OUTPUT + 1
}
result: OUTPUT
}


function: supdamage FLEVEL {
result:(8+((FLEVEL-2)/8*2))
}

function: supd ATTEMPTS FLEVEL LEVEL{
ATTACKS: ATTEMPTS
if ATTEMPTS > [supdice FLEVEL]{
ATTACKS: [supdice FLEVEL]
}

OUTPUT:0
loop N over {1..ATTACKS}{
OUTPUT: OUTPUT + 1d[supdamage FLEVEL] *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

function: csupd ATTEMPTS FLEVEL LEVEL{
ATTACKS: ATTEMPTS
if ATTEMPTS > [supdice FLEVEL]{
ATTACKS: [supdice FLEVEL]
}
OUTPUT:0
loop N over {1..ATTACKS}{
OUTPUT: OUTPUT + 2d(1d[supdamage FLEVEL]) *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

function: dsmite ATTEMPTS SPELLONE SPELLTWO LEVEL{
\ALWAYS ASSUMING THAT WE CAN CAST ALL SPELLS WE NEED TO\
OUTPUT:0
loop N over {1..SPELLTWO }{
OUTPUT: OUTPUT + 3d6 *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
loop N over {1..SPELLONE }{
OUTPUT: OUTPUT + 2d6 *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

function: cdsmite ATTEMPTS SPELLONE SPELLTWO LEVEL{
\ALWAYS ASSUMING THAT WE CAN CAST ALL SPELLS WE NEED TO\
OUTPUT:0
loop N over {1..SPELLTWO }{
OUTPUT: OUTPUT + 2d(3d6) *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
loop N over {1..SPELLONE }{
OUTPUT: OUTPUT + 2d(2d6) *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

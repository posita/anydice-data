function: X:n conditional {

if X - MOD = 1 {result: 0}

if X - MOD = 20 {result: 2}

if X >= AC {result: 1}

else { result: 0 }
}

function: X:n hittodamage D:d{

if X = 1 {result: 1dD+DAM}
if X = 2 {result: 2dD+DAM}
else { result: 0 }
}



AC: 12
MOD: 8
DAM: 4

HIT: [d20 + MOD conditional]
AHIT: [highest 1 of 1d[d20 + MOD conditional]]

output 2d[HIT hittodamage 1d6] named "twice without advantage"
output [AHIT hittodamage 1d6+3d6] named "once with advantage"
output 2d[AHIT hittodamage 1d6+3d6] named "twice with advantage"

CHECK: 14
MEDIC: 0
CHECK_IS_ALWAYS: 0
ALWAYS_HEALING: 0
SUCCESS_IS_CRIT: 0

function: heal ROLL:n vs DC:n with HEAL:n {
DOS: 0
if ROLL = 1 { DOS: -1 }
if ROLL = 20 { DOS: 1 }

CHECK_TOTAL: ROLL + CHECK
if CHECK_IS_ALWAYS > 0 {
CHECK_TOTAL: CHECK_IS_ALWAYS
DOS: 0
}

if CHECK_TOTAL >= DC + 10 { DOS: DOS + 2 }
if CHECK_TOTAL >= DC { DOS: DOS + 1 }
if CHECK_TOTAL <= DC - 10 { DOS: DOS - 1 }

if SUCCESS_IS_CRIT = 1 {
if DOS = 1 {
DOS: 2
}
}

if DOS = 0 { result: 0 + ALWAYS_HEALING }
if DOS < 0 { result: -1d8 + ALWAYS_HEALING }
if DOS = 1 { result: 2d8+HEAL + ALWAYS_HEALING }
if DOS > 1 { result: 4d8+HEAL + ALWAYS_HEALING }
result: 0 + ALWAYS_HEALING
}

output [heal d20 vs 15 with 0] named "VS DC 15"
output [heal d20 vs 20 with 10 + (MEDIC * 5)] named "VS DC 20"
output [heal d20 vs 30 with 30 + (MEDIC * 10)] named "VS DC 30"
output [heal d20 vs 40 with 50 + (MEDIC * 15)] named "VS DC 40"

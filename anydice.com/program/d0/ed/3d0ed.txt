function: crop SEQUENCE:n MIN:n MAX:n {                  \ limit the display range between [min;max] \
 if (SEQUENCE < MIN){result: MIN}
 if (SEQUENCE > MAX){result: MAX}
 result: SEQUENCE
}
function: crop SEQUENCE:n {
 if (SEQUENCE < 0){result: 0}
 result: SEQUENCE
}
function: halve SEQUENCE:n {                  \ divide per 2, rounded up \
 DIV: SEQUENCE/2
 TEMP: DIV*2
 if (DIV=TEMP){result: DIV}
 result: (SEQUENCE+1)/2
}

function: phaseone ATT:s DEF:s {                \Find the environment modifiers\
  SIZE_MODS: [compare bodies 13@ATT with 13@DEF]           \compare the nbr of bodies\
  VELOCITY_MODS: [compare 12@DEF with 11@ATT]        \compare the velocity for the dodge capacity\

  ATT_DICES: (1@ATT +SIZE_MODS)                      \final number of hits-dices to roll\
  DEF_DICES: (6@DEF +VELOCITY_MODS)                  \final number of saves-dices to roll\

  \freeze the hits-dices & saves-dices sequences\
  result: [phasetwo ATT_DICESdQ DEF_DICESdQ ATT DEF]
}

function: phasetwo ATT_ROLL:d DEF_ROLL:d  ATT:s DEF:s {       \Analysis of the rolls\
  ATT_NEWTARGET: 2@ATT+5@ATT                                  \apply target-modifiers for Hit\
  DEF_NEWTARGET: 7@ATT+10@ATT                                 \apply target-modifiers for Save\

  ATT_SUCCESSES: [overflowing pairing ATT_ROLL on ATT_NEWTARGET]     \find Hit-Successes\
  DEF_SUCCESSES: [overflowing pairing DEF_ROLL on DEF_NEWTARGET]     \find Save-Successes\

  ATT_FINAL: ATT_SUCCESSES + (4@ATT /ATT_NEWTARGET)        \find bonus successes from value-modifiers to hit roll\
  DEF_FINAL: DEF_SUCCESSES + (9@DEF /DEF_NEWTARGET)        \find bonus successes from value-modifiers to save roll\

  DAMAGES: ATT_FINAL - DEF_FINAL                           \TEST 1\
  \DAMAGES: [halve ATT_FINAL] - [halve DEF_FINAL]\
  result: [crop DAMAGES 0 4]
}


\******** Functions to find Successes on chosen target **********\

function: overflowing pairing SEQUENCE:s on TARGET:n {
  TOTAL: 0
  \count of number in sequence\
  UN:    SEQUENCE=1
  DEUX:  SEQUENCE=2
  TROIS: SEQUENCE=3
  TEMP: {UN, DEUX, TROIS}
  \sum of chosen numbers'occurencies\
  TOTAL: {TARGET..3}@TEMP                        \final count of Successes\

  \boolean checks : 0 or 1 multiplicators\
  DT: DEUX>=TROIS
  UT: UN>=TROIS
  UD: UN>=DEUX

  if(TARGET=6){result: TROIS/2}
  if(TARGET=5){result: (TROIS+ (DT*TROIS)+(!DT*DEUX)) /2}
  if(TARGET=4){result: (TROIS+ (UT*(TROIS+DEUX)) + (!UT*(UN+DEUX))) /2}
  if(TARGET=3){result: TOTAL + ( (DEUX+ (UD*DEUX)+(!UD*UN)) /2)}
  if(TARGET=2){result: TOTAL + (UN/2)}
  if(TARGET=1){result: TOTAL}
  result: TOTAL
}

  if(ATT_B > DEF_B) { result: ATT_B-DEF_B}
  result: 0
}
function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  result: -1       \VAL_X < VAL_Y\
}

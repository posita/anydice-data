DC: 10
CRIT_RANGE: {19, 20}
function: ATK_ROLL:n plus ATK_MOD:n with DMG_ROLL:s and DMG_MOD:n {
 ATK: ((ATK_ROLL + ATK_MOD) >= DC) * (DMG_ROLL + DMG_MOD)
 if [CRIT_RANGE contains ATK_ROLL] {
  result: ATK + ATK * ((1d20 + ATK_MOD) >= DC)
 }
 else {
  result: ATK
 }
}

output [1d20 plus 1 with 1d6 and 2]

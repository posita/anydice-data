set "maximum function depth" to 5
function: explode N:n {
  if N=-1 { result: N - [explode d{-1..6}] + 1 }
  if N= 6 { result: N + [explode d{-1..6}] - 1 }
  result: N
}

function: grade N:n with mod M:n {
  K: M + N
  if         K >= 12  { result:  3 }
  if (0 < K & K < 12) { result:  2 }
  if      K = 0       { result:  1 }
  if      K = -1      { result: -1 }
  if -6 < K & K < -3  { result: -2 }
  if -6 >= K          { result: -3 }
}

loop MOD over {-1..10} {
  output [grade [explode d{-1..6}] with mod MOD] named "mod [MOD]"
}

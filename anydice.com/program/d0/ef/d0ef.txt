function: pool R:n again N:n {
 if N >= R { result: 1 + [pool R again d6] }
 result: N >= 6
}

function: bigpool R:n again N:n {
 if N >= R { result: 1 + [bigpool R again d8] }
 result: N >= 6
}

HOT: [pool 6 again d6]
VERYHOT: [bigpool 6 again d8]

loop N over {1..2} {
 output [lowest of 10 and NdHOT] named "[N]d"
}

loop N over {1..2} {
 output [lowest of 10 and NdVERYHOT] named "[N]dVH"
}

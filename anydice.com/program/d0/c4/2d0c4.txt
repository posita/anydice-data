MAX_POOL_SIZE: 2
DICE_NUMBER_SIDES: 4

function: roll DICE_SIZE:n and NUM_DICE:n under THRESHOLD:n 
{
   OUT: 0
   loop I over {1..NUM_DICE}
   {
      OUT: OUT | 1dDICE_SIZE <= THRESHOLD
   }
   result: OUT
}

loop I over {1..MAX_POOL_SIZE}
{
   loop J over {1..DICE_NUMBER_SIDES}
   {
      output [roll DICE_NUMBER_SIDES and I under J] named "Pool Size: [I] - Success Under Or Equal To: [J]"
   }
}

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: 0 }
 if ROLL = 20 {
  result: 1
 }
 if ROLL + ATTACK >= DEFENSE { result: 1}
 result: 0
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = 1 {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

function: disadvantage{
 result: [damage for [attack [lowest 1 of 2d20] vs DEFENSE]]
}

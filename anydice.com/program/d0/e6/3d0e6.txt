
function: reroll REROLL:s on ROLL:n and ROLLOTHER:n to REPLACEMENT:d {
    if [REROLL contains ROLL + ROLLOTHER] {
      if ROLL > ROLLOTHER {
        result: REPLACEMENT + ROLL
      }
        result: REPLACEMENT + ROLLOTHER
    }
    result: ROLL + ROLLOTHER
}


output [reroll {1,2,3,4,5,6} on 1d6 and 1d6 to d6]

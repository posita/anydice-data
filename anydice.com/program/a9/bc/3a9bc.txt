CRIT: 2
PASS: 1
FAIL: 0
EVA: 10
GRIT: 2

function: lancer ROLL:n 
{
 if ROLL + GRIT >= 20 { result: CRIT }
 else if ROLL + GRIT >= EVA { result: PASS }
 else if ROLL + GRIT <= EVA - 1 { result: FAIL }
}

output [lancer d20 - [highest 1 of 6d6]] named "+6 Difficulty"
output [lancer d20 - [highest 1 of 5d6]] named "+5 Difficulty"
output [lancer d20 - [highest 1 of 4d6]] named "+4 Difficulty"
output [lancer d20 - [highest 1 of 3d6]] named "+3 Difficulty"
output [lancer d20 - [highest 1 of 2d6]] named "+2 Difficulty"
output [lancer d20 - [highest 1 of 1d6]] named "+1 Difficulty"

output [lancer d20] named "Standard Roll" 

loop ACCURACY over {1..6}
{ output [lancer d20 + [highest 1 of ACCURACYd6]] named "+[ACCURACY] Accuracy" }

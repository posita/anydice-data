function: roll A:n m B:n u Y:n s Z:n {
  if A=Y {
    result: Z+A+100
  } if A<Y { 
    result: Z+A
  } if A>Y {
    if A=B {
      result: Z-100
    } else {
      result: Z
    }
  }
}

function: adv A:n a B:n m C:n u Y:n s Z:n {
  if A=Y {
    P: Y+100
  } if A<Y {
    P: A
  } else {
    if A=C {
      P: -100
    } else {
      P: 0
    }
  }

  if B=Y {
    Q: Y+100
  } if B<Y {
    Q: B
  } else {
    if B=C {
      Q: -100
    } else {
      Q: 0
    }
  }

  if P<Q {
    if P<0 {
      if Q=0 {
        result: Z+P
      } else {
        result: Z+Q
      }
    } else {
      result: Z+Q
    } 
  } if Q<P {
    if Q<0 {
      if P=0 {
        result: Z+Q
      } else {
        result: Z+P
      }
    } else {
      result: Z+P
    }
  } else {
    if P<0 {
      result: Z+P+Q
    } else {
      result: Z+P
    }
  }
}

function: dis A:n a B:n m C:n u Y:n s Z:n {
  if A=Y {
    P: Y+100
  } if A<Y {
    P: A
  } else {
    if A=C {
      P: -100
    } else {
      P: 0
    }
  }

  if B=Y {
    Q: Y+100
  } if B<Y {
    Q: B
  } else {
    if B=C {
      Q: -100
    } else {
      Q: 0
    }
  }

  if P<Q {
    result: Z+P
  } if Q<P {
    result: Z+Q
  } else {
    if P<0 {
      result: Z+P+Q
    } else {
      result: Z+P
    }
  }
}

output [roll d20 m 20 u 10 s 0]
output [adv d20 a d20 m 20 u 10 s 0]
output [dis d20 a d20 m 20 u 10 s 0]

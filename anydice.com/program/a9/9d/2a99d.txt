function: rollcrit TOHIT:d plus MOD:n for DAM:d against AC:n {
TOHITMAX: [maximum of TOHIT]
result: [rollhelper TOHIT TOHITMAX plus MOD for DAM against AC]
}


function: rollhelper TOHIT:n TOHITMAX:n plus MOD:n for DAM:d against AC:n {
if TOHIT = TOHITMAX {result:DAM+DAM}
else if TOHIT+MOD >= AC {result:DAM}
else {result:0}
}

function: rolltohit TOHIT:n plus MOD:n for DAM:d against AC:n {
if TOHIT+MOD >= AC {result:DAM}
else {result:0}
}

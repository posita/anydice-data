function: two of A:n and B:n and C:n sum to RANGE:s {
  FOUND: 0
  loop TARGET over RANGE {
    if (A + B) = TARGET {
      FOUND: FOUND + 1
    } else if (A + C) = TARGET {
      FOUND: FOUND + 1
    } else if (B + C) = TARGET {
      FOUND: FOUND + 1
    }
  }
  result: FOUND
}

function: skill NUMBER:s dice SEQUENCE:s {
  SORTED: [sort SEQUENCE]
  R: 1@SORTED
  S: [count NUMBER in SEQUENCE]
  if S>0
  {
    S:S-1
  }
  result: R+S
}

function: attacker skill AS:s dice AD:s vs defender skill DS:s dice DD:s
{
  N:[skill AS dice AD]-[skill DS dice DD]
  if N<-1
  {
    result: -1
  } else
  {
    result: N
  }
}

output [skill {4,5,6} dice 2d6]
output [attacker skill {4,5,6} dice 2d6 vs defender skill {4,5,6} dice 2d6]

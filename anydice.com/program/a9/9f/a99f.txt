set "explode depth" to 5
set "maximum function depth" to 5

function: starwars DIES:d WILD:n
{
 if      WILD = 1 { result: DIES - [highest 1 of DIES]}
 else if WILD = 6 { result: DIES + WILD + [explode d6]}
 else             { result: DIES + WILD}
}

loop DIES over {1..4}{
loop ADD  over {0..2}
{
 output [lowest of [highest of
 ([starwars (DIES-1)d6 1d6] + ADD)
 and 1] and 30] named "[DIES]d6+[ADD] (StarWars D6)"
}}

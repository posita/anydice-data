function: rafpark UN:n DEUX:n TROIS:n {
  if ((UN + DEUX) = 2) {
    result: TROIS
  } else if ((UN + TROIS) = 2) {
    result: DEUX
  } else if ((TROIS + DEUX) = 2) {
    result: UN
  } else if ((UN + DEUX) = 12) {
    result: UN + DEUX + TROIS + 2
  } else if ((UN + TROIS) = 12) {
    result: UN + DEUX + TROIS + 2
  } else if ((TROIS + DEUX) = 12) {
    result: UN + DEUX + TROIS + 2
  } else {
    result: UN + DEUX + TROIS
  }
}

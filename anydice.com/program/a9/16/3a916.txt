set "explode depth" to 1

CRIT: d{0:5, 1}

function: hard crit N:n {
  EXTRAS: NdCRIT
  TOTAL: EXTRAS + EXTRAS dCRIT
  result: TOTAL >= (N + EXTRAS)/2
}

function: easy crit N:n {
  result: Nd[explode dCRIT] >= N/2
}

function: fumble N:n {
  result: NdCRIT >= N/2
}


loop N over {4..15} {
  output [easy crit N] named "easy crit [N]"
  output [hard crit N] named "hard crit [N]"
  output [fumble N] named "fumble [N]"
}

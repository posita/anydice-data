CRIT: 20
\ Target's Armour Class \
AC: 15

CON_MOD: 2

\ Attack Roll Against Target \
ATTACK_ROLL: 1d20

SAVING_THROW: 1d20

BASE_DMG: 1d10 + 1d6

DC: 15

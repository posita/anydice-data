function: reroll A:n if below three with B:d {
 if A < 3 { result: B }
 result: A
}

MOD: 4
CHA: 3

X: [reroll d6 if below three with d6]
Y: [reroll d6 if below three with d6]
Z: X + Y + MOD + d6 + CHA
output Z + Z named "GWF"

E: [reroll d10 if below three with d10] + MOD + d6 + CHA
F: [reroll d4 if below three with d4] + MOD + d6 + CHA
output E + F + E named "GWF and Polearm Mastery"

Q: d8 + MOD + d6 + CHA
R: d8 + MOD + d6 + CHA
output Q + R + Q named "TWF and DW Mastery"

BLAST: 1d10 + MOD + 1d6
output BLAST + BLAST + BLAST named "Blast"

DUEL: 1d8 + 2 + MOD + 1d6 + CHA
output DUEL + DUEL named "Duel"

ONE: {-1:1, 0:4, 1:5}
TWO: {-1:2, 0:3, 1:5}
THREE: {-1:3, 0:2, 1:5}
FOUR: {-1:4, 0:1, 1:5}
FIVE: {-1:5, 1:5}

loop X over {1..10} {
  output XdONE > 0 named "/// [X]d (1)"
  output XdTWO > 0 named "(2)"
  output XdTHREE > 0 named "(3)"
  output XdFOUR > 0 named "(4)"
  output XdFIVE > 0 named "(5)"
}

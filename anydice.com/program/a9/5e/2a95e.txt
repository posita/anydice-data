O:0
if D >= 2 {O:O+1}
if D >= 6 {O:O+1}
if D = 10 {O:O+1}
result:O
}

loop N over {0,1,4,6,8}{
if N = 0 & X = 0{}else{output Xd10 + dN named "Roll [X]d10 + d[N]"}
P: X
if N > 1 {P:P+1}
if N = 0 & X = 0{}else{output (Xd[lethal d10] + [lethal dN])+X named "Core [X]d10 + d[N]"}
if N = 0 & X = 0{}else{output (Xd[lethal d10] + [lethal dN])+P+X named "Potent Core [X]d10 + d[N] +[P]"}
}}

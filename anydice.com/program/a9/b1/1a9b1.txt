function: ROLL:n vs AC:n {
 if ROLL = 20 {
  result: (2d10)+4+2*HEX
 }
 if ROLL + ATTACK + PROF >= AC {
  result: (1d10)+4+HEX
 }
 result: 0
}

AC: d{14..20}
PROF: 2
ATTACK: 4

HEX: 1d6
output [d20 vs AC]+[d20 vs AC] named "Hex"

HEX: 0
output [[highest 1 of 2d20] vs AC]+[[highest 1 of 2d20] vs AC] named "No Hex, advantage"

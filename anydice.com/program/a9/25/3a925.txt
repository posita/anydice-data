function: attack A:n B:n roll D:n BONUS:n vs AC:n {
  if D = 20 {result: A+A+B}
  if D = 1 {result: 0}
  if D + BONUS >= AC {result: A+B}
  result: 0
}

AC: 15
D: d20

A: [attack d8 3 roll D 5 vs AC]
B: [attack d6 3 roll D 5 vs AC]
C: [attack d6 0 roll D 5 vs AC]
D: [attack 2d6 3 roll D 5 vs AC]

output A
output D
output B+C

function: reroll R:n under N:n {
   if R <= N { result: 3d{0, 0, 0, 1, 1, 2} } else {result: R}
}
function: reroll R:n charged under N:n {
   if R <= N { result: 3d{0, 0, 0, 1, 1, 2} + 1d{0, 1, 1, 1, 1, 2} } else {result: R}
}

output 3d{0, 0, 0, 1, 1, 2} named "Normal"
output [reroll 3d{0, 0, 0, 1, 1, 2} under 1] named "Reroll 1s"
output [reroll 3d{0, 0, 0, 1, 1, 2} + 1d{0, 1, 1, 1, 1, 2} charged under 1] named "Reroll Charged 1s"
output [reroll 3d{0, 0, 0, 1, 1, 2} under 2] named "Reroll 2s"
output [reroll 3d{0, 0, 0, 1, 1, 2} + 1d{0, 1, 1, 1, 1, 2} charged under 2] named "Reroll Charged 2s"
output [reroll 3d{0, 0, 0, 1, 1, 2} under 3] named "Reroll 3s"
output [reroll 3d{0, 0, 0, 1, 1, 2} + 1d{0, 1, 1, 1, 1, 2} charged under 3] named "Reroll Charged 3s"

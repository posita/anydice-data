function: success A:n {
 result: A >= 4
}

function: success in A:n and B:n {
 loop I over {1..5} {
   C: I * 2 + 2
  result: [success A] + [success B] + [success C]
 }
}

function: success in A:n and B:n and C:n {
 result: [success A] + [success B] + [success C]
}

loop II over {1..5}{
  I: II * 2 + 2
  loop IJ over {1..5}{
    J: IJ * 2 + 2
    output [success in dI and dJ] named "[I] [J]"
    loop IK over {1..5}{
      K: IK * 2 + 2
        \output [success in dI and dJ and dK] named "[I] [J] [K]"\
    }
  }
}

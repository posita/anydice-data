function: binarygenerator BITS:n
{
 RESULT: 0
 loop N over {0..BITS-1}
 {
  RESULT: RESULT + (2^N)*d{0,1}
 }
 result: RESULT
}

NUMBEROFBITS: 3
COMBINATIONS: 2^NUMBEROFBITS

output dCOMBINATIONS-1 named "d[COMBINATIONS]-1"
output [binarygenerator NUMBEROFBITS] named "d[COMBINATIONS] with Binary Generator [NUMBEROFBITS] Bits/Coins"

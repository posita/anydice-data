A: {0,1,1}
B: {0,0,1}

function: roll A:n max B:n under Y:n skill Z:n {
  if A=Y {
    result: 20
  } else {
    if A<Y { 
      result: Z+A
    } else {
      if A=B {
        result: -20
      } else {
        result: Z
      }
    }
  }
}

function: adv A:n and B:n max C:n under Y:n skill Z:n {
  if A<Y {
    P: A
  } else {
    if A=Y {
      P: 20
    } else {
      if A=C {
        P: -20
      } else {
        P: 0
      }
    }
  }
  if B<Y {
    Q: B
  } else {
    if B=Y {
      Q: 20
    } else {
      if B=C {
        Q: -20
      } else {
        Q: 0
      }
    }
  }
  if P=-20 {
    result: -20
  } else {
    if Q=-20 {
      result: -20
    } else {
      if P<=Q {
        result: Z+Q
      } else {
        result: Z+P
      }
    }
  }
}

output [roll d20 max 20 under 5 skill 0] named "Basic Roll under 5"
output [adv d20 and d20 max 20 under 5 skill 0] named "Advantage Roll under 5"
output [roll d20 max 20 under 10 skill 0] named "Basic Roll under 10"
output [adv d20 and d20 max 20 under 10 skill 0] named "Advantage Roll under 10"

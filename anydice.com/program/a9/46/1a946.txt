function: is ROLLS:s glitch  {
 BOOLEAN_ONES: ROLLS = 1
 result: BOOLEAN_ONES >= (#ROLLS + 1) / 2
}

function: is ROLLS:s critical glitch  {
 SUCESSES: ROLLS >= 5 
 if [is ROLLS glitch] & (SUCESSES = 0) {
  result: 1
 }
 result: 0
}

loop DICE over {1..10} {
 \ output [is DICE d6 glitch] named "[DICE]d6 p(glitch)" \
 output [is DICE d6 critical glitch] named "[DICE]d6 p(critical glitch)"
}

function: lethality L:n with DD:n and D:n {
     if DD = 10 {
          if D = 10 { P: 100 }
          else { P: D }
     }
     else if D = 10 { P: DD * 10 }
     else { P: DD * 10 + D }

     if P <= L { result: 0 }
     result: DD + D
}

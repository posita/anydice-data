STARZERO: d{0,0,0,0,0,3}
STARONE: d{0,0,0,0,2,3}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}



output STARZERO

output STARONE

output [reroll {0} on STARONE to STARONE]

output STARZERO + STARZERO

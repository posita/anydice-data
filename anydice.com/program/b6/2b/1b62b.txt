function: absnz A:n{
 if A>0 { result: A }
 if A=0 { result: 1 }
 if A>8 | A<-8 { result:8+[explodea [lowest 1 of 2d12 ]]}
 result: -A
}

function: explodea N:n {
 if N > 4 { result: 4 + [explodeb [lowest 1 of 3d12 ]] }
 result: N
}

function: explodeb N:n {
 if N > 4 { result: 4 + [explodec [lowest 1 of 4d12 ]] }
 result: N
}

function: explodec N:n {
 if N > 4 { result: 4 + [exploded [lowest 1 of 5d12 ]] }
 result: N
}

function: exploded N:n {
 if N > 4 { result: 4 + [lowest 1 of 2d4 ] }
 result: N
}

set "maximum function depth" to 4

output [absnz (2d8-2d8)]

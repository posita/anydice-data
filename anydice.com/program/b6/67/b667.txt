L: 1d12
M: 1d12
H: 1d12
TMP: 1d12


function: littledamage DA:n and DB:n and DC:n at NUM:n and TEMP:n re RER:n {
if RER >= DA {
 if TEMP > DA {DA: TEMP}
}
if RER >= DB {
 if TEMP > DB {DA: TEMP}
}
if RER >= DC {
 if TEMP > DC {DA: TEMP}
}

if DA >= DB {
 if DA >= DC {
  MAX: DA
  if DB >= DC {
   MID: DB
   LOW: DC
  }
  else {
   MID: DC
   LOW: DB
  }
 }
}
else {
  if DB >= DC {
   MAX: DB
   if DC >= DA {
    MID: DC
    LOW: DA
   }
   else {
    MID: DA
    LOW: DC
   }
  }
  else {
   MAX: DC
   MID: DB
   LOW: DA
  }
 }
if NUM = 0 {
 result: LOW + LOW}
else {
 if NUM = 1 {
 result: LOW + MID
 }
 else {
  if NUM = 2 {
  result: LOW + MAX }
 }
}

 
}

output [littledamage L and M and H at 0 and TMP re 8] named "little damage downgraded reroll 8"

function: roll A:n and B:n {
    if A = 1 | B = 1 { result: A + B }
    else if (A >= 10) & (B >= 10) { result: A + B + 4 }
    else if (A >= 10) | (B >= 10) { result: A + B + 2 }
    else { result: A + B }
}

function: disadvantage A:n and B:n and C:n {
    if (A <= C) & (B <= C)      { result: [roll A and B] }
    else if (A <= B) & (C <= B) { result: [roll A and C] }
    else                        { result: [roll B and C] }
}
function: advantage A:n and B:n and C:n {
    if (A >= C) & (B >= C)      { result: [roll A and B] }
    else if (A >= B) & (C >= B) { result: [roll A and C] }
    else                        { result: [roll B and C] }
}

output [roll d10 and d10]                 named "Standard"
output [roll d10 and d12]                 named "Profficiency"
output [roll d12 and d12]                 named "Expertise"

output [advantage d10 and d10 and d10]    named "Standard Advantage 1"
output [advantage d10 and d10 and d12]    named "Profficency Advantage 1"
output [advantage d12 and d12 and d12]    named "Expertise Advantage 1"

output [disadvantage d10 and d10 and d10] named "Standard Disadvantage 1"
output [disadvantage d10 and d10 and d12] named "Profficency Disadvantage 1"
output [disadvantage d12 and d12 and d12] named "Expertise Disadvantage 1"

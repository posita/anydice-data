function: damage DA:n dice DS:n plus DB:n minimum MIN:n{
  if MIN = 20{
    result: (2*DA)dDS+DB
  }
  result: DAdDS+DB
}

function: hit AC:n with TH:n {
  if TH > AC {
    result: 1
  }
  result: 0
}

output [damage 1 dice 4 plus 4 minimum 1d{13-4..20}] named "Player Damage"
G: 7
O: [hit 13 with 1d20+5]
F: O
loop N over {2..G} {
  F: F+O^N
}
output F named "Player Hit Chance Over {MAX} Turns"

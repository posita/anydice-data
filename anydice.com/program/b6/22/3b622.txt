\ 0:fumble, 1:failure, 2: succes, 3:special, 4:critical \
function: brp D:s S:n{
 if D = 100 | D >= (100-(100-S)/20) {
  result: 0 \ Fumble \
 } else if D > S | D >= 96{
  result: 1 \ Failure \
 } else if D <= S/20+1 {
  result: 4 \ Critical \
 } else if D <= (S/5+1) {
  result: 3 \ Special \
 } else if D <=5 | D <= S {
  result: 2 \ Success \
 } else {
  result: -1
 }
}

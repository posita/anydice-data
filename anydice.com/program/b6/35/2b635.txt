function: suchelper A:n {
 if A >= 24 {
  result: 6
 }
 if A >= 20 {
  result: 5
 }
 if A >= 17 {
  result: 4
 }
 if A >= 14 {
  result: 3
 }
 if A >= 11 {
  result: 2
 }
 if A >= 9 {
  result: 1
 }
 result: 0
}
function: suc A:n {
 if A > 18{
  A: A + 2d6
 }
 else{if A > 12{
  A: A + (1d6 * ((A-12)/6))
 }}
 result: [suchelper A]
}

function: highest N:n of A:s B:s C:s {
    result: {1..N}@[sort {A, B, C}]
}
function: highest N:n of A:s B:s {
    result: {1..N}@[sort {A, B}]
}

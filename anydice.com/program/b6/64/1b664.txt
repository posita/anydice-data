\
 Blades in the Dark
 0: Things go bad.
 1: Success, but with undesired consequences.
 2: Clean success.
 3: Critical success!
\

function: blades ROLL:s with PENALTY:n {
 PEN: PENALTY + 1
 CRIT: PENALTY + 2
 if PEN@ROLL = 6 {
  result: 2 + (CRIT@ROLL = 6)
 }
 result: PEN@ROLL >= 4
}

loop P over {2..7} {
 I: P-1
 output [blades Id6 with 0] named "[I]d"
 output [blades Pd6 with 1] named "[I]d+pen"
}

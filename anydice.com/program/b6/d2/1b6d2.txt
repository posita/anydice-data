JUMPS: 0
PERILS: 0
WAAAGH: 3
WEIRDBOYZ: 11

function: jump LEFT:n RIGHT:n {
 if LEFT + RIGHT + WAAAGH >= 7 {
  result: 1
 } else {
  result: 0
 }
}

function: peril LEFT:n RIGHT:n {
 if LEFT = RIGHT = 1 {
  result: 1
 } else if LEFT + RIGHT + WAAAGH >= 12 {
  result: 1
 } else {
  result: 0
 }
}

loop N over {1..WEIRDBOYZ} {
 JUMPS: JUMPS + [jump 1d6 1d6]
 PERILS: PERILS + [peril 1d6 1d6]
}

output JUMPS named "Units Jumped"
output PERILS named "Perils of the Warp Suffered"

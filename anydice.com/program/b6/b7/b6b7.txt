function: test ROLL:n vs AC:n {
 if ROLL = 1 { result: 0 }
 else {
  result: 1
 }
}

function: roll B:n A:n{
 result: BdA
}

function: gwf B:n A:n{
 X:0
 loop N over {1..B} {
  X: X + [reroll 1dA A]
 }
 result: X
}

function: reroll A:n B:n{ 
 if A <= 2 { 
  A: 1dB 
 }  
 result: A 
}

\--HIT VARIABLES--\
MISS: 0
HIT: 1
CRITICALHIT: 2
CRITMISS: -1

\--ATK_TYPE VARIABLES--\
NORMAL: 0
ADV:    1
DIS:    2
YES:    1
NO:     0

function: attackroll ROLL:n vs AC:n {
 ATTACK: ATK_AND_DMG_MOD + PROFICIENCY_MOD
 if ROLL = 1 { result: CRITMISS }
 if ROLL >= CRITICAL_ON {
  result: CRITICALHIT }
 if ROLL + ATTACK >= AC { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:s BONUSATTACKRESULT:s {
 if GREAT_WEAPON_FIGHTING = 1 {
  HITDMG:         [gwf  DMG_DICE_NUM        DMG_DICE]       + XTRA_DMG_PER_HIT
  BONUSHITDMG:    [gwf  BONUS_DMG_DICE_NUM  BONUS_DMG_DICE] + XTRA_DMG_PER_HIT
  CRITDMG:        HITDMG      + HITDMG      + [gwf  XTRA_CRIT_DICE  DMG_DICE]
  BONUSCRITDMG:   BONUSHITDMG + BONUSHITDMG + [gwf  XTRA_CRIT_DICE  BONUS_DMG_DICE]
 }
 else {
  HITDMG:         [roll  DMG_DICE_NUM        DMG_DICE]       + XTRA_DMG_PER_HIT
  BONUSHITDMG:    [roll  BONUS_DMG_DICE_NUM  BONUS_DMG_DICE] + XTRA_DMG_PER_HIT
  CRITDMG:        HITDMG      + HITDMG      + [roll  XTRA_CRIT_DICE  DMG_DICE]
  BONUSCRITDMG:   BONUSHITDMG + BONUSHITDMG + [roll  XTRA_CRIT_DICE  BONUS_DMG_DICE]
 }
 D: 0
 loop A over ATTACKRESULT {
  if A = CRITICALHIT {
   D: D + CRITDMG + (MOD_ON_ATTACK_DMG * ATK_AND_DMG_MOD) + MISC_DMG_MOD
  }
  else if A = HIT {
   D: D + HITDMG + (MOD_ON_ATTACK_DMG * ATK_AND_DMG_MOD) + MISC_DMG_MOD
  }
  else {
   D: D + 0
  }
 }
 loop B over BONUSATTACKRESULT {
  if B = CRITICALHIT {
   D: D + BONUSCRITDMG + (MOD_ON_ATTACK_DMG * ATK_AND_DMG_MOD) + MISC_DMG_MOD
  }
  else if B = HIT {
   D: D + BONUSHITDMG + (MOD_ON_ATTACK_DMG * ATK_AND_DMG_MOD) + MISC_DMG_MOD
  }
  else {
   D: D + 0
  }
 }
 result: D + [oncextradmg for {ATTACKRESULT,BONUSATTACKRESULT}]
}

function: oncextradmg for RESULTS:s {
 if [RESULTS contains CRITICALHIT] {
   result: ONCEXTRA_CRITABLE + ONCEXTRA_CRITABLE + ONCEXTRA_NOCRIT
 }
 if [RESULTS contains HIT] {
   result: ONCEXTRA_CRITABLE + ONCEXTRA_NOCRIT
 }
 result: 0
}

function: A:n B:n attacks {
 if ATK_TYPE = ADV {
  result: [damage for Ad[attackroll [highest 1 of 2d20] vs AC] Bd[attackroll [lowest 1 of 2d20] vs AC]]
 }
 else if ATK_TYPE = DIS {
  result: [damage for Ad[attackroll [lowest 1 of 2d20] vs AC] Bd[attackroll [lowest 1 of 2d20] vs AC]]
 }
 else {
  result: [damage for Ad[attackroll 1d20 vs AC] Bd[attackroll 1d20 vs AC]]
 }
}

SKILLMOD: 7
WAGESP: 100
LIFESTYLE: 20
WEEK: 7

function: run a business A:n {
 if A <= 4{
  result: -2
 }
 else if A <= 9{
  result: -1
 }
 else if A <= 14{
  result: 0
 }
 else if A <= 19{
  result: 1
 }
 else if A <= 24{
  result: 2
 }
 else if A <= 29{
  result: 3
 }
 else{
  result: 4
 }
}

function: work A:n {
 if A <= 9{
  result: 2*WEEK
 }
 else if A <= 14{
  result: 10*WEEK
 }
 else if A <= 20{
  result: 20*WEEK
 }
 else{
  result: 20*WEEK+250
 }
}


output ([run a business 1d20+SKILLMOD])*WAGESP named "RAB-Normal"
output ([run a business [highest 1 of 2d20]+SKILLMOD])*WAGESP named "RAB-Advantaged"
output ([run a business [lowest 1 of 2d20]+SKILLMOD])*WAGESP named "RAB-Disadvantaged"

output [work 1d20+SKILLMOD]-(LIFESTYLE*WEEK) named "WORK-Normal"
output [work [highest 1 of 2d20]+SKILLMOD]-(LIFESTYLE*WEEK) named "WORK-Advantage"
output [work [lowest 1 of 2d20]+SKILLMOD]-(LIFESTYLE*WEEK) named "WORK-Disadvantage"

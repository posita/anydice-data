loop N over {1, 2, 3, 4, 5} {
  output [highest 1 of Nd8] > 5 named "success on [N]d8"
  output ([highest 1 of Nd8] > 5) & ([lowest 1 of (N-1)d8] = 1) named "Crit on [N]d8"
  loop S over {1, 2, 3, 4, 5, 6, 7, 8} {
    output ([highest 1 of Nd8] > 5) & ([lowest 1 of (N-1)d8] <= S) named "Strong on [N]d8, Skill [S]"
  }
}

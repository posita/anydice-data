function: single dc arena B {
  DC: 2d10+5
  SUCCESSES: 0
  loop CHECK over {1..3} {
    SUCCESSES: SUCCESSES + (DC <= 1d20 + B)
  }
  result: SUCCESSES
}

function: multi dc arena B {
  SUCCESSES: 0
  loop CHECK over {1..3} {
    SUCCESSES: SUCCESSES + (2d10+5 <= 1d20 + B)
  }
  result: SUCCESSES
}

loop BONUS over {0..11} {
  output [single dc arena BONUS] named "1 DC +[BONUS]"
  output [multi dc arena BONUS] named "3 DC +[BONUS]"
}

POS: {4,6,8,10,12}

function: az X Y Z {
 result: [highest of [highest of 1@X and 1@Y] and [highest of 1@Z and 0]]
}

function: segmaior A:n B:n C:n {
 result: 2 @ [sort {A, B, C}]
}

loop N over POS {
 loop O over POS {
  loop P over POS {
   output [segmaior 1dN 1dO 1dP] named "[N],[O],[P]"
  }
 }
}

function: suc A:n {
 if A >= 24 {
  result: 6
 }
 if A >= 21 {
  result: 5
 }
 if A >= 18 {
  result: 4
 }
 if A >= 14 {
  result: 3
 }
 if A >= 11 {
  result: 2
 }
 if A >= 9 {
  result: 1
 }
 result: 0
}

function: critcheck ROLL:s CRIT:n EPICCRIT:n{
 A: 1@ROLL
 B: 2@ROLL
 C: 3@ROLL
 if A+B+C = EPICCRIT{
  result: ROLL + 2d(CRIT/2)
 }
 if A+B = CRIT{
  result: ROLL + 1d(CRIT/2)
 }
 result: ROLL
}

function: cc Q:s QM:n R:s RM:n S:s SM:n T:s TM:n{
 QC:0
 RC:0
 SC:0
 TC:0
 if Q = [maximum of Q]{
  QC: 1
 }
 if R = RM{
  RC: 1
 }
 if S = SM{
  SC: 1
 }
 if T = TM{
  TC: 1
 }
 if QC+RC+SC+TC = 3{
  result: 2
 }
 if QC+RC+SC+TC = 2{
  result: 1
 }
 result: 0
}

output [suc 3d6] named "Normal"

output [suc[cc 1d6 6 18]] named "NormalCrit"


output [suc[critcheck 3d6 12 18]] named "NormalCrit"

output [suc [lowest 3 of 4d6]] named "Desventaja"

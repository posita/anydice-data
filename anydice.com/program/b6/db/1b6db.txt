function: skillcheck MOD:n{
   result: 1d20 + MOD
}

function: crime MODA:n MODB:n MODC:n DC:n{
   SUCCESSES: 0
   if [skillcheck MODA] > DC{
      SUCCESSES: SUCCESSES + 1
   }
   if [skillcheck MODB] > DC{
      SUCCESSES: SUCCESSES + 1
   }
   if [skillcheck MODC] > DC{
      SUCCESSES: SUCCESSES + 1
   }
result: SUCCESSES
}

output [crime 7 7 10 15]

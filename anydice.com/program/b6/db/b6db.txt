set "maximum function depth" to 100

function: e N:n {
X: 1
loop I over{1..N} {
X: X*10
}
result:X
}
function: dtos D:s {
X:0
N:#D
  loop I over {1..N}{
    X: X + [e I-1]*(I@D)
  }
result: X
}

function: contest A:s yB:s {
I:6
loop N over {1..6} {
    if [count I in A] > [count I in B] {
      result: -1
    }
    if [count I in A] < [count I in B] {
      result: 1    
    }
    I: I - 1
}
result:0
}

output [contest 1d6y4d6]
output [contest 2d6y4d6]
output [contest 3d6y4d6]

function: suc A:n {
 if A >= 24 {
  result: 6
 }
 if A >= 20 {
  result: 5
 }
 if A >= 17 {
  result: 4
 }
 if A >= 14 {
  result: 3
 }
 if A >= 11 {
  result: 2
 }
 if A >= 9 {
  result: 1
 }
 result: 0
}

function: highest N:n of A:s B:s C:s {
    result: {1..N}@[sort {A, B, C}]
}
function: highest N:n of A:s B:s {
    result: {1..N}@[sort {A, B}]
}

function: crit{
 D:3d6
 result: [crithelp 1@D 2@D 3@D]
}
function: critadv{
 D:4d6
 result: [crithelp 1@D 2@D 3@D]
}
function: critdis{
 D:4d6
 result: [crithelp 2@D 3@D 4@D]
}
function: critdeter{
 D:3d6
 result: [crithelp [rerollsix 1@D under 4] [rerollsix 2@D under 4] [rerollsix 3@D under 4]]
}

function: crithelp X:n Y:n Z:n{
 if X >= 6 & Y >= 6 & Z >= 6{
  result: 3d6} 
 if X >= 6 & Y >= 6{
  result: 2d6}
 result: d0
}
function: rerollsix D:n under X:n{
   if D < X { result: 1d6 } else {result: D}
}
function: rerollten D:n under X:n{
   if D < X { result: 1d10 } else {result: D}
}
function: pasion {
 result: [pasionhelper 1d6] + [pasionhelper 1d6] + [pasionhelper 1d6] - 3d6
}
function: pasionhelper D:n{
  if D >= 6 {
   result: 6 + [pasionhelper 1d6]}
  result: D
}

output [suc 3d6+[crit]] named "Normal"

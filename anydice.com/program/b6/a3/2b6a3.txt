\ Goalsystem Revised \

function: goal roll ROLL:s target TARGET:n with REROLLS:n rerolls {
  SUPERSUCCESSES: [count {6} in ROLL]
  SUCCESSES: [count {4,5} in ROLL]
  BOTCHES: [count {1} in ROLL]

  TOTALSUCCESSES: SUCCESSES+2*SUPERSUCCESSES-BOTCHES

  result: TOTALSUCCESSES
}

loop NUMDICE over {1..5} {
  loop NUMREROLLS over {0..4} {
    output [goal roll NUMDICEd6 target 4 with NUMREROLLS rerolls] named "[NUMDICE]D[[NUMREROLLS]] Goal Roll"
  }
}

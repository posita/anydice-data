function: disadvantage DV:d{
    result: [lowest 1 of 2*DV]
}
function: advantage DV:d{
    result: [highest of DV and DV]
}
function: DICECK:n dc DC:n success ONSUCC failure ONFAIL{
    if DICECK>DC{
        result: ONSUCC
    }
    else{
        result: ONFAIL
    }
}
function: DICECK:n mod MOD:n dc DC:n success ONSUCC failure ONFAIL{
    if DICECK+MOD>DC{
        result: ONSUCC
    }
    else{
        result: ONFAIL
    }
}
function: bardstrat DICE:n bd BDICE mod MOD threshold T:n dc DC success SUCC failure FAIL{
    if DICE<T{
        result: [DICE mod MOD+BDICE dc DC success SUCC failure FAIL]
    }
    else{
        result: [DICE mod MOD dc DC success SUCC+BDICE failure FAIL]
    }
}
function: barddiff DICE:n bd BDICE mod MOD threshold T:n dc DC success SUCC failure FAIL{
  result: [bardstrat DICE bd BDICE mod MOD threshold T dc DC success SUCC failure FAIL] - [bardstrat DICE bd 0 mod MOD threshold T dc DC success SUCC failure FAIL]
}

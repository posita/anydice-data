\ change these parameters \
L: 5 \ Skill levels \
D: 10 \ Die faces \
C: 3 \ Dice cluster size (the number of dice counted) 1-3 recommended \

\ ----------- \

output CdD named "Reference [C]d[D]"

T: L+C-1 \ Number of dice to roll \
loop N over {1..L} {
  E: N + C - 1 \ end number of dice sequence \
  Q: L - N + 1
  output {N..E}@TdD named "Skill [Q]"
}

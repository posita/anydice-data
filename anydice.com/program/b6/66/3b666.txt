function: stress check C:n{
 if C >= 20 {result : 0}
else if C = 19 {result : 0}
else if C = 18 {result : 1}
else if C = 17 {result : 1}
else if C = 16 {result : 2}
else if C = 15 {result : 2}
else if C = 14 {result : 3}
else if C = 13 {result : 3}
else if C = 12 {result : 4}
else if C = 11 {result : 4}
else if C = 10 {result : 5}
else if C = 9 {result : 5}
else if C = 8 {result : 6}
else if C = 7 {result : 6}
else if C = 6 {result : 7}
else if C = 5 {result : 7}
else if C = 4 {result : 8}
else if C = 3 {result : 8}
else if C = 2 {result : 9}
else if C = 1 {result : 9}
else if C <= 0 {result : -100}
}


function: rout correct T:n
{
if T = -100 {result: 1}
result: 0
}


function: stress roll M:n
{
T: {0}
T:[stress check (1d20+M)]
result: T
}

function: disadvantaged stress roll M:n  and D
{
T: {}
T:[stress check ([lowest 1 of Dd20]+M)]
result: T 
}

function: advantaged stress roll M:n  and A
{
T: {0}
T:[stress check ([highest 1 of Ad20]+M)]
result: T 
}



function: stress rolls with morale mod M:n rolling N times
{
C: {1}
T: {0}
U: {0}

loop C over {1..N}
{

M: [loop internal check with M and T]

}

}

function: loop internal check with M:n and T:n
{

if [rout correct T]  = 1 {result: -100}
if [rout correct M]  = 1 {result: -100}
result: M-T
}

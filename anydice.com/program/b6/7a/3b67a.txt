function: pbta X:n {
  if X>=10 {result: 2}
  if X>=7 {result: 1}
  result: 0
}
loop N over {-3..3} {
  output [pbta 2d6+N+1] named "1 Forward to [N]"
}
loop N over {-2..2} {
  output [pbta {1,2}@3d6+N] named "Advantage to [N]"
}
loop N over {-2..2} {
  output [pbta {2,3}@3d6+N] named "Disadvantage to [N]"
}

\
 UnderTheMaar
 0: Things go bad.
 1: Success, but with undesired consequences.
 2: Clean success.
 3: Critical success!
\

function: maar ROLL:s {
 if 1@ROLL = 8 {
  result: 2 + (2@ROLL = 8)
 }
 result: 1@ROLL >= 5
}

output [maar 2@2d8] named "0d"
loop P over {1..7} {
 output [maar Pd8] named "[P]d"
}

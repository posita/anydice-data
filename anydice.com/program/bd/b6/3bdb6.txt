function: roll A:n and B:n and find the highest{
    if A >= B{result: A}
    result: B
}
function: roll A:n and B:n and find the lowest{
    if A <= B{result: A}
    result: B
}
function: roll A:n and B:n and find doubles then add C:n and subtract D:n{
    if A = B{result: -8}
    result: (A + B + C) - D
}
function: roll A:n and B:n and output higher die{
    if A = B{result: 0}
    if A > B{result: 1}
    if A < B{result: -1}
}
output [roll 3d6 and 3d6 and find the highest]

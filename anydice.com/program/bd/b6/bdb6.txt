function: ace D:n {
  if D = 1 {
    result: -1
  } else if D >= 4 {
    result: 1
  } else {
    result: 0
  }
}

function: limit N:n {
  if N < -1 {
    result: -1
  } else if N > 3 {
    result: 3
  } else {
    result: N
  }
}

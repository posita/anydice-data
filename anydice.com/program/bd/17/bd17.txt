function: C:n roteskill L:n {
  X: 0
  loop N over {1 .. C}{
    X: X+[L rote]
  }
  result: X
}

function: L:n rote {
  X: 0
  loop N over {1 .. L}{
    X: X+[roll d10]
  }
  result: X
}

function: roll R:n {
  if R=1 {X: [reroll d10]}
  if R=2 {X: [reroll d10]}
  if R=3 {X: [reroll d10]}
  if R=4 {X: [reroll d10]}
  if R=5 {X: [reroll d10]}
  if R=6 {X: [reroll d10]}
  if R=7 {X: [reroll d10]}
  if R=8 {X: 1}
  if R=9 {X: 1}
  if R=10 {X: 1}
  if R>9 {X: X + [roll d10]}
  result: X
} 

function: reroll R:n {
  if R=1 {X: 0}
  if R=2 {X: 0}
  if R=3 {X: 0}
  if R=4 {X: 0}
  if R=5 {X: 0}
  if R=6 {X: 0}
  if R=7 {X: 0}
  if R=8 {X: 1}
  if R=9 {X: 1}
  if R=10 {X: 1}
  if R>9 {X: X + [roll d10]}
  result: X
}

set "maximum function depth" to 6

output [6 roteskill 9]

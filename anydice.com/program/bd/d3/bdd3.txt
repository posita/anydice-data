function: hit R:d on N:n {
  result: R >= N
}

function: wound R:d on N:n {
  result: R >= N
}

function: save R:d on N:n {
  result: R < N
}

function: towound str S:n vs toughness T:n {
  if S <= T/2 { result: 6 }
  if S >= 2*T { result: 2 }
  if S >    T { result: 3 }
  if S =    T { result: 4 }
  if S <    T { result: 5 }
}

function: attack with A:d attacks at bs BS:n and str S:n vs toughness T:n and ap AP:n vs save SV:n and damage D:d {
  result: Ad([hit d6 on BS] * [wound d6 on [towound str S vs toughness T]] * [save d6 on (SV - AP)] * D)
}

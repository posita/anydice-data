function: reroll DIE:n on TRIGGER:n as REPLACEMENT:n {
  if DIE = TRIGGER {
    result: [highest of DIE and REPLACEMENT]
  }
  result: DIE
}


function: soft explode DIE:d as REPLACEMENT:n {
  result: [reroll DIE on [maximum of DIE] as REPLACEMENT]
}

DTWELVE: [soft explode d12 as d20]
DTEN:    [soft explode d10 as DTWELVE]
DEIGHT:  [soft explode d8  as DTEN]
DSIX:    [soft explode d6  as DEIGHT]
DFOUR:   [soft explode d4  as DSIX]

output DFOUR   named "d4"
output DSIX    named "d6"
output DEIGHT  named "d8"
output DTEN    named "d10"
output DTWELVE named "d12"

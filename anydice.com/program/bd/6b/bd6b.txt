\ based on http://anydice.com/articles/dnd4-attacks/ \

\ Level 5 Barbarian w/ Great Ax \
ATTACK: 7
HITDMG: d12 + 4
CRITDMG: 2d12 + 4
MISSDMG:0


MISS: 0
HIT: 1
CRIT: 2

\ ATTACK FUNCTION \
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

\ DAMAGE FUNCTION \
function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRIT {
  result: CRITDMG
 }
 if ATTACKRESULT = HIT {
  result: HITDMG
 }
 result: MISSDMG
}

loop AC over {16..17}{
output 2d[damage for [attack d20 vs AC]] named "Double ATK w/ Great Ax @ AC [AC]"
}

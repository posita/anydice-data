function: rank BEST:n SECOND:n {
 if SECOND = 6 { result: 3 }
 if BEST = 6 { result: 2 }
 if BEST >= 4 { result: 1 }
 result: 0
}

function: pool A {
 if A <= 0 {
  ROLL: 2d6
  result: [rank 2@ROLL 0]
 }
 else {
  ROLL: Ad6
  result: [rank 1@ROLL 2@ROLL]
 }
}

loop N over {0..10}{
 output [pool N] named "For [N] dice"
}

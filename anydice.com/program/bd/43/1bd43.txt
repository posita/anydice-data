CRITICALHIT: 0
HIT: 1
MISS: 2

function: attack ATTACK_ROLL:n confirm CRIT_CONFIRM_ROLL:n vs DEFENSE:n {
 if ATTACK_ROLL = 1 {result: MISS }
 if ATTACK_ROLL = 20 {
  if CRIT_CONFIRM_ROLL + ATTACK_BONUS >= DEFENSE {result: CRITICALHIT}
  result: HIT
 }
 if ATTACK_ROLL + ATTACK_BONUS >= DEFENSE {result: HIT}
 result: MISS
}

ATTACK_BONUS: 5
loop AC over {10..25} {
 output [attack d20 confirm d20 vs AC] named "[AC]"
}

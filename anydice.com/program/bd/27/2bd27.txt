function: negifodd X:n {
  result: 1 - 2 * ((X / 2 * 2) != X)
}

function: genesix IDX:s in X:s and Y:s and Z:s {
  I: 1@IDX
  A: I@[sort {X, Y, Z}]
  J: 2@IDX
  B: J@[sort {X, Y, Z}]
  if (A = 1) & (B = 1) {
    result: -100
  }
  if (A = 6) & (B = 6) {
    result: 100
  }
  if A = B {
    result: 10 * [negifodd A]
  }
  if [negifodd A] = [negifodd B] {
    result: 1 * [negifodd A]
  }
  result: 0
}

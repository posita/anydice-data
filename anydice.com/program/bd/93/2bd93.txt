\  VERGLEICHS-FUNKTION VON JASPER  \

function: A:s vs B:s {
 W: 0
 loop P over {1..[highest of #A and #B]} {
  W: W + (P@A > P@B)
 }
 result: W
}


\  MEINE VERGLEICHS-FUNKTION  \

function: A:s vsa B:s {
 W: 0
 loop P over {1..[lowest of #A and #B]} {
  W: W + (P@A > P@B)
 }
if #A >= #B {
 result: W + #A - #B
 } else {
 result: W
 }
}

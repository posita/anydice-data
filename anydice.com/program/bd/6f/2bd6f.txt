function: roll ROLL:s vs TARGET:n with optional REROLL:d reroll {
  SUM: ROLL + 0
  if SUM > TARGET {result: 0} 

else if SUM = TARGET {result: 1} 

else {REST: {2..#ROLL}@ROLL
    result: REST + REROLL = TARGET
  }
}

loop TARGET over {4..24} {
  output 2d6 = TARGET named "2d6 vs [TARGET]"
  output [roll 2d6 vs TARGET with optional d6 reroll] named "2d6 vs [TARGET] with optional reroll"
}

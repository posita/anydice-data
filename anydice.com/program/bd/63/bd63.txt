
function: DADO:n vs AC:n {
	if DADO >= MINACCIA & DADO + ATTACK >= AC {
		result: [critico d20]
	}
	if DADO + ATTACK >= AC | DADO = 20 {
		result: DANNO + FURTIVO + CRAVEN
	}
	else {result: 0}
}

function: critico DADO:n {
	if DADO + ATTACK >= AC | DADO = 20 {
		CD: CRAVEN * MOLT
		CDD: MOLT d DANNO
		result: FURTIVO + CD + CDD
	}
	result: DANNO + FURTIVO + CRAVEN
}

MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n confirm ROLL_TWO:n vs DEFENSE:n {
 if ROLL = 1 {result: MISS }
 if ROLL = 20 {
  if ROLL_TWO + ATTACK >= DEFENSE {result: CRITICALHIT}
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE {result: HIT}
 result: MISS
}

ATTACK: 5
loop AC over {10..25} {
 output [attack d20 confirm d20 vs AC] named "[AC]"
}

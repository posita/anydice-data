function: average damage of ATK:s having mod ATK_MOD:n with WPN:d having DEADLY:d and mod DMG_MOD:n against ac of AC:n {
  TOTAL_ATK: ATK + ATK_MOD
  TOTAL_DMG: WPN + DMG_MOD
  CRIT_DMG: (TOTAL_DMG * 2) + DEADLY
  
  if (TOTAL_ATK > (AC + 10) ) {
    result: CRIT_DMG
  }

  if (TOTAL_ATK > AC) { 
    if (ATK = 20) {
      result: CRIT_DMG
    }
    result: TOTAL_DMG
  }

  result: 0
}

function: adjusted damage of ATK:s having mod ATK_MOD:n with WPN:d having DEADLY:d and mod DMG_MOD:n against ac of AC:n {
  TOTAL_ATK: ATK + ATK_MOD
  TOTAL_DMG: WPN + DMG_MOD
  CRIT_DMG: [maximum of(WPN)] + (2 * DMG_MOD) + DEADLY
  
  if (TOTAL_ATK > (AC + 10) ) {
    result: CRIT_DMG
  }

  if (TOTAL_ATK > AC) { 
    if (ATK = 20) {
      result: CRIT_DMG
    }
    result: TOTAL_DMG
  }

  result: 0
}

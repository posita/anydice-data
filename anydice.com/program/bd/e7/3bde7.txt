function: rhpr A:n B:n M:n {
if A<(M/2) {result: B}
result: A
}
function: rolled hp A reroll {
result: [rhpr A A [maximum of A]]
}
function: rhpa A:n M:n {
if A<(M/2) {result: M/2}
result: A
}
function: rolled hp A average{
result: [rhpa A [maximum of A]]
}

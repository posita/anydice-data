function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE + FURTIVO }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE
 }
 result: DAMAGE
}

AC: d{14..20}
DES: 5
ATTACK: 2 + DES
CRAVEN: 6

DAMAGE: 1d8
FURTIVO: 4d6 + CRAVEN
THREAT: 18
CRITICAL: 2
output [d20 vs AC] named "Stocco"

function: D:n attack DC:n with BONUS:n{
 if D = 20 {result: 2}
 if D = 0 {result: 0}
 if D + BONUS < DC {result: 0}
 result : 1
}

function: roll DMG:d if HIT:n{
 if HIT = 2 {result: 2dDMG}
 if HIT = 1 {result: DMG}
 result: 0
}

EBH: d10 + d6 + 4

EBHC: d10 + d6 + 5 + d8

H: [d20 attack 15 with 8]
output 2d[roll EBH if 1@2dH] named "Current level"
H: [d20 attack 16 with 10]
output 2d[roll EBHC if 1@2dH] named "Level 9"
H: [d20 attack 17 with 10]
output 3d[roll EBHC if 1@2dH] named "Level 11"

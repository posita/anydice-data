function: evaluate ROLL:s
{
  SUCCESSES: ROLL >=4
   if [count  6 in ROLL] { result:SUCCESSES+1 }
   if [count  7 in ROLL] { result:SUCCESSES+1 }
   if [count  8 in ROLL] { result:SUCCESSES+2 }
   if [count  9 in ROLL] { result:SUCCESSES+2 }
   if [count 10 in ROLL] { result:SUCCESSES+3 }
   if [count 11 in ROLL] { result:SUCCESSES+3 }
   if [count 12 in ROLL] { result:SUCCESSES+4 }
   result:SUCCESSES
}
loop DICE over {1..5}
{

 output [evaluate DICE  d8] named "[DICE]d8"
 output [evaluate DICE d10] named "[DICE]d10"

}

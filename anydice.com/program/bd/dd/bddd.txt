set "explode depth" to 3 \adv explode will explode once more than this limit\

function: advexplode A:n B:n{  \Use the form [advexplode [highest/lowest X of (dice)] (die total e.g. 1d4 = 4)]\
  TOTAL: A
  if A = B {TOTAL: B+[explode 1dB]}
  result: TOTAL
}

function: limit A:n from B:n to C:n{
  if A > C {result:C}
  if A < B {result:B}
  result: A
}

LOWERBOUND: 9
UPPERBOUND: 25

loop N over {4,6}{
  AGILITY: 1dN
  output [limit [explode 1d20] + [advexplode [highest 1 of 5dAGILITY] N] from LOWERBOUND to UPPERBOUND] named "Invis Hide 1d[N]" 
}

loop N over {1..6}{output LOWERBOUND named ""}

loop N over {4,6,8,10}{
  PERCEPTION: 1dN
  output [limit [explode 1d20] + [explode 1dN] from LOWERBOUND to UPPERBOUND] named "Perception 1d[N]"
}

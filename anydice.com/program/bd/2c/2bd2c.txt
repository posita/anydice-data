\ Functions \
function: calculate modifier STAT {
    if STAT >= 10 {
        result: (STAT - 10) / 2
    }
    result: (STAT - 11) / 2
}

function: cantrip damage modifier CLERIC_LVL WISDOM_MODIFIER {
    if CLERIC_LVL >= 8 {
        result: WISDOM_MODIFIER
    }
    result: 0
}

function: cantrip damage LVL DAMAGE_DICE {
    if (LVL < 5) {
        result: 1dDAMAGE_DICE
    }
    if (LVL < 11) {
        result: 2dDAMAGE_DICE
    }
    if (LVL < 17) {
        result: 3dDAMAGE_DICE
    }
    result: 4dDAMAGE_DICE
}

function: barbarian attack with WEAPON:n on ARMOR_CLASS with ATTACK_MODIFIER {
    result: (1d20 + ATTACK_MODIFIER >= ARMOR_CLASS) * (WEAPON + 2)
}

function: cleric attack with CANTRIP:n on MODIFIER with SAVE_DIFFICULT {
    result: (1d20 + MODIFIER < SAVE_DIFFICULT) * CANTRIP
}

function: cleric attack with CANTRIP:n on ARMOR_CLASS with ATTACK_MODIFIER {
    result: (1d20 + ATTACK_MODIFIER >= ARMOR_CLASS) * CANTRIP
}

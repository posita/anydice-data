WDIE: {0,0,1}
BDIE: {-1,0,0,0,1,2}
POOL: 10
START: 0

function: roll WHITE:s BLACK:s {
  WHITE_SUCCESSES: [count {1} in WHITE]
  BLACK_SUCCESSES: [count {1} in BLACK] + 2*[count {2} in BLACK]
  BLACK_FAILURES: [count {-1} in BLACK]

  if BLACK_FAILURES > WHITE_SUCCESSES {
    result: -1
  } else {
    result: WHITE_SUCCESSES + BLACK_SUCCESSES
  }
}

loop B over {START..POOL} {
  W: POOL - B
  output [roll WdWDIE BdBDIE] named "[POOL] dice: [B] Black + [W] White"
}

\\\*** Base Damage Calc Info ***\\\
ENEMY_AC: 18
LEVEL: 20
STAT_MOD: 6
PROF: (LEVEL+7)/4

\\\*** Attack B *** (GWM) \\\
ATTACK_DICE_B: 2d6
BONUS_DAM_MOD_B: 3
BONUS_DICE_B: 1d8 + 2d8
HIT_BONUS_B: 0
MAGIC_B: 3
SINGLE_HIT_B: 4d8
N_ATTACKS_B: 3
BA_GWM_CRIT: 1 \"0" if no BA attack on crit, "1" if yes\

\\\*** General Info ***\\\
CRIT_RANGE: 20
CRIT_BONUS: 0 \Any additional dice used specifically for a critical\
NORMAL_HIT: d20
HIT_ADV: [highest 1 of 2d20]


\\\*** Functions ***\\\
function: attack ROLL:n vs ENEMY_AC:n {
 if ROLL = 1 { result: STAT_MOD }
 if ROLL >= CRIT_RANGE { result: CRITICALDAMAGE }
 if ROLL + ATTACK >= ENEMY_AC { result: HITDAMAGE }
 result: STAT_MOD
}

function: single ROLL:n vs ENEMY_AC:n {
 if ROLL = 1 { result: 0 }
 if ROLL >= CRIT_RANGE { result: CRIT_SINGLEDAMAGE }
 if ROLL + ATTACK >= ENEMY_AC { result: SINGLEDAMAGE }
 result: 0
}

ATTACK: STAT_MOD + PROF + MAGIC_B + HIT_BONUS_B
HITDAMAGE: [highest 1 of 2d(ATTACK_DICE_B + BONUS_DICE_B)] + STAT_MOD + MAGIC_B + BONUS_DAM_MOD_B
CRITICALDAMAGE: 2dATTACK_DICE_B + STAT_MOD + MAGIC_B + BONUS_DAM_MOD_B + 2dBONUS_DICE_B + CRIT_BONUS + BA_GWM_CRITd(ATTACK_DICE_B + MAGIC_B + BONUS_DAM_MOD_B + BONUS_DICE_B)
SINGLEDAMAGE: [highest 1 of 2dSINGLE_HIT_B]
CRIT_SINGLEDAMAGE: [highest 1 of 2d(2dSINGLE_HIT_B)]
output N_ATTACKS_Bd[attack NORMAL_HIT vs ENEMY_AC] + [single NORMAL_HIT vs ENEMY_AC] named "Glory DPR"
output N_ATTACKS_Bd[attack HIT_ADV vs ENEMY_AC] + [single NORMAL_HIT vs ENEMY_AC] named "Adv Glory DPR"

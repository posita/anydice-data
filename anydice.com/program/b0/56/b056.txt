MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

DEFENSE: d{12..22}

ATTACK: 3 + 4
HITDAMAGE: 1d10
CRITICALDAMAGE: 2d10
MISSDAMAGE: 0
output [attack] named "Eldritch Blast 1"
output [attack] named "Eldritch Blast 2"

ATTACK: 30
HITDAMAGE: 1d4 
CRITICALDAMAGE: 1d4
output [attack] named "Magic Missile 1"
output [attack] named "Magic Missile 2"
output [attack] named "Magic Missile 3"

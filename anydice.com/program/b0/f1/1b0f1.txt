function: hit X:n B:n A:n D:d P:n C:n R:n{
 if X+B+P>=A {result: [dam D]+B+[ref D P C R 1d20]}
 result: 0
}

function: dam D:n{
 result: D
}

function: ref D:d P:n C:n R:n X:n{
 if R+X<=(8+P+C) {result: [dam D]}
 result: 0
}

function: ttk D:n H:n{
 result: (H/D)+1
}

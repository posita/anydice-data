\
Apocalypse World dice with Disadvantage (Roll 3d6 and keep the worst 2 dice)
Result -1 is Miss; result 0 is Weak Hit; result 1 is Hit
\

function: aw ROLL:n {
if ROLL >= 10 { result: 1 }
if ROLL <= 6 { result: -1 }
result: 0
}

loop DICE over {-2..4} {
output [aw [lowest 2 of 3d6] + DICE] named "[DICE]"
}

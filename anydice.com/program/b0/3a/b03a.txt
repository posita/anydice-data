MISS: 0
HIT: 1
CRITICALHIT: 2

ENEMYAC: 20

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: 2*(d12+d6+d4)+2d12+23
 }
 if ATTACKRESULT = HIT {
  result: d12+d6+d4+23
 }
}

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRITICALHIT }
 if ROLL + 9 >= DEFENSE { result: HIT }
 result: MISS
}

output 5d[damage for [attack d20 vs 20]]

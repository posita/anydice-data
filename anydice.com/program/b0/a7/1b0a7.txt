function: test S with E effort and Y dice
{
D: E+Y
R: 0
R: 0 + [skill S verses Dd{0..9} with Y dice]
result: R
}

function: skill S verses SEQ:s with D dice
{
  K: S-1
  SEQR: [reverse SEQ]
  R: {}
  loop P over {1..D} { R: {R, P@SEQR}}
  if 1@R > K {result: 0}
  else
  {result: [count {0..K} in R] + [count {0} in R]}
}

loop S over {2..8}
{loop E over {0..0}
{loop Y over {5}
{output [test S with E effort and Y dice] named "With skill [S] and using [E] effort and [Y] dice"}
}}

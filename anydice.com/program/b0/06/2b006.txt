WEAPON_HITMOD: 5
AC: 11
DMG: 1d12+3

function: advantage attack with mod N {
  result: [attack with mod N] + [attack with mod N] != 0
}

function: attack with mod N {
  result: d20+HITMOD >= AC
}

output  DMG       *  (d20+HITMOD >= AC)                                named "Basic attack"
output  DMG       * ((d20+HITMOD >= AC) + (d20+HITMOD >= AC) != 0)     named "Advantaged Basic attack"

output (DMG+10)   *  (d20+HITMOD-5 >= AC)                              named "GWM attack"
output (DMG+10)   * ((d20+HITMOD-5 >= AC) + (d20+HITMOD-5 >= AC) != 0) named "Advantaged GWM attack"
output (DMG+10)/2 * ((d20+HITMOD-5 >= AC) + (d20+HITMOD-5 >= AC) != 0) named "(avged) True Strike + GWM attack"

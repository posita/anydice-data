\**** Damage calculator normal Vs advantage ****\

function: attackroll ROLL:n vs AC:n {
if ROLL >= CRITROLL { result: CRITDAMAGE }
if ROLL = 1 { result: 0 }
if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
result: 0
}

\looped against enemy AC from 15-25\
ENEMYAC: d{15..25}



ATTACKMOD: 0
PROFICIENCY: 3
ACCBONUS: 2
CRITROLL: 20
DMGBONUS: 14
ATTACKTIMES: 7
DAMAGE: 3d6 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2*(3d6 + ATTACKMOD + DMGBONUS)
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "X-BOW SHARPSHOOTER NORMAL"

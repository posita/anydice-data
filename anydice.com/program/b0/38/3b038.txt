\ Configuration A with rerolled 1s (sorted descending) \
DIE: d{2..6} \ Reroll 1s (dice range 2-6) \

function: config A INDEX:n of ROLLS:s {
  \ Sort descending and reverse to use 1-based indexing \
  SORTED: reverse [sort ROLLS] 
  result: 12 + (SORTED@INDEX) - (SORTED@(7 - INDEX))
}

\ Loop through stats (Strength=1, Dexterity=2, etc.) \
loop I over {1..6} {
  output [config A I of 6dDIE] named "Stat [I]"
}

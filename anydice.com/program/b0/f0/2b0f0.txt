function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}
function: strattack DIE:d with STRMOD:n {
    DMIN: [lowest of #{DIE}/2 and 1 + STRMOD]
    result: [highest of DIE and DMIN]
}

loop N over {4,6,8,10,12} {
loop STR over {0..5} {
if N/2 > STR {
if N/2 = STR+1 {output [strattack 1dN with STR] named "1d[N] with StrMod +[STR] and higher"} else {output [strattack 1dN with STR] named "1d[N] with StrMod +[STR]"}
}
}
}
output 2d[reroll {1,2} on 1d6 to 1d6] named "RAW greatsword/maul with GWF"

function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE
 }
 result: DAMAGE
}

AC: 12
STRENGTH: 3
ATTACK: 5

DAMAGE: 1d8 + STRENGTH
THREAT: 20
CRITICAL: 2
output [d20 vs AC] named "One hand"

DAMAGE: 1d10 + STRENGTH
THREAT: 20
CRITICAL: 2
output [d20 vs AC] named "Two hand"

DAMAGE: 1d6 + STRENGTH
THREAT: 20
CRITICAL: 2
output [d20 vs AC] named "Mace"

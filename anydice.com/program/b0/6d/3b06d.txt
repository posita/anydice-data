function: attack ATTACK:d {
  result: ATTACK
}

function: multiattack X:n Y:d {
  TOTAL: 0
  loop N over {1..X} {
    DAMAGE: [attack Y]
    TOTAL: TOTAL + DAMAGE
  }
  result: TOTAL
}

GWF: 2d{3,3,3,4,5,6}
MULTIATTACK: 3
ATTACK: 2d6

output [multiattack 1 ATTACK] named "1 Attack Basic"
output [multiattack 1 GWF] named "1 Attack GWF"

output [multiattack 2 ATTACK] named "2 Attack Basic"
output [multiattack 2 GWF] named "2 Attack GWF"

output [multiattack 8 ATTACK] named "8 Attack Basic"
output [multiattack 8 GWF] named "8 Attack GWF"

\damage per round (dpr)\

\smite damage roll, given spell level, and whether or not its undead\
function: smite level LVL:n against UN:n undead {
  result:(LVL+!!UN+1)d8
}

\calculate damage given hit roll, hit mod, enemy ac, damage die, damage mod, \
\smite level (0 for no smite), whether to smite only on crit, \
\and whether opponent is undead/fiend (for smite extra damage)\
function: damage if HIT:n plus MOD:n hits ac AC:n with DMG_DIE:d die plus D_MOD:n and level S:n smite on crit C against U undead{
  if HIT=20 {
    if S {
      result: 2*DMG_DIE+D_MOD+[smite level S against U undead]*2
    }
    else {
      result: 2*DMG_DIE+D_MOD
    }
  }
  else if HIT = 1 {
    result: 0
  }
  else if HIT+MOD>=AC {
    if S & !C {
      result: DMG_DIE+D_MOD+[smite level S against U undead]
    }
    else {
      result: DMG_DIE+D_MOD
    }
  }
  else {
    result:0
  }
}

function: max of X:n and Y:n {
  if X > Y {
    result: X
  } else {
    result: Y
  }
}

function: attra A:n attrb B:n attrc C:n wu U:n wv V:n ww W:n sg S:n fw F:n {
  if [count{1,1} in {U,V,W}] > 2 {result: [max of F and 1]}
  if [count{20,20} in {U,V,W}] > 2 {result: 0}

  L: [max of U - (A - S) and 0]
  M: [max of V - (B - S) and 0]
  N: [max of W - (C - S) and 0]
  R: L + M + N
  if R <= F {result: [max of F - R and 1]}

  result: 0
}
output [attra 10 attrb 10 attrc 10 wu 1d20 wv 1d20 ww 1d20 sg 0 fw 0]

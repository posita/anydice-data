function: test S with E effort
{
D: E+4
R: 0
R: 0 + [skill S verses Dd{0..9}]
result: R
}

function: skill S verses SEQ:s 
{
  K: S-1
  SEQR: [reverse SEQ]
  R: {}
  loop P over {1..4} { R: {R, P@SEQR}}
  if 1@R > K {result: -1}
  else
  {result: [count {0..K} in R] + [count {0} in R]}
}

loop S over {1..8}
{loop E over {0..0}
{output [test S with E effort] named "With skill [S] and using [E] effort"}
}

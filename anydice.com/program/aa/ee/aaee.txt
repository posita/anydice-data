set "maximum function depth" to 4

function: dxdice ROLL:s {
CRITS: ROLL = 1
if CRITS {
result: [dxdice CRITS d CD] + 10
}
result: 1@((#ROLL)dND)
}

CRITVALUE: 10
CD: d10 >= CRITVALUE
ND: d{1..CRITVALUE-1}

loop N over {1..5}{
output [dxdice NdCD] named "[N]d10 cv10"
}

CRITVALUE: 9
CD: d10 >= CRITVALUE
ND: d{1..CRITVALUE-1}

loop N over {1..5}{
output [dxdice NdCD] named "[N]d10 cv9"
}

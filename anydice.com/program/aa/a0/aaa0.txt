\this calculates damage for weapons of flaming or freezing\
\leave YOUR_TARGETS_ARMOR_CLASS and YOUR_TARGETS_ELEM_RESIST blank for general stats\

YOUR_WEAPONS_BASE_ATTACK: 10
YOUR_WEAPONS_ENCHANTMENT: 0
YOUR_TARGETS_ARMOR_CLASS: 0
YOUR_TARGETS_ELEM_RESIST: 0

\fill in the above then hit output\

function: clamp A:n to zero {
	if A < 0 {result: 0}
	else {result: A}
	}

function: brand damage A:n v B:n {
	if A = 0 {result: ((B * (1d51 - 1)) / 100)}
		else
		if A = 1 {result: ((B * (1d26 - 1)) / 100)}
			else
			if A = 2 {result: ((B * (1d11 - 1)) / 100)}
				else
				if A = 3 {result: 0}
					else
					if A = -1 {result: ((B * (1d76 - 1)) / 100)}
						else {result: -10}
	}

function: end result A:n B:n {
	if A = -10 {result: 35565}
	else {result: B}
}

W_M_ATK: YOUR_WEAPONS_BASE_ATTACK + (1d(YOUR_WEAPONS_ENCHANTMENT + 1) - 1)
T_F_INJ: W_M_ATK - 1dYOUR_TARGETS_ARMOR_CLASS
T_T_INJ: [clamp T_F_INJ to zero] + [brand damage YOUR_TARGETS_ELEM_RESIST v [clamp T_F_INJ to zero]]

output [end result [brand damage YOUR_TARGETS_ELEM_RESIST v [clamp T_F_INJ to zero]] T_T_INJ]

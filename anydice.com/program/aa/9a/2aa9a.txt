function: evaluate ROLL:s {
 SUCCESSES: ROLL >= 7
 if SUCCESSES > 0 { result: SUCCESSES + (ROLL = 10) }
 if [count 1 in ROLL] { result: -1 }
if [count 10 in ROLL] { result: +1 }
 result: 0
}

loop DICE over {1..5} {
 output [evaluate DICE d10] named "[DICE]d10"
}

loop DICE over {2..6} {
 output [evaluate 1@DICE d10] named "Bonus[DICE]d10"
}

loop DICE over {2..6} {
 output [evaluate DICE@DICE d10] named "Malus[DICE]d10"
}

set "maximum function depth" to 6 \ Notes: max 11 rolls... probably anydice can't handle any more \

function: age S:n from A:n minus D:n failed F:n rolls R:n {
  if (A<13) {
    result: [age d6 from (A+S) minus D failed F rolls R+1]
  } else {
    if ((S-D) > 0) { \ aged + dark \
      result: [age d6 from (A+(S-D)) minus (D+1) failed 0 rolls R+1]
    } else { \ fail, no age \
      if (F=1) { \ failed last time, end chargen \
        result: A \ replace A with D to get darkness, replace with R to get (6=6 or more) \
      } else {
        result: [age d6 from A minus D failed 1 rolls R+1]
      }
    }
  } 
}
output [age d6 from 0 minus 6 failed 1 rolls 1]
output [age d6 from 13 minus 0 failed 0 rolls 1]

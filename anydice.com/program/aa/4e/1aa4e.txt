function: compare R:n to X:n and Y:n {
 if R > X {
  if R > Y { result : 2 }
  else { result : 1 }
 }
 else {
  if R > Y { result : 1 }
  else { result : 0 }
 }
}

function: cap NUMBER:n at MAX:n {
 if NUMBER < MAX { result: NUMBER }
 result: MAX
}

loop N over {0..6} {
  output [compare [cap 1d6+N at 10] to d10 and d10] named "+[N]"
}

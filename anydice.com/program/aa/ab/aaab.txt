DEF: 2
HEALTH: 4
function: getdamage H:n and R:n and D:s {
    if(H < R) {result: 0}
    else if (H >= 6) {result: [lowest of D and HEALTH]}
    else {result: [lowest of [highest of 0 and (D - DEF)] and HEALTH]}
}

output [getdamage d8 and 4 and d8] * ([getdamage d8 and 4 and d8] = HEALTH) + [getdamage d8 and 4 and d8]

output 2d[getdamage d6 and 4 and d6]

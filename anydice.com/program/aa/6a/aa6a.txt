function: count sixes from SEQUENCE:s {
 COUNT: 0
 loop N over {1..#SEQUENCE} {
  COUNT: COUNT + (6=N@SEQUENCE)
 }
 result: COUNT
}

function: convert DICEPOOL:s {
 result: DICEPOOL
}

function: wtfroll N dice {
 SIXES: 0
 B: 0
 loop A over {1..N} {
  A: 1d6
  B: (((A/B)*2)/((A/B)+1))*B + (((B/A)*2)/((B/A)+1))*A
  SIXES: SIXES+(A=6)
 }
 result: B+SIXES
}

loop N over {2..10} {
 output [wtfroll N dice] named "[N] dice"
}

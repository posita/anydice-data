set "maximum function depth" to 12

function: caravanroll LV:n N:n M:n helper {
 if N = M { 
  if N=1 { result: 0 }
  if N=2 { result: 2 }
  if N=3 { result: 6 + LV }
  if N=4 { result: 8 + LV }
 }
 SUM: N+M
 result: SUM + LV
}

function: caravanroll LV:n {
 result: [caravanroll LV 1d4 1d4 helper]
}

loop ML over {-3, -2, -1, 1, 3, 5} {
 output [caravanroll 0] >= (1d8 + ML) named "[ML] ML"
}

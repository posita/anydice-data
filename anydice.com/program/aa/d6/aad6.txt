MISS: 0
HIT: 1
CRITICALHIT: 2
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}
function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

DEFENSE: d{12..22}

ATTACK: 6
HITDAMAGE: 1d8+6
CRITICALDAMAGE: 1d8 *2 + 6
MISSDAMAGE: 0
output [attack] named "Plusone"
HITDAMAGE: 1d8+5
CRITICALDAMAGE: 1d8 *2 + 5
output [attack] named "Battleaxe"
HITDAMAGE: 1d6+5
CRITICALDAMAGE: 1d6 *2 + 5
output [attack] named "Handaxe"

function: wtfroll N dice {
 SIXES: 0
 B: 1
 loop A over {1..N} {
  A: 1d6
  B: [special highest of A and B]
 }
 result: B+SIXES
}

function: special highest of A:n and B:n {
 result: (((A/B)*2)/((A/B)+1))*A + (((B/A)*2)/((B/A)+1))*B - (A/B)*(B/A)*A + (A/6)*(((B/6)*2)/((B/6)+1))
}

output 1d6-1 named "0 dice"
loop N over {1..10} {
 output [wtfroll N dice] named "[N] dice"
}

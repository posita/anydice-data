function: impregnation duel COYNN:s LINK:s {
 if LINK >= COYNN {
  result: 0 \ if Link has rolled higher than Coynn, he manages to prevent impregnation from this die roll / load \
 }
 else {
  result: COYNN \ otherwise, Coynn's sperm find their mark, adding his roll to the number of eggs fertilized \
 }
}

function : max A:n B:n { \ just a simple function to find the maximum of two numbers \
 if A >= B {
  result: A
 }
 else {
  result: B
 }
}

function: round up A:n divided by B:n { \ AnyDice rounds down by default on division, so this is needed as a workaround if we want to round up \
 result: [max A/B (A+1)/B]
}

loop N over {1..10}{
 output 3d[impregnation duel [explode 1d(N*2)] 1d([round up N divided by 2])]
}

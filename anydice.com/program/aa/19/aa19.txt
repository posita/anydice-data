function: roll DIEA:n u BONUSA:n s SKILLA:n vs DIEB:n u BONUSB:n s SKILLB:n {
  MAX: 20
  \ PLAYER A ROLLS \
  if DIEA <= BONUSA {
    if DIEA = BONUSA {
      A: 100
    } else {
      A: SKILLA + DIEA
    }
  } if DIEA > BONUSA {
    if DIEA = MAX {
      A: -100
    } else {
      A: SKILLA
    }
  }
  \ PLAYER B ROLLS \
  if DIEB <= BONUSB {
    if DIEB = BONUSB {
      B: 100
    } else {
      B: SKILLB + DIEB
    }
  } if DIEB > BONUSB {
    if DIEB = MAX {
      B: -100
    } else {
      B: SKILLB
    }
  }
  \ COMPARE ROLLS \
  if A > B {
    if A = 100 {
      if B = -100 {
        FINAL: 100
      } else {
        FINAL: 50
      }
    } else {
      if B = -100 {
        FINAL: 50
      } else {
        FINAL: A-B
      }
    }
  } if A < B {
    if A = -100 {
      if B = 100 {
        FINAL: -100
      } else {
        FINAL: -50
      }
    } else {
      if B = 100 {
        FINAL: -50
      } else {
        FINAL: A-B
      }
    }
  } if A = B {
    FINAL: 0
  }
  result: FINAL
}

output [roll d20 u 5 s 0 vs d20 u 5 s 0]
output [roll d20 u 5 s 1 vs d20 u 8 s 0]

CRIT: 2
PASS: 1
FAIL: 0

GRIT: 3 \ Adds a static bonus of (License Level / 2) to all rolls (maxes out at 6)  \
TN: 10 \ Target Number \

function: lancer ROLL:n 
{
 if ROLL + GRIT >= 20 { result: CRIT }
 else if ROLL + GRIT >= TN { result: PASS }
 else if ROLL + GRIT <= TN - 1 { result: FAIL }
}

output [lancer d20 - [highest 1 of 1d6]] named "+1 Difficulty"

output [lancer d20] named "Standard Roll" 

output [lancer d20 + [highest 1 of 1d6]] named "+1 Accuracy"

AC: 15
CRIT_THRESHOLD: 20
AMOD: 7
DMOD: 4
function: attack ROLL:n for DAMAGE:d{
 if ROLL >= CRIT_THRESHOLD { result: DAMAGE+DAMAGE+DMOD }
 if ROLL >= AC - AMOD { result: DAMAGE+DMOD }
 result: 0
}

function: turn ACTION:d BONUS:d{
 result: [attack d20 for ACTION] + [attack d20 for ACTION] + [attack d20 for BONUS]
}

TURN_A: [turn 1d6+1d6 0]
TENTURN_A: [turn 1d10+1d6 0]
CRIT_THRESHOLD: 19
DMOD: 7
TURN_B: [turn 1d6+1d6 0]
TENTURN_B: [turn 1d10+1d6 0]
TURN_C: [turn 1d6+1d6 1d4+1d6]
TENTURN_C: [turn 1d10+1d6 1d4+1d6]
output TURN_A named "turn1"
output TURN_B named "turn2"
output TURN_C named "turn3"
output (TURN_A + TURN_B + TURN_C)/3 named "Average"

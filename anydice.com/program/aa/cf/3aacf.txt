function: dualcrit on ROLLHOPE:n and ROLLFEAR:n at least X {
 if ROLLHOPE = ROLLFEAR {
  result: 1
 }
 if ROLLHOPE != ROLLFEAR {
  result: [highest of ROLLHOPE and ROLLFEAR] >= X
 }
}

loop P over {4..12} {
output [dualcrit on 1d12 and 1d12 at least P] named "At least [P]"
}
output [highest 1 of 2d12] >= 12 named "Natural 12"
output [dualcrit on 1d12 and d12 at least 13] named "Miracle"

AC: 15

function: hit D:n plus HIT_MOD:n for DMG:d plus DMG_MOD:n {
  if D = 20 {
    result: DMG + DMG + DMG_MOD
  } else if D+HIT_MOD >= AC {
    result: DMG + DMG_MOD
  } else {
    result: 0
  }
}

SAVE_MOD: 2
function: save D:n DC:n for DMG:d {
  if D + SAVE_MOD < DC {
    result: DMG
  } else {
    result: DMG / 2
  }
}

EVASION: 16
ARMOR: 0
NUCLEAR: 0

function: hammer R:n {
  if (R < EVASION) {
    result: 0
  } else if (R < 20) {
    result: [highest of (3+NUCLEAR)d{2..6}+3-ARMOR and 0]
  } else {
    result: [highest of {1..(3+NUCLEAR)}@(6+2*NUCLEAR)d{2..6}+3-ARMOR and 0]
  }
}

function: dsas R:n {
  if (R < EVASION) {
    result: 0
  } else if (R < 20) {
    result: [highest of 2d6-ARMOR and 0]
  } else {
    result: [highest of {1..2}@4d6-ARMOR and 0]
  }
}

output [hammer 1d20+3+1d6] named "Skirmish damage"
output [hammer 1d20+3+1d6]+[dsas 1d20+3] named "Barrage damage"

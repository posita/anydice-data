YEAR : 1

function : calc value CHAIN:s{
   R:0
   CRITFACTOR:0
   loop P over {1..#CHAIN}{
      R: R + P@CHAIN
      if P > -1 {
         if P@CHAIN = -3 {
            CRITFACTOR : CRITFACTOR - 1
         }
         if P@CHAIN = 2 {
            CRITFACTOR : CRITFACTOR + 1
         }
      }
   }
   R: R
   if(R < 0){R:0}
   if(R>20){R:20}
   result : CRITFACTOR
}

output [calc value (YEARd{-3:1,-2:4,0:5,1:9,2:1})]

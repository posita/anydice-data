function: percentile NUMBER under DIE:n{
if DIE<=NUMBER {result: 1}
result: 0}

function: level up SAVE for CHANCE{
result: SAVE+[percentile CHANCE under 1d100]
}

HSAVE: 0
HOLY: 30

WSAVE: 0
WITCH: 25

FSAVE: 0
FIGHTER: 20

BSAVE: 0
BARD: 15

SSAVE: 0
SWASH: 10

loop N over {1..14}{
HSAVE: [level up HSAVE for HOLY] 
HSAVE: [level up HSAVE for HOLY] 
output HSAVE named "HOLY [N]"

WSAVE: [level up WSAVE for WITCH]
WSAVE: [level up WSAVE for WITCH] 
output WSAVE named "WITCH [N]"

FSAVE: [level up FSAVE for FIGHTER]
FSAVE: [level up FSAVE for FIGHTER] 
output FSAVE named "FIGHTER [N]"

BSAVE: [level up BSAVE for BARD]
BSAVE: [level up BSAVE for BARD] 
output BSAVE named "BARD [N]"

SSAVE: [level up SSAVE for SWASH]
SSAVE: [level up SSAVE for SWASH] 
output SSAVE named "SWASH [N]"
}

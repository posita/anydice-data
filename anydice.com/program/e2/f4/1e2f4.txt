function: modifier A:n {
  if A < 4 { result: -4 }
  if A < 6 { result: -3 }
  if A < 8 { result: -2 }
  if A < 10 { result: -1 }
  if A < 12 { result: 0 }
  if A < 14 { result: 1 }
  if A < 16 { result: 2 }
  if A < 18 { result: 3 }
  if A < 20 { result: 4 }
}

output [modifier 3d6] named "traditional"
output 4d{-1..1} named "fudge"
output [modifier [highest 3 of 4d6]] named "traditional drop"
output [highest 4 of 5d{-1..1}] named "fudge drop 1"
output [highest 4 of 6d{-1..1}] named "fudge drop 2"

loop I over {0 .. 5}{
    DICEI: I
    TARGETI: {5 .. 10}
    if I = 0 {
       DICEI: 1
       TARGETI: {10} 
    }
    loop J over {0 .. 5}{
        DICEJ: J
        TARGETJ: {5 .. 10}
        if J = 0 {
            DICEJ: 1
            TARGETJ: {10}
        }
        output [count TARGETI in DICEId10] - [count TARGETJ in DICEJd10] named "[I]d10 - [J]d10"
    }
}

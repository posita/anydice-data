function: dubs SEQ:s {
  FOO: 0
  loop P over {1..10} {
    BAR: (P = SEQ)
    if BAR > 1 {
      FOO: FOO + 1
    }
  }
  if FOO > 0 {
    result: 1
  }
  result: 0
}

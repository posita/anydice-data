function: highest number of matches in ROLL:s {
  MAX: 0
  loop I over ROLL {
    COUNT: ROLL = I
    if MAX < COUNT { MAX: COUNT }
  }
  result: MAX
}

function: highest matched number in ROLL:s {
  MAX: 0
  loop I over ROLL {
    COUNT: ROLL = I
    if COUNT > 1 & MAX < I { MAX: I }
  }
  result: MAX
}

loop N over {1..5} {
  output [highest number of matches in Nd6] > 1 named "[N]d6 any success"
  output [highest matched number in Nd6] >= 5 named "[N]d6 full success"
}


function: attack ROLL:n vs DEFENSE:n {
 if ROLL + ATTACK >= DEFENSE { result: HITDAMAGE }
 result: 0
}

DEX: 3
AC: 15

loop AC over {10, 15, 20} {
  ATTACK: DEX - 5
  HITDAMAGE: 1d8 + 10 + DEX
  output [attack d20 vs AC] named "Longbow vs [AC]"

  ATTACK: DEX
  HITDAMAGE: 1d6 + DEX
  output [attack d20 vs AC] + [attack d20 vs AC] named "Crossbow expert vs [AC]"
}

function: ten to the power of A:n place of B:n {
  if (A - #B > 0) {
    result: 0
  }
  result: (#B - A)@B
}

function: sum digits of A:n {
  result: {1..10}@A
}

function: digital root of A:n {
  loop N over {1..5}{
    A: [sum digits of A]
  }
  result: A
}

D: d3-1 \d{0..2}\

\loop N over {1..8}{
 output NdD named "[N] dice"
}\

\N:4
loop M over {1..8}{
 output NdD > MdD named "[N] vs [M] dice"
}\

\loop N over {1..20} {
  output [ten to the power of d2-1 place of Nd20]
}\

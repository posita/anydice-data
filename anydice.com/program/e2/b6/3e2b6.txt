\ FIRE DAMAGE + BURN spell Balance Calculator \

FD: 2d6                                       \Fire Damage Per Hit\
BS: 2                                         \Burning Stacks Applied Per Hit\
H: 5                                          \Number of attempted Hits\

HD: FD+BSd4+BSd2+BS                           \Hit Damage\

SA: (HD*62/100)+(((38*HD)/200))             \Single Target Damage accounting for accuracy\

TD: SA*H                                      \Total damage, summing the damage against all targets attempted to hit and accounting for accuracy\

MPC: TD*4/5                         \Balanced Mana Point Cost, with damage accounted for accuracy\

output HD named "Hit Damage"
output SA named "Accuracy Hit Damage"
output TD named "All Target Total Accuracy Hit Damage"
output MPC named "Balanced Mana Point Cost"

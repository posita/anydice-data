function: adv X:d {
  result: 1@2dX
}

function: avg X:d {
  result: ((#{X})@{X} - 1@{X} + 1) / 2
}

function: average damage of X:d with Y:d to hit {
  result: (Y > 10)
}

W_DMG: 1d8 + 6 + 2d4
UA_DMG: 1d8 + 4 + 1d4

W_DMG_GWM: W_DMG + 10

ATTACK_MOD: 10
ATTACK_MOD_GWM: ATTACK_MOD - 5

DPR_BASE: 2dW_DMG + 2dUA_DMG
DPR_GWM: 2dW_DMG_GWM + 2dUA_DMG

HIT_BASE: 1d20 + ATTACK_MOD
HIT_GWM: 1d20 + ATTACK_MOD_GWM

\
output W_DMG named "Weapon (base)"
output UA_DMG named "Unarmed attack (base)"

output W_DMG + 10 named "Weapon (GWM)"
output [adv W_DMG] named "Weapon (SA)"
\
\
output DPR_BASE named "DPR (base)"
output [adv DPR_BASE] named "DPR (SA)"
output DPR_GWM named "DPR (GWM)"

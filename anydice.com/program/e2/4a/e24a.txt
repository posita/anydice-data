\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + CHA+ PROFICIENCY>= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

CHA: 3
PROFICIENCY: 3
CRITROLL: 19
ATTACKTIMES: 1
DAMAGE: 1d10 + CHA + 2d8
CRITDAMAGE: 2d10 + CHA + 4d8
output ATTACKTIMESd[attackroll d20 + CHA + PROFICIENCY vs ENEMYAC] named "Longsword +3 proficiency +5 CHA modifier"

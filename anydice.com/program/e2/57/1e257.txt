function: dice ROLL:n ac AC:n attack BONUS:n{
    if ROLL = 20 { result: 0 }
    if ROLL + BONUS >= AC {
        if ROLL + BONUS < AC + 5{
            result: 1
        }
    }
    result: 0 
}

output [dice [highest 1 of 2d20] ac 15 attack 3] named "Bandit vs Mage Armor (Advantage)"
output [dice [highest 1 of 2d20] ac 19 attack 3] named "Bandit vs Armor (Advantage)"
output [dice 1d20 ac 15 attack 3] named "Bandit vs Mage Armor"
output [dice 1d20 ac 19 attack 3] named "Bandit vs Armor"

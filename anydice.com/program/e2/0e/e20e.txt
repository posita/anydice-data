
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 20 { result: DAMAGEDICE * 2 + DAMAGEMOD }
 if ROLL + ATTACK >= DEFENSE { result: DAMAGEDICE + DAMAGEMOD }
 result: 0
}

DEX: 4

loop AC over {10, 15, 17, 20} {
  ATTACK: DEX - 5
  DAMAGEDICE: 1d8
  DAMAGEMOD: 10 + DEX
  output [attack d20 vs AC] named "Sharpshooter longbow vs [AC] AC"

  ATTACK: DEX
  DAMAGEDICE: 1d6
  DAMAGEMOD: DEX
  output [attack d20 vs AC] + [attack d20 vs AC] named "Crossbow expert vs [AC] AC"
}

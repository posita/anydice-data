function: brawl A:s vs B:s {
  if (#A@A < 1@B) {
    SWAPA: {1@A}
    loop X over {2..#A-1} {
      SWAPA: {SWAPA, X@A}
    }
    SWAPA: {SWAPA, 1@B}
    SWAPB: {#A@A}
    loop X over {2..#B} {
      SWAPB: {SWAPB, X@B}
    }
    A: [sort SWAPA]
    B: [sort SWAPB]
  }
  SA: A >= 1@B
  SB: B >= 1@A
  if SA-SB=0 {
    result:(A > B) - (A < B)
  }
  result:SA-SB
}
output [brawl 3d6 vs 3d6] named "A vs B Damage"

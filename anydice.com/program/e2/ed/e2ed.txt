POISON: POISONDAMG

function: proficiency LEVEL {
result: ((LEVEL-1)/4) + 2
}


function: hit DICE HITS LEVEL {
result: [hit DICE HITS LEVEL 0 0]
}

function: hit DICE HITS LEVEL offhand {
result: [hit DICE HITS LEVEL 0 1]
}

function: chit DICE HITS LEVEL {
result:[hit DICE HITS LEVEL 1 0]
}

function: chit DICE HITS LEVEL offhand{
result:[hit DICE HITS LEVEL 1 1]
}

function: hit DICE HITS LEVEL CRIT OFFHAND{
if OFFHAND>=1{
OFFHAND: 0
} else {
OFFHAND: 1
}
if CRIT>=1{
CRIT: CRITMULT
NORMCRIT: 2
} else {
CRIT: 1
NORMCRIT: 1
}
OUTPUT:0
loop N over {1..HITS}{
OUTPUT: OUTPUT + (CRITd(DICE)+(OFFHAND*(DAMAGEBONUS))+NORMCRITd(ENLARGE)+NORMCRITd(HEX)+(POISON*((1d20+ENEMYCON)<POISONSAVE)))*((1@NORMCRITd20+HITBONUS+[proficiency LEVEL])>=AC)
}
result: OUTPUT
}


function: sneak ATTACKS LEVEL RLEVEL {
SDICE:(RLEVEL+1/2)
result: SDICEd6 * [lowest of((1@ATTACKSd20+HITBONUS+[proficiency LEVEL])>= AC) and 1]
}

function: csneak ATTACKS LEVEL RLEVEL {
SDICE:(RLEVEL+1/2)
result: 2d(SDICEd6) * [lowest of ((1@ATTACKSd20+HITBONUS+[proficiency LEVEL])>= AC) and 1]
}


function: supdice FLEVEL {
OUTPUT: (4+((FLEVEL+1)/8))
if MARTIALFEAT >=1{
OUTPUT: OUTPUT + 1
}
result: OUTPUT
}


function: supdamage FLEVEL {
result:(8+((FLEVEL-2)/8*2))
}

function: supd ATTEMPTS FLEVEL LEVEL{
ATTACKS: ATTEMPTS
if ATTEMPTS > [supdice FLEVEL]{
ATTACKS: [supdice FLEVEL]
}

OUTPUT:0
loop N over {1..ATTACKS}{
OUTPUT: OUTPUT + 1d[supdamage FLEVEL] *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

function: csupd ATTEMPTS FLEVEL LEVEL{
ATTACKS: ATTEMPTS
if ATTEMPTS > [supdice FLEVEL]{
ATTACKS: [supdice FLEVEL]
}
OUTPUT:0
loop N over {1..ATTACKS}{
OUTPUT: OUTPUT + 2d(1d[supdamage FLEVEL]) *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

function: dsmite ATTEMPTS SPELLONE SPELLTWO LEVEL{
\ALWAYS ASSUMING THAT WE CAN CAST ALL SPELLS WE NEED TO\
OUTPUT:0
loop N over {1..SPELLTWO }{
OUTPUT: OUTPUT + 3d6 *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
loop N over {1..SPELLONE }{
OUTPUT: OUTPUT + 2d6 *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

function: cdsmite ATTEMPTS SPELLONE SPELLTWO LEVEL{
\ALWAYS ASSUMING THAT WE CAN CAST ALL SPELLS WE NEED TO\
OUTPUT:0
loop N over {1..SPELLTWO }{
OUTPUT: OUTPUT + 2d(3d6) *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
loop N over {1..SPELLONE }{
OUTPUT: OUTPUT + 2d(2d6) *((N@2d20+HITBONUS+[proficiency LEVEL])>= AC)
}
result: OUTPUT
}

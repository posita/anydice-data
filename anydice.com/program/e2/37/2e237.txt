function: number of dice in ROLL:s that match at least N:n other {
  COUNT: 0
  SKIP: 1@ROLL + 1
  loop I over ROLL {
    MATCHES: ROLL = I
    if MATCHES >= N & SKIP > I {
      COUNT: COUNT + MATCHES - 1
      SKIP: I
    }
  }
  result: COUNT
}

loop N over {1..5} {
  output [number of dice in Nd6 that match at least 1 other] >= 1 named "[N]d6 any success"
  output [number of dice in Nd6 that match at least 1 other] >= 2 named "[N]d6 full success"
}

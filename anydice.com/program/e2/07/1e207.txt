
function: attack ROLL:n plus BONUS:n plus BONUSWPN:n vs DEF:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= DEF {
        result: HIT
    }
    result: MISS
}

function: attack ROLL:n plus BONUS:n plus BONUSWPN:n vs DEF:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS plus BONUSWPN vs DEF hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}

function: dice ROLL:n ac AC:n attack BONUS:n{
    if ROLL = 20 { result: 1 }
    if ROLL + BONUS >= AC {
        if ROLL + BONUS < AC + 5{
            result: 1
        }
    }
    result: 0 
}

loop ARMOR over {1..20}{
    output [dice [highest 1 of 2d20] ac ARMOR attack 3] named "Bandit vs AC [ARMOR]"
}

\Greataxes\

set "maximum function depth" to 3

output 1d12 named "baseline"

function: rerollones N:n {
 if N = 1 { result: 0 + 1d12 }
 result: N
}

output [rerollones 1d12] named "GWF"

function: explode N:n {
 if N = 12 { result: 12 + [explode d12] }
 result: N
}

output [explode 1d12] named "BRUTAL"


function: xplode N:n {
 if N = 12 { result: 12 + [explode d12] }
 if N = 1 { result: 0 + [explode d12] }
 result: N
}

output [xplode 1d12] named "GWF-BRUTAL"

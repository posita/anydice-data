function: hit ROLL:n BONUS:n AC:n {
  if (ROLL = 1) {
    result: 0
  } else if (ROLL = 20) {
    result: 1
  }
  result: ROLL + BONUS >= AC
}

function: multiply DAMAGE:d MULTIPLIER:n {
  SUM: 0
  loop N over {1..MULTIPLIER} {
    SUM: SUM + DAMAGE
  }
  result: SUM
}

function: attack ROLL:n BONUS:n AC:n DAMAGE:d PRECISION_DAMAGE:d CRIT_ROLL:n THREAT_RANGE:n MULTIPLIER:n EXTRA_CRIT_DAMAGE:d {
  if (![hit ROLL BONUS AC]) { 
    result: 0
  }
  if (ROLL >= THREAT_RANGE & [hit CRIT_ROLL BONUS AC]) {
    result: [multiply DAMAGE MULTIPLIER] + PRECISION_DAMAGE + [multiply EXTRA_CRIT_DAMAGE MULTIPLIER - 1]
  }
  result: DAMAGE + PRECISION_DAMAGE
}

function: simpleattack BONUS:n AC:n DAMAGE:d THREAT_RANGE:n MULTIPLIER:n {
  result: [attack 1d20 BONUS AC DAMAGE 0 1d20 THREAT_RANGE MULTIPLIER 0]
}

loop AC over {2..20} {
  output [simpleattack 0 AC 1d6 20 2] named "AC [AC]"
}

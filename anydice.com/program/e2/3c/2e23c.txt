function: spawn chance per cell with N tiles {
 result: [highest 1 of Nd(d100 <= 30)]
}
function: spawn chance per cell {
 result: [spawn chance per cell with 13 tiles]
}
function: seed chance per death {
 result: 1d1000 <= 69
}
function: seed chance per spawn event {
 result: [spawn chance per cell] > 0 & [seed chance per death] > 0
}
function: seed loss chance per crop {
 result: 1d100 > 75
}
function: seeds lost per crop cycle {
 result: 8d[seed loss chance per crop]
}

function: net seeds per X crop cycles and Y spawn events with Z cells {
 result: Zd(Yd[seed chance per spawn event]) - Xd[seeds lost per crop cycle]
}

output [spawn chance per cell] named "Spawn Chance Per Cell"
output [seed chance per death] named "Seed Chance Per Death"
output [seeds lost per crop cycle] named "Seeds Lost Per Crop Cycle"
output [net seeds per 11 crop cycles and 15 spawn events with 22 cells] named "Net Seeds Per Eleven (Current) Crop Cycles"

function: attack TARGET:n ROLL:n ATTACK_BONUS:n CRIT_RANGE:n CRIT_MULT:n DAMAGE_DIE_COUNT:n DAMAGE_DIE_SIZE:n DAMAGE_BONUS:n NONCRIT_DAMAGE_BONUS:n{
  if ROLL = 20 {result: (CRIT_MULT*DAMAGE_DIE_COUNT)dDAMAGE_DIE_SIZE + (CRIT_MULT*DAMAGE_BONUS) + NONCRIT_DAMAGE_BONUS}
  if ROLL = 1 {result: 0}
  if ROLL > CRIT_RANGE-1 {if ROLL+ATTACK_BONUS > TARGET-1 {result:(CRIT_MULT*DAMAGE_DIE_COUNT)dDAMAGE_DIE_SIZE + (CRIT_MULT*DAMAGE_BONUS) + NONCRIT_DAMAGE_BONUS}}
  if ROLL + ATTACK_BONUS > TARGET-1 {result:DAMAGE_DIE_COUNTdDAMAGE_DIE_SIZE + DAMAGE_BONUS + NONCRIT_DAMAGE_BONUS}
  result: 0
}

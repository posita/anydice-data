function: roll adv A {
D: d20
if A >= 1 {result: [middle 1 of [highest 3 of (A+3)d20]]}
if A <= -1 {result: [middle 1 of [lowest 3 of (A+3)d20]]}
else {result: [middle 1 of 3d20]}
}

function: calc R:n mod M vs T ef E {
if R = 20 {result: 2 * E}
if R = 1 {result: - E}
R: R + M
if R > T + 5 {result: E + E/2}
if R > T {result: E}
if R > T - 5 {result: E/2}
if R <= T - 5 {result: 0}
}

function: roll M vs T ef E adv A{
R: [roll adv A]
result: [calc R mod M vs T ef E]
}
MOD: 0
TARGET: 10
EFFECT: 2
ADV: 0
output [roll MOD vs TARGET ef EFFECT adv ADV]

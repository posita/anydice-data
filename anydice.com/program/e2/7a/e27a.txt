function: interpret ROLL:s MOD:n {
  HITS: 0
  S: [sort ROLL]
  MODT: MOD
  loop RES over {1..#(ROLL)} {
    if RES@ROLL = 6 { HITS: HITS + 1 }
    else { if RES@ROLL + MODT > 5 {
      HITS: HITS + 1
      MODT: MODT - (6 - RES@ROLL)
    } }
  }
  result: HITS
}

loop MOD over {0..10} {
  output [interpret 6d6 MOD] named "Hits for 6d6 + [MOD]"
}

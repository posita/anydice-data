function : replacenum SEQ:s with NUM:n at POS:n {
    RESULTSEQ : {}
    loop I over {1 .. #SEQ} {
        if I = POS { RESULTSEQ : {RESULTSEQ, NUM}}
        else { RESULTSEQ : {RESULTSEQ, I@SEQ}}
    }
    result : RESULTSEQ 
}

function : interpret ROLL:s NEGATEROLL:s {
    ROLLB : ROLL
    NEGATEROLLB : NEGATEROLL
    ROLLRESULT : 0
    loop I over {1 .. #ROLLB} {
        loop J over {1 .. #NEGATEROLLB} {
            if I@ROLLB = J@NEGATEROLLB{ 
                ROLLB : [replacenum ROLLB with 0 at I]
                NEGATEROLLB : [replacenum NEGATEROLLB with 0 at J]
            }
        }
    }
    result : ROLLB
}

function: interpret ROLL:s {
  DHITS : 0
  SHITS : 0
  DMISS : 0
  SMISS : 0
  HITS : 0
  RROLL : [sort ROLL]
  loop I over {1 .. #RROLL} {
    if I@RROLL = 1 {DHITS : DHITS + 1}
    if I@RROLL < 4 {SHITS : SHITS + 1}
    if I@RROLL > 4 {SMISS : SMISS + 1}
    if I@RROLL = 6 {DMISS : DMISS + 1}
  }
  HITS : ([highest of (DHITS - DMISS) and 0]) + SHITS - SMISS
  result: HITS
}

loop I over {1 .. 10} {output [interpret Id6] named "[I]d6"}

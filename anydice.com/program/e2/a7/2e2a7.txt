function: models MODELS:n wounds WOUNDS:n dmg DMG:n {
    HP: MODELS*WOUNDS
    COUNT: 0
    loop H over {1..HP} {
        if H + HP - DMG > H {
            HP: HP - DMG
            COUNT: COUNT + 1
        }
    }
    result: COUNT
}


output [models 2 wounds 6 dmg d6] named "rolls"

function: models MODELS:n wounds WOUNDS:n damage DMG:n {
    HP: MODELS*WOUNDS
    COUNT: 0
    loop D over {1..HP} {
        if HP - D*DMG > 0 {
            COUNT: COUNT + 1
        }
    }
    result: COUNT
}

output [models 2 wounds 6 damage d6] named "rolls"

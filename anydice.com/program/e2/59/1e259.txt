function: dice ROLL:n ac AC:n attack BONUS:n{
    if ROLL = 20 { result: 0 }
    if ROLL + BONUS >= AC {
        if ROLL + BONUS < AC + 5{
            result: 1
        }
    }
    result: 0 
}

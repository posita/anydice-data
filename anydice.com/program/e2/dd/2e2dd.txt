function: count pbta with ROLLS:n {
  if ROLLS >= 10 {
    result: 2
  }
  if ROLLS >= 7 {
    result: 1
  }
  result: 0
}
  
function: count at least N:n in ROLLS:s {
  COUNT: 0
  loop X over ROLLS {
    if X >= N {
      COUNT: COUNT + 1
    }
  }
  result: COUNT
}

loop DC over {6..18} {
  MOD: -((DC - 13) / 2)
  output [count pbta with 2d6+MOD] named "PbtA [MOD]"
  output [count at least DC in 2d20] named "d20s DC [DC]"
}

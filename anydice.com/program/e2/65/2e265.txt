function: ROLL:s tn TARG:n { result: (ROLL = TARG) }

function: N:n sets of VALS:s in SEQ:s {
  SETS: 0
  loop I over VALS {
    if [count I in SEQ] = N {
      SETS: SETS + 1
    }
  }
  result: SETS
}

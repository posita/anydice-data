  result: 0

function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  result: -1       \VAL_X < VAL_Y\
}

\Distribute the dices adaptation between Attack Roll & Save Roll\
function: compare potentials ATT_VAL:n DEF_VAL:n {
 BASE_DICES: 9
 MAX_DICES: 15
 MIN_DICES: 3

 VAL: ATT_VAL-DEF_VAL                \VAL = difference between Att Potential & Def Potential\
 ATT_BONUS: 0
 DEF_BONUS: 0
 if (VAL>=7) { result: {16, 0} }       \if Att is too high, then there is no Def\
 if (VAL<=-7) { result: {0, 16} }      \if Def is too high, then there is no Att\

 ATT_BONUS: VAL
 DEF_BONUS: -VAL
 ATT_DICES: BASE_DICES + ATT_BONUS
 DEF_DICES: BASE_DICES + DEF_BONUS
 result: {ATT_DICES, DEF_DICES}
}

function: compare potentials bis ATT_VAL:n DEF_VAL:n {
 BASE_DICES: 9
 MAX_DICES: 15
 MIN_DICES: 3

 VAL: ATT_VAL-DEF_VAL                \VAL = difference between Att Potential & Def Potential\
 ATT_BONUS: 0
 DEF_BONUS: 0
 if (VAL>=13) { result: {16, 0} }       \if Att is too high, then there is no Def\
 if (VAL<=-13) { result: {0, 16} }      \if Def is too high, then there is no Att\

 if (VAL>=0 & VAL<13) {
  if VAL>6 {
    ATT_BONUS: 6
    DEF_BONUS: -VAL+6
  }else {
    ATT_BONUS: VAL
  }
 }
 if (VAL<0 & VAL>-13) {
  if VAL<-6 {
    ATT_BONUS: VAL+6
    DEF_BONUS: 6
  }else {
    DEF_BONUS: -VAL
  }
 }

 ATT_DICES: BASE_DICES + ATT_BONUS
 DEF_DICES: BASE_DICES + DEF_BONUS
 result: {ATT_DICES, DEF_DICES}
}

\compare potentials : test A : Att-Def -> +X Att & -X Def\
\compare potentials bis : test B : Att-Def -> +X Att then -X Def if surplus\

\************************** Subsets Search Functions ****************************\

function: find best sets ATT:s DEF:s {
 ATT_SET: [best subset ATT]
 DEF_SET: [best subset DEF]
 result: [compare sets ATT_SET DEF_SET]
}

function: best subset ROLL:s {       \Validated : best specialized for d6s : max 20d6\
 SIXES:  ROLL=6                      \test the value on each element of the Sequence parameter : optimized\
 FIVES:  ROLL=5
 FOURS:  ROLL=4
 THREES: ROLL=3
 TWOS:   ROLL=2
 ONES:   ROLL=1

 \check if a set size is more than half of the roll\
 SIZE: (1+#ROLL)/2
 if (SIXES>=SIZE){result: 6} 
 if (FIVES>=SIZE){result: 5}
 if (FOURS>=SIZE){result: 4}
 if (THREES>=SIZE){result: 3}
 if (TWOS>=SIZE){result: 2}
 if (ONES>=SIZE){result: 1}

 \compare the sets to find the more numerous one, choosing the higher if equal\
 if (SIXES>0) & (SIXES>=FIVES) & (SIXES>=FOURS) & (SIXES>=THREES) & (SIXES>=TWOS) & (SIXES>=ONES) { result: 6}
 if (FIVES>0) & (FIVES>=FOURS) & (FIVES>=THREES) & (FIVES>=TWOS) & (FIVES>=ONES) { result: 5}
 if (FOURS>0) & (FOURS>=THREES) & (FOURS>=TWOS) & (FOURS>=ONES) { result: 4}
 if (THREES>0) & (THREES>=TWOS) & (THREES>=ONES) { result: 3}
 if (TWOS>0) & (TWOS>=ONES) { result: 2}
 if (ONES>0) { result: 1}
 result: 0
}

function: compare sets ATT_SET:n DEF_SET:n {    \transform Subsets into Damages\
 if (ATT_SET<=3) {result: 1}
 if (ATT_SET<=5) {
  if (DEF_SET<=3) {result: 3}

 }
 result: 0
}

\************************** Sequence Functions ****************************\

function: phaseone ATT:s DEF:s {                     \find the environment modifiers\
  SIZE_MODS: [compare bodies 13@ATT with 13@DEF]     \compare the nbr of bodies\
  VELOCITY_MODS: [compare 12@DEF with 11@ATT]        \compare the velocity for the dodge capacity\

  ATT_DICES: (1@ATT +SIZE_MODS)                      \modifier to Hit-pool\
  DEF_DICES: (6@DEF +VELOCITY_MODS)                  \modifier to of Save-pool\

  ATT_NEWTARGET: 2@ATT+5@ATT                         \apply target-modifiers for Hit\
  DEF_NEWTARGET: 7@ATT+10@ATT                        \apply target-modifiers for Save\

  BASE_DICES: [compare potentials bis 1@ATT 6@DEF]
  ATT_BASE_DICES: 1@BASE_DICES                         \base number of hit-dices\
  ATT_MODS_DICES: ATT_BASE_DICES+SIZE_MODS              \final number of added hit-dices\

  DEF_BASE_DICES: 2@BASE_DICES                         \base number of save-dices\
  DEF_MODS_DICES: DEF_BASE_DICES+VELOCITY_MODS          \final number of added save-dices\

  result: [hit pass ATT_MODS_DICESd6 DEF_MODS_DICESd6 ATT_NEWTARGET DEF_NEWTARGET]
}

function: hit pass ATT_ROLL:s DEF_ROLL:s ATT_TAR:n DEF_TAR:n {
 ATT_SUCCESSES: ATT_ROLL>=ATT_TAR
 DEF_SUCCESSES: DEF_ROLL>=DEF_TAR
 RES: ATT_SUCCESSES - DEF_SUCCESSES
 if (RES>0){result: [find best sets ATT_ROLL DEF_ROLL]}
 result: 0
}

function: hit A:n for AD:s with AB:n and DB:n against AC:n{
  if A = 20 {result: 2d(AD) + DB}
  if A + AB >= AC & A != 1 {result: 1d(AD) + DB}
  result: 0
}


loop TOHIT over {2..20}{
output [hit 1d20 for 1d8 with 0 and 10+1 against TOHIT] named "DMG bonus, need [TOHIT] to hit"
output [hit 1d20 for 1d8 with 1 and 10+0 against TOHIT] named "ATK bonus, need [TOHIT] to hit"
}

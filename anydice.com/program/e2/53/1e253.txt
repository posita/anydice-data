function: ROLL:s boon X:n {
  Y: 7 - X
  Z: 0
  loop A over ROLL {
    Z: Z + A
  }
  B: (ROLL=X) - (ROLL=Y)
  if ((Z>6)&(B=2)) {
    result: 5
  }
  if ((Z>6)&(B=1)) {
    result: 4
  }
  if ((Z>6)&(B=0)) {
    result: 3
  }
  if ((Z>6)&(B=-1)) {
    result: 2
  }
  if ((Z>6)&(B=-2)) {
    result: 1
  }
  if ((Z<7)&(B=2)) {
    result: -1
  }
  if ((Z<7)&(B=1)) {
    result: -2
  }
  if ((Z<7)&(B=0)) {
    result: -3
  }
  if ((Z<7)&(B=-1)) {
    result: -4
  }
  if ((Z<7)&(B=-2)) {
    result: -5
  }
}
loop N over {2..6} {
  output [2d6 boon N]
}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: highest N:n of A:s B:s {
    result: {1..N}@[sort {A, B}]
}

X: [reroll 1 on d10 to d10]
Y: 3d10

output [highest 2 of 1d10 X] named "reroll 1's"
output 2d10 named "2d10"
output [highest 2 of 3d10] named "3d10"

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + CHA+ PROFICIENCY>= AC { result: DAMAGE }
 result: 0
}

ENEMYAC: d{18}

CHA: 3
PROFICIENCY: 2
CRITROLL: 19
ATTACKTIMES: 2
DAMAGE: 1d10 + CHA
CRITDAMAGE: 2d10 + CHA
output ATTACKTIMESd[attackroll 1@2d20 + CHA + PROFICIENCY vs ENEMYAC] named "Longsword +3 proficiency +5 CHA modifier"

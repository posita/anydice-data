ATK: 0
NORM: 1d20
ADV: [highest 1 of 2d20]
DISADV: [lowest 1 of 2d20]

loop AC over {1..20}{
    output [{AC..AC+4} contains NORM   + ATK] named "Normal vs AC [AC]"
    output [{AC..AC+4} contains ADV    + ATK] named "Advant vs AC [AC]"
    output [{AC..AC+4} contains DISADV + ATK] named "Disadv vs AC [AC]"
    output 0
}

function: hit on HIT:n in ROLL:s {
  HITS: 0
  FAILS: -1
  loop RES over {1..#ROLL} {
    if RES@ROLL >= HIT { HITS: HITS + 1 }
    else { FAILS: FAILS + 1 }
  }
  if FAILS > 0 {
    BPOOL: FAILS
    loop RESTWO over {1..#ROLL} {
      if RESTWO@ROLL < HIT {
        if (RESTWO@ROLL + BPOOL) >= HIT {
		  HITS: HITS + 1
		  BPOOL: BPOOL - (HIT - RESTWO@ROLL)
		}
      }
    }
  }
  result: HITS
}

function: scale of X:n plus Y:n with difficulty D:n {
  if X = 100 {
    result: 50
  } else if X = 1 {
    result: -50
  } else if X + Y >= D {
    if X + Y >= D + 85 {
      result: 50
    } else if X + Y >= D + 35 {
      result: 25
    } else if X + Y >= D + 10 {
      result: 10
    } else {
      result: 1
    }

  } else {
    if X + Y <= D - 85 {
      result: -50
    } else if X + Y <= D - 35 {
      result: -25
    } else if X + Y <= D - 10 {
      result: -10
    } else {
      result: -1
    }
  }
}

function: roll a A:n b B:n c C:n x X:n y Y:n {
AO: 0 \attack one\
AT: 0 \attack two\
DO: 0 \defence one\
DT: 0 \defence two\

\attack one is the highest of all attacks\
if A>AO {AO:A}
if B>AO {AO:B}
if C>AO {AO:C}

\attack two is the middle of each attack\
if (C>=A)*(A>=B) {AT:A}
if (A>=B)*(B>=C) {AT:B}
if (B>=C)*(C>=A) {AT:C}
if (B>=A)*(A>=C) {AT:A}
if (C>=B)*(B>=A) {AT:B}
if (A>=C)*(C>=B) {AT:C}

\defense one is the highest of defense\
if X>DO {DO:X}
if Y>DO {DO:Y}

\defense two is the lowest of defense\
if X<=Y {DT:X}
if Y<=X {DT:Y}

result: (AO>DO)+(AT>DT)
}

output [roll a 1d6 b 1d6 c 1d6 x 1d6 y 1d6]

output (1@3d6>1@2d6) + (2@3d6>2@2d6)
\output 1@2d6\

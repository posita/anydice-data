function: ROLL:s counting {
   if 2@ROLL >= B { result: 2 }
   if 1@ROLL >= B { result: 1 }
   result: 0}
loop B over {6}{loop C over {6}{loop A over {1..5}{output [AdC counting]named "[A]d[C]@[B]"}}}
function: ROLL:s countingblades {
   if 1@ROLL >= 6 { result: 2 }
   if 1@ROLL >= 4 { result: 1 }
   result: 0}
loop C over {6}{loop A over {1..5}{output [AdC countingblades]named "[A]d[C]"}}
function: ROLL:s countingpbta {
   if 1@ROLL >= 10 { result: 2 }
   if 1@ROLL >= 7 { result: 1 }
   result: 0}
loop C over {6}{loop A over {-1..3}{output [(A+2dC) countingpbta]named "2d6+[A]"}}

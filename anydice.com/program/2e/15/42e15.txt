function: nwod R:n again N:n {
 if N >= R { result: 1 + [nwod R again d10] }
 result: N >= 8
}

NWOD: [nwod 10 again d10]

function: znt S:n R:n again N:n {
 if N >= R { result: 1 + [znt S R again d6] }
 result: N >= S
}

ZNT: [znt 5 7 again d6]

loop N over {3..5} {
 output [lowest of 10 and NdNWOD] named "[N]dW"
 output [lowest of 10 and NdZNT] named "[N]dZ"
}

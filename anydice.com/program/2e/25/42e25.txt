\ Define the target and tag threshold. Change these 2 number as needed. \
TARGET: 6
TAG: 1

function: evaluate ROLL:n {
if ROLL <= TAG { result: 2 } \ Below or equal to Tag gives 2 success \
if ROLL <= TARGET { result: 1 } \ Below target and above tag gives 1 success \
if ROLL = 20 { result: -1 } 
result: 0 \ Anything else gives 0 successes \
}

\ Create the custom die based on the function \
CUSTOM_DIE: [evaluate d20]

function: combine negatives VAL:n {
   if VAL <= 0 { result: 0 }
   result: VAL
}

\ Roll X of these custom dice and output the result where X is your number of dice \
output 3dCUSTOM_DIE named "Successes on 3d20"
output [combine negatives 3dCUSTOM_DIE] named "Total Successes"

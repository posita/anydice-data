\ 0 is a miss, 1 is a weak hit, 2 is a strong hit \

function: progress PROG:n {
  result: 2d(PROG > d10)
}

function: action AROLL:n AMOD:n MOM:n {
  if MOM < 0 & AROLL = -MOM {AROLL: 0}
  result: 2d([highest of [lowest of AROLL + AMOD and 20] and MOM] > d20)
}

\ output [progress 5] named "Progress 5" \

loop X over {-5..20} {
  output [action 1d6 X 0] named "Action 1d6+[X] Momentum [0]"
}

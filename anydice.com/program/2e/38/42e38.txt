function: ROLL:s has doubles {
  loop R over {1..#ROLL} {
   if [count R@ROLL in ROLL] > 1 {
     result: 1
   }
  }
 result: 0
}

function: if CONDITION:n then A else B {
  if CONDITION { result: A } else { result: B }
}


function: tries until double on DIE:d {
  SIZE: [maximum of DIE]
  TRIES: SIZE + 1

  loop N over {2..6} {
    TRIES: [lowest of TRIES and [if [NdDIE has doubles] then N else SIZE + 1]]
  }
    
  result: TRIES
} 

function: tries until max on DIE:d{
  TRIES: 10

  loop N over {1..9} {
    TRIES: [lowest of TRIES and [if [NdDIE contains [maximum of DIE]] then N else 10]]
  }
    
  result: TRIES
}

output [tries until double on d6] named "tries to get a double on d6"
output [tries until max on d6] named "tries to get a 6 on d6"
output [tries until double on d4] named "tries to get a double on d4"

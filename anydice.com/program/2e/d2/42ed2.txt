MAP: {-1,0,0,1,1,2}
NONE: 0
EASY: 1
HARD: 2

function: if CONDITION:n then A else B {
  if CONDITION { result: A } else { result: B }
}

function: roll N:n MODE:n {
  if MODE = 1 {
    result: [highest N of (N+1)dMAP]
  }
  if MODE = 2 {
    result: [lowest N of (N+1)dMAP]
  }
  result: NdMAP
}

function: test N:n TN:n MODE:n {
  result: [roll N MODE] >= TN
}

loop N over {1..5}{
  output [test N 1 NONE] named "Roll [N] vs 1 Straight"
  output [test N 1 EASY] named "Roll [N] vs 1 Ease"
  output [test N 1 HARD] named "Roll [N] vs 1 Difficult"
}

PENALTYNET: 0
RPENALTY: 1d20+1-PENALTYNET
R: 1d20+1
AC: 18

function: doubleroll ROLL:s {
  TOTAL: 0
  DOUBLEFOUND: 0
  loop X over {ROLL} {
    if [count X in ROLL] > 1 {
      TOTAL: TOTAL + X
      DOUBLEFOUND: 1
    }
  }
  if DOUBLEFOUND = 0 {
  TOTAL: [highest 1 of 2d4]
  }
  result: TOTAL
}


output (RPENALTY >= AC) * [doubleroll 2d4]
output (R >= AC) * 1d4

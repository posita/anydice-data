
function: attacks of ATK:n damage on HP:n hit points  {
  R: [HP mod ATK]
  if R > 0 {
    result: HP/ATK + 1
  }
  else {
    result: HP/ATK
  }
}

function: A:n to five {
  R: [A mod 5]
  if R < 3 { result: A-R }
  else { result: A/5*5+5 }
}

function: A:n mod N:n {
  result: A-A/N*N
}

output [attacks of [1d8+6 to five] damage on [44 to five] hit points]

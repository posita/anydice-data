function: count BAD:s and GOOD:s in SEQUENCE:s {
 COUNT: 0
 loop P over {1..#BAD} {
  COUNT: COUNT - (P@BAD = SEQUENCE)
 }
 loop R over {1..#GOOD} {
  COUNT: COUNT + (R@GOOD = SEQUENCE)
 }
 result: COUNT
}

output [count 1 and {8..10} in 1d10] named "Guan"
output [count 1 and {8..10} in 6d10] named "Kenichi"

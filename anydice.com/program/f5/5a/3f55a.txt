function: ROLL:s has doubles {
  loop R over {1..#ROLL} {
   if [count R@ROLL in ROLL] > 1 {
     result: 1
   }
  }
 result: 0
}

function: if CONDITION:n then A else B {
  if CONDITION { result: A } else { result: B }
}


function: tries until double {
  TRIES: 7

  loop N over {2..6} {
    TRIES: [lowest of TRIES and [if [Nd6 has doubles] then N else 7]]
  }
    
  result: TRIES
} 

function: tries until six {
  TRIES: 10

  loop N over {1..9} {
    TRIES: [lowest of TRIES and [if [Nd6 contains 6] then N else 10]]
  }
    
  result: TRIES
} 

output [tries until double] named "tries to get a double"
output [tries until six] named "tries to get a 6"

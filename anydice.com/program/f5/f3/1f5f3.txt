SIZE: 8

BASE_ROLL: 1d20
BASE_TARGET: 10

MOD_DICE: {d6}
MOD_TARGET: 2

loop N over [reverse {1..SIZE}] {
output BASE_ROLL > BASE_TARGET | [lowest 1 of NdMOD_DICE] <= MOD_TARGET named "STAT -[N]"
}
output BASE_ROLL > BASE_TARGET named "STAT"

loop N over {1..SIZE} {
output BASE_ROLL > BASE_TARGET & [lowest 1 of NdMOD_DICE] > MOD_TARGET named "STAT +[N]"
}

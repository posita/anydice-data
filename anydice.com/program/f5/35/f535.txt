function: cr OCR:s skill OSKILL:n {
CR: OCR
SKILL: OSKILL
SIXES: [count 6 in CR]
FIVES: [count 5 in CR]
FOURS: [count 4 in CR]
ONES:  [count 1 in CR]

loop X over {1..FIVES}{
if SKILL > 0{
SKILL: SKILL - 1
CR: CR - {5}
CR: CR + 1d6
}
}

loop X over {1..ONES}{
if SKILL > 0{
SKILL: SKILL - 1
CR: CR - {1}
CR: CR + 1d6
}
}

SIXES: [count 6 in CR]
FIVES: [count 5 in CR]
FOURS: [count 4 in CR]
ONES:  [count 1 in CR]

CROSSBONES: FOURS + FIVES
FORTUNE: SKILL + SIXES - CROSSBONES
SKULLS: ONES + FIVES

result: (SKULLS > 0)
}

output [cr 5d6 skill 0]
output [cr 5d6 skill 1]
output [cr 5d6 skill 2]
output [cr 5d6 skill 3]

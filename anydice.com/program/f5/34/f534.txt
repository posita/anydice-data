function: cr CR:s skill OSKILL:n {

SKILL: OSKILL
FORTUNES: [count 6 in CR]
FIVES: [count 5 in CR]
FOURS: [count 4 in CR]
ONES: [count 1 in CR]

T: FIVES
loop X over {1..T} {
if FORTUNES > 1 {
FORTUNES: FORTUNES - 1
FIVES: FIVES - 1
}
}

T: ONES
loop X over {1..T} {
if FORTUNES > 1 {
FORTUNES: FORTUNES - 1
ONES: ONES - 1
}
}

T: FOURS
loop X over {1..T} {
if FORTUNES > 1 {
FORTUNES: FORTUNES - 1
FOURS: FOURS - 1
}
}

T: FIVES
loop X over {1..T} {
if SKILL > 2 {
SKILL: SKILL - 2
FIVES: FIVES - 1
}
}

T: ONES
loop X over {1..T} {
if SKILL > 1 {
SKILL: SKILL - 1
ONES: ONES - 1
}
}

T: FOURS
loop X over {1..T} {
if SKILL > 1 {
SKILL: SKILL - 1
FOURS: FOURS - 1
}
}

SKULLS: FIVES + ONES
CROSSBONES: FOURS + FIVES

result: SKULLS = 0
}

output [cr 1d6 skill 2]
output [cr 2d6 skill 2]
output [cr 3d6 skill 2]
output [cr 4d6 skill 2]
output [cr 5d6 skill 2]
output [cr 6d6 skill 2]
output [cr 7d6 skill 2]
output [cr 8d6 skill 2]
output [cr 9d6 skill 2]

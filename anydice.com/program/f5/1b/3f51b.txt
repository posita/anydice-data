MISS: 0
HIT: 1
CRIT: 2

function: tohit ROLL:n plus ATK:n vs DEF:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRIT }
 if ROLL + ATK >= DEF { result: HIT }
 result: MISS
}

function: damage DICE:d plus MOD:n which TOHIT:n {
 if TOHIT = 0 {result: MISS}
 DICE: (DICE * TOHIT) + MOD
 result: DICE
}

output IFHIT named "Normal tohit"

function: reroll NEWDIE:d for ORIGDIE:n { \ Great Weapon Fighting damage die rerolls on 1 or 2 \
	if ORIGDIE <= 2 { result: NEWDIE }
	result: ORIGDIE
}

output 1d12 named "Greataxe no GWF"
output [reroll 1d12 for 1d12] named "Greataxe with GWF"
output 2d6 named "Greataxe no GWF"
output [reroll 1d6 for 1d6]+[reroll 1d6 for 1d6] named "Greataxe with GWF"

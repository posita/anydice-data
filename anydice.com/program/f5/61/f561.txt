AC: 18 - 6
DC: 16 - 6
PATH: 1d20 + 6
EATH: 1d20 + 4
ATKROLL: 1d20
ADVROLL: [highest 1 of 2d20]
BLESS: 1d4

ADVROLLB: ADVROLL + BLESS
ATKROLLB: ATKROLL + BLESS

function: contest A:n vs B:n then C or D {
  if (A > B) {
    result: C
  }
  result: D
}
DISROLL: [lowest 1 of 2d20]
DISTOHIT: DISROLL >= AC
DISTOHITB: DISROLL + BLESS >= AC
ADVTOHIT: ADVROLL >= AC
ATKTOHIT: ATKROLL >= AC
ATKTOHITB: ATKROLL + BLESS >= AC
ADVTOHITB: ADVROLL + BLESS >= AC
PAPR: 5
HDMG:1d8+4
PDMG:1d6+3
LDMG:2d8
SDMG:2d8+4

output PAPRd(ADVTOHIT*PDMG) +ADVTOHIT*HDMG+ 1d(DISTOHIT*LDMG) named "prone group dpr"

output PAPRd(ADVTOHITB*PDMG) +ADVTOHITB*HDMG+ 1d(DISTOHITB*LDMG) named "prone group dpr blessed"

output PAPRd(ATKTOHIT*PDMG) + ATKTOHIT*HDMG  + ATKTOHIT*LDMG named "group dpr"

output PAPRd(ATKTOHIT*PDMG) + ATKTOHIT*HDMG  + ATKTOHIT*SDMG+ ATKTOHIT*LDMG named "group dpr SW"

output PAPRd(ATKTOHITB*PDMG) + ATKTOHITB*HDMG  + ATKTOHIT*SDMG+ ATKTOHITB*LDMG named "group dpr SW blessed"

output PAPRd(ATKTOHITB*PDMG) + ATKTOHITB*HDMG + ATKTOHITB*LDMG named "group dpr blessed"

output ATKROLL >= AC named "roll on AC"
output ATKROLLB >= AC named "bless on AC"
output [contest PATH vs EATH then ADVTOHIT or ATKTOHIT] named "contest then advantage"
output [contest PATH vs EATH then ADVTOHITB or ATKTOHITB] named "contest then advantage + bless"

\

function: hits CC:n in SEQUENCE:s against ARM {
 HITS: [rawhits CC in SEQUENCE]
 result: [unsaved ARM in HITSd6]
}

function: rawhits CC:n in SEQUENCE:s {
 result:(CC >= SEQUENCE)
}

function: unsaved ARM:n in SEQUENCE:s {
 result:(ARM < SEQUENCE)
}

function: newhits CC:n in SEQUENCE:s against ARM {
 result: (CC = SEQUENCE) + (CC >= SEQUENCE) - (ARM >= SEQUENCE)
}

ATTACKERS: 7
ACC: 4
DARM: 2

output [ hits ACC in ATTACKERSd6 against DARM ]
output [ newhits ACC in ATTACKERSd6 against DARM ]

function: attack ROLL:n plus PROF:n plus BONUS:n vs AC:n for DMG:d {
  if ROLL = 20 {
    result: 2d6 + BONUS
  }
  if ROLL + PROF + BONUS >= AC {
    result: 1d6 + BONUS
  }
  result: 0
}

function: sneak ROLL:n plus PROF:n plus BONUS:n vs AC:n {
  if ROLL = 20 {
    result: 6d6 + [sword ROLL plus PROF plus BONUS vs AC]
  }
  if ROLL + PROF + BONUS >= AC {
   result: 3d6 + [sword ROLL plus PROF plus BONUS vs AC]
  }
  result: 0
}

function: twf ROLLR:n ROLLL:n plus PROF:n plus BONUS:n vs AC:n {
  result: [sword ROLLR plus PROF plus BONUS vs AC] + [sword ROLLL plus PROF plus BONUS vs AC]
}

function: twf sneak ROLLR:n ROLLL:n plus PROF:n plus BONUS:n vs AC:n {
  if ROLLR = 20 {
    result: [sneak ROLLR plus PROF plus BONUS vs AC] + [sword ROLLL plus PROF plus BONUS vs AC]
  }
  if ROLLR + PROF + BONUS >= AC {
    result: [sneak ROLLR plus PROF plus BONUS vs AC] + [sword ROLLL plus PROF plus BONUS vs AC]
  }
  result: [sneak ROLLL plus PROF plus BONUS vs AC]
}

output [sword 1d20 plus 3 plus 4 vs 15] named "shortsword"
output [twf 1d20 1d20 plus 3 plus 4 vs 15] named "twf"
output [sneak 1d20 plus 3 plus 4 vs 15] named "sneak attack"
output [twf sneak 1d20 1d20 plus 3 plus 4 vs 15] named "twf sneak attack"

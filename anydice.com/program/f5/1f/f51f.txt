function: cr ROLL:s skill OSKILL:n {
  SKILL: OSKILL
  SIXES: [count 6 in ROLL]
  FIVES: [count 5 in ROLL]
  FOURS: [count 4 in ROLL]
  ONES: [count 1 in ROLL]

  TCROSSBONES: FOURS + FIVES - SIXES
  CROSSBONES: TCROSSBONES 
  loop X over {1..TCROSSBONES} {
    if SKILL >= 1 {
      SKILL: SKILL - 1
      CROSSBONES: CROSSBONES - 1
    }
  }

  TSKULLS: ONES + FIVES
  SKULLS: TSKULLS
  loop X over {1..TSKULLS} {
    if SKILL >= 1 {
      SKILL: SKILL - 1
      SKULLS: SKULLS - 1
    }
  }

  result: (SKULLS > 0) & (CROSSBONES < 0)
}

output [cr 1d6 skill 1]
output [cr 2d6 skill 1]
output [cr 3d6 skill 1]
output [cr 4d6 skill 1]
output [cr 5d6 skill 1]
output [cr 6d6 skill 1]

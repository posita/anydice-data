function: ROLL:n vs AC:n {
 if ROLL = 20 {
  result: [critical d20]
 }
 if ROLL + ATTACK >= AC { result: DAMAGEDICE + DAMAGEBONUS }
 result: 0
}

function: critical ROLL:n {
  result: 2d DAMAGEDICE + DAMAGEBONUS
}

function: ROLL:n with gwm vs AC:n {
 if ROLL = 20 {
  result: [gwmcritical d20]
 }
 if ROLL + ATTACK -5 >= AC { result: DAMAGEDICE + DAMAGEBONUS +10}
 result: 0
}

function: gwmcritical ROLL:n {
  result: 2d DAMAGEDICE + DAMAGEBONUS +10
}

AC: 13
ATTACK: 6
DAMAGEDICE: 2d6
DAMAGEBONUS: 6

output [[highest 1 of 2d20] vs AC] named "Without GWM"

output [[highest 1 of 2d20] with gwm vs AC] named "With GWM"

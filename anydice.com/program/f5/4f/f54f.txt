function: cr OCR:s skill OSKILL:n {
CR: OCR
SKILL: OSKILL

ONES: [count 1 in CR]
FOURS: [count 4 in CR]
FIVES: [count 5 in CR]
SIXES: [count 6 in CR]

CR: {}
loop X over {1..FIVES}{
if SKILL > 0 {
SKILL: SKILL - 1
N: [roll 1d6]
CR: {CR, N}
} else {
CR: {CR, 5}
}
}

ONES: [count 1 in CR]
FOURS: [count 4 in CR]
FIVES: [count 5 in CR]
SIXES: [count 6 in CR]

result: FIVES

}

function: roll N:s { result: N }

output [cr 5d6 skill 0]
output [cr 5d6 skill 1]
output [cr 5d6 skill 2]

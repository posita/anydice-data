AC: 16 - 6
DC: 16 - 6
PATH: 1d20 + 6
EATH: 1d20 + 4
ATKROLL: 1d20
ADVROLL: [highest 1 of 2d20]
BLESS: 1d4

ADVROLLB: ADVROLL + BLESS
ATKROLLB: ATKROLL + BLESS

function: contest A:n vs B:n then C or D vs AC {
  if (A > B) {
    result: C >= AC
  }
  result: D >= AC
}

output ATKROLL >= AC named "roll on AC"
output ATKROLLB >= AC named "bless on AC"
output [contest PATH vs EATH then ADVROLL  or ATKROLL  vs AC] named "contest then advantage"
output [contest PATH vs EATH then ADVROLLB or ATKROLLB vs AC] named "contest then advantage + bless"

\
output ADVROLL >= AC named "advantage on AC"
output ADVROLLB >= AC named "advantage + bless on AC"
\

function: cr ROLL:s skill OSKILL:n {
  SKILL: OSKILL
  SIXES: [count 6 in ROLL]
  FIVES: [count 5 in ROLL]
  FOURS: [count 4 in ROLL]
  ONES: [count 1 in ROLL]

  SKULLS: FIVES
  loop X over {1..SKULLS} {
    if SKILL >= 2 {
      SKILL: SKILL - 2
      FIVES: FIVES - 1
    }
  }

  SKULLS: ONES
  loop X over {1..SKULLS} {
    if SKILL >= 1 {
      SKILL: SKILL - 1
      ONES: ONES - 1
    }
  }

  result: FIVES + ONES
}

output [cr 5d6 skill 0]
output [cr 5d6 skill 1]
output [cr 5d6 skill 2]
output [cr 5d6 skill 3]
output [cr 5d6 skill 4]
output [cr 5d6 skill 5]
output [cr 5d6 skill 6]

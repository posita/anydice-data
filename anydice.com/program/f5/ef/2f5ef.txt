function: roll ROLL:s {
 result: (ROLL >= 4) - (ROLL = 1)
}

function: helper RESULT:n {
  if RESULT < 0 { result: -1 }
  else { result: RESULT }
}

function: check DICE:n BONUS:n  {
  RESULT: [roll DICEd6]
  result: [helper RESULT] + BONUS
}

loop I over {2..12}{
  loop J over {0..2}{
    output [check I J] named "[I]d6+[J]"
  }
}

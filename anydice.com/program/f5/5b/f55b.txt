HIT: 1d20
HITBONUS: 3
DMG: {0..1}

AC: 13
HP: 8

function: hit {
 result: (HIT + HITBONUS)
}

function: chance to hit {
 result: (HIT + HITBONUS) >= AC & HIT != 1
}

function: chance to one shot {
 result: DMG > HP
}

output [chance to hit] named "Chance to hit"
output [chance to one shot] named "Chance to one shot damage"
output [chance to hit] & [chance to one shot] named "Chance to one shot"

function: cr CR:n skill SKILL:n {
  ROLL: CRd6
  SIXES: [count 6 in ROLL]
  FIVES: [count 5 in ROLL]
  ONES: [count 1 in ROLL]

  FORTUNES: [lowest of SIXES and SKILL]
  if [FIVES is greater than FORTUNES] {
    SIXES: SIXES - FORTUNES
    SKILL: SKILL - FORTUNES
    FIVES: FIVES - FORTUNES
    FORTUNES: 0
  }
  else {
    SIXES: SIXES - FIVES
    SKILL: SKILL - FIVES
    FORTUNES: FORTUNES - FIVES
    FIVES: 0
  }
  
  result: ONES + FIVES - SKILL
}

output [cr 6 skill 3]



function: opt D:n R:n  {
   if(D = R) & D <= 3{
      result : D + 1d6
   
   } else {
     
     if(D < R)& D<=3{
       result : R + 1d6
     }else{
      
      if(R <= 3){
      result : D + 1d6
      }else{
       result : R + D
      }

     }
}

}



function: bad D:n R:n  {
   if(D = R) & D <= 3{
      result : D + 1d6
   
   } else {
     
     if(D < R){
       result : R + 1d6
     }else{
      result : D + 1d6
      }
}

}

output 2d6 named "normal distribution"

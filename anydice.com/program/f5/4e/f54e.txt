ERFOLG:{}
PATZER:{}
PATZER_NEU:{}
DIE: d6
loop N over {1..7}{
ERFOLG: ERFOLG + [count {5..6} in DIE]
PATZER: PATZER + [count {1} in DIE]
PATZER_NEU: PATZER_NEU + [count {1..2} in DIE]
}
output ERFOLG named "Erfolge"
output PATZER > N/2 named "Patzer"
output PATZER > N/2 & ERFOLG = 0 named "Kritischer Patzer"
output PATZER_NEU > N/2 named "Patzer 1-2"
output PATZER_NEU > N/2 & ERFOLG = 0 named "Kritischer Patzer 1-2"

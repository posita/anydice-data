function: cr ROLL:s stat OSTAT:n skill OSKILL:n {
  STAT: OSTAT
  SKILL: OSKILL
  SIXES: [count 6 in ROLL]
  FIVES: [count 5 in ROLL]
  FOURS: [count 4 in ROLL]
  ONES: [count 1 in ROLL]

  TFIVES: FIVES
  loop X over {1..TFIVES} {
    if STAT >= 1 {
      STAT: STAT - 1
      FIVES: FIVES - 1
      ONES: ONES + 1
    }
  }
  CROSSBONES: FOURS - (SIXES + STAT) 

  SKULLS: ONES
  loop X over {1..ONES} {
    if SKILL >= 1 {
      SKILL: SKILL - 1
      SKULLS: SKULLS - 1
    }
  }
  SKULLS: SKULLS + FIVES

  result: (SKULLS > 0) & (CROSSBONES > 0)
}

output [cr 5d6 stat 1 skill 4] named "1/4"
output [cr 5d6 stat 1 skill 3] named "1/3"
output [cr 5d6 stat 1 skill 2] named "1/2"
output [cr 5d6 stat 1 skill 1] named "1/1"

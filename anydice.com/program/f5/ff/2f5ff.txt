function: rrone DIE:d {
 MIN: 3
 MAX: 1
 result: [rrone DIE helper]
}

function: rrone N:n helper {
 if N <= MIN & N>= MAX { result: DIE }
 result: N
}
function: with N:n{
 CONT: 1
 loop M over {1..(N-1)}{
  CONT: CONT+[match d6 and d6]
 }
 result: CONT
}

function: match DO:d and DT:d{
 result: [match DO and DT helper]
}
function: match NO:n and NT:n helper{
 if NO=NT {result: 1}
 result: 0
}

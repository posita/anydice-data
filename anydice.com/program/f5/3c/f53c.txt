function: lowest A:n B:n C:n
{
  if (A <= B & A <= C) { result: A }
  if (B <= A & B <= C) { result: B }
  if (C <= A & C <= B) { result: C }
}

function: highest A:n B:n
{
  if (A >= B) { result: A }
  else { result: B }
}

output [lowest 1d6 1d6 1d6] named "lowest"
output [highest 1d6 1d6] named "higher"
output 1d3
output [highest 1d6 1d6] - 1d3 named "Motivator + Demotivator"

function: and A:n B:n {
    if A=1 {if B=1 {result: 1}}
    result: 0
}

function: attack X:n bonus B:n ac A:n {
    R: 0
    if X+B >= 10+A {R: 4}
    else if X+B >= A {R: 3}
    else if X+B <= A-10 {R: 1}
    else if X+B < A {R: 2}
	if [and R!=4 X=20] {R: R+1}
	if [and R!=1 X=1] {R: R-1}
    result: R
}

function: savthr X:n bonus B:n dc A:n {
    R: 0
    if X+B >= 10+A {R: 1}
    else if X+B >= A {R: 2}
    else if X+B <= A-10 {R: 4}
    else if X+B < A {R: 3}
	if [and R!=1 X=20] {R: R-1}
	if [and R!=4 X=1] {R: R+1}
    result: R
}

function: weapon X:n bonus B:n ac A:n {
    X: [attack X bonus B ac A]
    if X = 4 {result: WCRIT}
    if X = 3 {result: WDAM}
    if X = 2 {result: 0}
    if X = 1 {result: 0}
}

function: spell X:n bonus B:n dc A:n {
    X: [savthr X bonus B dc A]
    if X = 4 {result: SCRIT}
    if X = 3 {result: SDAM}
    if X = 2 {result: SHALF}
    if X = 1 {result: 0}
}

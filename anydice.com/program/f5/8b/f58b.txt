\d6 Dice Pool\

FOC:6  \ <---- set your focus\
TN:5   \ <---- set you target number for die success\

function: D:s target T:n focus F:n {
   if (D>=F) {result:2}
   else if (D>=T){result: 1}
   else {result:0}
}

loop S over {1..10} {
  output Sd[d6 target TN focus FOC] named "[S]d6 Pool Target [TN] Focus [FOC]"
}

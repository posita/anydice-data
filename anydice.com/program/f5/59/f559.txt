HIT: 1d20
DMG: 1d6
MOD: 3
PROF: 2
AC: 13
HP: 6

function: hit {
 result: (HIT + MOD + PROF)
}

function: chance to hit {
 result: (HIT + MOD + PROF) >= AC & HIT != 1
}

function: damage {
 result: (DMG + MOD + PROF)
}

function: chance to one shot {
 result: (DMG + MOD + PROF) > HP
}

output [hit]
output [chance to hit]
output [damage]
output [chance to one shot]
output [chance to hit] & [chance to one shot]

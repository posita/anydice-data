function: if CONDITION:n then A else B {
  if CONDITION { result: A } else { result: B }
}

function: dsix DICE:n {
  result: [dsixinner [explode d6] (DICE-1)d6]
}

function: dsixinner WILD:n OTHERS:d {
  if WILD = 1 {
    result: [lowest #OTHERS - 1 of OTHERS]
  }
  else {
    result: WILD + OTHERS
  }
}
set "explode depth" to 10
output 5d6/2d6

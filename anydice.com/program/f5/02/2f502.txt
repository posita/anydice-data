MISS:0
HIT:1
CRITICALHIT:2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

function: advattack {
 result: [damage for [attack [highest  1 of 2d20] vs DEFENSE]]
}

DEFENSE: d{15..22}

ATTACK: 13
HITDAMAGE: 2d6 + 7 + 4
CRITICALDAMAGE: 4d6 + 7 + 4
MISSDAMAGE: 7
output 2d[advattack] named "Greatsword"

ATTACK: 12
HITDAMAGE: 1d6 + 6 + 4
CRITICALDAMAGE: 2d6 + 6 + 4
MISSDAMAGE: 0
output 3d[advattack] named "Shortsword"

ATTAX: 10
TARGETHIT: 4
TARGETSAVE: 6

function: rolldefense SUCCESSES:s{
REMSUCCESSES:SUCCESSES d6 >=TARGETSAVE
result: SUCCESSES-REMSUCCESSES
}

function: rollattack ROLL:s {
 SUCCESSES: ROLL >= TARGETHIT
 if SUCCESSES > 0 { result: [rolldefense(SUCCESSES)]}
 result: 0
}

loop DICE over {1..ATTAX} {
 output [rollattack DICE d6] named "[DICE]d ATTACK"
}

\ System A: Roll Under with Success Counting \
\ Each die that rolls <= (10 + modifier) counts as a success \

function: systema MOD:n {
  result: (2d10 <= 10 + MOD)
}

\ System B: Roll Over with Single Result \
\ Roll 2d10 + modifier vs difficulty \

function: systemb MOD:n vs DIFF:n {
  result: (d10 + d10 + MOD) >= DIFF
}

\ Compare different modifier levels for System A \
output [systema -3] named "System A: Mod -3 (successes)"
output [systema -1] named "System A: Mod -1 (successes)"
output [systema 0] named "System A: Mod 0 (successes)"
output [systema 2] named "System A: Mod +2 (successes)"
output [systema 5] named "System A: Mod +5 (successes)"

\ Compare System B at different difficulties (mod +2) \
output [systemb 2 vs 12] named "System B: Mod +2 vs Diff 12"
output [systemb 2 vs 14] named "System B: Mod +2 vs Diff 14"
output [systemb 2 vs 16] named "System B: Mod +2 vs Diff 16"
output [systemb 2 vs 18] named "System B: Mod +2 vs Diff 18"

\ Direct comparison at roughly equivalent success rates \
output [systema 0] named "System A: 0 modifier"
output [systemb 2 vs 15] named "System B: +2 vs Diff 15"

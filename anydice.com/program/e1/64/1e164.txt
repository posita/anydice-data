function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}

X: [d10 reroll {1} as d10]

Y: [d10 reroll {1..3} as d10]

Z: [d4 reroll {1..2} as d4]

output 1dX named "Flamberge Normal"

output 3d4 named "Greatsword Normal"

output 1dY named "Flamberge GWM"

output 3dZ named "Greatsword GWM"


function: HUNDRED:n plus TEN:n {
 H: 0
 T: 0

 if HUNDRED = 10 & TEN = 10 { result: 100 }

 if HUNDRED = 10 { H: 0 }
 else { H: HUNDRED*10 }

 if TEN = 10 { T: 0 }
 else { T: TEN }

 result: H+T
}


output 2@2d100 named "2d100"
output [2@2d10 plus 2@2d10] named "2d10(h) + 2d10(t)"
output [2@2d10 plus 1d10] named "2d10(h) + 1d10(t)"

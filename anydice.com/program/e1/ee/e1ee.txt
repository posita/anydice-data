function: atk INF:n inf ARM:n arm FT:n ft BOM:n bom {
  result: [count {1} in INF d6] + [count {1,2,3} in (ARM + FT) d6] + [count {1,2,3,4} in BOM d6]
}
function: def INF:n inf ARM:n arm FT:n ft BOM:n bom {
  result: [count {1,2} in (INF + ARM) d6] + [count {1,2,3,4} in FT d6] + [count {1} in BOM d6]
}

function: diff ATK:s DEF:s {
  result: (ATK >= 3) + (DEF - ATK < 0) - (DEF >= 4)
}
ATK: [atk 4 inf 1 arm 3 ft 1 bom]
DEF: [def 6 inf 0 arm 2 ft 1 bom]
output [diff ATK DEF]

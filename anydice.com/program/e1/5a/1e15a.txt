function: hsd FIRSTRES:s plus SECONDRES:s with FIRST:d and SECOND:d {
  ROLL: FIRSTRES + SECONDRES
  if ROLL < 12 {
    if FIRSTRES <= SECONDRES {
      ROLL: ROLL - FIRSTRES
      ROLL: SECONDRES + FIRST
    }
    else {
      ROLL: ROLL - SECONDRES
      ROLL: FIRSTRES + SECOND
    }
  }

  result: ROLL
}

output [hsd d8 plus d10 with d8 and d10] named "Flat Roll w/ Focus or Boon Reroll"

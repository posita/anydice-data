\AD&D Henchmen locating calculator (implementing the rules on page 39 of the DMG)\

\Fill out the variables below\

NUM_TAVERNS:6    \Number of Taverns visited, this calculator assumes that you are spending 50 g.p. at each tavern (word to the wise: never go to more than 6 taverns)\
BOOL_NOTICES:1   \Set to 1 if you have posted notices, 0 otherwise\
BOOL_AGENTS:1    \Set to 1 if you have hired agents, 0 otherwise\
BOOL_CRIER:0     \Set to 1 if you have hired a crier, 0 otherwise\

\You don't need to mess with anything below here\

if NUM_TAVERNS > 0 {
   BOOL_TAVERNS:1
}else {
   BOOL_TAVERNS:0
}

NUM_METHODS:BOOL_TAVERNS+BOOL_NOTICES+BOOL_AGENTS+BOOL_CRIER

if NUM_METHODS = 1 {
   NUM_METHODS:0
}

TAVERNS:(((NUM_TAVERNS*5)d4)-(NUM_TAVERNS^2))
NOTICES:BOOL_NOTICES*(1d31+9)
AGENTS:BOOL_AGENTS*(1d31+19)
CRIER:BOOL_CRIER*(1d10)

output TAVERNS + NOTICES + AGENTS + CRIER - NUM_METHODS*5

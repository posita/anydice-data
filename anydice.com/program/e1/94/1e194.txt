function: adv ROLL:s {
 result: 1@ROLL-#ROLL@ROLL
}

function: dis ROLL:s {
 X: 1@ROLL
 loop N over {1..#ROLL-1}{
  Y: N@ROLL-(N+1)@ROLL
  if(Y<X){X: Y}
 }
 result: X
}

D:4

output [dis 5dD] named "-3"
output [dis 4dD] named "-2"
output [dis 3dD] named "-1"
output [adv 2dD] named "0"
output [adv 3dD] named "+1"
output [adv 4dD] named "+2"
output [adv 5dD] named "+3"

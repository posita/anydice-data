function: rolls DICE:n threshold DC:n{
 SUCCEEDS: 0
 loop N over {1..DICE} {
  ROLL: [explode 1d10]
  SUCCEEDS: SUCCEEDS + [decide ROLL DC]
 }
 result: SUCCEEDS
}
function: decide ROLL:n DC:n {
 if ROLL >= DC {
  result: 1 + (ROLL-DC)/4
 } 
 result: 0
} 
function: judge RESULTS:n {
 if RESULTS >= 2 {
  result: MORE
 } 
 if RESULTS = 1 {
  result: ONLY
 }
 result: FAIL
}
FAIL: 0
ONLY: 1
MORE: 2

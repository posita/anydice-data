function: atleastzero A:n {
 if A < 0 {result: 0}

result: A
}

function: def DEF:n {
 if 1@DEF >= SAMOUNT & 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + SBONUS + ARMOR + MAGICSHIELD)}
 if 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + ARMOR + MAGICSHIELD)}
 if 1@DEF >= SAMOUNT {result: (3@DEF + SBONUS + ARMOR + MAGICSHIELD)}

result: 3@DEF + ARMOR + MAGICSHIELD
}

function: defminus DEF:n {
 if 1@DEF >= SAMOUNT & 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + SBONUS + [atleastzero(ARMOR - MINUSARMOR)] + MAGICSHIELD)}
 if 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + [atleastzero(ARMOR - MINUSARMOR)] + MAGICSHIELD)}
 if 1@DEF >= SAMOUNT {result: (3@DEF + SBONUS + [atleastzero(ARMOR - MINUSARMOR)] + MAGICSHIELD)}

result: 3@DEF + [atleastzero(ARMOR - MINUSARMOR)] + MAGICSHIELD
}

function: TAW hammerschlag N:n D:n P:n {
  if N <= TAW-8 { result: (Dd6+P)*3 }  
 result: 0
}

function: TAW L wuchtschlag N:n D:n P:n {
  if N <= (TAW-L) { result: Dd6+P+L}
 result: 0
}

function: TAW L quali N:n D:n P:n {
  if N <= (TAW-L) { result: Dd6+P+L+((TAW-N)/2)}
 result: 0
}

function: TAW quali N:n D:n P:n {
  if N <= (TAW) { result: Dd6+P+((TAW-N)/2)}
 result: 0
}


TAW: 22+2-2
DICE: 1
PLUS: 4

\
output [TAW hammerschlag 1d20 DICE PLUS] named "Hammerschlag"

loop L over {1 .. 6} {
output [TAW L wuchtschlag 1d20 DICE PLUS] named "[L]er Wuchtschlag"
}
\
output [TAW quali 1d20 DICE PLUS] named "FK mit Quali"

loop L over {1 .. 10} {
output [TAW L quali 1d20 DICE PLUS] named "FK mit Quali + [L]er-Ansage"
}

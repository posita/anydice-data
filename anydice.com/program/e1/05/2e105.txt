\_________________________________________VAIKUTUKSET______________________________________________\

PLUSHIT: 0

BLUED: 2
ARMOR: 0
MAGICSHIELD: 0
LAMOUNT: 0
LBONUS: 0
SAMOUNT: 0
SBONUS: 0

\__________________________________________________________________________________________________\

set "position order" to "lowest first"

\_________________________________VIHOLLISEN PUOLUSTUSHEITTO________________________________________\

\1 = HITTITÃ„HTI, 2 = HITTISALAMA, 3 = KILPI, 4 = GREMLIN, 5 = HITIT, 6 = TUPLASALAMA, 7 = SS\

function: def DEF:n {
 if 1@DEF >= SAMOUNT & 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + SBONUS + ARMOR + MAGICSHIELD)}
 if 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + ARMOR + MAGICSHIELD)}
 if 1@DEF >= SAMOUNT {result: (3@DEF + SBONUS + ARMOR + MAGICSHIELD)}

result: 3@DEF + ARMOR + MAGICSHIELD
}

function: defnoarmor DEF:n {
 if 1@DEF >= SAMOUNT & 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + SBONUS + MAGICSHIELD)}
 if 2@DEF >= LAMOUNT {result: (3@DEF + LBONUS + MAGICSHIELD)}
 if 1@DEF >= SAMOUNT {result: (3@DEF + SBONUS + MAGICSHIELD)}

result: 3@DEF + MAGICSHIELD
}

\_________________________________________________AXE_IGNORE_PRIO____________________________________________________\

function: axedefa AB:n {
AUTOHIT: 2
HIT_MINUS_MS: (1@AB + 2@AB + AUTOHIT + PLUSHIT - MAGICSHIELD)
HIT_MINUS_ARMOR_MS: (1@AB + 2@AB + AUTOHIT + PLUSHIT - ARMOR - MAGICSHIELD)
HIT_AMOUNT: (1@AB + 2@AB + AUTOHIT + PLUSHIT)
L_AMOUNT: 2@AB

 if L_AMOUNT = 1 & HIT_MINUS_MS >= BLUED {result: HIT_AMOUNT - [defnoarmor BLUEDd{1:3,10:2,100:4,1000:1}]}

 if L_AMOUNT = 1 & HIT_MINUS_MS < BLUED & HIT_MINUS_MS >= 0 {result: HIT_AMOUNT - [defnoarmor HIT_MINUS_MSd{1:3,10:2,100:4,1000:1}]}
 
 if L_AMOUNT = 1 & HIT_MINUS_MS < 0 {result: 0}

 if L_AMOUNT = 0 & HIT_MINUS_ARMOR_MS >= BLUED {result: HIT_AMOUNT - [def BLUEDd{1:3,10:2,100:4,1000:1}]}

 if L_AMOUNT = 0 & HIT_MINUS_ARMOR_MS < BLUED & HIT_MINUS_ARMOR_MS >= 0 {result: HIT_AMOUNT - [def HIT_MINUS_ARMOR_MSd{1:3,10:2,100:4,1000:1}]}

 if L_AMOUNT = 0 & HIT_MINUS_ARMOR_MS < 0 {result: 0}

result: 1000
}
          output [axedefa 1d{1:3,10:2,100:4,1000:1}] named "Kirves vastaan puolustus (Ignore Armor Priority):"

HIT_AMOUNT_PLUS: (1@AB + 2@AB + AUTOHIT + PLUSHIT + 1)
HIT_AMOUNT: (1@AB + 2@AB + AUTOHIT + PLUSHIT)

 if L_AMOUNT = 1 & HIT_MINUS_ARMOR_MS_PLUS < 0 {result: 0}

 if L_AMOUNT = 0 & HIT_MINUS_ARMOR_MS < BLUED & HIT_MINUS_ARMOR_MS >= 0 {result: HIT_AMOUNT - [def HIT_MINUS_ARMOR_MSd{1:3,10:2,100:4,1000:1}]}

 if L_AMOUNT = 0 & HIT_MINUS_ARMOR_MS < 0 {result: 0}

result: 1000
}
          output [axedefb 1d{1:3,10:2,100:4,1000:1}] named "Kirves vastaan puolustus (Plus 1 Hit Priority):"

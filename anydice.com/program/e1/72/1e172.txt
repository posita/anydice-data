function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {result: 0}
    if ROLL >= CRIT_RANGE {result: DMG + CRIT_DMG}
    if ROLL + BONUS >= AC {result: DMG}
    result: 0
}

output 1d[attack [highest 1 of 2d20] plus 5+1d4 vs d{15} for 2d6+3+2 crit 2d6 on 20] named "Bless Normal"
output 1d[attack [highest 1 of 2d20] plus 5+1d4-5 vs d{15} for 2d6+3+2+10 crit 2d6 on 20] named "Bless GWM"
output 1d[attack [highest 1 of 2d20] plus 5 vs d{15} for 2d6+3+2 crit 2d6 on 20] named "Normal"
output 1d[attack [highest 1 of 2d20] plus 5-5 vs d{15} for 2d6+3+2+10 crit 2d6 on 20] named "GWM"

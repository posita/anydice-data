YES: 1
NO: 0

\ bonuses config \
BASE_MOVEMENT: 30
MOBILE_FEAT: 10
SQUAT_NIMBLENESS_FEAT: 5

LONGSTRIDER_SPELL: YES
BLADESINGING_FEATURE: YES
ASHARDALONS_STRIDE_LEVEL: 0

\ multipliers config \
DASH_ACTION: YES
DASH_BONUS_ACTION: YES
FELINE_AGILITY: YES
HASTE: YES
HASTE_ACTION_DASH: YES
BOOTS_OF_SPEED: YES
EAGLE_WHISTLE: NO



\ bonuses \
FLAT_MOVEMENT: BASE_MOVEMENT + MOBILE_FEAT + SQUAT_NIMBLENESS_FEAT

if LONGSTRIDER_SPELL {
  FLAT_MOVEMENT: FLAT_MOVEMENT + 10
}

if BLADESINGING_FEATURE {
  FLAT_MOVEMENT: FLAT_MOVEMENT + 10
}

if ASHARDALONS_STRIDE_LEVEL >= 3 {
  FLAT_MOVEMENT: FLAT_MOVEMENT + 20 + (ASHARDALONS_STRIDE_LEVEL - 3) * 5
}



\ multipliers \
MULTIPLIERS_COUNT: DASH_ACTION + DASH_BONUS_ACTION + FELINE_AGILITY + HASTE + HASTE_ACTION_DASH + BOOTS_OF_SPEED + EAGLE_WHISTLE
TOTAL_MOVEMENT_SPEED: FLAT_MOVEMENT

loop N over {1..MULTIPLIERS_COUNT} {
 TOTAL_MOVEMENT_SPEED: TOTAL_MOVEMENT_SPEED * 2
}

output TOTAL_MOVEMENT_SPEED

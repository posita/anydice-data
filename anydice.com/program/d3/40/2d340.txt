\ ----- Critical Damage Dice Function ----- \

function: critical D:n at C add A{
 if D = C { result: D + A }
 result: D
}

function: deadly D:n at C add A{
 if D < C { result: D + A }
 result: D
}

function: brutal D:n at C{
 if D < C { result: C }
 result: D
}

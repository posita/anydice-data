function: stunt DICE:s {
  \ Loop over looking for pairs \
  loop N over {1..6} {
    if [count N in DICE] > 1 {
      \ SUM of dice minus the sum of the pair will be the number
        on the remaining die \
      result: DICE - (2*N)
    }
  }
  result: 0
}

function: stuntandsucceed BONUS:n TARGET:n ROLL:s {
   RESULT: ROLL+BONUS
   if RESULT >= TARGET {
      result: [stunt ROLL] 
   }
   result: -1
}

output [stunt 3d6] named "Stunts"
output [stuntandsucceed 0 10 3d6] named "Succeed With A Stunt"

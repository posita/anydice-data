\ lightning calling spells Balance Calculator \

E_DD: 3d10                            \Electric Damage Dice\
S_DD: 3d6                             \Sonic Damage Dice\
B: 1                                  \Number of Bolts\

E_DA: (E_DD*62/100)+(((38*E_DD)/200)) \Electric Damage Accounting for Accuracy (62% hit chance)\
E_DE: E_DA*10/10                      \Electric damage for estimated 1 affected target(s) on average per bolt\

S_DA: (S_DD*62/100)\+(((38*S_DD)/200))\ \Sonic Damage Accounting for Accuracy (62% hit chance)\
S_DE: S_DA*27/10                      \Sonic damage for estimated 2.7 affected target(s) on average per bolt\

T_DE: (E_DE+S_DE)*B                   \Total damage estimate\

MPC: T_DE*2/3                         \Balanced Mana Point Cost\

output T_DE named "Total average damage estimate"
output MPC named "Balanced Mana Point Cost"

function: bd R:n with O:n reroll RE:s already A:n {
  if (R = RE) & !A {
    result: [bd d6 with O reroll RE already 1]
  }
  
  if R = 6 {
    result: [bd d6 with O + 5 reroll RE already 0]
  }

  result: R + O
}

BD: [bd d6 with 0 reroll {} already 0]
TW: [bd d6 with 0 reroll 1 already 0]

loop N over {1..5} {
  output [lowest of NdBD and 25] named "[N]BD"
  output [lowest of NdTW and 25] named "[N]BD TW"
}

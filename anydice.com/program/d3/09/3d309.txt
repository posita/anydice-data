function : split DICE:d on COIN:n{
if (COIN = 1){result: [highest of DICE and [highest of DICE and DICE]]}
result: [lowest of DICE and [lowest of DICE and DICE]]
}

function: choose FUNC:n for DICE:d{
if FUNC=1 {result: 1@ 3 d (DICE)}
if FUNC=2 {result: 2@ 3 d (DICE)}
if FUNC=3 {result: 3@ 3 d (DICE)}
if FUNC=4 {result: [split DICE on d2]}
if FUNC=5 {result: [explode DICE]}
result: DICE
}

output [choose 5 for d20]
output 3d[choose 5 for d6]
output [choose 5 for 2@3 d (2d12)]

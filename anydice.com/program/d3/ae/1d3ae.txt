\ Given a level, return what "tier" it is in the to-hit table. This is an index into the sequence of numbers \
function: tier level LEVEL:n {
  if LEVEL < 4 {
    result: 1
  }
  else if LEVEL < 7 {
    result: 2
  }
  else if LEVEL < 10 {
    result: 3
  }
  else if LEVEL < 13 {
    result: 4
  }
  else if LEVEL < 16 {
    result: 5
  }
  result: 6
}

\ Given a HD value (hit dice) for a monster, return what "tier" it is in the to-hit table. This is an index into the sequence of numbers \
function: tier hd HD:n plus PLUS:n {
  if HD < 1 {
    result: 1
  }
  else if HD = 1 {
    if PLUS > 0 {
      result: 2
    }
    else {
      result: 1
    }
  }
  else if HD = 2 {
    result: 3
  }
  else if HD = 3 {
    if PLUS > 0 {
      result: 4
    }
    else {
      result: 3
    }
  }
  else if HD = 4 {
    if PLUS > 0 {
      result: 5
    }
    else {
      result: 4
    } 
  }
  else if HD = 5 {
    result: 5
  }
  else if HD = 6 {
    if PLUS > 0 {
      result: 6
    }
    else {
      result: 5
    }   
  }
  else if HD < 9 {
    result: 6
  }
  else if HD < 11 {
    result: 7
  }
  else { 
    result: 8
  }
}

  if AC = 2 {

\ The result of to-hit roll for an attacker of given level attacking a defender with given AC \
function: hit for level LEVEL:n attacking ac AC:n {
  result: d20 >= [tohit for level LEVEL attacking ac AC]
}

\ The result of to-hit roll for a monster of given HD attacking a defender with given AC \
function: hit for hd HD:n plus PLUS:n attacking ac AC:n {
  result: d20 >= [tohit for hd HD plus PLUS attacking ac AC]
}

function: damage of level LEVEL:n attacking ac AC:n {
  result: [hit for level LEVEL attacking ac AC] * 1d6
}

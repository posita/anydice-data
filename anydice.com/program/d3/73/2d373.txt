function: selenebasic ROLL:s DIFF:s MOD:s {
   RESULT_ONE: ROLL + MOD
   if RESULT_ONE >= DIFF {
    result: 1
   } else {
    result:0
   }
}

loop DIFF over {5..12}{
  loop MOD over {0..3}{
    COSA: [selenebasic d8 DIFF MOD]
    COSADOS: [selenebasic d8 DIFF MOD]
    COSAFINAL: COSA + COSADOS
    output COSAFINAL named "Selene Doble Desventaja (2d8) dificultad [DIFF] mod +[MOD]"
  }
}

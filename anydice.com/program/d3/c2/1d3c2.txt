PROF: 3
CHA: 4
CON: 2
EBS: 2
ENEMY_AC: 15
ENEMY_DMG: d10+4
ENEMY_ATK: 3
HP: 37

function: blast against AC {
 ATK: d20
 HIT: ATK+PROF+CHA > AC
 CRIT: ATK >= 20-CUR
 DMG: d10+HEX*d6+CUR*PROF+CHA
 result: (HIT+CRIT) * DMG
}

function: blast X against AC {
 T_DMG: 0
 loop Z over {1..X}{
  T_DMG: T_DMG+ [blast against AC]
 }
 result: T_DMG
}


function: constitution save of VAL {
 result: d20+CON >= VAL
}

function: mirror image x X {
 result: d20 >= 20/(X+1) + 1
}

IMG: 3
HEX: 1
CUR: 1
TOTAL: 0
loop N over {1..7}{
 ALIVE: HP > 0
 DO_DMG: [blast EBS against ENEMY_AC] * ALIVE
 TOTAL: TOTAL + DO_DMG
 output DO_DMG named "Damage roll [N]"
 DO_DEFLECT: [mirror image x IMG]
 E_ATK: d20
 E_HIT: E_ATK + ENEMY_ATK >= 10 + CHA
 E_CRIT: E_ATK = 20
 IMG: IMG - DO_DEFLECT * E_HIT
 E_DMG: !DO_DEFLECT * ENEMY_DMG * (E_HIT + E_CRIT)
 output E_DMG named "Enemy damage roll [N]"
 HP: HP - E_DMG
 HEX: HEX = 1 & (E_DMG = 0 | [constitution save of 10])
}
output HEX named "Hex still active"
output IMG named "Remaining mirror images"
output TOTAL named "Total damage"
output HP named "HP left"
output ALIVE named "Chance alive"

function: hits AC:n B:n
{
   result: [highest of [lowest of [lowest 1 of 1d20] + B - AC + 1 and 1] and 0]
}

function: hitsadv AC:n B:n
{
   result: [highest of [lowest of [highest 1 of 2d20] + B - AC + 1 and 1] and 0]
}

function: hitsdis AC:n B:n
{
   result: [highest of [lowest of [lowest 1 of 2d20] + B - AC + 1 and 1] and 0]
}

output [hits 16 7]
output [hitsdis 20 8]
output [hitsadv 20 8]

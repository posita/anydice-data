
CRIT_THRESHOLD: 20
TARGET_AC: 16
CRITICAL_DICE: 2
BASE_DICE: 1
DISADVANTAGE_ROLL: [lowest 1 of 2d20]
ADVANTAGE_ROLL: [highest 1 of 2d20]
ROLL: 1d20

function: R:n attack{
    if R >= CRIT_THRESHOLD {result: CRITICAL_DICE d DMG + DMG_MOD}
    if (R + HIT_MOD) >= TARGET_AC {result: BASE_DICE d DMG + DMG_MOD}
    if (R + HIT_MOD) < TARGET_AC {result: 0}
}


HIT_MOD: 10
DMG: 1d4 + 5d6
DMG_MOD: 6 

output [ROLL attack]  named "Haybin's Blade"
output [ADVANTAGE_ROLL attack] named "advantage Haybin's Blade"

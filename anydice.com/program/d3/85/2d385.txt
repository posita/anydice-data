PROF : 4
MOD : 4
MAGIC: 1
BLESS: 1d4
DUELING:2
AC: 17

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n and SMITE:d {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG + 2dSMITE
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

HC_TRI      : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d8 + MOD + MAGIC + DUELING+PROF crit 1d8 on 19 and 2d8]
HC_TRI_PA   : [attack [highest 1 of 1d20] plus MOD+MAGIC vs AC for 1d8 + MOD + MAGIC + DUELING+ 3*PROF crit 1d8 on 19 and 2d8]
HC_TRI_BUD  : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d4 + MOD + MAGIC + DUELING+PROF crit 1d4 on 19 and 2d8]

HX_TRI      : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d8 + MOD + MAGIC + DUELING+1d6 crit 1d8+1d6 on 20 and 2d8]
HX_TRI_PA   : [attack [highest 1 of 1d20] plus MOD+MAGIC vs AC for 1d8 + MOD + MAGIC + DUELING+ 2*PROF+1d6 crit 1d8+1d6 on 20 and 2d8]
HX_TRI_BUD  : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d4 + MOD + MAGIC + DUELING+1d6 crit 1d4+1d6 on 20 and 2d8]

SS_TRI      : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d8 + MOD + MAGIC + DUELING+1d8 crit 1d8+1d8 on 20 and 2d8]
SS_TRI_PA   : [attack [highest 1 of 1d20] plus MOD+MAGIC vs AC for 1d8 + MOD + MAGIC + DUELING+ 2*PROF+1d8 crit 1d8+1d8 on 20 and 2d8]
SS_TRI_BUD  : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d4 + MOD + MAGIC + DUELING+1d8 crit 1d4+1d8 on 20 and 2d8]


HC_EB :[attack [highest 1 of 1d20] plus MOD+PROF+BLESS vs AC for 1d10+MOD+PROF crit 1d10 on 19]



output 2dHC_TRI_PA+HC_TRI_BUD     named "10th level Relius with Trident, power attack, bud, Hexblades Curse"

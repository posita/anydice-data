\ Torchdice 2e Probability Calculator\
\ This program circumvents the usual way AnyDice handles dice.                                     \
\ Instead, each die is a function.                                                                 \
\ Usage:                                                                                           \
\ [be X threshold 6 explode 12] will roll X dice at t 6 and e 12.                                  \
\ The dice can be added together, e.g. [be 10 thres 6 expl 12] + [be 5 thres 6 expl 12]            \

set "maximum function depth" to 7 
\do not set this below 4, as wrapper function count against the limit\

function: roll ROLL:n threshold T {
  if ROLL >=T {result: 1}
  result: 0
}

function: rollexploding ROLL:n threshold T explode E{
  if ROLL >= E {result: 1+[rollexploding 1d12 threshold T explode E]}
  result: [roll ROLL threshold T]
}

function: be DICE:n threshold T explode E{
  RES:0
  loop N over {1..DICE} {
    RES:RES+[rollexploding 1d12 threshold T explode E]
  }
  result:RES
}

output [be 50 threshold 6 explode 12]

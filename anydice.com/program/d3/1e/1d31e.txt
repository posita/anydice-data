AURA: 1
BLESS: 1
BEACON: 0
CEREMONY: 1
	
function: roll B C A D {
	E: d20
	if D = 1 {E: [highest 1 of 2d20]}
	if B = 1 {E: E + d4}
	if C = 1 {E: E + d4}
	if A = 1 {E: E + 3}
	
	result: E
}
	
function: did R:n save {
	if R >= 10 {result: 1} 
	result: 0
}
	
SUCCESSES: 0
loop N over {1..5} {
	ROLL: [roll BLESS CEREMONY AURA BEACON]
	SUCCESSES: SUCCESSES + [did ROLL save]
}
	
output SUCCESSES >= 3

MOD: 3

function: darts MODIFIER {
    FIRST: [dartpoints 1d20+MODIFIER]
    SECOND: [dartpoints 1d20+MODIFIER]
    THIRD: [dartpoints 1d20+MODIFIER]

    result: FIRST+SECOND+THIRD
}

function: dartpoints X:s {
    if X >= 20 {
        result: 50
    }
    if X >= 15 {
        result: 30
    }
    if X >= 10 {
        result: 10
    }
    if X >= 5 {
        result: 5
    }
    result: 0
}

function: bottlesdown X:s {
    if X >= 20 {
        result: 6
    }
    if X >= 18 {
        result: 5
    }
    if X >= 16 {
        result: 4
    }
    if X >= 13 {
        result: 3
    }
    if X >= 10 {
        result: 2
    }
    if X >= 5 {
        result: 1
    }
    result: 0
}

function: bottles MODIFIER {
    FIRST: [bottlesdown 1d20+MODIFIER]
    SECOND: [bottlesdown 1d20+MODIFIER]
    THIRD: [bottlesdown 1d20+MODIFIER]

    result: FIRST+SECOND+THIRD
}

output [darts MOD] named "Darts"
output [bottles MOD] named "Bottles"

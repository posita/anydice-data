\\\ Uplift Function \\\

function: uplift ROLL:s with DIE {
 R: 0
 loop P over {1..3}{
  if P@ROLL = 1 { R: R + DIE }
  else { R: R + P@ROLL }
 }
 result: R
}

\\\ Newdie Function \\\

function: newdie { 
    result: d6
}

\\\ Reroll Function \\\

function: rerolldie ROLL:n REROLLS:n {
if ((ROLL<=2)&(REROLLS>0))
{
result: [rerolldie [newdie] REROLLS-1]
}
else
{
result:ROLL
}
}

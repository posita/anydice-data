YES: 1
NO: 0

\ config \
EXTRA_ATTACK: YES
TURNS_IN_COMBAT: 1
WEAPON_DIE: d8
WEAPON_ENCHANT: 2d4
DAMAGE_MODIFIER: 6
ACTION_SURGE: YES
UNLEASH_INCARNATION_ATTACKS: 0
SNEAK_ATTACK: 2d6
DREAD_AMBUSHER: YES
IS_ENEMY_SURPRISED: YES
IS_HUNTER_MARKED: YES
PIERCER_FEAT: NO
POISONER_FEAT: NO
DIVINE_SMITES: 0
DIVINE_SMITES_TOTAL_LEVELS_SPENT:5+5+4+4+4+3+3+3



\ attacks \
ACTIONS: TURNS_IN_COMBAT

if ACTION_SURGE {
  ACTIONS: ACTIONS + 1
}

ATTACKS_PER_ACTION: 1 + EXTRA_ATTACK

if UNLEASH_INCARNATION_ATTACKS > ACTIONS {
  UNLEASH_INCARNATION_ATTACKS: ACTIONS
}

TOTAL_ATTACKS: ATTACKS_PER_ACTION * ACTIONS + UNLEASH_INCARNATION_ATTACKS

if DREAD_AMBUSHER {
  TOTAL_ATTACKS: TOTAL_ATTACKS + 1 + ACTION_SURGE
}



\ damage \
DICE_DAMAGE: (WEAPON_DIE + WEAPON_ENCHANT) * TOTAL_ATTACKS

if DREAD_AMBUSHER {
  DICE_DAMAGE: DICE_DAMAGE + d8 + ACTION_SURGEd8
}

if IS_HUNTER_MARKED {
  DICE_DAMAGE: DICE_DAMAGE + d6 * TOTAL_ATTACKS
}

if POISONER_FEAT {
  DICE_DAMAGE: DICE_DAMAGE + 2d8
}

if DIVINE_SMITES {
  DIVINE_SMITE_DICE: DIVINE_SMITES_TOTAL_LEVELS_SPENT + DIVINE_SMITES
  DICE_DAMAGE: DICE_DAMAGE + DIVINE_SMITE_DICEd8
}

DICE_DAMAGE: DICE_DAMAGE + SNEAK_ATTACK * TURNS_IN_COMBAT

\ TODO this should only double the first turn's damage \
if IS_ENEMY_SURPRISED {
  DICE_DAMAGE: DICE_DAMAGE * 2 
}

if PIERCER_FEAT & IS_ENEMY_SURPRISED {
  DICE_DAMAGE: DICE_DAMAGE + WEAPON_DIE * (ATTACKS_PER_ACTION + ATTACKS_PER_ACTION * ACTION_SURGE * 2)
}

DICE_DAMAGE: DICE_DAMAGE + DAMAGE_MODIFIER * TOTAL_ATTACKS

output TOTAL_ATTACKS named "attacks"
output DICE_DAMAGE / TOTAL_ATTACKS named "damage per attack"
output DICE_DAMAGE named "damage"

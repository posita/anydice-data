AURA: 1
CEREMONY: 1
BEACON: 1

function: roll C A B {
	D: d20
	if B = 1 {D: [highest 1 of 2d20]}
	if C = 1 {D: D + d4}
	if A = 1 {D: D + 3}

	result: D
}

function: did R:n save {
	if R >= 10 {result: 1} 
	result: 0
}

SUCCESSES: 0
loop N over {1..5} {
	ROLL: [roll CEREMONY AURA BEACON]
	SUCCESSES: SUCCESSES + [did ROLL save]
}

output SUCCESSES >= 3

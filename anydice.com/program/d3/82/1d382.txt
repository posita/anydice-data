function: evaluate ROLL:s {
 MAX_FAILURES: 5
 FAILURES: ROLL <= 2
 SUCCESSES: (ROLL >= 7) + (ROLL = 10)

 if FAILURES > MAX_FAILURES { BASE_RESULT: SUCCESSES - MAX_FAILURES }
 else { BASE_RESULT: SUCCESSES - FAILURES }
 
 if BASE_RESULT < 0 { result: 0 }
 else { result: BASE_RESULT }
}

loop DICE over {1..6} {
 output [evaluate DICE d10] named "[DICE]d"
}

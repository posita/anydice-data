  result: ROLL + ATTACK_BONUS >= ACS

  DAMAGE: 0
  loop I over {1..ATTACKS} {
    HIT: 0

    if (ATTACK_ROLL >= CRIT) { 
      DAMAGE: DAMAGE + 2*(WEAPON_DICE + (SUBCLASS_BERSERKER * USE_FRENZY * FRENZY)) + DAMAGE_BONUS
      USE_FRENZY: 0
      HIT: 1
      if BOON_OF_IRRESISTIBLE_OFFENSE & (ATTACK_ROLL = 20) { DAMAGE: DAMAGE + ATTACK_MOD }
    } else {
      if (ATTACK_ROLL + ATTACK_BONUS >= AC) {

        HIT: 1
      } else {
        DAMAGE: DAMAGE + WEAPON_MISS
      }

      if USE_BRUTAL {

        USE_BRUTAL: 0
      } else if USE_CLEAVE {

        USE_CLEAVE: 0

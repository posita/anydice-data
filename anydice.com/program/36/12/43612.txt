HT: 18

function: death roll D:n with ht HT:n {
  if HT-D <= -10 { result: 1 }
  if HT <= 15 & D >= 17 { result: 1}
  if D = 18 { result: 1}
  result: D > HT
}

loop SAVE_COUNT over {1..4} {
  if SAVE_COUNT <= 1 {
    output [death roll 3d6 with ht HT] named "-[SAVE_COUNT]*HP"
  }
  else {
    ROLLS: SAVE_COUNT d [death roll 3d6 with ht HT]
    output (1 @ ROLLS) named "-[SAVE_COUNT]*HP"
  }
}

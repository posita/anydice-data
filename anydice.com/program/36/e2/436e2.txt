function: minimum ROLL:n MIN:n {
  if ROLL < MIN { result: MIN }
  result: ROLL
}

function: bounce ROLL:n LOW:n {
  if ROLL <= LOW { result: ROLL + LOW }
  result: ROLL
}

function: reroll ROLL:n DIE:n FLR:n {
  if ROLL < FLR { result: 1dDIE }
  result: ROLL
}

function: combo DIE:d {
 result: [combo DIE 0 helper]
}

function: combo N:n L:n helper {
 if N >= L { result: N + [combo DIE N helper] }
 result: N
}

function: minimum ROLL:n MIN:n {
  if ROLL < MIN { result: MIN }
  result: ROLL
}

function: bounce ROLL:n LOW:n {
  if ROLL <= LOW { result: ROLL + LOW }
  result: ROLL
}

function: reroll ROLL:n DIE:n FLR:n {
  if ROLL < FLR { result: 1dDIE }
  result: ROLL
}

set "maximum function depth" to 5

function: combo DIE:d {
 LAST: 1
 result: [combo DIE helper]
}

function: combo N:n helper {
 if N >= LAST { result: N + [combo DIE helper] }
 result: N + 1dDIE
}

output [combo d4] named "d4 combo"
output [combo d6] named "d6 combo"
output [combo d8] named "d8 combo"

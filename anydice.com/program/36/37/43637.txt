\ Define the function for a single attack \
function: attack ROLL:n {
  if ROLL + 8 >= 18 {
    \ Hit: 2d6 + 2d6 + 2d10 + 4 \
    result: 4d6 + 2d10 + 4
  } else {
    \ Miss: 0 damage \
    result: 0
  }
}

\ Calculate the distribution of one attack \
ATTACK: [attack 1d20]

\ Output the results \
output ATTACK named "Damage from 1 Attack"
output ATTACK + ATTACK named "Total Damage from 2 Attacks"

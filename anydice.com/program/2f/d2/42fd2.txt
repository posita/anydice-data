function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}
function: split X:n on Y:n from Z:n{
if (Y > 1) {
 result: [highest 1 of ZdX]
}
else {
 result: [lowest 1 of ZdX]
}
}

function: split ONE and TWO on DICE:n{
if (DICE = 1){
result: ONE
}
result: TWO
}

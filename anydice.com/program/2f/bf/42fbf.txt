function: if CONDITION:n then A else B {
  if CONDITION { result: A } else { result: B }
}

function: skill check A w B {
 if A + B >= 18 {
  result: 0
 }
 else {
  if A + B >= 16 & A + B < 18 {
   result: 1
  }
  else {
   result: 2
  }
 }
}


loop N over {1..6}{
 output [skill check 1d6 w N]named "Att.: [N]"
}

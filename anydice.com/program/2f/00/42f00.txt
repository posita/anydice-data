 result: [highest ASSESSKEEPA of [explode ASSESSROLLAd10]]-[highest ASSESSKEEPB of [explode ASSESSROLLBd10]]
}

function: focus FOCUSROLLA:n FOCUSKEEPA:n FOCUSROLLB:n FOCUSKEEP:n ASSESSMENTMODIFIER:n
{
  if ASSESSMENTMODIFIER >= 10
  { 
    result: [highest FOCUSKEEPA+1 of [explode (FOCUSROLLA+1)d10]]-[highest FOCUSKEEPB of [explode (FOCUSROLLB)d10]]
  }
  if ASSESSMENTMODIFIER <= -10
  { 
    result: [highest FOCUSKEEPA of [explode (FOCUSROLLA)d10]]-[highest (FOCUSKEEPB+1) of [explode (FOCUSROLLB+1)d10]]
  }
  result: [highest FOCUSKEEPA of [explode (FOCUSROLLA)d10]]-[highest (FOCUSKEEPB) of [explode (FOCUSROLLB)d10]]
}

function: strike STRIKERESULTA:n STRIKERESULTB:n ARMOURTNA:n ARMOURTNB:n FOCUSMODIFIER:n {
 if FOCUSMODIFIER >= 5 {
   ALPHABONUS: ((FOCUSMODIFIER-5)/5)*5
   if STRIKERESULTA>(ARMOURTNB-ALPHABONUS)
   {
     result: 1
   }
   else
   {
     if STRIKERESULTB>ARMOURTNA
     {
       result: -2
     }
     else
     {
       result: 3
     }
   }
 }
 if FOCUSMODIFIER <= -5 {
   BETABONUS: (((-FOCUSMODIFIER)-5)/5)*5
   if STRIKERESULTB>(ARMOURTNA-BETABONUS)
   {
     result: -1
   }
   else
   {
     if STRIKERESULTA>ARMOURTNB
     {
       result: 2
     }
     else
     {
       result: -3
     }
   }
 }
 result: 0
}

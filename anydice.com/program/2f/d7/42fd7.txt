function: mfr ROLLS:s {
 SIX: [count 6 in ROLLS]
 FIVE: [count 5 in ROLLS]
 FOUR: [count 4 in ROLLS]
 ONE: [count 1 in ROLLS]
 if SIX >= 2 | FIVE >= 2 { result: 4 } \ crit success \
 if SIX = 1 | FIVE = 1 { result: 3 }   \ full success \
 if FOUR >= 2 {result: 2}              \ crit mixed \
 if FOUR = 1 {result: 1}               \ mixed \
 result: 0                             \ fail \
}

output [mfr 1d6] named "1d6"
output [mfr 2d6] named "2d6"
output [mfr 3d6] named "3d6"
output [mfr 4d6] named "4d6"

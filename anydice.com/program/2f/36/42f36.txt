function: oppose A:n and D:n {
  if A < 4 {result: 0}
  if A <= D {result: 0}
  result: 1
}

loop D over {4,6,8}{
  loop A over {4,6,8,10,12}{
    output [oppose [highest of 1d[explode d6] and 1d[explode dA]] and [highest of 1d[explode d6] and 1d[explode dD]]] named "SWADE Wildcard: oppose 1d[A] vs 1d[D]"
  }
}

loop D over {4,6,8}{
  loop A over {4,6,8,10,12}{
    output [oppose 1d[explode dA] and 1d[explode dD]] named "SWADE Extra: oppose 1d[A] vs 1d[D]"
  }
}

function: cap N:n at MAX:n {
  if N > MAX { result: MAX }
  result: N
}

function: skillsub A:n with AMAX:n and B:n with BMAX:n using N:n {
  if A <= B {
    if A < N {
      result: [cap N at AMAX] + B
    }
    result: A + B
  }

  if B < N {
    result: A + [cap N at BMAX]
  }
  result: A + B
}

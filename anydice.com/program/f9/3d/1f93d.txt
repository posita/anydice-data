\ This program allows you to set an attack modifier and an amount of damage on a hit                \
\ It will then allow you to input a number of attacks (X) vs a certain AC (Y) using this function:  \
\      output [X attacks vs Y]                                                                      \


ATK:5
DMG:1d12 + 3

function: ROLL:n vs AC:n {
 if ROLL + ATK >= AC { result: 1 }
 result: 0
}

function: X:n attacks vs Y:n{
 HITS: 0
 loop N over {1..X}{
   HITS : HITS + [d20 vs Y]
  }
 OUCH : [damage for HITS hits]
 result: OUCH
}


function: damage for Z:n hits {
 if Z = 0 {result: 0}
 PAIN:0
 loop N over {1..Z} {
  PAIN : PAIN + DMG
 }
 result: PAIN
}

output [(d6 + 10) attacks vs 22] named "orc archers"
output [8 attacks vs 22] named "orc soldiers"

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: battackroll BROLL:n vs AC:n {
 if BROLL >= BCRITROLL { result: BCRITDAMAGE } 
 if BROLL = 1 { result: 0 }
 if BROLL + BPROFICIENCY + BATTACKMOD + BACCBONUS >= AC { result: BDAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 6
DMGBONUS: 0
PROFICIENCY: 4
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 2d{{1..2}, {3..6}:4} + ATTACKMOD + DMGBONUS
CRITDAMAGE: 4d{{1..2}, {3..6}:4} + ATTACKMOD + DMGBONUS

BATTACKMOD: 3
BDMGBONUS: 0
BPROFICIENCY: 4
BACCBONUS: 0
BCRITROLL: 20
BATTACKTIMES: 1
BDAMAGE: 2d8
BCRITDAMAGE: 4d8

output ATTACKTIMESd[attackroll d20 vs ENEMYAC]

output BATTACKTIMESd[battackroll [highest 1 of 2d20] vs ENEMYAC]

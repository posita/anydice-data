function: attack DIE:n plus ATK ac AC for DMG crit CRIT {
 if (DIE = 20) {result: DMG + CRIT}
 if (DIE = 1) {result: 0}
 if (DIE + ATK >= AC) {result: DMG}
 else {result: 0}
}

}

function: craft DIE:n plus MOD {
 if (DIE = 1) {result: 150}
 if (DIE+MOD <= 10) {result: 115}
 if (DIE+MOD <= 15) {result: 100}
 if (DIE+MOD <= 20) {result:  85}
 if (DIE+MOD  > 20) {result:  50}
 else {result: 0}
}

function: gold DIE:n plus MOD {
 if (DIE = 1) {result: 0}
 if (DIE+MOD <= 10) {result: MOD * 2d4}
 if (DIE+MOD <= 15) {result: MOD * 3d4}
 if (DIE+MOD <= 20) {result: MOD * 3d6}
 if (DIE+MOD  > 20) {result: MOD * 3d8}
 else {result: 0}
}

function: reroll FIRST:n as SECOND:n {
 if (FIRST=0) {result: SECOND}
 else {result: FIRST}
}

function: froll A:n or B:n or M:n as R:n {
 if (A=0)|(B=0)|(M=0) {result: A+B+M+R}
 else {result: A+B+M}
}

function: froll A:n or B:n as R:n {
 if (A=0)|(B=0) {result: A+B+R}
 else {result: A+B}
}

loop N over {1..6} {
   T: d6 < 4
   U: d6 > 5
   V: ( d6 > 3 & d6 < 6 )
   TF: T
   UF: U
   VF: V
   loop N over {2..N} {
      TF: TF | T
      UF: UF | U
      VF: VF | V
   }
   output TF named "[N] Dice Disasters"
   output VF named "[N] Dice Conflicts"
   output UF named "[N] Dice Triumphs"
}

function: A:n eq B:n { result: A = B }

function: roll under ATTR:n explode SKILL:n times {
  A: {1:ATTR, 0:(6 - ATTR)}
  MAX_A: SKILL + 1
  SUCC_A: [explode 1dA]
  EXPL: SUCC_A - 1
  MAX_B: (SKILL + 1) - EXPL
  
  if ([SUCC_A eq MAX_A]) {
    SUCC_B: [lowest of [explode 1dA] and MAX_B]
  }
  else {
    SUCC_B: 1dA
  }
  result: SUCC_A + SUCC_B
}

ATTR: 3
SKILL: 4
A: {1:ATTR, 0:(6 - ATTR)}
set "explode depth" to SKILL
output [roll under ATTR explode SKILL times]

function: precision P:n R:n {
 if P > 10 {result: [highPrec P R]}
 else {result: [lowPrec P R]}
}
function: lowPrec P:n R:n {
 if P < R {result: ((R*(25/100))/P)*-1}
 else {result: 1}
}
function: highPrec P:n R:n {
 if ((P/2)-2) < R {result: ((R*(25/100))/P)*-1}
 else {result: 1}
}
output [precision 10+[explode 1d6] 30]

\ example \
\fireball 35+ P1 \ 
\fireflies 30+ P-1\
\flyingshards P0\
\bane fire 45+ P1\
\lightning bolt 40+ P4\
\thunder strike 100 P2\
\farstrike 50 P5\
\goh 100 P-3\
\falling fires 30 P1\
\smokeless flame 35 P1\

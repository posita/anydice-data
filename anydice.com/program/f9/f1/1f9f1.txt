function: luck ROLLA:n{
 RES: 0

 if ([count {1} in ROLLA]) {
   RES: RES -3
 }
 if ([count {2} in ROLLA]) {
   RES: RES -2
 }

 if ([count {3} in ROLLA]) {
   RES: RES -1
 }

 if ([count {8} in ROLLA]) {
   RES: RES +1
 } 
 if ([count {9} in ROLLA]) {
   RES: RES +2
 }
 if ([count {10} in ROLLA]) {
   RES: RES +3
 }

result: RES
}
LUCK: [luck d10]
output 3@3d10 + LUCK named "lowest 3D10"

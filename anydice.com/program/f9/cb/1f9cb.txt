function: hsd N:n plus M:n with O:d and P:d using A:n and B:n {
 if N+M < 12 {
  if A > B {
   if N <= M {
    result: M + O
   }
   else {
    result: N + P
   }
  }
  else {
   if N < M {
    result: M + O
   }
   else {
    result: N + P
   }
  }
 }
 result: N + M
}

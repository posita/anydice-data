function: hit DICE:n automiss MISS:n plus ACC:n vs DEF:n {
 if DICE = MISS { result: 0}
 if DICE + ACC < DEF { result: 0 }
 result: 1
}

function: damage DICE:n by MOD:n {
 if DICE + MOD < 0 { result: 0 }
 result: DICE + MOD
}

function: attack HIT:d plus ACC:n vs def DEF:n doing DAM:d plus PS:n minus ARM:n {
 result: [hit HIT automiss #HIT plus ACC vs DEF] * [damage DAM by PS-ARM]
}

output 
[attack 3d6 plus 5 vs def 9 doing 3d6 plus 10 minus 15] + 
[attack 3d6 plus 7 vs def 14 doing 3d6 plus 14 minus 15]

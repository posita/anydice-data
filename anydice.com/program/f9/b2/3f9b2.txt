set "maximum function depth" to 20
function: how many DIE:d to NUM:n {
  if NUM <= 0 {
    result: 0
  }
  MAX: [maximum of DIE]
  if #MAX = 0 {
    result: 1
  }
  MINIT: NUM / MAX
  if (MINIT > 1) {
   result: MINIT + [how many DIE to (NUM - MINITdDIE)]
  }
  result: 1 + [how many DIE to (NUM - DIE)]
}

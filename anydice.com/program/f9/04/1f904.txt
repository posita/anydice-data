\ Smallsword comparisons \

function: critmult ATTACK:n bonus BONUS:n versus AC:n damage DAMAGE:n plus PLUS:n {
if ATTACK = 1 { result: 0 }
if ATTACK + BONUS < AC { result: 0 }
if ATTACK = 20 { result: 2dDAMAGE + PLUS }
result: DAMAGE + PLUS
}

function: critmultfact ATTACK:n bonus BONUS:n versus AC:n damage DAMAGE:n plus PLUS:n {
if ATTACK = 1 { result: 0 }
if ATTACK + BONUS < AC { result: 0}
if ATTACK = 20 { result: 2dDAMAGE + PLUS + [critmult d20 bonus BONUS versus AC damage DAMAGE plus PLUS] }
result: DAMAGE + PLUS + [critmult d20 bonus BONUS versus AC damage DAMAGE plus PLUS]
}

\SNEAK: 4d6
EXPERT: d6
DEX: 3
PROF: 3
TARGET_AC: 20\

SNEAK: 10d6
EXPERT: d12
DEX: 5
PROF: 6
TARGET_AC: 18

\output [critmult d20 bonus DEX + PROF versus TARGET_AC damage d6 plus DEX] named "Standard d6"\
output [critmultfact d20 bonus DEX + PROF + EXPERT versus TARGET_AC damage d6 + EXPERT plus DEX] named "Duellist d6"
output [critmultfact d20 bonus DEX + PROF + EXPERT + 2 versus TARGET_AC damage d6 + EXPERT plus DEX] named "Duellist d6 + 2"
output [critmultfact d20 bonus DEX + PROF + EXPERT + PROF versus TARGET_AC damage d6 + EXPERT plus DEX - PROF] named "Duellist d6, original smallsword"
\output [critmult d20 bonus DEX + PROF versus TARGET_AC damage d6 + SNEAK plus DEX] named "Rogue d6"
output [critmult d20 bonus DEX + PROF + 2 versus TARGET_AC damage d6 + SNEAK plus DEX] named "Rogue d6 + 2"
output [critmult d20 bonus DEX + PROF + PROF versus TARGET_AC damage d6 + SNEAK plus DEX - PROF] named "Rogue d6, original smallsword"\

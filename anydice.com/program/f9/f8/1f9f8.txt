function: near dupes in A:s B:s C:s {
  DICE: [sort {A, B, C}]
  DICE: {1@DICE + 999, DICE, #DICE@DICE - 999} \ pad list with dummy values \
  DUPES: 0
  loop X over {2 .. #DICE-1} {
    SAME: (X-1)@DICE = X@DICE | (X+1)@DICE = X@DICE
    NEAR: (X-1)@DICE = X@DICE+1 | (X+1)@DICE = X@DICE-1
    if SAME | NEAR { DUPES: DUPES + 1 }
  }
  result: DUPES
}

output [near dupes in 1d12 2d10 1d8]

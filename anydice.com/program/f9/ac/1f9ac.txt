function: ROLL:n vs AC:n {
 if ROLL = 1 {result: 0}
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK + CONFIRM >= AC | ROLL = 99 {
  result: CRITICAL d DAMAGE
 }
 result: DAMAGE
}

AC: 31
STRENGTH: 5
TWOHANDED: STRENGTH + STRENGTH/2
OFFHAND: STRENGTH/2
DEXTERITY: 6
MODIFIER: 6
ATTACK: 9 + DEXTERITY + MODIFIER 

DAMAGE: 1d8 + 3 +2d6+2
THREAT: 20
CRITICAL: 2
CONFIRM: -99
output [d20 vs AC] named "Attack vs AC 31"

LVL:1
PROF:4
STR:4
DEX:1
ITEM:0
MAP:-5
AGIMAP:-4

DEFAULT: LVL+PROF+STR+ITEM
FINESSE: LVL+PROF+DEX+ITEM

WEAPON_A: 1d12+STR
WEAPON_B: 2d12+STR

FOELVL:4
FOEDEX:3
FOEARMOR:3
FOEBONUS:2

AC:10+FOELVL+FOEDEX+FOEARMOR+FOEBONUS

function: X damage D:d E:n F:n {
DMG:D+E-F
if X<=0 {result: 0}
if X=1 {result: DMG}
if X>=2 {result:(DMG)*2}
}


function: N attack A:s plus B minus P vs ac AC dealing D plus E damage minus F {
if N>3 {N:3}
TOHIT:A+B-P-((N-1)*5)
if TOHIT>=AC+10 {X:2} else if TOHIT>=AC {X:1} else {X:0}
if A=20 {X:X+1} else if A=1 {X:X-1}
result:[X damage D E F]
}

output [1 attack d20 plus DEFAULT minus 0 vs ac AC dealing WEAPON_A plus 0 damage minus 0]
+ [2 attack d20 plus DEFAULT minus 0 vs ac AC dealing WEAPON_A plus 0 damage minus 0]
+ [3 attack d20 plus DEFAULT minus 0 vs ac AC dealing WEAPON_A plus 0 damage minus 0] named "3 strikes"

output [1 attack d20 plus DEFAULT minus 0 vs ac AC dealing WEAPON_B plus 0 damage minus 0]
+ [3 attack d20 plus DEFAULT minus 0 vs ac AC dealing WEAPON_A plus 0 damage minus 0] named "power attack + strike"

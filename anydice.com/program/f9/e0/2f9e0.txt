function: biplode DIE:d {
 MAX: [maximum of DIE]
 result: [biplode DIE helper]
}

function: biplode N:n helper {
 if N = MAX { result: MAX + 1dMAX}
 if N = 1 { result: -MAX + 1dMAX}
 result: N
}

function: extreme A:d B:d
{
 HALF: [maximum of A] / 2
 result: [extreme A B helper]
}

function: extreme A:n B:n helper
{
 ADELTA:[absolute A - HALF]
 BDELTA:[absolute B - HALF]
 if ADELTA > BDELTA { result:A }
 result: B
}

output 1d20 named "1d20"
output [biplode 1d20] named "Biplode"
output [extreme 1d20 1d20] named "Extreme"

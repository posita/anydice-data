function: attack with ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT:d on CRIT_RANGE:n {
  if ROLL = 1           { result: 0 }
  if ROLL >= CRIT_RANGE { result: CRIT }
  if ROLL + BONUS >= AC { result: DMG }
  result: 0
}

SUPER_ADVANTAGE: [highest 1 of 3d20]
CRIT_RANGE: 20

BOOMING_FLAME_BLADE_DMG_ROLL: 1d8
BOOMING_FLAME_BLADE_DMG_ROLL_CRIT: 2d8

SHADOW_BLADE_HIT_BONUS: 7
SHADOW_BLADE_DMG_BONUS: 4
SHADOW_BLADE_DMG_ROLL: 3d8
SHADOW_BLADE_DMG_ROLL_CRIT: 6d8
SHADOW_BLADE_ATTACK_DMG: SHADOW_BLADE_DMG_ROLL + SHADOW_BLADE_DMG_BONUS + BOOMING_FLAME_BLADE_DMG_ROLL
SHADOW_BLADE_ATTACK_DMG_CRIT: SHADOW_BLADE_DMG_ROLL_CRIT + SHADOW_BLADE_DMG_BONUS + BOOMING_FLAME_BLADE_DMG_ROLL_CRIT
SHADOW_BLADE_ATTACK_BONUS_DMG: SHADOW_BLADE_DMG_ROLL
SHADOW_BLADE_ATTACK_BONUS_DMG_CRIT: SHADOW_BLADE_DMG_ROLL_CRIT

loop AC over {10..25} {

  SHADOW_BLADE_ATTACK: [attack with SUPER_ADVANTAGE plus SHADOW_BLADE_HIT_BONUS vs AC for SHADOW_BLADE_ATTACK_DMG crit SHADOW_BLADE_ATTACK_DMG_CRIT on CRIT_RANGE]
  SHADOW_BLADE_ATTACK_BONUS: [attack with SUPER_ADVANTAGE plus SHADOW_BLADE_HIT_BONUS vs AC for SHADOW_BLADE_ATTACK_BONUS_DMG crit SHADOW_BLADE_ATTACK_BONUS_DMG_CRIT on CRIT_RANGE]
  output SHADOW_BLADE_ATTACK + SHADOW_BLADE_ATTACK_BONUS named "Shadow Blade in dim light or darkness vs AC [AC]"
}

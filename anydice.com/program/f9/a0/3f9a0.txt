\ Angrepsfunksjonen \
\ 1 er treff, 2 er raise, 0 er bom \
function: old tohit DEFENSE with DIE:n {
  if DIE = 1 { result: -2 }
  if DIE >= DEFENSE+4 { result: 1 } 
  if DIE >= DEFENSE { result: 0 }
  result: -1
}

\ Angrepsfunksjonen \
\ 1 er treff, 2 er raise, 0 er bom \
function: tohit DEFENSE with DIE:n {
  if DIE >= DEFENSE+4 { result: 2 } 
  if DIE >= DEFENSE { result: 1 }
  result: 0
}


\ WOUNDS-funksjonen \
\ returnerer antall wounds. 0 er shaken, -1 er bom, 1 er 1 wound, 2 er 2 wounds osv. \
\ input er tohit-sjansen, skaden, fiendens toughness \
function: wounds toughness TN atk result ATK_RES:n dmg NDMG:n ifraise RAISE:n {
  DMG: NDMG
  if ATK_RES = 0  { result: -1 }
  if ATK_RES = 2 { DMG: NDMG + RAISE }
  if DMG - TN < 0 { result: -1 }
  if ((DMG - TN) / 4) >= MAXWOUNDS { result: MAXWOUNDS }
  \ divisjon runder ned til heltall sÃ¥ 1-3 over gir 0, 4-7 over er 1. Hva med minus? \
  result: (DMG-TN)/4
}

\ lag et resultat for begge piler eller skudd \
\ vi har -1 som bom, 0 som shaken, positiv som wounds \
function: addwounds DA:n and DB:n {
  if DB < 0 { result: DA }
  if DA < 0 { result: DB }
  \ DA og DB er begge nÃ¥ 0 eller positiv \
  \ nÃ¥ vet vi at vi har treff pÃ¥ begge. Komplikado at 2x shaken = 1 wound \
  if DB = 0 {
    if DA = 0 { result: 1 }
    result: DA
  }
  result: DA + DB
}

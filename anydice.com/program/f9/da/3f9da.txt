function: keep A:n over B:n if C:n{
 if A = 10 { result: 1 }
 if C = 1 { result: A }
 else {
  result: B
 }
}

function: reroll X:n and Y:n and Z:n{
  if Z > 2 { result: [keep X over Y if 1d2] }
  result: [blm 1d20 and 1d20 and Z]
}

function: blm X:n and Y:n and Z:n{
    FIRST: [absolute X - 10]
    SECOND: [absolute Y - 10]
    if FIRST<SECOND{
        result: Y
    } 
    else {
       if SECOND<FIRST {
         result: X
       } 
       else {
         result: [reroll X and Y and Z+1]
       }
    }
}

output [blm 1d20 and 1d20 and 1] named "brennan lee mulligan"

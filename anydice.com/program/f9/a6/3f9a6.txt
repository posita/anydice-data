X: {0:5, 1}
function: roll ROLL:s keep K:n {
  result: [lowest of K and ROLL]
}
function: push ROLL:s keep K:n {
  SUCC: [lowest of K and ROLL]
  P_N: #ROLL - SUCC
  P_K: 6 - SUCC
  PUSH: [lowest of P_K and P_NdX]
  result: SUCC + PUSH
}

SKILL: 0
ATTRIBUTE: 1
N: SKILL + ATTRIBUTE
K: SKILL
X: {0:5, 1}

loop A over {1..6} {
  N: SKILL + A
  output [roll NdX keep K] named "Roll [N] Keep [K]"
  output [push NdX keep K] named "Roll [N] Keep [K], Pushed"
}

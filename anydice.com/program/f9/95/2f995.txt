if X= 20 {result: [d20 AC BONUS crit]}
if X= 1 {result: 0}
if X>= CRIT & X>=AC-BONUS {result: [d20 AC BONUS crit]}
if X>=AC-BONUS {result: [damage]}
else {result: 0}
}

function: X:n AC:n BONUS:n crit {

}

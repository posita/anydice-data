function: scaling N:n {
 if N > 1 { result: 2^N}
 if N < 1 { result: 0}
}
OFFSET: 3

loop REALM over {0..9} {
  POOL: 10*(REALM) + (8*REALM)d4-(8*REALM) + [scaling (REALM-OFFSET)]
  output POOL named "d4-1 R[REALM]"
}

loop REALM over {0..9} {
  POOL: 10*(REALM)  + (4*REALM)d6 + [scaling (REALM-OFFSET)]
  output POOL named "d6 R[REALM]"
}

loop REALM over {0..9} {
  POOL: 10*(REALM) + (4*REALM)d10 + [scaling (REALM-OFFSET)]
  output POOL named "4d10 R[REALM]"
}

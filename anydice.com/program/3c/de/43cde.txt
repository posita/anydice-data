BASEDIE: d6
REROLLCOUNT: 3

set "maximum function depth" to 3

function: target TARGET:n reroll REROLL:n VALUE:n {
  if VALUE = REROLL {
    result: 1 + [target TARGET reroll REROLL BASEDIE]
  }
  if VALUE  > TARGET {
    result: 1
  }
  if REROLLCOUNT > 0{
  REROLLCOUNT: REROLLCOUNT - 1
  result: [wewolla 3 reroll 6 BASEDIE]
  }
  result: 0
}

function: wewolla TARGET:n reroll REROLL:n VALUE:n {
  if VALUE = REROLL {
    result: 1 + [target TARGET reroll REROLL BASEDIE]
  }
  if VALUE  > TARGET {
    result: 1
  }
  result: 0
}

DIE: [target 3 reroll 6 BASEDIE]

loop N over {1..10}{
  output NdDIE named "[N]d6"
}

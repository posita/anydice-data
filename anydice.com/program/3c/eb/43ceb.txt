function: calc S:n r N:s m M:n vs TN:n {
R: 4
X: 10

if N + M < TN + X{R: 3}
if N + M < TN {R: 2}
if N + M < TN - X {R: 1}
if S@N = 1 {R: R - 1}
if S@N = 12 {R: R + 1}

result: [lowest 1 of 4 and [highest 1 of 1 and R]]
}

loop X over {-5..5}{
Y: X * 4
output [calc d3 r 3d12 m Y vs 20] named "[Y]"
}

\ d100 roll-under with digit swap \
\ Roll a tens d10 and a ones d10; you may swap the digits if it lowers the result \
\ Double-zero counts as 100, Call of Cthulhu style \

function: swapdigit T:n O:n {
  if T < O { result: T * 10 + O }
  if O < T { result: O * 10 + T }
  if T = 0 { result: 100 }
  result: T * 10 + O
}

function: positive ROLL:n {
  if ROLL < 0 {
    result: 0
  } else {
    result: ROLL
  }
}

DIGIT: d10 - 1

output 8d6
output [swapdigit DIGIT DIGIT] <= 20 named "swap"
output [positive d100-20] <= 20 named "-20"

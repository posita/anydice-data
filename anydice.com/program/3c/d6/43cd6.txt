function: good A:n bad B:n and ugly C:n {
 if A = B & B = C {
  result: 0
 }
 if A = B & A > C {
  result: 0
 }
 if A = C & A > B {
  result: 0
 }
 if B = C & B > A {
  result: 0
 }
 if A > B & A > C {
  result: 1
 }
 if B > A & B > C {
  result: 2
 }
 if C > B & C > A {
  result: 3
 }
 result: 10
}


function: impact I:n neutral N:n and cost C:n {
 S: 0
 F: 0
 if I >= N {
  S: S + 1
 }
 if I >= C {
  S: S + 1
 }
 if C >= N {
  F: F + 1
 }
 if C >= I {
  F: F + 1
 }
 result: 10 * S + 3 - F
}

output [good 1d6 bad 1d6 and ugly 1d6]
output [good 1d7 bad 1d7 and ugly 1d7]

output [impact 1d6 neutral 1d6 and cost 1d6]
output [impact 1d7 neutral 1d7 and cost 1d7]

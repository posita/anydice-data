set "explode depth" to 5
function: normal C:n contra T:n{
 SM:(d(C-1)>T)
 E:(dC=C)*(C>T)
 EXP:[explode E]
 result:EXP+SM
}
function: desventaja Q:n de C:n contra T:n{
 S:(dC>T)
 E:(dC=C)
 result:[lowest Q of (2*Q)dS]+[lowest Q of (2*Q)dE]*[normal C contra T]
}
function: doble Q:n de C:n contra T:n{
 S:(dC>T)
 E:(dC=C)
 result:[lowest Q of (3*Q)dS]+[lowest Q of (3*Q)dE]*[normal C contra T]
}
loop TARGET over {0..8}{
 output 4d[normal 10 contra TARGET] named "[TARGET] Heavy"
 output 2d[normal 10 contra TARGET]+[desventaja 2 de 10 contra TARGET] named "[TARGET] Normal"
 output 3d[normal 8 contra TARGET]+[desventaja 3 de 8 contra TARGET] named "[TARGET] Light"
 output 3d[normal 4 contra TARGET]+[desventaja 3 de 4 contra TARGET] named "[TARGET] Dagger"
}

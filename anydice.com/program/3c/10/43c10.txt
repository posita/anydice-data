function: closest POOL:s to TN:n {
  MIN:10
  MIN_RESULT:0
  loop NUM over POOL {
    DIST:[absolute (TN-NUM)]
    if(DIST<MIN){
      MIN:DIST
      MIN_RESULT:NUM
    }
  }
  result:MIN_RESULT
}
loop TN over {5..8}{
  loop DICE over {2..4}{
    output [closest DICEd10 to TN] named "[DICE]d10 @[TN]"
  }
}

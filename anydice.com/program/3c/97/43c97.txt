DICE: 6

function: success R:n vs TARGET:n {
    if R = 12 {
        result: 1 + (d12 >= TARGET)
    }
    result: R >= TARGET
}

loop TARGET over {3..12} {
    output DICE d[success d12 vs TARGET] >= 1 named "T[TARGET]: 1+"
    output DICE d[success d12 vs TARGET] >= 2 named "T[TARGET]: 2+"
    output DICE d[success d12 vs TARGET] >= 3 named "T[TARGET]: 3+"
    output DICE d[success d12 vs TARGET] >= 4 named "T[TARGET]: 4+"
    output DICE d[success d12 vs TARGET] >= 5 named "T[TARGET]: 5+"
    output DICE d[success d12 vs TARGET] >= 6 named "T[TARGET]: 6+"
}

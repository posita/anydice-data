\roll 4 drop 4s reroll lowest\

function: A:s drop B:n {
 LOWEST: 4@[sort A]
 if LOWEST != B {
  result: ((A - LOWEST) - B*(A = B)) + 1d6
 }
 if [count 4 in A] = B {
  result: 0
 }
 if LOWEST = B {
  if [A contains 5] {
   result: ((A - 5) - B*(A = B)) + 1d6
  }
  else {
   result: ((A - 6) - B*(A = B)) + 1d6
  }
 }
}

output [4d6 drop 4]

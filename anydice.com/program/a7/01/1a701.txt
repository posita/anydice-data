\ Computes probability of successful Counterspell against 4-9 level spells \
CHA: 4
PROF: 3
JACK: PROF/2
BASE: 10
PEERLESS: d10

CHECK: d20+CHA+JACK

loop N over {4..9}{
   \ DC = 10+Spell_Level \
   output CHECK>=BASE+N named "[N]th level spell"
   output CHECK+PEERLESS>=BASE+N named "[N]th level spell (Peerless Skill)"
}

function: wbsroll N:s {
 if N = 6 { result: N + [explode 1d6] }
 if #(N) > 1 {
  if N = 1 { result: N - 1@N}
 }
 result: N
}

set "maximum function depth" to 5

loop THRESH over {1..30} {
 output ([wbsroll 3d6] + 6) > THRESH named "Standard vs [THRESH]"
 output ([wbsroll 1d6]+12) > THRESH named "Controlled Effort vs [THRESH]"
 output ([wbsroll 4d6]+2) > THRESH named "Wild Effort vs [THRESH]" 
}

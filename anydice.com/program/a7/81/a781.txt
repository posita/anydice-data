MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  result: CRITICALHIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: advantageattack FIRSTROLL:n and SECONDROLL:n vs DEFENSE:n {
 if FIRSTROLL= 20 | SECONDROLL= 20{
 result: CRITICALHIT }
 if FIRSTROLL+ ATTACK >= DEFENSE | SECONDROLL+ ATTACK >= DEFENSE{
 result: HIT}
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: attack {
 result: [damage for ([attack d20 vs DEFENSE])]
}

function: advantageattack {
 result: [damage for ([advantageattack d20 and d20 vs DEFENSE])]
}

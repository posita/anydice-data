REROLL: 2 \fighter two handed reroll on damage\

function: weapon damage DMG:n crit CRIT{
 result: DMG*CRIT + WM + AM \dmg dice + weapon bonus + damage attribute bonus\
}

function: weapon damage DMG:n plus DMOD crit CRIT {
 result: [weapon damage DMG crit CRIT] + DMOD
}

function: dmg DMG:n dice DICE:n reroll REROLL  {
 if DMG <= REROLL {
  result: dDICE
 }
 result: DMG
}

function: dmg DICE multiple N reroll REROLL {
DMG:0
loop I over {1..N}{
 DMG: DMG + [dmg dDICE dice DICE reroll REROLL]
}
result: DMG
}

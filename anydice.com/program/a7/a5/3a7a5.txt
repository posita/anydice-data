function: arma dmg D str F leth L prot P{
 result: [count {P..(F*2)} in Dd(F*2)] * L
}

function: mazza dmg D str F leth L prot P{
 SUCC:0
 loop N over {1..D}{
  RIS: [highest 1 of 2d(F * 2)]
  SUCC: SUCC + (RIS >= P)
 }
 result: SUCC * L
}

loop PROT over {1..10}
{
 output [arma dmg 1 str 1 leth 1 prot PROT ] named "SPADA"
}
loop PROT over {1..10}
{
 output [mazza dmg 1 str 1 leth 1 prot PROT ] named "MAZZA"
}

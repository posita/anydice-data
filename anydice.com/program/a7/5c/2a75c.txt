function: test ONE:n and TWO:n vs T:n {
  R: 0
  if ONE >= T { R: 1 }
  if TWO >= T { result: R + 1 }
  result: R
}

loop N over {-10..5} {
  \output [test d6 and d6 vs d6-N] named "[N]"\
  \output [test [explode d6] and [explode d6] vs d6-N] named "[N]"\
  output [test [explode d6] and d6 vs d6-N] named "[N]"
}

function: zbeats RANK {
 result: [highest 1 of 2d14 < RANK]
}

function: beats PULLS RANK {
 result: PULLS d14 < RANK
}

loop RANK over {2..15} {
 output [zbeats RANK] named "0c < [RANK]"
}

loop CARDS over {1..4} {
 loop RANK over {2..15} {
  output [beats CARDS RANK] named "[CARDS]c < [RANK]"
 }
}

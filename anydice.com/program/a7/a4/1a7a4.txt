function: roll ROLL:s {
  HI: 1@ROLL
  LO: 2@ROLL
  if LO = HI           { result: LO + HI }
  if LO > HI           { result: LO + HI - d6 }
  if LO < HI           { result: LO + HI - d6 }

}

output [roll 2d6]

output d6 +3 -d6

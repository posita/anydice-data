function: reroll DICE:n SECONDDICE:n {
  if DICE > 2 {
    result: DICE
  } else {
    result: SECONDDICE
  }
}

  result: [reroll 1d6 1d6]+[reroll 1d6 1d6]
}

function: savageattack DICE:n SECONDDICE:n {
  if DICE >= SECONDDICE {
    result: DICE
  } else {
    result: SECONDDICE
  }
}

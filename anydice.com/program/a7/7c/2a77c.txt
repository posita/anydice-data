REROLL: 2 \fighter two handed reroll on damage\
CRITON: 20 \default crit value\

MISS: 0
HIT: 1
CRITICALHIT: 2

\attack rols start\
function: advantage A:n plus B:n{
 if  A > B{
  result: A
 }
 else {
  result: B
 }
}

function: disadvantage A:n plus B:n{
 if  A > B{
  result: B
 }
 else {
  result: A
 }
}

function: halfling A:n roll {
 if A = 1 {
  result: d20
 }
 result: A
}

function: attack ROLL:n modifier MOD vs DEFENSE:n crit on CRITON{
 if ROLL = 1 { result: MISS }
 if ROLL >=CRITON {
  if ROLL + MOD >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + MOD >= DEFENSE { result: HIT }
 result: MISS
}

\attack rolls end\

\damage rolls start\

function: damage DMG:n plus DMOD {
 result: DMG + DMOD
}

function: dmg DMG:n dice DICE:n reroll REROLL  {
 if DMG <= REROLL {
  result: dDICE
 }
 result: DMG
}

function: dmg DICE multiple N reroll REROLL {
DMG:0
loop I over {1..N}{
 DMG: DMG + [dmg dDICE dice DICE reroll REROLL]
}
result: DMG
}
\damage rolls end\

\full attack action start\
function: attack HITDAMAGE:n CRITICALDAMAGE:n for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: 0
}

function: spell HITDAMAGE:n for SAVINGTHROW:n with ABILITY:n{
 if SAVINGTHROW >= ABILITY {
  result: HITDAMAGE/2
 }
 result: HITDAMAGE
}

function: two ATK {
 result: ATK + ATK
}

function: three ATK {
 result: ATK + ATK + ATK
}

function: attack DMG:n plus DMOD atk roll ATTACK:n plus MOD:n crit on CRITON crit dmg CRITDMG{
 result:[attack [damage DMG plus DMOD] [damage DMG*CRITICALHIT plus CRITDMG+DMOD] for 
        [attack ATTACK modifier MOD vs AC crit on CRITON ]]
}

function: attack DMG:n plus DMOD atk roll ATTACK:n plus MOD:n crit on CRITON {
 result:[attack [damage DMG plus DMOD] [damage DMG*CRITICALHIT plus DMOD] for 
        [attack ATTACK modifier MOD vs AC crit on CRITON ]]
}

function: attack DMG:n plus DMOD atk roll ATTACK:n plus MOD:n {
 result:[attack [damage DMG plus DMOD] [damage DMG*CRITICALHIT plus DMOD] for 
        [attack ATTACK modifier MOD vs AC crit on CRITON ]]
}

\full attack action end\

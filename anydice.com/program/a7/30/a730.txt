function: minmax ROLL:n MOD{
 if ROLL+MOD > 6 {result: 6}
 if ROLL+MOD < 1 {result: 1}
 result: ROLL+MOD
}

function: roll POS:n mod MOD:n {
 D: POS@3d6
 result: [minmax D MOD]
}

loop P over {-5..5}{
 loop M over {-5..5}{
  if M-6 < P & P-6 < M{
   loop N over {1,2,3}{
    output [roll N mod M] > [roll N mod P] named "[N]@3d6 mod [M] vs [P]"
   }
  }
 }
}

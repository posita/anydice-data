\**** Damage calculator Shadow Blade (lv 3) w/Action Surge ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from X1-X2\
ENEMYAC: d{16..20}

ATTACKMOD: 4
DMGBONUS: 0
PROFICIENCY: 5
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 6
DAMAGE: 3d8 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 6d8 + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Shadow Blade w/Action surge"

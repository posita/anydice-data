ATTACK: 2 * (1d20 + 1d6 + 4) + 2
ACCURACY: 20













function: max X:n Y:n {
   if X > Y {
      Z: X
   }
   else {
      Z:Y
   }
   result: Z
}

function: min X:n Y:n {
   if X < Y {
      Z: X
   }
   else {
      Z:Y
   }
   result: Z
}

function: ishit X:n {
   if X = 1 {
      result: 0
   }

   if X > (20 - ACCURACY) {
      result: 1
   }
   else {
      result: 0
   }   
}



HIT: [ishit d20]




output ATTACK * HIT named "Attack"
output ATTACK * [max HIT HIT] named "Roll Advantage"
output ATTACK * [min HIT HIT] named "Roll Disadvantage"
output [max ATTACK ATTACK] * HIT named "Attack Advantage"
output [min ATTACK ATTACK] * HIT named "Attack Disadvantage"

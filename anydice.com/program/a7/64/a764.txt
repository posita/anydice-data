function: check TN:n SKILL:n CORRUPTION:n{
X: {-1, 2>=TN, 3>=TN, 4>=TN, 5>=TN, 6>=TN, 7>=TN, 8>=TN, 9>=TN, 10>=TN}
Y: {-1, -1, 0, 0, 0, 1}

DTEN: SKILL
DSIX: 5-SKILL
if (CORRUPTION > DSIX) {
DTEN: SKILL - (CORRUPTION-DSIX)
DSIX: 0
}

result: SKILLdX + (5-SKILL)dY + CORRUPTION
}

function: plusonesuccess ROLL:n {
if (ROLL > 0) {result:ROLL+1}
else {result: ROLL}
}

function: ignoreonebotch ROLL:n {
if (ROLL < 0) {result:ROLL+1}
else {result: ROLL}
}

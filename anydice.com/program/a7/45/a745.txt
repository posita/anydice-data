function: tinyrpg D:n again N:n reroll R:n{
 if N=10 { result: 1+[tinyrpg D again d10 reroll R] }
 if N < D {
  if R > 1 { result: [tinyrpg D again d10 reroll R-1] }
 }
 result: N >= D
}

R: 0

loop D over {6..6} {
 loop N over {1..5} {
  output [lowest of 10 and Nd[tinyrpg D again d10 reroll R]] named "Number of success proba with [N]d10 difficulty [D] and [R] reroll"
 }
}

function: roll TOHIT:n plus MOD:n for DAM:d against AC:n {
if TOHIT = 20 {result:DAM+DAM}
else if TOHIT+MOD >= AC {result:DAM}
else {result:0}
}

function: rollchamp TOHIT:n plus MOD:n for DAM:d against AC:n {
if TOHIT > 18 {result:DAM+DAM}
else if TOHIT+MOD >= AC {result:DAM}
else {result:0}
}

function: rollgw TOHIT:n plus MOD:n for DAM:d against AC:n {
if TOHIT > 18 {result:DAM+DAM+[rollchamp d20 plus 0 for 2d{d6,d6,3,4,5,6}+12 against 15]}
else if TOHIT+MOD >= AC {result:DAM}
else {result:0}
}

function: rollcs A:n {
if A = 0 {result:[roll d20 plus 5 for d8+2d4+2 against 15]}
else {result:A+[roll d20 plus 5 for 2d8+2d4+2 against 15]}
}

function: rollcss A:n {
if A = 0 {result:[roll d20 plus 0 for d6+2+2d4+10 against 15]}
else {result:A+[roll d20 plus 0 for d6+2+2d4+10+d8 against 15]}
}

function: zbeats RANK{
 result: [highest 1 of 2d14] < RANK
}

function: beats ELEM RANK{
 SUM: 0
 loop CARD over {1..ELEM}{
  SUM: SUM + (1d14 < RANK)
 }
 result: SUM
}

loop RANK over {2..14}{
 output [zbeats RANK] named "0 [RANK]"
}

loop ELEM over {1..5}{
 loop RANK over {2..14}{
  output [beats ELEM RANK] named "[ELEM] [RANK]"
 }
}

BLUE:0
RED:1
function: X:n ATT:n vs DEF:n{
\ BvB \
if ATT+DEF= 0  {
if X <= 4 {result: 1}
}
\ BvR, RvB \
if ATT+DEF= 1  {
if X <= 2 {result: 1}
}
\ RvR \
if ATT+DEF= 2 {
if X = 8 {result: 2}
if X >= 5 {result: 1}
}
result: 0
}

output 3d(d10 >= 5) named "3 Dreadnought d10 5+"
output 3d[d8 RED vs RED] + 3d[d8 BLUE vs RED] named "3 Dreadnought 1red/1blue VESUS red health type"
output 3d[d8 RED vs BLUE] + 3d[d8 BLUE vs BLUE] named "3 Dreadnought 1red/1blue VESUS red health type"

output 3d(d10 >= 4) named "3 Dreadnought d10 4+"
output 4d[d8 RED vs RED] + 3d[d8 BLUE vs RED] named "3 Dreadnought 1red/1blue + 1 red VESUS red health type"
output 3d[d8 RED vs BLUE] + 4d[d8 BLUE vs BLUE] named "3 Dreadnought 1red/1blue + 1 blue VESUS red health type"

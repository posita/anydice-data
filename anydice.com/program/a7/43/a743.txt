function: tinyrpg D:n again N:n reroll R:n {
  result: [tinyrpgrec D again N reroll R count 0]
}

function: tinyrpgrec D:n again N:n reroll R:n count C {
  if C < 1 { \ stop counting after 10 succes \
    if N=10 { result: 1+[tinyrpgrec D again d10 reroll R count C+1] }
    if N < D {
      if R > 1 { result: [tinyrpgrec D again d10 reroll R-1 count C+1] }
    }
  }
  result: N >= D
}

R: 0

loop D over {6..6} {
 loop N over {1..5} {
  output Nd[tinyrpg D again d10 reroll R] named "Number of success proba with [N]d10 difficulty [D] and [R] reroll"
 }
}

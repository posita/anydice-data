BASE_DIE: d6

function: target TARGET:n reroll REROLL:n VALUE:n {
  if VALUE >= REROLL {
    result: 1 + [target TARGET reroll REROLL BASE_DIE]
  }
  if VALUE  > TARGET {
    result: 1
  }
  result: 0
}

ATTACKER_ATTRIBUTE: 5
ATTACKER_SKILL: 1
DEFENDER_ATTRIBUTE: 5
DEFENDER_SKILL: 1
ATTACK_POOL: ATTACKER_ATTRIBUTE + ATTACKER_SKILL
ATTACK_DIFFICULTY: 1 + DEFENDER_SKILL

DIE: [target ATTACK_DIFFICULTY reroll [maximum of BASE_DIE] BASE_DIE]
ATTACK_SUCCESSES: ATTACK_POOLdDIE
HITS: ATTACK_SUCCESSES - DEFENDER_ATTRIBUTE

output ATTACK_SUCCESSES named "successes"
output HITS named "hits"

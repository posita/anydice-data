function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}
function: ROLL:n extrahelper BADA:s and BADB:s as REROLLA:d and REROLLB:d{
  if ROLL = BADA { result: REROLLA }
  else if ROLL = BADB { result: REROLLB}
  result: [[[[ROLL failsless HIT] reroll {HIT..6} as [[d6 reroll BADB as d6] failsless WOUND]] reroll {WOUND..6} as [d6 failsmore SAVE]] reroll {1..(SAVE-1)} as DMG]
}
function: ROLL:n failsless THRESHOLD:n{
  if ROLL < THRESHOLD { result: 0}
  result: ROLL
}
function: ROLL:n failsmore THRESHOLD:n{
  if ROLL >= THRESHOLD { result: 0}
  result: ROLL
}

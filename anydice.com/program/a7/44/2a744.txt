function: keep three ACCURACY:s PRECISION:s
{TOPS: {1@ACCURACY, 2@ACCURACY, 3@ACCURACY, 4@ACCURACY, 5@ACCURACY, 6@ACCURACY, 7@ACCURACY, 8@ACCURACY, 9@ACCURACY, 10@ACCURACY, 1@PRECISION, 2@PRECISION, 3@PRECISION, 4@PRECISION, 5@PRECISION, 6@PRECISION}
  SORTED: [sort TOPS]
  result: {1,2,3}@SORTED}

output [keep three 5d{0,0,0,1,2,3} 0d{0,1,1,1,1,1}] named "5d"
output [keep three 4d{0,0,0,1,2,3} 1d{0,1,1,1,1,1}] named "5d1w"
output [keep three 3d{0,0,0,1,2,3} 2d{0,1,1,1,1,1}] named "5d2w"
output [keep three 2d{0,0,0,1,2,3} 3d{0,1,1,1,1,1}] named "5d3w"
output [keep three 1d{0,0,0,1,2,3} 4d{0,1,1,1,1,1}] named "5d4w"
output [keep three 0d{0,0,0,1,2,3} 5d{0,1,1,1,1,1}] named "5d5w"

function: tn TN { result: 1d10 >= TN }

function: check RATING tn TN {
  SUCCESSES: 0
  loop N over {1..RATING} {
    SUCCESSES: SUCCESSES + [tn TN] 
  }
  result: SUCCESSES
}

function: novice    RATING { result: [check RATING tn 6] }
function: expert    RATING { result: [check RATING tn 5] }
function: master    RATING { result: [check RATING tn 4] }
function: legendary RATING { result: [check RATING tn 3] }
function: divine    RATING { result: [check RATING tn 2] }

function: check     RATING tn TN ob OB { result: [check RATING tn TN] >= OB }
function: novice    RATING       ob OB { result: [check RATING tn 6 ob OB] }
function: expert    RATING       ob OB { result: [check RATING tn 5 ob OB] }
function: master    RATING       ob OB { result: [check RATING tn 4 ob OB] }
function: legendary RATING       ob OB { result: [check RATING tn 3 ob OB] }
function: divine    RATING       ob OB { result: [check RATING tn 2 ob OB] }

\
output [novice 10 ob 3] named "Novice 10 Ob 3"
output [expert 6 ob 3] named "Expert 6 Ob 3"
output [divine 3 ob 3] named "Divine 3 Ob 3"
\

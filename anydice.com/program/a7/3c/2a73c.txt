function: advantage A:n plus B:n{
 if  A > B{
  result: A
 }
 else {
  result: B
 }
}

function: disadvantage A:n plus B:n{
 if  A > B{
  result: B
 }
 else {
  result: A
 }
}

function: halfling A:n roll {
 if A = 1 {
  result: d20
 }
 result: A
}

function: attack ROLL:n modifier MOD vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + MOD >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + MOD >= DEFENSE { result: HIT }
 result: MISS
}

\attack rolls end\

}

}

  result: dDICE
 }

}

result: [weapon damage DMG crit CRIT] + HWM

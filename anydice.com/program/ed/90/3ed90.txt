function: success of ROLL:s when using dice F:n {
  S: 0
  NO_ONES: ![ROLL contains 1]
  loop DIE_VALUE over {1,4..F} {
    COUNT: ROLL = DIE_VALUE
    if DIE_VALUE = 1 & COUNT >= 1{
      S: S - #(ROLL)
    }
    if DIE_VALUE >= 4{
      WEIGHT: 0
      if DIE_VALUE = F & NO_ONES{
        WEIGHT: #(ROLL)
      }else{
        WEIGHT: ((DIE_VALUE - 4) / 2 ) + 1
      }
      S: S + COUNT * WEIGHT
    }

  }
  MAX_CNT: ROLL = F

  if S < 0{
    S: 0
  }
  result: S
}

output [success of 5d8 when using dice 8]

function: explode N:n {
 if N = 1 {
  result: N - d10
 }
 if N = 10 {
  result: N + d10
 }
 result: N
}

function: damage N:n {
 result: N
}

function: crit N:n {
 if N = 6 {
 result: 1
 }
 result: 0
}

DAMAGE: 3d([damage d6])
CRIT: 3d([crit d6])
ONE: [explode 1d10]+14
DUAL: [explode 1d10]+10
AIM: [explode 1d10]+6
DVCLOSE: 13
DVMID: 15
DVFAR: 20

output ONE > DVCLOSE named "One Weapon (CLOSE)"
output ONE > DVMID named "One Weapon (MID)"
output ONE > DVFAR named "One Weapon (FAR)"
output DUAL > DVCLOSE named "Dual wield (CLOSE)"
output DUAL > DVMID named "Dual wield (MID)"
output DUAL > DVFAR named "Dual wield (FAR)"
output AIM > DVCLOSE named "Headshot (CLOSE)"
output AIM > DVMID named "Headshot (MID)"
output AIM > DVFAR named "Headshot (FAR)"

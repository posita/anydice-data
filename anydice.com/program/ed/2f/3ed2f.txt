\ 0 is a miss, 1 is a weak hit, 2 is a strong hit \


function: action AROLL:n AMOD:n CROLL:s {
  result: (AROLL + AMOD) > CROLL
}
function: actionb AROLL:n AMOD:n CROLL:s {
  result: (AROLL + AMOD) > CROLL
}


loop X over {-1..5} {
  output [actionb 2d6 X {6,9}] named "2d6 Modifier [X]"
  output [action 1d8 X 2d10] named "d10 vs 2d12 Modifier [X]"
}

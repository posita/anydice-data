function: performance ROLL:n plus MOD:n inspo INS:n four FOUR:n {

if FOUR < 4 {
  if ROLL+MOD+INS <= 12 {result: 1d4}
  else if ROLL+MOD+INS <= 18 {result: 10*(2d8+4)}
  else if ROLL+MOD+INS <= 23 {result: 10*(3d12+10)}
  else {result: 10*(5d10+20)}
}
else {
  if ROLL+MOD+INS <= 16 {result: 100*(2d4)}
  else if ROLL+MOD+INS <= 21 {result: 100*(1d10+2)}
  else {result: 100*(1d12+4)}
}

}

set "position order" to "highest first"

output [performance 1@2d20 plus 7 inspo 1d8 four 1d4] named "+d8, +7"
output [performance 1@2d20 plus 10 inspo 1d8 four 1d4] named "+d8, +10"
output [performance 1@2d20 plus 9 inspo 1d10 four 1d4] named "+d10, +9"
output [performance 1@2d20 plus 13 inspo 1d10 four 1d4] named "+d10, +13"

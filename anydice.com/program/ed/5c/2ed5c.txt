function: defense
{result: 15}

function: SPEAR:n spear {

if SPEAR <[defense]
{result: [1d20+2d6 haft]}
if SPEAR >[defense]-1
{result: 3d6+[1d20+2d6 haft]}
}

function: HAFT:n haft {

if HAFT <[defense]
{result: 0}
if HAFT >[defense]-1 
{result: 1d6}
}

function: SHIELD:n shield {

if SHIELD <[defense]
{result: [d20+1d6 maul]}
if SHIELD >[defense]-1 
{result: 2d6+[d20+3d6 maul]}
}

function: MAUL:n maul {

if MAUL <[defense]
{result: 0}
if MAUL >[defense]-1
{result: 4d6}
}

function: HMAUL:n hmaul {

if HMAUL <[defense]
{result: 0}
if HMAUL >[defense]-1
{result: 6d6}
}


output [d20+2d6 shield] named "Shield Bash"
output [d20+2d6 spear] + [d20+2d6 spear] named "Haft Attack"
output [d20 hmaul]+[d20 hmaul] named "Powerful Attack"

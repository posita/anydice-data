NUMDICE: 6     \ number of dice in the pool \
DICESIZE: 10   \ size of the dice \
SETSIZE: 2     \ set of 2 (doubles), set of 3 (triples), set of 4 (quadruples), etc. \

function: countset SET:n in ROLL:s {
 MATCHES: 0 
 loop X over {1..DICESIZE}{
  if [count X in ROLL] >= SET { MATCHES: MATCHES + 1 }
 }
 result: MATCHES
}

output [countset SETSIZE in NUMDICE d DICESIZE] named "sets of [SETSIZE]"

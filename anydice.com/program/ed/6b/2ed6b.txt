function: lucky ROLL:n added ADDEDROLL:n vs DEFENSE:n {
 if LUCKY > 0 {
  if ROLL = 1 {
    LUCKY: LUCKY - 1
    result: ADDEDROLL
  } else {
    \if ROLL + TOHIT < DEFENSE {
      LUCKY: LUCKY - 1
      result: ADDEDROLL
    }\
  }
 }
 result: ROLL
}

function: attack ROLL:n added ADDEDROLL:n vs DEFENSE:n {
 FINALROLL: [lucky ROLL added ADDEDROLL vs DEFENSE]

 if FINALROLL = 1 {
  result: MISSRESULT
 }
 if FINALROLL >= CRITRANGE {
  result: CRITRESULT
 }
 if FINALROLL + TOHIT >= DEFENSE {
   result: HITRESULT
 }

 result: MISSRESULT
}

function: damage for ATTACKRESULTS:s {
 D: 0
 loop A over ATTACKRESULTS {
  if A = CRITRESULT {
   D: D + CRITDAMAGE
  }
  else if A = HITRESULT {
   D: D + HITDAMAGE
  }
  else {
   D: D + MISSDAMAGE
  }
 }
 result: D + [bonus damage for ATTACKRESULTS]
}

function: bonus damage for ATTACKRESULTS:s {
 if [ATTACKRESULTS contains CRITRESULT] {
  result: 2 * ONCEDAMAGE
 }
 if [ATTACKRESULTS contains HITRESULT] {
  result: ONCEDAMAGE
 }
 result: 0
}

once ONCE:d advantage ADVANTAGE:n vs TARGET:n {
 TOHIT: HIT
 HITDAMAGE: DMG
 CRITDAMAGE: CRIT
 CRITRANGE: RANGE
 ONCEDAMAGE: ONCE
 DEFENSE: TARGET
 if ADVANTAGE = ATTACKNORMAL {
   result: [damage for Nd[attack [highest 1 of 1d20] added [highest 1
of 1d20] vs DEFENSE]]
 }
 if ADVANTAGE = ATTACKADV {
   result: [damage for Nd[attack [highest 1 of 2d20] added [highest 1
of 1d20] vs DEFENSE]]
 }
 if ADVANTAGE = ATTACKDIS {
   result: [damage for Nd[attack [lowest 1 of 2d20] added [highest 1
of 1d20] vs DEFENSE]]
 }

 result: [damage for Nd[attack [highest 1 of 1d20] added [highest 1 of
1d20] vs DEFENSE]]
}

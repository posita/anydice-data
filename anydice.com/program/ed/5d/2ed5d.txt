function: defense
{result: 15}

function: bonus
{result: 2d6+3}

function: SPEAR:n spear {

if SPEAR <[defense]
{result: [1d20+[bonus] haft]}
if SPEAR >[defense]-1
{result: 3d6+[d20+[bonus] haft]}
}

function: HAFT:n haft {

if HAFT <[defense]
{result: 0}
if HAFT >[defense]-1 
{result: 1d6}
}

function: SHIELD:n shield {

if SHIELD <[defense]
{result: [d20+[bonus]-1d6 maul]}
if SHIELD >[defense]-1 
{result: 2d6+[d20+[bonus]+1d6 maul]}
}

function: MAUL:n maul {

if MAUL <[defense]
{result: 0}
if MAUL >[defense]-1
{result: 4d6}
}

function: HMAUL:n hmaul {

if HMAUL <[defense]
{result: 0}
if HMAUL >[defense]-1
{result: 6d6}
}


output [d20+[bonus] shield] named "Shield Bash"
output [d20+[bonus] spear] + [d20+[bonus] spear] named "Haft Attack"
output [d20+[bonus]-2d6 hmaul]+[d20+[bonus]-2d6 hmaul] named "Powerful Attack"

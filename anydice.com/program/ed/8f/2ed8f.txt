function: cut R:n at T:n {
  if R > T { result: 0 }
  result: R
}

function: decide R:s {
  if 1@R = 0 { result: 0 }
  result: 1@R + (R > 0) - 1
}

function: effort R:s {
  M: [ count 0 in R ]
  result: MdE
}

function: total R:s {
  M: [ count 0 in R ]
  result: [ decide R ] + MdE
}

D: [ cut 1d6 at 3 ]
E: { 1, 0 }

loop S over { 1..10 } {
  output [ decide SdD ] named "[S]d"
  output [ effort SdD ] named "[S]d Effort"
  output [ total SdD ] named "[S]d Total"
}

function: expld DIE:d with threshold O:n T:n times {
  MAX: [maximum of DIE]
  BEAT: MAX - O
  result: [expld DIE with threshold O T times helper]
}

function: expld DIEH:n with threshold OH:n TH:n times helper {
  if (TH > 0) & (DIEH > BEAT) {
    result: DIEH + [expld DIE with threshold OH (TH - 1) times]
  }
  result: DIEH
}

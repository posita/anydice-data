function: helper ROLL:n C:n {
  if ROLL > C {
     result: ROLL
  }
  else {
    result: C+1
  }
}

function: level L:n hit die H:n {
  O: 0   
  loop N over {1..L}{
    Z: N
    if N > 10 {Z: 10}
    O: [helper NdH O]
  }
  result: O
}

LVL: 12
HD: 6

loop LVLZ over {1..15}
{
output [level LVLZ hit die HD] named "adjusted [LVLZ]"
}

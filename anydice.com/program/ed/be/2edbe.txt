function: dupes in DICE:s {
  D: 0
  loop X over {2..#DICE} {
    if ((X-1)@DICE = X@DICE) { D: D + 1}
  }
  result: D
}

function: roll ROLL:s {
  DUPES: [dupes in ROLL]
  HITS: [count {8..20} in ROLL]
  if (HITS = 0) {
   MISSES: DUPES*-1 - #ROLL
   if (MISSES < -4) { result: -4 } else { result: MISSES }
  }
  result: HITS + DUPES - #ROLL
}

output [roll 2d20]

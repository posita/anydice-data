function: defense
{result: 15}

function: attack
{result: 1d20+2d6+3}

function: bonus
{result: 2d6}

function: PIKE:n pike {

if PIKE <[defense]
{result: [[attack]-1d6 haft]}
if PIKE >[defense]-1
{result: 1d6+[bonus]+[[attack]-1d6 haft]}
}

function: HAFT:n haft {

if HAFT <[defense]
{result: 0}
if HAFT >[defense]-1 
{result: 1d6}
}

function: SHIELD:n shield {

if SHIELD <[defense]
{result: [[attack] sword]}
if SHIELD >[defense]-1 
{result: 1d6+[bonus]+[[attack]+2d6 sword]}
}

function: SWORD:n sword {

if SWORD <[defense]
{result: 0}
if SWORD >[defense]-1
{result: 1d6+2+[bonus]}
}

function: MAUL:n maul {

if MAUL <[defense]
{result: 0}
if MAUL >[defense]-1
{result: 3d6+2d6+[bonus]}
}

function: DUAL:n dual {

if DUAL <[defense]
{result: 0}
if DUAL >[defense]-1
{result: 1d6+2+[bonus]+1d6+2+[bonus]}
}

output [[attack] standard]+[[attack] standard] named "Standard"
output [[attack] pike] + [[attack] pike] named "Pike + Haft Attack"
output [[attack] shield] named "Shield Bash + Shield & Sword"
output [[attack]-2d6 maul]+[[attack]-2d6 maul] named "Powerful Attack + Maul"
output [[attack] dual]+[[attack] dual] named "Two Weapon + Swords"

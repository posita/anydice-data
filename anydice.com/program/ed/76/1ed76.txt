NUMDICE: 6     \ number of dice in the pool \
DICESIZE: 10   \ size of the dice \
MAXSETSIZE: 3  \ set of 2 (doubles), set of 3 (triples), set of 4 (quadruples), etc. \

function: countn SETSIZE:n in ROLL:s {
 MATCHES: 0 
 loop X over {1..DICESIZE}{
  if [count X in ROLL] >= SETSIZE { MATCHES: MATCHES + 1 }
 }
 result: MATCHES
}

loop SET over {2..MAXSETSIZE}{
 output [countn SET in NUMDICE d DICESIZE] named "sets of [SET]"
}

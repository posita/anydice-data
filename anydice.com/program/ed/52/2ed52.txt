function: SPEAR:n spear {

if SPEAR = {1..14}
{result: 0}
if SPEAR = {15..20} 
{result: 3d6}
}

function: SHIELD:n shield {

if SHIELD = {1..14}
{result: [d20 maul]}
if SHIELD = {15..20} 
{result: 1d3+[d20+2d6 maul]}
}

function: MAUL:n maul {

if MAUL = {1..14}
{result: 0}
if MAUL = {15..20} 
{result: 4d6}
}

output [d20+1d6 shield] named "Shield & Maul"
output [d20+1d6 spear] + [d20+1d6 spear] named "Spear"
output [d20 maul]+[d20 maul] named "Only Maul"

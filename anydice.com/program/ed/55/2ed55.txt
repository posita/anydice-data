function: plusone ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS + 2 >= AC { result: DAMAGE  }
 result: 0
}

function: onedfour ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: ACRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + 2 >= AC { result: ADAMAGE  }
 result: 0
}

ENEMYAC: d{14..25}

ATTACKMOD: 5
PROFICIENCY: 6
CRITROLL: 20
ATTACKTIMES: 2
DAMAGEBONUS: 3
ACCBONUS: 3

DAMAGE: 1d8 + ATTACKMOD + DAMAGEBONUS + 2
CRITDAMAGE: 2d8 + ATTACKMOD + DAMAGEBONUS + 2

ADAMAGE: 1d8 + ATTACKMOD + 2d4 + 2
ACRITDAMAGE: 2d8 + ATTACKMOD + 4d4 + 2

output ATTACKTIMESdDAMAGE named "+3 ammo on hit"
output ATTACKTIMESd[plusone d20 vs ENEMYAC] named "+3 v 14-25 AC"
output ATTACKTIMESdADAMAGE named "Warpstone Damage on hit"
output ATTACKTIMESd[onedfour d20 vs ENEMYAC] named "Warpstone v 14-25 AC"

set "position order" to "lowest first"
function: rerollone A:n B:n C:n D:n under N:n {
    SEQ: [sort {A, B, C}]
    if 1@SEQ > N {result: 1@SEQ + 3@SEQ}
    SEX: [sort {D, 2@SEQ, 3@SEQ}]
    result: 1@SEX + 3@SEX

}

output [rerollone 1d10 1d10 1d10 1d10 under 2] named "3d10 reroll 1 or 2 once, edges"

function: edges K:d
{result: 1@K + (#K)@K}

output [edges 3d10] + 1 named "3d10+1, edges"

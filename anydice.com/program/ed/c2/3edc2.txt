function: success of ROLL:s of size Z:n with level L:n {
  S: 0
  NO_ONES: ![ROLL contains 1]
  loop DIE_VALUE over {1..Z} {
    COUNT: [count DIE_VALUE in ROLL]
    if DIE_VALUE = 1{
      S: S - #(ROLL) * COUNT
    }
    if DIE_VALUE >= (4 - (L >= 1)){
      S: S + COUNT
    }
    if DIE_VALUE >= (6 - (L >= 2)){
      S: S + COUNT
    }
    if DIE_VALUE >= (8 - (L >= 3)){
      S: S + COUNT
    }
    if DIE_VALUE >= (10 - (L >= 4)){
      S: S + COUNT
    }
    if DIE_VALUE >= (12 - (L >= 5)){
      S: S + COUNT
    }
    if DIE_VALUE >= (20 - (L >= 6)){
      S: S + COUNT
    }
  }
  MAX_CNT: ROLL = Z
  if MAX_CNT >= 2 & NO_ONES{
    S: S * 2
  }
  if S < 0{
    S: 0
  }
  result: S
}
function: roll N:n dice of tier T:n with level L:n{
  Z: (2+T*2)
  if T=6{
    Z: 20
  }
  result: [success of NdZ of size Z with level L]
}
output [roll 1 dice of tier 6 with level 0]

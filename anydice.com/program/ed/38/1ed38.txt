function: ROLL:n gwf REROLL:d reroll SEQUENCE:s{
  if [SEQUENCE contains ROLL] {
    result: REROLL
  }
  else {
    result: ROLL
  }
}

STR: 4
DIE_ONE: [d6 gwf d6 reroll {1,2}]
DIE_TWO: [d6 gwf d6 reroll {1,2}]
DIE_THREE: [d6 gwf d6 reroll {1,2}]
DIE_FOUR: [d6 gwf d6 reroll {1,2}]
RESULT_ONE: DIE_ONE+DIE_TWO
RESULT_TWO: DIE_THREE+DIE_FOUR
output [highest of RESULT_ONE and RESULT_TWO] named "savage"
output DIE_ONE+DIE_TWO named "Greatsword"

X: d{0..9}
Y: d6

function: A:n vs B:n {
  if A>B {
    result: A
} else {
    result: 0
} }

function: test C:n {
  if C > 6 {
    result: 2
} if C > 3 {
    result: 1
} else {
    result: 0
} }

Z: d[Y vs X]
K: 0

loop I over{1..6}{
  N: I-K
  if N<1 {
    output Id[test Z]
} else {
    output Kd[test Z] + Nd[test Y]
} }

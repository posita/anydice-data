function: dupes in DICE:s {
  D: 0
  loop X over {2..#DICE} {
    if ((X-1)@DICE = X@DICE) { D: D + 1}
  }
  result: D
}

function: roll ROLL:s {
  DUPES: [dupes in ROLL]
  HITS: [count {11..20} in ROLL]
  if (HITS = 0) {
   result: DUPES*-1 - #ROLL
  }
  result: HITS + DUPES - #ROLL
}

output [roll 2d20]

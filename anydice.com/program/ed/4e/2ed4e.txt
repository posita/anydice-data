
function: drama roll N:n {
  A: dN
  B: dN
  result: [drama compare helper A B]
}

function: drama compare helper A:n B:n {
  MID: N/2
  if (A = B) {
    result: A
  }

  ABS_A: [absolute (A - MID)]
  ABS_B: [absolute (B - MID)]
  if ( ABS_A > ABS_B ) {
    result: A
  }
  if ( ABS_B > ABS_A ) {
    result: B
  }

  result: -1
}

output [drama roll 20]

function: success of ROLL:s when using dice F:n {
  if [ROLL contains 1] { result: 0 }
  S: 0
  loop DIE_VALUE over {4..F} {
    COUNT: ROLL = DIE_VALUE
    WEIGHT: ((DIE_VALUE - 4) / 2 ) + 1
    S: S + COUNT * WEIGHT
  }
  MAX_CNT: ROLL = F
  if MAX_CNT >= 2 {
    S: S * MAX_CNT
  }
  result: S
}

output [success of 5d8 when using dice 8]

function: ROLL:n vs AC:n {
 if ROLL + ATTACK >= AC | ROLL = 20 { result: [hit d20] }
 result: 0
}

function:  hit ROLL:n {
 if ROLL = 20 {
  result: CRITICAL d DAMAGE
 }
 result: DAMAGE
}

AC: d{10..20}
STRENGTH: 5
ATTACK: 3 + STRENGTH

DAMAGE: 4d6 + STRENGTH
THREAT: 20
CRITICAL: 2
output [d20 vs AC] named "Shortsword"

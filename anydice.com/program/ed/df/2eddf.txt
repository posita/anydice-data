MISSRESULT: 0
HITRESULT: 1

function: marks for DICERESULTS:s range RANGE:n {
  MARKS: 0

  loop P over {1..10} {
    NUM: [count P in DICERESULTS]
    if NUM >= 2 {
      MARKS: MARKS + NUM
    }

    if P >= RANGE {
      MARKS: MARKS + NUM
    }
  }

  result: MARKS
}


function: strikehit for DICERESULTS:s  {
  if 1@DICERESULTS + TOHIT >= TARGET {
    result: 1
  } else {
    result: 0
  }
}


function: strikedamage for DICERESULTS:s {
  MARKS: [marks for DICERESULTS range CRITRANGE]
  DIDHIT: [strikehit for DICERESULTS]

  CRITDAMAGE: 0
  INDEX: 1
  loop A over CRITDAMAGES {
    if MARKS = INDEX {
      CRITDAMAGE: A
    }
    INDEX: INDEX + 1
  }
 
  if DIDHIT = 1 {
    result: MARKS + BASEDAMAGE + CRITDAMAGE - DAMAGERESISTANCE
  } else {
    result: 0
  }
}

function: N:n dicestrike hit HIT:n damage DAMAGE:n range RANGE:n crits CRITS:s vs GUARD:n armor ARMOR:n {
  TOHIT: HIT
  BASEDAMAGE: DAMAGE
  CRITRANGE: RANGE
  CRITDAMAGES: CRITS
  TARGET: GUARD
  DAMAGERESISTANCE: ARMOR

  result: [strikedamage for Nd10]
}

function: blastdamage for DICERESULTS:s {
  SUM: 1@DICERESULTS

  if HARDSTRINGS + 1 < DAMAGERESISTANCE {
    result: 0
  }

  if SUM + TOHIT >= TARGET {
    \ assume each wide string gives an additional target \
    result: (HARDSTRINGS + 1 - DAMAGERESISTANCE) * (WIDESTRINGS + 1)
  } else {
    result: 0
  }
}

function: N:n diceblast hit HIT:n hard HARD:n soft SOFT:n wide WIDE:n long LONG:n armor ARMOR:n {
  TOHIT: HIT
  TARGET: 5 + HARD + SOFT + WIDE + WIDE + LONG
  HARDSTRINGS: HARD
  SOFTSTRINGS: SOFT
  WIDESTRINGS: WIDE
  LONGSTRINGS: LONG
  DAMAGERESISTANCE:ARMOR

  result: [blastdamage for Nd10]
}

function: invokedamage for DICERESULTS:s {
  DAMAGE: 0
  loop VALUE over DICERESULTS {
    if VALUE + TOHIT >= TARGET {
      DAMAGE: DAMAGE + 3
    } else {
      DAMAGE: DAMAGE + MISSDMG
    }
  }

  result: DAMAGE
}

function: N:n diceinvoke hit HIT:n missdamage MISSDAMAGE:n vs GUARD:n {
  TOHIT: HIT
  TARGET: GUARD
  MISSDMG: MISSDAMAGE
  
  result: [invokedamage for Nd10]
}

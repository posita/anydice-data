function: fifth edition attack ROLL:n {
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*(DMG_DICE + DMG_MOD)}
if ROLL+TO_HIT >= AC {result: DMG_DICE + DMG_MOD}
result: 0}

DMG_DICE: 1d8
DMG_MOD: 3 + 7d6 + 4d8
TO_HIT: 5
CRIT: 20

loop AC over {8..23} {
Y:0
W:0

loop X over {1..2} {
  Y: Y + [fifth edition attack d20]
  W: W + [fifth edition attack [highest 1 of 2d20]]
}

DMG_MOD: 7d6 + 4d8

loop Z over {1..2} {
  Y: Y + [fifth edition attack d20]
  W: W + [fifth edition attack [highest 1 of 2d20]]
}

output Y named "Normale [AC]"
output W named "Vantaggio [AC]"
}

ASTR: 5
ASKILL: 5    
DSTR: 5     
DSKILL: 5

function: modify attack SEQUENCE:s{
 R: {}
 loop P over {1..#SEQUENCE}{
    E: P@SEQUENCE
    if E = 0 { E: -1 }
    R: {R, E}
 }
 result: R
}

ATTACK: [count {1..ASKILL} in ASTRd12]
ATTACKMOD: [modify attack ATTACK]
DEFENSE: [count {1..DSKILL} in DSTRd12]
DIFF: ATTACK - DEFENSE
DIFFMOD: ATTACKMOD - DEFENSE
output DIFF
output DIFFMOD


function: double AT:n PA:n if above DICEI:n DICE:n {
 ATQ: (AT-DICEI)
 PAC: (PA-DICE)
 if PAC > ATQ{ result: -100 }
 result: (ATQ-PAC)/10
}

loop AT over {12..30}{
output {} named "-------------------------------------------------[AT]--------------------------------------------------"
loop PA over {AT-10..AT+10}{
output [double AT PA if above 1d20 1d20] named "[PA]"}
}

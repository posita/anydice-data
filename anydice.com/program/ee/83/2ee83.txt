function: ROLL:n vs AC:n {
 if ROLL = 20 {
  result: CRITICAL
 }
 if ROLL + ATTACK >= AC { 
  result: DAMAGE
 }
 result: 0
}

\ Target AC \
AC: d{13..18}

\ Proficiency Bonus \
PROF: 2

\ Mod +3 \
MOD: 3
ATTACK: PROF + MOD
DAMAGE: 1d8 + MOD
CRITICAL: 2d8 + MOD
output [d20 vs AC] named "MOD +3"

\ Mod +5 \
MOD: 5
ATTACK: PROF + MOD
DAMAGE: 1d8 + MOD
CRITICAL: 2d8 + MOD
output [d20 vs AC] named "MOD +5"

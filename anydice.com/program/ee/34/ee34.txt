function: evaluate X:s R:s {
 C: 2*(R/2)
 HALF: R/2
 BONUS: 1d6
 if C != R { HALF: HALF + 1}
 if X = 12 { result: R + BONUS}
 if X >= 9 { result: R }
 if X >= 6 { result: HALF }
 result: 0
}
loop RANK over {1..6} {
 loop DICE over {2..2} {
  XS: DICE d6
  ROLL: [highest 2 of XS]
  output [evaluate ROLL RANK] named "[DICE] Dice, Rank [RANK]"
 }
}

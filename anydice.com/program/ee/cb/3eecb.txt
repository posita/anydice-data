function: full auto attack ROLL:n {
NUM_HIT: TO_HIT - ROLL + 1
if ROLL = 1  {result: (5*DMG_DICE + 5*DMG_MOD)-5*armor}
if ROLL >= CRIT {result: 2*DMG_DICE + DMG_MOD}
if ROLL+TO_HIT >= AC {result: DMG_DICE + DMG_MOD}
result: 0}

DMG_DICE: 1d8
DMG_MOD: 3
TO_HIT: 5
CRIT: 10

loop armor over {1..10} {
 output [full auto attack d10] named "[AC] AC"
}

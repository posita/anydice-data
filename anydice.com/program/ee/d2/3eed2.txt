DMG_DICE: 1d10
DMG_MOD: 3
BS: 4
CRIT: 10

function: check if crit DMGROLL:n {
if DMGROLL = 10 {result: DMGROLL + 1d5}
else {result: DMGROLL}
}

function: full auto attack ROLL:n {

TO_HIT : BS + 1
if TO_HIT - ROLL < 0  {result: 0}
if TO_HIT - ROLL >= 0 {result: (TO_HIT - ROLL + 1)*([check if crit DMG_DICE] + DMG_MOD - ARMOR)}
result: 0}

function: check if positive DMG:n {
if DMG < 0 {result: 0} else {result: DMG}
}

function: accurate attack ROLL:n {
TO_HIT : BS + 4
if TO_HIT - ROLL < 0  {result: 0}
if TO_HIT - ROLL >= 0 & TO_HIT - ROLL < 2 {result: [check if crit DMG_DICE] + DMG_MOD - ARMOR}
if TO_HIT - ROLL >= 2 & TO_HIT - ROLL < 4 {result: 2*([check if crit DMG_DICE]) + DMG_MOD - ARMOR}
if TO_HIT - ROLL >= 4 {result: 3*([check if crit DMG_DICE]) + DMG_MOD - ARMOR}
result: 0}

loop ARMOR over {1..10} {
output [check if positive [full auto attack d10]] named "fullauto [ARMOR] ARMOR "
}

loop ARMOR over {1..10} {
output [check if positive [accurate attack d10]] named "accurate [ARMOR] ARMOR "
}

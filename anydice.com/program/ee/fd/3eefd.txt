function: result of A:n and B:n versus TARGET:n {
  LOW: (A <= TARGET) + (B <= TARGET)
  IS_DOUBLE: A = B
  RESULT: 2

  if LOW = 2 {
    if IS_DOUBLE { RESULT : 4 }
    else { RESULT : 3 }
  }
  if LOW = 0 {
    if IS_DOUBLE { RESULT : 0 }
    else { RESULT : 1 }
  }

  result: RESULT
}

loop TARGET over {2..8} {
  output [result of d10 and d10 versus TARGET] named "Target <= [TARGET]"
}

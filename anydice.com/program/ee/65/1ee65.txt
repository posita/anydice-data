A: {4, 5, 7, 10, 12, 15, 16, 18, 20, 21, 24, 28, 35, 36, 42, 49, 54, 56, 64, 72}
B: {1, 2, 3, 6, 8, 9, 12, 14, 18, 24, 25, 27, 30, 32, 36, 40, 45, 48, 63, 81}
C: {4, 6, 8, 9, 12, 16, 21, 25, 27, 28, 32, 35, 36, 42, 48, 49, 54, 56, 64, 72}

NUMBERS: {1..9}

function: ROLL:n is divisible by DIVISOR:n {
  \ also checks that the quotient is another valid choice in NUMBERS \
  QUOTIENT: ROLL / DIVISOR
  result: ROLL = QUOTIENT * DIVISOR & NUMBERS = QUOTIENT
}

loop N over NUMBERS {
  output [dA is divisible by N] | [dB is divisible by N] | [dC is divisible by N]
    named "[N] is a possible factor"
}

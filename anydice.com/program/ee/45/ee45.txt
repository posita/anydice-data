CRIT_CHANCE: 1+(1d20=20)

FIRST_SMITE_UNDEAD: 3d8*CRIT_CHANCE
SECOND_SMITE_UNDEAD: 4d8*CRIT_CHANCE
BRANDING_SMITE: 2d6*CRIT_CHANCE
GUIDED_STRIKE: 10

TUGARIN_HIT_MAGIC: 1d20+3+4+1
TUGARIN_DAMAGE_MAGIC: CRIT_CHANCE*1d8+4+2+1
XI_HIT_MAGIC: [highest 1 of 2d20]+2+3+1
XI_DAMAGE_MAGIC: CRIT_CHANCE^2*1d12+2+1
FATHEL_HIT: 1d20+3+3
FATHEL_DAMAGE: (CRIT_CHANCE*1d8+3)/2
ENEMY_AC: 20
RADIANT_MULTIPLIER:2


\Assumptions:\
\Blessing of the Forge on Tugarin, Magic Weapon Spell on Xi. \
\Everyone on their turn casts Branding Smite, Uses Guided Strike on attacks of 10-19, 2nd Level Smites\
\Deathwalker is Vulnerable to Radiant, Resistant to non-magic B/S/P assumed 20AC\


TUGARIN_FIRST_STRIKE: ((TUGARIN_HIT_MAGIC+GUIDED_STRIKE)>=ENEMY_AC)*(TUGARIN_DAMAGE_MAGIC+(SECOND_SMITE_UNDEAD+BRANDING_SMITE)*RADIANT_MULTIPLIER)
TUGARIN_SECOND_STRIKE: (TUGARIN_HIT_MAGIC>=ENEMY_AC)*(TUGARIN_DAMAGE_MAGIC+(FIRST_SMITE_UNDEAD)*RADIANT_MULTIPLIER)
TUGARIN_TURN: TUGARIN_FIRST_STRIKE+TUGARIN_SECOND_STRIKE


XI_FIRST_STRIKE: ((XI_HIT_MAGIC+GUIDED_STRIKE)>=ENEMY_AC)*(XI_DAMAGE_MAGIC+(SECOND_SMITE_UNDEAD+BRANDING_SMITE)*RADIANT_MULTIPLIER)
XI_SECOND_STRIKE: (XI_HIT_MAGIC>=ENEMY_AC)*(XI_DAMAGE_MAGIC+(FIRST_SMITE_UNDEAD)*RADIANT_MULTIPLIER)
XI_TURN: XI_FIRST_STRIKE+XI_SECOND_STRIKE


FATHEL_FIRST_STRIKE: ((FATHEL_HIT+GUIDED_STRIKE)>=ENEMY_AC)*(FATHEL_DAMAGE+(SECOND_SMITE_UNDEAD+BRANDING_SMITE)*RADIANT_MULTIPLIER)
FATHEL_SECOND_STRIKE: (FATHEL_HIT>=ENEMY_AC)*(FATHEL_DAMAGE+(FIRST_SMITE_UNDEAD)*RADIANT_MULTIPLIER)
FATHEL_TURN: FATHEL_FIRST_STRIKE+FATHEL_SECOND_STRIKE

ROUND_ONE_DAMAGE: TUGARIN_TURN+XI_TURN+FATHEL_TURN

output ROUND_ONE_DAMAGE

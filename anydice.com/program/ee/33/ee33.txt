function: evaluate X:s R:s {
 C: 2*(R/2)
 HALF: R/2
 if C != R { HALF: HALF + 1}
 if X >= 6 { result: R }
 if X >= 4 { result: HALF }
 result: 0
}
loop RANK over {1..6} {
 loop DICE over {1..RANK} {
  MODIFIED: RANK - DICE + 1
  XS: DICE d6
  ROLL: [highest 1 of XS]
  output [evaluate ROLL MODIFIED] named "[DICE] Dice, Rank [RANK]"
 }
}

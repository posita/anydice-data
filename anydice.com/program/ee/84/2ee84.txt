BASE_DC: 0
WIND_DC: 0 \(1d10 - 5)\
STEP_DC: 2
ATK: 1
DEF: 5

DC: BASE_DC + WIND_DC + ATK + DEF

loop STEP_COUNT over {0..7}{
  output 1d20 + 0 >= (DC + (STEP_COUNT * STEP_DC)) named "[STEP_COUNT] Steps"
  output 1d20 + 0 >= (DC + ATK + (STEP_COUNT * STEP_DC)) named "[STEP_COUNT] Steps ATK"
  output 1d20 + 0 >= (DC + ATK + DEF + (STEP_COUNT * STEP_DC)) named "[STEP_COUNT] Steps ATK DEF"
}

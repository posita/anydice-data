function: evaluate X:s R:s {
 C: 2*(R/2)
 HALF: R/2
 if C != R { HALF: HALF + 1}
 if X >= 6 { result: R }
 if X >= 4 { result: HALF }
 result: 0
}
loop RANK over {1..8} {
 C: 2*(RANK/2)
 DICE: RANK / 2
 if C != RANK { DICE: DICE + 1}
 XS: DICE d6
 ROLL: [highest 1 of XS]
 output [evaluate ROLL RANK] named "[DICE] Dice, Rank [RANK]"
}

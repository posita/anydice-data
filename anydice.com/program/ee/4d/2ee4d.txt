\ this rolls 2d6 and compares to a roll of 1d10, then the same 1d10 + the target's defense. The first roll must be higher to be considered a hit \
\ 0 is a miss, 1 is a weak hit, 2 is a strong hit \


function: action AROLL:n CROLL:n DEF:n {
  result: 1d(AROLL > CROLL)+1d(AROLL > CROLL + DEF)
}

loop X over {1..5} {
  output [action 2d6 1d10 X] named "Action 2d6 against 1d10+[X]"
}

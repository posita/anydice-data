function: reroll DIE:d under LIMIT:n {
  result: [reroll DIE as DIE if under LIMIT]
}

function: reroll ROLL:n as DIE:d if under LIMIT:n {
  if ROLL < LIMIT { result: DIE }
  else { result: ROLL }
}

function: crit D:d on N:n{
result: [critcheck d20 D as D on N]
}

function: critcheck D:n N:n as M:n on C:n{
 if D<C{result: N} 
 result: N+M+[reroll d6 under 3]
}

function: hit N:n against M:n crit on X:n{
 result: [hitcheck N against M with d20 crit on X]
}

function: hitcheck N:n against M:n with D:n crit on X:n{
 if D<M{result: 0} else { result: [critcheck D N as N on X]}
 result: N
}

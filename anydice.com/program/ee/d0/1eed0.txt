set "maximum function depth" to 80

function: rare N:n {
  if N < 120 {result: 0}
  if N = 120 {result: 1}
}

function: tail N:n {
  if N < 8 {result: 0}
  if N >= 8 {result: 1 + [tail d8]}
}

function: poisson N:n {
  if N < 4 {result: 0}
  if N < 7 {result: 1}
  if N < 8 {result: 2}
  if N = 8 {result: 3 + [tail d8]}
}

output [lowest of 2d[poisson d8] and 8]
output [lowest of 240d[rare d120] and 8]

function: explode DIE:d {
 EXPLODE: 1
 result: [explode DIE helper]
}

function: explode N:n helper {
 if N = EXPLODE { result: N + [explode DIE helper] }
 result: N
}
loop N over {1..9}{
X: 2+2*((N+2)/2)
Y: 2+2*((N+1)/2)
loop DIFFICULTY over{8..12}{
output [explode(d(X))]+[explode(d(Y))]>=DIFFICULTY named "[N]>=[DIFFICULTY]"
}
}

function: A:n attacks B:n {
 if A > SKILL_A {
  result: -1 - (B <= SKILL_B)
 }
 result: B > SKILL_B - A
}

SKILL_A: 20
SKILL_B: 20
output [d100 attacks d100]
\ -2 = A fails, B succeeds: some reverse effect?       \
\ -1 = A and B fail: nothing?                          \
\  0 = A and B succeed: mitigation, reverse effect?    \
\  1 = A succeeds, B fails: val(A) * val(B)/100 damage \

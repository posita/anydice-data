function: reroll N :n as M:d{
 if N < 3 { result: M }
 result: N
}

function: critcheck D:n N:n as M:n{
 if D<19{result: N} 
 result: N+M+[reroll d6 as d6]
}
function: critcheckdouble D:n N:n as M:n{
 if D<19{result: N} 
 result: N+N+[reroll d6 as d6]
}

Z: [reroll d6 as d6]+[reroll d6 as d6]+[reroll d4 as d4]+[reroll d4 as d4]

function: daje LUCK:n with MOD:n {
 if LUCK+MOD > 10 {
  result: [highest 3 of 4d6] + MOD
 } else {
  result: [lowest 3 of 4d6] + MOD
 }
}

output 1d20 named "dnd roll"

loop M over {0..10}
{
 output [daje 1d20 with M] named "daje with mod [M]"
}

output [highest 1 of 2d20] + 10 named "highest 1 of 2d20+10"

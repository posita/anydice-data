BONUSA: 2
BONUSB: 2 

function: X:n joust Y:n {
  if (X+BONUSA) - (Y+BONUSB) >= 16 {result: 3}
  if (X+BONUSA) - (Y+BONUSB) >= 6 & (X+BONUSA) - (Y+BONUSB) < 16 {result: 1}
  if (X+BONUSA) - (Y+BONUSB) > 0 & (X+BONUSA) - (Y+BONUSB) < 6 {result: 0}
  if (X+BONUSA) - (Y+BONUSB) = 0  {result: 8}
  if (X+BONUSA) - (Y+BONUSB) < 0 & (X+BONUSA) - (Y+BONUSB) > -6 {result: 0}
  if (X+BONUSA) - (Y+BONUSB) <= -6  & (X+BONUSA) - (Y+BONUSB) > -16 {result: -1}
  if (X+BONUSA) - (Y+BONUSB) <= -16  {result: -3}
}

output [1d20 joust 1d20] named "Joust"

function: ROLL:n success chart {
 \result: ROLL\
 if ROLL < 7 {result: 0} 
 if ROLL < 12 {result: 1} 
 if ROLL > 11 {result: 2}
}

function: ROLL:n injury chart {
 if ROLL < 2 {result: 0} 
 if ROLL < 7 {result: 1} 
 if ROLL < 9 {result: 2} 
 if ROLL >= 9 {result: 3}
}

loop I over {0..3}{
 output [2d6-I injury chart] named "2d6 - [I] Injury Chart"
 loop N over {1..6}{
  Y: 2+N 
  output [[highest 2 of Yd6]-I injury chart] named "2d6 + [N] DICE  - [I] Injury Chart" 
 }

 output [3d6-I injury chart] named "3d6 (Blood Bath) - [I] Injury Chart" 
 loop N over {1..3}{
  Y: 3+N 
  output [[highest 3 of Yd6]-I injury chart] named "3d6 (Blood Bath) + [N] DICE - [I] Injury Chart"
 }
}

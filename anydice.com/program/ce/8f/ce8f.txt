LOW: 8
FIRST: d12
SECOND: d12
HIT: d100 > 48

function: savage X:n {
  result: [highest of X and 1d12] + 10
}

function: damage HITS:n FIRST:n SECOND:n LOW:n {
  if HITS = 2 {
    if FIRST < LOW {
      result: [savage FIRST] + SECOND + 10
    }
    else {
      result: FIRST + [savage SECOND] + 10
    }
  }
  else if HITS = 1 {
    if FIRST < LOW {
      result: [savage FIRST]
    }
    else {
      result: FIRST + 10
    }
  }
  else {
    result: 0
  }
}

output [damage 2dHIT FIRST SECOND LOW]

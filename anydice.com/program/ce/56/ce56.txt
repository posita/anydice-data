function: count CHALLENGE plus in ROLL:s mod MOD{
 SUCCESSES: ROLL >= CHALLENGE
 if [count {CHALLENGE-MOD..CHALLENGE-1} in ROLL] > 0 {
  MOD_SUCCESSES:1
 }
 else{
  MOD_SUCCESSES:0
 }
 result: MOD_SUCCESSES + SUCCESSES
}

\=================== User Options ===================\
 USE_MOD: 1      \Whether or not to use the mod system\
 CHALLENGE: 6    \What challenge level to check against\
\=======================================================\

loop DICE over {1..10} {
 if USE_MOD = 1 {MOD: (DICE-3)/2} else {MOD: 0}
 output [count CHALLENGE plus in DICEd6 mod MOD ] named "[DICE]d"
}

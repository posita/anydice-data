CHECK: 11

function: heal ROLL:n vs DC:n with HEAL:n {
if ROLL = 1 { result: -1d8}
if ROLL = 20 { result: 4d8+HEAL }
if ROLL + CHECK >= DC + 10 { result: 4d8+HEAL }
if ROLL + CHECK >= DC { result: 2d8+HEAL }
result: 0
}

output [heal d20 vs 15 with 0] named "VS DC 15"
output [heal d20 vs 20 with 10] named "VS DC 20"
output [heal d20 vs 30 with 30] named "VS DC 30"
output [heal d20 vs 40 with 50] named "VS DC 40"

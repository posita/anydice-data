function: success A:n and B:n{
if A < B {
result: 1
}
result: 2
}

function: A:n greater than B:n{
result: A >= B
}

function: A:n under threshold B:n{
if B<A{
result:0
}
result: A
}

function: attack A:n threshold B:n defense C:n threshold D:n{
E:[A under threshold B]
if [E greater than [C under threshold D]]{
if E >0{
result: 1
}
}
result: 0
}

function: forgiving attack A:n threshold B:n defense C:n threshold D:n{
E:[A under threshold B]
F:[C under threshold D]
if [A greater than C]{
if E > 0{
if F > 0{
result: 4
}
result: 3
}
if F > 0{
result: 2
}
result: 1
}
result: 0
}

function: attack A:n difficulty B:n defense C:n difficulty D:n{
if [A greater than C]{
F:[success A and B]
E:[success C and D] 
if F = 2{
if E = 2{
result: 4
}
result: 3
}
if E = 2{
result: 2
}
if E = 1{
result: 1
}
}
result: 0
}

}

function: A:n and D:n resolution{
DIF: A-D
if DIF>6{
result: 2
}
if DIF>0{
result: 1
}
if DIF<-6{
result: -2
}
result: -1
}

function: ve ATTACK:s AMOD:n DAMAGE:s DMOD:n DEF:n CRIT:n{
  HIT: (ATTACK+AMOD)
  DMG: (DAMAGE+DMOD)
  if HIT = 20 { result: CRIT }  \ CRITICAL \
  if HIT >= DEF { result: DMG } \ HIT \
  result: 0                     \ FAIL \
}

output [ve 1d20 1 1d4 0 10 5] named "Hechicero"
output [ve [highest 1 of 2d20] 1 1d4 0 10 5] named "Hechicero - ADV"
output [ve [lowest 1 of 2d20] 1 1d4 0 10 5] named "Hechicero - DIS"

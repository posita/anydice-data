function: ROLL:n success chart {
 if ROLL < 7 {result: 0} \fail\
 if ROLL < 12 {result: 1} \success\
 if ROLL > 11 {result: 2} \crit\
}

function: ROLL:n injury chart {
 if ROLL < 2 {result: 0} \No effect\ 
 if ROLL < 7 {result: 1} \Minor Hit\ 
 if ROLL < 9 {result: 2} \Down\
 if ROLL >= 9 {result: 3} \OOA\
}

\output [2d6 success chart] named "2d6 Success Chart" 
loop N over {1..6}{
 Y: 2+N 
 output [[highest 2 of Yd6] success chart] named "2d6 + [N] DICE Success Chart" 
}\

\output 2d6 named "2d6" 
loop N over {1..6}{
 Y: 2+N 
 output [highest 2 of Yd6] named "2d6 + [N] DICE" 
}\


output [2d6 injury chart] named "2d6 Injury Chart" 
loop N over {1..6}{
 Y: 2+N 
 output [[highest 2 of Yd6] injury chart] named "2d6 + [N] DICE Injury Chart" 
}

output [3d6 injury chart] named "3d6 (Blood Bath) Injury Chart" 
loop N over {1..3}{
 Y: 3+N 
 output [[highest 3 of Yd6] injury chart] named "3d6 (Blood Bath) + [N] DICE Injury Chart" 
}

function: roll D:s T:n {
 if 2@D > T {result: 2}
 if 1@D > T {result: 1}
 else {result: 0}
}

function: pbta R:n B:n F:n S:n {
 if R+B>S {result: 2}
 if R+B<F {result: 0}
 else {result: 1}
}

function: iron R:s T:n {
 if 2@R > T {result: 2}
 if 1@R > T {result: 1}
 else {result: 0}
}

function: steel R:s T:n {
 result: [count {1,1..T} in R] - [count 10 in R]
}

function: pool N:n {
 result: [count {9..10} in Nd10]
}

function: plus B:n {
 result: [pbta 2d10 B 11 16]
}

function: adv N:n {
 result: [pbta {1,2}@(2+N)d10 0 12 17]
}

loop X over {0..5}{
 output [plus X] named "+[X]"
}

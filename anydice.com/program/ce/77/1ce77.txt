function: compare A to B {
 result: [highest of 0 and (A - B)]
}
function: fail A:n {
 if A < 0 { result: 0 - 1 }
 result: A
}

loop TALENTWERT over {0..10}{
   T:10
   U:10
   V:10
   X: TALENTWERT - ([compare T to d20] + [compare U to d20] + [compare V to d20])
   if TALENTWERT < 0 {X:0 - 1}
   output [fail X] named "TaW [TALENTWERT]"
}

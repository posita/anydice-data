function: advantage for A:n and B:n{
if A < 5 {
result: d(B)
}
result: A
}

function: disadvantage for A:n and B:n{
if A > 4 {
result: d(B)
}
result: A
}

function: true advantage for A:n{
result: [highest 1 of 2d(A)]
}

function: true disadvantage for A:n{
result: [lowest 1 of 2d(A)]
}

R: 8

output d(R) named "Straight"
output [advantage for d(R) and R] named "Advantage"
output [disadvantage for d(R) and R] named "Disadvantage"
output [true advantage for R] named "True Advantage"
output [true disadvantage for R] named "True Disadvantage"

function: base ROLL:s tn TN:n pwr PWR:n {
 result: (ROLL<=TN) + PWR
}
function: crit ROLL:s tn TN:n pwr PWR:n {
 result: (ROLL<=TN) + (ROLL=1) + PWR
}
function: double ROLL:s tn TN:n pwr PWR:n {
 DBL:TN/2
 result: (ROLL<=TN) + (ROLL<=DBL) + PWR
}
function: double crit ROLL:s tn TN:n pwr PWR:n {
 DBL:TN/2
 result: (ROLL<=TN) + (ROLL<=DBL) + (ROLL=1) + PWR
}

function: opp base ROLLA:n tn TNA:n pwr PWRA:n and ROLLB:n tn TNB:n pwr PWRB:n {
  ENDA: (ROLLA<=TNA) + PWRA
  ENDA: (ROLLB<=TNB) + PWRB

  if ENDA = ENDB {
    if ROLLA = ROLLB { result: 0 } else {
      if ROLLA > ROLLB { result: 1 } else { result: -1 }
    }
  } else {
    result: ENDA-ENDB
  }
}
function: opp crt ROLLA:n tn TNA:n pwr PWRA:n and ROLLB:n tn TNB:n pwr PWRB:n {
  ENDA: (ROLLA<=TNA) + (ROLLA=1) + PWRA
  ENDA: (ROLLB<=TNB) + (ROLLB=1) + PWRB

  if ENDA = ENDB {
    if ROLLA = ROLLB { result: 0 } else {
      if ROLLA > ROLLB { result: 1 } else { result: -1 }
    }
  } else {
    result: ENDA-ENDB
  }
}
function: opp double ROLLA:n tn TNA:n pwr PWRA:n and ROLLB:n tn TNB:n pwr PWRB:n {
  DBLA: TNA/2
  DBLB: TNB/2
  ENDA: (ROLLA<=TNA) + (ROLLA<=DBLA) + PWRA
  ENDA: (ROLLB<=TNB) + (ROLLB<=DBLB) + PWRB

  if ENDA = ENDB {

}

  DBLA: TNA/2
  DBLB: TNB/2
  ENDA: (ROLLA<=TNA) + (ROLLA<=DBLA) + (ROLLA=1) + PWRA
  ENDA: (ROLLB<=TNB) + (ROLLA<=DBLB) + (ROLLB=1) + PWRB

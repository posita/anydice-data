function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

DROP_ONE_TWO:[reroll {1..2} on 1d6]

MODIFIER:0

output 2d6+MODIFIER named "Standard"

output [reroll {1..8-MODIFIER} on 2d6]+MODIFIER named "Full Reroll"

output 2dDROP_ONE_TWO+MODIFIER named "Reroll One and Two"

output [highest 2 of 3d6]+MODIFIER named "Drop Lowest"

output [reroll {1..8-MODIFIER} on [highest 2 of 3d6]]+MODIFIER named "Drop Lowest + Full Reroll"

output [highest 2 of 3dDROP_ONE_TWO]+MODIFIER named "Drop Lowest + Reroll One and Two"

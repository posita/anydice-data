loop SKILL over {10..20}{
loop DIF over {0..6}{
    if DIF = 0 {
         output 2d10 <= SKILL named "SKILL = [SKILL]; DIF = [DIF]"
    } else if DIF = 1 {
         output 2d10 + 1d4 <= SKILL named "SKILL = [SKILL]; DIF = [DIF]"
    } else if DIF = 2 {
         output 2d10 + 1d8 <= SKILL named "SKILL = [SKILL]; DIF = [DIF]"
    } else if DIF = 3 {
         output 2d10 + 1d12 <= SKILL named "SKILL = [SKILL]; DIF = [DIF]"
    } else if DIF = 4 {
         output 2d10 + 1d12 + 1d4 <= SKILL named "SKILL = [SKILL]; DIF = [DIF]"
    }
}}

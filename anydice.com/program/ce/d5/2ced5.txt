function: respf A:n {
 if A < 3 {
  result: 50 * A
 }
 else {
  result: 200
 }
}

function: resw A:n {
 if A < 10 {
  result: 2/10
 }
 if A < 15 {
  result: 1
 }
 if A < 20 {
  result: 2
 }
 else {
  result: 27
 }
}

function: resss A:n {
 if A < 11 {
  result: -75
 }
 if A < 21 {
  result: 125
 }
 else {
  result: 325
 }
}

EARNEDPF : 0
EARNEDW : 0
EARNEDSSTIO : 0
EARNEDSSMAI : 0

loop N over {1..20}{
 EARNEDPF : EARNEDPF + [respf ((d20 + 7 < 5 + 2d10) + (d20 + 7 < 5 + 2d10) + (d20 + 2 + d10 < 5 + 2d10))]
 EARNEDW : EARNEDW + [resw (d20 + 7)]
 EARNEDSSTIO : EARNEDSSTIO + [resss (d20 + 6)]
 EARNEDSSMAI : EARNEDSSMAI + [resss (d20 + 2)]
}
output EARNEDPF named "Pit-Fighting"
output EARNEDW  named "Working"
output EARNEDSSTIO  named "Spell scribing TIO"
output EARNEDSSMAI  named "Spell scribing Mailia"

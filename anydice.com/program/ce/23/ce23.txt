function: success A:n and B:n{
if A < B {
result: 1
}
result: 2
}

function: A:n greater than B:n{
result: A >= B
}

function: A:n under threshold B:n{
if B>A{
result:0
}
result: A
}

function: attack A:n threshold B:n defense C:n threshold D:n{
if [A under threshold B]{
if [A greater than C]{
result: 1
\Hit\
}
if ![C under threshold D]{
result: 1
\Hit\
}
}
result: 0
\Miss\
}

function: attack A:n difficulty B:n defense C:n difficulty D:n{
if [A greater than C]{
F:[success A and B]
E:[success C and D] 
if F = 2{
if E = 2{
result: 4
}
result: 3
}
if E = 2{
result: 2
}
if E = 1{
result: 1
}
}
result: 0
}

\0- Miss, 1- Hit\

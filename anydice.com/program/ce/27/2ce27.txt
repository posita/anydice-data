function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

DROP_ONE_TWO:[reroll {1..2} on 1d6]

output 2d6 >= 9 named "Standard"

output [reroll {1..8} on 2d6] >= 9 named "Full Reroll"

output 2dDROP_ONE_TWO >= 9 named "Reroll One and Two"

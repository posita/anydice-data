SECONDARY: FALSE
MISS: 0
HIT: 1
CRIT: 2
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n with BONUS:n {
 if ATTACKRESULT = CRIT {
  if BONUS = TRUE {
   result: SECONDARY_CRIT_DAMAGE
  }
  result: PRIMARY_CRIT_DAMAGE + ABILITY
 }
 if ATTACKRESULT = HIT {
  if BONUS = TRUE {
   result: SECONDARY_DAMAGE
  }
  result: PRIMARY_DAMAGE + ABILITY
 }
 result: MISS_DAMAGE
}

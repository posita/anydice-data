function: set element I:n in SEQ:s to N:n {
  NEW: {}
  loop J over {1 .. #SEQ} {
    if I = J { NEW: {NEW, N} }
    else { NEW: {NEW, J@SEQ} }
  }
  result: [sort NEW]
}

function: roll sequence length L:n{
  T:1
  NEW: {}
  AVGABILITY: Ld[highest 3 of 4d6]
  loop I over {1..L}{
    NEW: {NEW,  I@AVGABILITY }
  }
  
  result: AVGABILITY
}

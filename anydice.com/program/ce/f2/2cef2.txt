function: AMOUNT:n dice lower ADJUST:n {
 if 10-ADJUST > 20 {result: d{}}
 result: AMOUNTd{8-ADJUST:9-ADJUST, 10-ADJUST..20}
}

function: dice adjusted ADJUST:n {
 result: [1 dice lower ADJUST] + ADJUST
}

function: dice adjusted ADJUST:n with advantage {
 result: [highest 1 of [2 dice lower ADJUST]] + ADJUST
}

function: modifier from SCORE:n {
 result: [highest of [lowest of (SCORE/2)-5 and 10] and -5]
}


\-- Elemental Burst --\

DMG_DICE: 3d8
SAVE_DC: 14
SAVE_FOR_HALF: 1


function: saving throw ROLL:n{
  if ROLL < SAVE_DC {result: DMG_DICE}
  if SAVE_FOR_HALF {result: DMG_DICE/2}
  result: 0
}

loop MOD over {-5..SAVE_DC}{
  ROLL:(d20 + MOD)
  if MOD < 0 {output [saving throw (ROLL)] named "[MOD] Save"}
  else {output [saving throw  ROLL] named "+[MOD] Save"}
}

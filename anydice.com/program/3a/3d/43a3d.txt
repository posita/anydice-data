\ Crows power roll \
\ 0 = Doom, 1 = T1, 2 = T2, 3 = T3, 4 = Critical \

function: crows ROLL:n mod MOD:n {
  if ROLL <= 3  { result: 1 }
  if ROLL >= 19 { result: 3 }
  TOTAL: ROLL + MOD
  if TOTAL <= 11 { result: 1 }
  if TOTAL <= 16 { result: 2 }
  result: 3
}

loop M over {-8..8} {
  output [crows 2d10 mod M] named "mod [M]"
}

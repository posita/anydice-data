  if A = B & A <= 3 {
    result: 2
  }
  if A = B & A >= 4 {
    result: 9
  }
  if A = 1 {
    result: 3
  }
  if A = 2 {
    result: 3
  }
  if A = 3 {
    result: 5
  }
  if A = 4 {
    result: 5
  }
  if A = 5 {
    result: 5
  }
  if A = 6 {
    result: 7
  }
}

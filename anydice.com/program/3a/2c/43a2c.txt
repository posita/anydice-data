
WENDY_WIS_MOD: +4
WENDY_PROFICIENCY: +3
WENDY_TOTAL_ROLL_BONUS: WENDY_WIS_MOD + WENDY_PROFICIENCY

FRANK_STR_MOD: +2
FRANK_PROFICIENCY: +2
FRANK_TOTAL_ROLL_BONUS: FRANK_STR_MOD + FRANK_PROFICIENCY

WENDY_SPELL_DC: 8 + WENDY_TOTAL_ROLL_BONUS

function: frank strength save {
  result: (d20 + FRANK_TOTAL_ROLL_BONUS) >= WENDY_SPELL_DC
}

output [frank strength save] named "Does Frank win?"

output (WENDY_TOTAL_ROLL_BONUS - FRANK_TOTAL_ROLL_BONUS) named "Roll bonus difference"

function: validate X:n {
    if X >=  30 { result:  5 } \ Magnífico!!!!! \
    if X >=  25 { result:  4 } \ Fantástico!!!! \
    if X >=  20 { result:  3 } \ Excelente!!! \
    if X >=  15 { result:  2 } \ Ótimo!! \
    if X >=  10 { result:  1 } \ Bom! \
    if X >=   5 { result:  0 } \ Ruim. \
    if X >=   0 { result: -1 } \ Péssimo.. \
    if X >=  -5 { result: -2 } \ Terrivel... \
    if X >= -10 { result: -3 } \ Trágico.... \
                  result: -4   \ Catastrófico..... \
}

HABILIDADE: 1
DIFICULDADE: 1
function: modificador { result: HABILIDADE - DIFICULDADE }

function: rolar {
    result: 2d10
}

loop H over {-20..20}{
    output [validate [rolar] + H] named "[H]"
}

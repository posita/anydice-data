function : again A{
   if A=10 { result : 10+[again A]}
   result : A
}


function : roll A:n B:n C:n D:n E:n X:n {   
   if X=0 {
     result : 0
   }
   if X=1{
     if A=10{
        result : 10 + 1d10
     }
     result : A
   }
   if X=2{
      if [highest of A and B]=10{
         if[lowest of A and B]=10{
            result: 20 + 1d10
         }
         else{
            result : A+B
         }
      }
      else{
         result : [highest of A and B]
      }
   }
   if X=3{
      COUNT : (A=10) + (B=10) + (C=10)
      if COUNT = 0{
         result : [highest of A and [highest of B and C]]
      }
      if COUNT = 1 {
         result : A+B+C - [lowest of A and [lowest of B and C]]
      }
      if COUNT = 2{
         result : A+B+C
      }
      if COUNT = 3{
         result : 30+1d10
      }
   } 
   if X=4{
      COUNT : (A=10) + (B=10) + (C=10) + (D=10)
      if COUNT = 0{
         result : 1@[sort {A,B,C,D} ]
      }
      if COUNT = 1 {
         result : 10 + 2@[sort {A,B,C,D}]
      }
      if COUNT = 2{
         result : 20 + 3@[sort {A,B,C,D}]
      }
      if COUNT = 3{
         result : A+B+C+D
      }
      if COUNT = 4{
         result : 40 + 1d10
      }
   }
}

output [roll 1d10 1d10 1d10 1d10 0 4] - [roll 1d10 1d10 1d10 1d10 0 4] named "4d contest"
output 1d20 - 1d20 named "1d20 contest"

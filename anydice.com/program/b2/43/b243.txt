\First comment counts a natural 10 as two successes, and the second analyzes dice outputs\

function: evaluate ROLL:s {
 SUCCESSES: ROLL >= 6
 if SUCCESSES > 0 { result: SUCCESSES \+ (ROLL = 10)\ }
 if [count 1 in ROLL] { result: -1 }
 result: 0
}

\
loop DICE over {1..5} {
 output [evaluate DICE d10] named "[DICE]d"
}
\

output [evaluate 5d10] named "# of Successes"
output 1d10 named "Dice probabilities"

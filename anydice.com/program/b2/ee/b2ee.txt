function: roll DICE:n DIFF:n THRESH:n {
  SUCCESS: 0
  loop N over {1..DICE} {
    SUCCESS: SUCCESS + [roll 1d6 DIFF]
  }
  SUCCESS: SUCCESS - THRESH
  result: [highest of SUCCESS and 0]
}

function: roll DIE:n DIFF:n {
  B: 0
  if DIE = 6 {
    B: 1d6 > DIFF
  }
  result: (DIE > DIFF) + B
}

loop DICE over {2..12} {
  loop DIFF over {4..4} {
    output [roll DICE DIFF 0] named "[DICE]d6 difficulty [DIFF]"
  }
}

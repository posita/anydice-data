ATKS: 1
MOD: 4
PROF: 3
TOHIT: MOD+PROF+1
AC: 15
DAMAGEMOD: 0 + MOD


function: attack roll plus HIT:n vs DC:n dice DICE:d{
ROLL: 1d{-200,{2+HIT..18+HIT},200,200}
result: [attack ROLL DC DICE]
}
function: attack ROLL:n DC:n DICE:d{
result:[damage roll with [hit check ROLL vs DC] hits and [crit check ROLL] crits dice DICE]
}
function: hit check ATTACKROLL:n vs CHECK:n{result: ATTACKROLL>=CHECK}
function: crit check ATTACKROLL:n{result: ATTACKROLL>150}
function: damage roll with HITS hits and CRITS crits dice DICEDAMAGE{result: (HITSdDICEDAMAGE)+(CRITSdDICEDAMAGE)+(HITS*DAMAGEMOD)}

output 3d[attack roll plus TOHIT vs AC dice 2d6]+2d[attack roll plus TOHIT vs AC dice (1d10)]

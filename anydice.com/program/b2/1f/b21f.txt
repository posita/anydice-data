MONTH: {31,28,31,30,31,30,31,31,30,31,30,31}

function: ROLL:n riot select {
if ROLL = 1 {result: 21}
if ROLL = 2 {result: 1dMONTH}
if ROLL = 3 {result: 1dMONTH+14}
}

K: [lowest of 1d{{1}:1, {10}:9} and 1d{{2}:2, {10}:8}]
L: [lowest of 1d{{3}:3, {10}:7} and 1d{{4}:4, {10}:6}]
M: [lowest of 1d{{5}:5, {10}:5} and 1d{{6}:6, {10}:4}]
N: [lowest of 1d{{7}:7, {10}:3} and 1d{{8}:8, {10}:2}]
O: [lowest of K and L]
P: [lowest of M and N]
Q: [lowest of O and P]
R: [lowest of Q and 1d{{9}:9, {10}:1}]

WAIT: R*7+(1d7-1)
RIOT: [d3 riot select]

output WAIT/7 named "weeks between riots"
output RIOT/7 named "duration of riot in weeks"
output (WAIT+RIOT)/7 named "total interval in weeks"

output WAIT named "days between riots"
output RIOT named "duration of riot in days"
output WAIT+RIOT named "total interval in days"

Y: 0
loop Z over {1..12} {
 Y: Y+(WAIT+RIOT)
}
output Y/7 named "duration of twelve riot intervals in weeks"
output Y named "duration of twelve riot intervals in days"

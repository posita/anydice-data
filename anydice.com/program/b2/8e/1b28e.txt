function : Roll E:n {   \ E is amount of die to be rolled \ 
   if E=0 {
     result : 0
   }
   if E=1{
      set "explode depth" to 1
      result : [explode 1d10]
   }
   if E=2{
      A : 1d10
      B : 1d10
      if [highest of A and B]=10{
         if[lowest of A and B]=10{
            result: 20 + 1d10
         }
         else{
            result : A+B
         }
      }
      else{
         result : [highest of A and B]
      }
   }
}

output [Roll 2] named "2d"

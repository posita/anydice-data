function: successes in ROLL:s vs TN:n {
  result: [count {TN..6} in ROLL]
}
function: duds in ROLL:s vs TN:n {
  result: [count {1..TN-1} in ROLL]
}

function: successes in ROLL:s vs TN:n with P:n points {
  SUCC: 0
  REM: P
  loop X over {1..(#ROLL)}{
    N: X@ROLL
    if N >= TN {
      SUCC: SUCC + 1
    }
    else {
      REQ: TN - N
      if REQ <= REM {
        REM: REM - REQ
        SUCC: SUCC + 1
      }
    }
  }
  result: SUCC
}
function: stress from ROLL:s vs TN:n {
  STRESS: 0
  LOWEST: 6
  loop X over {1..(#ROLL)}{
    N: X@ROLL
    if N < LOWEST {
      LOWEST: N
    }
  }
  if LOWEST > STRESS & LOWEST < TN {
    STRESS: LOWEST
  }
  result: STRESS
}
function: successes vs TN:n with S:n skill L:n leverage {
  N: S + L
  ROLL: Nd6
  result: [lowest of [count {TN..6} in ROLL] and 6]
}

TN: 6
SKILL: 6
PUSH: {1d6:(TN-1), {TN..6}}

output [lowest 1 of SKILLdPUSH] named "Success = 6 Stress"
output [stress from SKILLdPUSH vs TN] named "Success = 0 Stress"

output [successes in SKILLd6 vs TN] named "[SKILL]d6 vs [TN]"
output [successes in SKILLdPUSH vs TN] named "[SKILL]d6 vs [TN] (Pushed)"

loop P over {1..3}{
  output [successes vs TN with SKILL skill P leverage] named "[SKILL]d6 vs [TN] ([P] Leverage)"
  output [successes in SKILLd6 vs TN with P points] named "[SKILL]d6 vs [TN] ([P] Stamina)"
  output [successes in SKILLdPUSH vs TN with P points] named "[SKILL]d6 vs [TN] ([P] Stamina, Pushed)"
}

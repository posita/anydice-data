function: ATTACK:n plus ATTACK_MOD:n vs AC:n ac with DAMAGE plus DAMAGE_MOD:n damage {
  if ATTACK = 1 {
    result: 0
  } else if ATTACK = 20 {
    result: 2d(DAMAGE) + DAMAGE_MOD
  } else if ATTACK + ATTACK_MOD >= AC {
    result: DAMAGE + DAMAGE_MOD
  } else {
    result: 0
  }
}

function: no crit ATTACK:n plus ATTACK_MOD:n vs AC:n ac with DAMAGE plus DAMAGE_MOD:n damage {
  if ATTACK = 1 {
    result: 0
  } else if ATTACK + ATTACK_MOD >= AC {
    result: DAMAGE + DAMAGE_MOD
  } else {
    result: 0
  }
}

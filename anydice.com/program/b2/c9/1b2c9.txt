function : roll A:n B:n C:n D:n E:n X:n {   

   if X=2{
      if [highest of A and B]>=10{
         if[lowest of A and B]>=10
         {result: A + B}}
         else{
            result : [highest of A and B]
             }
         }
   if X=3{
      COUNT : (A>=10) + (B>=10) + (C>=10)
      if COUNT = 0{
         result : [highest of A and [highest of B and C]]
      }
      if COUNT = 1 {
         result : [highest of A and [highest of B and C]]
      }
      if COUNT = 2{
         result : A+B+C - [lowest of A and [lowest of B and C]]
      }
      if COUNT = 3{
         result : A+B+C
      }
   } 
   if X=4{
      COUNT : (A>=10) + (B>=10) + (C>=10) + (D>=10)
      if COUNT = 0{
         result : 1@[sort {A,B,C,D} ]
      }
      if COUNT = 1 {
         result : 1@[sort {A,B,C,D}]
      }
      if COUNT = 2{
         result :  1@[sort {A,B,C,D}] + 2@[sort {A,B,C,D}]
      }
      if COUNT = 3{
         result : A+B+C+D - 4@[sort {A,B,C,D}]
      }
      if COUNT = 4{
         result : A+B+C+D
      }
   }
}

set "explode depth" to 1

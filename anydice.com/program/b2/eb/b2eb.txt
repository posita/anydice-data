ROLL:1d20
AC:15
HIT:5
CRIT_ON:20
DAMAGE: 2d6+3
CRIT_DAMAGE:2d6
MULTIATTACK:2
function: attack ROLL:n {
 if ROLL = 1 { result: 0 }
 if ROLL >= CRIT_ON { result: DAMAGE+CRIT_DAMAGE }
 if ROLL + HIT >= AC { result: DAMAGE }
 result: 0
}
output [attack ROLL] named "normal"
output [highest of [attack ROLL] and [attack ROLL]] named "Advantage"
output [lowest of [attack ROLL] and [attack ROLL]] named "Disadvantage"

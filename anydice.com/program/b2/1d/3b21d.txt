function: roll DICE {
   result: [lowest of 1@DICEd20 and 1d20]
}

function: mix ROLL with OTHER {
   result: [highest of [lowest of ROLL and OTHER] and OTHER]
}

function: NUM:n mod N:n {
    result: NUM - ((NUM / N)*N)
}

DC: 15
loop DICE over {1..5} {
output d{} named ""
output d{} named ""
output d{} named ""
output d{} named ""
output d{} named "With [DICE] dice"
ADV: 1@2d20
STD: 1d20
loop N over {0..10} {
    if [N mod 2] = 0 {
        output ADV>=DC named "Complex*[N] Adv"
        output STD>=DC named "Complex*[N] Std"
    }

    ADV: [mix ADV with 1@DICEd20]
    STD: [mix STD with 1@DICEd20]
}
}

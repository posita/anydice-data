function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

TARGET_AC: 18

\ sharpshooter - 5 to hit + 10dmg \
output [attack d20 plus 3 vs TARGET_AC for 1d6+14 crit 1d6 on 20] named "sharpshooter without precision shot"
output [attack d20+d8 plus 3 vs TARGET_AC for 1d6+14 crit 1d6 on 20] named "sharpshooter with precision shot"

\ normal to hit, dex +4 \
output [attack d20 plus 9 vs TARGET_AC for 1d6+5 crit 1d6 on 20] named "without sharpshooter"

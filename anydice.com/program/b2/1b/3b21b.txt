function: damage ROLL:n TOHIT:n AC:n HIT:n CRIT:n {
 if (ROLL = 20) { result: CRIT }
 if (ROLL+TOHIT > AC+10) { result: CRIT }
 else if (ROLL > AC) { result: HIT }
}

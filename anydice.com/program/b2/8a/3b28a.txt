function: A:n damages B:n {
 if A > B {result: A - B} 
 if A = B {result: 0}
 result: 0
}
function: A:n damages B:n weapon W:n {
 if A >= B {result: A - B + W}
 result: 0
}
F: {-1,0,1}
loop N over {-6..6} {
 loop M over {0..6} {
  output 3d[8dF + N damages 0 weapon M] named "3 rolls at +[N] W[M]"
  output 6d[8dF + N damages 0 weapon M] named "6 rolls at +[N] W[M]"
 }
}

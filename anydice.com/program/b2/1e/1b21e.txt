function: reroll DICE:n SECONDDICE:n {
  if DICE > 2 {
    result: DICE
  } else {
    result: SECONDDICE
  }
}

function: gwf {
  result: [reroll 1d6 1d6]+[reroll 1d6 1d6]
}

output [gwf]+[gwf]+[gwf]+9 named "3 attacks"

FIGHT:2d6
WARLO:1d8
MOD:3

function: great N:n X:n {
 if N < 3 { result: 1dX }
 result: N
}
function: gsword {
 result: [great 1d6 6] + [great 1d6 6]
}
function: pole {
 result: [great 1d10 10] + [great 1d10 10]
}
function: butt {
 result: [great 1d4 4] + [great 1d4 4]
}

\dual = dual wielder feat\
\pole = polearm mastery feat \

\warlock with hex spell active\
\output 1d8 + 1d6 + MOD named "warlock lvl 1"\
\output 2d8 + 2d6 + MOD named "dual warlock lvl 1"\

\fighter\
\output [gsword] + MOD named "great great fighter lvl 1"\
\output 2d8 + 2*MOD named "dual fighter lvl 1"\
output [pole] + MOD + [butt] + MOD named "pole fighter lvl 1"
\output 1d6 + MOD + 2 + 1d4 + MOD + 2 named "staff duel fighter lvl 1"\

\ Warlock with Improved Pact Weapon\
\output 2d6 + 1d6 + MOD + 1 named "great warlock lvl 3"\
\output 2d8 + 2d6 + MOD + 1 named "dual warlock lvl 3"\
output 1d10 + 1d4 + 2d6 + 2 + 2*MOD named "pole warlock lvl 3"


\fighter\
\output [gsword] + MOD + [gsword] + MOD named "great great fighter lvl 5"\
\output 3d8 + 2*MOD named "dual fighter lvl 5"\
output [pole] + [pole] + [butt] + 3*MOD named "pole fighter lvl 5"

\ Warlock with Improved Pact Weapon, Cloak of Flies, Extra Attack\
\output (2d6 + 1d6 + MOD + 1) + (2d6 + 1d6 + MOD + 1) + MOD  named "great warlock lvl 5"\
\output 3d8 + 3d6 + 2*MOD + 2 + MOD named "dual warlock lvl 5"\
output 2d10 + 1d4 + 3d6 + 3*MOD + 3 + MOD named "pole warlock lvl 5"
output [pole] + [pole] + [butt] + 3d6 + 3*MOD + 3 + MOD named "pole warlock lvl 5 fighter 1"


\ Warlock with Improved Pact Weapon, Cloak of Flies, Extra Attack, Maddening Hex\
\output (2d6 + 1d6 + MOD + 1) + (2d6 + 1d6 + MOD + 1) + 2*MOD  named "great Mad warlock lvl 7"\
\Note: Cannot use Maddening on dual or pole as it requires bonus action.\



\ Warlock with lifedrinker and +2 weapon\
output (2d6 + 1d6 + 2*MOD + 2) + (2d6 + 1d6 + 2*MOD + 2) + 2*MOD  named "great Mad warlock lvl 12"
\output 1d8 + (1d8 + 2*MOD)+ (1d8 + 2*MOD) + 3d6 + 4 + MOD named "dual warlock lvl 12"\
output 2d10 + 1d4 + 3d6 + 6*MOD + 6 + MOD named "pole warlock lvl 12"
output [pole] + [pole] + [butt] + 3d6 + 6*MOD + 6 + MOD named "pole warlock lvl 12 fighter 1"

\hybrid build, +2 weapon\
output ([pole]+MOD+2) + ([pole]+MOD+2) + ([pole]+MOD+2) + ([butt]+MOD+2) + 4d6 + MOD named "pole warlock lvl 5 + Fighter 11"

\Eldritch Knight with +1 weapon\
\output ([gsword] + MOD + 1) + (([gsword] + MOD + 1 + 1d8)+ 1d8 + MOD ) named "EK Great 7"\
\output ([gsword] + MOD + 1) + (([gsword] + MOD + 1 + 1d8)+ 1d8 + MOD ) + 2d6 named "EK Great 7 + hex"\

\Eldritch Knight with +2 weapon\
output ([gsword] + MOD + 2) + (([gsword] + MOD + 2 + 2d8)+ 2d8 + MOD ) named "Eldritch Knight Great 11"
\output ([gsword] + MOD + 2) + (([gsword] + MOD + 2 + 2d8)+ 2d8 + MOD ) + 2d6 named "Eldritch Knight Great 11 hex"\
\output (1d6 + MOD + 2 + 2) + ((1d6 + MOD + 2 + 2 + 2d8)+ 2d8 + MOD ) named "Eldritch Knight duel staff 11"\
\output (1d8 + MOD + 2 + 2) + ((1d8 + MOD + 2 + 2 + 2d8)+ 2d8 + MOD ) named "Eldritch Knight duel spear pole 11"\
output ([pole] + MOD + 2) + (([pole] + MOD + 2 + 2d8)+ 2d8 + MOD ) named "Eldritch Knight pole sniper 11"

\fighter - no subclass bonus, +2 weapon(s)\
\output [gsword] + MOD + 2 + [gsword] + MOD + 2 + [gsword] + MOD + 2 + 3d6 named "great fighter lvl 11 hex"\
\output 1d8 + MOD + 2 + 2 + 1d8 + MOD + 2 + 2 + 1d8 + MOD + 2 + 2 + 1d4 + MOD + 2 + 2 named "duel spear pole fighter lvl 11"\
\output 4d8 + 3*MOD + 4*2 named "dual fighter lvl 11"\
output [pole] + [pole] + [pole] + [butt] + 4*MOD + 4*2 named "pole fighter lvl 11"

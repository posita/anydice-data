function: roll DICE:d {
  result: [helper 1@DICE 2@DICE 3@DICE]
}

function: helper HIGHEST:n MIDDLE:n LOWEST:n {
  if MIDDLE > 10 { result: HIGHEST }
  result: LOWEST
}

function: helper odd HIGHEST:n MIDDLE:n LOWEST:n {
  if MIDDLE / 2 = (MIDDLE + 1) / 2 { result: HIGHEST }
  result: LOWEST
}

output [roll 3d20]

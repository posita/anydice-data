function: hitroll ATT:n QUL:n {
RES: [count {QUL..6} in ATTd6]
result: RES
}

function: defroll HIT:n APP:n DEF:n {
CAP: DEF+APP-1
if CAP > 5 {
 CAP: 5
}
else {
}
RES: [count {1..(CAP)} in HITd6]
result: RES
}

function: roll ATT:n QUL:n APP:n DEF:n {
ROL: [hitroll ATT QUL]
RES: [defroll ROL APP DEF]
result: RES
}

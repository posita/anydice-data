\H = roll to hit, A = roll needed to hit, D = Average damage of weapon, C = Crit range\ 
function: hit H:n A:n D:n C:n{
 if H=20 {result: [crit 1d20 A D]}
 if H=1 {result: 0}
 result: [hit help H A D C]
}

function: hit help H:n A:n D:n C:n{
 if H>=C {result: [crit 1d20 A D]}
 if H>=A {result: D}
 result: 0
}

function: crit H:n A:n D:n{
 if H>=A {result: 2*D}
 result: D
}

output [hit 1d20 10 27 17]
output [hit 1d20 10 25 15]

\ Farm \

\ Workers \
MAN: 1
SPE: 2
WOR: 10

\ Modifiers \
MODMAN: 2
MODSPE: 1
MODWOR: 1/2

\ Costs \
COSTMAN: 15
COSTSPE: 25/2
COSTWOR: 10

COST: MAN*COSTMAN+SPE*25/2+WOR*COSTWOR

\ Dice \
MULTI: 6

TIERA: MULTI*(0)
TIERB: MULTI*(15d6)
TIERC: MULTI*(20d8)
TIERD: MULTI*(25d10)

\ Roll \
ROLL: 1d100
BONUS: MAN*MODMAN + SPE*MODSPE + WOR*1/2
TOTAL: ROLL+BONUS

COUNT: 1

function: gold function A:n {
 if A <= 50 {
result: (TIERA*15/100-COST)
 }
 if A <= 80 {
result: (TIERB*15/100-COST)
 }
 if A <= 90 {
result: (TIERC*15/100-COST)
 }
 if A <= 200 {
result: (TIERD*15/100-COST)
 }
}

output [gold function TOTAL] named "Gold"

function: set element I:n in SEQ:s to N:n {         \change 1 element in a sequence\
  NEW: {}
  loop J over {1 .. #SEQ} {
    if I = J { NEW: {NEW, N} }
    else { NEW: {NEW, J@SEQ} }
  }
  result: NEW
}

function: modify all dices SEQ:s MODS:n {     \apply Value-Modifier of +1 or -1 on all Dices\
 THREES: ROLL=3
 TWOS:   ROLL=2
 ONES:   ROLL=1
 ZEROS:  ROLL=0
 if(MODS = 1){
   THREES: THREES+TWOS
   TWOS: ONES
   ONES: ZEROS
   ZEROS: 0}
 if(MODS = -1){
   ZEROS: ZEROS+ONES
   ONES: TWOS
   TWOS: THREES
   THREES: 0}
 result: {THREES,TWOS,ONES,ZEROS}
}

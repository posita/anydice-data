function: cascade KEEP:n DICE:s {
 SIX: DICE = 6
 FIVE: DICE = 5
 FOUR: DICE = 4
 THREE: DICE = 3
 TWO: DICE = 2
 ONE: DICE = 1
 if SIX > KEEP {
  SIX: KEEP
 }
 if FIVE > SIX {
  FIVE: SIX
 }
 if FOUR > FIVE {
  FOUR: FIVE
 }
 if THREE > FOUR {
  THREE: FOUR
 }
 if TWO > THREE {
  TWO: THREE
 }
 if ONE > TWO {
  ONE: TWO
 }
 SUM: 0
 SORTED: [sort DICE]
 loop N over {1..KEEP}
 {
  SUM: SUM + N@SORTED
 }
 result: SUM + FIVE + FOUR + THREE + TWO + ONE
}

loop SKILL over {1..12}
{
output [cascade ((SKILL+1)/2) (SKILLd6)] named "skill [SKILL]"
}

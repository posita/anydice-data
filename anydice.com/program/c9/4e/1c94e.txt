function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}

function: A:n and B:n cmd reroll BAD:s as REROLL:d {
  if A+B = BAD { result: [highest of A and B] + REROLL }
  result: A+B
}

output 2d6 named "charge success with no reroll"
output [d6 and d6 cmd reroll {2..10} as d6] named "charge success with reroll one, keep highest"
output [2d6 reroll {2..10} as 2d6] named "charge success with reroll both"

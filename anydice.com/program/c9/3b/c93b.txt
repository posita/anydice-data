function: cascade KEEP:n DICE:s {
 SIX: DICE = 6
 FIVE: DICE = 5
 FOUR: DICE = 4
 THREE: DICE = 3
 TWO: DICE = 2
 ONE: DICE = 1
 if SIX > KEEP {
  SIX: KEEP
 }
 if FIVE > SIX {
  FIVE: SIX
 }
 if FOUR > FIVE {
  FOUR: FIVE
 }
 if THREE > FOUR {
  THREE: FOUR
 }
 if TWO > THREE {
  TWO: THREE
 }
 if ONE > TWO {
  ONE: TWO
 }
 SUM: 0
 SORTED: [sort DICE]
 loop N over {1..KEEP}
 {
  NUMBAH: N@SORTED
  SUM: SUM + NUMBAH
  if NUMBAH = 5 & SIX {
   FIVE: FIVE + 1
  } else if NUMBAH = 4 & FIVE {
   FOUR: FOUR + 1
  } else if NUMBAH = 3 & FOUR {
   THREE: THREE + 1
  } else if NUMBAH = 2 & THREE {
   TWO: TWO + 1
  } else if NUMBAH = 1 & TWO {
   ONE: ONE + 1
  }
 }
 result: SUM + (FIVE + FOUR + THREE + TWO + ONE)*6
}

loop SKILL over {1..11}
{
output [cascade ((SKILL+1)/2) (SKILLd6)] named "skill [SKILL]"
}

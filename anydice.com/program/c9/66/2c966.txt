{
 result: N - N/10 * 10
}

function: tens digit of N:n
{
 result: N/10
}

 if (ROLL < SKILL) { result: 1 }

 if ([ones digit of ROLL] < [tens digit of SKILL]) {result: 0} 

 \* Complete failure *\
 result: -1 
}

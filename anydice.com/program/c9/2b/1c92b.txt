loop DICE over {1..20} {
 output DICE d (d6 = 6) >= 1 named "[DICE]d"
}


function: roller REROLLSEQ:s WINSEQ:s SEQUENCE:s{
 RR: [count REROLLSEQ in SEQUENCE]
 WIN: [count WINSEQ in SEQUENCE]
 result: [count WINSEQ in RRd6] + WIN
}

loop P over {1..5} {
output [roller {} {6} Pd6] >= 1 named "straight [P]d6"
output [roller {2,3,4,5} {6} Pd6] >= 1 named "rerolling [P]d6 no ones"
output [roller {1,2,3,4,5} {6} Pd6] >= 1 named "rerolling [P]d6"
}

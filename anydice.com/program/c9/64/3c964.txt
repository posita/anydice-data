function: FDICE:s vs EDICE:s with fight FFV:n vs EFV:n{
  FROLL: {1..1}@FDICE
  EROLL: {1..1}@EDICE
  WIN: 1
  if FROLL > EROLL {WIN: 1}
  if FROLL = EROLL {
    if FFV > EFV {WIN: 1} 
    else {if FFV < EFV {WIN: -1}
       else {WIN: d{-1, 1} * 1}}}
  if FROLL < EROLL {WIN: -1}

  result: WIN
}

function: NWOUND:n wounds on WOUNDROLL:n with ATTACKS:s attacks{
  WOUNDS: [count {WOUNDROLL..6} in ATTACKS]
  
  result: WOUNDS >= NWOUND
}

function: get NWOUND:n wounds on WOUNDROLL:n with ATTACKS:s attacks{
  WOUNDS: [count {WOUNDROLL..6} in ATTACKS]
  
  result: WOUNDS >= NWOUND
}

output [2d6 vs 2d6 with fight 6 vs 3] named "win fight"

function: cascade KEEP:n of DICE:s {
 SIX: DICE = 6
 FIVE: DICE =  5
 FOUR: DICE = 4
 THREE: DICE = 3
 TWO: DICE = 2
 ONE: DICE = 1
 if SIX > KEEP {
  SIX: KEEP
 }
 if FIVE > SIX {
  FIVE: SIX
 }
 if FOUR > FIVE {
  FOUR: FIVE
 }
 if THREE > FOUR {
  THREE: FOUR
 }
 if TWO > THREE {
  TWO: THREE
 }
 if ONE > TWO {
  ONE: TWO
 }
 result: [highest KEEP of DICE] + FIVE + FOUR + THREE + TWO + ONE
}

loop SKILL over {1..13}
{
output [cascade ((SKILL+1)/2) of (SKILLd6)] named "skill [SKILL]"
}

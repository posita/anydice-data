DIE: 2d6
BONUS: 4
MODIFIER: 0
EXTRA: 0
MINORWOUNDS: 1
MAJORWOUNDS: 1
GRIEVOUSWOUNDS: 1

function: reroll R:n above N {
   if R > N-1 { result: 1d6 } else {result: R}
}

if MAJORWOUNDS = 1 {
  DIE: [reroll 1d6 above 6] + [reroll 1d6 above 6]
}
if MAJORWOUNDS = 2 {
  DIE: [reroll 1d6 above 5] + [reroll 1d6 above 5]
}

SHIT: DIE + BONUS + MODIFIER + EXTRA

if GRIEVOUSWOUNDS = 1 {
  DIE: [lowest 2 of 3d6]
}

ASS: DIE + BONUS + MODIFIER + EXTRA

if MAJORWOUNDS = 1 & GRIEVOUSWOUNDS = 1 {
  STOOGES: 3d6
  LARRY: [reroll 2@STOOGES above 6]
  CURLY: [reroll 3@STOOGES above 6]
  DIE: LARRY+CURLY
}

FUCK: 2d6 + BONUS + MODIFIER + EXTRA - MINORWOUNDS

Y: DIE + BONUS + MODIFIER + EXTRA - MINORWOUNDS
Z: 2d6 + BONUS + MODIFIER + EXTRA


function: partial success X:n {
 if X > 6 {
  result: 1
} else {
  result: 0
}
}
function: success X:n {
 if X > 9 {
  result: 1
} else {
  result: 0
}
}

function: critical success X:n {
 if X > 12 {
  result: 1
} else {
  result: 0
}
}

output [partial success Y] + [success Y] + [critical success Y] named "Wounded"
output [partial success Z] + [success Z] + [critical success Z] named "Unwounded"

\output [partial success FUCK] + [success FUCK] + [critical success FUCK] named "Minor Wound Only"\
\output [partial success SHIT] + [success SHIT] + [critical success SHIT] named "Major Wound Only"\
\output [partial success ASS] + [success ASS] + [critical success ASS] named "Grievous Wound Only"\

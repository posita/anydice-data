      else {result: A}

   if H = 20 {
      if A < N & A < B {result: B + C + ((N-1)*2)}
      else {
         if B < N & B < A {result: A + C + ((N-1)*2)}
         else {result: A + B + ((N-1)*2)}
      }
   }
   else {
      if H = 1 {result: 0}
      else {
         if A < N  {result: B}
         else {result: A}
      }
   }
}

function: calculate SHOOTER DEFENDER DEFENDARM {
RESULT: {}
loop ATK over {1d20 + SHOOTER}{
 loop DEF over {1d20 + DEFENDER}{
  if DEF != 20 + DEFENDER {
   if ATK != 20 + SHOOTER {
    RESULT: {RESULT, (ATK > DEF) * (ATK - DEFENDARM > 0) * (ATK - DEFENDARM)}
   } else {
    RESULT: {RESULT, ATK + 5 - DEFENDARM}
   }
  } else {
   RESULT: {RESULT, 0}
  }
 }
}
result: RESULT
}

output [calculate 3 4 13] named "Shoot +3, Fight +4, Armour 13"
output [calculate 3 5 13] named "Shoot +3, Fight +5, Armour 13"
output [calculate 3 4 14] named "Shoot +3, Fight +4, Armour 14"

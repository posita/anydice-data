set "maximum function depth" to 22

function: pull PULL:n ZEROS:n ONES:n REMAINING:n TARGET:n TOTAL:n {
  if (TARGET = 1 & PULL = 1) | TARGET < 1 | ZEROS < 1 {
    result: TOTAL - REMAINING
  }

  if REMAINING < 1 | ONES < 1{
    result: -1
  }

  if PULL {
    result: [pull (1d{0:ZEROS, 1:(ONES-1)}) ZEROS (ONES-1) (REMAINING-1) (TARGET-1) TOTAL]
  }

  result: [pull (1d{0:(ZEROS-1), 1:ONES}) (ZEROS-1) ONES (REMAINING-1) TARGET TOTAL]
}


function: draw ZEROES:n ONES:n TARGET:n {
  TOTAL: ZEROES + ONES
  result: [pull (1d{0:ZEROES, 1:ONES}) ZEROES ONES (TOTAL-1) TARGET TOTAL]
}


output [draw 19 2 1]
output [draw 19 2 2]

function: flat K:d {
  COUNT: [count 6 in K]
  result: [lowest of COUNT and 2]
}
function: adv K:d {
  COUNT: [count 6 in K]
  CONDITION: [count 5 in K] > 0
  result: [lowest of COUNT + CONDITION and 2]
}
function: disadv K:d {
  COUNT: [count 6 in K]
  CONDITION: [count 1 in K] > 0
  result: [highest of [lowest of COUNT - CONDITION and 2] and 0]
}
function: norm K:n {
  SUM: 2d6 + K
  HIT: SUM > 6
  CRIT: SUM > 9
  result: HIT + CRIT
}
function: normadv K:n {
  SUM: [highest 2 of 3d6] + K
  HIT: SUM > 6
  CRIT: SUM > 9
  result: HIT + CRIT
}
function: normdis K:n {
  SUM: [lowest 2 of 3d6] + K
  HIT: SUM > 6
  CRIT: SUM > 9
  result: HIT + CRIT
}

loop N over {3,5,7,9,11}{
  S: (N-5)/2
  output [flat Nd6] named "[N]d6 FLAT"
  output [norm S] named ""
  output [disadv Nd6] named "[N]d6 DISADVANTAGE"
  output [normdis S] named ""
  output [adv Nd6] named "[N]d6 ADVANTAGE"
  output [normadv S] named ""
}

function: cascade ROLL:s {
 SIX: ROLL = 6
 FIVE: ROLL =  5
 FOUR: ROLL = 4
 THREE: ROLL = 3
 TWO: ROLL = 2
 ONE: ROLL = 1
 if FIVE > SIX {
  FIVE: SIX
 }
 if FOUR > FIVE {
  FOUR: FIVE
 }
 if THREE > FOUR {
  THREE: FOUR
 }
 if TWO > THREE {
  TWO: THREE
 }
 if ONE > TWO {
  ONE: TWO
 }
 result: ROLL + FIVE + FOUR * 2 + THREE * 3 + TWO * 4 + ONE * 5
}

{
output [cascade (SKILLd6)] named "skill [SKILL]"

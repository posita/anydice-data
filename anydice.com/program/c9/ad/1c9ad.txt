function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

AC: d{15..20}

output 2d[attack 1d20 plus 11 vs AC for 2d{1d6:2,3..6}+1d8+5 crit 4d{1d6:2,3..6}+2d8+5 on 20] named "Marius normal"
output 2d[attack 1d20 plus 6 vs AC for 2d{1d6:2,3..6}+1d8+15 crit 4d{1d6:2,3..6}+2d8+15 on 20] named "Marius GWM"

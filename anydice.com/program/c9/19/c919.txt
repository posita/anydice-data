function: roller DICE:d {
 NEW: {1@DICE, 2@DICE}
 EXTRA: [helper 1@DICE] + [helper 2@DICE]
 POOL: (#DICE) d NEW
 result: POOL + EXTRA
}

function: helper DIE:n {
 RES : 0
 if DIE >= 10 {RES : [explode 1d10]}
 result : RES
}

output 1d20
output [highest 1 of 2d20]
output [roller 2d10]
output [roller 3d10]

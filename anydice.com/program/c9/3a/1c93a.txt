function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}

function: A:n and B:n cmd reroll BAD:s as REROLL:d {
  if A+B = BAD { result: [highest of A and B] + REROLL }
  result: A+B
}

output [2d6 reroll {1..7} as 2d6] >= 8 named "charge success with reroll both"
output [d6 and d6 cmd reroll {1..7} as d6] >=8 named "charge success with reroll one, keep highest"

function: cascade KEEP:n DICE:s {
 SIX: [count 6 in DICE]
 FIVE: [count 5 in DICE]
 FOUR: [count 4 in DICE]
 THREE: [count 3 in DICE]
 TWO: [count 2 in DICE]
 ONE: [count 1 in DICE]
 if SIX > KEEP {
  SIX: KEEP
 }
 if FIVE > SIX {
  FIVE: SIX
 }
 if FOUR > FIVE {
  FOUR: FIVE
 }
 if THREE > FOUR {
  THREE: FOUR
 }
 if TWO > THREE {
  TWO: THREE
 }
 if ONE > TWO {
  ONE: TWO
 }
 SUM: 0
 SORTED: [sort DICE]
 loop N over {1..KEEP}
 {
  SUM: SUM + N@SORTED
 }
 result: SUM + FIVE + FOUR + THREE + TWO + ONE
}

loop SKILL over {1..11}
{
output [cascade ((SKILL+1)/2) (SKILLd6)] named "skill [SKILL]"
}

function: ROLL:s count fives and above and subtract ones {
 result: (ROLL >= 5) - (ROLL = 1)
}

function: ROLL:s count sixes {
 result: (ROLL >= 6)
}

loop N over {1..10}{
 output [count {5,6} in Nd6] - [count {1} in Nd6] named "[N]d6 Total Successes (original)"
 output [Nd6 count fives and above and subtract ones] named "[N]d6 total successes (new)"
 output [Nd6 count sixes] named "[N]d6 total successes (alternative)"
}

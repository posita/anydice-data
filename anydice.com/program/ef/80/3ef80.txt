loop N over {0..12} {
  DEGREE: N-6
  if DEGREE < -2 { X: DEGREE * -1 -2 output [lowest 1 of 3d8]+1d12-X named "Negative N [DEGREE]" }
  else if DEGREE < 0 { X: 1 + DEGREE * -1 output [lowest 1 of Xd8]+d12 named "Negative [DEGREE]" }
  else if DEGREE  = 0 { output 1d12+1d8 named "Neutral [DEGREE]" }
  else if DEGREE <= 2{ X: DEGREE+1 output [highest 1 of Xd8]+d12 named "Positive [DEGREE]" }
  else { X: DEGREE-2 output [highest 1 of 3d8]+d12+X named "Positive N [DEGREE]" }
}

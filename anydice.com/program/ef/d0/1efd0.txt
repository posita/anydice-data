
function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

output
[attack d20 plus 3 + 5 vs 16 for 1d10+1d6+5 crit 1d10+1d6 on 20]
named "Hex"

output
[attack [highest 1 of 2d20] plus 3 + 5 vs 16 for 1d10+5 crit 1d10 on 20]
named "Advantage"

function: ROLL:s count fives and above and subtract ones {
 result: (ROLL >= X) - (ROLL = 1)
}

DC : 4

loop N over {1 .. 10} {
  X : DC
  output [Nd6 count fives and above and subtract ones] > 0 named "success DC [DC], roll [N]"
  output [Nd6 count fives and above and subtract ones] < 0 named "blunder DC [DC], roll [N]"
}

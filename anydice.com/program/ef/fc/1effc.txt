 if ROLLEDVALUE >= ARMOUR { result: DAMAGEDICE } else { result: 0}
}

function: reroll FIRSTROLL:n DAMAGEDICE:d TRIGGER:n{
 if FIRSTROLL <= TRIGGER { result: DAMAGEDICE } else { result: FIRSTROLL}
}

  if DICEONE + DICETWO = 12 { result: DAMAGEDICE } 
  else{
   if DICEONE = 6 { result: DAMAGEDICE }
   else {
    if DICETWO = 6 { result: DAMAGEDICE }
    else{
     result: 0
    }
   }
  }
 }
}

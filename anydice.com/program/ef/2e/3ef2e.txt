HOPE: 1d12
HOPETWENTY: 1d20
FEAR: 1d12

function: get hope from H:n and F:n {
   if H > F {result: 1}
   if F > H {result: -1}
   if F = H {result: 100}
}
function: total A:n B:n {
   if A != B { result: A + B }
}

output [get hope from HOPE and FEAR] named "Fear/Hope/Crit"
output [get hope from HOPETWENTY and FEAR] named "Fear/Hope/Crit d20"
output [total HOPE FEAR] named "Result (excl. crit)"
output [total HOPETWENTY FEAR] named "Result d20 (excl. crit)"

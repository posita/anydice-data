\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: reroll R:n under N:n {
   if R < N { result: 1d12 } else {result: R}
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 4
DMGBONUS: 10
PROFICIENCY: 5
ACCBONUS: -5
CRITROLL: 19
ATTACKTIMES: 2
DAMAGE: [reroll 1d12 under 3] + ATTACKMOD + DMGBONUS
CRITDAMAGE: [reroll 3d12 under 3] + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll [highest 1 of 2d20] vs ENEMYAC] named "Krog Greataxe v1"

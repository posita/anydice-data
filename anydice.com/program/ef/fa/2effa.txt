DISASTER: 0 
DISADVANTAGE: 1 
MAINTAIN: 2 
LOW: 10
HIGH: 20
DN: 1
DS: 20 
LOMOD: 2
HIMOD: 6

function: save ROLL:n {
 if ROLL <= LOW { result: DISASTER }
 if ROLL >= HIGH { result: MAINTAIN  }
 result: DISADVANTAGE
}

loop ADJ over {LOMOD..HIMOD}{ 
output [save DNdDS + ADJ] named"[DN]d[DS] + [ADJ]" }

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {result: 0}
    if ROLL >= CRIT_RANGE {result: DMG + CRIT_DMG}
    if ROLL + BONUS >= AC {result: DMG}
    result: 0
}

AC: d{10..15}
MOD_ATTACK: 6
MOD_DAMAGE: 4
RAGE_DAMAGE: 2

output 1d[attack [highest 1 of 2d20] plus MOD_ATTACK+1d4 vs AC for 2d6+MOD_DAMAGE+RAGE_DAMAGE crit 2d6+2d6 on 20] named "Bless Normal"
output 1d[attack [highest 1 of 2d20] plus MOD_ATTACK+1d4-5 vs AC for 2d6+MOD_DAMAGE+RAGE_DAMAGE+10 crit 2d6+2d6 on 20] named "Bless GWM"
output 1d[attack [highest 1 of 2d20] plus MOD_ATTACK vs AC for 2d6+MOD_DAMAGE+RAGE_DAMAGE crit 2d6+2d6 on 20] named "Normal"
output 1d[attack [highest 1 of 2d20] plus MOD_ATTACK-5 vs AC for 2d6+MOD_DAMAGE+RAGE_DAMAGE+10 crit 2d6+2d6 on 20] named "GWM"

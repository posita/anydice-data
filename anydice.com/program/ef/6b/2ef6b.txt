\pool 1\ POOLONE: [count {5..6} in 3d6]
\pool 2\ POOLTWO: [count {5..6} in 3d6]

\test for a win\ SUCCESS: POOLONE >= POOLTWO

\test for any hits\ X: POOLONE > 0

\for the output, multiply the hit test by the SUCCESS chance. If there are no hits, it will return 0\

output SUCCESS*X named "No hits is a failure"
output SUCCESS named "No hits doesn't matter"

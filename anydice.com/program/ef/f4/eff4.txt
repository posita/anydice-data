DEPTH: 20
set "explode depth" to DEPTH
set "maximum function depth" to DEPTH

function: gwf SIZE:n {
  result: [gwfhelp d(SIZE) and SIZE]
}
function: gwfhelp FIRST:n and SIZE:n {
  if FIRST <= 2 { result: d(SIZE) } else { result: FIRST }
}

function: explode N:n on SIZE:n {
 if N = SIZE {
  result: N + [explode d(SIZE) on SIZE]
 }
 result: N
}

function: greatexplode SIZE:n {
  result: [gehelp d(SIZE) and SIZE and 1]
}
function: gehelp N:n and SIZE:n and CANGWF:n {
  if N <= 2 & CANGWF & SIZE > 2 {
    result: [gehelp d(SIZE) and SIZE and 0]
  } else if N = SIZE {
    result: N + [gehelp d(SIZE) and SIZE and 1]
  } else {
    result: N
  }
}

function: petersburg max MAX_DMG:n{
  result: [petersburghelp d2 base 1 max MAX_DMG]
}
function: petersburghelp D:n base BASE:n max MAX_DMG:n {
  if BASE > MAX_DMG { result: MAX_DMG }
  else if D = 1 { result: BASE }
  else { result: [petersburghelp d2 base BASE*2 max MAX_DMG] }
}

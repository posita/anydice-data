
function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}





output [reroll {1,2} on d6] named "reroll 1,2 on a d6"

output [reroll {1} on d6] named "reroll 1 on a d6"

output 1d6 named "normal 1d6"

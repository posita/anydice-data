function: rolltwo MIN:n to MAX:n {
  \result: [explode [clamp [absolute d6-d6] MIN to MAX] MIN to MAX]\
  A: d6
  B: d6
  result: [clamp [diff between A B] MIN to MAX]
}

function: diff between A:n B:n {
  if A = B { result: A+1 }
  result: [absolute A-B]
}

function: explode N:n MIN:n to MAX:n {
  if N = 0 { N: 1 + [rolltwo MIN to MAX] }
  result: N
}

function: absolute NUMBER:n {
 if NUMBER < 0 { result: -NUMBER }
 result: NUMBER
}

function: clamp NUMBER:n MIN:n to MAX:n {
 if NUMBER < MIN { result: MIN }
 if NUMBER > MAX { result: MAX }
 result: NUMBER
}

\rolling two same values yields value+1, i.e. 3+3 = 4\
\rolling different values yields a difference, i.e. 3+5 = 2\

output [rolltwo 1 to 6]

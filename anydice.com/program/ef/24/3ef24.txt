A: {3, 4, 5, 6, 8, 11, 13, 16, 18}
B: {6, 8, 10, 12, 14, 16, 18, 22, 26}
C: {6, 9, 12, 16, 20, 22, 25, 28, 32}

PROFICIENCY: 0


function: helper N:n O:n T:n S:s {
 result: [highest of 0 and ((N+PROFICIENCY >= S) + T - O)]
}

function: succ ROLLS:s with D:n in SUCC:s {
 KEPT: {1..((D+1)/2)} @ [sort ROLLS]
 ONES: [count 1 in ROLLS]
 TENS: [count 10 in ROLLS]
 result: [helper KEPT ONES TENS SUCC]
}

output [succ 6d10 with 6 in C] named "6d"
output [succ 5d10 with 5 in C] named "5d"
output [succ 4d10 with 4 in B] named "4d"
output [succ 3d10 with 3 in B] named "3d"
output [succ 2d10 with 2 in A] named "2d"
output [succ 1d10 with 1 in A] named "1d"

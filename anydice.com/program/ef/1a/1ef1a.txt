set "maximum function depth" to 4

function: standard ROLL:s  {
 result: (ROLL > 5) + ((ROLL = 10)/2)*2
}

function: doubleten ROLL:s  {
 result: (ROLL > 5) + (ROLL = 10)
}

function: reroll ROLL:s  {
 result: (ROLL > 5) + (ROLL = 10)d{0..1}
}


loop N over {1..8}{
 output [lowest of 10 and [standard Nd10]] named "[N] standard"
 output [lowest of 10 and [reroll Nd10]] named "[N] reroll 10s"
}

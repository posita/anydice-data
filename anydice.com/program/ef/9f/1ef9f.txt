
function: precise reroll R:n aka RESULT:n under N:n {
 if R < N { result: RESULT } else { result: R }
}

function: save DIE:n SAVE rend REND {
 if DIE = 6 { result: 0 } 
 if DIE = 1 { result: 1 }
 result: (DIE - REND) < SAVE
}


function: stabbas auxiliary LOONBOSS loonboss HIT tohit DHIT:n and WOUND wound DWOUND:n vs SAVE save REND rend {
 result: ((DHIT >= HIT) * (DWOUND >= WOUND) * [save d6 SAVE rend REND]) + LOONBOSS * ((DHIT >= HIT) * (DWOUND = 6))
}

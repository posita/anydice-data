\**** Damage calculator full basic damage dealing list ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: savehalfroll ROLL:n vs DC:n {
 if ROLL + SAVINGTHROW < DC { result: DAMAGE }
 if ROLL + SAVINGTHROW >= DC { result: HALFDAMAGE }
 result: 0
}

function: saveroll ROLL:n vs DC:n {
 if ROLL + SAVINGTHROW < DC { result: DAMAGE }
 if ROLL >= DC { result: 0 }
 result: 0
}

function: sneakattackdualwield ROLLA:n vs ROLLB:n vs AC:n {
 if ROLLA + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC & ROLLB + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC {result: [calculatedmg ROLLA] +  [calculatedmg ROLLB]+ [calculatesneakdmg ROLLA]}
 if ROLLA + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC & ROLLB + PROFICIENCY + ATTACKMOD + ACCBONUS < AC {result: [calculatedmg ROLLA] + [calculatesneakdmg ROLLA]}
if ROLLA + PROFICIENCY + ATTACKMOD + ACCBONUS < AC & ROLLB + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC {result: [calculatedmg ROLLB] + [calculatesneakdmg ROLLB]}
 if ROLLA + PROFICIENCY + ATTACKMOD + ACCBONUS < AC & ROLLB + PROFICIENCY + ATTACKMOD + ACCBONUS < AC { result: 0}
}

function: calculatedmg ROLL:n {
 if ROLL >= CRITROLL {result: CRITDAMAGE}
 if ROLL < CRITROLL {result: DAMAGE}
}

function: calculatesneakdmg ROLL:n {
 if ROLL >= CRITROLL {result: CRITSNEAKDAMAGE}
 if ROLL < CRITROLL {result: SNEAKDAMAGE}
}
\**** End of damage calculation functions ****\

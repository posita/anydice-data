\   1,  2,  3,  4,  5,  6,  7,  8,  9   \A: {6,  8,  10, 12, 14, 15, 16, 17, 18}
B: {7,  10, 12, 15, 17, 19, 22, 25, 28}
C: {8,  11, 13, 16, 19, 22, 26, 29, 33}

PROFICIENCY: 0

function: helper N:n F:n C:n S:s {
 result: [highest of 0 and ((N+PROFICIENCY >= S) - F + C)]
}

function: succ ROLLS:s with D:n dice in SUCC:s table {
 KEPT: {1..((D+1)/2)} @ [sort ROLLS]
 FAILS: [count 1 in ROLLS]
 CRITS: [count {10..12} in ROLLS]
 result: [helper KEPT FAILS CRITS SUCC]
}

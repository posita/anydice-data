function: explode N:n {
  COUNT: 0
  if N >= THRESHOLD & N <= ATTRIBUTE & THRESHOLD <= ATTRIBUTE {
    result: COUNT+1 + [explode STAKEd6]
  }
  result: 0
}

set "maximum function depth" to 2
loop THRESHOLD over {8,11,14} {
  loop ATTRIBUTE over {11..21} {
    loop STAKE over {2..5} {
      output [explode STAKEd6] named "Success for Stake of [STAKE]d6 rolls [THRESHOLD] to [ATTRIBUTE]"
    }
  }
}

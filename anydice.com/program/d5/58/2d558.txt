function: outcome of A:s B:s C:s {
T: {1}@[sort {A, B, C}]
G: {2}@[sort {A, B, C}]
    result: T + G
}
loop DIFF over {1..10}{
loop MOD over {4,6,8,10,12}{
loop A over {4,6,8,10,12}{
loop B over {4,6,8,10,12}{
output [outcome of (dMOD > DIFF) (dA > DIFF) (dB > DIFF)] named "d[A] + d[B] with Difficulty of [DIFF] and Mod of [MOD]"
output (dA > DIFF) + (dB > DIFF) named "d[A] + d[B] with Difficulty of [DIFF]"
}
}
}
}

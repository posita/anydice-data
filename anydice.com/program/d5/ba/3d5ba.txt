function: crop SEQUENCE:n {
 if (SEQUENCE < 0){result: 0}
 result: SEQUENCE
}
function: limits SEQUENCE:n ZERO:n MAX:n {
 if (SEQUENCE < ZERO){result: 0}
 if (SEQUENCE >= MAX){result: MAX}
 result: SEQUENCE
}

\************************** Functions : Highest/Lowest into Target Count (& vice versa) *******************\

\apply Highest then Target Number\
function: highest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLL >= TARGET) }
 result: COUNT
}
\apply Lowest then Target Number\
function: lowest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 ROLLR: [reverse ROLL]
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLLR >= TARGET) }
 result: COUNT
}

\apply Target Number then limit the max (Highest or Lowest doesn't matter)\
function: count from ROLL:s on TARGET:n keep NUMBER:n {
 COUNT: ROLL >= TARGET
 if(COUNT > NUMBER) { COUNT: NUMBER }
 result: COUNT
}



\apply Highest then Target Number\
function: highestbis NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLL <= TARGET) }
 result: COUNT
}
\apply Lowest then Target Number\
function: lowestbis NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 ROLLR: [reverse ROLL]
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLLR <= TARGET) }
 result: COUNT
}
function: countbis from ROLL:s on TARGET:n keep NUMBER:n {
 COUNT: ROLL <= TARGET
 if(COUNT > NUMBER) { COUNT: NUMBER }
 result: COUNT
}

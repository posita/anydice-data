set "maximum function depth" to 13

function: legend A:n B:n {
 if B < A { result: [lowest of 8 and B]d{0,1,2} < 5 & [legend A B+1] }
 result: [lowest of 8 and B]d{0,1,2} >= 5
}

function: baldur A:n B:n {
 if B < A { result: [lowest of 8 and B]d{0,1,2} < 6 & [baldur A B+1] }
 result: [lowest of 8 and B]d{0,1,2} >= 6
}

function: classic A:n B:n {
 if B < A { result: 6d{0,1,2} > B & [classic A B+1] }
 result: 6d{0,1,2} <= B
}

loop X over {1..12} {
 output [classic X 1] named "Classic [X]"
}
loop X over {1..12} {
 output [baldur X 1] named " Baldur's [X]"
}
loop X over {1..12} {
 output [legend X 1] named "Legacy [X]"
}
\

output 3d{4,3:2,2:3,1:2,0} named "S&D"
output 6d{0,1,2} named "Classic"

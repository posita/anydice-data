function: explode N:n {
  COUNT: 0
  if N >= THRESHOLD & N <= ATTRIBUTE {
    result: COUNT+1 + [explode 6d6]
  }
  result: COUNT
}

set "maximum function depth" to 3
loop THRESHOLD over {8,11,14} {
  loop ATTRIBUTE over {11..21} {
    output [explode d20] named "explode between [THRESHOLD] and [ATTRIBUTE]"
  }
}

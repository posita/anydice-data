function: reroll lower of A:n or B:n if less than T:n {
C: A + B
D: T - C
 if C >= T { result: C }
 if D >= 6 { result: C }
 if A < B { result: d6 + B }
 result: d6 + A
}

function: highest two of A:n B:n C:n {
 OA: A
 OB: C
 if A < B { OA: B}
 if A > C { OB: A}
 result: OA + OB
}

loop TARGET over {6..12} {
  output [reroll lower of d6 or d6 if less than TARGET] named "reroll for [TARGET]"
}
output [highest 2 of 3d6] named "3k2d6"
output [highest two of d6 d6 d6] named "custom 3k2d6"
output 2d6 named "2d6"

function: owod N:n diff D:n spec S:n {
 if N = 1 { result: -1 }
 if N < D { result: 0 }
 if N = 10 {
   if S = 1 { 
     result: 2
   }
 } 
 result: 1
}

POOLS: {10}
DIFFS: {6..9}

loop N over POOLS {
 loop D over DIFFS {
   output Nd[owod d10 diff D spec 0] named "[N]d10 >= [D]"
 }

 loop D over DIFFS {
  output Nd[owod d10 diff D spec 1] named "[N]d10 (spec) >= [D]"
 }
}

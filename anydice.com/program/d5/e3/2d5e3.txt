function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}

\ Prof Bonus \
A: 3

\ Dice Size \
B: 10

\ Ability Mod \
C: 5

D: B/2

X: [dB reroll {1,2} as dB]
Y: [dB reroll {1..D} as dB]
Z: [dY reroll {1,2} as dB]

\ 1 Attack \

output 1dB+C named "weapon with FS:GWF"
output 1dY+C named "weapon with SA"
output 1dZ+C named "weapon with SA and FS:GWF"
output 1dB+C+A named "weapon with GWM"
output 1dX+C+A named "weapon with GWM and FS:GWF"
output 1dY+C+A named "weapon with GWM and SA"
output 1dZ+C+A named "weapon with SA and FS:GWF and GWM"

\ 2 Attacks \

output 2dB+C+C named "(2 hits) weapon"
output 1dY+1dB+C+C named "(2 hits) weapon with SA"
output 2dX+C+C named "(2 hits) weapon with FS:GWF"
output 1dZ+1dX+C+C named "(2 hits) weapon with SA and FS:GWF"
output 2dB+C+C+A named "(2 hits) weapon with GWM"
output 1dY+1dB+C+C+A named "(2 hits) weapon with GWM and SA"
output 2dX+C+C+A named "(2 hits) weapon with GWM and FS:GWF"
output 1dZ+1dX+C+C+A named "(2 hits) weapon with SA and FS:GWF and GWM"

\ 3 Attacks \

output 3dB+C+C+C named "(3 hits) weapon"
output 1dY+2dB+C+C+C named "(3 hits) weapon with SA"
output 3dX+C+C+C named "(3 hits) weapon with FS:GWF"
output 3dB+C+C+C+A named "(3 hits) weapon with GWM"
output 1dZ+2dX+C+C+C named "(3 hits) weapon with SA and FS:GWF"
output 1dY+2dB+C+C+C+A named "(3 hits) weapon with GWM and SA"
output 3dX+C+C+C+A named "(3 hits) weapon with GWM and FS:GWF"
output 1dZ+2dX+C+C+C+A named "(3 hits) weapon with SA and FS:GWF and GWM"

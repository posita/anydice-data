function: attack A:n plus B:n vs AC:n for D:n crit C:n {
  if (A = 20) {
    result: D+C
  } else if (A+B >= AC) {
    result: D
  } else {
    result: 0
  }
}

function: adv A:n and B:n {
  if (A > B) {
    result: A
  } else {
    result: B
  }
}

function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}

\ Prof Bonus \
A: 3

X: [d10 reroll {1,2} as d10]
Y: [d10 reroll {1,2,3,4,5} as d10]
Z: [dY reroll {1,2} as d10]

\ 1 Attack \

output 1d10+5 named "glaive with FS:GWF"
output 1dY+5 named "glaive with SA"
output 1dZ+5 named "glaive with SA and FS:GWF"

output 1dY+5+A named "glaive with GWM and SA"
output 1dZ+5+A named "glaive with SA and FS:GWF and GWM"

  SUCCESSES: ROLL <= TARGET
  if SUCCESSES < #ROLL {
    result: SUCCESSES + (REROLL <= TARGET)
  }
  else {
   result: SUCCESSES
  }
}

  }
  else {
   result: 0
  }
}

function: crop SEQUENCE:n {
 if (SEQUENCE < 0){result: 0}
 result: SEQUENCE
}
function: limits SEQUENCE:n ZERO:n MAX:n {
 if (SEQUENCE < ZERO){result: 0}
 if (SEQUENCE >= MAX){result: MAX}
 result: SEQUENCE
}

function: count negativepositive target number SEQUENCE:s TARGET:n {
 if (TARGET >= 1){result: SEQUENCE >= TARGET}
 if (TARGET <= -1){result: SEQUENCE <= [absolute TARGET]}
 result: 0
}

\apply Highest then Target Number\
function: highest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLL >= TARGET) }
 result: COUNT
}
\apply Lowest then Target Number\
function: lowest NUMBER:n of ROLL:s on TARGET:n {
 COUNT: 0
 ROLLR: [reverse ROLL]
 loop P over {1..NUMBER} { COUNT: COUNT + (P@ROLLR >= TARGET) }
 result: COUNT
}

function: count TARGET:n in highest NUMBER:n of ROLL:s {
  TARGET_SEQ: {}
  loop P over {1..NUMBER} { TARGET_SEQ: {TARGET_SEQ, P@ROLL} }
  result: [count negativepositive target number TARGET_SEQ TARGET]
}
function: count TARGET:n in lowest NUMBER:n of ROLL:s {
  ROLLR: [reverse ROLL]
  TARGET_SEQ: {}
  loop P over {1..NUMBER} { TARGET_SEQ: {TARGET_SEQ, P@ROLLR} }
  result: [count negativepositive target number TARGET_SEQ TARGET]
}

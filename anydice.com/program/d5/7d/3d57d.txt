\ Implementation details \
MISS: 0
NORMAL_HIT: 1
CRITICAL_HIT: 2

function: hit with BONUS:n vs AC:n rolling ROLL:n {
   if ROLL >= CRITICAL_THRESHOLD { result: CRITICAL_HIT }

   if ROLL = 1 { result: MISS }

   if ROLL + BONUS - AC >= 0 { result: NORMAL_HIT }

   result: MISS
}


function: attack with BONUS:n vs AC:n rolling ROLL:n with DICE:d and MOD:n {
   HIT: [ hit with BONUS vs AC rolling ROLL ]

   if HIT = MISS { result: 0 }
   
   if HIT = CRITICAL_HIT { result: DICE + DICE + DICE + 2d6 + MOD }
   
   result: DICE + MOD
}
\ End implementation details \

PROFICIENCY: 3
CRITICAL_THRESHOLD: 20

NORMAL_ROLL: 1d20
ADVANTAGE: [ highest 1 of 2d20 ]

ONE: 1d8
TWO: 1d10

STR_MOD: 4

loop TARGET_AC over { 12..20 } {

   DAMAGE_BONUS: STR_MOD + 2 + 2d6
   HIT_BONUS: STR_MOD + PROFICIENCY
   output [ attack with HIT_BONUS vs TARGET_AC rolling NORMAL_ROLL with ONE and DAMAGE_BONUS ] named "One-handed vs AC[TARGET_AC]"

   DAMAGE_BONUS: STR_MOD + 2d6
   HIT_BONUS: STR_MOD + PROFICIENCY
   output [ attack with HIT_BONUS vs TARGET_AC rolling NORMAL_ROLL with TWO and DAMAGE_BONUS ] named "Two-handed vs AC[TARGET_AC]"

   DAMAGE_BONUS: STR_MOD + 2 + 2d6
   HIT_BONUS: STR_MOD + PROFICIENCY
   output [ attack with HIT_BONUS vs TARGET_AC rolling ADVANTAGE with ONE and DAMAGE_BONUS ] named "One-handed w/ Fighting Spirit vs AC[TARGET_AC]"

   DAMAGE_BONUS: STR_MOD + 2d6
   HIT_BONUS: STR_MOD + PROFICIENCY
   output [ attack with HIT_BONUS vs TARGET_AC rolling ADVANTAGE with TWO and DAMAGE_BONUS ] named "Two-handed w/ Fighting Spirit vs AC[TARGET_AC]"

   \ Group the results prettier. \
   output {0} named ""
}

function: owod N:n diff D:n spec S:n {
 if N = 1 { result: -1 }
 if N < D { result: 0 }
 if N = 10 {
   if S = 1 { 
     result: 1 + [owod 1d10 diff D spec S] 
   }
 } 
 result: 1
}

function: out N:n pool M:n {
  result: [highest of [lowest of N and M + M/2] and -5]  
}

POOLS: {10}
DIFFS: {6..9}

loop N over POOLS {
 loop D over DIFFS {
   output [out Nd[owod d10 diff D spec 0] pool N] named "[N]d10 >= [D]"
 }

 loop D over DIFFS {
  output [out Nd[owod d10 diff D spec 1] pool N] named "[N]d10 (spec) >= [D]"
 }
}

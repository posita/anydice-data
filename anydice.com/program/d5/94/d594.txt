function: best two of I:n and J:n and K:n {
 OA: I
 OB: K
 COMP: J
 if I < J { OA: J COMP: I}
 if COMP > K { OB: COMP}
 result: OA + OB
}

function: reroll lower of A:n or B:n if less than T:n {
C: A + B
D: T - C
 if C >= T { result: C }
 if D >= 6 { result: C }
 if A < B { result: d6 + B }
 result: d6 + A
}

loop TARGET over {6..12} {
  output [reroll lower of d6 or d6 if less than TARGET] named "reroll for [TARGET]"
}
output [highest 2 of 3d6] named "3k2d6"
output [best two of d6 and d6 and d6] named "custom 3k2d6"
output 2d6 named "2d6"

DICE_POOL:5 \* Maximum number of dice to be rolled *\
SUCCESS_CHANCE:5
DOUBLE_CHANCE: 10
AUTO_SUCCESS: 1

function: evaluate ROLL:s
{
	SUCCESS: ROLL >= SUCCESS_CHANCE
	DOUBLES: ROLL >= DOUBLE_CHANCE
	SUCCESSES: SUCCESS + DOUBLES
	result: SUCCESSES + AUTO_SUCCESS
}

loop DICE over {1..DICE_POOL} 
{
	output [evaluate DICE d10] named "[DICE]d"
}

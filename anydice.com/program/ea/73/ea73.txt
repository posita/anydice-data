PROF: 3
MOD: 4
AC: 16
CRITRANGE: 20

function: D:n gwf{ result: {{1..2}:2, {3..D}:D+2} }

function: A:n BA:n N:n D:s BD:n attack{
     if A >= CRITRANGE {
          result: (N+N)dD + MOD + BD
     }
     else if A = 1 { 
          result: 0 
     }
     else if A+PROF+MOD+BA < AC {
          result: 0
     }
     result: NdD + MOD + BD
}


output 3d[d20 2 2 1d6 0 attack] named "crossbow ranger"
output 3d[d20 2 2 1d6 0 attack] named "crossbow blood hunter"
output 2d[d20 0 1 1d8+1d6 2 attack] named "rapier"
output 2d[d20 0 3 [6 gwf] 0 attack] named "maul blood hunter"

output 3d[d20 (-3) 2 1d6 10 attack] named "crossbow sharpshooter"
output 2d[d20 0 1 1d12+1d6 2 attack] named "dueling lance"

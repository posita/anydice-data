\calculate consve depending on proficiency\
function: consave{if CONPROF {result: DICE + CONMOD + PROF} result: DICE + CONMOD}
CONSAVE: [consave]



\############### FUNCTIONS ######################\
\Main Function that runs the program Calculates the number of \
\Hours spent traveling if trying to cover a distance in a day\
   \@param M: miles to travel\
   \@param SPEED: Travel Pace\
   \@param E: exhaustion level at the start of travel\
   \@result: return EXHAUSTION\
function: travel M miles at SPEED speed in one day starting with E:n exhaustion {
 H: 8 + (MILES-2@PACE)/1@PACE           \calculate number of full hours traveling\
 \loop over 1 to H and if over 8 check if need to apply exhaustion\
 loop N over {1..H}{
  if N > 8 {
   DC: DC + 1
   EX: [chose save E]
   E: E + EX
   E: [lowest of E and MAXEXHAUST]
  }
 }
 EXHAUSTION: E
 result: EXHAUSTION
}



\Check if the character succeeds on a normal CON save\
   \@param A: Dice roll for CON save\
   \@param DC: DC of the CON save\
   \@result: value 1 or 0\
function: normal save A:n DC:n{
 result: [A is less than DC]
}

\Check if the character succeeds on a disadvantage CON save\
   \@param A: Dice roll for consave\
   \@param B: Dice roll for consave\
   \@param DC: DC of the CON save\
   \@result: value 1 or 0\
function: disadvantage save DICE_ONE:n DICE_TWO:n DC:n {
 LOWEST: [lowest of DICE_ONE and DICE_TWO]
 result: [LOWEST is less than DC]
}

function: A:n is less than B:n {
 if A < B {result: TRUE}
 result: FALSE
}

\Check current exhaustion level and make the correct number of dice rolls\
    \@param A: Current EXHAUSTION\
    \@result: 2, 1 or 0\
function: chose save A:n { 
 if A < 2 {result: [normal save CONSAVE DC]}
 if A = 2 {result: [normal save CONSAVE DC] + [normal save CONSAVE DC]}
 if A > 2 {result: [disadvantage save CONSAVE CONSAVE DC] + [disadvantage save CONSAVE CONSAVE DC]}
}


TRAVEL: [travel MILES miles at SPEED speed in one day starting with EXHAUSTION exhaustion]
output TRAVEL named "Exhaustion"

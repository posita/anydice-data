set "maximum function depth" to 20


EXHAUSTION:0
CON: 16
CONMOD: ((CON-10)/2)
DICE: (d20 + CONMOD)
TIME: 0



\this function will check the probability of exhaustion for traveling\

function: travel H hours {
 TEMP: EXHAUSTION
 X: H-1
 if H < 9 {result: EXHAUSTION}
 TEMP: [save DICE DICE]
 result: [travel X hours]
}


function: save A:n {
 RET: EXHAUSTION
 if A < 10 {
  RET: RET + 1
  result: RET
 }
 result: RET
}

function: save A:n B:n {
 RET: EXHAUSTION
 if A < 10 {
  RET: RET + 1
  result: RET
 }
 if B < 10 {
  RET: RET + 1
  result: RET
 } 
 result: RET
}

function: Boolean {
 if A > B {result: A}
 result: B
}

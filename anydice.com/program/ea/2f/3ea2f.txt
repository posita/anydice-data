function: cut R:n at T:n {
  if R <= T { result: R }
  result: 0
}

D: [ cut 1d6 at 3 ]
R: 1d6 > 3
loop M over { -3..7 } {
   output 1dD + MdR named "Dado de Base + [M]d6 Dados de Apoio"
}

E: [ cut 1d10 at 4 ]
S: 1d10 > 4
loop M over { -3..7 } {
   output 1dE + MdS named "Dado de Base + [M]d10 Dados de Apoio"
}

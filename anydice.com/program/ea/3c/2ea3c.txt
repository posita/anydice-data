function: draw X {
 result: ((X+3)/4)+1
}

function: quick X:n {
 if [draw X] > 5 { result: [draw X] }
 result: [draw 1d(54-16)+16]
}

function: level headed X Y {
 result: [highest of [draw X] and [draw Y]]
}

function: level headed X:n Y:n choosing quick below Z {
 if [lowest of [draw X] and [draw Y]] > 5 {
  result: [level headed X Y]
 }
 else {
  if [level headed X Y] >= Z {
   result: [quick [highest of X and Y]]
  }
  else {
   result: [quick [lowest of X and Y]]
  }
 }
}


output [draw 1d54] named "Deadlands Draw"
output [quick 1d54] named "Deadlands Quick"

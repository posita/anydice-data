ATK: 1
DEF: 1

ATK_ROLL: ATK d {2, 1, 1, 0, 0, 0}
DEF_ROLL: DEF d {-1, 0, 0, 0, 0, 0}

function: combat outcome of ATK vs DEF {
  HITS: [count 1 in ATK]
  PIERCE: [count 2 in ATK]

  BLOCK: [highest of [count 1 in DEF] - PIERCE and 0]
  RIPOSTE: [count 2 in DEF]

  result: HITS + BLOCK - RIPOSTE
}

output [combat outcome of ATK_ROLL vs DEF_ROLL] named "damage done"

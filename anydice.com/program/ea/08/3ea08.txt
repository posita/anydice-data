\ returns 1 if X is 1, 2 if X is 2, else 0 \
function: plusvalue X:n {
  if X = 1 { result: 1 }
  else { if X = 2 { result: 2 } else { result: 0 } }
}

\ returns the larger of A and B \
function: maxtwo A:n B:n {
  if A > B { result: A } else { result: B }
}

\ roll ROLL:s, keep the highest die, add +1 per extra 1 and +2 per extra 2 \
function: score ROLL:s {
  MAX:   0
  BONUS: 0
  loop D over ROLL {
    MAX:   [maxtwo MAX D]
    BONUS: BONUS + [plusvalue D]
  }
  BONUS: BONUS - [plusvalue MAX]   \ remove the plus from the kept die \
  result: MAX + BONUS
}

\ probability tables for 1, 2, and 3 dice \
output [score 1d6] named "Probe mit 1D6"
output [score 2d6] named "Probe mit 2D6"
output [score 3d6] named "Probe mit 3D6"

set "position order" to "lowest first"

function: reroll DIE:d if under LIMIT:n {
  result: [reroll DIE as DIE if under LIMIT]
}
function: reroll ROLL:n as DIE:d if under LIMIT:n {
  if ROLL < LIMIT { result: DIE }
  else { result: ROLL }
}

function: hit H:n mod M crit C dmg D bonus B extra E ac A {
   if H = 1 {result: 0}
   else if H >= C {result: D+D+B+E}
   else if H+M >= A {result: D+B} 
   else {result: 0}
}


function: empower ROLL:s rerollvalue REROLLVAL:n dice D:d charisma CHA:n minrerollcount MIN:n {
   MOD: (CHA - 10) / 2
   MAX: [highest of MOD and 1]

   R: {}

   loop P over {1..#ROLL} {
      if P <= MAX & (P@ROLL <= REROLLVAL) > MIN {
         R: d{R, [reroll P@ROLL if under REROLLVAL+1]}
      } else {
         R: d{R, P@ROLL}
      }
   }

   result: R
}

function: if CONDITION:n then A else B {
  if CONDITION { result: A } else { result: B }
}

function: damage K:n BONUS ARMOR{
ATTACK: K+BONUS
result: [if ATTACK >=ARMOR then ATTACK-ARMOR else 0]

}

function: maybecrit K:d{
COUNT: [count 6 in K]
FOO: [if COUNT >1 then K+5 else K]
result: FOO
}
function: attack K:n D:n DAMAGE_ROLL:d BONUS ARMOR{
  TOTAL: K - D
if TOTAL < 0{
result: 0
}
else{
result: [damage [maybecrit DAMAGE_ROLL] BONUS ARMOR]

}
}

output 1d( [attack 1d10+4 1d10+18 6d6 4 18])

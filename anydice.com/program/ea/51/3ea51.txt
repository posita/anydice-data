function: D:n plus B:n hit AC:n{
  if D=20 { result: 4d8+8 }
  if (D+B)>=(AC+10){
    result: 4d8+8
  }
  if (D+B)>=AC{
    result: 2d8+4
  }
  else{ result: 0 }
}

function: D:n plus B:n hitt AC:n{
  if D=20 { result: 6d8+8 }
  if (D+B)>=(AC+10){
    result: 6d8+8
  }
  if (D+B)>=AC{
    result: 3d8+4
  }
  else{ result: 0 }
}

loop N over {20..30}{
  A: [1d20 plus 17 hit N] + [1d20 plus 17-5 hit N] + [1d20 plus 17-10 hit N]
  B: [1d20 plus 17 hitt N] + [1d20 plus 17-5 hit N]
  C: [[highest 1 of 2d20] plus 17 hit N] + [1d20 plus 17-5 hit N]
  D: [[highest 1 of 2d20] plus 17 hitt N]
  output A named "A AC [N]"
  output B named "B AC [N]"
  output C named "C AC [N]"
  output D named "D AC [N]"
}

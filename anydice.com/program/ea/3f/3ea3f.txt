function: die A:s die B:s vs TN:n plus BONUS:n step STEP:n {
ROLL: A + B + BONUS
MODIFIER: 0
if A = B { MODIFIER: 1}
if A + B = 2 { MODIFIER: -1 }

if ROLL >= TN { SUCCESS: 3 + MODIFIER}
if ROLL > TN + STEP { SUCCESS: 4 + MODIFIER}
if ROLL < TN { SUCCESS: 2 + MODIFIER}
if ROLL < TN - STEP {SUCCESS: 1 + MODIFIER}
if SUCCESS > 4 { SUCCESS: 4 }
if SUCCESS < 1 { SUCCESS: 1 }
result: SUCCESS
}

output [ die d10 die d10 vs 10 plus 1 step 10 ] named "degrees of success"

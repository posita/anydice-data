function: adv ROLL:s {
 result: 1@ROLL-#ROLL@ROLL
}

function: dis ROLL:s {
 X: 1@ROLL
 loop N over {1..#ROLL-1}{
  Y: N@ROLL-(N+1)@ROLL
  if(Y<X){X: Y}
 }
 result: X
}

D:4

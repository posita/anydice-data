function: successes ROLL:s {
  if ROLL >= 25 {result: 3 } \ 3 successes \
  if ROLL >= 20 {result: 2 } \ 2 successes \
  if ROLL >= 15 {result: 1 } \ 1 success \
  if ROLL >= 10  {result: 0 } \ partial \ 
  result: -1 \ failure \
}

loop TYPE over {-1, 0 , 1} {
  loop MOD over {0..6} {
    if TYPE = -1 { output [successes [lowest 2 of 3d{0..9}]+MOD] named "+[MOD] Disadvantage" }
    if TYPE = 0 { output [successes 2d{0..9}+MOD] named "+[MOD]" }
    if TYPE = 1 { output [successes [highest 2 of 3d{0..9}]+MOD] named "+[MOD] Advantage" }
  }
}

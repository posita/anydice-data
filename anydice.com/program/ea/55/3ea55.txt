function: with ROLL:n attack DICE:d plus ATK:n attack on hit plus HIT:n against AC:n{
	if ROLL = 20 {
		result: 2*(DICE+ATK)
	} 
	else {
		if (ROLL+HIT)>=AC {
			result: DICE+ATK
		}
		else {
			result: 0
		}
	}
}

\Make Attacks:\
ENEMY_HPS: {50,60,40}
ENEMY_ACS: {14,16,12}
\Karl:\
\Normal:\

loop E_AC over ENEMY_ACS {
	output [with 1d20 attack 1d12 plus 0 attack on hit plus 5 against E_AC] named "Attack (Karl Lance) against AC [E_AC]. Damage Type: Piercing"
}

\- Tavern Brawler -\

DAMAGE_BONUS: 0

function: tavern brawler D:d {result: [tavern brawler D D]}
function: tavern brawler N:n M:n{
  if N = 1 {result: M}
  result: N
}

SIDES: {6,8,10,12}


loop DIE over SIDES {
  output dDIE + DAMAGE_BONUS named "Roll d[DIE]"
  output [tavern brawler dDIE]+ DAMAGE_BONUS named "Tavern Brawler d[DIE]"
}

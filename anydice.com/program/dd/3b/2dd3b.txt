function: burst A:n B:n C:n {
  if A = 1 & B = 1 & C = 1 { result: 1 }
  if A = 1 & B = 1 & C != 1 { result: C }
  if A = 1 & B != 1 & C = 1 { result: B }
  if A != 1 & B = 1 & C = 1 { result: A }
  if A = B | A = C | B = C {
    result: A + B + C + 1d{[maximum of A]}
  }
  result: A + B + C
}

output [burst 1d6 1d6 1d6]

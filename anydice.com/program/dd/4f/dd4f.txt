D: 3d6
AWEAPON: 5d6
A: 5

function: attack ATTACKER:n using AWEAPON:s attacks DEFENDER:s {
SORTEDA: [sort AWEAPON]
SORTEDD: [sort DEFENDER]
FINALA: {1..ATTACKER}@SORTEDA
FINALD: {1..5}@SORTEDD
result: [attack helper FINALA FINALD]
}

function: attack helper FINALDICE:s DEFENDER:s {
result: FINALDICE > DEFENDER
}

output [attack A using AWEAPON attacks D]

\****Defensive Stuff Calculator ****\


function: concentration ROLL:n plus BONUS:n plus LUCK:n {
if ROLL + BONUS + LUCK >= DAMAGE {result: 1}
result: 0
}

function: save ROLL:n plus BONUS:n plus LUCK:n {
if ROLL + BONUS + LUCK >= TARGET {result: 1}
result: 0
}

DAMAGE: 10
TARGET: 15

output [concentration 1d20 plus 1 plus 0] named "Nothing (conc.)"
output [concentration 1d20 plus 1 plus 1d10] named "Nothing (conc.)"
output [save 1d20 plus 1 plus 0] named "Nothing (save)"
output [save 1d20 plus 1 plus 1d10] named "Nothing (save)"

output [concentration 1d20 plus 6 plus 0] named "Resilient (conc.)"
output [concentration 1d20 plus 6 plus 1d10] named "Resilient (conc.) + Luck"
output [save 1d20 plus 6 plus 0] named "Resilient (save)"
output [save 1d20 plus 6 plus 1d10] named "Resilient (save) + Luck"

output [concentration [highest 1 of 2d20] plus 1 plus 0] named "Eldritch Mind (conc.)"
output [concentration [highest 1 of 2d20] plus 1 plus 1d10] named "Eldritch Mind (conc.) + Luck"
output [save 1d20 plus 1 plus 0] named "Eldritch Mind (save)"
output [save 1d20 plus 1 plus 1d10] named "Eldritch Mind (save) + Luck"

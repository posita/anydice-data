NUMDICE: 2

function: explrankhelp LOWEST:n of DICE:d rank RANK:n {
  if (LOWEST > RANK) {
    result: {1..NUMDICE}@DICE
  } else {
    result: {1..NUMDICE-1}@DICE + d6
  }
}

function: explrank DICE:d rank RANK:n {
  result: [explrankhelp NUMDICE@DICE of DICE rank RANK]
}

loop N over {1..3} {
  output [explrank NUMDICEd6 rank N] named "Rank=[N]"
}

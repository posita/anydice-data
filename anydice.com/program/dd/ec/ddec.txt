function: ROLLSIZE:n k KEEPSIZE:n {
 result: [convert [highest KEEPSIZE of ROLLSIZE d{2..5, 600}]]
}

function: convert SUM:n {
 if SUM >= 600 {
  SIXESROLLED: SUM / 600
  result: SUM - SIXESROLLED * 594 + SIXESROLLED d6
 }
 result: SUM
}

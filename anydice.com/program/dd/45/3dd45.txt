\- Tavern Brawler with Damage Bonus-\

DAMAGE_BONUS: 0

function: tavern brawler D:d {result: [tavern brawler D D]}
function: tavern brawler N:n M:n{
  if N = 1 {result: M}
  result: N
}

SIDES: {6,8,10,12}

if DAMAGE_BONUS {
  loop DIE over SIDES {
    output dDIE + DAMAGE_BONUS named "Roll d[DIE] + [DAMAGE_BONUS]"
    output [tavern brawler dDIE]+ DAMAGE_BONUS named "Tavern Brawler d[DIE] + [DAMAGE_BONUS]"
  }
}
else
{
  loop DIE over SIDES {
    output dDIE named "Roll d[DIE]"
    output [tavern brawler dDIE] named "Tavern Brawler d[DIE]"
  }
}

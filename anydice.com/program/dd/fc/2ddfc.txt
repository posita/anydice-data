\**** Damage calculator Effect Roll ****\

function: attackroll ROLL:n vs FORT:n {
 if ROLL + EFFECTBONUS + SPLDAMAGE - FORT > 0 { result: ROLL + EFFECTBONUS - FORT + SPLDAMAGE} 
 result: 0
}

ENEMYFORT: 12

EFFECTBONUS: 2
BONUSDMG: 0
SPLDAMAGE: 8

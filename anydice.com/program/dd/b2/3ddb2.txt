function: roll ACC plus BONUS {
  if ACC >= 0 {
    result: 1d8+1d4 + BONUS + [highest 1 of ACCd3]
  }
  else {
    result: 1d8+1d4 + BONUS + [lowest 1 of ACCd3]
  }
}

\loop N over {-2..2} {\
\  output [roll N plus 0] named "[N] roll"\
\}\

output [roll 0 plus 0] named "raw"
output [roll 1 plus 0] named "acc"
output [roll 0 plus 2] named "bonus"

\output d20 named "raw"\
\output [highest 1 of 2d20] named "adv"\
\output d20+3 named "bonus"\

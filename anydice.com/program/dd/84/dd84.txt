function: mine DIG:s vs ORE:n {
    result: (DIG >= ORE) * ORE - #DIG
}
function: test R:s vs TARG:n {
    result: [lowest of (R >= TARG) and TARG]
}

function: poolroll R:s {
    result: ((R >= #R)) * 2
}

loop T over {1}{
    loop R over {1..6}{
        \output [mine Rd6 vs T] named "[R] dig [T]"\
        \output [test Rd6 vs T] named "[R] test [T]"\
        \output Rd{0,1} >= Td{0,1} named "[R] vs [T]"\
        output [poolroll Rd6] named "roll [R]"
    }
}

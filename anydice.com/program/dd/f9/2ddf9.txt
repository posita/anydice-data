\**** Damage calculator Lightning spell ****\

function: attackmain ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: (WPNDAMAGEA*2) + LIGHTNINGC} 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS - 0 >= DEF { result: WPNDAMAGEA + LIGHTNINGC}
 result: 0
}

function: attackcombo ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: WPNDAMAGEB*2 + LIGHTNINGC} 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS >= DEF { result: WPNDAMAGEB + LIGHTNINGC}
 result: 0
}

\looped against enemy DEF\
ENEMYDEF: d{9}

ATTACKBONUS: 3
RAGEBONUS: 0
CRITROLL: 12
WPNDAMAGEA: 12
WPNDAMAGEB: 12
LIGHTNINGC: 6
COMBOTIMES: 1
output [attackmain 2d6 vs ENEMYDEF] + ([attackcombo 2d6 vs ENEMYDEF] * COMBOTIMES) named "Lightning Chain"

output 2d6

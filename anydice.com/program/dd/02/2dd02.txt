GRIT: 20
EVA: 87

function: wep DIE:n todo DMG:d{
  if (DIE < EVA) {
    result: #DMG 
  } else if (DIE < 20) {
    result: DMG
  } else {
    result: {1..#(DMG*2)}@{1,47}
  }
}

function: hmg D:n {
  if (D < EVA) {
    result: 0
  } else if (D < 20) {
    result: 2d6+4
  } else {
    result: {1,2}@4d6+4
  }
}

output [wep 1d20+GRIT-1d6 todo 2d6+4] named "HMG?"
output [hmg 1d20+GRIT-1d6] named "HMG"

function: tier A skill B multiroll {
 result: Ad10+B*A
}

function: tier A skill B {
 result: (1d10+B*A)*A
}

function: check R:n against dc C tier T {
 if(R < (C*T / 2)) {
  result: -1
 } else {
  if(R < C) {
   result: 0
  } else {
   if(R < (C*T + (C*T/2))) {
    result: 1
   } else {
    if(R < (C*T*2)) {
     result: 2
    } else {
     result: 3
    }
   }
  }
 }
}

loop T over {1..5} {
 loop S over {0..4} {
  DC: 5
  DCXT: DC*(T)
  DCT: T-1
  output [check [tier T skill S] against dc DC tier T] named "t[T]d10+[S] DC[DCXT]"
 }
}

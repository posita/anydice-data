
function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

\FIEND: 1 if target is fiend or undead, 0 if neither \
\FIREDMG: 0 means immune, 1 means normal, 2 means resists, 3 means vulnerable \

FIEND: 0
FIREDMG: 1
SMITESLOT: 0
RAGEDMG: 0

FIREDICE: 8
IMPSMITE: 1

if SMITESLOT = 0 {
SMITEDICE: IMPSMITE
}
if SMITESLOT = {1..4} {
SMITEDICE: IMPSMITE + SMITESLOT + FIEND + 1
}
if SMITESLOT = {5..9} {
SMITEDICE: IMPSMITE + 4 + FIEND + 1
}

STRDMG: 8
AURADMG: 4
WPNDMG: 3
GWMDMG: 10

GWFFOUR: [reroll {1,2} on d4 to d4]
GWFSIX: [reroll {1,2} on d6 to d6]
GWFEIGHT: [reroll {1,2} on d8 to d8]
GWFTEN: [reroll {1,2} on d10 to d10]
GWFDOZEN: [reroll {1,2} on d12 to d12]
SWORDGWF: 2 d GWFSIX
SMITEGWF: SMITEDICE d GWFEIGHT
FIREGWF: FIREDICE d GWFEIGHT

if FIREDMG = 0 {
FIRETOTAL: 0
CRITFIRE: 0
}
if FIREDMG = 1 {
FIRETOTAL: FIREGWF
CRITFIRE: FIREDICE*8
}
if FIREDMG = 2 {
FIRETOTAL: FIREGWF/2
CRITFIRE: FIREDICE*4
}
if FIREDMG = 3 {
FIRETOTAL: FIREGWF + FIREGWF/2
CRITFIRE: FIREDICE*12
}

if FIEND = 0 {
AVENGER: 0
NUMTEN: 0
}
else {
AVENGER: 2 d GWFTEN
NUMTEN: #(AVENGER)
}

NUMSIX: #(SWORDGWF)
NUMEIGHT: #(SMITEGWF)+#(FIREGWF)
NUMSMITE: #(SMITEGWF)

if FIEND = 0 {
output SWORDGWF + SMITEGWF + FIRETOTAL + GWMDMG + STRDMG + WPNDMG + AURADMG + AVENGER + RAGEDMG named "GWM+GWF ([NUMSIX]d6, [NUMEIGHT]d8, [NUMTEN]d10)"
output SWORDGWF + SMITEGWF + FIRETOTAL + GWMDMG + STRDMG + WPNDMG + AURADMG + AVENGER + RAGEDMG + NUMSIX*6 + NUMSMITE*8 + CRITFIRE + NUMTEN*10 named "Crit GWM+GWF ([NUMSIX]d6, [NUMEIGHT]d8, [NUMTEN]d10)"
}
else {
output SWORDGWF + SMITEGWF + FIRETOTAL + GWMDMG + STRDMG + WPNDMG + AURADMG + AVENGER + RAGEDMG named "GWM+GWF v fiend/undead ([NUMSIX]d6, [NUMEIGHT]d8, [NUMTEN]d10)"
output SWORDGWF + SMITEGWF + FIRETOTAL + GWMDMG + STRDMG + WPNDMG + AURADMG + AVENGER + RAGEDMG + NUMSIX*6 + NUMSMITE*8 + CRITFIRE + NUMTEN*10 named "Crit GWM+GWF v fiend/undead([NUMSIX]d6, [NUMEIGHT]d8, [NUMTEN]d10)"
}

output 8d8 named "fire damage TWF"

USE_HERO_POINT: 0
USE_REWRITE_FATE: 0
USE_CALLING: 1

function: ROLL:n plus CHECK:n vs DC:n {
DOS: 0
if ROLL + CHECK <= DC - 10 { DOS: -1 }
if ROLL + CHECK >= DC      { DOS: 1 }
if ROLL + CHECK >= DC + 10 { DOS: 2 }

if ROLL = 1 { DOS: DOS - 1 }
if ROLL = 20 { DOS: DOS + 1 }

if DOS < -1 { DOS: -1 }
if DOS > 2 { DOS: 2 }

result: DOS
}

function: LEVEL:n DC:n ROLL:n REROLL:n {
BONUS: LEVEL + 2 MYTHIC: 8
if LEVEL >= 3 { BONUS: LEVEL + 4 MYTHIC: 6 }
if LEVEL >= 7 { BONUS: LEVEL + 6 MYTHIC: 4 }
if LEVEL >= 15 { BONUS: LEVEL + 8 MYTHIC: 2 }

if USE_CALLING = 1  { BONUS: BONUS + MYTHIC }

DOS: [ROLL plus BONUS vs DC]
if DOS > 0 { result: DOS }

if USE_REWRITE_FATE = 1 {
if USE_CALLING = 1 {
DOS: [REROLL plus BONUS vs DC]
}
if USE_CALLING = 0 {
DOS: [REROLL plus BONUS + MYTHIC vs DC]
}
}
if USE_HERO_POINT {
DOS: [REROLL plus BONUS vs DC]
}

result: DOS
}

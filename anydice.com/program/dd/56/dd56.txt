  HIGH: ONE + TWO
  if ONE + THREE > HIGH { HIGH: ONE + THREE }
  if ONE + FOUR > HIGH { HIGH: ONE + FOUR }
  if TWO + THREE > HIGH { HIGH: TWO + THREE }
  if TWO + FOUR > HIGH { HIGH: TWO + FOUR }
  if THREE + FOUR > HIGH { HIGH: THREE + FOUR }
  if HIGH > 6 { result: HIGH - 8 }
} output [stats d6 and d6 and d6 and d6] named "stats"

    if ONE + TWO + THREE = 3 { result: 20 }
    else { result: 0 } }
  if ONE = TWO {
    if ONE = 6 { result: 20 }

} output [tech d6 and d6 and d6] named "technique roll"

function: reverse SEQUENCE:s{
 R: {}
 loop P over {1..#SEQUENCE}{
  R: {P@SEQUENCE, R}
 }
 result: R
}

function: HOPE:n FEAR:n plus MOD:n for DAM:d plus X:n {
 if HOPE = FEAR {result: 1@[reverse {DAM}]+DAM+X}
 if HOPE+FEAR+MOD >= DIFFICULTY {result: DAM+X}
 else {result: 0}
}

HOPE: d12
FEAR: d12
DIFFICULTY: 10

function: countsuccess in POOL:n against DEFENSE:n with DEGAT_A and ARMOR_A against DEGAT_B and ARMOR_B {
  
  if POOL - DEFENSE >= 0{
      if ARMOR_B > (POOL - DEFENSE + DEGAT_A) {
         result: 0}
      else {
         result: POOL - DEFENSE + DEGAT_A}
      }
   else {
      if ARMOR_A > (-(POOL - DEFENSE) + DEGAT_B) {
         result: 0}
      else {
         result: -(-(POOL - DEFENSE) + DEGAT_B)}
      } 
}


\ ---- elegy ---- \

\ Elegy 3.5 \
function: elegy three with ATTR:n {
  A: d10 < ATTR
  B: d10 < ATTR
  result: A + B
}

function: elegy four D:n A:n {
  SUM: D + A
  if SUM >= 15 { result: 2 }
  if SUM >= 10 { result: 1 }
  result: 0
}

function: elegy alt D:n A:n {
  SUM: D + A
  if SUM >= 12 { result: 2 }
  if SUM >= 7 { result: 1 }
  result: 0
}

loop ATTR over {5..7}{
  output [elegy three with ATTR] named "E-3.5, stat=[ATTR]"
}

loop ATTR over {-1..1} {
  output [elegy four 2d10 ATTR] named "E-4.0, stat: [ATTR]"
}

loop ATTR over {-1..1} {
  output [elegy alt 2d8 ATTR] named "E-4.0 with 2d8, stat: [ATTR]"
}

output 2d6 named "basic 2d6 odds"
output 2d8 named "basic 2d8 odds"
output 2d10 named "basic 2d10 odds"

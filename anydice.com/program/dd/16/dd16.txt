function: X:n attack {
if X = 20 {result: 2d10 + MOD }

if X = 1 {result: 0}

if X + MOD + PROF + MISC >= AC { result: 1d10 + MOD } 

else{ result: 0}
}

function: X:n sharpshooter{
if X = 20 {result: 2d10 + MOD +10}

if X = 1 {result: 0}

if X + MOD + PROF + MISC - 5 >= AC{ result: 1d10 + MOD +10} 

else{ result: 0}
}

function: X:n supdice Y:n supsharpshooter{
if X = 20 {result: 2d10 + MOD +10}

if X = 1 {result: 0}

if (X + MOD + PROF + MISC - 5) >= AC{ result: 1d10 + MOD +10 +Y} 

else{ result: 0}
}

function: X:n supdice Y:n persharpshooter{
if X = 20 {result: 2d10 + MOD +10}

if X = 1 {result: 0}

if (X + MOD + PROF + MISC - 5 +Y) >= AC{ result: 1d10 + MOD +10} 

else{ result: 0}
}


AC: 15
MOD: 5
PROF: 6
MISC: 2

output [d20 sharpshooter] + [d20 sharpshooter] + [d20 sharpshooter] named "sharpshooter attacks"

output [d20 supdice d10 supsharpshooter] + [d20 supdice d10 supsharpshooter] + [d20 supdice d10 supsharpshooter] named "supsharpshooter attacks"

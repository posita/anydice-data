
# Define possible values for each die
standard_die = [1, 2, 3, 4, 5, 6]
modified_die = [2, 3, 4, 5, 6, 6]

total_outcomes = 0
favorable_outcomes = 0

# Iterate over all possible combinations
for d1 in standard_die:
    for d2 in standard_die:
        for d3 in modified_die:
            total_outcomes += 1
            if d1 + d2 + d3 >= 11:
                favorable_outcomes += 1

# Calculate probability
probability = favorable_outcomes / total_outcomes
print(f"Favorable outcomes: {favorable_outcomes}")
print(f"Total outcomes: {total_outcomes}")
print(f"Probability (sum â‰¥ 11): {probability:.4f}")

function: roll ROLL:s sweep SWEEP:s{
   GLOOM: [count 1 in ROLL] + [count 1 in SWEEP]
   BRILL: [count 6 in ROLL]
   if GLOOM + BRILL = 0 {result: 0}
   if GLOOM > BRILL {result: -1}
   else {result: BRILL}
}

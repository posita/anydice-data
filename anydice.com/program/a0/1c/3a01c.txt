function: range on DIE:n between LOW:n and HIGH:n {
  if DIE >= LOW & DIE <= HIGH { result: 1 }
  result: 0
}


output 1d20 = 1 named "The creature gains an additional level of Blood Debt, and must re-roll this table."
output [range on 1d20 between 2 and 14] = 1 named "The creature does not gain the benefit of a long rest"
output [range on 1d20 between 15 and 19] = 1 named "The dosage required to gain [[#Blood Frenzied]] doubles."
output 1d20 = 20 named "The creature looses a level of [[#Blood Debt]]"

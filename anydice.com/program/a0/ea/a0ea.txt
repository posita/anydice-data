TN: 6

function: skilled N:n {
 result: (N >= TN) - (N = 1)
}

function: unskilled N:n {
 result: (N = 6) - (N = 1) - (N = 2)
}

STAT: [skilled d10]
BOTCH: [unskilled d6]

output 1dSTAT + 4dBOTCH named "Stat 1"
output 2dSTAT + 3dBOTCH named "Stat 2"
output 3dSTAT + 2dBOTCH named "Stat 3"
output 4dSTAT + 1dBOTCH named "Stat 4"
output 5dSTAT + 0dBOTCH named "Stat 5"

function: reroll lowest of ROLL:s as REROLL:d if less than MIN:n {
  LOWEST: (#ROLL)@ROLL       \ the lowest die is sorted last \
  if LOWEST >= MIN {
    result: ROLL + 0
  } else {
    REST: {1..#ROLL-1}@ROLL  \ all but the lowest die \
    result: REST + REROLL
  }
}

function: reroll highest of ROLL:s as REROLL:d if more than MAX:n {
  HIGHEST: 1@ROLL       \ the highest die is sorted first \
  if HIGHEST <= MAX {
    result: ROLL + 0
  } else {
    REST: {2..#ROLL}@ROLL  \ all but the lowest die \
    result: REST + REROLL
  }
}

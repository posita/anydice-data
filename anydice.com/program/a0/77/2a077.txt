function: gwf DIE:d {
  result: [reroll DIE as DIE if under 3]
}
function: reroll ROLL:n as DIE:d if under LIMIT:n {
  if ROLL < LIMIT { result: DIE }
  else { result: ROLL }
}

loop SIDES over {10} {
  output 1d[gwf dSIDES]+2d[gwf dSIDES]*2+11+5d10 named "Holy Avg GWF-HF + BanishingSm"
}

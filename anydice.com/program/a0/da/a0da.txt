function: explode SC N:n {
 if N = 1 { result: 1 + [explode SC 1d10] }
 else if N >= SC {result: 1}
 else {result: 0}
}


function: roll dice DC for success SC {
  TOTAL:0
  loop N over  {1..DC} {
     TOTAL: TOTAL + [explode SC 1d10]
  }
  result: TOTAL
}

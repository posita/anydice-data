function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

output [attack 1d20 plus 10 vs 18 for 1d8+5 crit 1d8 on 20] named "Normal Attack"
output [attack 1d20 plus 5 vs 18 for 1d8+15 crit 1d8 on 20] named "Gambling Shot"
output [attack [highest 1 of 2d20] plus 10 vs 18 for 1d8+5+3d6 crit 1d8+3d6 on 20] named "Oath Bow"
output [attack [highest 1 of 2d20] plus 5 vs 18 for 1d8+15+3d6 crit 1d8+3d6 on 20] named "Gambling Shot + Oath Bow"

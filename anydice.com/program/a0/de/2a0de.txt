function: reroll N:n dice showing REROLLED_VALUES:s in S:s {
 TOTAL: S
 loop D over REROLLED_VALUES{
  NUM: [lowest of N and [count D in S]] \Number of dice being rerolled at this step\
  TOTAL: TOTAL - (D * NUM) + NUMd6 \Subtract the values of the dice being rerolled, then roll NUM d6s and add them to the total\
  N: N - NUM
 }
 result: TOTAL
}

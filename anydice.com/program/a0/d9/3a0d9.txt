function: chaotic of FIRST:n and SECOND:n tiebreak by TIEBREAK:n {
  if FIRST > SECOND {
    result: -FIRST
  } else if FIRST < SECOND {
    result: SECOND
  } else {
    if TIEBREAK = 1 {result: -FIRST} else {result: SECOND}
  }
}

function: chaos {result: [chaotic of 1d6 and 1d6 tiebreak by 1d2]}

loop N over {0..8} {
output 4d10+N*[chaos] named "Level [N] chaos"
}

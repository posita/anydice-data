function: to hit AC:n {
  result: (d20+7)>=AC
}

function: to hit AC:n advantage {
  result: ([highest 1 of 2d20]+7)>=AC
}
function: to hit AC:n disadvantage {
  result: ([lowest 1 of 2d20]+7)>=AC
}

function: X:n crit {
if X = 20 {result: 2d6}
result: 0
}

function: X:n miss {
  if X = 1 {result: 0}
  result: 1
}

loop AC over {10..25} {
  output 4d4+4 named "Magic Missile at AC [AC]"

  output 
    ((2d6)*[to hit AC disadvantage]+[[lowest 1 of 2d20] crit])*[[lowest 1 of 2d20] miss] + 
    ((2d6)*[to hit AC disadvantage]+[[lowest 1 of 2d20] crit])*[[lowest 1 of 2d20] miss] + 
    ((2d6)*[to hit AC disadvantage]+[[lowest 1 of 2d20] crit])*[[lowest 1 of 2d20] miss]  
    named "Scorching Ray with Disadvantage at AC [AC]"

output 
    ((2d6)*[to hit AC]+[1d20 crit])*[1d20 miss] + 
    ((2d6)*[to hit AC]+[1d20 crit])*[1d20 miss] +
    ((2d6)*[to hit AC]+[1d20 crit])*[1d20 miss]  
    named "Scorching Ray (Normal) at AC [AC]"

  }

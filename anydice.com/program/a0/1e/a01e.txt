function: fight AA:s and BA:s and CA:s and DA:s vs AD:s and BD:s and CD:s and DD:s {
  AWIN: 0
  DWIN: 0
  COUNT: [lowest of #AA+#BA+#CA+#DA and #AD+#BD+#CD+#DD]
  loop N over {1..COUNT}{
    if N@[sort{AA, BA, CA, DA}] > N@[sort{AD, BD, CD, DD}] {
      AWIN: AWIN + 1
    }
    if N@[sort{AA, BA, CA, DA}] < N@[sort{AD, BD, CD, DD}] {
      DWIN: DWIN + 1
    }
  }

  result:AWIN-DWIN
}

function: combine A:s and B:s and C:s and D:s {
  R: {}
  R: {{1..3}@[sort{A, B, C, D}]}
  result: R
}

R: {0,0,4,4,4,4}
G: {2,2,2,2,6,6}
B: {5,5,5,1,1,1}
N: 3

A: 2dR
D: 2dG

output [fight 1dR and {} and {} and {} vs 1dB and {} and {} and {}]

function: normalroll TENS:s ONES:s {
  if TENS = 10 & ONES = 10 {result: 100}
  if TENS = 10 & ONES < 10 {result: ONES}
  if TENS < 10 & ONES = 10 {result: 10 * TENS}
  if TENS < 10 & ONES < 10 {result: 10 * TENS + ONES}
}

function: advantageroll TENS:s ONES:s {
  if TENS = 10 & ONES = 10 {result: 100}
  if TENS = 10 & ONES < 10 {result: ONES}
  if TENS < 10 & ONES = 10 {result: TENS}
  if TENS < 10 & ONES < 10 {
    if TENS < ONES {result: 10 * TENS + ONES}
    if TENS > ONES {result: 10 * ONES + TENS}
    if TENS = ONES {result: 10 * TENS + ONES}
  }
}

function: disadvantageroll TENS:s ONES:s {
  if TENS = 10 & ONES = 10 {result: 100}
  if TENS = 10 & ONES < 10 {result: 10 * ONES}
  if TENS < 10 & ONES = 10 {result: 10 * TENS}
  if TENS < 10 & ONES < 10 {
    if TENS < ONES {result: 10 * ONES + TENS}
    if TENS > ONES {result: 10 * TENS + ONES}
    if TENS = ONES {result: 10 * TENS + ONES}
  }
}

\ Define dice for the tens and ones place \
TENS: d10
ONES: d10

\ Calculate probabilities for each mechanic \
output [normalroll TENS ONES] named "Normal Roll"
output [advantageroll TENS ONES] named "Advantage"
output [disadvantageroll TENS ONES] named "Disadvantage"

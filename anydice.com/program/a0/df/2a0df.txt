function: reroll N:n dice showing REROLLED_VALUES:s in S:s {
 TOTAL: S
 loop D over REROLLED_VALUES{
  NUM: [lowest of N and [count D in S]] \Number of dice being rerolled at this step\
  TOTAL: TOTAL - (D * NUM) + NUMd6 \Subtract the values of the dice being rerolled, then roll NUM d6s and add them to the total\
  N: N - NUM
 }
 result: TOTAL
}

output 3d6 named "Baseline"
output [reroll 1 dice showing {1} in 3d6] named "1 reroll, only 1s"
output [reroll 2 dice showing {1} in 3d6] named "2 rerolls, only 1s"
output [reroll 3 dice showing {1} in 3d6] named "3 rerolls, only 1s"
output [reroll 1 dice showing {1,2} in 3d6] named "1 reroll, 1-2 ascending"
output [reroll 2 dice showing {1,2} in 3d6] named "2 rerolls, 1-2 ascending"
output [reroll 3 dice showing {1,2} in 3d6] named "3 rerolls, 1-2 ascending"
output [reroll 1 dice showing {2,1} in 3d6] named "1 reroll, 1-2 descending"
output [reroll 2 dice showing {2,1} in 3d6] named "2 rerolls, 1-2 descending"
output [reroll 3 dice showing {2,1} in 3d6] named "3 rerolls, 1-2 descending"
output [reroll 1 dice showing {1,2,3} in 3d6] named "1 reroll, 1-3 ascending"
output [reroll 2 dice showing {1,2,3} in 3d6] named "2 rerolls, 1-3 ascending"
output [reroll 3 dice showing {1,2,3} in 3d6] named "3 rerolls, 1-3 ascending"
output [reroll 1 dice showing {3,2,1} in 3d6] named "1 reroll, 1-3 descending"
output [reroll 2 dice showing {3,2,1} in 3d6] named "2 rerolls, 1-3 descending"

function: minimum of DIE:d {
  result: 1@{DIE}
}

function: N:n between L:n and U:n {
  if ((L <= N) & (N <= U)) {result: N}
  else {
    if (L > N) {result: L}
    if (U < N) {result: U}
  }
}

function: K:n of DIE:d mitigate N:n by M:n {
  \roll K copies of DIE, add up to M to the smallest N without exceeding the extreme values of DIE\
  if (N <= 0) {result: K d DIE}
  if (N > K) {N: K}
  MAX: [maximum of DIE]
  MIN: [minimum of DIE]
  R: 0
  loop I over {1..N}{
    R: R + [(DIE + M) between MIN and MAX]
  }
  if (K > N) {
    loop J over {1..K-N}{
      R: R + DIE
    }
  }
  result: R
}

if R = 2 { result: E + 2 }
else if R < 7 { result: E + 1 }
else if R = 20 { 
 if E < 2 { result: 0 }
 else { result: E - 2 }
 }
else if R > 16 {
 if E < 1 { result: 0 }
 else { result: E - 1 }
 }
result: E
}


\ attack die value \
X: 3d8
\ defense die value \
Y: 3d8





\ Block function \

function: block A B {
 C: A / 2
 if(C > B) { result: A - B }
 result: A - C
}



\ Armor Rolltable Functions \

function: arm light leather A:n B:n {
 if(B >= 17) { result: 0 }
 result: A
}

function: arm medium chain A:n B:n {
 if(B >= 19) { result: 0 }
 if(B >= 13) { result: [block A 5] }
 result: A
}

function: arm heavy plate A:n B:n {
 if(B >= 15) { result: [block A 10] }
 if(B >= 10) { result: [block A 5] }
 result: A
}


\ Attack Function (set the target armor here) \

function: atk A {
 result: [arm heavy plate A Y]
}




\ Weapon Rolltable Functions \

function: wep dagger A:n {
 if A >= 18 { result: 24 }
 if A >= 9 { result: 4 }
 result: 0
}

function: wep greatsword A:n {
 if A >= 18 { result: 16 }
 if A >= 15 { result: 10 }
 if A >= 10 { result: 6 }
 result: 0
}

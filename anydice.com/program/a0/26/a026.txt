function: combat ATKDEF:n ATKATK:n vs DEFDEF:n DEFATK:n {
  O: (DEFDEF - ATKATK) <= 0
  P: (ATKDEF - DEFATK) <= 0

  if(O & P) {
   result: 3
  } else {
   if(O) {
    result: 1
   } else {
    if(P) {
     result: 2
    } else {
     result: 0
    }
   }
  }

}

function: ctwo ATKDEF:n ATKATK:n vs DEFDEF:n DEFATK:n {
  O: (DEFDEF - ATKATK) <= 0
  P: (ATKDEF - DEFATK) <= 0
  Q: (DEFDEF - ATKATK) <= -4
  R: (ATKDEF - DEFATK) <= -4

  if(O & P) {
   if(Q & R) {
    result: 3
   } else{if(Q) {
    result: 1
   } else{if(R) { 
    result: 2
   } else {
    result: 3
   }
   }
   }
  } else {
   if(O) {
    result: 1
   } else {
    if(P) {
     result: 2
    } else {
     result: 0
    }
   }
  }

}

function: cthree ATKDEF:n ATKATK:n vs DEFDEF:n DEFATK:n {
  O: (DEFDEF - ATKATK)
  P: (ATKDEF - DEFATK)

  if(O < 0 & P < 0) {
   result: 3
  } else {
   if(P < 0 & O > P) {
    result: 2
   } else {
    if(O < 0 & P > O) {
     result: 1
    }
   }
  }
  result: 0

}

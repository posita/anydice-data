AC: 25
DMG_D: 4d12
DMG_F: 10
ATK: 19
PROF: 6

function: swing ROLL:n {
  if ROLL = 20 {
    result: 2dDMG_D + DMG_F
  } else if ROLL + ATK >= AC {
    result: DMG_D + DMG_F
  } else {
    result: 0
  }
}

function: protecc DMG:n {
  REDUCED: DMG - d10 - PROF
  result: (REDUCED >= 0) * REDUCED
}

output [swing d20] named "Nothing"
output [swing 2@2d20] named "Disadvantage"
output [protecc [swing d20]] named "Protecc"

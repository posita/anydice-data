function: minimum of N:n and M:n {
  if (N < M) {result: N}
  else {result: M}
}

function: K:n of DIE:d mitigate N:n by M:n {
  \roll K copies of DIE, add up to M to the smallest N without exceeding the maximum value of DIE\
  if (N <= 0) {result: K d DIE}
  if (N > K) {N: K}
  MAX: [maximum of DIE]
  R: 0
  loop I over {1..N}{
    R: R + [minimum of DIE + M and MAX]
  }
  if (K > N) {
    loop J over {1..K-N}{
      R: R + DIE
    }
  }
  result: R
}

BIGDICE: 12d6
MEDDICE: 8d6
SMALLDICE: 4d6
HIT: 5
MISS: 1
GLITCH: 1
HARDMISS: 2
DRAINRESIST: 4

function: COUNT:s at least THRESHOLD:n {
  result: (COUNT >= THRESHOLD)
}

function: COUNT:s at most THRESHOLD:n {
  result: (COUNT <= THRESHOLD)
}

function: COUNT:s glitch at THRESHOLD:n {
  result: [COUNT at most THRESHOLD] >= ((#(COUNT)+1) / 2)
}

function: expected drain for COUNT:s at THRESHOLD:n with RESIST:n  {
  result: [COUNT at most THRESHOLD] - RESIST
}

loop N over {1..4}{
 output Nd4 named "[N]d4"
 output 1dN named "1d[N]"
}

CRIT: 20
TARGET: 17

function: attack ROLL:n p AB:n dmg D p DB {
  if ROLL >= CRIT {
    result: 2dD + DB
  }
  else {
    if ROLL + AB >= TARGET {
      result: D + DB
    }
    else {
      result: 0
    }
  }
}

function: N multiattack p AB dmg D p DB {
  if N = 0 {
    result: 0
  } else {
    result: [attack 1d20 p AB dmg D p DB] + [N-1 multiattack p AB dmg D p DB]
  }
}

function: N multiadv p AB dmg D p DB {
  if N = 0 {
    result: 0
  } else {
    result: [attack 1@2d20 p AB dmg D p DB] + [N-1 multiattack p AB dmg D p DB]
  }
}

output [ 2 multiattack p 5 dmg 1d8 p 15] named "Archery + XBE + SS"

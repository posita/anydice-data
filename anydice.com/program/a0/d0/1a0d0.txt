function: skillroll SKILL:n {
  RESULT: 0
  if SKILL = 0 {
    RESULT: 3d6
  }

  if SKILL > 0 {
    loop I over {1..3} {
      RESULT: RESULT + I@(SKILL+3)d6
    }
    loop I over {4..(SKILL+3)} {
      RESULT: RESULT + (I@(3+SKILL)d6 = 6)
    }
  }

  if SKILL < 0 {
    loop I over {1..(-SKILL)} {
      RESULT: RESULT - (I@(3-SKILL)d6 = 1)
    }
    loop I over {(1-SKILL)..(3-SKILL)} {
      RESULT: RESULT + I@(3-SKILL)d6
    }
  }
  
  result: RESULT
}

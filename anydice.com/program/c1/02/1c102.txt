function: attackroll ROLL:n {
  \ return the number of times damage is applied on an attack roll of ROLL \
  if ROLL <= 1 { result: 0 }     \ natural 1 is always a miss \
  if ROLL >= CRIT { result: CRITMULTI }  \ critical hit multiplies damage dice \
  result: ROLL + ATKBONUS >= AC
}

ATKRESULT: [attackroll ATK]  \ optimization: precalculate the miss/hit/crit probabilities \
BONUSATKRESULT: [attackroll ATK] \ allows for different attack mod for TWF attacks \

function: reroll NEWDIE:d for ORIGDIE:n { \ Great Weapon Fighting damage die rerolls on 1 or 2 \
 if ORIGDIE <= 2 { result: NEWDIE }
 result: ORIGDIE
}

function: baseattack ATKA:n times and bonusattack ATKB:n times {
  result: [roll ATKRESULT to baseattack ATKA times and bonusattack ATKB times]
}

function: bonusattack ATKC:n times {
  result: [roll BONUSATKRESULT to bonusattack ATKC times]
}

function: roll BASEDMGMULTI:n to baseattack ATKD:n times and bonusattack ATKE:n times {
	ATKCOUNT: ATKD
	BONUSATKS: ATKE
	GWMATK: 0
	TOTALDMG: 0
	if ATKCOUNT > 0 {
		if BASEDMGMULTI > 0 {
			TOTALDMG: TOTALDMG + (BASEDMGMULTI d DMGOPTDICE + DMGOPTMODS)
			if BASEDMGMULTI > 1 {
				if GWMASTER > 0 {
					TOTALDMG: TOTALDMG + (BASEDMGMULTI d (DMG + DMGBONUSDICE) + (BRUTALMULTI d BRUTALDIE) + DMGMODS)
					GWMATK: 1
					GWMASTER: 0
				}
				TOTALDMG: TOTALDMG + ATKCOUNT d (BRUTALMULTI d BRUTALDIE)
			}
			TOTALDMG: TOTALDMG + ATKCOUNT d (BASEDMGMULTI d (DMG + DMGBONUSDICE) + DMGMODS)
		} 
	}
	if BONUSATKS > 0 {
		if GWMATK != 1 {
			BONUSTOTALDMG: [bonusattack BONUSATKS times]
			TOTALDMG: TOTALDMG + BONUSTOTALDMG
		}
	}
	result: TOTALDMG
}

function: roll BONUSDMGMULTI:n to bonusattack AF:n times {
	BONUSATKCOUNT: AF
	TWFTOTALDMG: 0
	if BONUSATKCOUNT > 0 {
		if BONUSDMGMULTI > 0 {
			if BONUSDMGMULTI > 1 {
				TWFTOTALDMG: TWFTOTALDMG + BONUSATKCOUNT d (BRUTALMULTI d TWFBRUTALDIE)
			}
			TWFTOTALDMG: TWFTOTALDMG + BONUSATKCOUNT d (BONUSDMGMULTI d (TWFDMG + TWFDMGBONUSDICE) + TWFDMGMODS)
		}
	}
	result: TWFTOTALDMG
}

MODIFIER: 0
ADVANTAGE: 0

if ADVANTAGE > 0 {
  DICE: 1@(ADVANTAGE + 1)d20
}
if ADVANTAGE = 0 {
  DICE: d20
}
if ADVANTAGE < 0 {
  DICE: (-1 * ADVANTAGE + 1)@(-1 * ADVANTAGE + 1)d20
}

function: roll D:n add M:n dc T:n {
  if D = 1 {result: 0}
  if D = 20 {result: 1}
  result: (D + M) >= T
}

output 1 named "DC:"
loop DC over {1..30} {
  output [roll DICE add MODIFIER dc DC] named "[DC]"
}

function: roll ROLL:s {
  COUNTONE: [count 1 in ROLL]
  COUNTFIVESIX: [count {5,6} in ROLL]
  
  if COUNTONE = COUNTFIVESIX {
  if COUNTONE = 0 {
    result: 0
  }
    result: -1
  } 
  if COUNTONE > COUNTFIVESIX {
   result: -2
  }
  result: COUNTFIVESIX
}

loop N over {1..10} {
  output [roll Nd6] named "Original"
}

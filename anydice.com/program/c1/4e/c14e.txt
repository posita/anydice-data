function: gwf DIE:n DS:n {
if DIE < 3 {result: d DS}
result: DIE
}

function: attack ROLL:n vs AC:n {
if ROLL >= THREAT {result: 2}
if ROLL = 1 {result:0}
if ROLL >= AC {result: 1}
result: 0
}

function: damage for ATTACK:n {
 if ATTACK = 2 {result: CRITDAMAGE}
 if ATTACK = 1 {result: DAMAGE}
 result: 0
}

function: miss half ROLL:n vs DC:n {
 if ROLL >= DC {result:2}
 result: 1
}

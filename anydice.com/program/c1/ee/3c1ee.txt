function: evaluate ROLLA:s ROLLB:s MAXA MAXB{
 result: (ROLLA + ROLLB)/5 + (ROLLA = MAXA) + (ROLLB = MAXB) - (ROLLA = 1) - (ROLLB = 1)
}

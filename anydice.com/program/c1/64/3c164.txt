function: crop SEQUENCE:n MIN:n MAX:n {                  \ limit the display range between [min;max] \
 if (SEQUENCE < MIN){result: MIN}
 if (SEQUENCE > MAX){result: MAX}
 result: SEQUENCE
}
function: halve SEQUENCE:n {                  \ divide per 2, rounded up \
 DIV: SEQUENCE/2
 TEMP: DIV*2
 if (DIV=TEMP){result: DIV}
 result: (SEQUENCE+1)/2
}

function: phaseone ATT:s DEF:s {
  ATT_ROLL: 1@ATT dQ                       \freeze the hits-dices sequence\
  DEF_ROLL: 3@DEF dQ
  result: [phasetwo ATT_ROLL DEF_ROLL ATT DEF]
}

function: phasetwo ATT_ROLL:n DEF_ROLL:n  ATT:s DEF:s {
  VELOCITY_MODS: [compare 5@ATT with 6@DEF]

  HIT_SETS: [sets in ATT_ROLL]
  SAVE_SETS: [sets in DEF_ROLL]

  DAMAGES: ATT_ROLL - DEF_ROLL
  result: DAMAGES
}

function: sets in DICES:s {
  ZEROS:  [count 0 in DICES]
  ONES:   [count 1 in DICES]
  TWOS:   [count 2 in DICES]
  THREES: [count 3 in DICES]
  SETS: {ZEROS,ONES,TWOS,THREES}
  result: SETS
}
function: compare VAL_ATT:n with VAL_DEF:n {
  if(VAL_ATT > VAL_DEF) { result: -1}
  if(VAL_ATT = VAL_DEF) { result: 0}
  if(VAL_ATT < VAL_DEF) { result: 1}
  result: 0
}

function: hit ATTACKS:n {
  DICE: ATTACKS dQ                       \freeze the hits-dices sequence\
  result: [hit roll DICE]
}

function: hit roll DICES:s {
  HITS: [count {WS..3} in DICES]
  ZEROS:  [count 0 in DICES]
  ONES:   [count 1 in DICES]
  TWOS:   [count 2 in DICES]
  THREES: [count 3 in DICES]
  SETS: {ZEROS,ONES,TWOS,THREES}
  result: [save DICES with SETS]
}

function: save HITS:n with SETS:s {
  DICES: RD dQ                              \freeze the saves-dices sequence\
  result: [damages DICES with SETS]
}

function: hitme NN:n {
  A: [count {6} in NN d6]
  B: [count {1} in NN d6]
  result: A*2 + B*(-1)
}

output [hitme 5]

function: challenge MAIN:n SEC {
  ROLL: MAIN
  loop N over {1..SEC}{
   if d6 > MAIN {ROLL: ROLL + 1}
   if d6 = 1 {ROLL: ROLL -1}
  }
  result: ROLL
}

output [challenge 1d6 1]
\
\
function: challenge MM:n NN:n {
  MAIN: MM d6
  SEC: NN d6
  A: [count {SEC > MAIN} in SEC]
  B: [count {1} in SEC]
  result: A - B
}

output [challenge 1 2]
\

function: challenge MAIN:n SEC:n {
  MOD: 0
  DMAIN: [highest 1 of MAIN d6]
  loop N over {1..SEC}{
   DSEC: d6
   if DSEC > DMAIN {MOD: MOD+1}
   if 1 < DSEC <= DMAIN {MOD: MOD+0}
   if DSEC = 1 {MOD: MOD-1}
  }
  result: DMAIN + MOD
}

output [challenge 1 2]

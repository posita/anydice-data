MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n blessed BLESS:n baned BANED:n {
 if ROLL = 1 { result: MISS }
 MOD: ATTACK + BLESS - BANED
 if ROLL = 20 {
  if ROLL + MOD >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + MOD >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

ATTACK: 4+2
HITDAMAGE: 1d8 + 4
MISSDAMAGE: 0
CRITICALDAMAGE: HITDAMAGE + 1d8

ROLL: [highest 1 of 1d20]
output [damage for [attack ROLL vs 15 blessed 0 baned 0]] named "Straight AC 15"
output [damage for [attack ROLL vs 15 blessed 1d4 baned 0]] named "Blessed AC 15"
output [damage for [attack ROLL vs 15 blessed 0 baned 1d4]] named "Baned AC 15"

ROLL: [highest 1 of 2d20]
output [damage for [attack ROLL vs 15 blessed 0 baned 0]] named "Straight AC 15 ADV"
output [damage for [attack ROLL vs 15 blessed 1d4 baned 0]] named "Blessed AC 15 ADV"
output [damage for [attack ROLL vs 15 blessed 0 baned 1d4]] named "Baned AC 15 ADV"

ROLL: [lowest 1 of 2d20]
output [damage for [attack ROLL vs 15 blessed 0 baned 0]] named "Straight AC 15 DIS"
output [damage for [attack ROLL vs 15 blessed 1d4 baned 0]] named "Blessed AC 15 DIS"
output [damage for [attack ROLL vs 15 blessed 0 baned 1d4]] named "Baned AC 15 DIS"

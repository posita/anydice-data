DRAIN:10

function: roll ROLL:s {
  COUNTONE: [count 1 in ROLL]
  COUNTFIVESIX: [count {5,6} in ROLL]
  if COUNTONE = 0 {
    result: [highest of 0 and DRAIN-COUNTFIVESIX ]
  }
  if COUNTFIVESIX = 0 {
  if COUNTONE > 0 {
    result: DRAIN*2
  }
    result: DRAIN
  }
  if COUNTONE = COUNTFIVESIX {
    result: [highest of 0 and (DRAIN-COUNTFIVESIX)]
  }
if COUNTONE > COUNTFIVESIX {
    result: [highest of 0 and (DRAIN-COUNTFIVESIX)*2]
  }


  result: [highest of 0 and DRAIN-COUNTFIVESIX]
}

loop N over {1..10} {
  output [roll Nd6] named "[N] Dice"
}

function: move SLOT:n with DICE:n {
  if SLOT = 1 {
    result: [count {3, 4, 5, 6} in DICEd6]
  }
  if SLOT = 2 {
    result: [count {4, 5, 6} in DICEd6]
  }
  if SLOT = 3{
    result: [count {5, 6} in DICEd6]
  }
}

function: jam A_STRATEGY:n with A_SUCCESSES:n vs B_STRATEGY:n with B_SUCCESSES:n {  
  if A_STRATEGY = 1 & B_STRATEGY = 1 {
    A_RESULT: 2d6 + A_SUCCESSESd6
    B_RESULT: 2d6 + B_SUCCESSESd6
  }
  if A_STRATEGY = 1 & B_STRATEGY = 2 {
    A_RESULT: 2d6 + A_SUCCESSESd6 - B_SUCCESSESd6
    B_RESULT: 2d6
  }
  if A_STRATEGY = 2 & B_STRATEGY = 1 {
    A_RESULT: 2d6
    B_RESULT: 2d6 + B_SUCCESSESd6 - A_SUCCESSESd6
  }
  else {
    A_RESULT: 2d6 - B_SUCCESSESd6
    B_RESULT: 2d6 - A_SUCCESSESd6
  }

  result: [winner A_RESULT B_RESULT]
}

function: winner A_SUCCESSES:n B_SUCCESSES:n {
  if A_SUCCESSES > B_SUCCESSES {
    result: 1
  } 
  if A_SUCCESSES < B_SUCCESSES {
    result: 2
  }
  if A_SUCCESSES = B_SUCCESSES {
    result: 3
  }
}

loop I over {1..3} {
  loop J over {1..3} {
    loop K over {1..2} { 
      loop L over {1..2} {
        A_SUCCESSES: [move I with 2]
        B_SUCCESSES: [move J with 2]
        
        WINNER: [ jam K with A_SUCCESSES vs L with B_SUCCESSES ]
        if K = 1 & L = 1 {
          output WINNER named "[I]O v [J]O"
        }
        if K = 1 & L = 2 {
          output WINNER named "[I]O v [J]D"
        }
        if K = 2 & L = 1 {
          output WINNER named "[I]D v [J]O"
        }
        if K = 2 & L = 2 {
          output WINNER named "[I]D v [J]D"
        }
      }
    }
  }
}

function: con {
result: 6 - 1d11
}
function: health A to B {
result: [con] + 10 + (B - 1d(B - A) - 1)d10
}

loop N over {1 .. 3} {
output [health N to N] / 2 - 2d6 named "Lvl [N]"
}
loop N over {4 .. 5} {
output [health N to N] / 2 - 4d6 named "Lvl [N]"
}
loop N over {6 .. 8} {
output [health N to N] / 2 - 6d6 named "Lvl [N]"
}
loop N over {9 .. 10} {
output [health N to N] / 2 - 8d6 named "Lvl [N]"
}

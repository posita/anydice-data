function: run PLAY:n with DICE:n stat {
  if PLAY = 1 {
    result: [count {3, 4, 5, 6} in DICEd6]
  }
  if PLAY = 2 {
    result: [count {4, 5, 6} in DICEd6]
  }
  if PLAY = 3{
    result: [count {5, 6} in DICEd6]
  }
}

function: jam using A_STRATEGY:n with A_SUCCESSES:n vs B_STRATEGY:n with B_SUCCESSES:n {  
  if A_STRATEGY = 1 & B_STRATEGY = 1 {
    A_RESULT: 2d6 + A_SUCCESSESd6
    B_RESULT: 2d6 + B_SUCCESSESd6
  }
  if A_STRATEGY = 1 & B_STRATEGY = 2 {
    A_RESULT: 2d6 + A_SUCCESSESd6 - B_SUCCESSESd6
    B_RESULT: 2d6
  }
  if A_STRATEGY = 2 & B_STRATEGY = 1 {
    A_RESULT: 2d6
    B_RESULT: 2d6 + B_SUCCESSESd6 - A_SUCCESSESd6
  }
  else {
    A_RESULT: 2d6 - B_SUCCESSESd6
    B_RESULT: 2d6 - A_SUCCESSESd6
  }

  result: [winner A_RESULT B_RESULT]
}

function: winner A_SUCCESSES:n B_SUCCESSES:n {
  if A_SUCCESSES > B_SUCCESSES {
    result: 1
  } 
  if A_SUCCESSES < B_SUCCESSES {
    result: 2
  }
  if A_SUCCESSES = B_SUCCESSES {
    result: 3
  }
}

loop PLAY_A over {1..3} {
  loop PLAY_B over {1..3} {
    loop STRATEGY_A over {1..2} { 
      loop STRATEGY_B over {1..2} {
        A_SUCCESSES: [run PLAY_A with 2 stat]
        B_SUCCESSES: [run PLAY_B with 2 stat]
        
        WINNER: [ jam using STRATEGY_A with A_SUCCESSES vs STRATEGY_B with B_SUCCESSES ]
        if STRATEGY_A = 1 & STRATEGY_B = 1 {
          output WINNER named "[PLAY_A]O v [PLAY_B]O"
        }
        if STRATEGY_A = 1 & STRATEGY_B = 2 {
          output WINNER named "[PLAY_A]O v [PLAY_B]D"
        }
        if STRATEGY_A = 2 & STRATEGY_B = 1 {
          output WINNER named "[PLAY_A]D v [PLAY_B]O"
        }
        if STRATEGY_A = 2 & STRATEGY_B = 2 {
          output WINNER named "[PLAY_A]D v [PLAY_B]D"
        }
      }
    }
  }
}

function: shadowrun ROLL:s {
  COUNTONES: [count 1 in ROLL]
  COUNTSUCCESSES: [count {5,6} in ROLL]
  POOLSIZE: #ROLL
  GLITCHTHRESHOLD: (POOLSIZE + 1) / 2 

  if COUNTONES >= GLITCHTHRESHOLD {
    if COUNTSUCCESSES = 0 {
      result: -1  
    }
    result: -2  
  }

  result: COUNTSUCCESSES
}

loop N over {1..10} {
  output [shadowrun Nd6] named "SR Glitch Test"
}

function: roll A {
  result: [count {6..12} in A] + [count {8..12} in A] + [count {10..12} in A] + [count 12 in A] 
}


loop N over {6,8,10,12} {
  output [roll 1dN] named "Basic Tier D[N] (1D[N])"
  loop M over {6,8,10,12} {
    output [roll 1dN] + [roll 1dM] named "Basic D[N] + Skill D[M] (1D[N]+D[M])"
  }
}

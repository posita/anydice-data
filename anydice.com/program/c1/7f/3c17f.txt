function: silhouette ROLL:s {
  H: {1@ROLL}          \ Highest roll from the pool
  S: (ROLL = 6) - 1    \ Count extra sixes beyond the first
  if S < 0 { S: 0 }    \ Ensure no negative values
  result: H + S        \ Final result with exploding sixes
}

loop N over {1..5} {   \ Outputs for 1d6 to 5d6
  output [silhouette Nd6] named "Silhouette Nd6"
}

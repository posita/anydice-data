function: move STRATEGY:n with DICE:n {
  if STRATEGY = 1 {
    result: [count {3, 4, 5, 6} in DICEd6]
  }
  if STRATEGY = 2 {
    result: [count {4, 5, 6} in DICEd6]
  }
  if STRATEGY = 3{
    result: [count {5, 6} in DICEd6]
  }
}

function: jam using MY_STRATEGY:n in MY_SLOT:n with MY_DICE:n vs OPP_STRATEGY in OPP_SLOT:n with OPP_DICE:n {
  ME: [move MY_SLOT with MY_DICE]
  THEM: [move OPP_SLOT with OPP_DICE]
  
  if MY_STRATEGY = 1 & OPP_STRATEGY = 1 {
    MY_RESULT: 2d6 + MEd6
    THEIR_RESULT: 2d6 + THEMd6
  }
  if MY_STRATEGY = 1 & OPP_STRATEGY = 2 {
    MY_RESULT: 2d6 + MEd6 - THEMd6
    THEIR_RESULT: 2d6
  }
  if MY_STRATEGY = 2 & OPP_STRATEGY = 1 {
    MY_RESULT: 2d6
    THEIR_RESULT: 2d6 + THEMd6 - MEd6
  }
  else {
    MY_RESULT: 2d6 - THEMd6
    THEIR_RESULT: 2d6 - MEd6
  }

  result: [winner MY_RESULT THEIR_RESULT]
}

function: winner ME:n THEM:n {
  if ME > THEM {
    result: 1
  } 
  if ME < THEM {
    result: 2
  }
  if ME = THEM {
    result: 3
  }
}

function: nonnegative ROLL:n {
  if ROLL < 0 {
    result : 0
  }
  else { 
    result: ROLL
  }
}


loop I over {1..3} {
  loop J over {1..3} {
    loop K over {1..2} { 
      loop L over {1..2} {
        WINNER: [jam using K in I with 2 vs L in J with 2 ]
        if K = 1 & L = 1 {
          output WINNER named "#[I]O/#[J]O"
        }
        if K = 1 & L = 2 {
          output WINNER named "#[I]O/#[J]D"
        }
        if K = 2 & L = 1 {
          output WINNER named "#[I]D/#[J]O"
        }
        if K = 2 & L = 2 {
          output WINNER named "#[I]D/#[J]D"
        }
      }
    }
  }
}

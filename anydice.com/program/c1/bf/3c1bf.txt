DRAIN:10

function: shadowrun ROLL:s {
  COUNTONES: [count 1 in ROLL]
  COUNTSUCCESSES: [count {5,6} in ROLL]
  POOLSIZE: #ROLL
  GLITCHTHRESHOLD: (POOLSIZE+1) / 2 

  if COUNTONES >= GLITCHTHRESHOLD {
    if COUNTSUCCESSES = 0 {
      result: DRAIN*2
    }
    result: (DRAIN-COUNTSUCCESSES)
  }

  result: DRAIN-COUNTSUCCESSES
}

loop N over {1..10} {
  output [shadowrun Nd6] named "[N] SR Glitch Test"
}

STR: 2
FIN: 2
MELEE:1
DEFENSEBONUS: 4
ARMOR: 0
ATTACK: STR + MELEE
DEFENSE: 4 + DEFENSEBONUS

function: floor NUM:n {
     if NUM < 0 { result: 0 }
     result: NUM
}

function: bound NUM:n {
     if NUM > 0 { result: 0 }
     result: NUM
}

function: damage of DAMAGE:n PIERCING:n {
     result: [floor DAMAGE + [bound PIERCING - ARMOR]]
}

function: battack ROLL:n damage DAM:n pierce PIERCE:n {
 if ROLL + ATTACK >= DEFENSE { result: [damage of DAM+STR PIERCE] }
 result: 0
}

function: sattack ROLL:n damage DAM:n pierce PIERCE:n {
 if ROLL + ATTACK >= DEFENSE { result: [damage of DAM+FIN PIERCE] }
 result: 0
}

function: aattack ROLL:n damage DAM:n pierce PIERCE:n {
 if ROLL + ATTACK >= DEFENSE { result: [damage of DAM+(2*FIN) PIERCE] }
 result: 0
}

function: fattack ROLL:n damage DAM:n {
 if ROLL + ATTACK - 2 >= DEFENSE { result: DAM+(2*STR) }
 result: 0
}


output [battack 2d4+2 damage 1 pierce 4] named "Fist"

output [sattack 2d4 damage 1d4+3 pierce 0] named "Small Blade"
output [fattack 2d4+2 damage 1] named "Fist with Iron Fist"
output [battack 2d4+1 damage 2d4 pierce 6] named "Medium Blunt"
output [sattack 2d4 damage 2d4+4 pierce 0] named "Medium Blade"
output [battack 2d4+1 damage 3d4+2 pierce 10] named "Large Blunt"
output [sattack 2d4 damage 3d4+8 pierce 3] named "Large Blade"

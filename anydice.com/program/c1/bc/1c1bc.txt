function: run PLAY:n with DICE:n stat {
  if PLAY = 1 {
    result: [count {3, 4, 5, 6} in DICEd6]
  }
  if PLAY = 2 {
    result: [count {4, 5, 6} in DICEd6]
  }
  if PLAY = 3{
    result: [count {5, 6} in DICEd6]
  }
}

function: winner A_SUCCESSES:n vs B_SUCCESSES:n {
  if A_SUCCESSES > B_SUCCESSES {
    result: 1
  } 
  if A_SUCCESSES < B_SUCCESSES {
    result: 2
  }
  if A_SUCCESSES = B_SUCCESSES {
    result: 3
  }
}

loop PLAY_A over {1..3} {
  loop PLAY_B over {1..3} {
    A_SUCCESSES: [run PLAY_A with 2 stat]
    B_SUCCESSES: [run PLAY_B with 2 stat]
        
    WINNER: [ winner A_SUCCESSES vs B_SUCCESSES ]
    A: PLAY_A
    B: PLAY_B
    output WINNER named "A:[A] v B:[B]"
  }
}

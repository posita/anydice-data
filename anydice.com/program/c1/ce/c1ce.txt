
function: ROLL:s hits vs DC:n{
 BADONES: (ROLL = 1)
 SUCCESSES: (ROLL >= DC)
 CRITICAL: (ROLL=12)
 MODS: SUCCESSES-BADONES+CRITICAL
 if (MODS>8) {result: 8}
 if (MODS<-8) {result: -8}
 result: MODS
}

DICE:5
DC:9
ATTR:3
RISK:1

loop DICE over {0..5}{
output ATTR+[(DICE)d12 hits vs (DC)] named "[DICE]d12 successes vs [DC]" 
loop RISK over {1..2}{
output ATTR-RISK+[(DICE+1+RISK)d12 hits vs (DC)] named "[DICE]d12 successes vs [DC] risk"
}
}

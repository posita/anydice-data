function: spec N:n tn TN:n {
 if N = 1 { result: -1 }
 if N < TN { result: 0 }
 if N < 10 { result: 1 }
 if N = 10 { result: 2 }
 result: [Spec d10 tn TN]
}

SPEC: [spec d10 tn 2]

loop N over {1..20} {
 output [highest of [lowest of NdSPEC and 20] and -1] named "[N]d diff 2"
}

function: cap N:n {
 if N < 1 { result: 0}
 else {result: N}
}


loop REALM over {0..9} {
  POOL: 10*(REALM) + (8*REALM)d3-(8*REALM) + 2^[cap (REALM-3)]
  output POOL named "d4 R[REALM]"
}

loop REALM over {0..9} {
  POOL: 10*(REALM)  + (4*REALM)d4 + 2^[cap (REALM-3)]
  output POOL named "d4 R[REALM]"
}

loop REALM over {0..9} {
  POOL: 10*(REALM) + (4*REALM)d8 + 2^[cap (REALM-3)]
  output POOL named "d8 R[REALM]"
}

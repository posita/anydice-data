function: pairexplode A:n and B:n at DEPTH:n depth {
  TOTAL: A + B
  if (A  = B)&(DEPTH>0) {
    TOTAL: TOTAL + [pairexplode d6 and d6 at DEPTH-1 depth]
  }
  result: TOTAL
}

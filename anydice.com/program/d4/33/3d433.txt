function: round R:n {
  if R=1 {
    X: d{1, 3, 5}
    Y: d{[round 2]:5, [round 1]:1}
    result: X + Y
  }
  if R=2 {
    X: d{5, 7, 9}
    Y: d{[round 3]:5, [round 2]:1}
    result: X + Y
  }
  if R=3 {
    X: d{9, 11, 13}
    Y: d{[round 2]:1, [round 4]:4, [round 3]:1}
    result: X + Y
  }
  if R=4 {
    X: d{13, 15, 17}
    Y: d{[round 3]:1, [round 5]:4, [round 4]:1}
    result: X + Y
  }
  if R=5 {
    X: d{17, 19, 21}
    Y: d{[round 4]:1, [round 5]:1, [end d6]:3, [round 6]:1}
    result: X + Y
  }
  if R=6 {
    X: d{21, 23, 25}
    Y: d{[round 2]:1, [end d6]:5}
    result: X + Y
  }
}

function: end X:n {
  if X=1 { result: 0 }
  if X=2 { result: [bonus] }
  if X=3 { result: [round 1] }
  if X=4 { result: 0 }
  if X=5 { result: [round 2] }
  if X=6 { result: [bonus]+[round 1] }
}

  X: d{9, 13, 17, 21, 25, 30}
  result: X
}

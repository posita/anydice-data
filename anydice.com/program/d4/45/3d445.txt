A: [highest 1 of 2d6]
B: [highest 1 of 2d6]
C: [highest 1 of 2d6]
D: [highest 1 of 2d6]
E: [highest 1 of 2d6]

SNKATK_LV_IV: A+B
SNKATK_LV_V: A+B+C
SNKATK_LV_VII: A+B+C+D
SNKATK_LV_XI: A+B+C+D+E+F

MODIFIER: 5
PROFICIENCY: 4
ENCHANTMENT: 3

ENEMY_AC: 16

function: attack roll X:d {
  result: X+MODIFIER+PROFICIENCY+ENCHANTMENT-ENEMY_AC
}

function: check hit X:s {
  if X<0 {
    result: 0
  }else {
    result: 1
  }
}

function: if crit X:s {
  if X=20 {
    result: 1
  }else {
    result: 0
  }
}

WEAPON_ATTACK_A: [highest 1 of 2d8] + MODIFIER
WEAPON_ATTACK_B: [highest 1 of 2d8] + MODIFIER
BOOMING_BLADE_A: [highest 1 of 2d8] + [highest 1 of 2d8] + MODIFIER

CRIT_BONUS_WEAPON_A: [highest 1 of 2d8]
CRIT_BONUS_WEAPON_B: [highest 1 of 2d8]
CRIT_BONUS_BOOBLD_A: [highest 1 of 2d8] + [highest 1 of 2d8]

CRIT_BONUS_SNKATK_LV_V: [highest 1 of 2d6] + [highest 1 of 2d6] + [highest 1 of 2d6]
CRIT_BONUS_SNKATK_LV_VII: [highest 1 of 2d6] + [highest 1 of 2d6] + [highest 1 of 2d6] + [highest 1 of 2d6]
CRIT_BONUS_SNKATK_LV_XI: [highest 1 of 2d6] + [highest 1 of 2d6] + [highest 1 of 2d6] + [highest 1 of 2d6] + [highest 1 of 2d6] + [highest 1 of 2d6]

ATTACK_ROLL_A: [check hit [attack roll d20]]
ATTACK_ROLL_B: [check hit [attack roll d20]]

IF_CRIT_A: [if crit d20]
IF_CRIT_B: [if crit d20]

ATTACK_ROLL_SNKATK: ATTACK_ROLL_A | ATTACK_ROLL_B
IF_CRIT_SNKATK: IF_CRIT_A | IF_CRIT_B

output ATTACK_ROLL_A named "attack roll"
output ATTACK_ROLL_SNKATK named "attack roll sneak attack"

output WEAPON_ATTACK_A named "weapon attack damage"
output WEAPON_ATTACK_A + IF_CRIT_A * CRIT_BONUS_WEAPON_A named "weapon attack damage w/ crit"

output BOOMING_BLADE_A named "booming blade damage"
output BOOMING_BLADE_A + IF_CRIT_A * CRIT_BONUS_BOOBLD_A named "booming blade damage w/ crit"

output SNKATK_LV_VII named "sneak attack 4d6 damage"
output SNKATK_LV_VII + IF_CRIT_A * CRIT_BONUS_SNKATK_LV_VII named "sneak attack 4d6 damage w/ crit"

output (SNKATK_LV_VII + WEAPON_ATTACK_A) * ATTACK_ROLL_A + IF_CRIT_A * (CRIT_BONUS_WEAPON_A + CRIT_BONUS_SNKATK_LV_VII) named "plv7 savage attacker 4d6 damage per turn"

output (SNKATK_LV_V + BOOMING_BLADE_A) * ATTACK_ROLL_A + IF_CRIT_A * (CRIT_BONUS_BOOBLD_A + CRIT_BONUS_SNKATK_LV_V) named "plv7 savage attacker 3d6 w/BooBlade damage per turn"

output (SNKATK_LV_XI + WEAPON_ATTACK_A) * ATTACK_ROLL_A + IF_CRIT_A * (CRIT_BONUS_WEAPON_A + CRIT_BONUS_SNKATK_LV_XI) named "plv12 savage attacker 6d6 damage per turn"

output SNKATK_LV_VII * ATTACK_ROLL_SNKATK + WEAPON_ATTACK_A * ATTACK_ROLL_A + WEAPON_ATTACK_B * ATTACK_ROLL_B + IF_CRIT_A * CRIT_BONUS_WEAPON_A + IF_CRIT_B * CRIT_BONUS_WEAPON_B + IF_CRIT_SNKATK * CRIT_BONUS_SNKATK_LV_VII named "plv12 savage attacker 4d6 extra attack damage per turn"

function: TOHIT:s hits on HITVALUE:n {
  result: [count {HITVALUE..6} in TOHIT]
}

function: TOSAVE:s saves on SAVEVALUE:n {
  loop N over {TOSAVE} {
    result: [count {1..SAVEVALUE-1} in Nd6]
  }
}

function: TOKILL:s kills on FIREPOWER:n {
  loop N over {TOKILL} {
    result: [count {FIREPOWER..6} in Nd6]
  }
}


output [[[8d6 hits on 5] saves on 4] kills on 3]

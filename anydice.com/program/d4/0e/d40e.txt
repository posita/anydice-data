loop N over {-20..20}{
if N<0 {
output 1d20+N <= 1d20+2 named "Reactionary v.s. initiative modifier [N]"
output 1d20+N <= 1d20 named "Normal v.s. initiative modifier [N]"
}
else {
if N<2 {
output 1d20+N <= 1d20+2 named "Reactionary v.s. initiative modifier [N]"
output 1d20+N < 1d20 named "Normal v.s. initiative modifier [N]"
}
else {
output 1d20+N < 1d20+2 named "Reactionary v.s. initiative modifier [N]"
output 1d20+N < 1d20 named "Normal v.s. initiative modifier [N]"
}}
}

function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: DAMAGE + DAMAGE - STRENGTH
 }
 result: DAMAGE
}

AC: d{14..20}
STRENGTH: 3
ATTACK: 2 + STRENGTH

DAMAGE: 1d4 + STRENGTH
THREAT: 20
output [d20 vs AC] - d4 named "dagger"

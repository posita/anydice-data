function: roll P:n and N:n {
  if P = N {
    result: 0
  }
  else if P < N {
    result: P
  }
  else {
    result: -N
  }
}

  S: {A, B, C, D}
  T: [sort S]
  
  U: 3@T
  V: 4@T

  if X = 1 {
    U: -U
  }

  if X = Y & X = 1 {
    V: -V
  }
  else if X = 2 {
    if Y > 1 {
      V: -V
    }
  }

}

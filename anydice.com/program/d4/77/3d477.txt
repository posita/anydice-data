function: frontir ROLL:s reroll N:n ones {
 SIXES: 6 = ROLL
 ONES: 1 = ROLL

 if (ONES <= N) {
  SECN: #ROLL - SIXES - ONES
  result: [frontir reroll SECNd6]
 }
 else { result: 0 }
}

function: frontir reroll REROLL:s {
 RESIXES: 6 = REROLL
 REONES: 1 = REROLL

 SIXSUM: SIXES + RESIXES
 ONESUM: ONES + REONES

 if (ONESUM - SIXSUM) > 0 { result: ONESUM }
 else { result: 0 }
}

loop N over {2..6}{
 output [frontir Nd6 reroll 1 ones] named "[N]d6"
}

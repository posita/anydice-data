function: round R:n {
  if R=1 {
    X: d{1, 3, 5}
    Y: d{2:5, 1:1}
    result: X + [round Y]
  }
  if R=2 {
    X: d{5, 7, 9}
    Y: d{3:5, 2:1}
    result: X + [round Y]
  }
  if R=3 {
    X: d{9, 11, 13}
    Y: d{2:1, 4:4, 3:1}
    result: X + [round Y]
  }
  if R=4 {
    X: d{13, 15, 17}
    Y: d{3:1, 5:4, 4:1}
    result: X + [round Y]
  }
  if R=5 {
    X: d{17, 19, 21}
    Y: d{4:1, 5:1, 7:3, 6:1}
    result: X + [round Y]
  }
  if R=6 {
    X: d{21, 23, 25}
    Y: d{2:1, 7:5}
    result: X + [round Y]
  }
  if R=7 {
    result: [end d6]
  }
}

function: end X:n {
  if X=1 { result: 0 }
  if X=2 { result: [bonus] }
  if X=3 { result: [round 1] }
  if X=4 { result: 0 }
  if X=5 { result: [round 2] }
  if X=6 { result: [bonus]+[round 1] }
}

function: bonus {
  X: d{9, 13, 17, 21, 25, 30}
  result: X
}

set "maximum function depth" to 12
output [round 1]

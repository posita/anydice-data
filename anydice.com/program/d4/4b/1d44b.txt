function: ROLL:n attack AC:n exp EXP:n
{
  if (ROLL = 20)
  {
    result: [d20 threat AC exp EXP]
  }
  if ((ROLL + ATTACK_MODIFIER) >= AC)
  {
    if (ROLL >= RANGE)
    {
      result: [d20 threat AC exp EXP]
    }
    result: DAMAGE + AFTER
  }
  result: MISS
}

function: ROLL:n threat AC:n exp EXP:n
{
  if (ROLL = 20)
  {
    if (EXP)
    {
      result: [d20 threat AC exp (EXP + 1)]
    }
    result: (MULTIPLIER d DAMAGE) + AFTER
  }
  if ((ROLL + ATTACK_MODIFIER) >= AC)
  {
    if (ROLL >= RANGE)
    {
      if (EXP)
      {
        result: [d20 threat AC exp (EXP + 1)] 
      }
      result: (MULTIPLIER d DAMAGE) + AFTER
    }
    result: (1 + (EXP * (MULTIPLIER - 1))) d DAMAGE + AFTER
  }
  result: (1 + [highest of (EXP - 1) and 0] * (MULTIPLIER - 1)) d DAMAGE + AFTER
}

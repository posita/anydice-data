function: scaling N:n {
 if N > 1 { result: 2^N}
 if N < 1 { result: 0}
}
OFFSET: 3

loop REALM over {0..9} {
  POOL: 10*(REALM) + (8*REALM)d3-(8*REALM) + [scaling (REALM-OFFSET)]
  output POOL named "d3-1 R[REALM]"
}

loop REALM over {0..9} {
  POOL: 10*(REALM)  + (4*REALM)d4 + [scaling (REALM-OFFSET)]
  output POOL named "d4 R[REALM]"
}

loop REALM over {0..9} {
  POOL: 10*(REALM) + (4*REALM)d8 + [scaling (REALM-OFFSET)]
  output POOL named "d8 R[REALM]"
}

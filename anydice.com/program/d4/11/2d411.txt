function: attack A:n damage D:d {
  OUT: {}
  EMPTY: {}
  EMPTY: {EMPTY, D}
  loop I over {1..20} {
    if (I >= A) {
      OUT: {OUT, D}
    } else {
      loop J over {1..#EMPTY} {
        OUT: {OUT, 0}
      }
    }
  }
  result: OUT
}
loop I over {1..20} {
  output [attack I damage d12] named "Hits on [I]+"
}

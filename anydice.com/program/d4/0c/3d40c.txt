function: ones in POOL:s {
  RESULT: 0
  loop I over {1..#POOL} {
    if I@POOL = 1 {
      RESULT: RESULT + 1
    }
  }
  result: RESULT
}

function: hits in POOL:s {
  RESULT: 0
  loop I over {1..#POOL} {
    if I@POOL >= 4 {
      RESULT: RESULT + 1
    }
  }
  result: RESULT
}

function: net success POOL:s {
  H: [hits in POOL]
  O: [ones in POOL]
  result: H - O
}

function: failure rate POOL:s {
  O: [ones in POOL]
  if O > 0 {
    result: 1
  }
  result: 0
}

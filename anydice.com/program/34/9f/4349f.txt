function: damagee ROLL:n {
    if ROLL >=16 {
        result: (4d8+4)/2
    }

    if ROLL <16 {
        result: 4d8+4
    }

    result: 0
}

function: damagea ROLL:n {
    if ROLL > 19 {
        result: 2d10 + 4
    }

    if ROLL > 11 {
        result: 1d10 + 4
    }

    result: 0
}

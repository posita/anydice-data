{
FAIL: DICE = 0
PASS: DICE = 1
REROLLS: [lowest of PASS and LIMIT]
result: (PASS - REROLLS) + REROLLSdDIE
}


function: DICE:s reroll LIMIT:n 
{
FAIL: DICE = 0
PASS: DICE = 1
REROLLS: [lowest of FAIL and LIMIT]
result: PASS + REROLLSdDIE
}

{

function: hit V:n M:n {
if V+M >5 {
result: 1
}
result:0
}

function: hitcount D:n M:n {
H: 0
loop N over {1..D}
{
H: H+[hit 1d10 M]

}
result: H
}
function: routcheck V {
if V < 0 {result: 1}
result: 0
}
function: morale check V:n and M:n
{
if V+M > 18 {result: 0}
if V+M >16 {result: -1}
if V+M >14 {result: -2}
if V+M >12 {result: -3}
if V+M >10 {result: -4}
if V+M >8 {result: -5}
if V+M >6 {result: -6}
if V+M >4 {result: -7}
if V+M >2 {result: -8}
if V+M >0 {result: -9}
result: -20
}
function: morale check roll I:n and M:n
{
T: [morale check I and M]
if T<= -20 {result: -20}
result: I+T
}
function: morale rolls N:n and M:n
{
S: 0
loop C over {1..N}
{
M: [morale check roll 1d20 and M]
}
result: M
}

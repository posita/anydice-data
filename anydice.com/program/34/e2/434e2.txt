function: dos ROLL:n plus MOD:n {
  TIER2: (84 - 5 * MOD) / 8
  TIER3: (140 - 5 * MOD) / 8
  TIER4: (172 - 9 * MOD) / 8

  if ROLL >= TIER4 { result: 4 }
  if ROLL >= TIER3 { result: 3 }
  if ROLL >= TIER2 { result: 2 }
  result: 1
}

loop MOD over {0..8} {
  output [dos 1d20 plus MOD] named "1d20+[MOD]"
}

DIE: 6
POOL: 6
START: 0

function: roll WHITE:s BLACK:s {
  WHITE_SUCCESSES: [count {5..6} in WHITE]
  BLACK_SUCCESSES: [count {5} in BLACK] + 2*[count {6} in BLACK]
  BLACK_FAILURES: [count {1} in BLACK]

  if BLACK_FAILURES > WHITE_SUCCESSES {
    result: -1
  } else {
    result: WHITE_SUCCESSES + BLACK_SUCCESSES
  }
}

loop B over {START..POOL} {
  W: POOL - B
  output [roll WdDIE BdDIE] named "[POOL] dice: [B] Black + [W] White"
}

\
   Azimuth Dice Pool System
   0 = failure
   1 = mid success
   2 = full success
\

function: azimuth ROLL:s {
 if 1@ROLL >= 5 {
  result: 1 + (2@ROLL >= 5)
 }
 result: 0
}
output [azimuth 3@3d6] named "-1 ranks"
output [azimuth 2@2d6] named "no ranks"
loop R over {1..5} {
 output [azimuth Rd6] named "[R] ranks"
}

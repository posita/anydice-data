\**** Damage calculator full basic damage dealing list ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL = CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: 15

ATTACKMOD: 5
DMGBONUS: 10
PROFICIENCY: 6
ACCBONUS: -10
CRITROLL: 20
ATTACKTIMES: 10
DAMAGE: 1d12 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 3d12 + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Half-Orc Samurai Fighter with Greataxe"

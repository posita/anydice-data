function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}
\ advantage on 2 attacks, vs AC18 with thrown hammer \
output [attack [highest 1 of 2d20] plus 11 vs 18 for 2d8 + 2 + 5 crit 2d8  on 20] +
  [attack [highest 1 of 2d20] plus 11 vs 18 for 2d8 + 2 + 5 crit 2d8 on 20]

\ as above no advantage \
output [attack d20 plus 11 vs 18 for 2d8 + 2 + 5 crit 1d8  on 20] +
  [attack d20 plus 11 vs 18 for 2d8 + 2 + 5 crit 1d8 on 20]

\ no advantage vs giant \
output [attack d20 plus 11 vs 18 for 3d8 + 2 + 5 crit 3d8  on 20] +
  [attack d20 plus 11 vs 18 for 3d8 + 2 + 5 crit 3d8 on 20]

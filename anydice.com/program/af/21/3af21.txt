function: attack ROLL:n DMG:d CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG*2
    }
    if ROLL <= CRIT_RANGE {
        result: DMG
    }
    result: 0
}


output
[attack 1d20(1d6+1d8+1d6+8)18]
named"18"

[attack 1d20(1d8+1d10+1d6+8)19]

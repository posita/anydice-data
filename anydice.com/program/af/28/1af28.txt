function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

output 1d10+5 named "longsword 2h"
output 1d8+5 + 1d6+4 named "magic longsword/shortsword twf+dw"
output 1d6+4 + 1d6+4 named "shortsword/shortswor twf"

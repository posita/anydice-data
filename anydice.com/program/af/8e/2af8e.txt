DS: 6
PB: 17
EB: 5
BD: 8
R: 0

function: normal A bonus B {
 result: AdDS + B
}
function: empowered A bonus B {
 result: [normal A bonus B] + ([normal A bonus B] / 2)
}
function: maximized A bonus B {
 result: A * DS + B
}
function: empowered maximized A bonus B {
 result: [maximized A bonus B] + ([maximized A bonus B] / 2)
}
function: physical A {
 result: A - R
}
function: energy A {
 result: A - R
}
function: physical composit A {
 result: (A / 2) + (A / 2) - (R * 2)
}
function: energy composit A {
 result: (A / 2) + (A / 2) - (R * 2)
}
function: composit bonus A {
 result: A + (A / 2)
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}
\ advantage on 2 attacks, vs AC18 with lance of the charger both charging with advantage \
output [attack [highest 1 of 2d20] plus 11 vs 18 for 2d12 + 2 + 5 crit 2d12  on 20] +
  [attack [highest 1 of 2d20] plus 11 vs 18 for 2d12 + 2 + 5 crit 2d12 on 20]

\ 2 attacks, vs AC18 with lance of the charger one charging \
output [attack 1d20 plus 11 vs 18 for 2d12 + 2 + 5 crit 2d12  on 20] +
  [attack 1d20 plus 11 vs 18 for 1d12 + 2 + 5 crit 1d12 on 20]

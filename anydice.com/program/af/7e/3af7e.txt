\ converts number of successes "SUC" and number of advantages "ADV" into an integer \
function: scav SUC ADV {
 result: (2^SUC)*(3^ADV)
}

\ example d8 with 1 blank side, 2 success sides, 1 double success side, 2 advantage sides, 1 double advantage side, and one success and advantage side \
GEN: d{[scav 0 0], [scav 1 0], [scav 1 0], [scav 2 0], [scav 0 1], [scav 0 1], [scav 0 2], [scav 1 1]}

\ how many times does DIVISOR go into DIVIDEND (up to CAP times)? \
function: factor DIVIDEND DIVISOR CAP {
 COUNTER: 0
 loop N over {1..CAP}{
  DIVISIBLE: DIVIDEND = (DIVISOR^N)*(DIVIDEND/(DIVISOR^N))
  COUNTER: COUNTER + DIVISIBLE
 }
 result: COUNTER
}

\ how many groups of successes "SUC" and advantages "ADV" show in DIVIDEND (up to CAP times)? \
function: vacs DIVIDEND SUC ADV CAP {
 DIVISOR: [scav SUC ADV]
 result: [factor DIVIDEND DIVISOR CAP]
}

output [vacs GEN 1 0 10] named "success"
output [vacs GEN 0 1 10] named "advantage"
output [vacs GEN 1 1 10] named "success and advantage"

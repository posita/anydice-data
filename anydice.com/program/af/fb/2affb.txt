function: count VALUES:s in highest NUMBER:n of ROLL:s {
  R: {}
  loop P over {1..NUMBER} { R: {R, P@ROLL} }
  result: [count VALUES in R]  \ here, R is still a sequence \
}

function: modifier VALUE:n {
  result: [count {1,6} in ([absolute VALUE])d6]
}

function: KEEP:n swap SWAP:n {
 result: [count {6} in highest KEEP of (KEEP + SWAP)d6]
}

function: flatten SEQUENCE:n {
 X: SEQUENCE
 if X > 0 {
  result: X
 } else {
  result: 0
 }
}

function: num SEQUENCE:n {
 result: SEQUENCE
}

function: KEEP:n swap SWAP:n mod MOD:n {
 PLAYER: [num [KEEP swap SWAP]]
 MOD_RESULT: [num [modifier MOD]]
 if MOD >= 0 {
  X: PLAYER + MOD_RESULT
 } else {
  X: PLAYER - MOD_RESULT
 }
 result: [flatten X]
}

output [4 swap 2 mod 3] named "+3 mod"

output [4 swap 2 mod 2] named "+2 mod"

output [4 swap 2 mod 1] named "+1 mod"

output [4 swap 2 mod 0] named "0 mod"

output [4 swap 2 mod -1] named "-1 mod"

output [4 swap 2 mod -2] named "-2 mod"

output [4 swap 2 mod -3] named "-3 mod"



FAILURE: -1
TIE: 0
SUCCESS: 1
SUCCESSWITHSTYLE: 2

function: fate ROLL:n using SKILLTARGETDIFF:n {
  \ Assumes target number (difficulty) is 0 \
  if ROLL + SKILLTARGETDIFF < 0 { result: FAILURE }
  if ROLL + SKILLTARGETDIFF = 0 { result: TIE }
  if ROLL + SKILLTARGETDIFF < 3 { result: SUCCESS }
  result: SUCCESSWITHSTYLE 
}


loop SKILLTARGETDIFF over {-3..-1} {
   ASKTD: [absolute SKILLTARGETDIFF]
   output [ fate 4d{-1,0,1} using SKILLTARGETDIFF ] named "4dF - [ASKTD]" 
}
loop SKILLTARGETDIFF over {0..5} {
   ASKTD: [absolute SKILLTARGETDIFF]
   output [ fate 4d{-1,0,1} using SKILLTARGETDIFF ] named "4dF + [ASKTD]" 
}

 if A = 1 {result: 5}
 else {result: 4}
}

function: mag rp A:n B:n {
 if A < 5 {
  if B < 6 {result: 2}
  else {result: 1}
 }
 else {
  if B < 6 {result: 2}
  else {result: 0}
 }
}

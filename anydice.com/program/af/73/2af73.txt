\ function: highest N:n of A:s B:s { \
\    result: {1..N}@[sort {A, B}] \
\ } \

function: pbta ROLL:s {
 if ROLL >= 10 {
  result: 2
 }
 result: ROLL >= 7
}

loop P over {-1..5} {
  output [pbta 2d6+P] named "2d6+[P]"
}

function: pbtadiceonly ROLL:s {
 if {1,2}@ROLL >= 10 {
  result: 2
 }
 result: {1,2}@ROLL >= 7
}

loop R over {1..7} {
  output [pbtadiceonly Rd6] named "[R]d6"
}

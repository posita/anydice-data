function: ataque ROLLEDVALUE:n DEFESA:n {
 if ROLLEDVALUE < DEFESA { result: 0 }
 result: ROLLEDVALUE
}

function: crit ROLLEDVALUE:d MULTIPLIER:n {
 A : 0
 loop N over {1..MULTIPLIER}{
 A: A + ROLLEDVALUE
 }
 result: A
}

function: dano ROLLEDVALUE:n DADO:d MARGEM:n MULTIPLICADOR:n {
 if ROLLEDVALUE >= MARGEM { result: [crit DADO MULTIPLICADOR] }
 if ROLLEDVALUE = 0 { result: 0 }
 result: DADO
}

A: [dano [ataque d20 10] 1d8 20 2]
output A

function: damage N:n goe THRESHOLD:n {
 if N >= THRESHOLD { result: d6 }
 result: 0
}

function: explode DICE:n on THRESHOLD:n {
  X: 0
  loop P over {1..DICE} {
    X: X + [damage d6 goe THRESHOLD]
  }

  result: X
}

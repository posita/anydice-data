function: ROLL:n counting {
   if ROLL >= 10 { result: 1 }
   if ROLL >= 7 { result: 0 }
   result: -1}

output 2@2d{-1,-1,0,0,1} named "lower of 2d6"
loop A over {1..6}{output 1@Ad{-1,-1,0,0,1} named "best of [A]d6"}

output 2@2d{-1,-1,-1,0,0,1} named "lower of 2d6"
loop A over {1..6}{output 1@Ad{-1,-1,-1,0,0,1} named "best of [A]d6"}

output 2@2d{-1,-1,0,0,0,1} named "lower of 2d6"
loop A over {1..6}{output 1@Ad{-1,-1,0,0,0,1} named "best of [A]d6"}

loop A over {-1..3}{output [2d6+A counting] named "2d6+[A]"}

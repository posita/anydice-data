\\\*** Functions ***\\\
function: attack ROLL:n vs ENEMY_AC:n {
 if ROLL = 1 { result: 0 }
 if ROLL >= CRIT_RANGE { result: CRITICALDAMAGE }
 if ROLL + ATTACK >= ENEMY_AC { result: HITDAMAGE }
 result: 0
}


function: bite ROLL:n vs ENEMY_AC:n {
 if ROLL = 1 { result: 0 }
 if ROLL + ATTACK >= ENEMY_AC { result: [spell NORMAL_HIT vs SPELL_DC]}
 result: 0
}

function: spell ROLL:n vs SPELL_DC:n {
 if ROLL + SAVE_A >= SPELL_DC{ result: SUCC_SAVE_A}
 if ROLL + SAVE_A < SPELL_DC{ result: FAIL_SAVE_A}
}

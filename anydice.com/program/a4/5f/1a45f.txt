

function: roll NN:n with B:n
{
  BO: B
  N: NN
  if NN = 0 {
    N: 4
    BO: BO - 2
  }
  result: [explode 1dN + BO]
}

function: roll wild NN:n with B:n
{
  BO: B
  N: NN
  if NN = 0 {
    N: 4
    BO: BO - 2
  }
  result: [highest of [explode 1dN + BO] and [explode 1d6 + BO]]
}

function: raises D:n
{
  result: D / 4 - 1
}

output [raises [roll wild 8 with 0]]

AC: 15
THRS: 20
QLTY: 4
ACR: 0
TOHIT: QLTY + ACR
DMGROLL: 8d10
NUM: 10

\===============================================================\

\ Hit chance when rolling with advantage \
HITADV: [lowest of [highest of (1@2d20+TOHIT-AC+1) and 0] and 1]
CRITADV: [lowest of [highest of (1@2d20-20+1) and 0] and 1]
\output HITADV named "Hit Chance w/ Advantage"\
\output CRITADV named "Crit Chance w/ Advantage"\

\ Hit chance when rolling normally \
HITSTR: [lowest of [highest of (d20+TOHIT-AC+1) and 0] and 1]
CRITSTR: [lowest of [highest of (d20-20+1) and 0] and 1]
\output HITSTR named "Hit Chance Normally"\
\output CRITSTR named "Crit Chance Normally"\

\ Hit chance when rolling with disadvantage \
HITDIS: [lowest of [highest of (2@2d20+TOHIT-AC+1) and 0] and 1]
CRITDIS: [lowest of [highest of (2@2d20-20+1) and 0] and 1]
\output HITDIS named "Hit Chance w/ Disadvantage"\
\output CRITDIS named "Crit Chance w/ Disadvantage"\

\============================================================\

DPRABS: (DMGROLL + CRITADV*DMGROLL)*NUM
output DPRABS named "DPR with direct hits"

DPRADV: (HITADV*DMGROLL + CRITADV*DMGROLL)*NUM
output DPRADV named "DPR rolling w/ advantage"

DPRSTR: (HITSTR*DMGROLL + CRITSTR*DMGROLL)*NUM
output DPRSTR named "DPR rolling normally"

DPRDIS: (HITDIS*DMGROLL + CRITADV*DMGROLL)*NUM
output DPRDIS named "DPR rolling w/ disadvantage"

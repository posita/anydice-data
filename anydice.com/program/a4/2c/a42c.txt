MAGIC: 5
SKILL: 6
EDGE: 3
FORCE: 11

function: postedge N:n {
  if N >= 5 { result: 1 }
  result: [count {5, 6} in 1d6]
}

function: preedge N:n {
  if N = 6 { result: 1 + [preedge 1d6] }
  else if N = 5 { result: 1 }
  else { result: 0 }
}

POOL: MAGIC + SKILL * 2
EDGEPOOL: POOL + EDGE
RFORCE: FORCE * 2
LIMIT: RFORCE + 2

RITUAL: [count {5, 6} in RFORCEd6]
output EDGEPOOLd[preedge d6] - RITUAL named "Pre-Edged"
output POOLd[postedge d6] - RITUAL named "Post-Edged"

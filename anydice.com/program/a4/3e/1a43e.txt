function: dupes in DICE:s {
  MAX: 0
  loop X over {2..#DICE} {
    if MAX=0 {MAX:(X-1)@DICE}
    
    if ((X-1)@DICE = X@DICE) { MAX: X@DICE+10}
  }
  result: MAX
}
output [dupes in 2d10]-[dupes in 2d10] named "2d10"
output [dupes in 4d10]-[dupes in 4d10] named "4d10"
output [dupes in 6d10]-[dupes in 6d10] named "6d10"

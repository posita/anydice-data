function: average of DIE:d{
 TOTAL: 0
 COUNT: 0
 loop N over {DIE}{
  TOTAL: TOTAL + N
  COUNT: COUNT + 1
 }
 result: TOTAL/COUNT
}

loop N over {1..15}{
 loop SIDED over {2, 4, 6, 8, 10, 12, 20}{
  loop MOD over {0..15}{
   DIE: NdSIDED+MOD
   if [maximum of DIE] = 50 & [average of DIE] = 20{
   output DIE named "[N]d[SIDED]+[MOD]"
   }
  }
 }
}

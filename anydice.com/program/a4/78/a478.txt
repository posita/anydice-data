loop CON over {2..7}
{
    loop DICE over {1..4}
    {
        if (DICE == 1)
        {
            output [CON + d6] named "Damage Threshold Con:[CON], 1d6"
        }
        if (DICE == 2)
        {
            output [CON + d8] named "Damage Threshold Con:[CON], 1d8"
        }
        if (DICE == 3)
        {
            output [CON + d10] named "Damage Threshold Con:[CON], 1d10"
        }

        if (DICE == 4)
        {
            output [CON + d12] named "Damage Threshold Con:[CON], 1d12"
        }
    }
}

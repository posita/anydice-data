LUCK: {-1..1}
function: roll with SKILL:s and UNSKILL:s vs DIFF:n {
  if DIFF>0 { 
   result: [highest 1 of SKILLd6] - [highest 1 of (UNSKILL+[absolute DIFF])d6] 
  }
  if DIFF=0 { 
   result: [highest 1 of SKILLd6] - [highest 1 of UNSKILLd6]
  }
  if DIFF<0 {
   result: [highest 1 of (SKILL+[absolute DIFF])d6] - [highest 1 of UNSKILLd6]
  }
}

SKILLS:{0..4}
DIFFS:{0}

loop SKILL over SKILLS {
 loop DIFF over DIFFS {
  UNSKILL: 4-SKILL
  output [roll with SKILL and UNSKILL vs DIFF] named "hab.[SKILL] vs dif.[DIFF]"
 }

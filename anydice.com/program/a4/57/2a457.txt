\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 3
DMGBONUS: 0
PROFICIENCY: 2
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 1d10 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d10 + ATTACKMOD + DMGBONUS

output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "1 Eldritch Blast +2 proficiency +3 Attack modifier"

output 2*[attackroll d20 vs ENEMYAC] named "2 Eldritch Blast +2 proficiency +3 Attack modifier"

output ATTACKTIMESd[attackroll [highest 1 of 2d20] vs ENEMYAC] named "Eldritch Blast with advantage"

output 2*[attackroll [lowest 1 of 2d20] vs ENEMYAC] named "Eldritch Blast with disadvantage"

output 1d10 +3 named "Eldritch Blast with Auto Hit"

output 5d4 named "gold"

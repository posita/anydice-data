\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n hand ISMAIN:n {
 if ROLL >= 20 { result: DAMAGE + DAMAGE + WPNBONUS + (SKILLMOD * ISMAIN) } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + SKILLMOD + WPNBONUS >= AC { result: DAMAGE + WPNBONUS + (SKILLMOD * ISMAIN) }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

PROFICIENCY: 3
SKILLMOD: 4
WPNBONUS: 1
DAMAGE: 1d6 
output d[attackroll d20 vs ENEMYAC hand 1] + d[attackroll d20 vs ENEMYAC hand 0] named "Two +1 Shortswords"
DAMAGE: 1d8 
output d[attackroll d20 vs ENEMYAC hand 1] named "One +1 Rapier"

function: blast POW:n ARM:n {
  result: [damage [result POW] ARM]
}

function: strike POW:n {
    STRIKE: POWd{0,0,0,1,1,2}
  result: STRIKE
}
function: result POW:n { 
result: [strike POW]
}

function: damage STRIKE:n ARM:n {
  if STRIKE < ARM {
    result: 0
  }
  if STRIKE < 2*ARM {
    result: 1
  }
  if STRIKE < 3*ARM {
    result: 2
  }
  if STRIKE < 4*ARM {
    result: 3
  }
 if STRIKE < 5*ARM {
    result: 4
  }
  result: 5
}

output [blast 3 6]

function: o score ROLLEDVALUE:n HANDICAP:n {
  TOTAL: ROLLEDVALUE - HANDICAP
  if TOTAL > 0 {
    result: TOTAL
  } else {
    result: 0
  }
}

function: o roll TIMES:n HANDICAP:n {
  TOTAL: 0
  loop N over {1..TIMES} {
    TOTAL: TOTAL + [o score 1d6 HANDICAP]
  }
  result: TOTAL
}

loop ATTRIBUTE over {1..5} {
  loop SKILL over {1..5} {
    output [o roll ATTRIBUTE (6 - SKILL)] named "Attribute at [ATTRIBUTE] and skill at [SKILL]"
 }
}

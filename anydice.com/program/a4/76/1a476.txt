DICE: 1d20

function: check X:n vs Y:n debuff Z:n {
  if X >= Y { result: 0 }
  result: Z
}

  if X >= 13 { result: 1d7 }
  result: 0
}

output OFFENSE>= 50 named "Offense"

function: horse X:n and Y:n {
  if X >= 15 { result: 3 }
  if X >= 11 { result: 2 }
  if X < 11 & Y >= 11 {
    result: 1
  }
  result: 0
}

function: horsecha X:n and Y:n {
  if X >= 20 { result: 3 }
  if X >= 14 { result: 2 }
  if X < 14 & Y >= 11 {
    result: 1
  }
  result: 0
}

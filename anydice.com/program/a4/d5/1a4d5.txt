function: o score ROLLEDVALUE:n HANDICAP:n {
  TOTAL: ROLLEDVALUE - HANDICAP
  if TOTAL > 0 {
    result: TOTAL
  } else {
    result: 0
  }
}

function: o roll TIMES:n HANDICAP:n {
  TOTAL: 0
  loop N over {1..TIMES} {
    TOTAL: TOTAL + [o score 1d6 HANDICAP]
  }
  result: TOTAL
}

output [o roll 2 5]

output [o roll 2 4]

output [o roll 2 3]

output [o roll 2 2]

output [o roll 3 5]

output [o roll 3 4]

output [o roll 3 3]

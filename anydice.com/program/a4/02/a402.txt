X: 4
WRFL_N: {-1:1, 0:2, 1:3}
WRFL_K: {-1:1, 0:2, 1:1}
WRFL_G: {-1:1, 0:2, 1:5}

function: werfe X modifikator MOD {
 DN: X - [absolute MOD]
 DK: 0
 DG: 0

 if MOD > 0 {
  DG: MOD
 }
 else {
  DK: [absolute MOD]
 }

 result: DNdWRFL_N + DGdWRFL_G + DKdWRFL_K
}

loop A over {-X..X}{
  output [werfe X modifikator A] named "X=[X], MOD=[A]"
}

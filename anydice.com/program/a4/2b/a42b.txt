MAGIC: 5
SKILL: 6
EDGE: 3

function: postedge N:n {
  if N >= 5 { result: 1 }
  result: [count {5, 6} in 1d6]
}

POOL: MAGIC + SKILL * 2
EDGEPOOL: POOL + EDGE

function: preedge N:n {
  if N = 6 { result: 1 + [preedge 1d6] }
  else if N = 5 { result: 1 }
  else { result: 0 }
}

RITUAL: [count {5, 6} in 22d6]
output EDGEPOOLd[preedge d6] - RITUAL
output POOLd[postedge d6] - RITUAL

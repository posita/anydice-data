set "maximum function depth" to 6

function: explode DIE:n count successes at TARGET with a fate point worth FATE_POINT {
  if DIE >= 12 { 
    result: 1 + [explode d12 count successes at TARGET with a fate point worth FATE_POINT]
  }
  result: DIE+FATE_POINT >= TARGET
}

TARGET: 6
FATE_POINT_BONUS: 2

output 4d[explode d12 count successes at TARGET with a fate point worth FATE_POINT_BONUS] named "Number of successes"

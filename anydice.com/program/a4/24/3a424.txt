function: roll stats {
    result: [highest 3 of 4d6]
}

function: reroll stat STAT:n if lt N:n {
    if STAT < N {
        result: [roll stats]
    } else {
        result: STAT
    }
}

function: roll all stats {
    result: 6d[roll stats]
}

function: roll all stats with reroll lt N:n {
    STATS: [roll all stats]
    LOW: 6@STATS
    REST: [highest 5 of STATS]
    result: [reroll stat LOW if lt N] + REST
}

output [roll all stats] / 6 named "Normal Method"
output [roll all stats with reroll lt 11] / 6 named "Single Reroll"

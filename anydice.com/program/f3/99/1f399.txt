set "position order" to "lowest first"

function: onedie A:n {
 if A = 1 {result: 0}
 result: A
}

function: onemore A:s of B:s {
 if B = 1 | A = 0 {result: 0}
 if B > A {result: B}
 result: A
}

X: [onedie d20]

loop N over {1..15}{
 output X named "[N]d20"
 if N < 15 {
  X: [onemore X of d20]
 }
}

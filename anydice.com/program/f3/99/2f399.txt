DIE: {0..9}

function: ROLL:n stat {
  if ROLL = 0 {
    result: -4
  }
  else if ROLL <= 1 {
    result: -3
  }
  else if ROLL <= 2 {
    result: -2
  }
  else if ROLL <= 4 {
    result: -1
  }
  else if ROLL <= 10 {
    result: 0
  }
  else if ROLL <= 12 {
    result: 1
  }
  else if ROLL <= 14 {
    result: 2
  }
  else if ROLL <= 16 {
    result: 3
  }

  result: 4
}

output [2dDIE stat] named "Rolling stats"

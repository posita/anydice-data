\output 5+2d10 - (d20) < 0 named "gambling check (flat roll)"\

\output 3d(5+2d10 - (d20) < 0) named "gambling (flat roll)"\
\output 3d(5+2d10 - (d20+2) < 0) named "gambling (+2 PB)"\

function: gamble SUCCESSES:n{
 if SUCCESSES = 0 { result: -20 }
 if SUCCESSES = 1 { result: -5 }
 if SUCCESSES = 2 { result: 15 }
 if SUCCESSES = 3 { result: 20 }
}

output [gamble 3d(5+2d10 - (d20) < 0)] named "gambling EV (flat roll)"
output [gamble 3d(5+2d10 - (d20+2) < 0)] named "gambling EV (+2 PB)"
output [gamble 3d(5+2d10 - (d20+3) < 0)] named "gambling EV (+3 PB)"
output [gamble 3d(5+2d10 - (d20+3+3) < 0)] named "gambling EV (16 WIS/CHA, +3 PB)"
output [gamble 3d(5+2d10 - (d20+3+3+d4) < 0)] named "gambling EV (16 WIS/CHA, +3 PB, guidance)"
output [gamble 3d(5+2d10 - ([highest 1 of 2d20]+3+3) < 0)] named "gambling EV (16 WIS/CHA, +3 PB, lucky)"
output [gamble 3d(5+2d10 - ([highest 1 of 2d20]+3+3+d4) < 0)] named "gambling EV (16 WIS/CHA, +3 PB, guidance, lucky)"

DISASTER: 0
DISADVANTAGE: 1
MAINTAIN: 2
LOW: 12
HIGH: 17
DN: 5
DS: 6
ADJ_LO: 3
ADJ_HI: 5
ADJ: 0

function: save ROLL:n {
 if ROLL <= LOW { result: DISASTER }
 if ROLL >= HIGH { result: MAINTAIN  }
 result: DISADVANTAGE
}

loop DN over {ADJ_LO..ADJ_HI}{
output [save DNdDS + 0] named"[DN]d[DS] + [ADJ]"}

MISS: 0
HIT: 1
CRITICALHIT: 2
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL > 17 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}
function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}
ATTACK: 5 + 4
HITDAMAGE: (2d8+1d6+27)
CRITICALDAMAGE: 22+(2d8+1d6+26)
MISSDAMAGE: 0
loop AC over {15, 20, 25, 30} {
 output ([damage for [attack [highest 1 of 3d20] vs AC]])*4 named "[AC]"
}

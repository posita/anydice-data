
function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

output [reroll {1,2} on d8] named "reroll 12 on a d8"
output [reroll {1,2,3} on d8] named "reroll 1-3 on a d8"
output [reroll {1,2,3,4} on d8] named "reroll 1-4 on a d8"
output [reroll {1,2,3,4,5} on d8] named "reroll 1-5 on a d8"

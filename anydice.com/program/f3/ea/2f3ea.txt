function: sorcerer ROLLED:n ROLLS:n MANA:n {
if ROLLS > 0 {
 \ if 3, then +1 rolls \
 if ROLLED = 3  {
  ROLLS: ROLLS + 1
 } else {
 \ OR if 4 - 6, +1 mana \
 if ROLLS >= 4 {
  MANA: MANA + 1
 }
 }
 \ expend roll, recursive \
 ROLLS: ROLLS - 1
 result: [sorcerer d6 ROLLS MANA]
}
\ if no rolls, return mana \
\ result: MANA \
result: ROLLS
}

output [sorcerer d6 10 0]
named "Sorcerer"

function: odds of NUMERATOR in DENOMINATOR {
   result: 1dDENOMINATOR <= NUMERATOR
}


DODGE: 6


loop MISSES over {0..9}{
   OUTCOME_ODDS: 1

   \ Output odds of missing this many times in a row, *then* hitting \
   loop I over {1..MISSES}{
      OUTCOME_ODDS: OUTCOME_ODDS & [odds of (DODGE-I) in (11-I)]
   }
   OUTCOME_ODDS: OUTCOME_ODDS & [odds of (11-DODGE) in (10-MISSES)]
   output OUTCOME_ODDS
}

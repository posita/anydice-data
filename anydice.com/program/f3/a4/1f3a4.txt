function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE
 }
 result: DAMAGE
}


STRENGTH: 5
BAB: 5

loop AC over {14..26}{

ATTACK: BAB + STRENGTH
DAMAGE: 2d6 + STRENGTH * 3/2
THREAT: 20
CRITICAL: 3
output [d20 vs AC] named "Earth Breaker, AC [AC], rage"

ATTACK: BAB + STRENGTH - 2
DAMAGE: 2d6 + STRENGTH * 3/2 + 6
output [d20 vs AC] named "Earth Breaker, AC [AC], rage, Power Attack"

ATTACK: BAB + STRENGTH
DAMAGE: d6 + STRENGTH
CRITICAL: 2
output 2d[d20 vs AC] named "Claws (2), AC [AC], rage"

ATTACK: BAB + STRENGTH - 2
DAMAGE: d6 + STRENGTH + 4
output 2d[d20 vs AC] named "Claws (2), AC [AC], rage, Power Attack"
}

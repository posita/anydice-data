function: sorcerous burst DIE:d {
  DAMAGE_SUM: 0
  loop N over {1..#(DIE)} {
    DAMAGE_SUM: DAMAGE_SUM + [sorcerous burst DIE helper]
  }
  result: DAMAGE_SUM
}

function: sorcerous burst N:n helper {
  if N = 6 & ABILITY_MODIFIER > 0 {
    ABILITY_MODIFIER: ABILITY_MODIFIER - 1
    result: N + [sorcerous burst DIE helper] 
  }
  result: N
}

ABILITY_MODIFIER: 3
CANTRIP_DAMAGE: 2d6

output [sorcerous burst CANTRIP_DAMAGE]

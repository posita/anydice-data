
function: A:n thdr N:n {
if A < 3 { result: 1dN}
else { result: A}
}

function:  A:n thd D:n {
 TOTAL : 0
 loop N over {1..A}{
 TOTAL: TOTAL + [1dD thdr D]
}
 result: TOTAL
}

function: crit{ result: [4 thd 6] +3}
function: dmg{ result: [2 thd 6] +3}

function: average damage AC:n MOD:n ROLL:n {
 if ROLL + MOD >= AC | ROLL = 20 { if ROLL >= 19 { result: [crit] } result: [dmg] }
else { result: 0}

}

output [average damage 15 5 1d20]

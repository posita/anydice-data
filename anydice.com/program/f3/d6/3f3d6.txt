DMG_MOD: +0
SHACKLED_BISECTORS_RAMP: 1

function: adv {
    result: [highest 1 of 2d20]
}

function: hit R:n S:n F:n M:n {
    if (R <= 1 + F) {
        result: 0
    } else if (R >= 20 - S) {
        result: M
    } else {
        result: 1
    }
}

\ apply bonus iff roll was a hit \
function: bonus R:n B:n {
    if (R > 0) {
        result: R + B
    }
    result: R
}

function: modulo N:n M:n {
    if N < M {
        result: N
    }
    result: [modulo N-M M]
}

function: bisector N:n {
    M: [modulo N 5]
    R: (N / 5)d12
    if M = 0 {
        result: R + d0
    } else if M = 1 {
        result: R + d4
    } else if M = 2 {
        result: R + d6
    } else if M = 3 {
        result: R + d8
    } else {
        result: R + d10
    }
}

function: shackled bisectors ramped up to N:n {
    A: [hit d20 4 0 2]*[bisector N]
    B: [hit d20 4 0 2]*[bisector N+1]
    result: [bonus A DMG_MOD+3] + [bonus B 3]
}

output [bonus [hit [adv] 4 0 2]d10 DMG_MOD+3] named "Viewing Scythe"
output [bonus [hit d20 4 4 4]*2d6 DMG_MOD] named "Pulsating Blade"
output [shackled bisectors ramped up to SHACKLED_BISECTORS_RAMP] named "Shackled Bisectors"
output [bonus [hit d20 4 0 2]*2*(d6+d8) 3] named "Salamandus Bolter"

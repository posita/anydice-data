function: roll A:n astrology dice {
  if A <= 1 {
    result: 1d4
  }
  if A >= 6 {
    result: [roll A-1 astrology dice] + 1d20
  }
  result: [roll A-1 astrology dice] + 1d{1..A*2+2}
}


loop N over {1..6} {
  output [roll 1dN astrology dice] named "[N] dice"
}


output 1d4+1d6+1d8+1d10+1d12 named "second-last roll"
output 1d4+1d6+1d8+1d10+1d12+1d20 named "final roll"

DEX: 5
PROF: 3
AC: d{12..18}


MISS: 0
HIT:  1
CRIT: 2
function: attack ROLL:n vs DEFENSE:n {
   if ROLL = 1 {result: MISS}
   if ROLL = 20 {result: CRIT}
   if ROLL + ATTACK >= DEFENSE {result: HIT}
   result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = MISS {result: MISSDAMAGE}
 if ATTACKRESULT = HIT  {result: HITDAMAGE}
 if ATTACKRESULT = CRIT {result: CRITDAMAGE}
}


ATTACK: DEX + PROF
MISSDAMAGE: 0
HITDAMAGE:  1d6+DEX
CRITDAMAGE: 2d6+DEX
output [damage for [attack d20 vs AC]] named "Attack"

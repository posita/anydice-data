function: pass of P:n and C:n and T:n {
if P>=T & C<P { result: 2}
if P<T & C<P { result: 1}

if P>=T & C>=P & C<=P-T { result: 0}
if P>=T & C>=P & C>P-T { result: -1}

if P<T & C>=P & C<=P-T { result: 0}
if P<T & C>=P & C>P-T { result: -1}
}

loop T over {0..7} {
  output [pass of 1d8 and 1d8 and T] named "[T]"
}

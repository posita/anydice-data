DAGGER: 1d4
SNEAK: 2d6
MOD: 3
WEAPON_CRIT: (DAGGER + SNEAK) + (DAGGER + SNEAK)
WEAPON_TOTAL: WEAPON_CRIT + MOD

POISON: 1d6
POISON_CRIT: POISON + POISON
POISON_HALF: POISON_CRIT / 2

DAMAGE_TOTAL: WEAPON_TOTAL + POISON_CRIT
DAMAGE_TOTAL_SAVED: WEAPON_TOTAL + POISON_HALF

output WEAPON_TOTAL named "weapon"
output POISON_CRIT named "poison"
output DAMAGE_TOTAL named "total damage (full)"
output DAMAGE_TOTAL_SAVED named "total damage (saved)"

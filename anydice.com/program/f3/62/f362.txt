function: DICE:s TN:n FOCUS:n {
 X: DICE<=FOCUS
 Y: (DICE<=TN) - X
 result: 2*X + Y
}

function: DICE:s TN:n {
 X: DICE<=TN
 Y: DICE<=(TN-5)
 Z: DICE<=(TN-10)
 W: DICE<=(TN-15)
 result: 4*W + 3*(Z-W) + 2*(Y-Z) + 1*(X-Y)
}

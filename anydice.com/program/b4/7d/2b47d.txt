function: evaluate ROLL_A:s STAT_A:n sup ROLL_B:s STAT_B:n{
	if ROLL_A<=STAT_A & ROLL_B<=STAT_B {
		if STAT_A-ROLL_A!=STAT_B-ROLL_B{
  			result : STAT_A-ROLL_A>STAT_B-ROLL_B
		}
		else{ if STAT_A=STAT_B{result : -1}else{result : STAT_A>STAT_B}}
	}
}

loop STAT_A over {2..7}{
loop STAT_B over {2..7}{
output [evaluate 1d10 STAT_A sup 1d10 STAT_B] named "[STAT_A]/[STAT_B]"
}
}

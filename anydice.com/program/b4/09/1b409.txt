function: cash for N:n {
  if N>114 {A:1600} else if N>109 {A:1400} else if N>104 {A:1200} else if N>94 {A:1100} else if N>84 {A:1000} else if N>75 {A:900} else if N>65 {A:750} else if N>20 {A:500} else {A:0}
  result:A
}

function: revenues for M:n N:n {
  result: (8+2*M)d6+N
}

loop M over {1..5} {
  loop N over {10*1d10-1} {
      output [cash for [revenues for M N]] named "[N]@[M]"
  }
}

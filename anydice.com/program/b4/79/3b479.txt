\This is the number of dice you expect to roll\
DIENUM:1
\This is the minimum die face you need to Explode. Anything other then 8,9, or 10 is set to "Cannot explode", 0 shows all graphs\
EXPNUM:10

function: explode ROLLEDVALUE:n EXPNUM:n EXPTIMES:n {
 if (EXPTIMES = 6) {result: 5}
 if (ROLLEDVALUE = 2 & (EXPNUM = 10 | EXPNUM = 0)) { result: 1 + [explode d{0,0,0,0,0,0,0,1,1,2} EXPNUM (EXPTIMES + 1)] }
 result: ROLLEDVALUE
}
output DIENUMd[explode d{0,0,0,0,0,0,0,1,1,2} 10 0] named "explode on 10"

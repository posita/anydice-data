ABILITYA: [highest 3 of 4d6]
ABILITYB: [highest 3 of 4d6]
ABILITYC: [highest 3 of 4d6]
ABILITYD: [highest 3 of 4d6]
ABILITYE: [highest 3 of 4d6]
ABILITYF: [highest 3 of 4d6]

function: valuate for ABILITYRESULTS:s{
D: 0
loop A over ABILITYRESULTS{
if A = 3 {D: D-5}
if A = 4 {D: D-4}
if A = 5 {D: D-3}
if A = 6 {D: D-2}
if A = 7 {D: D-1}
if A = 8 {D: D}
if A = 9 {D: D+1}
if A = 10 {D: D+2}
if A = 11 {D: D+3}
if A = 12 {D: D+4}
if A = 13 {D: D+5}
if A = 14 {D: D+7}
if A = 15 {D: D+9}
if A = 16 {D: D+12}
if A = 17 {D: D+15}
if A = 18 {D: D+18}}
result: D
}

output ([valuate for ABILITYA] + [valuate for ABILITYB] + [valuate for ABILITYC] + [valuate for ABILITYD] + [valuate for ABILITYE] + [valuate for ABILITYF])

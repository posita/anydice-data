function: matches A:s and B:n {
  M: 0
  loop N over A{
    if N = B {
      M: M+1
    }
  }
  result: M
}

function: myo caracteristica CAR:s habilidad HAB:s objeto OBJ:s dificultad DIF:n {
  E_CAR: [matches CAR and 6]
  E_HAB: [matches HAB and 6]
  E_OBJ: [matches OBJ and 6]
  P_CAR: [matches CAR and 1]
  P_OBJ: [matches OBJ and 1]
  
  TMP_RESULT: E_CAR+E_HAB+E_OBJ
  if TMP_RESULT >= DIF{
    result: TMP_RESULT
  }
  else { 
    result: 0
  }
}

output [myo caracteristica 4d6 habilidad 2d6 objeto 1d6 dificultad 1]

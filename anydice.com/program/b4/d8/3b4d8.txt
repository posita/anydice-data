function: hp at level X:n {
 CUR: 0
 loop N over {1..X}{
  CUR: [highest of CUR + 1 and Nd6]
 }
 result: CUR
}

function: hp at level X:n max first reroll ones forever {
 CUR: 6
 loop N over {1..(X-1)}{
  CUR: [highest of CUR + 1 and 6 + Nd{2,3,4,5,6}]
 }
 result: CUR
}

function: A:n reroll one once {
 if A = 1 {
  result: 1d6
 }
 else {
  result: A
 }
}


function: hp at level X:n max first reroll ones once {
 CUR: 6
 loop N over {1..(X-1)}{
  CUR: [highest of CUR + 1 and 6 + Nd[1d6 reroll one once]]
 }
 result: CUR
}


output 10d6
output [hp at level 10]
output 6 + 9d{2,3,4,5,6}
output [hp at level 10 max first reroll ones forever]

output 6 + 9d[1d6 reroll one once]

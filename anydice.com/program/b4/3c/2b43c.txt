function: range on DIE:n between LOW:n and HIGH:n {
  if DIE >= LOW & DIE <= HIGH { result: 1 }
  result: 0
}

DICE_POOL:4d6
BOON:[highest 2 of 2d6]
BANE:[highest 1 of 1d6]

RESULT:DICE_POOL + BOON - BANE 

output [range on RESULT between -100 and 4]named "-4 Tragic No!"
output [range on RESULT between 5 and 9]named "5-9 No, And"
output [range on RESULT between 10 and 14] named "10-14 No, But"
output [range on RESULT between 15 and 19] named "15-19 Yes, But"
output [range on RESULT between 20 and 24] named "20-24 Yes, And"
output [range on RESULT between 25 and 100] named "25+ Triumphant Yes!"

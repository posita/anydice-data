function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRITICALHIT }
 if ROLL + ATTACKMOD >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {  result: CRITICALDAMAGE }
 if ATTACKRESULT = HIT {  result: HITDAMAGE }
 result: MISSDAMAGE
}

function: roundof NUMBEROFATTACKS:n attacks with ATTACKMOD:n bonus and DMGDICE:s and dmgmod DAMAGEMOD:n {
HITDAMAGE: DMGDICE + DAMAGEMOD
CRITICALDAMAGE: (DMGDICE)*2 + DAMAGEMOD
if NUMBEROFATTACKS = 1 { result: [damage for [attack d20 vs AC]] }
if NUMBEROFATTACKS = 2 { result: [damage for [attack d20 vs AC]] + [damage for [attack d20 vs AC]] }
if NUMBEROFATTACKS = 3 { result: [damage for [attack d20 vs AC]] + [damage for [attack d20 vs AC]] + [damage for [attack d20 vs AC]]}
result: MISSDAMAGE
}

MISS: 0
HIT: 1
CRITICALHIT: 2
MISSDAMAGE: 0

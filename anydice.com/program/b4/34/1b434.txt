function: succ ROLL:n DIFFICULTY:n MOD:n {
  if ROLL <= DIFFICULTY-MOD {
    result: 0
  }
  if ROLL < DIFFICULTY {
    result: 1
  }
  if ROLL < DIFFICULTY+MOD {
    result: 2
  }
  if ROLL >= DIFFICULTY+MOD {
    result: 3
  }
}

function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 { result: MISS }
    if ROLL >= CRIT_RANGE { result: CRIT }
    if ROLL + BONUS >= AC { result: HIT }
    result: MISS
}
function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS vs AC hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}

AC: 24
MYCRIT: 17
ADVANTAGE: [highest 1 of 1d20]
ABONUS: 16
DBONUS: 15
DAMAGE: 3d6
HITDMG: DAMAGE + DBONUS
CRITDMG: DAMAGE + DBONUS + DAMAGE + DBONUS + DAMAGE + DBONUS + DAMAGE + DBONUS
STONEBOW_ATTACK: [attack ADVANTAGE plus ABONUS vs AC for HITDMG crit CRITDMG on MYCRIT]
RAPIDSHOT: [attack ADVANTAGE plus ABONUS - 2 vs AC for HITDMG crit CRITDMG on MYCRIT]
STONEBOW_ATTACKS: [attack ADVANTAGE plus ABONUS - 5 vs AC for HITDMG crit CRITDMG on MYCRIT]
RAPIDSHOTS: [attack ADVANTAGE plus ABONUS - 7 vs AC for HITDMG crit CRITDMG on MYCRIT]
TOTAL_FULLATTACK: STONEBOW_ATTACK + STONEBOW_ATTACKS
TOTAL_ATTACKRAPIDSHOT: RAPIDSHOT + RAPIDSHOT+ RAPIDSHOTS

output STONEBOW_ATTACK named "Stonebow"
output TOTAL_FULLATTACK named "Fullattack"
output TOTAL_ATTACKRAPIDSHOT named "Rapid Shot"

function: damage DMG:n att ATTACK:s vs DEFENCE:s {
  SUCCESS: 0
  CRITICAL: 0
  loop DA over ATTACK {
    if DA = 12 {
      CRITICAL: CRITICAL + 1
    }
    if DA >= 7 {
      SUCCESS: SUCCESS + 1
    }
  }
  loop DD over DEFENCE {
    if DD = 12 {
      SUCCESS: SUCCESS - 1
    }
    if DD >= 7 {
      SUCCESS: SUCCESS - 1
    }
  }
  if SUCCESS > 0 {
    result: (DMG + SUCCESS) * (1 + CRITICAL)
  } else {
    result: -1
  }
}

function: maximize with pair R:s {
 H: 1@R
 L: 3@R
 
 if H = L & H <= 2 {
  \ triple, reroll pair \
  result: 2d4 + L
 }
 
 if 2@R = L {
  \ low pair \
  if L <= 2 {
   \ reroll pair \
   result: H + 2d4
  }
  if H <= 2 {
   \ reroll single \
   result: 1d4 + L + L
  }
 }
 
 if H = 2@R {
  \ high pair \
  if L <= 2 {
   \ reroll single \
   result: H + H + 1d4
  }
  if H <= 2 {
   \ reroll pair \
   result: 2d4 + L
  }
 }
 
 \ no pair or no reroll \
 result: R
}

function: minimize with pair R:s {
 H: 1@R
 L: 3@R
 
 if H = L & H > 2 {
  \ triple, reroll pair \
  result: 2d4 + L
 }
 
 if H = 2@R {
  \ high pair \
  if H > 2 {
   \ reroll pair \
   result: 2d4 + L
  }
  if L > 2 {
   \ reroll single \
   result: H + H + 1d4
  }
 }
 
 if 2@R = L {
  \ low pair \
  if H > 2 {
   \ reroll single \
   result: 1d4 + L + L
  }
  if L > 2 {
   \ reroll pair \
   result: H + 2d4
  }
 }
 
 \ no pair or no reroll \
 result: R
}

output [minimize with pair 3d4] named "minimize"
output [maximize with pair 3d4] named "maximize"

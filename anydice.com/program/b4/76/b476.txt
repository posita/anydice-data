MISS: 0
HIT: 1
CRITICALHIT: 2
CRITON: 19

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= CRITON {
  result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

ATTACK: 4 + 4 - 5
loop AC over {15..20} {
output [attack [highest 1 of 2d20] vs AC] named "[AC]"
}

function: average damage for attack ATTACK:n miss MISS:s vs defense DEFENSE:n {
  AROLL: [remove MISS from ATTACKd6]
  DROLL: DEFENSEd6
  result: [calculate damage AROLL vs DROLL]
}

function: remove MISS:s from ROLL:s {
  R: {}
  loop P over {1..#ROLL}{
    N: P@ROLL
    if ![MISS contains N]{ R: {R, N} }
  }
  result: R
}

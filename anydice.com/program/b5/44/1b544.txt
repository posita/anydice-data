\ Outputs the XdY - Z that produce numbers in the M-N range. \
M: 15
N: 18
loop X over {1..N - M} {
 if ((N - M) / X) * X = N - M {
  Y: (X + N - M) / X
  Z: X - M
  if Z > 0 {
   output 4*[highest 3 of 4d6] + 2*(XdY - Z) named "[X]d[Y] - [Z]"
  }
  else {
   Z: -Z
   output 4*[highest 3 of 4d6] + 2*(XdY + Z) named "[X]d[Y] + [Z]"
  }
 }
}

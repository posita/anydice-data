function: at least N from D:s{
ROLL: D


if [ROLL contains 1] {
	if N =1 {
		result: 1
	}
	else {
		result: 2
	}
}
else {
	if [ROLL contains 10] {
		if N = 10 {
			result: 0
		}
		else {
			if N = 11 {
				result: 1
			}
			else {
				result: -1
			}
		}
	}
	else {
		result: ROLL<N
	}
}
}



function: roll X at least Y{
L: 0
loop N over {1..X}{
 L:  L+ [at least Y from d10]
}
result: L
}

output [roll 10 at least 11]

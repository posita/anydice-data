function: roll PA:n and PB:n vs NA:n and NB:n {

  if [lowest  of PA and PB] > [highest of NA and NB] { ANS:  2 }
  if [lowest  of NA and NB] > [highest of PA and PB] { ANS: -2 }

  if [highest of PA and PB] > [highest of NA and NB] & [lowest of PA and PB] <= [highest of NA and NB]{ ANS:  1 }
  if [highest of NA and NB] > [highest of PA and PB] & [lowest of NA and NB] <= [highest of PA and PB]{ ANS: -1 }

  if [highest of PA and PB] = [highest of NA and NB] { ANS:  0 }

  result: ANS
}

output [roll d6 and d6 vs d6 and d6]

function: attack A:n and D:d crits on C:n and adds CD:d {
  if A >= C {
    result: D+CD
  } else {
    result: D
  }
}

output   [attack 1d20 and 1d12 crits on 20 and adds 2d12]
output   [attack 1d20 and 2d 6 crits on 20 and adds 2d 6]
output 2d[attack 1d20 and 1d12 crits on 20 and adds 2d12]
output 2d[attack 1d20 and 2d 6 crits on 20 and adds 2d 6]

function: X:n w Y:n {
result: 1@[sort {X, Y}]
}

function: X:n w Y:n w Z:n {
result: {1..2}@[sort {X, Y, Z}]
}

function: X:n e D:n {
if X = 1 {result: X + [dD e D]}
result: X
}

function: X:n ee D:n {
if X = 1 {result: X + [dD ee D]}
if X = D {result: X + [dD ee D]}
result: X
}

set "maximum function depth" to 5
set "explode depth" to 5

AC: 17
MOD: 5
PROF: MOD + 4
CON: 4

function: X:n triss {
if X = 20 {result: 3d10 + MOD}
if X = 1 {result: 0}
if X + PROF >= AC {result: d10 + MOD}
result: 0
}

function: X:n barb {
if X = 20 {result: 3d12 + MOD + 3}
if X = 1 {result: 0}
if X + PROF >= AC {result: d12 + MOD + 3}
result: 0
}

output 4d([1@3d20 triss]) named "triss"
output 3d((d20 + PROF >= AC)*(2d6 + MOD)) named "greatsword fighter"

output 3d((d20 + PROF - 5 >= AC)*(d10 + MOD + 10)) + (d20 + PROF - 5 >= AC)*(d4 + MOD + 10) named "pam gwm fighter"

output 4d((d20 + PROF >= AC)*(d8 + MOD)) named "standard monk"
output 2d((d20 + PROF >= AC)*(d10 + MOD)) + 2d((d20 + PROF >= AC)*(d8 + MOD)) + d8 named "raw kensei deft + flurry"

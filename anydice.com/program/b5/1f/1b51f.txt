function: mock N:n against SEQ:s returning X:n {
  if [SEQ contains N] {
    result: X
  }
  result: 0
}

function: battle NUM:n with COMBAT:n {
  X: 0
  ONE: {-1, 3, 5, 8, 11}
  TWO: {10, 12, 13, 17}
  THREE: {14}
  loop N over {1..NUM}{
    X: X + [mock (COMBAT - 8 + 2d6) against ONE returning 1] 
    X: X + [mock (COMBAT - 8 + 2d6) against TWO returning 2]
    X: X + [mock (COMBAT - 8 + 2d6) against THREE returning 3]
  }
  result: X
}

output [battle 11 with 3] named "11 goblins"
output [battle 1 with 6] named "1 hobgoblin"

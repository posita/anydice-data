function: sort helper add N:n to S:s {
 if #S = 0 { result: {N} }
 if N >= 1@S { result: {N, S} }
 if N <= (#S)@S { result: {S, N} }

 R: {}
 loop P over {1..#S} {
  if N >= P@S {
   R: {R, N, P@S}
   N: {}
  }
  else { R: {R, P@S} }
 }
 if #N { R: {R, N} }
 result: R
}

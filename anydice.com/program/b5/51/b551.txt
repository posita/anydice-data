function: test ROLL:n vs AC:n {
 if ROLL = 1 { result: 0 }
 else {
  result: 1
 }
}

function: roll B:n A:n{
 result: BdA
}

function: gwf B:n A:n{
 X:0
 loop N over {1..B} {
  X: X + [reroll 1dA A]
 }
 result: X
}

function: reroll A:n B:n{ 
 if A <= 2 { 
  A: 1dB 
 }  
 result: A 
}

MISS: 0
HIT: 1
CRITICALHIT: 2
CRITMISS: -1

function: attackroll ROLL:n vs AC:n {
 if ROLL = 1 { result: CRITMISS }
 if ROLL >= CRITON {
  result: CRITICALHIT }
 if ROLL + ATTACK >= AC { result: HIT }
 result: MISS
}

function: damage for ATTACKROLLRESULT:s {
 D: 0
 loop A over ATTACKROLLRESULT {
  if A = CRITICALHIT {
   D: D + CRITDMG
  }
  else if A = HIT {
   D: D + HITDMG
  }
  else {
   D: D + 0
  }
 }
 result: D + [oncextradmg for ATTACKROLLRESULT]
}

function: bonusdmg for ATTACKROLLRESULT:s {
 D: 0
 loop A over ATTACKROLLRESULT {
  if A = CRITICALHIT {
   D: D + BONUSCRITDMG
  }
  else if A = HIT {
   D: D + BONUSHITDMG
  }
  else {
   D: D + 0
  }
 }
 result: D + [oncextradmg for ATTACKROLLRESULT]
}

function: oncextradmg for ATTACKROLLRESULT:s {
 if [ATTACKROLLRESULT contains CRITICALHIT] {
   result: [roll (2*ONCEXTRADMGNUM) ONCEXTRADMGDICE]
 }
 if [ATTACKROLLRESULT contains HIT] {
   result: [roll ONCEXTRADMGNUM ONCEXTRADMGDICE]
 }
 result: 0
}

function: N:n attacks {
 result: [damage for Nd[attackroll [highest 1 of ATKDICENUMd20] vs AC]]
}

function: N:n bonusattacks {
 result: [bonusdmg for Nd[attackroll [highest 1 of ATKDICENUMd20] vs AC]]
}

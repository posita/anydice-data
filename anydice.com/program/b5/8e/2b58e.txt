function: suc A:n {
 if A >= 18 {
  result: 4
 }
 if A >= 14 {
  result: 3
 }
 if A >= 11 {
  result: 2
 }
 if A >= 9 {
  result: 1
 }
 result: 0
}

function: high N:n of A:s B:s C:s{
    result: N@[sort {A, B, C}]
}

output [suc 2d6 + 1d8] named "Proef"
output [suc 2d6 + 1d10] named "Experto"
output [suc 2d6 + 1d12] named "Maestro"

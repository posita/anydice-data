function: attack A:n and D:d crits on C:n and adds CD:d {
  if A >= C {
    result: D+CD
  } else {
    result: D
  }
}

function: X:n damage N:n players T:n turns D:d damage {
    result: (2*T)d[attack 1d20 and D crits on 20 and adds D] >= X/N
}

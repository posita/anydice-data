\ Change these values to view different/the same Establishment       \
\ levels in the generated plot. Valid levels are 1 through 5.        \
\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\
LOWLEVEL:1                                                            
HIGHLEVEL:2                                                           
\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\
function: cash for N:n {                                              
  if      N>114 {A:1500}                                              
  else if N>109 {A:1400}                                              
  else if N>104 {A:1250}                                              
  else if N> 94 {A:1100}                                              
  else if N> 84 {A:1000}                                              
  else if N> 75 {A: 900}                                              
  else if N> 65 {A: 750}                                              
  else if N> 20 {A: 600}                                              
  else          {A: 250}                                              
  result:A                                                            
}                                                                     
\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\
function: revenues for M:n N:n {                                      
  result: (8+2*M)d6+N                                                 
}                                                                     
\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\
function: levels L:n H:n {                                            
  if L=H {result:{L}} else {result:{L,H}}                             
}                                                                     
\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\\
loop M over [levels LOWLEVEL HIGHLEVEL] {  \ establishment levels    \
  loop N over {10*1d10-1} {                \ @ leverage 9 through 99 \
      output [cash for [revenues for M N]] named "[M]@[N]"            
  }                                                                   
}

function: reroll lowest of ROLL:s as REROLL:d if less than MIN:n {
  LOWEST: (#ROLL)@ROLL       \ the lowest die is sorted last \
  if LOWEST > MIN {
    result: ROLL + 0
  } else {
    REST: {1..#ROLL-1}@ROLL  \ all but the lowest die \
    result: REST + REROLL
  }
}

output [reroll lowest of 2d6 as d6 if less than 2] named "2d6 replace lowest if <= 2"
output [reroll lowest of 1d12 as d12 if less than 2] named "1d12 replace lowest if <= 2"

function: vtmd DIE:n over COMPL:n {
 SUCCESSES: 0
 if DIE=1 { SUCCESSES: SUCCESSES-1 }
 if DIE>=COMPL { SUCCESSES: SUCCESSES+1 }
 if DIE=10 { SUCCESSES: SUCCESSES+[vtmd d10 over COMPL] }
 result: SUCCESSES
}

function: vtm NUMBER:n over COMPL:n {
 R: 0
 loop N over {1..NUMBER} {
  R: R+[vtmd d10 over COMPL]
 }
 result: R
}

output [vtm 6 over 6]

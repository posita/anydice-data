function: limit A:s to B:n above {
  OUT: {}
  loop IDX over {1..(#A)} {
    RAW: IDX@A
    OUT: {OUT, [lowest of RAW and B]}
  }
  result: OUT
}
function: limit A:s to B:n below {
  OUT: {}
  loop IDX over {1..(#A)} {
    RAW: IDX@A
    OUT: {OUT, [highest of RAW and B]}
  }
  result: OUT
}
function: cull A:s above B:n {
  OUT: {}
  loop IDX over {1..(#A)} {
    RAW: IDX@A
    if RAW <= B {
      OUT: {OUT, RAW}
    } 
  }
  result: OUT
}
function: cull A:s below B:n {
  OUT: {}
  loop IDX over {1..(#A)} {
    RAW: IDX@A
    if RAW >= B {
      OUT: {OUT, RAW}
    } 
  }
  result: OUT
}

FA: 2d4 FD:2d2 FT:2 FS:0 \Fighter. Use 1\
RA: 1d6 RD:1d4 RT:2 RS:0 \Ranger. Use 2\
HA: 2d6 HD:1d3 HT:3 HS:0 \Heavy. Use 3\
WA: 1d4 WD:1d10 WT:1 WS:0 \Wizard. Use 4\ 


\Computes an attack, turns neg values to 0\
function: atk A:n def D:n tuf T:n str S:n{
  if (A+S-D)/T < 0{
    result: 0
  }
  else{
    result: (A+S-D)/T
  }
}

\returns the attack using the number references\
function: X:n attacks Y:n{
  if X=1 { \Fighter attacks\
    if Y=1 {
      result: [atk FA def FD tuf FT str FS]
    }
    if Y=2 {
      result: [atk FA def RD tuf RT str FS]
    }
    if Y=3 {
      result: [atk FA def HD tuf HT str FS]
    }
    if Y=4 {
      result: [atk FA def WD tuf WT str FS]
    }
  }
  if X=2 { \Ranger attacks\
    if Y=1 {
      result: [atk RA def FD tuf FT str RS]
    }
    if Y=2 {
      result: [atk RA def RD tuf RT str RS]
    }
    if Y=3 {
      result: [atk RA def HD tuf HT str RS]
    }
    if Y=4 {
      result: [atk RA def WD tuf WT str RS]
    }
  }
  if X=3 { \Heavy attacks\
    if Y=1 {
      result: [atk HA def FD tuf FT str HS]
    }
    if Y=2 {
      result: [atk HA def RD tuf RT str HS]
    }
    if Y=3 {
      result: [atk HA def HD tuf HT str HS]
    }
    if Y=4 {
      result: [atk HA def WD tuf WT str HS]
    }
  }
  if X=4 { \Wizard Attacks\
    if Y=1 {
      result: [atk WA def FD tuf FT str WS]
    }
    if Y=2 {
      result: [atk WA def RD tuf RT str WS]
    }
    if Y=3 {
      result: [atk WA def HD tuf HT str WS]
    }
    if Y=4 {
      result: [atk WA def WD tuf WT str WS]
    }
  }
}

\use to check against all defenses\
loop N over {1..4}{
  output [1 attacks N]
}

\use to check against all attacks\
loop N over {1..4}{
  output [N attacks 1]
}

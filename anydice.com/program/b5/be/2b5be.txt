function: cut R:n at T:n {
  if R > T { result: 0 }
  result: R
}

function: decide R:s {
  if 1@R=0 { result: 0 }
  result: 1@R + (R>0) - 1
}

function: margin N:n vs T:n {
  result: N - T
}

function: success R:s {
  result: (R>0)
}
\
loop T over { 1..8 } {
  D: [ cut 1d10 at T ]
  loop N over { 1..8 } {
    output [ margin [ decide NdD ] vs 3 ] named "[N]d x [T]"
    output [ success NdD ] named "Success [N]d x [T]"
    output [ decide NdD ] / 3 named "[N]d x [T] / 3"
  }
}
\

loop T over { 1..5 } {
  output 1d20 - 1@Td6 named "1d20 - [T]d6"
}

loop T over { 0..5 } {
  output 1d20 + 1@Td6 named "1d20 + [T]d6"
}

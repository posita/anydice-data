
function: ROLL:s hits vs DC:n{
 BADONES: (ROLL = 1)
 SUCCESSES: (ROLL >= DC)
 CRITICAL: (ROLL=12)
 MODS: SUCCESSES-BADONES+CRITICAL
 if (MODS>6) {result: 6}
 if (MODS<-6) {result: -6}
 result: MODS
}

DICE:5
DC:9
ATTR:1

loop DICE over {1..6}{
output ATTR+[(DICE)d12 hits vs (DC)] named "[DICE]d12 successes vs [DC]" 
}

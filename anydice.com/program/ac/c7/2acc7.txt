function: ROLL:n vs AC:n {
 if ROLL >= 20 {
  result: DAMAGE + CRIT
 }
 if ROLL + ATTACK >= AC { 
 result: DAMAGE
 }
 result: 0
}

function: drop ROLL:n {
 if ROLL <= 2 {
   result: 1d6
 }
 result: ROLL
}

LEVEL: 3
AC: d{13..16}
MOD: 3
PROF: 2
OVERFLOW: 2

ATTACK: PROF + MOD
DAMAGE: 1d10 + MOD
CRIT:   1d10
output [d20 vs AC] named "Eldritch Blast lvl 3"

ATTACK: PROF + MOD + OVERFLOW
DAMAGE: 2d6 + MOD
CRIT:   2d6
output [d20 vs AC] named "Kinetic Blast lvl 3"

ATTACK: PROF + MOD
DAMAGE: [drop 1d6] + [drop 1d6] + MOD + 1
CRIT:   [drop 1d6] + [drop 1d6] + 1
output [d20 vs AC] named "Greatsword lvl 3"

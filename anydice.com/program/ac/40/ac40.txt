function: attk ROLL:n ROLLTWO:n vs ac AC:n with tohit BAB:n str ABI:n dmg DMG:n {
  if (ROLL >= 20) {
    if (ROLLTWO + BAB + ABI >= AC) {
      result:  [absolute DMG * 2]
    }
  }

  if (ROLL + BAB + ABI > AC) {
    result: [absolute DMG]
  }
  result: 0
}

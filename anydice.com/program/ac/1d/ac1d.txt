function: crt DIEA:n n DIEB:n r RANGE:n {
  if DIEA <= RANGE {
    A: DIEA
  } if DIEA > RANGE {
    if DIEA = 20 {
      A: 100
    } else {
      A: 0
    }
  }
  if DIEB <= RANGE {
     B: DIEB
  } if DIEB > RANGE {
    if DIEB = 20 {
      B: -100
    } else {
      B: 0
    }
  }

  if A = 100 {
    result: 100
  } if B = -100 {
    result: -100
  } else {
    if A >= B {
      result: A
    } if A < B {
      result: B
    }
  }
}

function: two DIEA:n n DIEB:n r RANGE:n {
  if DIEA <= RANGE {
    A: DIEA
  } if DIEA > RANGE {
    A: 0
  }
  if DIEB <= RANGE {
     B: DIEB
  } if DIEB > RANGE {
    B: 0
  }
  if A >= B {
    result: A
  } if A < B {
    result: B
  }
}

function: sub DIEA:n n DIEB:n r RANGE:n {
  if DIEA >= DIEB {
    TOTAL: DIEA - DIEB
  } else {
    TOTAL: DIEB - DIEA
  }
  if TOTAL <= RANGE {
    result: TOTAL
  } else {
    result: 0
  }
}

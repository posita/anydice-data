\PARA IGNORAR UMA LINHA DE CODIGO COLOQUE CONTRABARRAS COMO FIZ NESTA LINHA\
\PARA ATIVAR UMA LINHA DE CODIGO RETIRE AS CONTRABARRAS\
\ N E O NUMERO DE DADOS. J E O SEGUNDO DADO SE MAIS DE UM TIPO DE DADO ESTIVER SENDO USADO\

\DEFINICAO DOS DADOS\

INST: {1..6}
APT: {1..10}
ASS: {1..12}

\DEFINICAO DAS FACES DOS DADOS\

NADA: {1,2}
PRESS_NADA: {3, 12}
PRESS_ADAP: {4, 5}
PRESS: {3, 4, 5, 9, 10, 11, 12}
SUCESSO: {6, 7, 8, 9, 10, 11}
SUCESSO_NADA: {6, 7}
SUCESSO_ADAP: {8}
SUCESSO_SEM_PRESS: {6, 7, 8}
SUCESSO_ADAP_PRESS: {9, 11}
SUCESSO_PRESS_NADA: {10}
SUCESSO_PRESS: {9, 10, 11}

\CHANCE DE SUCESSO COM N DE INSTINTO E J DE APTIDAO\

loop N over {1..3}{
 loop J over {0..2}{
  output [count SUCESSO in NdINST] + [count SUCESSO in JdAPT] >0 named "SUCESSO_[N]INST_[J]APT"
 }
}

\CHANCE DE SUCESSO COM N DE INSTINTO EM J DE APTIDAO VIR COM PRESSOES\

\loop N over {1..3}{\
\ loop J over {0..2}{\
\  output [count SUCESSO_PRESS in NdINST] + [count SUCESSO_PRESS in JdAPT] >0 named "SUCESSO_PRESS_[N]INST_[J]APT"\
\ }\
\}\

\CHANCE DE SUCESSO COM N DE INSTINTO EM J DE APTIDAO VIR COM PRESSOES\

\loop N over {1..3}{\
\ loop J over {0..2}{\
\  output [count SUCESSO_SEM_PRESS in NdINST] + [count SUCESSO_SEM_PRESS in JdAPT] >0 named "SUCESSO_SEM_PRESS_[N]INST_[J]APT"\
\ }\
\}\

\CHANCE DE SUCESSO AGINDO POR INSTINTO\

\loop N over {1..5}{\
\ output [count SUCESSO in NdASS] >0 named "SUCESSO_[N]ASS"\
\}\

\CHANCE DE SUCESSO COM PRESSOES AGINDO POR INSTINTO\

\loop N over {1..5}{\
\ output [count SUCESSO_PRESS in NdASS] >0 named "SUCESSO_PRESS_[N]ASS"\
\}\

\CHANCE DE SUCESSO SEM PRESSOES AGINDO POR INSTINTO\

\loop N over {1..5}{\
\ output [count SUCESSO_SEM_PRESS in NdASS] >0 named "SUCESSO_SEM_PRESS_[N]ASS"\
\}\

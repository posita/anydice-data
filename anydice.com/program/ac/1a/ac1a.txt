function: two DIEA:n n DIEB:n r RANGE:n {
  if DIEA <= RANGE {
    A: DIEA
  } if DIEA > RANGE {
    A: 0
  }
  if DIEB <= RANGE {
     B: DIEB
  } if DIEB > RANGE {
    B: 0
  }
  if A >= B {
    result: A
  } if A < B {
    result: B
  }
}

function: sub DIEA:n n DIEB:n r RANGE:n {
  if DIEA >= DIEB {
    TOTAL: DIEA - DIEB
  } else {
    TOTAL: DIEB - DIEA
  }
  if TOTAL <= RANGE {
    result: TOTAL
  } else {
    result: 0
  }
}

function: vstwo SKILLA:n r RANGEA:n vs SKILLB:n r RANGEB:n {
  result: SKILLA - SKILLB + [two d20 n d20 r RANGEA] - [two d20 n d20 r RANGEB]
}
function: vssub SKILLA:n r RANGEA:n vs SKILLB:n r RANGEB:n {
  result: SKILLA - SKILLB + [sub d20 n d20 r RANGEA] - [sub d20 n d20 r RANGEB]
}

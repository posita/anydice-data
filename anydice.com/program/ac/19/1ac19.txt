N: 3
DADO: 20
DIFS: [reverse {1..19}]

function: closest of RESULT:n die to DIFFICULTY:n {
   if RESULT = 0 {
      result: 0
   }
   if RESULT = 1 {
      result: 100000 + [lowest 1 of d{21-DIFFICULTY..20}]
   }
   if RESULT = 2 {
      result: 200000 + [lowest 1 of 2d{21-DIFFICULTY..20}]
   }
   if RESULT = 3 {
      result: 300000 + [lowest 1 of 3d{21-DIFFICULTY..20}]
   }
}

loop D over DIFS {
  DD: 21-D
  output [closest of [count {1..D} in NdDADO] die to D] named "DIF [DD] ([N]D[DADO])"
}

output d20

function: choice of A:n and B:n and C:n target T:n {
  if A=T { result: A }
  if B=T { result: B }
  if C=T { result: C }
  result: 1d{A, B, C}
}
function: combo of A:n and B:n and C:n target T {
  result: [choice of A+B and B+C and A+C target T]
}

loop T over {1..12} {
  output [combo of 1d6 and 1d6 and 1d6 target T] named "2/3d6 [T]"
}

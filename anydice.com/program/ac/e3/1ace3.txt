function: attackroll ROLL:n vs AC:n for DAMAGE:n and CRITDAMAGE: n {
 if ROLL >= CRITROLL { result: CRITDAMAGE + DAMAGE }
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}
ATTACKMOD: 5
DMGBONUS: 2
PROFICIENCY: 4
ACCBONUS: 0
CRITROLL: 20
CRITDAMAGE: 1d8
DAMAGE: 1d8 + ATTACKMOD + DMGBONUS
output 3d[attackroll [highest 1 of 2d20] vs ENEMYAC for 1d8+ATTACKMOD + DMGBONUS and 1d8] named "Shield Master Paladin 12"

ATTACKMOD: 5
DMGBONUS: 10
PROFICIENCY: 4
ACCBONUS: -5
CRITROLL: 20
output 2d[attackroll [highest 1 of 2d20] vs ENEMYAC for 1d10+ATTACKMOD + DMGBONUS and 1d10] + [attackroll [highest 1 of 2d20] vs ENEMYAC for 1d4+ATTACKMOD + DMGBONUS and 1d4] + [attackroll d20 vs ENEMYAC for 1d10+ATTACKMOD + DMGBONUS and 1d10] named "GWM/PAM Paladin 12"

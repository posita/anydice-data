function: work ROLL:s N:n D:n {
 loop P over {1..N}{
  loop M over {1..D}{
   if ROLL + 1 = M*N + P {result: P}
  }
 }
 result: -1 \if -1 shows up in the results, it means something has gone wrong\
}
function: who goes first of N:n players rolling D:n sided dice {
 result:[work NdD N D]
}
output [who goes first of 4 players rolling 6 sided dice]

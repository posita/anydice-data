function: make attack against AC:n ac with ROLL:n plus ROLLMOD:n and PROF:n proficiency and MOD:n ability modifier for DMG:d plus DMGMOD:n damage critting on CRITMIN:n {
  if ROLL >= CRITMIN {
    result: DMG + DMG + DMGMOD + MOD
  } else if ROLL + ROLLMOD + PROF + MOD >= AC {
    result: DMG + DMGMOD + MOD
  } else {
    result: 0
  }
}

function: make attack against AC:n ac with ROLL:n and PROF:n proficiency and MOD:n ability modifier for DMG:d {
  result: [make attack against AC ac with ROLL plus 0 and PROF proficiency and MOD ability modifier for DMG plus 0 damage critting on 20]
}

\ Standard attack with 2 prof and a +3 ability modifier against 12 AC doing 1d8 damage \

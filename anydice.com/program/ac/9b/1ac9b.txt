FIGHT:2d6
WARLO:1d8
MOD:4

function: great N:n X:n {
 if N < 3 { result: 1dX }
 result: N
}

function: greatest {
  result: [great 1d6 6] + [great 1d6 6]
}


\damage die, level, extra enemy\
function: green N:n L:n X:n {
  result: (N + 1 + MOD) + Ld8 + X*(MOD + Ld8)
}



\pew pew + hex\
function: eldritch L:n {
  result: (1d10 + MOD + 1d6) + Ld10 + L*MOD + Ld6
}

UPGRADE:0

\output 2d6+1+MOD named "basic"\
\output [green 1d8 LVL 0] named "green rapier solo"\
output [green 2d6 UPGRADE 0] + 1d6 named "green great solo"
\output [green [greatest] LVL 0] + 1d6 named "green great great solo 5"\

\output [green 1d8 LVL 1] named "green rapier extra"\
output [green 2d6 UPGRADE 1] + 1d6 named "green great extra"
\output [green [greatest] LVL 1] + 1d6 named "green great great extra 5"\

output [eldritch UPGRADE] named "blast"


\hexblade warlock using longsword and hex spell active\
output 1d8 + 1d6 + MOD named "hex hex warlock lvl 1"

\switch to greatsword using invocation Improved Pact Weapon\
output 2d6 + 1d6 + 2*MOD + 1 named "hex hex warlock lvl 3"

\Adding invocations Extra Attack, Maddening Hex\
\output 2d6 + 1d6 + MOD + 1 + 2d6 + 1d6 + MOD + 1 + MOD named "hex hex warlock lvl 5"\

\taking one level of fighter for great weapon fighting\
output [great 1d6 6] + [great 1d6 6] + 1d6 + MOD + 1 + [great 1d6 6] + [great 1d6 6] + 1d6 + MOD + 1 + MOD named "hexlock 5, fighter 1"

output [greatest] + 1d6 + MOD + 1 + [greatest] + 1d6 + MOD + 1 + [greatest] + 1d6 + MOD + 1 + MOD named "hex 5, fight 11"

\Adding LifeDrinker\
output [great 1d6 6] + [great 1d6 6] + 1d6 + 2*MOD + 1 + [great 1d6 6] + [great 1d6 6] + 1d6 + 2*MOD + 1 + MOD named "hex 12 fight 1"

\Adding Cloak of Flies\
output [great 1d6 6] + [great 1d6 6]  + 1d6 + 2*MOD + 1 + [great 1d6 6] + [great 1d6 6] + 1d6 + 2*MOD + 1 + MOD + MOD named "hex 18 fight 1"

output [greatest] + MOD + [greatest] + MOD + [greatest] + MOD + [greatest] + MOD named "fight 20"

function: roll A:s vs B:n severe crit C:n D:n {
  BEST: 1@A
  if (A <= C) > 0 { result: -1 }
  if BEST >= D { result: 2 }
  result: BEST >= B
}

function: roll A:s vs B:n crit C:n D:n {
  BEST: 1@A
  if BEST <= C { result: -1 }
  if BEST >= D { result: 2 }
  result: BEST >= B
}

function: roll disadvantage A:s vs B:n crit C:n D:n {
  WORST: (#A)@A
  if WORST <= C { result: -1 }
  if WORST >= D { result: 2 }
  result: WORST >= B
}

function: ROLL:s less than TARGET:n {
  TOTAL: 0
  loop N over {1..#(ROLL)} {
    X: N@ROLL
    if X < TARGET{
      TOTAL: TOTAL + 1
    }
  }
  result: TOTAL
}

BASE_LEVEL: 6

loop N over {BASE_LEVEL..16} {
  LEVEL: N - BASE_LEVEL
  output [2d20 less than N] > 1 named "2d20 lvl [LEVEL] ([N])"
}

function: roll E:n S:n F:n C:n X:n {
 if E=1 { result: -2 }
 if X=1 { result: E }
 if S=1 { result: -4 }
 if X=2 { result: E+S }
 if F=1 { result: -6 }
 if X=3 { result: E+S+F }
 if C=1 { result: -8 }
 result: 2*(E+S+F)
}

loop X over {1..4} {
 output [roll 1d8 1d6 1d4 1d2 X] named "Stop after [X]"
}

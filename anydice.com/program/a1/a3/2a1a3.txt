function: blades ROLL:s { result: (1@ROLL>=4) + (1@ROLL=6) }
function: ogre A:s B:s C:s {
 RES: [sort {A, B, C}]
 result: (1@RES>=5) + (2@RES>=5)
}

output [blades 2@2d6] named "0d"
loop P over {1..7} { output [blades Pd6] named "blades[P]d" }
loop A over {6,8,10,12}{
 loop B over {6,8,10,12}{
  loop C over {6,8,10,12}{
   output [ogre 1dA 1dB 1dC] named "d[A]+d[B]+d[C]"
  }
 }
}

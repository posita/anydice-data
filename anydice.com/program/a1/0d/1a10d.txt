function: mydamage A:n extra B:n off O:n hex H:n curse C:n flourish F:n{
TOTAL: 0
NUMHEX: 0
if A>1{ 
TOTAL: TOTAL + 2d6 + 6
NUMHEX: NUMHEX + 1
}
if A=1 {
TOTAL: TOTAL + 1d6 + 6
NUMHEX: NUMHEX + 1
}
if B>1{ 
TOTAL: TOTAL + 2d6 + 6
NUMHEX: NUMHEX + 1
}
if B=1 {
TOTAL: TOTAL + 1d6 + 6
NUMHEX: NUMHEX + 1
}
if O=1 {
TOTAL: TOTAL + 1d6 + 7
NUMHEX: NUMHEX + 1
}
if H=1 {
TOTAL: TOTAL + (NUMHEX)d6
}
if C=1 {
TOTAL: TOTAL + 12
}
if F=1 {
TOTAL: TOTAL + 1d8
}
result: TOTAL
}

output [mydamage d6 extra d6 off 1 hex 1 curse 1 flourish 1]

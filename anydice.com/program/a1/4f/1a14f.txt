function: dicepool sides SIDES:n tn TN:n botch BOTCH:n {
 result: [rolldicepool dSIDES sides SIDES tn TN botch BOTCH]
}

function: rolldicepool N:n sides SIDES:n tn TN:n botch BOTCH:n {
 if N = SIDES { result: 1 + [rolldicepool dSIDES sides SIDES tn TN botch BOTCH] }
 if N >= TN { result: 1 }
 if N <= BOTCH {result: -1}
 result: 0
}

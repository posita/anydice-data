function: max N of AA:n{
 if AA > N {
   result: N
 }
 else {
   result: AA
 }
}


function: min N of AA:n{
 if AA < N {
   result: N
 }
 else {
   result: AA
 }
}


function: bounded M:n N:n of D:n {
 if D < N | D > M {
   result: -1
 }
 else {
   result: D
 }
}

function: reroll ROLL:n on VAL:n to D:d {
   
   if ROLL = VAL { result: [reroll D on VAL to D] }
   else { result: ROLL }
}

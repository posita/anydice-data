set "explode depth" to 2

function: D:n b B:n {
 result: [highest D of (D+B) d6]
}

function: D:n x B:n {
 if D <= 1 {
  result: [explode d6]
 }
 else {
  result: [highest (D-1) of ((D-1)+B) d6] + [explode d6]
 }
}

function: D:n xx B:n {
 if D <= 1 {
  result: [explode d6]
 }
 else {
  result: [highest of ([(D-1)bB] + [explode d6]) and ([DbB])]
 }
}

loop D over {2..2} {
 loop B over {0..D} {
  output [DbB] named "[D]b[B]"
  output [DxxB] named "[D]xx[B]"
 }
}

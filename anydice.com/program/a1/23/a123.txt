HIT: 1
MISS: 0
MISSDAMAGE: 0

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

loop DAM over {0, 10, 20} {
 HITDAMAGE: 18 + 14 + DAM
  loop ARM over {0, 20, 40} {
   ARMOR: (100 - ARM)
   HITDAMAGE: (18 + 14 + DAM) * ARMOR / 100
    loop AC over {-20, -10, 0, 10, 20} {
     output [damage for d100 >= d100 + AC] named "DAM:[DAM] - AC:[AC] - ARM:[ARM]"
    }
    \ output [damage for d100 >= d100] named "DAM:[DAM] - ARM:[ARM]" \ 
  }
}

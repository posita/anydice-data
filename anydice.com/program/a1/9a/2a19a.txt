function: blades ROLL:s { result: (1@ROLL>=4) + (1@ROLL=6) }
function: ogre A:s B:s C:s {
 RES: [sort {A, B, C}]
 result: (3@RES<=4) + (2@RES<=4)
}

output [blades 2@2d6] named "0d"
loop P over {1..6} { output [blades Pd6] named "blades[P]d" }
loop A over {12,10,8,6}{
 loop B over {12,10,8,6}{
  loop C over {12,10,8,6}{
   output [ogre 1dA 1dB 1dC] named "d[A]+d[B]+d[C]"
  }
 }
}

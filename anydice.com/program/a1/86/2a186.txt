function: blades ROLL:s { result: (1@ROLL>=4) + (1@ROLL=6) }
function: ogre ROLL:s ATTR:s {
 FSTDIE: 1@ROLL
 SECDIE: 2@ROLL
 if ATTR > 2@ROLL { SECDIE: ATTR }
 if ATTR > 1@ROLL {
  FSTDIE: ATTR
  SECDIE: 1@ROLL
 }
 result: (FSTDIE>=5) + (SECDIE>=6)
}

loop P over {1..6} { output [blades Pd6] named "blades[P]d" }
loop A over {6,8,10,12}{
 loop P over {1..8} { output [ogre Pd6 1dA] named "d[A]+[P]d" }
}

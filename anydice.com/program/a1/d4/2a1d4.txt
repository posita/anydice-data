function: hitme D:n H:n {
  result: [count {H..6} in Dd6]
}

function: comparison A:s D:s {
  R: 0
  loop N over {1..[highest of #A and #D]} {
   if (N@A > N@D) & (N@A > 0) & (N@D > 0) { R: R + 1 }
   if (N@D > N@A) & (N@A > 0) & (N@D > 0) { R: R }
   if (N@A = 0) & (N@D >= 1) { R: R }
   if (N@D = 0) & (N@A >= 1) { R: R + 1}
  }
  result: R
}

output [hitme 2 1] named "2d6 hitting on 1"
output [hitme 2 1]d6  named "(2d6 hitting on 1)d6"
output 2d6 named "2d6"
output [comparison [hitme 2 1]d6 1d6] named "on hit"
output [comparison 2d6 1d6] named "risk"

\D6 system with "critical failures" on a wild die subtracting highest roll and "critical successes" exploding\
SIDES: 6
WILDS: 1
set "explode depth" to 2

loop DICE over {0..9}
{
\PIPS: SKILL-3*(SKILL/3)\
\DICE: SKILL-WILDS\

function: wildroll WILD:n {
 if WILD = WILDS{
  result: {2..DICE}@DICEdSIDES}
 if WILD = WILDS*SIDES{
  result: (WILDS*SIDES + [explode WILDSdSIDES]) + DICEdSIDES}
 result: WILD + DICEdSIDES
}
output [wildroll WILDSdSIDES] named "([WILDS] wild and [DICE])d[SIDES]"
}

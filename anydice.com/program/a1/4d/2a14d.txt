function: multi helper N:n SIDES:n VALUE:n {
  if VALUE=SIDES {
    result: VALUE + [multi N+1]
  } else {
    result: VALUE
  }
}

function: multi N:n {
  DICE: {4,6,8,10,12,20}
  if N < 1 { result: -1000 }
  else { if N > #DICE { result: 0 }
  else {

  SIDES: N@DICE
  ROLL: 1dSIDES
  result: [multi helper N SIDES ROLL]

  }}
}

set "maximum function depth" to 15

output [multi 1] named "Result"

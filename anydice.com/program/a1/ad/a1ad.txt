function: A:n prog B:n{
if A <= B{ result: B }
else{ result: A }
}

function: noright A:d{
MAX: [maximum of A]
result: [noright A helper]
}

function: noright N:n helper{
if N = MAX {result: N+2d10}
else {result: N}
}

function: accurate N:n D:d{
if N >= 4 {result: D+2d10}
else if N < 4 & N >= 2 {result: D+1d10}
else {result: D}
}

function: charlie accurate N:n{
if N >=6 {result: 6}
else if N >= 2 & N < 6 {result: 2+[charlie accurate N-2]}
else {result: 0}
}

function: maser D:d{
MAX: [maximum of D]
result: [maser helper D]
}

function: maser helper N:n {
 if N = MAX | N = MAX-1 { result: N + [maser helper D] }
 result: N
}

function: nocompact D:d{
MAX: [maximum of D]
C: 1d2
CMAX: [maximum of C]
result: [nocompact helper D C]
}

function: nocompact helper A:n B:n{
if A = MAX & B = CMAX {result: A + [nocompact helper D C]}
else {result: A}
}

function: falling D:n{
if D<= 6 {result: 1d5}
else if D > 6 & D <= 15 {result: 1d10}
else if D > 15 & D <= 30 {result: 1d10+2}
else if D > 30 & D <= 50 {result: 1d10+4}
else if D > 50 & D <= 100 {result: 1d10+6}
else if D > 100 & D <= 200 {result: 2d10+2} \the book includes 100 in this range, I'm assuming it's a mistake\ 
else if D > 200 & D < 16000 {result: 2d10+5} \the book has a dead zone of 201-299. I'm assuming whoever wrote that was retarded\
else {result: 2d10+8}
}

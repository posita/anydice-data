
\ attack die value \
X: 3d8+5
\ defense die value \
Y: 3d8





\ Block function \

function: block A B {
 C: A / 2
 D: 0
 if(C > B) { D: A - B }
 D: A - C
 if(D < A) { result: A - D }
 result: 0
}

function: negate A B {
 if(B < A) { result: A - B }
 result: 0
}

\ Armor Rolltable Functions \

function: arm light none A:n B:n {
 if(B >= 21) { result: [negate A 40] }
 result: A
}

function: arm light leather A:n B:n {
 if(B >= 17) { result: [negate A 50] }
 result: A
}

function: arm medium chain A:n B:n {
 if(B >= 21) { result: [negate A 40] }
 if(B >= 13) { result: [block A 20] }
 result: A
}

function: arm heavy plate A:n B:n {
 if(B >= 15) { result: [block A 20] }
 if(B >= 11) { result: [block A 20] }
 result: A
}


\ Attack Function (set the target armor here) \

function: atk A {
 result: [arm heavy plate A Y]
}




\ Weapon Rolltable Functions \

function: wep dagger A:n {
 if A >= 18 { result: 24 }
 if A >= 9 { result: 4 }
 result: 0
}

function: wep greatsword A:n {
 if A >= 18 { result: 16 }
 if A >= 15 { result: 10 }
 if A >= 10 { result: 6 }
 result: 0
}

AC: 13
PROF: 4
MOD: 5
SPELL: MOD + PROF
FIREBOLT: 2d10
FIREBOLTM: 20
SHOCKING_GRASP: 2d8
SHOCKING_GRASPM: 16

function: attack ROLL:n plus ATTACK:n vs DEFENCE:n {
 if ROLL = 20 {
  result: 2
 }
 if ROLL + ATTACK >= DEFENCE {
  result:1
 }
 result: 0
}

function: damage NAME:n HIT:n {
 if NAME = 1 {
  if HIT = 1 {result: FIREBOLT}
  if HIT = 2 {result: FIREBOLT + FIREBOLTM}
 }
 if NAME = 2 {
  if HIT = 1 {result: SHOCKING_GRASP}
  if HIT = 2 {result: SHOCKING_GRASP + SHOCKING_GRASPM}
 }
 result: 0
}


output [damage 1 [attack d20 plus SPELL vs AC]] named "firebolt"
output [damage 2 [attack d20 plus SPELL vs AC]] named "shocking grasp"

APR: 3
RNDS: 10

function: attack DMG misfire MF ammo AM {
 TOTALDMG: 0
 AMMOTRACK: AM
 MFLOSTATTACKS: ((APR*RNDS-(APR*RNDS/(1+AM)))*MF*APR)/20

 loop I over {MFLOSTATTACKS..APR*RNDS}{
   if AMMOTRACK > 0 {
     TOTALDMG: TOTALDMG + DMG
     AMMOTRACK: AMMOTRACK - 1
   } else {
     AMMOTRACK: AM
   }
 }

 result: TOTALDMG
}

output [attack 1d10 misfire 1 ammo 4] named "Pistol"
output [attack 1d12 misfire 2 ammo 1] named "Musket"
output [attack 1d10 misfire 2 ammo 6] named "Pepperbox"
output [attack 1d8 misfire 3 ammo 2] named "Scattergun"
output [attack 2d12 misfire 3 ammo 1] named "Bad News"
output [attack 2d8 misfire 3 ammo 1] named "Hand Mortar"

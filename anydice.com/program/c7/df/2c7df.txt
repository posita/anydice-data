function: advantage A:s with B:s {
  \ Roll dice set A, replace at most one of them with a die from dice set B \
  BASE: {1..(#A-1)}@A
  ADD: [highest of (#A)@A and 1@B]
  result: BASE + ADD
}

output [highest of 1d6 and 1d6] + 1d6 named "minor advantage"
output [highest of 1d6 and 1d6] + [highest of 1d6 and 1d6] named "major advantage"

output [advantage 2d6 with 2d4] named "custom advantage 2d4"
output [advantage 2d6 with 2d6] named "custom advantage 2d6"

function: count VALUES:s in SEQUENCE:s {
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
 }
 result: COUNT
}

output [count 3 in 3d6] named "rolling 6s on 3d6"
output [ count {6..12} in 3d6 ] + [ count {6..12} in 1d8 ] named "3d6,d8"
output [ count {6..9} in 1d10 ] + [ count {10..12} in 1d10 ] named "3d6,d8"

function: succ SUM:n {
  if SUM > 0 {result: SUM/100 + 1}
  if SUM > -10 {result: 1}  
  result: 0
}

function: oldsucc SUM:n {
  if SUM = 0 {result: 1}
  if SUM > 0 {result: SUM/5 + 1}
  result: 0
}

function: roll ATTR:d with SKILL:d vs DIFFICULTY:n {
  SUM: ([explode ATTR] + [explode SKILL] - DIFFICULTY)
  result: [succ SUM]
}

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}
MISS: 0
HIT: 1
CRITICALHIT: 2

DEFENSE: d{12..22}

ATTACK: 5 + 2
HITDAMAGE: 2d10
CRITICALDAMAGE: 4d10
MISSDAMAGE: 0
output [attack] named "Firebolt"

ATTACK: 5 + 2
HITDAMAGE: 1d10
CRITICALDAMAGE: 2d10
output [attack] + [attack] named "Eldritch blast"

ATTACK: 5 + 3
HITDAMAGE: d8 + 4
CRITICALDAMAGE: 12
output [attack] named "Longsword"

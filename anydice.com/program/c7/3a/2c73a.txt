function: round VALUE:n {
  if VALUE > 0 {
    result: VALUE
  } else {
    result: 0
  }
}

function: score ATTRIBUTE:n SKILL:n {
  HANDICAP: (4 - SKILL) * ATTRIBUTE
  ROLL: ATTRIBUTE d6
  TOTAL: ROLL - HANDICAP

  result: [round TOTAL]
}

loop ATTRIBUTE over {1..5} {
  loop SKILL over {0..5} {
    output [score ATTRIBUTE SKILL] named "Attribute at [ATTRIBUTE] and bonus at +[SKILL]"
  }
}

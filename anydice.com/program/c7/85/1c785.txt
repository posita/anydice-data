\ FU dice with crits and fumbles \
function: helper DICE:s P:n {
 H: P@DICE \ highest number rolled if P = 1 (bonus dice), or lowest if P = #DICE (penalty dice) \
 \ if rolling bonus dice, add the number of extra 6s rolled \
 if P = 1 & H = 6 {H: H + ((DICE = 6) - 1)} 
 \ if rolling penalty dice, subtract the number of extra 1s rolled \
 if P > 1 & H = 1 {H: H - ((DICE = 1) - 1)}
 result: H
}
function: fu with crits DICE:s high {
 result: [helper DICE 1]
}
function: fu with crits DICE:s low {
 result: [helper DICE (#DICE)]
}
output [fu with crits 5d6 low] named "-4"
output [fu with crits 4d6 low] named "-3"
output [fu with crits 3d6 low] named "-2"
output [fu with crits 2d6 low] named "-1"
output [middle 1 of 3d6] named "0"
output [fu with crits 2d6 high] named "+1"
output [fu with crits 3d6 high] named "+2"
output [fu with crits 4d6 high] named "+3"
output [fu with crits 5d6 high] named "+4"

function: apoc A:n {
   if A >= 10 { result: 10 }
   if A >= 7 { result: 7 }
   result: 0
}
function: apockinda A:n plus B:n {
   X: A + B
   if X >= 18 { result: 10 }
   if X >= 11 { result: 7 }
   result: 0
}
function: apoctwenty A:n {
   if A >= 18 { result: 10 }
   if A >= 10 { result: 7 }
   result: 0
}

output [apoc 2d6] named "+0"
output [apoc 2d6+1] named "+1"
loop N over {2..10} {
  output [apoc [highest 2 of Nd6]] named "2d6 of [N] advantage"
  output [apoc 2d6+N] named "aw +[N]"
}

loop N over {1..5} {
  output [apoctwenty [highest 1 of Nd20]] named "adv [N]d20"
}

P: 4
loop N over {0..8} {
  output [apockinda 1d20 plus [highest 1 of NdP]] named "highest of [N]d[P]"
}

loop N over {2..4} {
  output [apockinda 1d20 plus [lowest 1 of NdP]] named "lowest of [N]d[P]"
}

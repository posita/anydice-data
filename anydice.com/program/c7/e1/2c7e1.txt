function: minor advantage A:s with B:s {
  \ Roll dice set A, replace lowest one with lowest die from dice set B, if it is to your advantage \
  BASE: {1..(#A-1)}@A
  ADD: [highest of (#A)@A and (#B)@B]
  result: BASE + ADD
}

function: major advantage A:s with B:s {
  \ Roll dice set A, replace lowest one with highest die from dice set B, if it is to your advantage \
  BASE: {1..(#A-1)}@A
  ADD: [highest of (#A)@A and 1@B]
  result: BASE + ADD
}

function: minor disadvantage A:s with B:s {
  \ Roll dice set A, replace highest one with highest die from dice set B, if it is to your disadvantage \
  BASE: {2..#A}@A
  ADD: [lowest of 1@A and 1@B]
  result: BASE + ADD
}

  BASE: {2..#A}@A
  ADD: [lowest of 1@A and (#B)@B]
  result: BASE + ADD
}

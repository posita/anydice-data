function: check mod M:n skill S:n adv A:n focus F:n {
 \ Permanent advantage \
 if M = 4 {
  M: 3
  A: A + 1
 }
 if M = 5 {
  M: 3
  A: A + 2
 }

 if S = 3 {
  S: 2
  A: A + 1
 }

 R: [highest (F+2) of (A+F+2)d6]

 \ No crits with focus \
 if F = 0 {
  result: R + M + S + [crit 2d6 mod M]
 } else {
  result: R + M + S
 }
}

function: crit D:s mod M:n {
 if 1@D = 2@D {
  result: 1d6
 }
 else { result: 0 }
}

output [check mod 0 skill 0 adv 0 focus 0]

function: is DIE:n lte LTEV:n {
  result: DIE <= LTEV
}

function: is DIE:n gte GTEV:n {
  result: DIE >= GTEV
}

function: is DIE:n gt GTV:n and lte LTEV:n {
  result: DIE > GTV & DIE <= LTEV
}

X: 2d6
RANGES: {7,11,15}

loop N over {1..#RANGES} {
  R: N @ RANGES

  if N = 1 {
    output [is X lte R] named "X <= [R]"
  }
  
  if N = #RANGES {
    output [is X gte R] named "X >= [R]"
  }
  else {
    NN: N + 1
    RR: NN @ RANGES
    output [is X gt R and lte RR] named "[R] <= X <= [RR]"
  }
}

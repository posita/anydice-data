function: apoc A:n {
   if A >= 10 { result: 10 }
   if A >= 7 { result: 7 }
   result: 0
}
function: apockinda A:n plus B:n {
   X: A + B
   if X >= 18 { result: 10 }
   if X >= 11 { result: 7 }
   result: 0
}
function: apoctwenty A:n {
   if A >= 18 { result: 10 }
   if A >= 10 { result: 7 }
   result: 0
}

output [apoc 2d6] named "2d6 (aw)"
output [apoctwenty 1d20] named "1d20 (aw)"
output [apockinda 1d20 plus 1d6] named "+1d6"

loop N over {2..4} {
  output [apockinda 1d20 plus [highest 1 of Nd6]] named "highest of [N]d6"
}

loop N over {2..4} {
  output [apockinda 1d20 plus [lowest 1 of Nd6]] named "lowest of [N]d6"
}

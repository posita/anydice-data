function: maxplode N:n {
 if N = 10 { result: 10 + [explode d10] }
 result: N
}

function: implode N:n
{
 if N = 2 {result: -10 + [implode 2d10]}
 result: N
}

set "maximum function depth" to 4

set "position order" to "lowest first"
loop N over {3}
{
D:N-2
R:Nd10
\*output [implode ([maxplode 1@R] + [maxplode 2@R])] named "Advantage -[D]"*\
}

R:2d10
loop N 
output [implode ([maxplode 1@R] + [maxplode 2@R])] < 20 named "Advantage 0"

set "position order" to "highest first"
loop N over {3}
{
D:N-2
R:Nd10
\*output [implode ([maxplode 1@R] + [maxplode 2@R])] named "Advantage +[D]"*\
}

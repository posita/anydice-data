loop N over {0..10}{
   if N < 4 {
   output 1d10-1d10+2*N
   } else if N < 7 {
   output [highest of 1d10-1d10+2*N and 1d10-1d10+2*N]
   } else if N < 10 {
   output [highest of 1d10-1d10+2*N and [highest of 1d10-1d10+2*N and 1d10-1d10+2*N]]
   } else {
   output [highest of 1d10-1d10+2*N and [highest of 1d10-1d10+2*N and [highest of 1d10-1d10+2*N and 1d10-1d10+2*N]]]
   }
}

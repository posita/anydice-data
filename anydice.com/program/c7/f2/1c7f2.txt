function: reroll DIE:n with NEW:n {
 ROLL: 0
   if DIE < 6 { ROLL: NEW } else { ROLL: DIE }
   if ROLL = 6 { result: 1 } else { result: 0 }
}

function: take DIE:n {
   if DIE = 6 { result: 1 } else { result: 0 }
}

function: fell N:n {
 END: 0
 A: 1@5d6
 B: 2@5d6
 C: 3@5d6
 D: 4@5d6
 E: 5@5d6

  if N  >= 5 { E: [reroll E with 1d6] } else { E: [take E] }
  if N  >= 4 { D: [reroll D with 1d6] } else { D: [take D] }
  if N  >= 3 { C: [reroll C with 1d6] } else { C: [take C] }
  if N  >= 2 { B: [reroll B with 1d6] } else { B: [take B] }
  if N  >= 1 { A: [reroll A with 1d6] } else { A: [take A] }

result: A+B+C+D+E
  
}

output [count 6 in 5d6]

loop N over {0..5}{
output [fell N] named "[N]"
}

function: rolla ROLL:s {

 if [count 6 in ROLL] >= 2 {
  result: 4
 }
 if 1@ROLL = 6 {
  result: 2 + ([count 1 in ROLL] <= 1)
 }

 if 1@ROLL >= 4 {
  result: 1 + ([count 1 in ROLL] <= 1)
 }
 
 if 1@ROLL >= 2 {
  result: 0 + ([count 1 in ROLL] <= 1)
 
 }
 
 result: 0
 
}

function: rollb ROLL:s {

 if 1@ROLL = 6 & 2@ROLL = 6  {
  result: 3 + ([count 1 in ROLL] <= 1)
 }
 if 1@ROLL = 6 {
  result: 2 + ([count 1 in ROLL] <= 1)
 }

 if 1@ROLL >= 4 {
  result: 1 + ([count 1 in ROLL] <= 1)
 }
 
 if 1@ROLL >= 2 {
  result: 0 + ([count 1 in ROLL] <= 1)
 
 }
 
 result: 0
 
}

function: match A:n and B:n {
   RES: 0
   if (A = B) {
     RES: A + B
   }

   result: RES
}

function: match A:n and B:n and C:n {
   RES: 0
   if ((A = B) & (B = C)) {
     RES: A + B + C
   }
   if ([match A and B] >= RES) {
      RES: [match A and B]
   }
   if ([match A and C] >= RES) {
      RES: [match A and C]
   }
   if ([match B and C] >= RES) {
      RES: [match B and C]
   }

   result: RES
}

function: match A:n and B:n and C:n and D:n {
   RES: 0
   if ((A = B) & (B = C) & (C = D)) {
       RES: A + B + C + D
   }
   if ([match A and B and C] >= RES) {
      RES: [match A and B and C]
   }
   if ([match A and B and D] >= RES) {
      RES: [match A and B and D]
   }
   if ([match A and C and D] >= RES) {
      RES: [match A and C and D]
   }
   if ([match B and C and D] >= RES) {
      RES: [match B and C and D]
   }
}

loop VAL over {1..20} {
   output [match d10 and d10] >= VAL named "match 2d10 > [VAL]"
}

\ Sentinel + Hill Giant Belt + GWF + GWM + GreatSword+1 \
AC: 15
GWF: 2d{{1..2}, {3..6}:4}   \This simulates 4D6 rerolling the first 1 or 2 of each d6 \
output 2d6 named "2d6"
output GWF named "2d6 with rerolls (1,2)"

NORMAL_HIT: (1d20+9) >= AC    \ 0 or 1 result.    5 for str, 3 for prof, 1 for sword \
GWM_HIT: (1d20+4) >= AC    \ 0 or 1 result.    5 for str, 3 for prof, 1 for sword, -5 for GWM \

NORMAL_HIT_ADV: ([highest 1 of 2d20]+9) >= AC    \ 0 or 1 result.    5 for str, 3 for prof, 1 for sword \
GWM_HIT_ADV: ([highest 1 of 2d20]+4) >= AC    \ 0 or 1 result.    5 for str, 3 for prof, 1 for sword, -5 for GWM \

NORMAL_DMG: GWF + 6   \ 5 for Str + 1 for sword \
GWM_DMG: GWF + 16   \ 10 from GWM + 5 for Str + 1 for sword \

output NORMAL_HIT named "Normal Swing - To Hit AC"
output (NORMAL_HIT* NORMAL_DMG) + (NORMAL_HIT* NORMAL_DMG) named "Normal Swing - Damage per Round"

output GWM_HIT named "GWM Swing - To Hit AC"
output (GWM_HIT * GWM_DMG) + (GWM_HIT * GWM_DMG) named "GWM Swing - Damage per Round"

output NORMAL_HIT_ADV named "Normal Swing (ADV) - To Hit AC"
output (NORMAL_HIT_ADV * NORMAL_DMG) + (NORMAL_HIT_ADV * NORMAL_DMG) named "Normal Swing (ADV) - Damage per Round"

output GWM_HIT_ADV named "GWM Swing (ADV)- To Hit AC"
output (GWM_HIT_ADV * GWM_DMG) + (GWM_HIT_ADV * GWM_DMG) named "GWM Swing (ADV) - Damage per Round"

function: A:n plus B tohit AC
{
 if A        = 20 {result: 2} \Crit result.\
 if A        =  1 {result: 0} \Crit Miss result.\
 if (A + B) >= AC {result: 1} \Meet or exceed AC, hit\
 if (A + B) <  AC {result: 0} \Fall short of AC, miss.\
}

TOHIT:   [1d20 plus 10 tohit 10] \Let's define a test AC.\
PLUSTWO: [1d20 plus 12 tohit 10] \Elemental Weapon gains a +2 to hit.\

ATTACK: 1d10 + 12 \Glaive\
POLEARM: 1d4 + 12 \Polearm master bonus action attack\

EWEAP: 2d4 \Elemental Weapon level 5\
SMITE: 6d8 \Eldritch Smite   level 5\

CANTRIP: 2d8 \Booming Blade and GFlame blade have the same damage output if neither of them trigger their additional effect.\
BBLADE:  3d8 \If the target moves.\
GBLADE:  2d8 + 6 \If there is a second available target.\

output (TOHIT * ATTACK) + (TOHIT * ATTACK) + (TOHIT * POLEARM) named "Basic Attacks"
\output (PLUSTWO * (ATTACK + EWEAP)) + (PLUSTWO * (ATTACK + EWEAP)) + (PLUSTWO * (POLEARM + EWEAP)) named "Elemental Weapon"\
\output (PLUSTWO * (ATTACK + EWEAP + SMITE)) + (PLUSTWO * (ATTACK + EWEAP + SMITE)) + (PLUSTWO * (POLEARM + EWEAP + SMITE)) named "EWeap + Triple Smite"\
output (TOHIT * (ATTACK + CANTRIP)) named "Booming-GFlame Blade"
output (TOHIT * (ATTACK + CANTRIP+ BBLADE)) named "Booming Blade + Move"
output (TOHIT * (ATTACK + CANTRIP+ GBLADE)) named "GFlame Blade + Target"

BLUE:0
RED:1
function: X:n ATT:n vs DEF:n{
\ BvB \
if ATT+DEF= 0  {
if X <= 4 {result: 1}
}
\ BvR, RvB \
if ATT+DEF= 1  {
if X <= 2 {result: 1}
}
\ RvR \
if ATT+DEF= 2 {
if X = 8 {result: 2}
if X >= 5 {result: 1}
}
result: 0
}

output 3d(d10 >= 3) named "Org d10"

output 3d[d8 RED vs RED] named "3R RvR"
output 3d[d8 RED vs BLUE] named "3R RvB"

output 2d[d8 RED vs RED] + 1d[d8 BLUE vs RED] named "2R1B RvR"
output 2d[d8 RED vs BLUE] + 1d[d8 BLUE vs BLUE] named "2R1B RvB"

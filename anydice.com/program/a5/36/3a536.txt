function: strength E:n of A:n B:n C:n D:n {
if E = 1 {result: {1,2} @ [sort{A,B,C}]}
if E = 2 {result: {1,2} @ [sort{A,B,C,D}]}
}

function: weakness E:n of A:n B:n C:n D:n {
if E = 1 {result: {2,3} @ [sort{A,B,C}]}
if E = 2 {result: {3,4} @ [sort{A,B,C,D}]}
}

function: diceoption A:n {
DICE: [absolute A]
if A = 0 {result: 1d12+1d8}
if A > 0 {result: [strength DICE of 1d12 1d8 1d10 1d10]}
if A < 0 {result: [weakness DICE of 1d12 1d8 1d10 1d10]}
}



output [diceoption 2] named "2 Strengths"
output [diceoption 1] named "1 Strengths"
output [diceoption 0] named "Base Roll"
output [diceoption -1] named "1 Weakness"
output [diceoption -2] named "2 Weakness"

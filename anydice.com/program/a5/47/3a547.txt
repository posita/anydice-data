function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: vexattack TIMES:n times ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if TIMES = 0 {
        result: 0
    }
    if (ROLL = 1) | ((ROLL < 20) & (ROLL + BONUS < AC)) {
        result: [vexattack TIMES - 1 times 1d20 plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE]
    }
    else {
        if ROLL < CRIT_RANGE {
            DAMAGE: DMG
        }
        else {
            DAMAGE: DMG + CRIT_DMG
        }
        result: DAMAGE + [vexattack TIMES - 1 times [highest 1 of 2d20] plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE]
    }
}

function: effect VALUE:n skill SKILL:n ALPHA:n BRAVO:n{
 MULTIPLE: VALUE-10
if MULTIPLE <0 { MULTIPLE:0}
 if ALPHA + BRAVO <= SKILL {
   DAMAGE: ALPHA+MULTIPLE
      if DAMAGE >VALUE {DAMAGE:VALUE}
      if ALPHA=BRAVO {result: DAMAGE+VALUE}
    result: DAMAGE}
 result: 0
}
loop N over {1..10}{
output [effect N skill 12 1d10 1d10] named "Stat [N] Skill 12" }

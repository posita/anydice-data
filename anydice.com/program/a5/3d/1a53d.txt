M:5
S:6
F:1
D:{M+S+F}d6
function: is V:d success of T:n{
   result: V>=T
}
function: attack{
   C:0
   loop N over {1..M+S+F}{
      C: C+[is N@D success of 3]
   }
   result: C
}
function: A:n ge B:n{
   if(A){result:B}
   result:0
}
function: save{
   C:0
   A:(M-1)
   B:(M-S)+A
   if(B<A){B:A}
   loop N over {1..M+S+F}{
      C: C+[is N@D success of 4]}
   B:[attack]>=B
   C:C>=A
   result:[B ge C]
}
output [save]
output [attack]

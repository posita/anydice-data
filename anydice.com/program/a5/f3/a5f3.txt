function: attk ROLL:n ROLLTWO:n vs ac AC:n with tohit BAB:n str STR:n dmg DMG:n {
  if (ROLL = 20) {
    if (ROLLTWO + BAB + STR > AC) {
      result: DMG * 2
    }
  }

  if (ROLL + BAB + STR > AC) {
    result: DMG
  }
  result: 0
}

BAB: 3
STR: 4
AC: 18

DMGPNSHSTANCE: 1d6
DMGPRIM: 1d4+STR+1+DMGPNSHSTANCE
DMGCLAW: 1d4+STR/2+1+DMGPNSHSTANCE
DMGWYRM: 1d6+STR/2+1+DMGPNSHSTANCE

output (
  [attk 1d20 1d20 vs ac AC with tohit BAB str STR dmg DMGPRIM]
  + [attk 1d20 1d20 vs ac AC with tohit BAB str STR dmg DMGCLAW]
  + [attk 1d20 1d20 vs ac AC with tohit BAB str STR dmg DMGCLAW]
  + [attk 1d20 1d20 vs ac AC with tohit BAB str STR dmg DMGCLAW]
)

function: minimum of DIE:d {
 result: 1@{DIE}
}

function: aroll regular double N:n M:n O:n {
 result: N + M + O
}

function: aroll DIE:d N:n M:n helper {
 if N = M { 
  if N=1 { result: 0 }
  if N=2 { result: 111 }
  if N=3 { result: 4 }
  if N=4 { result: 6 }
  result: 100
 }
 SUM: N+M
 if SUM={3,4} {result: 1}
 if SUM=5 { result: 2}
 if SUM={6,7} { result: 3 }

 result: N + M
}

function: aroll DIE:d {
 MAX: [maximum of DIE + DIE]
 MIN: [minimum of DIE + DIE]
 result: [aroll DIE DIE DIE helper]
}

function: maybe add level N:n LEVEL:n {
 if N = 0 { result: 0 }
 if N = 111 { result: LEVEL }
 result: N + LEVEL
}

function: aroll DIE:d at LEVEL:n {
 result: [maybe add level [aroll DIE] LEVEL]
}

function: repeat ROLL:d N:n times {
 if N=1 { result: ROLL }
 result: ROLL + [repeat ROLL N-1 times]
}

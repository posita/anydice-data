LUCK: {-1..1}
function: roll with SKILL:s vs DIFF:n {
   result: [highest 1 of 2d6] - [highest 1 of 2d6] + SKILL - DIFF
  }

SKILLS: {-2..8}
DIFFS:{0}

loop SKILL over SKILLS {
 loop DIFF over DIFFS {
  output [roll with SKILL vs DIFF] named "[SKILL] vs dif.[DIFF]"
 }
}

loop SKILL over {-2..8} {
 output SKILL + 4d{-1,-1,0,0,1,1} named "FATE [SKILL]"
}

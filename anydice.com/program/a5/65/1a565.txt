function: DIE:n hits TARGET:n {
  VAL: 0
  if DIE >= TARGET { VAL: VAL + 1 }
  if (DIE >= (TARGET + 10)) | (DIE = 20) { VAL: VAL + 1 }
  result: VAL
}

function: attackreal ATTACK:n AC:n DAMAGE:d DEADLY:d {
  HITS: [ATTACK hits AC]
  TOTAL: (HITS * DAMAGE) + ([HITS hits 2] * DEADLY)  
  result: TOTAL
}

function: attack AC:n for DAMAGE:d deadly DEADLY:d {
  result: [attackreal 1d20 AC DAMAGE DEADLY]
}

function: explode N:n {
 
 if N >= THRESHOLD { result: N + [explode d{2,3,4,5,6}] }
 result: N
}

 output [explode d{2,3,4,5,6}] + [explode d{2,3,4,5,6}] named "explode on at least [THRESHOLD]"
}

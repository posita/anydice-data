function: attk ROLL:n ROLLTWO:n vs ac AC:n with tohit BAB:n str STR:n dmg DMG:n {
  if (ROLL = 20) {
    if (ROLLTWO + BAB + STR > AC) {
      result: DMG * 2
    }
  }

  if (ROLL + BAB + STR > AC) {
    result: DMG
  }
  result: 0
}

BAB: 4
STR: 7
AC: 18

DMGBITE: 1d6+STR
DMGCLAW: 1d3+STR/2

CLAWBAB: BAB - 5

output (
  [attk 1d20 1d20 vs ac AC with tohit BAB str STR dmg DMGBITE]
  + [attk 1d20 1d20 vs ac AC with tohit CLAWBAB str STR dmg DMGCLAW]
  + [attk 1d20 1d20 vs ac AC with tohit CLAWBAB str STR dmg DMGCLAW]
)

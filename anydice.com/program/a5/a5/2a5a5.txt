function: ROLL:n plus MOD:n to hit against AC:n dealing DICE:d plus EXTRA:n{
 if ROLL = 20 {
  result: DICE + DICE + EXTRA
 }
 else {
  if ROLL = 1 {
   result: 0
  }
  else {
   result: (ROLL+MOD >= AC) * (DICE + EXTRA)
  }
 }
}

function: ROLLMOD:n against SAVEDC:n dealing DAMAGE:n against N:n targets{
 TOTAL: 0
 loop D over {1..N} {
  TOTAL: TOTAL + (1d20 + ROLLMOD < SAVEDC) * (DAMAGE)
 }
 result: TOTAL
}

function: ROLLMOD:n against SAVEDC:n dealing DAMAGE:n against N:n targets half on save{
 TOTAL: 0
 loop D over {1..N} {
  TOTAL: TOTAL + (((1d20 + ROLLMOD < SAVEDC) * (DAMAGE)/2) + DAMAGE/2)
 }
 result: TOTAL
}

function: enhanced bond applied to N:n
{
 if (N > 0)
 {
  N: N + 1d8
 }
}

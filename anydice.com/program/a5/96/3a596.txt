function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n hit DMG:d miss MISS:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: MISS
}

function: vextwo ROLL:d baseroll BASE_ROLL:d plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS vs AC 
        hit  [attack [highest 1 of 2dBASE_ROLL] plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE] + DMG
        miss [attack                 BASE_ROLL  plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE]
        crit [attack [highest 1 of 2dBASE_ROLL] plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE] + DMG + CRIT_DMG
        on CRIT_RANGE]
}

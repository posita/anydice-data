function: action ACTROLL:n ACTMOD:n CHALROLL:s {
  MAX_RESULT: 20
  result: [lowest of (ACTROLL + ACTMOD) and MAX_RESULT] > CHALROLL
}

CHALDIE: 2d20
MIN_BONUS: 0
MAX_BONUS: 8

loop B over {MIN_BONUS..MAX_BONUS} {
  output [action 1d4 B CHALDIE] named "Action 1d4+[B] vs [CHALDIE] Challenge"
}

loop B over {MIN_BONUS..MAX_BONUS} {
  output [action 1d6 B CHALDIE] named "Action 1d6+[B] vs [CHALDIE] Challenge"
}

loop B over {MIN_BONUS..MAX_BONUS} {
  output [action 1d8 B CHALDIE] named "Action 1d8+[B] vs [CHALDIE] Challenge"
}

loop B over {MIN_BONUS..MAX_BONUS} {
  output [action 1d10 B CHALDIE] named "Action 1d10+[B] vs [CHALDIE] Challenge"
}

loop B over {MIN_BONUS..MAX_BONUS} {
  output [action 1d12 B CHALDIE] named "Action 1d12+[B] vs [CHALDIE] Challenge"
}

MISS: 0
HIT: 1
CRIT: 2

function: attack ROLL:n plus ATKBONUS:n vs DEFENSE:n {
 if ROLL + ATKBONUS >= DEFENSE {
  if ROLL = 20 { result: CRIT }
  result: HIT
 }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRIT {
  result: CRITDMG
 }
 if ATTACKRESULT = HIT {
  result: HITDMG
 }
 result: MISSDMG
}

\ Scimitar Booming Blade \
CRITDMG: 1d6 + 1d8 + 15
HITDMG:  1d6 + 1d8 + 1
MISSDMG: 0
output [damage for [attack d20 plus 3 vs 15]] named "Scimitar Booming Blade"

CRITDMG: 2d8 + 16
HITDMG:  2d8
MISSDMG: 0

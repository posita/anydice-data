function: R:n die D:n {
  X:0
  if D=8 {X: 1}
  if D=9 {X: 1}
  if D=10 {X: 1}
  if D>=R {X: X + [R die d10]}
  result: X
} 

function: R:n rotedie D:n {
  if D=1 {X: [R rerolldie d10]}
  if D=2 {X: [R rerolldie d10]}
  if D=3 {X: [R rerolldie d10]}
  if D=4 {X: [R rerolldie d10]}
  if D=5 {X: [R rerolldie d10]}
  if D=6 {X: [R rerolldie d10]}
  if D=7 {X: [R rerolldie d10]}
  if D=8 {X: 1}
  if D=9 {X: 1}
  if D=10 {X: 1}
  if D>=R {X: X + [R rotedie d10]}
  result: X
} 

function: R:n rerolldie D:n {
  X:0
  if D=8 {X: 1}
  if D=9 {X: 1}
  if D=10 {X: 1}
  if D>=R {X: X + [R rerolldie d10]}
  result: X
}

function: R:n roteskill L:n {
  T: 0
  loop N over {1 .. L}{
    T: T+[R rotedie d10]
  }
  result: T
}

function: R:n skill L:n {
  T: 0
  loop N over {1 .. L}{
    T: T+[R die d10]
  }
  result: T
}

set "maximum function depth" to 4

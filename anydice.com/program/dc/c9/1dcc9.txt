function: progress DICE:s {
  if DICE <= 1 {
    result: 0 \ Absolute Failure \
  }
  if DICE <= 8 {
    result: 1 \ No, and \
  }
  if DICE <= 11 {
    result: 2 \ No \
  }
  if DICE <= 14 {
    result: 3 \ No, unless \
  }
  if DICE <= 18 {
    result: 4 \ Yes \
  }
  if DICE <= 19 {
    result: 5 \ Yes, and \
  }
  if DICE <= 20 {
    result: 6 \ Ultimate Success \
  }
}

output [progress 1@1d20] named "1d20"

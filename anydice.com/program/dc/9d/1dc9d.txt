\ roll AdX + BdY, target 4, where X is one size less than Y and A+B = 6 \

DICE: {6,10}

loop I over {2..#DICE} {
  X: (I-1)@DICE
  Y: I@DICE

  output 3d(dX >= 4) named "3d[X]"

  loop B over {1..3} {
    A: 3-B
    output Ad(dX >= 6) + Bd(dY >= 6) named "[A]d[X] and [B]d[Y]"
  }
}
output 3d(dY >= 6) named "3d[Y]"

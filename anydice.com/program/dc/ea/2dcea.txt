\**** Damage calculator Assassin Rogue ****\

function: attackmain ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: ((ROLL + ATTACKBONUS - DEF) *2 + BONUSDMG + WPNDAMAGEA)*2 + CINDERB} 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS - 0 >= DEF { result: (ROLL + ATTACKBONUS - DEF) *2 + (BONUSDMG) + WPNDAMAGEA + CINDERB}
 result: 0
}

function: dualwield ROLL:n vs DEF:n {
 if ROLL >= CRITROLL { result: WPNDAMAGEB*2 + CINDERB} 
 if ROLL <= 2 { result: 0 }
 if ROLL + ATTACKBONUS >= DEF { result: WPNDAMAGEB + CINDERB}
 result: 0
}

\looped against enemy DEF\
ENEMYDEF: d{10}

ATTACKBONUS: 5
BONUSDMG: 0
CRITROLL: 12
CINDERB: 0
WPNDAMAGEA: 12
WPNDAMAGEB: 12
COMBOTIMES: 1
output [attackmain 2d6 vs ENEMYDEF] + ([dualwield 2d6 vs ENEMYDEF] * COMBOTIMES) named "Trickster Rogue Dual Wield"

output 2d6

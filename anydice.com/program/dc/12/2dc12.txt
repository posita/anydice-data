DMG_MOD: 3
TO_HIT: 5
CRIT: 20

function: maul attack ROLL:n {
DMG_DIE: 1d6
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*([gwf DMG_DIE] + [gwf DMG_DIE]) + DMG_MOD}
if ROLL + TO_HIT >= AC {result: [gwf DMG_DIE] + [gwf DMG_DIE] + DMG_MOD}
}

function: greataxe attack ROLL:n {
DMG_DIE: 1d12
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*[gwf DMG_DIE] + DMG_MOD}
if ROLL + TO_HIT >= AC {result: [gwf DMG_DIE] + DMG_MOD}
}

function: gwf ROLL:n {
if ROLL <= 2  {result: DMG_DIE}
if ROLL >= 3  {result: ROLL}
}

loop AC over {8..23} {
 output [maul attack d20] named "Maul vs [AC] AC"
 output [greataxe attack d20] named "Greataxe vs [AC] AC" 
}

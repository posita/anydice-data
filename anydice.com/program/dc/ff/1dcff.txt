ACTIONMOD: 0
CHALLENGEMOD: 0
ACTION: 1d20 + ACTIONMOD
CHALLENGEA: 1d20 + CHALLENGEMOD
CHALLENGEB: 1d20 + CHALLENGEMOD

output
  (ACTION > CHALLENGEA & ACTION <= CHALLENGEB) |
  (ACTION <= CHALLENGEA & ACTION > CHALLENGEB)
named "Weak Hit"
output
  ACTION > CHALLENGEA & ACTION > CHALLENGEB
named "Strong Hit"
output
  ((ACTION > CHALLENGEA & ACTION <= CHALLENGEB) |
  (ACTION <= CHALLENGEA & ACTION > CHALLENGEB)) |
  (ACTION > CHALLENGEA & ACTION > CHALLENGEB)
named "Hit"
output
  ACTION <= CHALLENGEA & ACTION <= CHALLENGEB
named "Miss"

function: results ROLL:n {
    if ROLL <= 9 {result: 1}
    if ROLL >= 9 & ROLL <= 13 {result: 2}
    if ROLL >= 14 & ROLL <= 22 {result: 4}
    if ROLL >= 23 & ROLL <= 26 {result: 3}
    result: 5
}

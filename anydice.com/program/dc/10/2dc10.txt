DMG_DICE: 1d12
DMG_MOD: 3
TO_HIT: 5
CRIT: 20

function: maul attack ROLL:n {
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*([gwf 1d6] + [gwf 1d6]) + DMG_MOD}
if ROLL + TO_HIT >= AC {result: [gwf 1d6] + DMG_MOD}
}

function: greataxe attack ROLL:n {
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*[gwf 1d12] + DMG_MOD}
if ROLL + TO_HIT >= AC {result: [gwf 1d12] + DMG_MOD}
}

function: gwf ROLL:n {
if ROLL <= 2  {result: 1d12}
if ROLL >= 3  {result: ROLL}
}

loop AC over {8..23} {
 output [maul attack d20] named "Maul vs [AC] AC"
 output [greataxe attack d20] named "Greataxe vs [AC] AC" 
}

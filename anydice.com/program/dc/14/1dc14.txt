function: action ACTROLL:n ACTMOD:n CHALROLL:s {
  result: [lowest of (ACTROLL + ACTMOD) and 10] > CHALROLL
}

  output [action 1d4 B 2d20] named "Action 1d4+[B] vs 2d20 Challenge"
}

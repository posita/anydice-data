function: reduce RESULT:n by TARGETS:s {
  TARGET_RESULT: 0
  loop N over TARGETS{
    if RESULT < N{result: TARGET_RESULT}
    TARGET_RESULT: N
}
}

output [reduce 1d20 by {5, 10, 15, 20, 25, 30}]
output [reduce 1d20 +3 by {5, 10, 15, 20, 25, 30}]
output [reduce 1d20 +5 by {5, 10, 15, 20, 25, 30}]

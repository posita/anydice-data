function: if CONDITIONA:n then A else if CONDITIONB:n then B else C {
  if CONDITIONA { result: A } else if CONDITIONB { result: B } else {result: C}
}

TARGET:6
DICE: 10

loop N over {3..6}{
loop M over {N..6}{
loop O over {M..6}{

if N < 4 {
 I: 0
} else {
 I: N*2
}

if M < 4 {
 J: 0
} else {
 J: M*2
}

if O < 4 {
 K: 0
} else {
 K: O*2
}

ROLL:  [count {TARGET..12} in 2dDICE] + [count {TARGET..12} in 1dI] + [count {TARGET..12} in 1dJ] + [count {TARGET..12} in 1dK]
FOO: [if ROLL < 2 then 0 else if ROLL > 3 then 2 else 1]

output FOO  named "d[I]+d[J]+d[K]"
}
}
}

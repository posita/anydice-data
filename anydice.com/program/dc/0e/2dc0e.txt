function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE+STRENGTH }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE+STRENGTH
 }
 result: DAMAGE+STRENGTH
}

AC: 24
STRENGTH: 4
ATTACK: 4 + STRENGTH

DAMAGE: 1d8
THREAT: 19
CRITICAL: 2
output [d20 vs AC] named "Espada Longa"

\output d20 + 5 named "AC"\
\output (1d10+3)*100/100  named "av dam"\
\loop N over {1..20}{
 output (1d10+3)*[armor chance]
 output d20 + 5 named "armor chance"
}\
MISS: 0
HIT: 1
CRITICALHIT: 2
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 if ROLL + ATTACK <= DEFENSE { result: MISS }
}

loop AC over {10..25} {
 ATTACK: 0
 output [attack d20 vs AC] named "[AC]"
}

loop AC over {10..25} {
 ATTACK: 5
 output [attack d20 vs AC] named "[AC]"
}

function: X:n to tier {
  if X < 12 {
    result: 1
  }
  if X < 17 {
    result: 2
  }
  result: 3
}

function: power roll plus X:n {
  SUM: 2d10 + X
  result: [SUM to tier]
}

loop X over {-5..-1} {
  ABS: 0 - X
  output [power roll plus X] named "Power Roll - [ABS]"
}
loop X over {0..5} {
  output [power roll plus X] named "Power Roll + [X]"
}

function: fifth edition attack ROLL:n {
if ROLL = 1  {result: 0}
if ROLL >= CRIT {result: 2*[gwm DMG_DICE] + DMG_MOD}
if ROLL + TO_HIT >= AC {result: DMG_DICE + DMG_MOD}
}

function: gwm ROLL:n {
if ROLL <= 2  {result: 1d12}
if ROLL >= 3  {result: ROLL}
}

DMG_DICE: 1d12
DMG_MOD: 3
TO_HIT: 5
CRIT: 20

output [gwm DMG_DICE] named "GWM"

loop AC over {8..23} {
 output [fifth edition attack d20] named "[AC] AC"
}

function: ROLL:n strength STR:n damage DMG:n minimum OVW:n vs defense DEFENSE:n and soak SOAK:n {

 if ROLL >= DEFENSE { 
  EXTRA: ROLL - DEFENSE
  result: [highest of (STR + DMG + EXTRA - SOAK) and OVW]d{0,0,0,0,0,0,1,1,1,2}
 }
}

COOPERATIVESTRIKE: 0
DOUBLEDAMAGE: 1
SHOWRAISES: 0
SMITE: 1
function: attack{ 
TOHIT :[highest of d8 and d6]+6
RAISES: [highest of TOHIT -4 and 0]/4
if(COOPERATIVESTRIKE = 0){RAISES: [lowest of 1 and RAISES]}
DAMAGE: 2d6 + RAISES * 1d6
if SMITE > 0 {DAMAGE: DAMAGE + 4}
if DOUBLEDAMAGE > 0 {DAMAGE: 2 * DAMAGE}
if SHOWRAISES > 0 {result: RAISES }
result: DAMAGE
}
output [highest of [attack] and 0]

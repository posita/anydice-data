DICEFIRST:2
DICESECOND:3
BON:8

function: roll A:n B:n{
ATEN:0
BTEN:0
if A=20 {ATEN:1}
if B=20 {BTEN:1}

if A>B+BON {result:2}
if A<B+BON & ATEN=1 & BTEN=0 {result:1}
if A=B+BON {result:1}
result:0
}

set "explode depth" to 1
output [highest 2 of DICEFIRSTd [explode d10]]>= [highest 2 of DICESECONDd [explode d10]]+BON named " explosive [DICEFIRST]d vs [DICESECOND]d + [BON] "

output [highest 2 of DICEFIRST d10]>= [highest 2 of DICESECONDd10]+BON named " basic [DICEFIRST]d vs [DICESECOND]d + [BON] "

output [roll [highest 2 of DICEFIRSTd10] [highest 2 of DICESECONDd10]] named " ten crit [DICEFIRST]d vs [DICESECOND]d + [BON] "

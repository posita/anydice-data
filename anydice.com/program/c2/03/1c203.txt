function: partial explode DICE:d depth DEPTH:n{
 OLD: d{0}
 ADD: d{1}
 TEMP: d{0}
 PERM: d{1}
 
 loop N over {1..DEPTH}{
  loop M over {1..N}{
   OLD: ADD
   ADD: OLD * DICE
  }
  TEMP: PERM
  PERM: TEMP + ADD
 }

result: PERM
}
D:6
loop Q over{1..D}{
 loop R over{1..D}{
  if (R!=Q | (R=1 & Q=1)){
   output [partial explode d{{0}:Q,{1}:R} depth 20]
  }
 }
}

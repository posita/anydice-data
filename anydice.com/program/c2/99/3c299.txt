function: halve SEQUENCE:n {                  \ divide per 2, rounded up \
 DIV: SEQUENCE/2
 TEMP: DIV*2
 if (DIV=TEMP){result: DIV}
 result: (SEQUENCE+1)/2
}

\************************************************************************\

function: phaseone ATT:s DEF:s {
  SIZE_MODS: [compare 11@ATT with 11@DEF]                 \compare the nbr of bodies\
  ATT_ROLL: (1@ATT +SIZE_MODS) dQ + 4@ATT                      \freeze the hits-dices sequence\

  VELOCITY_MODS: [compare 10@DEF with 9@ATT]             \compare the velocity for the dodge capacity\
  DEF_ROLL: (5@DEF +VELOCITY_MODS) dQ + 8@DEF

  result: [phasetwo ATT_ROLL DEF_ROLL ATT DEF]
}


function: phasetwo ATT_ROLL:n DEF_ROLL:n  ATT:s DEF:s {
  HITS: [crop ATT_ROLL 2@ATT 3@ATT]
  SAVES: [crop DEF_ROLL 6@DEF 7@DEF]
  HIT_SETS: [sets in ATT_ROLL]
  SAVE_SETS: [sets in DEF_ROLL]

  \DAMAGES: HITS - SAVES\
  DAMAGES: [halve HITS] - [halve SAVES]
  result: [crop DAMAGES 0 4]
}

function: sets in DICES:s {
  ZEROS:  [count 0 in DICES]
  ONES:   [count 1 in DICES]
  TWOS:   [count 2 in DICES]
  THREES: [count 3 in DICES]
  SETS: {ZEROS,ONES,TWOS,THREES}
  result: SETS
}
function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  if(VAL_X < VAL_Y) { result: -1}
  result: 0
}

\************************************************************************\

function: hit ATTACKS:n {
  DICE: ATTACKS dQ                       \freeze the hits-dices sequence\
  result: [hit roll DICE]
}

function: hit roll DICES:s {
  HITS: [count {WS..3} in DICES]
  ZEROS:  [count 0 in DICES]
  ONES:   [count 1 in DICES]
  TWOS:   [count 2 in DICES]
  THREES: [count 3 in DICES]
  SETS: {ZEROS,ONES,TWOS,THREES}
  result: [save DICES with SETS]
}

function: save HITS:n with SETS:s {
  DICES: RD dQ                              \freeze the saves-dices sequence\
  result: [damages DICES with SETS]
}

function: damages POOL:s with THRESHOLDS:s {
  if(POOL >= 4@THRESHOLDS) { result: 4}
  if(POOL >= 3@THRESHOLDS) { result: 3}
  if(POOL >= 2@THRESHOLDS) { result: 2}
  if(POOL >= 1@THRESHOLDS) { result: 1}
  result: 0
}


\************************************************************************\
function: finaldamages THRESHOLDS:s from POOL:n{
  if(POOL >= 4@THRESHOLDS) { result: 4}
  if(POOL >= 3@THRESHOLDS) { result: 3}
  if(POOL >= 2@THRESHOLDS) { result: 2}
  if(POOL >= 1@THRESHOLDS) { result: 1}
  result: 0
}
function: test PDICES:s {
  RDICES: RD dQ
  HITS: [finaldamages PD_THRESHOLDS from PDICES]
  SAVES: [finaldamages RD_THRESHOLDS from RDICES]
  result: HITS - SAVES
}

function: vert T:s G:n S:n {
 WIN: 0
 if #T > 1 {
  if (#T-1)@T <= G {result: -1}
 }
 loop N over {1..#T}{
  if N@T <= S {WIN: WIN + 1}
 }
 result: WIN
}
function: verp P:s T:s G:n S:n {
 WIN: 0
 if #T > 0 {
  if #T@T <= G {result: -1}
 }
 if P <= S {WIN: WIN + 1}
 loop N over {1..#T}{
  if N@T <= S {WIN: WIN + 1}
 }
 result: WIN
}

output [vert 3d12 1 8]
output [verp 1d12 2d12 1 8]

STR: 7
DEX: 5
CON: 5

PRF: 4
MBF: (STR + 1) / 2

QS: 0
US: 1
SD: 2
UD: 3

QSD: 1d8 + STR + MBF + 2
USD: 1d8 + STR + CON
SDD: 1d8 + 2d4 + DEX
UDD: 1d8 + DEX + CON

QSA: PRF + STR + 2
USA: PRF + STR
SDA: PRF + DEX + 1
UDA: PRF + DEX

function: real damage WPN vs AC {
  WDMG: 0
  HDMG: 0
  WATK: 0
  HATK: 0

  if WPN = QS | WPN = US {
    WDMG: USD
    HDMG: USD
    WATK: USA
    HATK: USA
    if WPN = QS {
      WDMG: QSD
      WATK: QSA
    }
  }

  if WPN = SD | WPN = UD {
    WDMG: UDD
    HDMG: UDD
    WATK: UDA
    HATK: UDA
    if WPN = SD {
      WDMG: SDD
      WATK: SDA
    }
  }

  result: WDMG + WDMG + HDMG + HDMG
}

output [real damage QS vs 17]
output [real damage SD vs 17]

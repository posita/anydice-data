
function: hp MISS:n to GLANCE:n to FULL:n to VITAL:n on ROLL:n {
if [{-100..MISS} contains ROLL]{ result:0}
if [{MISS+1..GLANCE} contains ROLL]{ result:1}
if [{GLANCE+1..FULL} contains ROLL]{ result:2}
if [{FULL+1..VITAL} contains ROLL]{ result:3}
else { result:4}
}

output [hp 30 to 60 to 90 to 120 on 1d100+30]

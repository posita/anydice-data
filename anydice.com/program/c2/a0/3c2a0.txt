  AC: 1+ARM+WT+ADD+SHIELD
  WPN: HAND*([maximum of WEAP]-4)/2
    if WPN > 0 {AC: AC + PROF + WPN + !F*STR + F*DEX}
  DIF: STR-WT
    if DIF < -1 {AC: AC - DIF}
    if DIF <  0 {AC: AC + DEX/2}
    if DIF =  0 {AC: AC + DEX}
    if DIF >  0 {AC: AC + DEX + [lowest of DEX and DIF]}
  result: AC
}

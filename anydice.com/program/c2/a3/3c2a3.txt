\\\*** Base Damage Calc Info ***\\\
ENEMY_AC: d{14..18} \looped against enemy AC from 14-18, DMG suggested AC from CR4-16\

\\\"0" = off, "1" = on\\\
SHOW_DIS: 0
SHOW_ADV: 0

\\\*** Attack A ***\\\
TO_HIT_A: 10
ATTACK_DICE_A: 4d12
ATTACK_DAM_A: 7
N_ATTACKS_A: 1

\\\*** Attack B ***\\\
TO_HIT_B: 10
ATTACK_DICE_B: 4d8
ATTACK_DAM_B: 7
N_ATTACKS_B: 1

\\\*** General Info ***\\\
CRIT_RANGE: 20
CRIT_BONUS: 0 \Any additional dice used specifically for a critical\
NORMAL_HIT: d20
DIS_ADV: [lowest 1 of 2d20]
HIT_ADV: [highest 1 of 2d20]


\\\*** Functions ***\\\
function: attack ROLL:n vs ENEMY_AC:n {
 if ROLL = 1 { result: 0 }
 if ROLL >= CRIT_RANGE { result: CRITICALDAMAGE }
 if ROLL + ATTACK >= ENEMY_AC { result: HITDAMAGE }
 result: 0
}

function: single ROLL:n vs ENEMY_AC:n {
 if ROLL = 1 { result: 0 }
 if ROLL + ATTACK >= ENEMY_AC { result: SINGLEDAMAGE }
 result: 0
}

HITDAMAGE: ATTACK_DICE_A + ATTACK_DAM_A
CRITICALDAMAGE: 2dATTACK_DICE_A + ATTACK_DAM_A
if SHOW_DIS = 1 {
output N_ATTACKS_Ad[attack DIS_ADV vs ENEMY_AC] named "Adv Atk A" }
output N_ATTACKS_Ad[attack NORMAL_HIT vs ENEMY_AC] named "Atk A"
if SHOW_ADV = 1 {
output N_ATTACKS_Ad[attack HIT_ADV vs ENEMY_AC] named "Adv Atk A"}

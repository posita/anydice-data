MAX: 20
function: arkhandie THR:n ROLL:n {
 if THR <= 1 {
  result: 100
 }
 if (ROLL >= THR) {
  result: 1+[arkhandie (THR-1) (d6)]
 }
 result: 0
}

function: limit V:n {
 if V > MAX {
  result: MAX
 }
 result: V
}

 }

 result: [limit TOTAL]
}

\ Do not touch this defines please \
NO:0
YES:1
\ DEAR USER: ENTER DATA BELOW \
\ attacker: \
ATTACKS:2
BS: 1 \ enter 2 for BS 2+ \
S: 4
AP: 0
DAMAGE: 3

REROLLS_ONES_TO_HIT: NO
REROLLS_FAILED_TO_HIT: NO
REROLLS_ONES_TO_WOUND: NO
REROLLS_FAILED_TO_WOUND: NO

\ target: \
T: 4
W: 3
REMAINING_WOUNDS: 1 \ Enter "W" if all models in unit have full wound pools. If opposite is true, enter wound pool of wounded one \
SAVE: 7 \ enter 2 for save 2+, and\

QUANTITY:1

\ No need to go below this line. Just press "Calculate" \
set "maximum function depth" to ATTACKS+6

function: X:n save{
  if X >= SAVE-AP {result: 0}
  else { result: 1}
}

function: wounding on{
  if 2*T <= S {result:2}
  if 2*S <= T {result:6}
  if S > T {result:3}
  if T > S {result:5}
  result:4
}
WOUNDING_ON:[wounding on]

function: X:n wound without reroll{
  if X < WOUNDING_ON{result: 0}
  else {
    result: [d6 save]
  }
}

function: X:n wound{
  if X < WOUNDING_ON{
    if REROLLS_ONES_TO_WOUND & X=1{
      result: [d6 wound without reroll]}
    else{
      if REROLLS_FAILED_TO_WOUND{
        result: [d6 wound without reroll]
      }
      else {result: 0}
    }
  }
  else{
    result: [d6 save]
  }
}

function: X:n hit without reroll{
  if X < BS{result: 0}
  else 
  {
    result: [d6 wound]
  }
}

function: X:n hit{
  if X < BS{
    if REROLLS_ONES_TO_HIT & X=1{
      result: [d6 hit without reroll]}
    else{
      if REROLLS_FAILED_TO_HIT
      {
        result: [d6 hit without reroll]
      }
      else {result: 0}
    }
  }
    else 
  {
    result: [d6 wound]
  }
}

function: attack  X:n times{
  if X>0{
    result: [d6 hit] + [attack X-1 times]
  }
  else {result: 0}
}

 
function: calculate damage inflicted by X:n failed saves{
  WOUNDS_LOST: 0
  LOCAL_REMAINING_WOUNDS: REMAINING_WOUNDS
  LOCAL_QUANTITY: QUANTITY
  TEMP_DAMAGE_DONE: 0
  if X > 0{
    loop Z over {1..X}{
      TEMP_DAMAGE_DONE: DAMAGE
      \model died\
      if LOCAL_REMAINING_WOUNDS - TEMP_DAMAGE_DONE < 1 {
        WOUNDS_LOST: WOUNDS_LOST + LOCAL_REMAINING_WOUNDS
        LOCAL_REMAINING_WOUNDS: W
        LOCAL_QUANTITY: LOCAL_QUANTITY - 1
        if LOCAL_QUANTITY < 1{
          result: WOUNDS_LOST
        }
      }
      \model survived\
      else{
        LOCAL_REMAINING_WOUNDS: LOCAL_REMAINING_WOUNDS - TEMP_DAMAGE_DONE
        WOUNDS_LOST: WOUNDS_LOST + TEMP_DAMAGE_DONE
      }
    }
  }
result: WOUNDS_LOST

}

output [calculate damage inflicted by [attack ATTACKS times] failed saves] named "Wounds lost"
output [attack ATTACKS times] named "Failed saves"
output [wounding on] named "Wounding on [WOUNDING_ON]+"

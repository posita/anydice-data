function: count ones S:s {
  result: S = 1
}
function: count above ones S:s {
  result: S > 1
}
\ this will give same results (2 out of 3) \
function: count one or three S:s {
  result: S != 2
}

output [count ones 80d2] \ 1 in 2 \
output [count ones 80d3] \ 1 in 3 \
\ output [count above ones 80d2] \ \ 1 in 2 \
output [count above ones 80d3]   \ 2 in 3 \
\ output [count one or three 80d3] \ \ 2 in 3 \

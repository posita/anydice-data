
function: iseven N:n {
  \returns 1 if N is even, or 0 if N is odd\
  result: (N/2 - (N-1)/2) 
}

function: eval ROLL:s {
 if [iseven #ROLL] {
  FUM: ROLL < 4
  if FUM = #ROLL/2 {
   result: FUM+[eval d6]
  }
 }
 result: ROLL < 4
}

loop PLAYERS over {1..10} {
 FUMBLES: PLAYERS/2
 output [eval PLAYERS d6] named "[PLAYERS] Players, no more than [FUMBLES] fumbles"
}

function: check D:n for T:n {
  if D>=T {result: 1}
  result: 0
}

function: explodecheck X:n size D:d target T:n {
  if X=[maximum of D] {result: [check X for T]+[check D for T]}
  result: [check X for T]
}

function: explodepool N:n dice D:d target T:n {
  result: Nd[explodecheck D size D target T]
}

output [explodepool 2 dice 1d8 target 6] named "basic roll"

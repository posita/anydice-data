MAX: 14
function: arkhandie THR:n ROLL:n {
 if THR <= 1 {
  result: 100
 }
 if (ROLL >= THR) {
  result: 1+[arkhandie (THR-1) (d6)]
 }
 result: 0
}

function: limit V:n {
 if V > MAX {
  result: MAX
 }
 result: V
}

function: arkhanhit {
 TOTAL: 0
 loop N over {1..10} {
  TOTAL: TOTAL + [arkhandie (6) (d6)]
 }

 result: [limit TOTAL]
}

function: cast VALUE:n roll ROLL:n {
 if ROLL >= VALUE {
  result:1
 }
 result: 0
}

function: cast VALUE:n {
 result: [cast VALUE roll 2d6]
}

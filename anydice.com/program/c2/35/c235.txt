ACR: 17
DEX: 7
PROFR: 6
SA: 6d6
WR: 1d8
FIRR: SA + WR
SECR: WR
RIPR: SA + WR + 1d8
Z: 0

ACW: 18
STR: 9
PROFW: 7
WW: 2d6+1d10

function: rip D:n{
  if D = 20 {result: RIPR + RIPR + DEX}
  if D + PROFR + DEX >= ACW {result: RIPR +DEX}
  result: 0
}

function: sneak D:n{
  if D = 20 {result: FIRR + FIRR + DEX}
  if D + PROFR + DEX >= ACW {result: FIRR +DEX}
  result: 0
}

function: att F:n{
  if Z = 0 {Z:1 result:[sneak F]}
  if F = 20 {result: WR + DEX}
  if F + PROFR + DEX >= ACW {result: RIPR + DEX}
  result: 0
}

output [att d20] + [att d20] + [att d20] + [rip d20] named "Rogue"

function: witch E:n{
  if E >= 18 {result: WW + WW + STR + 10}
  if E + PROFW + STR -5 >= ACR {result: WW + STR + 10}
  result: 0
}
output [witch d20] + [witch d20] + [witch d20] + [witch d20] named "Witch"

function: X:n witch {
if X >= 30 {result: 3}
if X >= 10 {result: 2}
result: 1
}

function: X:n fate {
if X >= 40 {result: 3}
if X >= 20 {result: 2}
result: 1
}

function: X:n persuasion {
if X >= 52 {result: 3}
if X >= 32 {result: 2}
if X >= 12 {result: 1}
result: 0
}

function: X:n empathy {
if X >= 60 {result: 3}
if X >= 40 {result: 2}
if X >= 20 {result: 1}
result: 0
}

THRESHOLDS: [d100 witch] + [d100 fate] 

output THRESHOLDS named "Vein Pass Count"
output THRESHOLDS >= 3 named  "Vein Success Chance"
output THRESHOLDS >= 5 named  "Vein Major Success Chance"

THRESHOLDSG: [d100 persuasion] + [d100 empathy] 

output THRESHOLDSG named "Ghost Pass Count"
output THRESHOLDSG >= 3 named  "Ghost Success Chance"
output THRESHOLDSG >= 5 named  "Ghost Major Success Chance"

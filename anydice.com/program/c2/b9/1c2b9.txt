function: attack ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS + ARCH >= AC { result: DAMAGE }
 result: 0
}

ENEMYAC: 15

ATTACKMOD: 3

PROFICIENCY: 2
CRITROLL: 20
EXTRAATTACK: 1
DMGBONUS: 0
ACCBONUS: 0
EXTRADMGDICE: 0

ARCH: 0

EXTRADMGDICE: 2d6

DAMAGE: 1d8 + ATTACKMOD + DMGBONUS + EXTRADMGDICE
CRITDAMAGE: 2d8 + ATTACKMOD + DMGBONUS + 2dEXTRADMGDICE

output d[attack d20 vs ENEMYAC] named "Rogue Sneak Attack"

EXTRADMGDICE: 0

DAMAGE: 2d6 + ATTACKMOD + DMGBONUS + EXTRADMGDICE
CRITDAMAGE: 4d6 + ATTACKMOD + DMGBONUS + 2dEXTRADMGDICE

output EXTRAATTACKd[attack d20 vs ENEMYAC] named "Greatsword"

DAMAGE: 1d6 + ATTACKMOD + DMGBONUS + EXTRADMGDICE
CRITDAMAGE: 2d6 + ATTACKMOD + DMGBONUS + 2dEXTRADMGDICE

output (EXTRAATTACK + 1)d[attack d20 vs ENEMYAC] named "TWF Shortswords"

ARCH: 2
DAMAGE: 1d8 + ATTACKMOD + DMGBONUS + EXTRADMGDICE
CRITDAMAGE: 2d8 + ATTACKMOD + DMGBONUS + 2dEXTRADMGDICE

output EXTRAATTACKd[attack d20 vs ENEMYAC] named "Archery Longbow"

SNIPER: 6
EXTRADMGDICE: (EXTRAATTACK-1)dSNIPER + (EXTRAATTACK-1)dSNIPER
DAMAGE: 1d10 + ATTACKMOD + DMGBONUS + EXTRADMGDICE + PROFICIENCY 
CRITDAMAGE: 2d10 + ATTACKMOD + DMGBONUS + 2dEXTRADMGDICE + PROFICIENCY 

output d[attack d20 vs ENEMYAC] named "Archery Sniper"

output d[attack [highest 1 of 2d20] vs ENEMYAC] named "Archery Sniper Advantage"

DAMAGE: 1d10 + ATTACKMOD + DMGBONUS + EXTRADMGDICE
CRITDAMAGE: 2d10 + ATTACKMOD + DMGBONUS + 2dEXTRADMGDICE

output EXTRAATTACKd[attack d20 vs ENEMYAC] named "Archery Hvy Crossbow"

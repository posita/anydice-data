\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n DAMAGE:n{
 if DAMAGE <0 {result: 0}
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..25}

ATTACKMOD: 10
DMGBONUS: 10
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 1d8 + DMGBONUS -(4+1d6)
CRITDAMAGE: 2d8 + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC DAMAGE] named "Normal Attack"

function: chaos N:n {
 if N = 6 { result: 1 + [chaos d6] }
 if N > 3 { result: 1 }
 result: 0
}

function: normaldie N {
  result: [count {4..6} in Nd6]
}

function:rolldice N {
  result: [normaldie N-1] + [chaos d6]
}

function:contest A B {
  if A > B {result: 1}
  if B < A {result: 2}
  {result: 3}
}

set "maximum function depth" to 20
output [contest [rolldice 1] [rolldice 1]]

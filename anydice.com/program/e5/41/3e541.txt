AC: 13
HIT: 5

function: tohit ROLL:n dmg DMG:n flat FLAT:n {
  if ROLL = 20 {
    result: 2 * DMG + FLAT
  }
  if ROLL = 1 {
    result: 0
  }
  if ROLL + HIT >= AC {
     result: DMG + FLAT
  }
  result: 0
}

output [tohit 1@2d20 dmg (1d8+2d6) flat 3] named "Sneak (Advantage)"
output [tohit 1@3d20 dmg (1d8+2d6) flat 3] named "Sneak (Elven Accuracy)"
output [tohit 1@2d20 dmg (2d6) flat 5] named "Greatsword"
output [tohit (1@2d20 - 5) dmg (2d6) flat 15] named "Greatsword (GWM)"

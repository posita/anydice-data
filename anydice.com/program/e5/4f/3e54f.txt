function: trespasser ROLL:n MOD:n DC:n {

if (ROLL + MOD <= DC-5) { RES: 0 }
else if (ROLL + MOD < DC) { RES: 1 }
else if (ROLL + MOD < DC + 5) { RES: 2 }
else { RES: 3 }

if (ROLL >= 17) {RES: RES+1}
if (RES < 0) { result: 0 }
else if (RES > 3) { result: 3 }
else { result: RES }

}

output [trespasser 3d6 0 10] named "3d6 vs 10"
output [trespasser 3d6 1 10] named "3d6 + 1 vs 10"
output [trespasser 3d6 2 10] named "3d6 + 2 vs 10"
output [trespasser 3d6 3 10] named "3d6 + 3 vs 10"

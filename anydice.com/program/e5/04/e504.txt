function: reroll R:n under N:n {
   if R < N { result: 1d6 } else {result: R}
}

function: attack D:n vs AC:n for DMG:n {
   if D = 20 {result: 2*DMG} 
   else if (D + MOD) >= AC { result: DMG }
   else { result: 0 }
}


MON_PROF: {5,5,6,7,9,9,9,10,11}

PLAYER_AC: 16
loop CR over {1..9} {
  DMG_LOW: 6*CR + 3
  DMG_HIGH:6*CR + 8
  MON_DMG:d{DMG_LOW..DMG_HIGH}
  
  MOD: CR@MON_PROF
  NORM: [attack d20 vs PLAYER_AC for MON_DMG]
  DEFENSE_AC: PLAYER_AC + 1
  DEF: [attack d20 vs DEFENSE_AC for MON_DMG]
  output NORM named "Average damage for CR [CR]"
  output DEF named "Average damage for CR [CR] with defense"
}

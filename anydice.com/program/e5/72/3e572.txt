\ This function generates basic odds for a d10 equal or higher then Target Number "TN" with rule of 1s \
function: odds DICE:s TN:n{
 ASUCCESS: DICE >=TN
 AONES: DICE = 1
 result: ASUCCESS - AONES
}

\ This function directly compares two sets of rolls. The A should be higher than B to calculate odds of victory \

function: comparison A:s and B:s {
 ASUCCESS: A >=6
 AONES: A = 1
 ARESULT: ASUCCESS - AONES
 BSUCCESS: B >=6
 BONES: B = 1
 BRESULT: BSUCCESS - BONES
 result: ARESULT - BRESULT
}

function: roll at DICE:n with A and B {

 AC : 18
 HIT : 9

 if DICE=20 { result: 2*A + B }
 if DICE=1 { result : 0 }
 if DICE>=(AC-HIT) { result: A + B } else { result: 0 }
}

DMG : 4d10
DMG_MOD : 0
TOTAL : 0
ADV : [highest 1 of 2d20]
DIS : [lowest 1 of 2d20]
STRAIGHT : 1d20

loop N over {1..5}{
TOTAL : TOTAL + [roll at STRAIGHT with DMG and DMG_MOD]
}

output TOTAL

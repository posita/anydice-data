HIT: 1d20
TOHIT: 5
TOHITB: 3
AC: 18
DMG: 1d8+1d6+3
CRITRANGE: 20
CRITDMG: 2d8+2d6+3

function: attack HIT:n TOHIT:n AC:n DMG:n CRITRANGE:n CRITDMG:n {
 if HIT >= CRITRANGE {result: CRITDMG}
 if (HIT+TOHIT) >= AC {result: DMG}
 result: 0
}

function: crit attack HIT:n TOHIT:n AC:n DMG:n {
 if (HIT+TOHIT) >= AC {result: DMG}
 result: 0
}

output [attack HIT TOHIT AC DMG CRITRANGE CRITDMG]
output [crit attack HIT TOHITB AC CRITDMG]

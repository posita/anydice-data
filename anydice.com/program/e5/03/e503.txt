function: reroll R:n under N:n {
   if R < N { result: 1d6 } else {result: R}
}

function: attack D:n vs AC:n for DMG:n {
   if D = 20 {result: 2*DMG} 
   else if (D + MOD) >= AC { result: DMG }
   else { result: 0 }
}

GWM_DMG: 2d[reroll d6 under 3]
NORM_DMG: 2d6
MOD: 7

output GWM_DMG + [attack d20 vs 16 for GWM_DMG] named "Great Weapon Master vs CR 5-7 (AC 16)"
output NORM_DMG + [attack d20 vs 16 for NORM_DMG] named "Normal vs CR 5-7 (AC 17)"

output GWM_DMG + [attack d20 vs 17 for GWM_DMG] named "Great Weapon Master vs CR 7-9 (AC 17)"
output NORM_DMG + [attack d20 vs 17 for NORM_DMG] named "Normal vs CR 7-9 (AC 17)"
output GWM_DMG + [attack d20 vs 17 for GWM_DMG] - (NORM_DMG + [attack d20 vs 17 for NORM_DMG]) named "Difference"

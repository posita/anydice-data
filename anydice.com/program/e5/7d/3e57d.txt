function: autofire A:n {
 if A > (DV+4) {result: [highest of 0 and (2d6 * 4) - SP]}
 if A > DV {result: [highest of 0 and (2d6 * (A - 17)) - SP]}
 result: 0
}

function: singleshot A:n {
 if A > DV {result: [highest of 0 and DMG - SP]}
 result: 0
}

function: melee A:n {
 if A > DV {result: [highest of 0 and DMG - (SP/2)]}
 result: 0
}

set "explode depth" to 1

DV: 17
SP: 11
DMG: 5d6

output [singleshot ATK] named "AR Single Shot"

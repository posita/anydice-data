function: hitcheck ROLL:n vs AC:n {
  if ROLL>=19 { result: HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

loop AC over {10, 15, 20} {
  CHA: 4
  TOHIT: CHA+4+1
  HITDIE: 1d10
  HITDAMAGE: CHA+4
  output [hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]  named "Hexblade Warlock vs [AC]"
}

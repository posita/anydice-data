function: hitcheck ROLL:n vs AC:n {
  if ROLL=20 { result: HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

loop AC over {10, 15, 20} {
  CHA: 4
  TOHIT: CHA+4+1
  HITDIE: 1d10+1d6
  HITDAMAGE: CHA
  output [hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]  named "Basic Warlock vs [AC]"

  DEX: 4
  TOHIT: DEX+2+4-5
  HITDIE: 1d6
  HITDAMAGE: DEX+10
  output [hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC] named "Basic Fighter vs [AC] AC"
}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

output 1d6 named "1d6" 

output [reroll {1} on d6 to d6] named "1d6, reroll 1"

output [reroll {1} on d6 to 2d3] named "1d6, on a 1 roll 2d3"

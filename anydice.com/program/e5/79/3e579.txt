function: autofire A:n {
 if A < DV {result: 0}
 if A < (DV+4) {result: 2d6 * (A - 17)}
 result: 2d6 * 4
}

function: singleshot A:n {
 if A < DV {result: 0}
 result: 5d6
}

set "explode depth" to 1
ATK: {6..16}
DV: {22, 20, 17, 20, 25}

output [autofire [explode 1d10] + ATK] named "Autofire"
output [singleshot [explode 1d10] + ATK] named "Single shot"

function: epo ROLL:n { 
if ROLL = 2*(ROLL/2) { result: ROLL + 1 }
else { result: ROLL }
}

loop N over {2..4}  {
  output 2d6 + N named "2d6+[N]"

}
loop N over {2..4}  {

  output [epo 2d6] + N named "2d6+[N].5 (even bump)"
}

loop N over {2..4}  {
  output 2d6 + d{0,1} + N named "2d6+?+[N] (+flip)"
}

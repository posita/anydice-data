function: hit ARMOR and TOUGHNESS with PENETRATION and DAMAGE {
 OVERCAME_ARMOR: [check armor with ARMOR and PENETRATION and d6]
 result: [check toughness with TOUGHNESS and DAMAGE and d6 and OVERCAME_ARMOR]
}

function: check armor with ARMOR:n and PENETRATION:n and ROLL:n {
  if ROLL + PENETRATION < ARMOR {
    result: 0
  }
  if ROLL + PENETRATION = ARMOR {
    result: 1
  }
  result: 2
}

function: check toughness with TOUGHNESS:n and DAMAGE:n and ROLL:n and OVERCAME_ARMOR:n{
  if OVERCAME_ARMOR < 2 {
    result: OVERCAME_ARMOR
  }
  if ROLL = 1 {
    result: 1
  }
  if ROLL + DAMAGE <= TOUGHNESS {
    result: 2
  }
  result: 3
}

output [hit 3 and 5 with 0 and 0] named "Armor 3, Toughness 5, +0/+0"
output [hit 3 and 5 with 1 and 0] named "Armor 3, Toughness 5, +1/+0"
output [hit 3 and 5 with 0 and 1] named "Armor 3, Toughness 5, +0/+1"

output d6 + 3 - d6
output d6 - d6


function: attackint DICE:n AC:n BONUS:n CRIT:n DMGDIE DMGBON SMITEDIE REDUCTION:n {
  if DICE >= CRIT { result: (2*DMGDIE + DMGBON)/REDUCTION + 2*SMITEDIE }
  if (DICE + BONUS) >= AC { result: (DMGDIE + DMGBON)/REDUCTION }
  result: 0
}

    result: [attackint d20 AC BONUS CRIT DMGDIE DMGBON SMITEDMG REDUCTION]
}

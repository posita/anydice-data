function: evaluate ROLL:s RESISTANCE:n {
 DAMAGE: ROLL - RESISTANCE
 if DAMAGE > 0 { result: DAMAGE }
 result: 0
}
loop TOUGHNESS over {1..5} {
output [evaluate 1d4 TOUGHNESS] named "1d4 damage against Toughness [TOUGHNESS]"

output [evaluate 1d12 TOUGHNESS] named "1d12 damage against Toughness [TOUGHNESS]"

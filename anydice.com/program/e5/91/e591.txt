function: hitcheck ROLL:n vs AC:n {
  if ROLL=20 { result: HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

function: precisionhitcheck ROLL:n vs AC:n {
  if ROLL=20 { result: HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

loop AC over {10, 15, 20} {
  DEX: 5
  TOHIT: DEX+2-5
  HITDIE: 1d8
  HITDAMAGE: DEX+10
  output [precisionhitcheck d20 vs AC]named "Fighter vs [AC] AC"
}

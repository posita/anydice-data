function: highest unless one of DICE:d {
  LOWEST: (#DICE)@DICE
  HIGHEST: 1@DICE
  result: [helper use LOWEST or HIGHEST]
}
function: helper use LOW:n or HIGH:n {
  if (LOW = 1) {
    result: LOW
  } else {
    result: HIGH
  }
}
loop D over {4, 6, 8, 10, 12, 20} {
  output [highest unless one of 2dD] named "2d[D]"
  output [highest unless one of 3dD] named "3d[D]"
}

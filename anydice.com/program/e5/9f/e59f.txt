function: hitcheck ROLL:n vs AC:n {
  if ROLL=20 { result: HITDIE+HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

loop AC over {10, 15, 20} {
  STR: 4
  TOHIT: STR+4
  HITDIE: 1d12
  HITDAMAGE: STR+3
  output [hitcheck [highest 1 of 2d20] vs AC]+[hitcheck [highest 1 of 2d20] vs AC]  named "Mounted Barbarian vs [AC]"
}

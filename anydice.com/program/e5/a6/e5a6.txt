function: hitcheck ROLL:n vs AC:n {
  if ROLL=20 { result: HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

function: precisionhitcheck ROLL:n and PDIE:n vs AC:n {
  if ROLL=20 { result: HITDIE+HITDIE+HITDAMAGE }
  else if ROLL+PDIE+TOHIT >= AC { result: HITDIE+HITDAMAGE }
  else {result: 0 }
}

loop AC over {10, 15, 20} {
  DEX: 4
  TOHIT: DEX+2+4-5
  HITDIE: 1d6
  HITDAMAGE: DEX+10
  output [precisionhitcheck d20 and d10 vs AC]+[precisionhitcheck d20 and d10 vs AC]+[precisionhitcheck d20 and d10 vs AC]+[precisionhitcheck d20 and d10 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC]+[hitcheck d20 vs AC] named "Precision Figher 2round w/surge vs [AC]"
}

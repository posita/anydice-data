function: compare A:n or B:n against C:n {
  R: [lowest of A and B]
  if R < C { result: 1 }
  if R = C { result: 0 }
  result: 0
}

function: threesuccesses MOD:n {
  A: [compare d8 or d8 against d6+MOD]
  B: [compare d8 or d8 against d6+MOD]
  C: [compare d8 or d8 against d6+MOD]
  result: A & B & C
} 

function: fivesuccesses MOD:n {
  A: [compare d8 or d8 against d6+MOD]
  B: [compare d8 or d8 against d6+MOD]
  C: [compare d8 or d8 against d6+MOD]
  D: [compare d8 or d8 against d6+MOD]
  E: [compare d8 or d8 against d6+MOD]
  result: A & B & C & D & E
} 

loop N over {0..2}{
  output [compare d8 or d8 against d6+N] named "Check STAT + [N]"
  output [threesuccesses N] named "STAT(3) + [N]"
  output [fivesuccesses N] named "STAT(5) + [N]"
}

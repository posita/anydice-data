function: attacksequence AS WAD BAD RAD DS {
 \Shortcut Variables\
  ATTACKDICE: WAD + BAD + RAD
 \Set ATTACKROLL\
   \No Surge\
    if AS = 0 {ATTACKROLL: WADd{1,1,0,0,0,0,0,0} + BADd{1,1,1,1,0,0,0,0} + RADd{1,1,1,1,1,1,0,0}}
   \Hit or Crit Surge\
    if AS = 1 {ATTACKROLL: WADd{1,1,1,0,0,0,0,0} + BADd{1,1,1,1,1,0,0,0} + RADd{1,1,1,1,1,1,1,0}}
 \Set DEFENSEROLL\
   \White\
    if DS = 1 {DEFENSEROLL: (ATTACKDICE)d{1,0,0,0,0,0}}
   \White with Surge\
    if DS = 2 {DEFENSEROLL: (ATTACKDICE)d{1,1,0,0,0,0}}
   \Red\
    if DS = 3 {DEFENSEROLL: (ATTACKDICE)d{1,1,1,0,0,0}}
   \Red with Surge\
    if DS = 4 {DEFENSEROLL: (ATTACKDICE)d{1,1,1,1,0,0}}
 \Function\
  result: ATTACKROLL - DEFENSEROLL
}
function: attacksequencefiltered F:s {
  if F < 0 {result: 0}
  result: F
}

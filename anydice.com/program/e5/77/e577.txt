\ to get a reroll effect, create a custom function that only includes results for the values you want to keep \

set "position order" to "lowest first"


function: ignore below N:n for ROLL:n {
 if N < ROLL { result: ROLL }
}

function: reroll R:n under N:n {
   if R < N { result: 2d6 } else {result: R}
}

function: cprr R:n and V:n under N:n {
 if R+V < N {
if R>V { result:R+1d6} else {result:V+1d6}
} else {result:R+V}
}

function: cprrh R:n and V:n and J:n under N:n {
if R+V+J < N { result: (R+V+J)-1@{R, V, J}+d6 } else { result:R+V+J }
}


\output [ignore below 4 for 2d6]\
output 2d6 named "2d6 summon"
output [cprr d6 and d6 under 8] named "Summon 2d6 with CP" 
output 3d6 named "3d6 summon"
output [cprrh d6 and d6 and d6 under 8] named "Summon 3d6 with CP"

\output [cprr d6 and d6 under 9] named "CP charge reroll" \
\output [reroll 2d6 under 9] named "full charge reroll" \

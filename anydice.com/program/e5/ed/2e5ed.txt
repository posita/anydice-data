function: strong ACTION:s BONUS:n CHALLENGEA:s CHALLENGEB:s {
  if ACTION + BONUS > CHALLENGEA {
    if ACTION + BONUS > CHALLENGEB {
      result: 1@[sort {ACTION, CHALLENGEA, CHALLENGEB}]
    }
  }
}

function: weak ACTION:s BONUS:n CHALLENGEA:s CHALLENGEB:s {
  if ACTION + BONUS > CHALLENGEA {
    if ACTION + BONUS <= CHALLENGEB {
      result: 2@[sort {ACTION, CHALLENGEA, CHALLENGEB}]
    }
  }
  else {
    if ACTION + BONUS > CHALLENGEB {
      if ACTION + BONUS <= CHALLENGEA {
        result: 2@[sort {ACTION, CHALLENGEA, CHALLENGEB}]
      }
    }
  }
}

function: miss ACTION:s BONUS:n CHALLENGEA:s CHALLENGEB:s {
  if ACTION + BONUS <= CHALLENGEA {
    if ACTION + BONUS <= CHALLENGEB {
      result: 3@[sort {ACTION, CHALLENGEA, CHALLENGEB}]
    }
  }
}
output [strong 1d6 1 1d10 1d10] named "+1 Strong Hit"
output [strong 1d6 2 1d10 1d10] named "+2 Strong Hit"
output [strong 1d6 3 1d10 1d10] named "+3 Strong Hit"
output [strong 1d6 4 1d10 1d10] named "+4 Strong Hit"
output [weak 1d6 1 1d10 1d10] named "+1 Weak Hit"
output [weak 1d6 2 1d10 1d10] named "+2 Weak Hit"
output [weak 1d6 3 1d10 1d10] named "+3 Weak Hit"
output [weak 1d6 4 1d10 1d10] named "+4 Weak Hit"
output [miss 1d6 1 1d10 1d10] named "+1 Miss"
output [miss 1d6 2 1d10 1d10] named "+2 Miss"
output [miss 1d6 3 1d10 1d10] named "+3 Miss"
output [miss 1d6 4 1d10 1d10] named "+4 Miss"

function: regattack A {
 if A=0 { result: 0 }
 else {
  result: [regattack 1 to A]
 }
}

function: regattack A to B {
 if A=B{
  result: 1d6+4
 }
 else {
  result: 1d6+4+[regattack A+1 to B]
 }
}

function: favorattack A {
 result: [favorattack 1 to A]
}

function: favorattack A to B {
 if A=B{
  result: 1d6+1d4+4
 }
 else {
  result: 1d6+1d4+4+[favorattack A+1 to B]
 }
}
set "maximum function depth" to 100
X:10
output [regattack X] named "Regular Attacks"
output [favorattack X] named "Divine Favor"
output 1d6+2d8+4+[regattack X-1] named "Divine Smite"

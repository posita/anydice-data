function: rerollone A:n B:n C:n D:n under N:n {
    \Only reroll if less than N\
    if A + B + C < N {
        \Find the lowest dice\
             if A < B & A < C { result: B + C + D }
        else if B < A & B < C { result: A + C + D }
        else                  { result: A + B + D }
    }
    else { result: A + B + C }
}

function: rerolltwod A:n B:n C:n D:n F:n  under N:n {
    \Only reroll if less than N\
    if A + B + C < N {
        \Find the lowest dice\
             if A < B & A < C { if D + B + C < N  { if B < C { result: D + F + C} else { result: D + B + F}} else { result: D + B + C }}
        else if B < A & B < C { if A + D + C < N  { if A < C { result: F + D + C} else { result: A + D + F}} else { result: A + D + C }}
        else                    if A + B + D < N  { if A < B { result: F + B + D} else { result: A + F + D}} 
    }
    else { result: A + B + C }
}

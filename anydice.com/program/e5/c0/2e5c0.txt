
function: reroll if under proficiency PROF:n {
 result: [reroll proficiency d20 helper]
}

function: reroll proficiency N:n helper {
 if N <= PROF {
   result: d20
 }
 result: N
}

function: reroll advantage if under proficiency PROF:n {
  result: [reroll advantage d20 helper]
}

function: reroll advantage N:n helper {
 if N <= PROF {
   result: [reroll advantage (N) with d20 advantage helper]
 }
 result: N
}
function: reroll advantage N:n with M:n advantage helper {
 if N <= M {
   result: M
 }
 result: N
}



loop P over {0..20} {
  output [reroll if under proficiency P] named "[P] reroll"
}
loop P over {0..20} {
  output [reroll advantage if under proficiency P] named "[P] advantage reroll"
}


\output d20
output [highest 1 of 2d20]\

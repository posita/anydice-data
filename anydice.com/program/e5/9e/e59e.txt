function: hitcheck ROLL:n vs AC:n {
  if ROLL=20 { result: 1 }
  else if ROLL+TOHIT >= AC { result: 1 }
  else {result: 0 }
}

function: advhitcheck ROLL:n and ROLL:n vs AC:n {
  if ROLL=20 { result: 1}
  else if ROLL+TOHIT >= AC { result: 1 }
  else {result: 0 }
}

loop AC over {1, 15, 20} {
  STR: 4
  TOHIT: STR+4
  HITDIE: 1d12
  HITDAMAGE: STR+3
  output [advhitcheck d20 and d20 vs AC]  named "Mounted Barbarian vs [AC]"
  output [hitcheck d20 vs AC] named "basic hit vs [AC]"
}

set "maximum function depth" to 5

function: reroll if under proficiency PROF:n {
 result: [reroll proficiency d20 helper 0]
}

function: reroll proficiency N:n helper ITER:n {
 USE_N: N - ITER
 if ITER < 5 & USE_N <= PROF {
   result: [reroll proficiency d20 helper (ITER + 1)]
 }
 result: USE_N
}

loop P over {0..10} {
  output [reroll if under proficiency P] named "[P] reroll"
}

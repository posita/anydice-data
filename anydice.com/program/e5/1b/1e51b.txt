  SCORE: 5 + 15*A
  if(ROLL > SCORE) { SCORE: 2}
  else if(ROLL > (SCORE - 10)) {result: 1}
  else if(ROLL > (SCORE - 30)) {result: 0}
  else if(ROLL > (SCORE - 40)) {result: -1}
  else {result: -2}
}

function: reroll R:n under N:n {
   if R < N { result: 1d6 } else {result: R}
}

function: attack D:n vs AC:n for DMG:d {
   if D = 20 {result: 2dDMG} 
   else if (D + MOD) >= AC { result: DMG }
   else { result: 0 }
}

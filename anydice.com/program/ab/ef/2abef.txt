function: blades ROLL:s { result: (1@ROLL>=4) + (1@ROLL=6) + (2@ROLL=6) }
function: any doubles of N:n or above in ROLL:s {
  loop X over ROLL {
   if X >= N & (ROLL = X) > 3 { result: 3 }
   if X >= N & (ROLL = X) > 2 { result: 2 }
   if X >= N & (ROLL = X) > 1 { result: 1 }
  }
  result: 0
}

output [blades 2@2d6] named "blades0d"
loop P over {1..6} { output [blades Pd6] named "blades[P]d" }
loop P over {1..12}{
 output [any doubles of 1 or above in Pd6] named "doubles [P]d"
}

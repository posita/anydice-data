MISS: 0
HIT_ONE: 1
HIT_TWO: 2
HIT_THREE: 3

function: action ROLL:n {
 if ROLL >= 20 { result: HIT_THREE }
 if ROLL >= 18 { result: HIT_TWO }
 if ROLL >= 11 { result: HIT_ONE }
 result: MISS
}

loop N over {1..25} {
  output Nd[action 1d20] named "[N] Actions"
  Y: N-1
  output Yd[action 1d20+1] named "[Y] Actions @ +1"
}

N: 8
D: d8

function: ROLL:n reroll under THRESHOLD:n as REROLL:d {
  if ROLL < THRESHOLD { result: REROLL }
  result: ROLL
}

REROLL_D: [dD reroll under 3 as dD]

output NdD named "Regular spell damage"
output (N-2)dD + 2dREROLL_D named "5e empowered spell damage (+2 CHA, re-roll if under 3)"
output (N-4)dD + 4dREROLL_D named "5e empowered spell damage (+4 CHA, re-roll if under 3)"
output [highest 1 of 2d(NdD)] named "N1300 empowered spell damage"
output (N+1)dD named "Roll an additional die spell damage"
output Nd([explode dD]) named "Exploding spell damage"
output [highest N of (2*N)dD] named "Keep N-highest spell damage"
output ((NdD)*3)/2 named "N1300 amplified spell damage"
output [maximum of NdD] named "N1300 maximized spell damage"

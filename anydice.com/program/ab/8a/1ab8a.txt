MISS: 0
HIT: 1
CRITHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

ATTACK: 10
loop AC over {10..30} {
 output [attack d20 vs AC] named "[AC]"
}

output [attack d20 vs d{10..30}]

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITHIT {
  result: CRITDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}
ATTACK: 5+3+6
HITDAMAGE: 2d6+1d8+1d8+5d8+5d8+20+8+3d6+2d6
CRITDAMAGE: 2*(2d6+1d8+1d8+5d8+5d8+3d6+2d6)+20+10
MISSDAMAGE: 0
loop AC over {10, 20, 30} {
 output [damage for [attack d20 vs AC]] named "[AC]"
}

output [damage for [attack d20 vs d{10..30}]]

output CRITDAMAGE named "true crit damage"
output HITDAMAGE named "true damage"

function: round X:n Y:n {
  result: (1dX > 1dY) - (1dX < 1dY)
}

function: total score X:n Y:n R:n {
  result: Rd[round X Y]
}

function: win probability X:n Y:n R:n T:n {
  result: [count {T..R} in [total score X Y R]] / [count {0..R} in [total score X Y R]]
}

X: 15
Y: 10
R: 10

output [total score X Y R] named "Total score after 10 rounds"
output [win probability X Y R 3] named "Probability of >= 3 points (Trivial)"
output [win probability X Y R 5] named "Probability of >= 5 points (Important)"
output [win probability X Y R 7] named "Probability of >= 7 points (Crucial)"

function: roll TIMES:n die SIDES:n count PASS:n explode on EXPL:n {
  SUCCESSES: 0
  DIE: 1dSIDES
  loop N over {1..TIMES} {
    SUCCESSES: SUCCESSES + [DIE success on PASS explode on EXPL]
  }
  result: SUCCESSES
}

function: DIETYPE:n success on PASS:n explode on EXPL:n {
  SUCCESSES: 0
  if DIETYPE >= PASS { SUCCESSES: 1 }
  if DIETYPE >= EXPL { SUCCESSES: SUCCESSES + [DIE success on PASS explode on EXPL] }
  result: SUCCESSES
}

output [roll 7 die 10 count 8 explode on 10] named "player"

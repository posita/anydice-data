MISS: 0
HIT: 1
CRITICALHIT: 2
MISSDAMAGE: 0

ATTACK: +5
HITDAMAGE: d6 + 3
CRITICALDAMAGE: 2d6 +3

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

output [damage for [attack d20 vs d{10..20}]] named "Regular Attack"

ATTACK: +5
HITDAMAGE: 1d4
CRITICALDAMAGE: 2d4

output [damage for [attack d20 vs d{10..20}]] named "Off hand Attack"

ATTACK: +5
HITDAMAGE: 2d6 + 3
CRITICALDAMAGE: 4d6 +3

output [damage for [attack [highest 1 of 2d20] vs d{10..20}]] named "Critical Attack"

function: evaluate ROLL:s {
SUCCESSES: ROLL >= 7
result: SUCCESSES  + (ROLL = 10)+ (-1*(ROLL = 1))
}

function: degree ROLL:s {
RES: (1@ROLL>=7)+(1@ROLL=10)-(#ROLL@ROLL<7)-(#ROLL@ROLL=1)
output [RES] named "degreeRes"
result: RES
}

function: bonus ROLL:s {
if [degree ROLL]>0{
result: [evaluate ROLL]
}
else{
result :[evaluate ROLL]
}
}

loop DICE over {1..5} {
output [bonus DICE d10] named "[DICE]d10"
}

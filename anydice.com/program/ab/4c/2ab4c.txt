function: ROLL:n gwf REROLL:d reroll SEQUENCE:s{
  if [SEQUENCE contains ROLL] {
    result: REROLL
  }
  else {
    result: ROLL
  }
}

DEX: 4
DIE_ONE: [d4 gwf d4 reroll {1,2}]
DIE_TWO: [d4 gwf d4 reroll {1,2}]
DIE_THREE: [d8 gwf d8 reroll {1,2}]
DIE_FOUR: [d10 gwf d10 reroll {1,2}]

output DIE_ONE+DIE_TWO+DEX named "One-Hand d4 gwf"
output 2d4+DEX+2 named "One-Hand d4 dueling"
output DIE_THREE+DEX named "One-Hand d8 gwf"
output 1d8+DEX+2 named "One-Hand d8 dueling"
output DIE_FOUR+DEX named "Two-Hand d10 gwf"
output 1d10+DEX named "Two-Hand Current"

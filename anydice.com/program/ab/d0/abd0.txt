function: bas DIE:n m MAX:n s SKILL:n p BONUS:n {
  if DIE <= BONUS {
    if DIE = BONUS {
      result: 25
    } else {
      result: SKILL + DIE
    }
  } if DIE > BONUS {
    if DIE = MAX {
      result: -25
    } else {
      result: SKILL
    }
  }
}

function: sub DIEA:n m DIEB:n s SKILL:n r RANGE:n {
  if DIEA = 20 {
    result: 25
  } if DIEA = 1 {
    result: -25
  } else {
    TOTAL: DIEA - DIEB
    if TOTAL > RANGE {
      result: 0 + SKILL
    } if TOTAL < -RANGE {
      result: 0 + SKILL
    } else {
      result: TOTAL + SKILL
    }
  }
}

function: min DIEA:n m DIEB:n s SKILL:n r RANGE:n {
  if DIEA > DIEB {
    if DIEA = 20 {
      result: 25
    } else {
      TOTAL: DIEA - DIEB
      if TOTAL > RANGE {
        result: 0 + SKILL
      } if TOTAL < RANGE {
        result: TOTAL + SKILL
      } else {
        result: TOTAL + SKILL
      }
    }
  } if DIEA < DIEB {
    if DIEB = 20 {
      result: -25
    } else {
      TOTAL: DIEB - DIEA
      if TOTAL > RANGE {
        result: 0 + SKILL
      } if TOTAL < RANGE {
        result: TOTAL + SKILL
      } else {
        result: TOTAL + SKILL
      }
    }
  } if DIEA = DIEB {
    TOTAL: DIEA - DIEB
    if TOTAL > RANGE {
      result: 0 + SKILL
    } if TOTAL < RANGE {
      result: TOTAL + SKILL
    } else {
      result: TOTAL + SKILL
    }
  }
}

output [bas d20 m 20 s 0 p 6]
output [sub d20 m d20 s 0 r 6]
output [min d20 m d20 s 0 r 6]

function: ROLL:n gwf REROLL:d reroll SEQUENCE:s{
  if [SEQUENCE contains ROLL] {
    result: REROLL
  }
  else {
    result: ROLL
  }
}

DIE_ONE: [d6 gwf d6 reroll {1,2}]
DIE_TWO: [d8 gwf d6 reroll {1,2}]
DIE_THREE: [d10 gwf d10 reroll {1,2}]
DIE_FOUR: [d12 gwf d12 reroll {1,2}]

output DIE_ONE
output DIE_TWO
output DIE_THREE
output DIE_FOUR

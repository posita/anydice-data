WARLOCK_HIT: 1d20+7
WARLOCK_DMG: 1d10+4
MONK_HIT: 1d20+8
MONK_DMG: 1d8+5
function: switch S:n {
 if S > 0 { result: 1 }
 result: 0
}
function: HIT:n then DMG:n {
 if [switch HIT - 15] = 1 { result: DMG }
 result: 0
}
output 2d{[WARLOCK_HIT then WARLOCK_DMG]} named "Warlock"
output 2d{[MONK_HIT then MONK_DMG]} named "Monk"

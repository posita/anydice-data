function: sub DIEA:n m DIEB:n s SKILL:n r RANGE:n {
  if DIEA = 20 {
    result: 100
  } if DIEA = 1 {
    result: -100
  } else {
    TOTAL: DIEA - DIEB
    if TOTAL > RANGE {
      result: 0 + SKILL
    } if TOTAL < -RANGE {
      result: 0 + SKILL
    } else {
      result: TOTAL + SKILL
    }
  }
}

function: min DIEA:n m DIEB:n s SKILL:n r RANGE:n {
  if DIEA > DIEB {
    if DIEA = 20 {
      result: 100
    } else {
      TOTAL: DIEA - DIEB
      if TOTAL > RANGE {
        result: 0 + SKILL
      } if TOTAL < RANGE {
        result: TOTAL + SKILL
      } else {
        result: TOTAL + SKILL
      }
    }
  } if DIEA < DIEB {
    if DIEB = 20 {
      result: -100
    } else {
      TOTAL: DIEB - DIEA
      if TOTAL > RANGE {
        result: 0 + SKILL
      } if TOTAL < RANGE {
        result: TOTAL + SKILL
      } else {
        result: TOTAL + SKILL
      }
    }
  } if DIEA = DIEB {
    TOTAL: DIEA - DIEB
    if TOTAL > RANGE {
      result: 0 + SKILL
    } if TOTAL < RANGE {
      result: TOTAL + SKILL
    } else {
      result: TOTAL + SKILL
    }
  }
}

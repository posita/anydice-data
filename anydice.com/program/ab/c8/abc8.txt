function: bas DIE:n m MAX:n s SKILL:n p BONUS:n {
  if DIE <= BONUS {
    if DIE = BONUS {
      result: 10
    } else {
      result: SKILL + DIE
    }
  } if DIE > BONUS {
    if DIE = MAX {
      result: -10
    } else {
      result: SKILL
    }
  }
}

function: sub DIEA:n m DIEB:n s SKILL:n r RANGE:n {
  TOTAL: DIEA - DIEB
  if TOTAL > RANGE {
    result: 0 + SKILL
  } if TOTAL < -RANGE {
    result: 0 + SKILL
  } else {
    result: TOTAL + SKILL
  }
}

function: min DIEA:n m DIEB:n s SKILL:n r RANGE:n {
  if DIEA >= DIEB {
    TOTAL: DIEA - DIEB + SKILL
  } if DIEA < DIEB {
    TOTAL: DIEB - DIEA + SKILL
  }
  if TOTAL > RANGE {
    result: 0 + SKILL
  } if TOTAL < RANGE {
    result: TOTAL + SKILL
  } else {
    result: TOTAL + SKILL
  }
}

output [sub d20 m d20 s 0 r 10]
output [min d20 m d20 s 0 r 10]

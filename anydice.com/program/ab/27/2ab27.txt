\Damage Distrobution Calculator
Takes crits, probability to hit, (dis)advantage, etc it to account 
This is written for Ian's House rules\
\Global Constants\
MISS: 0
HIT: 1
CRIT: 2

\User Variable Input\
TOHIT: 14 \to hit modifier - should be a constant - 999 for guaranteed hit\
DMG: 1d10+8 \attack damage - should be dice + modifier\
MISSDMG: 0 \damage done on a miss - usually 0 but some abilities may deal damage on miss\
AC: 25 \enemy AC - should be constant\

\Normal Attack Function\
function: attack ROLL:n vs AC:n {
 if ROLL = 1 {
  if TOHIT = 999 {result: HIT} 
  result: MISS}
 if ROLL = 20 {result: CRIT}
 if ROLL + TOHIT > AC {result: HIT}
 else {
 if TOHIT = 999 {result: HIT}
 result: MISS}
}

function: damage for RESULT:n {
 if RESULT = CRIT {result: CRIT * DMG}
 if RESULT = HIT {result: DMG}
 if RESULT = MISS {result: MISSDMG}
}

function: attack {
 result: [damage for [attack 1d20 vs AC]]
}

function: advantage attack {
 result: [damage for [attack [highest 1 of 2d20] vs AC]]
}

function: super advantage attack {
 result: [damage for [attack [highest 1 of 3d20] vs AC]]
}

function: disadvantage attack {
 result: [damage for [attack [lowest 1 of 2d20] vs AC]]
}

function: super disadvantage attack {
 result: [damage for [attack [lowest 1 of 3d20] vs AC]]
}

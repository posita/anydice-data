function: outlaw A:n {
  if A >= 10 {result: 10}
  else if A >= 8 {result: 5}
  else {result: 0}
}

loop DICE over {0..7}
{
  if (DICE <= 0) {
    output [outlaw [lowest 1 of 2d10]] named "Outlaw d10: [DICE] dice"
  }
  else {
    output [outlaw [highest 1 of DICEd10]] named "Outlaw d10: [DICE] dice"
  }
}

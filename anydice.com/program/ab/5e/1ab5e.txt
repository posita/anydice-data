\ variables \
CRIT: 20
HIT: 15
PROF: 3
HITBONUS: 0

STR: 5
DMGBONUS: 0
RAGEBONUS: 2



\ rolls the correct number of d20s \
function: attack NUMBER:n RAGE:d {
 VALUE: 1@NUMBERd20
 result: [check VALUE RAGE]
}



\ checks to see if the attack is a crit \
function: check NUMBER:n RAGE:d {
 if NUMBER >= CRIT {
  result: [damage 3 RAGE]
 } else if NUMBER + STR + HITBONUS + PROF >= HIT {
  result: [damage 1 RAGE]
 } else {
  result: 0
 }
}



\ calculates damage for each damage die \
function: damage NUMBER:n RAGE:d {
 ROLL: 0
 
 loop N over {1..NUMBER} {
  LEFT: [reroll 1d12]
  RIGHT: [reroll 1d12]
  CENTER: [compare LEFT RIGHT]
  
  ROLL: ROLL + CENTER
 }
 
 result: ROLL + STR + DMGBONUS + RAGE
}



\ rerolls a damage die once if it is a 1 or a 2 \
function: reroll ROLL:n {
 if ROLL < 3 {
  result: 1d12
 } else {
  result: ROLL
 }
}



\ compares two damage dice for the higher roll \
function: compare ROLL:n CHECK:n {
 if ROLL > CHECK {
  result: ROLL
 } else {
  result: CHECK
 }
}



\ program outputs \
output [attack 1 0] named "regular"
output [attack 2 0] named "advantage"

output [attack 1 0] + [attack 1 0] named "action surge"
output [attack 2 0] + [attack 2 0] named "action surge w/advantage"

output [attack 1 RAGEBONUS] + [attack 1 RAGEBONUS] named "frenzy w/disadvantage"
output [attack 2 RAGEBONUS] + [attack 2 RAGEBONUS] named "frenzy"
output [attack 2 RAGEBONUS] + [attack 2 RAGEBONUS] + [attack 2 RAGEBONUS] named "frenzied action surge"

function: straight D:s {
C:0
L:0
loop N over {1..6}{
if [D contains N]{C:C+1}else{if C>L{L:C}C:0}
}
if C>L{L:C}
result:L
}

function: count equals D:s{
R:{}
loop N over {1..6}{
R:{R,[count N in D]}
}
result: R
}

function: paschs D:s
{
if 1@[sort [count equals D]]>1 {result: 1@[sort [count equals D]]}else{result:0}
}

function: fullhouse D:s{
if [[count equals D] contains 2] & [[count equals D] contains 3]{result: 1000}else{result: 0}
}

function: two pairs D:s \works because of "at least". Does not need to be combined with full house\{
if 1@[sort [count equals D]]>1 & 2@[sort [count equals D]]>1{result:  100}else{result: 0}
}

function: no two alike D:s{
if 1@[sort [count equals D]]>1{result:0}else{result:1}
}

function: nothing under aka get lowest D:s{
result:#D@[sort D]
}

function: paschs of x D:s{
if 1@[count equals D]>1 {result: 1@[sort [count equals D]]}else{result:0}
}

PROF : 3
MOD : 4
AC: 15

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

X : [attack 1d20 plus PROF+MOD+1 vs AC for 2d10+1d8 crit 2d10+1d8 on 20]
Y : [attack 1d20 plus PROF+MOD+1 vs AC for 2d8 crit 2d8 on 20]

output X+Y named "5th level artillerist artificer with firebolt and force ballista (enhanced arcane focus infusion and arcane firearm)"

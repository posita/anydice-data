FAIL: 0
SUCC: 1

function: luckcheck ROLL:n vs LUCK:n {
 if ROLL > LUCK { result: FAIL }
 result: SUCC
}

FAILLUCK: 2d10+10
SUCCLUCK: 1d10+5

function: luckresult for RESULT:n and LUCK:n {
 if RESULT = FAIL { result: FAILLUCK + LUCK }
 result: SUCCLUCK + LUCK
}

loop LUCK over {0..99} {
 output [luckresult for [luckcheck d100 vs LUCK] and LUCK] named "[LUCK]"
}

function: clamp N:n between MIN:n and MAX:n {
  if N > MAX { result: MAX }
  if N < MIN { result: MIN }
  result: N
}

RANGE: 25
FUDGE: {-1..1}
ZNINE: {0..9}

\ These make tails more broad but greatly increase chance of additional rolls \
\output [clamp 2d[explode dZNINE]-2d[explode dZNINE] between -RANGE and RANGE] named "2z9!-2z9!"\
\output [clamp 2d[explode dZNINE]-8 between -RANGE and RANGE] named "2z9!-8"\

output [clamp d12+[explode dZNINE]-d12-[explode dZNINE] between -RANGE and RANGE] named "d12+z9! vs"

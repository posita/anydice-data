function:  rollone withbonus BONUS:n against DC:n {
  DIE: d100
  CRITBY: DIE + BONUS - DC
  result: (DIE + BONUS) + (CRITBY >= 0) * (CRITBY + BONUS)
}

function:  roll ROLLS:n withbonus BONUS:n against DC:n {
  TOTAL: 0
  loop N over {1..ROLLS} {TOTAL: TOTAL + [rollone withbonus BONUS against DC]}
  result: TOTAL 
}

output [roll 2 withbonus (5+6+5) against 50] + 261 + 25 >= 500

function: boon ROLL:s {
  SUCCESSES: ROLL > 3
  if(SUCCESSES < #ROLL) { result: SUCCESSES + 1 }
  result: SUCCESSES
}

  if(SUCCESSES > 0) { result: SUCCESSES - 1 }
  result: SUCCESSES
}

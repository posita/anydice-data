\ Firebolt\
B: 1d10
C:4
X:0

function: reroll N:n for ROLL:n {
 if N < ROLL {result: ROLL }
 else {result: B}
}


\FireBolt\
output B named "Weak FireBolt"

\FireBolt+Cha\
output B+C named "FireBolt"

\FireBolt,reroll 1's,+Cha\
output [reroll 1 for B] + C named "Infernal FireBolt"

\FireBolt,+Cha+Cha\
output B+C+C named "Powerful FireBolt"

\FireBolt,reroll 1's,+Cha+Cha\
output [reroll 1 for B]+C+C named "Powerful Infernal FireBolt"

\** 2D6 reroll Advantages **\
function: rerolldisadvantage DICEA:n vs DICEB:n {
 if DICEA >= DICEB { result: DICEB + 1d6 } 
 if DICEA < DICEB { result: DICEA + 1d6 }
}

function: rerolladvantage DICEA:n vs DICEB:n {
 if DICEA <= DICEB { result: DICEB + 1d6 } 
 if DICEA > DICEB { result: DICEA + 1d6 }
}

function: onlysmalldisadvantage DICEA:n vs DICEB:n {
 if DICEA > DICEB { result: DICEB + 1d6 } 
 if DICEA < DICEB { result: DICEA + 1d6 }
 result: DICEA + DICEB
}

function: onlysmalladvantage DICEA:n vs DICEB:n {
 if DICEA < DICEB { result: DICEB + 1d6 } 
 if DICEA > DICEB { result: DICEA + 1d6 }
 result: DICEA + DICEB
}

function: picksmallest DICEA:n vs DICEB:n {
 if DICEA < DICEB { result: DICEA + DICEA } 
 if DICEA > DICEB { result: DICEB + DICEB }
 result: DICEA + DICEB
}

function: pickbiggest DICEA:n vs DICEB:n {
 if DICEA > DICEB { result: DICEA + DICEA } 
 if DICEA < DICEB { result: DICEB + DICEB }
 result: DICEA + DICEB
}

output [rerolldisadvantage 1d6 vs 1d6] named "Disadvantage"
output 2d6 named "Normal"
output [rerolladvantage 1d6 vs 1d6] named "Advantage"

output [onlysmalldisadvantage 1d6 vs 1d6] named "Keep Same Disadvantage"
output [onlysmalladvantage 1d6 vs 1d6] named "Keep Same Advantage"

output [picksmallest 1d6 vs 1d6] named "Pick Smallest"
output [pickbiggest 1d6 vs 1d6] named "Pick Biggest"

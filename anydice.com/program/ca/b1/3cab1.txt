function: tocci D:s {
 if D >= 5 {
  result: 5
 }
 else if D >= 3 {
  result: 3
 }
 else {
  result: 1
 }
}

loop N over {6, 8, 10, 12} {
 output [tocci dN] named "d[N]"
}

output [tocci [highest of d6  and d6]] named "d6+d6"
output [tocci [highest of d8  and d6]] named "d8+d6"
output [tocci [highest of d10 and d6]] named "d10+d6"
output [tocci [highest of d12 and d6]] named "d12+d6"

output [tocci [highest of d6  and 1@2d6]] named "d6+d6+d6"
output [tocci [highest of d8  and 1@2d6]] named "d8+d6+d6"
output [tocci [highest of d10 and 1@2d6]] named "d10+d6+d6"
output [tocci [highest of d12 and 1@2d6]] named "d12+d6+d6"

function: explosive C:n {
 if C > 3{
result: C+10
}
result: C
}


function: minum A:n to B:n {
if A > B{
  result: A
}
  result: B
}

function: check A:n to B:n {
if A = 20 {
  \result: [minum( (6d4+2) - ( [bothexp 1d4 and 1d4] ) )to 0]\
  \result: [minum( (6d4+2) - ( [threeexp 1d2 and 1d2 and 1d2] ) )to 0]\
  \result: [minum(6d4+2)-( 1d8)to 0]\
}
if A > B { 
  \result: [minum( (3d4+2) - ( [bothexp 1d4 and 1d4] ) )to 0]\
  \result: [minum( (3d4+2) - ( [threeexp 1d2 and 1d2 and 1d2] ) )to 0]\
  result: [minum(3d4+2)-( 1d8)to 0]
}
result: 0
}

function: checka A:n to B:n {
if A = 20 {
  result : 6d4+2 
}
if A > B { 
  result: 3d4+2
}
result: 0
}

function: bothexp A:n and B:n{
if A = 2 & B = 2{
  result: A+B+6
}
  result: A+B
}

function: threeexp A:n and B:n and C:n{
if A = 2 & B = 2 & C=2{
  result: A+B+C+6
}
  result: A+B+C
}

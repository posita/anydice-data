function: stat {
 result: [highest 3 of 4d6]
}

function: thresh ROLL:s LIM {
 if (ROLL < LIM ) {
  OUT: 0
 }
 else {
  OUT: 1
 }
 result: OUT
}

function: total LIM NUM {
 OUT: 0
 loop N over {1..NUM}{
  OUT: OUT + [thresh [stat] LIM]
 }
 result: OUT
}

loop M over {10..18}{
 output [total M 6] named "At least X of [M]+"
}

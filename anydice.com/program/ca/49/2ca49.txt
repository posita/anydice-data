PROF : 4
MOD : 5
AC: 17
SPELL: 5

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: save ROLL:n plus BONUS:n vs DC:n for DMG:d fail FAIL_DMG:d {
    if ROLL + BONUS < DC {
        result: DMG
    }
    result: FAIL_DMG
}

X : [attack 1d20 plus PROF+MOD vs AC for 3d10 crit 3d10 on 20]
Y : [attack 1d20 plus PROF+MOD vs AC for 1d6+4+SPELL crit 1d6 on 20]
Z : [save 1d20 plus PROF vs 8+PROF+MOD for 2d6 fail 2d6/2]

output X+Y+Y+Z named "11th level caster attacking with firebolt and using a 5th level 'Summon Draconic Spirit'"

PROF : 4
MOD : 5
AC: 17

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

X : [attack 1d20 plus PROF+MOD+2 vs AC for 3d10+1d8 crit 3d10+1d8 on 20]
Y : [attack 1d20 plus PROF+MOD+2 vs AC for 3d8 crit 3d8 on 20]

output X+Y named "11th level artillerist artificer with firebolt and force ballista (enhanced arcane focus infusion and arcane firearm"

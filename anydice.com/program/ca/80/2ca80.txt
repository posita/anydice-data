set "explode depth" to 2   \by default, anydice only explodes twice\

DIE: 6   \custom die with five zeros and one five eq. To {0,0,0,0,0,5}\
POOL: 6   \number of dice being rolled\
NKEEP: {1..4}  \number of highest dice to keep\

\output [highest NKEEP of POOL d [explode d DIE]] named "[POOL] dice"\

loop NUMBER over {1..5} {
 output [highest NUMBER of POOL d [explode d DIE]] named "[POOL] d [DIE] exp keep [NUMBER]"
}

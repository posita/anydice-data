\ If 1 is less than roll (2+) then do nothing, if it is one, reroll\

function: reroll N:n for ROLL:n {
 if N < ROLL {result: ROLL }
 else {result: 1d8}
}


\Scorching Ray\
X : 0
loop N over{1..3}{
 loop M over{1..2}{
  X:X+1d8
 }
 output X named "[N] Scorching Rays"
}

\Scorching Ray with CHA\
CHA : 4
X : 0
loop N over{1..3}{
 loop M over{1..2}{
  X:X+1d8
 }
 X:X+CHA
 output X named "[N] Scorching Rays + Charisma to damage"
}

\Scorching Ray with rerolled ones\
X : 0
loop N over{1..3}{
 loop M over{1..2}{
  X:X+[reroll 1 for 1d8]
 }
 output X named "[N] Scorching Rays, rerolled 1's"
}

\Scorching Ray with rerolled ones and CHA bonus damage\
CHA : 4
X : 0

loop N over{1..3}{
 loop M over{1..2}{
  X:X+[reroll 1 for 1d8]
 }
 X:X+CHA
 output X named "[N] Scorching Rays, rerolled 1's, + Charisma to damage"
}

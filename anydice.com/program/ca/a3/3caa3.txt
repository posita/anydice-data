
function: edamage on HIT:n{
if HIT = 0 {result: 0}
result: HITd10 + HITd6 + 5
}
function: blightdamage on HIT:n{
result: 9d8/HIT
}
function: fballdamage on HIT:n{
result: 10d6/HIT
}
function: rebukedamage on HIT:n{
result: 6d10/HIT
}
function: ROLL:n plus BONUS vs AC{
if ROLL = 20 {result: 2}
result: (ROLL + BONUS > AC)
}
function: try SPELL vs THROW:n plus BONUS{
result:(SPELL > THROW + BONUS) + 1
}
output [edamage on [d20 plus 9 vs 18]] + [edamage on [d20 plus 9 vs 18]] named "2 Eldritch Blast with Hex (AC)"
output [blightdamage on [try 17 vs d20 plus 3]] named "Blight (Con Save)"
output [fballdamage on [try 17 vs d20 plus 3]] named "Fireball (Dex Save)"
output [rebukedamage on [try 17 vs d20 plus 3]] named "Hellish Rebuke (Dex Save)"

loop N over {1..5} {
 loop M over {-1..1} {
  DIFFICULTY: N*5+M*5
  FULL_DIFFICULTY: DIFFICULTY + 10
  CRITICAL_DIFFICULTY: DIFFICULTY + 20

  PARTIAL: Nd12 >= DIFFICULTY
  FULL: Nd12 >= FULL_DIFFICULTY
  CRIT: Nd12 >= CRITICAL_DIFFICULTY
  output PARTIAL + FULL + CRIT  named "[N]d12 vs [DIFFICULTY]"
 }
}

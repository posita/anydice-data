function: calc A:n {
 result: DMIN+A*DPL
}

function: comp A:n B:n {
 if A > B {result: -1}
 if B > A {result: 1}
 else {result: 0}
}

function: throw A:n {
 result: [calc A]d6
}

function: center A {
 result: [throw A] - [calc A]*3 - [calc A]/2
}

function: fold A:n {
 if A < 0 {result: -A}
 else {result: A}
}

function: roll A:n B:n {
 result: [comp [fold [throw A]] [fold [throw B]]]
}

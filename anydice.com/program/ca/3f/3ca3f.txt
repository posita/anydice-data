DICE: {4,6,8,10,12,20}

function: fumble A:n {
  if A = 1 { result: 0 }
  result: A
}

function: check A:d {
  result: [fumble [explode A]]
}

function: cost J:n {
  COSTDIE: 0
  loop DIE over DICE {
    COSTDIE: COSTDIE + DIE ^ 2
    if J = DIE { result: COSTDIE - 4^2}
  }
  result: COSTDIE -4^2
}

function: cost N:n by J:n {
  COST: 0
  loop I over {2..N} {
    COST: COST + I * 4
  }
  result: COST + [cost J]
}

loop J over DICE {
  loop I over {1..10} {
    output [cost I by J]/Id[check dJ] named "cost [I]d[J]"
  }
}

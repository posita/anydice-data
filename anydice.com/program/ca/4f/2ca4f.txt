PROF : 4
MOD : 5
AC: 17
MAGIC: 0

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

Z : [reroll {1,2} on d10 to d10]
ZZ: [reroll {1,2} on d4 to d4]

Y : [attack 1d20 plus PROF+MOD-5+MAGIC vs AC for ZZ+MOD+10+MAGIC crit ZZ on 20]

output 3dX+Y named "11th level fighter with polearm master and great weapon master and great weapon fighting style"

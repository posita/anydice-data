function: chaos CONDITION:n {
  if CONDITION <= 3 { result: -2 }
  else if CONDITION <= 5 { result: -1 }
  else if CONDITION <= 8 { result: 0 }
  else if CONDITION <= 10 { result: 1 }
  else { result: 2 }
}

function: reroll INITIAL:n REROLL:n THRESHOLD:n {
  if INITIAL < THRESHOLD { result: REROLL }
  else { result: INITIAL }
}

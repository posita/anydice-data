function: ROLL:n vs AC:n {

  if ROLL = 1 { result: MISS }
  if ROLL = 20 & ROLL + ATTACK >= AC { result: CRIT }
  if ROLL = 20 & ROLL + ATTACK < AC { result: HIT }
  if ROLL >= AC { result: HIT }
  result: MISS

}

ATTACK: 4
HIT: 1d12
CRIT: 2d12+1d6
MISS: 0

loop AC over {10..25} {
  output [d20 vs AC] named "1x vs [AC]"
}

ATTACK: 0

loop AC over {10..25} {
  output [d20 vs AC] + [1d20 vs AC] named "2x vs [AC]"
}

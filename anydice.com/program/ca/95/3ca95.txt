\SKILL: 2

CHALLENGE: 2
SUCCESS: {4,5,6,6}

SUCCESS_COUNT: [count SUCCESS in DICE]

output SUCCESS_COUNT named "Possible # of successes"
output SUCCESS_COUNT >= CHALLENGE named "Chance of success"\

function: find SUCCESS:s in SKILL:n {
 result: [count SUCCESS in SKILLd6]
}

SUCCESS_N: {4,5,6,6} \6 is a crit and is counted as two successes\

loop I over {1..4} {
 loop V over {1..6} {
 output [find SUCCESS_N in I] >= V named "Chance to beat challenge lvl [V] w/ Skill Lvl [I]"
 }
}

output [find SUCCESS_N in 1] named "# Successes w/ Skill Lvl 1"

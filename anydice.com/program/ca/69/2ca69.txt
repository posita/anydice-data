PROF : 3
MOD : 4
AC: 15

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}
Y : [reroll {1,2} on d6 to d6]
X : [attack 1d20 plus MOD vs AC for Y+Y + MOD + 2*PROF crit Y+Y on 20]


output 2dX named "5th level fighter with a greatsword using heavy strike"

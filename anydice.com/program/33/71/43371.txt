\ Set your Spell Save DC and the target's Wisdom Save modifier \
DC: 15
TARGET_SAVE_MOD: 4

\Spell Casting Level\
LEVEL: 2

\ Define the damage dice (3d6 for 1st level) \
DAMAGE_DICE: (LEVEL + 2)d6

\ Function to calculate the outcome \
function: dissonantwhispers SAVE:n {
  if SAVE + TARGET_SAVE_MOD >= DC {
    result: DAMAGE_DICE / 2 \ Success: half damage, rounded down \
  }
  else {
    result: DAMAGE_DICE \ Failure: full damage \
  }
}

\ Output the combined probability density function \
output [dissonantwhispers 1d20] named "Dissonant Whispers Total Damage PDF"

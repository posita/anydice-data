\calculate damage per round\
function: dpr ROLL:n vs DEF:n DMG:n {
  if ROLL = 1 { result: 0 }
  if ROLL >= DEF { result: DMG }
  result: 0
}

\calculate the accuracy roll from lancer\
function: acc N:n {
   if N = 0 {result: 0}
   result: [highest of 1d6 and [acc (N - 1)]]
}

\ Define what a success looks like for the reroll phase (1 on a 6, 0 otherwise) \
HIT: d{0, 0, 0, 0, 0, 1}

\ The function takes the max rerolls and the sequence of the initial roll \
function: ROLL:s reroll LIMIT:n {
  SUCCESSES: ROLL = 1
  FAILURES: ROLL = 0
  REROLLS: [lowest of FAILURES and LIMIT]
  
  result: SUCCESSES + REROLLS d HIT
}

loop X over {4..8} {
loop N over {0..4} {

\ Execute the function: 4d6, rerolling up to 2 failures \
output [XdHIT reroll N] named "[X]d6, reroll up to [N] failures"

}
}

function: pbta STAT:n {
  result: (2d6+STAT > 6) + (2d6+STAT > 8) 
}

function: fitd ACTION:n {
  if ACTION = 0 {
    result: (2@2d6 > 3) + (2@2d6 > 5)
  } else {
    result: (1@ACTIONd6 > 3) + (1@ACTIONd6 > 5)
  }
}

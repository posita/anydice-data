MALF_ROLL: 3d6

function: N:n in range MIN:n to MAX:n {
  if N < MIN { result: 0 }
  if N > MAX { result: 0 }
  result: 1
}

function: malfunction for N:n {
  if [N in range 3 to 4] { result: 3 }
  if [N in range 5 to 8] { result: 2 }
  if [N in range 9 to 11] { result: 1 }
  if [N in range 12 to 14] { result: 2 }
  result: 4
}

\ 0 = No Malfunction \
\ 1 = Stoppage \
\ 2 = Misfire \
\ 3 = Mechanical or Electrical Problem \
\ 4 = Mech. or Elec. Problem OR Explosion \
loop MALF over {12, 14, 16, 17} {
  output (3d6 > MALF)*[malfunction for 3d6] named "Malf = [MALF]"
}

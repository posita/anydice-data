function: montroll amount N:n dice R:n streak S:n {
 NS: R = 4
 NCONT: N > 1
 \Continue if streak is alive or more dice have to be rolled\
 if NS | NCONT {
  result: R + [montroll amount N - 1 dice d4 streak NS & S]
 }
 else {
  result: R
 }
}

function: success rate R:n against DC:n {
 result: R >= DC
}

set "maximum function depth" to 7

ROLLF: [montroll amount 1 dice d4 streak 1]
ROLLD: [montroll amount 2 dice d4 streak 1]
ROLLC: [montroll amount 3 dice d4 streak 1]
ROLLB: [montroll amount 4 dice d4 streak 1]
ROLLA: [montroll amount 5 dice d4 streak 1]

output [success rate ROLLF against 3]

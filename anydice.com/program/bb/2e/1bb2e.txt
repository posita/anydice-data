function: pool SET:s has any values le SUCC:n {
  SUCCESSES: 0
  loop N over SET {
    DIEROLL: dN
    SUCCESSES: SUCCESSES + 1d (dN <= SUCC) 
  }
  result: SUCCESSES > 0
}

function: dupes in DICE:s {
  D: 0
  loop X over {2..#DICE} {
    if ((X-1)@DICE = X@DICE) { D: D + 1}
  }
  result: D > 0
}

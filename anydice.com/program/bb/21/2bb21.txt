function: resolve TENS:n ONES:n {
  if (TENS = 10) {
    if (ONES = 10) {
      result: 100
    }
    result: ONES
  }
 
  result: 10*TENS + (ONES-1)
}

function:  bonusone {
  ONES: 1d10
  FIRSTROLL: [resolve 1d10 ONES]
  SECONDROLL: [resolve 1d10 ONES]
  result: [lowest of FIRSTROLL and SECONDROLL]
}

function:  bonustwo {
  ONES: 1d10
  FIRSTROLL: [resolve 1d10 ONES]
  SECONDROLL: [resolve 1d10 ONES]
  THIRDROLL: [resolve 1d10 ONES]
  result: [lowest of [lowest of FIRSTROLL and SECONDROLL] and THIRDROLL]
}


function:  penaltyone {
  ONES: 1d10
  FIRSTROLL: [resolve 1d10 ONES]
  SECONDROLL: [resolve 1d10 ONES]
  result: [highest of FIRSTROLL and SECONDROLL]
}

function:  penaltytwo {
  ONES: 1d10
  FIRSTROLL: [resolve 1d10 ONES]
  SECONDROLL: [resolve 1d10 ONES]
  THIRDROLL: [resolve 1d10 ONES]
  result: [highest of [highest of FIRSTROLL and SECONDROLL] and THIRDROLL]
}

output [bonusone] named "Bonus 1"
output [bonustwo] named "Bonus 2"
output [resolve 1d10 1d10] named "Base"
output [penaltyone] named "Penalty 1"
output [penaltytwo] named "Penalty 2"

MISS: 0
HIT: 1
CRIT: 2

function: attack ROLL:n against AC:n
{
if ROLL = 20 { result: CRIT }
if ROLL + ATTACK_MOD < AC | ROLL = 1 { result: MISS }
result: HIT
}

function: attack X:n and Y:n against AC:n with advantage
{
if X = 20 | Y = 20 { result: CRIT }
if X = 1 & Y = 1 { result: MISS }
if X + ATTACK_MOD >= AC | Y + ATTACK_MOD >= AC { result: HIT }
result: MISS
}

function: damage ROLL:n
{
if ROLL = MISS {result: 0} 
if ROLL = HIT {result: DAMAGE_DIE + DAMAGE_MOD}
if ROLL = CRIT {result: [maximum of DAMAGE_DIE] + DAMAGE_DIE + DAMAGE_MOD}
}

function: damage barbarian ROLL:n
{
if ROLL = MISS {result: 0} 
if ROLL = HIT {result: DAMAGE_DIE + DAMAGE_MOD}
if ROLL = CRIT {result: [maximum of DAMAGE_DIE] + DAMAGE_DIE + DAMAGE_DIE + DAMAGE_MOD}
}

function: damage barbarian crit ROLL:n
{
if ROLL = MISS {result: 0} 
if ROLL = HIT {result: DAMAGE_DIE + DAMAGE_MOD}
if ROLL = CRIT {result: [maximum of DAMAGE_DIE] + DAMAGE_DIE + DAMAGE_DIE + DAMAGE_DIE + DAMAGE_DIE + DAMAGE_MOD}
}

function: damage pam ROLL:n
{
if ROLL = MISS {result: 0} 
if ROLL = HIT {result: PAM_DIE + DAMAGE_MOD}
if ROLL = CRIT {result: [maximum of PAM_DIE] + PAM_DIE + PAM_DIE + DAMAGE_MOD}
}

function: damage pam crit ROLL:n
{
if ROLL = MISS {result: 0} 
if ROLL = HIT {result: PAM_DIE + DAMAGE_MOD}
if ROLL = CRIT {result: [maximum of PAM_DIE] + PAM_DIE + PAM_DIE + PAM_DIE + PAM_DIE + DAMAGE_MOD}
}

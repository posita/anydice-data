\ roll AdX + BdY, target 4, where X is one size less than Y and A+B = 6 \
DICE: {4,6,8,12,20}
loop I over {1..(#DICE-1)} {
  X: I@DICE
  Y: (I+1)@DICE
  output 6d{0:3, 1:X-3} named "6d[X]"
  loop B over {1..5} {
    A: 6-B
    output Ad{0:3, 1:X-3} + Bd{0:3, 1:Y-3} named "[A]d[X] and [B]d[Y]"
  }
}
output 6d{0:3, 1:Y-3} named "6d[Y]"

function: advantage A disadvantage D {
  if A > D {
    result: [lowest 1 of (A-D+1)d20]
  }
  if D > A {
    result: [highest 1 of (D-A+1)d20]
  }
  result: d20
}

loop SUM over {0..3} {
  loop ADV over {0..SUM} {
    DIS: SUM - ADV
    output [advantage ADV disadvantage DIS] named "advantage [ADV], disadvantage [DIS]"
  }
}

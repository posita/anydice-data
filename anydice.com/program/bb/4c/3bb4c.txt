A: 12  \ Attribute \
D: 2  \ # Dices   \
MS:3  \ Min skill \
MX:6  \ Max skill \

function: roll S F {
  result: {2:F, 1:(A +S-F), 0:(20-(A+S)) }
}
if MS<3 {
  output Dd[roll 2 1] named "skill 1"
  output Dd[roll 2 1] named "skill 2"
} else if MS=3 {
  output Dd[roll MS 1] named "skill 3"
} else if MS=4 {
  output Dd[roll MS 1] named "skill 4"
} else if MS=5 {
  output Dd[roll MS 1] named "skill 5"
} else if MS=6 {
  output Dd[roll MS 1] named "skill 6"
}
loop S over {MS..MX} {
    output Dd[roll S S] named "focus"
    if S < 6 {
      output Dd[roll S+1 1] named "or skill +1"
    }
}

function: max S:s {result: 1@[sort S]}
function: max M:n N:n {result: [max {M, N}]}
function: min S:s {result: #S@[sort S]}
function: min M:n N:n {result: [min {M, N}]}
function: N:n vs T:n {
  if N > T {result: 1} if N < T {result: -1} result: 0
}
function: N:n vs L:n H:n {
  if N > H {result: 1} if N < L {result: -1} result: 0
}
function: F:s seqsum S:s {
  R: {}
  FSIZE: #F
  SSIZE: #S
  MAXI: [max FSIZE SSIZE]
  loop I over {1..MAXI} {
    R: {R, I@F + I@S}
  }
  result: R
}
function: F:s seqprod S:s {
  R: {}
  FSIZE: #F
  SSIZE: #S
  MAXI: [max FSIZE SSIZE]
  loop I over {1..MAXI} {
    R: {R, I@F * I@S}
  }
  result: R
}
function: N:n vs CM:n L:n H:n CH:n {
  if N >= CH {result: 2} if N <= CM {result: -2}
  if N > H {result: 1} if N < L {result: -1}
  result: 0
}
function: roll N:n D:n plus MOD:n with ADV:n {
  if ADV < 0 {result: {1-ADV..N-ADV}@(N-ADV)dD+MOD}
  result: {1..N}@(N+ADV)dD+MOD
}
function: roll N:n D:n with ADV:n {result: [roll N D plus 0 with ADV]}
function: roll N:n D:n plus MOD:n {result: [roll N D plus MOD with 0]}
function: F:s within N:n of S:s {
  result: [F within N N of S ]
}
function: F:s within M:n N:n of S:s {
  R: {}
  FSIZE: #F
  SSIZE: #S
  MAXI: [max FSIZE SSIZE]
  loop I over {1..MAXI} {
    CMP: I@F - I@S
    if CMP < M {R: {R, -1}}
    if CMP > N {R: {R, 1}}
    if CMP > M & CMP < N {R: {R, 0}}
  }
  result: R
}

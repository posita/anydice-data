DICE: 4d6 - 14
HIT: 2

function: roller VAL:n vdiv VDIV:n dmg DMG:n bonus BONUS:n {
  if VAL >= 0 {
    if VAL < 2 {
      TDMG: (VAL/VDIV)+(DMG/2)
      if TDMG < 1 {
        result: 1
      } else {
        result: TDMG
      }
    } else {
      if VAL >= 6 {
        result: (6/VDIV)+DMG+BONUS
      } else {
        result: (VAL/VDIV)+DMG
      }
    }
  } else {
     result: 0
  }
}

output [roller DICE+HIT+1 vdiv 1 dmg d6 bonus d6]
output [roller DICE+HIT vdiv 1 dmg 1 bonus 1] + [roller DICE+HIT vdiv 1 dmg 1 bonus 1]

\Step Die PbtA\
\1 Step Die for Attributes D6 to D12 (these don't change unless XP)\
\1 Step Die for Modifiers D6 to D20 (these change situationally)\
\Modifers: -2:d6, -1:d8, 0:d10, +1:d12, +2:d20\
\Results: 6+: 1 success, 10+: 2 successes\
\Complications: 1 success = success with complications\

function: stepdiepbta A:s M:s {
RA: [count {6..9} in A]+[count {10..20} in A]*2
RM: [count {6..9} in M]+[count {10..20} in M]*2
SUM : RA+RM
if SUM>2 {SUM:2}
result: SUM
}

\--1d6--\
output [stepdiepbta 1d6 1d6] named "Successes on d6,d6"
output [stepdiepbta 1d6 1d8] named "Successes on d6,d8"
output [stepdiepbta 1d6 1d10] named "Successes on d6,d10"
output [stepdiepbta 1d6 1d12] named "Successes on d6,d12"
output [stepdiepbta 1d6 1d20] named "Successes on d6,d20"

\--1d8--\

QUAL: {1:2, 2:4, 3:5, 4:4, 5:3, 6:7}
PROF: {2:6, 3:6, 4:6, 5:6, 6}

function: slam level L:n {
 result: (L@QUAL)d10+3
}

function: lay waste level L:n {
 result: (L@PROF)d6+3
}

function: sum of S:s {
 SUM: 0
 loop N over S {
  SUM: SUM + N
 }
 result: SUM
}

function: average of S:s {
 result: [sum of S]/#S
}

DIFFS: {1..25}

loop N over {1..25} {
 \output [slam level N]  named "Slam Level [N]"\
 \output [lay waste level N] named "Lay Waste Level [N]"\
 \output [slam level N]<[lay waste level N] named "Comparison Level [N]"\
 output [average of {[slam level N]-[lay waste level N]}] named "Subtraction Level [N]"
}

\output DIFFS\

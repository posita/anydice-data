function: hit N:n {
 result: (N)d{0,0,1}
}
function: hit N:n def M:n {
 result: [clamp ([hit N] - [hit M]) to 0 and 2]
}

function: clamp N:n to X:n and Y:n {
if (N > Y) { result: Y }
if (N < X) { result: X }
result: N
}

loop N over {1..5}{
loop M over {0..4}{
 output [hit N def M] named "hit [N] def [M]"
}
}

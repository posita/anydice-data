QUAL: {1:2, 2:4, 3:5, 4:4, 5:3, 6:7}
PROF: {2:6, 3:6, 4:6, 5:6, 6}

function: slam level L:n {
 result: (L@QUAL)d10+3
}

function: waste level L:n {
 result: (L@PROF)d6+3
}

loop N over {1..25} {
 output [slam level N]  named "Slam Level [N]"
 output [waste level N] named "Lay Waste Level [N]"
 \output [slam level N]<[waste level N] named "Comparison Level [N]"\
}

\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 17\
ENEMYAC: d{17}

ATTACKMOD: 4
DMGBONUS: 0
PROFICIENCY: 3
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 3
DAMAGE: 3*[highest 1 of 2d6] + [highest 1 of 1d4] + 4*ATTACKMOD + DMGBONUS
CRITDAMAGE: 6*[highest 1 of 2d6] + 2*[highest 1 of 1d4] + 4*ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Scimitar+Dagger"

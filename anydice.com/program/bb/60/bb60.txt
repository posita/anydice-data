

function: woundstarget STRENGTH:n vs TOUGHNESS:n {
  if STRENGTH < TOUGHNESS {
     if 2*STRENGTH <= TOUGHNESS {result: 6}
     result: 5}  
  if STRENGTH > TOUGHNESS {
     if STRENGTH >= 2*TOUGHNESS {result: 2}
     result: 3}
  result: 4
}


function: hitroll {
  if FLAMER > 0 {BS: 1}
  result: d6 >= BS
}

function: woundroll {
  result: d6 >= [woundstarget S vs T]
}

function: saveroll  {
  SAVE: SV-AP-CV 
  if INV < SAVE{result: d6 >= INV}
  result: d6 >= SAVE
}

function: damage {
  if MELTA > 0 {D: [highest 1 of 2d6]}
  result: D
}

\loop ATTEMPTS over {1..20} {
output ATTEMPTSd(Ad(([hitroll] * [woundroll] * ![saveroll])*D)) named "No. Attacks [ATTEMPTS]"
}\

function: limit A:n to B:n{
 if A > B {result: B}
 result: A
}

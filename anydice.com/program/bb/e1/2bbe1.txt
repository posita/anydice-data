function: lower N:n M:n {
if N < M {
result: N
}
else {
result: M
}
}
function: higher N:n M:n {
if N > M {
result: N
}
else {
result: M
}
}
function: luck {
result: [lower 3d6 [lower 3d6 3d6]]
}
function: advantage {
result: [higher 1d20 1d20]
}

output 3d6 named "GURPS"
output 21-[luck] named "GURPS Luck, inverted" \ Note that GURPS is roll low and D&D is roll high, subtracting the value from 21 turns the values into equivalents for a roll-high version \
output 1d100 / 5 named "d100 adjusted"
output 1d20 named "D&D"
output [advantage] named "D&D Advantage"

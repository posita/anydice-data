function: roll ROLL:s TN {

 if (1@ROLL >= 10) {
  A: 1@ROLL-9
 } else {
  A: 1@ROLL
 }
 
 if (1@ROLL >= 10) {
  B: 2@ROLL-9
 } else {
  B: 2@ROLL
 }
 
 if (1@ROLL >= 10) {
  C: 3@ROLL-9
 } else {
  C: 3@ROLL
 }

 R:0
 
if (#(ROLL) >= 1) {
 if (1@ROLL <= TN) {
  R: R+1
 } else {
  R: R-1
 }
}

if (#(ROLL) >= 2){
 if (2@ROLL <= TN) {
  R: R+1+(A=B)
 } else {
  R: R-1-(A=B)
 }
}

if (#(ROLL) >= 3) {
 if (3@ROLL <= TN) {
  R: R+1+(A=C)
 }else {
  R: R-1-(A=C)
 }
}

 result: R
}

loop D over {2..2}{
 loop N over {1..20} {
  output [roll Dd20 N] named "[D]v[N]"
 }
}

function: atk ATK:n vs DEF:n res DC:n vs RES:n {
  if  ATK >= DEF { 
    result: [res DC vs RES]
  }
  result: 0
}

function: multiatk ATK:n vs DEF:n res DC:n vs RES:n {
  if  ATK >= DEF + 10 { 
    result: [res DC + 5 vs RES]
  }
  if  ATK >= DEF + 5 { 
    result: [res DC + 2 vs RES]
  }
  if  ATK >= DEF { 
    result: [res DC vs RES]
  }
  result: 0
}

function: res DC:n vs RES:n{
  if  DC > RES {
     if (1 = 1) {result: 1} 
     if (0 = 1) {result: DC - RES} 
     result: ((DC - RES - 1) / 5) + 1
  }
  result: 0
}

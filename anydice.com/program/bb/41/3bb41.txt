A: 14 \ Attribute \
D: 2 \ Dices \

function: roll T F {
  result: {2:F, 1:(A +T-F), 0:(20-(A+T+F)) }
}
output Dd[roll 1 1] named "skill 1"
output Dd[roll 2 1] named "skill 2"


loop T over {2..6} {
    output Dd[roll T T] named "focus"
    if T < 6 {
      output Dd[roll T+1 1] named "skill +1"
    }
}

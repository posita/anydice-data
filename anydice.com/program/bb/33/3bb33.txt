ANTI_MISSILE_STRENGTH: 40
ANTI_MISSILE_VELOCITY: 360
ANTI_MISSILE_RELOAD_XTEN: 200
MISSILE_STRENGTH: 100
MISSILE_ACCELERATION_XHUNDRED: 145
MISSILE_DRAG_XHUNDRED: 21


MISSILE_VELOCITY: MISSILE_ACCELERATION_XHUNDRED / MISSILE_DRAG_XHUNDRED
NUM_HITS: ANTI_MISSILE_VELOCITY / MISSILE_VELOCITY * 600 / ANTI_MISSILE_RELOAD_XTEN

function: antimissile check {
result: 1dANTI_MISSILE_STRENGTH+1>1dMISSILE_STRENGTH+1
}

C: 0

loop N over {1..NUM_HITS} {
C: C+[antimissile check]
}

output C

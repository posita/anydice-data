FAILURE: 0
MODERATESUCCESS: 1
GREATERSUCCESS: 2
CRITICALSUCCESS: 3

function: interpret ROLL:n{
if ROLL <=10 {result: FAILURE}
if ROLL >=11 & ROLL <= 20 {result: MODERATESUCCESS}
if ROLL >= 21 & ROLL <= 30 {result: GREATERSUCCESS}
if ROLL >= 31 {result: CRITICALSUCCESS}
}

loop N over {0..12}{
output [interpret 2d12+N] named "Chances at a +[N]"
}

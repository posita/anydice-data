set "explode depth" to 2

MAXWOUNDS: 4

EDF: [explode d4]
EDS: [explode d6]
EDA: [explode d8]
EDTE: [explode d10]
EDTW: [explode d12]



\ RESULTS are: -1 for no damage, 0 for shaken, and 1,2,3,etc for number of WOUNDS \ 
function: roll DMG:n damage vs toughness TOUGH:n {
  if DMG < TOUGH { result: -1 }
  if (DMG-TOUGH)/4 > MAXWOUNDS { result: MAXWOUNDS }
  result: (DMG-TOUGH)/4
}

\ RESULTS are: -1 for critfail, 0 for miss, 1 for hit, and 2 for raise \
function: roll ATK:n attack vs defense DEF:n {
  if ATK = 1 { result: -2 }       \ CRITFAIL \
  if ATK < DEF { result: -1 }     \ NORMAL MISS \
  if ATK - DEF >= 4 { result: 1 } \ RAISE \
  result: 0                       \ normal hit \
}


function: ATK:n attack with DMG:n damage vs defense DEF:n and tougness TOUGH:n {
  ATKRESULT: [roll ATK attack vs defense DEF]
  if ATKRESULT = -2 { result: -2 }              \ CRITFAIL \
  if ATKRESULT = -1 { result: -1 }              \ NORMAL MISS, no DMG \
  if ATKRESULT >=1 { DMG: DMG + [explode d6] }  \ RAISE, so +1d6 damage \

  result: [roll DMG damage vs toughness TOUGH] \ Roll damage \
}

function: evaluate ROLL:s {
  SUCCESSES: 0
  if ROLL = 6 { SUCCESSES: 2 }
  if ROLL = 4 | ROLL = 5 { SUCCESSES: 1 }
  if ROLL = 1 & [count {6} in ROLL] = 0 { SUCCESSES: -1 }

  if SUCCESSES >= 2 { result: 2 }
  else if SUCCESSES = 1 { result: 1 }
  else { result: 0 }
}

loop DICE over {1..3} {
  output [evaluate DICE d6] named "[DICE]D"
}

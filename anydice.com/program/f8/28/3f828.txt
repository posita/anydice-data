E: {0, 1, 1}
F: {-1, 0, 1}

function: roll DICE:n TARGET:n {
  CRIT: DICE >= (TARGET + 3)
  HIT: DICE >= TARGET
  result: HIT + CRIT
}

output 4dF named "Normal Roll"
output 3dF+1dE named "Replaced Roll"

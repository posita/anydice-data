\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 19-20\
ENEMYAC: d{18..18}

ATTACKMOD: 4
DMGBONUS: 4
PROFICIENCY: 0
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 2 * 5d12 + 1d12 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 4 * 5d12 + 1d12 + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Greataxe"

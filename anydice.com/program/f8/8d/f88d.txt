\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: sixgwfight ROLLX:n {
 if ROLLX <= 2 { result:d6 }
 result: ROLLX
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 8
DMGBONUS: 8
PROFICIENCY: 6
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 2d6 + DMGBONUS
CRITDAMAGE: 4d6 + DMGBONUS
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Maul with GWF"

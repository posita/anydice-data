function: positive N {
if N <= 0 {
result: 1
}
result: N
}

function: damage D:n SKILL {
if D <= SKILL {
result: [positive D*2]
}
}

function: limited D:n LIMIT {
if D > LIMIT {
result: LIMIT
}
result: D
}

LIMIT: 8

loop SKILL over {3..7} {
output [limited [damage d10 SKILL] LIMIT] named "Skill [SKILL]"
}

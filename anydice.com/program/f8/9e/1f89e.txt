function: positive N {
if N <= 0 {
result: 1
}
result: N
}

function: melee D:n SKILL LIMIT {
if D < SKILL {
result: [positive D]
}
if D = SKILL {
result: LIMIT
}
}

function: firearm D:n SKILL LIMIT {
if D < SKILL {
result: [positive D]*2
}
if D = SKILL {
result: LIMIT
}
}

function: limited D:n LIMIT {
if D > LIMIT {
result: LIMIT
}
result: D
}


LIMIT: 6

loop SKILL over {1..10} {
output [limited [firearm d10 SKILL LIMIT] LIMIT] named "Skill [SKILL]"
}

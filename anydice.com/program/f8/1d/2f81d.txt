  R: 0
  if PLAYER >= LOCATION & PLAYER = BONUS { R: 4 }     \ Critical Player Win \
  else if PLAYER >= LOCATION { R: 3 }      \ Player Win \
  \else if PLAYER = CL { R: 0 }\    \ Critical Player Loss defined globally like 2-4 \

  else { \ Loss Scenarios -- Police Rolls \
    if POLICE >= JAIL & POLICE = HIT { R: 0 }  \ Critical Catch from Police \
    else if POLICE >= JAIL { R: 1 }   \ Police Win, Player is caught - deal loss \
    else { R : 2 }       \ Neutral - Player escapes \
  }
  result: R
}

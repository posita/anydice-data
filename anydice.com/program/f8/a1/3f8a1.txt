HORSEA: 1d8
HORSEB: 1d8

LANCEA: 1d4
LANCEB: 1d4

RIDERA: 1d12
RIDERB: 1d12

function: jousta HORSEA:n LANCEA:n RIDERA:n LANCEB:n {
  if LANCEB > HORSEA {RIDERA: 0}
  result: (HORSEA + LANCEA + RIDERA)
}

function: joustb HORSEB:n LANCEB:n RIDERB:n LANCEA:n {
  if LANCEA > HORSEB {RIDERB: 0}
  result: (HORSEB + LANCEB + RIDERB)
}

function: joust HORSEA:n LANCEA:n RIDERA:n HORSEB:n LANCEB:n RIDERB:n{
  TEMPA: RIDERA
  TEMPB: RIDERB
  if LANCEB > HORSEA {TEMPA: 0}
  if LANCEA > HORSEB {TEMPB: 0}  
  result: (HORSEA + LANCEA + TEMPA) - (HORSEB + LANCEB + TEMPB)
}

output [jousta HORSEA LANCEA RIDERA LANCEB]
output [joustb HORSEB LANCEB RIDERB LANCEA]
output [joust HORSEA LANCEA RIDERA HORSEB LANCEB RIDERB]

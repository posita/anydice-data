function: prov R:n D:n F:n C:n {
  if R = 0 {
    if D = C {
      R: R + 1
      result: [prov R FdC F C]
    }
    if D = F {
      R: R - 1
      result: [prov R FdC F C]
    }
  }
  if R > 0 {
    if D = C {
      R: R + 1
      result: [prov R FdC F C]
    }
  }
  if R < 0 {
    if D = F {
      R: R - 1
      result: [prov R FdC F C]
    }
  }
  result: R
}
output [prov 0 1d6 1 6]
output [explode 1d6] named "dmg 6"
output 3d4 named "chance"
output [explode 1d4] named "dmg 4"

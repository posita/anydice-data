function: dctwenty ATTACK:n vs DC:n with DAMAGE:n and BONUS:n heavy HEAVY:n{
  if ATTACK = 1 {
    result: 0
  }
  
  else if ATTACK = 20 {
    result: 2 + [strike (DAMAGE) attack result (ATTACK + BONUS) vs (DC) heavy HEAVY]
  }

  else if ATTACK + BONUS >=DC{
    result: [strike (DAMAGE) attack result (ATTACK + BONUS) vs (DC) heavy HEAVY]
  }

  else if ATTACK+BONUS < DC {
    result: 0
  }  
}

function: strike DAMAGE:n attack result OUTCOME:n vs DC:n heavy HEAVY:n{
  if OUTCOME<DC{
    result: 0
  }else if OUTCOME >=DC+5{
    result: DAMAGE + ((OUTCOME - DC) / 5) + HEAVY
  }else{
    result: DAMAGE + ((OUTCOME - DC) / 5)
  }
}

DAMAGE_DIE: 3
ATTACK_MODIFIER: 7
TARGET_DC: 20
HEAVY_DAMAGE: 1

output [dctwenty [highest 1 of 1d20] vs TARGET_DC with DAMAGE_DIE and ATTACK_MODIFIER heavy HEAVY_DAMAGE] named "Attack vs DC"
output [dctwenty [highest 1 of 2d20] vs TARGET_DC with DAMAGE_DIE and ATTACK_MODIFIER heavy HEAVY_DAMAGE] named "ADV vs DC"
output [dctwenty [highest 1 of 1d20] vs TARGET_DC with DAMAGE_DIE and (ATTACK_MODIFIER+1d8) heavy HEAVY_DAMAGE] named "1d8 vs DC"
output [dctwenty [highest 1 of 2d20] vs TARGET_DC with DAMAGE_DIE and (ATTACK_MODIFIER+1d8) heavy HEAVY_DAMAGE] named "ADV + 1d8 vs DC"

function: chime A:n {
  if A >= 3 {result: 1}
  if A >= 2 {result: 0}
  result: -1
}

function: blades A:n {
  if A >= 6 {result: 1}
  if A >= 4 {result: 0}
  result: -1
}

function: bten A:n {
  if A >= 10 {result: 1}
  if A >= 7 {result: 0}
  result: -1
}

function: test A:n {
  if A >= 10 {result: 1}
  if A >= 8 {result: 0}
  result: -1
}

\loop D over {2..7} {\
\  output [test [highest {2} of Dd6]] named "[D]d6k2"\
\}\

loop D over {1..7} {
  output [bten [highest {1} of Dd10]] named "blades but d10s: [D]d10"
}



\loop D over {2..6} {\
\  B: D-2\
\  output [chime [count {4..6} in Dd6]] named "chime [D]d6; blades equiv-[B]"\
\}\


output [blades [lowest {1} of 2d6]] named "blades 0d6"
loop D over {1..5} {
  output [blades [highest {1} of Dd6]] named "blades [D]d6"
}

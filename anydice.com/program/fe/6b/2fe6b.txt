function: score SKILL:n and U:s and V:s and W:s and X:s and Y:s {
  A:[absolute SKILL-U]
  B:[absolute SKILL-V]
  C:[absolute SKILL-W]
  D:[absolute SKILL-X]
  E:[absolute SKILL-Y]
\score is taken as the closest roll to the SKILL target number\
  RES:U
  MIN:A
  if B<MIN {MIN:B}  if C<MIN {MIN:C}  if D<MIN {MIN:D}  if E<MIN {MIN:E}
  if MIN=A { RES:U
    if A=B & U<V {result:V}
    if A=C & U<W {result:W}
    if A=D & U<X {result:X}
    if A=E & U<Y {result:Y}
  }
  if MIN=B { RES:V
    if B=A & V<U {result:U}
    if B=C & V<W {result:W}
    if B=D & V<X {result:X}
    if B=E & V<Y {result:Y}
  }
  if MIN=C { RES:W
    if C=A & W<U {result:U}
    if C=B & W<V {result:V}
    if C=D & W<X {result:X}
    if C=E & W<Y {result:Y}
  }
  if MIN=D { RES:X
    if D=A & X<U {result:U}
    if D=B & X<V {result:V}
    if D=C & X<W {result:W}
    if D=E & X<Y {result:Y}
  }
  if MIN=E { RES:Y
    if E=A & Y<U {result:U}
    if E=B & Y<V {result:V}
    if E=C & Y<W {result:W}
    if E=D & Y<X {result:X}
  }
  result:RES
}

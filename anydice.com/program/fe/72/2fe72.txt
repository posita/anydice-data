function: evaluate R:s {
 C: 1000
 F: 1000
 loop N over R {
  if N >= 0 {
   if N <= C {
    C: N
    F: N
   }
  }
  else if -N < C {
   C: -N
   F: N
  }
 }
 result: F
}

function: evaluate A:s and B:s {
 R: {A, B}
 C: 1000
 F: 1000
 loop N over R {
  if N >= 0 {
   if N <= C {
    C: N
    F: N
   }
  }
  else if -N < C {
   C: -N
   F: N
  }
 }
 result: F
}

function: ignore N:n outside R:n {
 if -R <= N & N <= R {
  result: N
 }
 result: 1000
}

SKILL: 3
DA: d8 - SKILL
DB: [ignore d20 - 10 - SKILL outside [maximum of DA]]

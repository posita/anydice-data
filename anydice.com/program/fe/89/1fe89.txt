
function: attack ATT:n defence DEF:n armour AC:n {
    \ Check if attack roll is higher than defence roll \
    if(ATT > DEF) {
        \ Damage is attack minus opponents armour \        
        DAMAGE: (ATT - AC) 

        \ Damage cannot be less than zero \
        result: [highest of DAMAGE and 0]
    } 
    \ Attack roll failed, 0 damage \
    result: 0
}

output [attack 1d20 + 3 defence 1d20 + 3 armour 11] named "grenade"
output [attack 1d20 + 7 defence 1d20 + 3 armour 11] named "elemental bolt"

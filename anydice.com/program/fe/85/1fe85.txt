function: ROLL:s twice {
 result: ROLL = 1
}


\ output 1@3d20 = 1 named "super" \
\ output [2d20 twice] named "twice" \


function: attack R:n vs ac A:n prof P:n score S:n {
 if R = 1 {result: 0}
 if R >= 19 {result : 2d8 + S}
 if R+P+S > A {result : 1d8 + S}
 result: 0
}

loop A over {15..30}{
  SUPER: [attack 1@3d20 vs ac A prof 3 score 3]
  TWICE: [attack 1d20 vs ac A prof 3 score 3] + [attack 1d20 vs ac A prof 3 score 3]
  \output SUPER named "one attack - super advantage vs [A]"\
  \output TWICE named "two attack vs [A]"\

  output  TWICE - SUPER named "[A]"



}

function: hgiant ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

ATTACKMOD: 5
DMGBONUS: 0
PROFICIENCY: 3
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 3d8 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 6d8 + ATTACKMOD + DMGBONUS


output ATTACKTIMESd[hgiant d20 vs 15] named "vs Mollymauk"
output ATTACKTIMESd[hgiant d20 vs 18] named "vs Fjord"

output ATTACKTIMESd[hgiant d20 vs 16] named "vs Mollymauk with cloak"
output ATTACKTIMESd[hgiant d20 vs 17] named "vs Fjord without cloak"

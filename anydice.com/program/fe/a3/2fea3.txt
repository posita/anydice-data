\separate weapon dmg and other sources\
\crit\
\variable crit number\
\second dmg source with separate attacks amount (e.g. maneuvers)\
ATTACKS:6
ADVANTAGE: 1
TO_HIT_FLAT: 6
DEFENSE: 18
WEAPON_DMG_DIE: 1d6
EXTRA_DMG_DIE: 0
FLAT_DMG: 5 + 10 + 4
MIN_CRIT_PROC: 19

function: if HIT:n hit if CONDITION:n crit {
  if CONDITION {
    result: 2
  }
  else {
    if HIT {result: 1} else {result: 0}
  }
}

HIT_RESULT: 0
HIT_ROLL: d20
if ADVANTAGE = 1 {
  HIT_ROLL: [highest 1 of 2d20]
}

\hit chance\
HIT_RESULT: HIT_ROLL + TO_HIT_FLAT >= DEFENSE
CRIT: HIT_ROLL >= MIN_CRIT_PROC
output [if HIT_RESULT hit if CRIT crit] named "crit chance: 0 miss, 1 hit, 2 crit"

\normal dmg\
DAMAGE: WEAPON_DMG_DIE + EXTRA_DMG_DIE + FLAT_DMG
DAMAGE: DAMAGE * ATTACKS
output DAMAGE named "normal dmg"

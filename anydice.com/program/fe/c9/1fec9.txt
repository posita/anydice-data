function: savethrow SAVE:n dc DC:n {
if SAVE >= DC {result: 0}
if SAVE < DC {result: 2d12}
}

function: attack DICE:n plus TOHIT:n ac AC:n bless BLESS:n{
if DICE = 20 {result: 2d10}
if (TOHIT + DICE + BLESS) >= AC {result: 2d10}
if (TOHIT + DICE + BLESS) < AC {result: 0}
}

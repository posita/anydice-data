function: dctwenty ATTACK:n vs DC:n with DAMAGE:n and BONUS:n {
  if ATTACK = 1 {
    result: 0
  }
  
  if ATTACK = 20 {
    if 20+BONUS<DC{
        result: 2
    }
    result: 2 + DAMAGE + ((ATTACK + BONUS - DC) / 5)
  }
  
  if ATTACK+BONUS < DC {
    result: 0
  }
  
  result: DAMAGE + ((ATTACK + BONUS - DC) / 5)
}

DAMAGE_DIE: 1
ATTACK_MODIFIER: 5
TARGET_DC: 14

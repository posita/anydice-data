Z: {1,2}@2d8+2

function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}

function: point buy eq AN:n first{
MAXA:1@{AN,25-AN}
if MAXA<14{result:9}
else{if MAXA=14{result:10}
if MAXA=15{result:11}
if MAXA=16{result:13}
if MAXA=17{result:15}
if MAXA=18{result:18}}
}

function: point buy eq BN:n second{
MAXB:1@{BN,23-BN}

if MAXB<14{result:7}
else{if MAXB=14{result:8}
if MAXB=15{result:9}
if MAXB=16{result:11}
if MAXB=17{result:13}
if MAXB=18{result:16}}
}

function: point buy eq CN:n third{
MAXC:1@{CN,21-CN}

if MAXC<14{result:5}
else{if MAXC=14{result:6}
if MAXC=15{result:7}
if MAXC=16{result:9}
if MAXC=17{result:11}
if MAXC=18{result:14}}
}

function: AN:n BN:n CN:n first {
A: 1@[sort{AN,BN,CN}]
B: 2@[sort{AN,BN,CN}]
C: 3@[sort{AN,BN,CN}]
if C<7{if B<7{result:A}
else{result:B}}
else{result:C}}

function: AN:n BN:n CN:n second {
A: 1@[sort{AN,BN,CN}]
B: 2@[sort{AN,BN,CN}]
C: 3@[sort{AN,BN,CN}]
if C<7{if B<7{result:C}}
else{result:B}}


function: AN:n BN:n CN:n third {
A: 1@[sort{AN,BN,CN}]
B: 2@[sort{AN,BN,CN}]
C: 3@[sort{AN,BN,CN}]
if C<7{if B<7{result:B}
else{result:A}}
else{result:A}}


function: total point buy eq of A:n and B:n and C:n{
AN: [A B C first]
BN: [A B C second]
CN: [A B C third]

PBA: [point buy eq AN first]
PBB: [point buy eq BN second]
PBC: [point buy eq CN third]



result:PBA+PBB+PBC

}

output [total point buy eq of Z and [Z reroll 4 as d{}] and [Z reroll {4..6} as d{}]]

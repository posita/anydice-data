\**** Damage calculator normal Vs advantage ****\

function: gsattackroll ROLL:n vs AC:n {
if ROLL >= CRITROLL { result: GSCRITDAMAGE }
if ROLL = 1 { result: 0 }
if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: GSDAMAGE }
result: 0
}
function: gaattackroll ROLL:n vs AC:n {
if ROLL >= CRITROLL { result: GACRITDAMAGE }
if ROLL = 1 { result: 0 }
if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: GADAMAGE }
result: 0
}
function: ROLL:n gwf REROLL:d reroll SEQUENCE:s{
  if [SEQUENCE contains ROLL] {
    result: REROLL
  }
  else {
    result: ROLL
  }
}
GREATSWORD_DIE_ONE: [d6 gwf d6 reroll {1,2}]
GREATSWORD_DIE_TWO: [d6 gwf d6 reroll {1,2}]
GREATSWORD_DIE_ONE_CRIT: [d6 gwf d6 reroll {1,2}]
GREATSWORD_DIE_TWO_CRIT: [d6 gwf d6 reroll {1,2}]
GREATSWORD_DIE_THREE_CRIT: [d6 gwf d6 reroll {1,2}]
GREATAXE_DIE_ONE: [d12 gwf d12 reroll {1,2}]
GREATAXE_DIE_ONE_CRIT: [d12 gwf d12 reroll {1,2}]
GREATAXE_DIE_TWO_CRIT: [d12 gwf d12 reroll {1,2}]

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 5
DMGBONUS: 5
PROFICIENCY:4
ACCBONUS: 0
CRITROLL: 19
ATTACKTIMES: 2

GSDAMAGE: GREATSWORD_DIE_ONE + GREATSWORD_DIE_TWO + ATTACKMOD + DMGBONUS
GSCRITDAMAGE: GREATSWORD_DIE_ONE + GREATSWORD_DIE_TWO + GREATSWORD_DIE_ONE_CRIT + GREATSWORD_DIE_TWO_CRIT + GREATSWORD_DIE_THREE_CRIT + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[gsattackroll [highest 1 of 2d20] vs ENEMYAC] named "GREATSWORD Attack w/ advantage"

GADAMAGE: GREATAXE_DIE_ONE + ATTACKMOD + DMGBONUS
GACRITDAMAGE: GREATAXE_DIE_ONE + GREATAXE_DIE_ONE_CRIT + GREATAXE_DIE_TWO_CRIT + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[gaattackroll [highest 1 of 2d20] vs ENEMYAC] named "GREATAXE Attack w/ advantage"

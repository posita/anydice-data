function: evaluate A:s and B:s {
R: {A, B}
C: 1000
F: 1000
loop N over R {
  if N >= 0 {
    if N <= C {
      C: N
      F: N
    }
  }
  else if -N < C {
    C: -N
    F: N
    }
  }
  result: F
}

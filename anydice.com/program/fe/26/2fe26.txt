Z: 2d8+2

function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  else{result: ROLL}
}

function: point buy eq AI:n first{
MAXA:1@[sort {AI,25-AI}]
if MAXA<14{result:9}
else{if MAXA=14{result:10}
if MAXA=15{result:11}
if MAXA=16{result:13}
if MAXA=17{result:15}
if MAXA=18{result:18}
if MAXA>18{result:10000}}
}

function: point buy eq BI:n second{
MAXB:1@[sort {BI,23-BI}]

if MAXB<14{result:7}
else{if MAXB=14{result:8}
if MAXB=15{result:9}
if MAXB=16{result:11}
if MAXB=17{result:13}
if MAXB=18{result:16}
if MAXB>18{result:0}}
}

function: point buy eq CI:n third{
MAXC:1@[sort {CI,21-CI}]

if MAXC<14{result:5}
else{if MAXC=14{result:6}
if MAXC=15{result:7}
if MAXC=16{result:9}
if MAXC=17{result:11}
if MAXC=18{result:14}
if MAXC>18{result:200000}}
}

function: AN:n BN:n CN:n first {
ITEMA: 1@[sort {AN,BN,CN}]
ITEMB: 2@[sort {AN,BN,CN}]
ITEMC: 3@[sort {AN,BN,CN}]

if ITEMC<7{if ITEMB<7{result:ITEMA}
else{result:ITEMB}}
else{result:ITEMC}
}

function: AN:n BN:n CN:n second {
ITEMA: 1@[sort {AN,BN,CN}]
ITEMB: 2@[sort {AN,BN,CN}]
ITEMC: 3@[sort {AN,BN,CN}]

if ITEMC<7{if ITEMB<7{if ITEMC<5{result:ITEMB}
else{result:ITEMC}}
else{if ITEMC<5{result:ITEMA}
else{result:ITEMC}}}
else{if ITEMB<5{result:ITEMA}
else{result:ITEMB}}
}

function: AN:n BN:n CN:n third {
ITEMA: 1@[sort {AN,BN,CN}]
ITEMB: 2@[sort {AN,BN,CN}]
ITEMC: 3@[sort {AN,BN,CN}]

if ITEMC<7{if ITEMB<7{if ITEMC<5{result:ITEMC}
else{result:ITEMB}}
else{if ITEMC<5{result:ITEMC}
else{result:ITEMA}}}
else{if ITEMB<5{result:ITEMB}
else{result:ITEMA}}
}

}

function: total point buy eq of A:n and B:n and C:n {
AOPT: [A B C first]
BOPT: [A B C second]
COPT: [A B C third]

PBA: [point buy eq AOPT first]
PBB: [point buy eq BOPT second]
PBC: [point buy eq COPT third]

result:PBA+PBB+PBC

}


output [total point buy eq of Z and [Z reroll 4 as d{}] and [Z reroll {4..6} as d{}]]

function: ROLL:n vs AC:n {
 if ROLL = 20 {
  result: 2d DAMAGE
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

 if ROLL = 20 {
  result: OD + 2d DAMAGE
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE + OD}
 result: 0
}

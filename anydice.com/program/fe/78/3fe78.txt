function: count VALUES:s in A:s and B:s {
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = A)
  COUNT: COUNT + (P@VALUES = B)
 }
 result: COUNT
}

}

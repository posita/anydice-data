
function : attack DICE:s vs AC:n
{
	if DICE = 20 { result : 2 }
	else if DICE = 1 { result : 0 }
	else if DICE >= (AC) { result : 1 }
	else {result : 0}
}

HPS : {6, 12, 8, 10}
ACS : {14, 15, 16, 18}

loop N over {1..#HPS}
{
	output N@HPS - ([attack 1d20 + 5 vs N@ACS]d10 + [attack 1d20 + 3 vs N@ACS]d6)
}

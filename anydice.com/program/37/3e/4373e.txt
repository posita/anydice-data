function: simplify D:n dice plus A:n {
  if A >= 7 { result: [simplify D+2 dice plus A-7] }
  if A >= 4 { result: [simplify D+1 dice plus A-4] }
  result: Dd6 + A
}

function: abs N:n {
  if N < 0 { result: -N }
  result: N
}

loop D over {1..3} {
  loop ADD over {4..11} {
    ROLL: Dd6+ADD
    APPROX: [simplify D dice plus ADD]
    \calculate error\
    output ROLL-APPROX named "[D]d+[ADD]"
  }
}

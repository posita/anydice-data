function: compare DX:n and DY:n to X:n and Y:n with A:n {
 H: 0

 if A > 0 { \ Extra chance to succeed on Advantage \
  if  DX = DY {
   result : H+2
  }
 }

 if A < 0 { \ Extra chance to miss on Disadvantage \
  if  DX = DY {
   result : H
  }
 }

 if DX+DY >= X { \ Success \
  H: H+1
 }
 if DX+DY >= Y { \ Full Success \
  H: H+1
 }
 result : H

}

\ Variable used to set an offset for the curves. Set to 1 or 0 \
O:1

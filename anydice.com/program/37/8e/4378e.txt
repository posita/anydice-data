NUM: 4
A: 1000
B: 100
C: 10
D: 1
SYMBOLS: {A, B, C, D}
RESULT: NUMd{A, B, B, C, C, D}
GOAL: {0, 2, 0, 1}

function: check result {
 loop S over {1..#SYMBOLS}{
  if ([count S@SYMBOLS in RESULT] < S@GOAL) {
   result: 0
  }
 }
 result: 1
}

output [count C in RESULT]
output ([count 1@SYMBOLS in RESULT] >= 1@GOAL) & ([count 2@SYMBOLS in RESULT] >= 2@GOAL) & ([count 3@SYMBOLS in RESULT] >= 3@GOAL) & ([count 4@SYMBOLS in RESULT] >= 4@GOAL)
output RESULT

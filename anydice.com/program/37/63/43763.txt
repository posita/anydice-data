function: crit ROLL:s {
  SIX: [count {6} in ROLL]
  FIVE: [count {5} in ROLL]
  result: (SIX >= 2) | ((SIX = 1) & (FIVE >= 1))
}

function: dmg raw ROLL:s {
  if [count {6} in ROLL] >= 2 { result: ROLL + 5 }
  result: ROLL + 0
}

function: dmg promoted ROLL:s {
  if [crit ROLL] { result: ROLL + 5 }
  result: ROLL + 0
}

output 5d6 named "baseline, no crit"
output [dmg raw 5d6] named "RAW crit"
output [dmg promoted 5d6] named "promoted crit"

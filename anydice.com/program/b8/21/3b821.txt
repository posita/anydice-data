function: die size DIE:d {
result: (1@[reverse {DIE}])/(1@{DIE})
}
function: h R:n pl P:n for D:d plus B:n ag A {
if R>=20 {result: D+D+B}
if (R+P)>A {result: D+B}
result: 0
}
function: hit ARMOR at plus PLUS for DAMAGE plus BONUS damage {
result: [h d20 pl PLUS for DAMAGE plus BONUS ag ARMOR]
}

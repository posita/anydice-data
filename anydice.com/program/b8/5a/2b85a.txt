function: un R:n {
 if (R=20) {
  result: WPN + WPN + DEX + (SNK*2)d6
 }
 if (R+BNS>=AC) {
  result: WPN + DEX + SNKd6
 }
 result: SNKd6
}

function: normal R:n {
 
 if (R=20) {
  result: WPN+WPN + DEX + (SNK*2)d6
 }
 if (R=1) { result: 0 }
 if (R+BNS>=AC) {
  result: WPN+DEX+SNKd6
 }
 result: 0 
}

function: uss R:n {
 if (R=20) {
  result: WPN + WPN + DEX + (SNK*2)d6 + 10
 }
 if (R+BNS-5>=AC) {
  result: WPN + DEX + SNKd6 + 10
 }
 result: SNKd6
}

function: ss R:n {
 
 if (R=20) {
  result: WPN + WPN + DEX + (SNK*2)d6 + 10
 }
 if (R=1) { result: 0 }
 if (R+BNS-5>=AC) {
  result: WPN + DEX + SNKd6 + 10
 }
 result: 0 
}

WPN:1d8
SNK:3
DEX:3
PRF:3
BNS:DEX+PRF
CNT:2

N:0
U:0
S:1
US:1
C:0
CS:0

loop AC over {15..20} {
 if (N) {
 output [normal 1@CNTd20] named "Normal [AC]"
 }
 if (S) {
 output [ss 1@CNTd20] named "Sharpshooter [AC]"
 }
}
loop AC over {15..20} {
 if (U) {
 output [un 1@CNTd20] named "Unerring [AC]"
 }
 if (US) {
 output [uss 1@CNTd20] named "Unerring Sharpshooter [AC]"
 }
}
loop AC over {15..20} {
 if (C) {
  output [un 1@CNTd20]-[normal 1@CNTd20] named "Comparison [AC]"
 }
 if (CS) {
  output [uss 1@CNTd20]-[ss 1@CNTd20] named "Sharpshooter Comparison [AC]"
 }
}

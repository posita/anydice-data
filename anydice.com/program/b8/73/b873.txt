function: highest N:n of A:s and B:s and C:s {
  result: {1..N} @ [sort {A, B, C}]
}

function: lowest N:n of A:s and B:s and C:s {
  LEN: #A + #B + #C
  result: {LEN-N+1 .. LEN} @ [sort {A, B, C}]
}

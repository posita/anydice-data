function: fate as fours {
  D: 1d4 - 1d4
  result: D + (D/3) * (1d4/4)
}
function: fate with twenties D:n {
  if D<3 { result: -4 }
  if D<5 { result: -3 }
  if D<7 { result: -2 }
  if D<9 { result: -1 }
  if D>18 { result: +4 }
  if D>16 { result: +3 }
  if D>14 { result: +2 }
  if D>12 { result: +1 }
  result: 0
}
output 4d{-1..1} named "4dF"
output [fate as fours] named "1d4-1d4 w/crits"
output 3d6/2 -5 named "3d6 w/math"
output 2d{0,1} + 2d{-1,0} + 2d{-1,1} named "6cF"
output [fate with twenties 2@3d20] named "3d20 w/breaks"

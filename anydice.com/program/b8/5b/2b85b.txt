function: digit N:n {
  result: (1d10-1)*10^(N-1)
}

function: for total M:n digits {
  X: 0

  loop I over {1..M}{
    X: X + [digit I]
  }

  result: X
}

function: reroll R:n digits above M:n{
  if R > M{
    result: -1
  } else {
    result: R
  }
}


output [reroll [for total 2 digits] digits above 62]

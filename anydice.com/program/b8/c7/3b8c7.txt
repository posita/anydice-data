\\\*** Base Damage Calc Info ***\\\
ENEMY_AC: 27
LEVEL: 9
BLADE_WARD: 0
HEAVY_ARMOR_MASTER: 1
PROF: (LEVEL+7)/4


\\\*** Attack A ***\\\
ATTACK: 9
DICE_A: 2d6+1d10
FLAT_A: 5
N_ATTACKS: 3


\\\*** General Info ***\\\
CRIT_RANGE: 20
NORMAL_HIT: d20


\\\*** Functions ***\\\
function: attack ROLL:n vs ENEMY_AC:n {
 if ROLL = 1 { result: 0 }
 if ROLL >= CRIT_RANGE { result: CRITICALDAMAGE }
 if ROLL + ATTACK >= ENEMY_AC { result: HITDAMAGE }
 result: 0
}

if BLADE_WARD = 1 {
 if HEAVY_ARMOR_MASTER = 1 {
  HITDAMAGE: DICE_A+FLAT_A-PROF
  CRITICALDAMAGE: DICE_A+FLAT_A
  output N_ATTACKSd[attack NORMAL_HIT vs ENEMY_AC]/2 named "Attack"
 }
 else {
  HITDAMAGE: DICE_A+FLAT_A
  CRITICALDAMAGE: DICE_A+FLAT_A
  output N_ATTACKSd[attack NORMAL_HIT vs ENEMY_AC]/2 named "Attack"
 }
}
else {
 if HEAVY_ARMOR_MASTER = 1 {
  HITDAMAGE: DICE_A+FLAT_A-PROF
  CRITICALDAMAGE: DICE_A+FLAT_A
  output N_ATTACKSd[attack NORMAL_HIT vs ENEMY_AC] named "Attack"
 }
 else {
  HITDAMAGE: DICE_A+FLAT_A

  output N_ATTACKSd[attack NORMAL_HIT vs ENEMY_AC] named "Attack"
 }
}

function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 { result: MISS }
    if ROLL >= CRIT_RANGE { result: CRIT }
    if ROLL + BONUS >= AC { result: HIT }
    result: MISS
}
function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS vs AC hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}

AC: 18
MYCRIT: 20
ADVANTAGE: [highest 1 of 1d20]
ABONUS: 10
DBONUS: 7
DAMAGE: 1d8
HITDMG: DAMAGE + DBONUS
CRITDMG: DAMAGE + DBONUS + DAMAGE + DBONUS
GORE: [attack ADVANTAGE plus ABONUS vs AC for HITDMG crit CRITDMG on MYCRIT]
GOREPOWERATTACK: [attack ADVANTAGE plus ABONUS - 1 vs AC for HITDMG +2 crit CRITDMG +4 on MYCRIT]
SLAM: [attack ADVANTAGE plus ABONUS -1 vs AC for 1d6 + DBONUS - 1 crit CRITDMG -2 on MYCRIT]
SLAMPOWERATTACK: [attack ADVANTAGE plus ABONUS -2 vs AC for 1d6 + DBONUS +1 crit CRITDMG +2 on MYCRIT]
GORENON: [attack ADVANTAGE plus ABONUS -3 vs AC for HITDMG -3 crit CRITDMG -6 on MYCRIT]
GOREPOWERATTACKNON: [attack ADVANTAGE plus ABONUS - 4 vs AC for HITDMG -1 crit CRITDMG -2 on MYCRIT]
SLAMNON: [attack ADVANTAGE plus ABONUS -4 vs AC for 1d6 + DBONUS - 4 crit CRITDMG -8 on MYCRIT]
SLAMPOWERATTACKNON: [attack ADVANTAGE plus ABONUS -5 vs AC for 1d6 + DBONUS -2 crit CRITDMG -4 on MYCRIT]

TOTAL_FULLATTACK: GORE + SLAM
TOTAL_ATTACKPOWERATTACK: GOREPOWERATTACK + SLAMPOWERATTACK
TOTAL_FULLATTACKNON: GORENON + SLAMNON
TOTAL_ATTACKPOWERATTACKNON: GOREPOWERATTACKNON + SLAMPOWERATTACKNON

output TOTAL_FULLATTACK named "Full attack"
output TOTAL_ATTACKPOWERATTACK named "Full power attack"
output TOTAL_FULLATTACKNON named "Full attack NON"
output TOTAL_ATTACKPOWERATTACKNON named "Full power attack NON"

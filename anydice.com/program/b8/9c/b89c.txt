function: roll S:n x X:n over T:n {
 R: 0
 loop N over {1..S} { R: R + [rolltester [rolloverexploder d6 X] T] }
 result: R
}
function: rolloverexploder S:n X:n {
 if S >= X { result: S + d6 }
 result: S
}
function: rolltester S:n T:n {
 if S > T { result: 1 }
 result: 0
}

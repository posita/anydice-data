set "maximum function depth" to 20

function: A mod B {
  if [absolute A] >= B { 
   if A > 0 { result: [ A - B mod B] }
   result: [ A + B mod B]
  }
  result: A
}


function: get modifier for A:n{
 \ turns an ability score into a modifier \
 X: 0
 if A < 10 {X: -1}
 result: [(A - 10) invmod 2 - X]
}

function: A invmod B{
  \ returns the amount of times A fits in B completely \
  result: (A - [A mod B])/B
}

MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
  if ROLL = 1 { result: MISS }
  if ROLL >= CRIT_MINIMUM {
    result: CRITICALHIT
  }
  if ROLL + ATTACK >= DEFENSE { result: HIT }
  result: MISS
}


function: damage for ATTACKRESULT:n {
  if ATTACKRESULT = CRITICALHIT {
    result: CRITICALDAMAGE
  }
  if ATTACKRESULT = HIT {
    result: HITDAMAGE
  }
  result: MISSDAMAGE
}

function: attack {
  result: [damage for [attack d20 vs DEFENSE]]
}

function: attack with advantage {
  result: [damage for [attack [highest 1 of 2d20] vs DEFENSE]]
}

function: attack with disadvantage {
  result: [damage for [attack [lowest 1 of 2d20] vs DEFENSE]]
}

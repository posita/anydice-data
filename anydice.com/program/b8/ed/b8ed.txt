
function: middle of A:n B:n and C:n {
  if A >= B & B >= C { result: B }
  if A >= C & C >= B { result: C }
  if B >= A & A >= C { result: A }
  if B >= C & C >= A { result: C }
  if C >= A & A >= B { result: A }
  if C >= B & B >= A { result: B }
  result: 0
}

loop N over {1..6} {
   output [middle of 1d6 1d6 and N]
}

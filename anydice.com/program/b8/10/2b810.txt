function: bs AR:n {
 if (AR=20) {
  result: 1@2d10 + 10 + 2d6 + DMG
 }
 if (AR=1) { result: 0 }
 if (AR+BNS>=AC) {
  result: 1@2d10 + 1d6 + DMG
 }
 result: 0
}

function: p AR:n {
 if (AR=20) {
  result: 1@3d10 + 10 + 2d6 + DMG
 }
 if (AR=1) { result: 0 }
 if (AR+BNS>=AC) {
  result: 1@3d10 + 1d6 + DMG
 }
 result: 0
}

function: off AR:n {
 if (AR=20) {
  result: 1@2d4 + 4 + 2d6 + DMG
 }
 if (AR=1) { result: 0 }
 if (AR+BNS>=AC) {
  result: 1@2d4 + 1d6 + DMG
 }
 result: 0
}

STR:5
PRF:5
M:2
BNS:STR+PRF+M
DMG:STR+M

loop AC over {18..18} {
output [bs 1@2d20] + [bs 1@2d20] + [p 1@2d20] + [off 1@2d20] + [bs 1@2d20] + [bs 1@2d20] + [bs 1@2d20] named "Surge Lawbringer vs [AC]"
}

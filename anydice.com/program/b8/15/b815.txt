MOD: 5
PROF: 6
MISC: 0
DMG: 1d8+MOD

\ ------------------------------- \
\ DON'T CHANGE ANYTHING PAST HERE \
\ ------------------------------- \

NORM: 1d20+MOD+PROF+MISC
SS: 1d20+MOD+PROF-5+MISC
function: ROLL:s numhits {
 result: (ROLL >= AC)*DMG
}
function: ROLL:s numhitsSS {
 result: (ROLL >= AC)*(DMG+10)
}

loop AC over {10..26}{

output 0 named "--------vs. AC [AC]---------"

output [NORM numhits] named "Normal"
output [1@2d{NORM} numhits] named "Advantage"
output [SS numhitsSS] named "Sharpshooter"
output [1@2d{SS} numhitsSS] named "Sharpshooter w/Ad"
}

set "explode depth" to 1

function: roll N targeting T {
 result: [count {T..12} in Nd[explode d6]]
}
function: limit NUMBER:n{
 if NUMBER < 0 { result: 0 }
 result: NUMBER
}

loop T over {6} {
 loop N over {6..8} {
  output [limit ([roll T targeting 4] - [roll N targeting 4])] named "[T] - [N]"
  output [roll T targeting N+3] named "[T] | [N]"
 }
}

loop STAT over {-50, -25, -10, 0, 10, 25, 50, 100, 200, 400} {
  POCDICE: STAT/100
  POCHALF: STAT - POCDICE*100
  TMPDICE: d100

  if STAT < 0 {
    output [lowest of TMPDICE and d(100+STAT)] named "Stat: [STAT]"
  }else{
    loop N over {1..POCDICE}{
      TMPDICE: [highest of TMPDICE and d100]
    }
    output [highest of TMPDICE and d(POCHALF)] named "Stat: [STAT]"
  }
}

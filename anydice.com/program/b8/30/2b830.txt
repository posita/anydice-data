DMG: STRENGTH + MAGIC + RAGE

function: X:n crit with Y:n strength and Z:n advantage {

if X = 20 | Z=20  {result: 3d12+DMG}
if X = 1 & Z=1 {result: 0}
if X >= AC-MOD | Z >= AC-MOD {result: 1d12+DMG}
result: 0
}

result: 0
}

function: A:n max B pierces{
 if A = B {
  result: A + dB
 }
 else{
  result: A
 }
}

function: A:n max B extra six pierces{
 if A = B {
  result: A + d6
 }
 else{
  result: A
 }
}

loop N over {4,6,8,10,12}{
 output [dN max N pierces] named "d[N] with reroll rules"
 output [dN max N extra six pierces] named "d[N] with d6 rules"
 output dN named "d[N] without rules"
}

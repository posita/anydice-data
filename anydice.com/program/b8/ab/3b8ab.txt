function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 { result: MISS }
    if ROLL >= CRIT_RANGE { result: CRIT }
    if ROLL + BONUS >= AC { result: HIT }
    result: MISS
}
function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    result: [attack ROLL plus BONUS vs AC hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}

AC: 18
MYCRIT: 18
ADVANTAGE: [highest 1 of 1d20]
ABONUS: 10
DBONUS: 5
DAMAGE: 2d4
HITDMG: DAMAGE + DBONUS
CRITDMG: DAMAGE + DBONUS + DAMAGE + DBONUS
ATTACK: [attack ADVANTAGE plus ABONUS vs AC for HITDMG crit CRITDMG on MYCRIT]
FULLATTACKONE: [attack ADVANTAGE plus ABONUS - 2 vs AC for HITDMG crit CRITDMG on MYCRIT]
FULLATTACKTWO: [attack ADVANTAGE plus ABONUS - 2 vs AC for HITDMG - 3 crit CRITDMG -6 on MYCRIT]
ATTACKBUFF: [attack ADVANTAGE plus ABONUS +3 vs AC for HITDMG +3 crit CRITDMG + 6 on MYCRIT]
FULLATTACKONEBUFF: [attack ADVANTAGE plus ABONUS + 1 vs AC for HITDMG + 3 crit CRITDMG + 6 on MYCRIT]
FULLATTACKTWOBUFF: [attack ADVANTAGE plus ABONUS + 1 vs AC for HITDMG - 3 crit CRITDMG on MYCRIT]

TOTAL_FULLATTACK: FULLATTACKONE + FULLATTACKTWO
TOTAL_FULLATTACKBUFF: FULLATTACKONEBUFF + FULLATTACKTWOBUFF

output ATTACK named "Attack"
output ATTACKBUFF named "Attack buffed"
output TOTAL_FULLATTACK named "Full attack"
output TOTAL_FULLATTACKBUFF named "Full attack buffed"

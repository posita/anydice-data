function: roll A:n and B:n {
    if A = 1 | B = 1 { result: A + B }
    else if (A >= 10) & (B >= 10) { result: A + B + 4 }
    else if (A >= 10) | (B >= 10) { result: A + B + 2 }
    else { result: A + B }
}

function: disadvantage A:n and B:n and C:n {
    if (A <= C) & (B <= C)      { result: [roll A and B] }
    else if (A <= B) & (C <= B) { result: [roll A and C] }
    else                        { result: [roll B and C] }
}
function: advantage A:n and B:n and C:n {
    if (A >= C) & (B >= C)      { result: [roll A and B] }
    else if (A >= B) & (C >= B) { result: [roll A and C] }
    else                        { result: [roll B and C] }
}

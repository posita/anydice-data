function: roll damage {
  result: d6+5
}

function: deal SNEAK_DAMAGE:n if ATTACK:n hits AC:n with ATTNUM:n attacks {
  TOTAL_DAMAGE: SNEAK_DAMAGE
  loop N over {1..ATTNUM}{
    if ATTACK >= AC { TOTAL_DAMAGE: TOTAL_DAMAGE + 8}
  }
  if TOTAL_DAMAGE > SNEAK_DAMAGE { result: TOTAL_DAMAGE }
  result: 0
}

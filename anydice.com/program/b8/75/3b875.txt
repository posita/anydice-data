B: 1 \ # of Boost Dice. ALWAYS 1 \
S: 0 \ # of Setback Dice \

BS: Bd{0,0,0,0,0,0,0,1,1,3} \ d10 full or half moons \

SF: Sd{0, 0, 0, 0, 0, 0}      \ Failure table for setback die \

function: explode N:n {
 if N = 3 { result: N + [explode d(BS)] - 1 }
 result: N
}

set "maximum function depth" to 2

output 3d[explode d(BS)] - SF named "3d"
output 4d[explode d(BS)] - SF named "4d"
output 5d[explode d(BS)] - SF named "5d"
output 6d[explode d(BS)] - SF named "6d"

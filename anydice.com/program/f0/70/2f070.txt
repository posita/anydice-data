function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}
MISS: 0
HIT: 1
CRITICALHIT: 2

AC: 15
ATTACK: 7

SUM: 0

loop N over {1,1,1,1,1}{
SUM : SUM + [attack d20 vs AC]*2d6 + 2
}

output SUM
output [attack d20 vs AC]*10d6 + 10

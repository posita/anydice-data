function: outcome ROLL:s {
  SHIP: 0
  CAPTAIN: 0
  CREW: 0

  loop D over ROLL {
    if D = 6 { SHIP: 1 }
    if D = 5 { CAPTAIN: 1 }
    if D = 4 { CREW: 1 }
  }

  if SHIP = 1 & CAPTAIN = 1 & CREW = 1 { result: 4 }
  else if SHIP = 1 & CAPTAIN = 1 { result: 3 }
  else if SHIP = 1 { result: 2 }
  else { result: 1 }
}

loop N over {1..10} {
  output [outcome Nd6] named "[N]d6 Outcome Distribution"
}

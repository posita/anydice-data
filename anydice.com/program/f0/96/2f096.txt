DISASTER: 0 
DISADVANTAGE: 1 
MAINTAIN: 2 
LOW: 6
HIGH: 16 
DICENUMBER: 1
DICESIZES: 20
LOWERBONUS: -4
UPPERBONUS: 0


function: save ROLL:n {
 if ROLL <= LOW { result: DISASTER }
 if ROLL >= HIGH { result: MAINTAIN  }
 result: DISADVANTAGE
}

loop ADJ over {LOWERBONUS..UPPERBONUS}{ 
output [save DICENUMBERdDICESIZES + ADJ] named"[DICENUMBER]d[DICESIZES] + [ADJ]" }

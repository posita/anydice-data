\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: thirdhit FROLL:n SROLL:n {
 if FROLL >= CRITROLL | SROLL >= CRITROLL { result: 1}
 result: 0
}

function: dpr FROLL:n SROLL:n TROLL:n vs AC:n {
 result: [attackroll FROLL vs AC] + [attackroll SROLL vs AC] + [attackroll TROLL vs AC]*[thirdhit FROLL SROLL]
}

function: totaldpr DMG:n {
 if DMG = 0 { result: 0 }
 result: DMG + 6d8
}



ATTACKMOD: 4
DMGBONUS: 7
PROFICIENCY: 4
ACCBONUS: 0
CRITROLL: 19
DAMAGE: 2d6 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 4d6 + ATTACKMOD + DMGBONUS
ENEMYAC: 15
DICE: 1@3d20

AVERAGE_DPR: [totaldpr [dpr DICE DICE DICE vs ENEMYAC]]

output AVERAGE_DPR named "Hexbld Greatsword vs [ENEMYAC] AC"

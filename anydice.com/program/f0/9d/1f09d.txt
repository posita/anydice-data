\ poisson dice for various sided dice \

PFOUR: {0,0,1,2}

PSIX: {0,0,1,1,2,3}

PEIGHT: {0,0,0,1,1,1,2,3}

PTEN: {0,0,0,0,1,1,1,2,2,3}

function: tail N:n D M T {
  if N < M {result: N}
  if (T = 1) {result: 1 + [tail dD D M T]}
  result: N + [tail dD D M T]
}

\ 
  Poisson function
   N = a single roll of a poisson die
   D = the poisson die e.g. PEIGHT
   M = the value at which to chain; normally the maximum value of the poisson die e.g. 3
   T = 0: no exploding dice
   T = 1: Nerfed Value chained dice
   T = 2: Face Value chained dice
\
function: poissonimpl N:n D M T {
  if (N < M) {result: N}
  if (T != 0) {result: N + [tail dD D M T]}
  result: N
}

\ Generally, call this function not the poissonimpl function \
function: poisson D T {
  M: [maximum of dD]
  result: [poissonimpl dD D M T]
}

function: success D:n on D at least N explode M {
 if (D >= M) {result: 1 + [success dD on D at least N explode M]}
 if (D >= N) {
  result: 1
 }
 result: 0
}

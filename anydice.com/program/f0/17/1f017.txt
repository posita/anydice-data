\functions\

function: S:s cull X:s {
ES:{}
  loop Y over S {
    if (Y = X)=0{
      ES:{ES,Y}
    }
  }
result:{ES}
}

function: singledie D:d {
result: d([maximum of D]/#D)
}

function: D:d reroll S:s {
SINGLEDIE:[singledie D]
REROLLD:{{SINGLEDIE}:(#S),[{SINGLEDIE} cull S]:[maximum of SINGLEDIE]}
result:(#D)dREROLLD
}



TDICE:10d6
loop C over {6,5,4,3,2}{
  output [count {C..6} in TDICE] named "BS [C] hits"
  output [count {C..6} in [TDICE reroll 1]] named "BS [C] hits reroll 1"
  output [count {C..6} in [TDICE reroll {1..(C-1)}]] named "BS [C] hits reroll any"
}

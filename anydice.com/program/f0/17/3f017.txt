DIE: d{0:5,1:1}

function: push REROLL:s SIXES:n {
 NEW_SIXES: [count 1 in REROLL]
 if SIXES + NEW_SIXES >= 3 {
  result: 3
 }
 if SIXES + NEW_SIXES >= 1 {
  result: 1
 }
 result: 0
}

function: roll INITIAL:s {
 SIXES: [count 1 in INITIAL]
 BLANKS: [count 0 in INITIAL]
 if SIXES >= 3 {
  result: 3
 }
 if BLANKS >= 1 {
  result: [push BLANKSdDIE SIXES]
 }
 result: 1
}

loop C over {1..10} {
 output [roll CdDIE] named "[C]d"
}

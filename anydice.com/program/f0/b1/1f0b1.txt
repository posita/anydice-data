function: attack ATTACK:d {
  result: ATTACK
}

function: multiattack X:n Y:d {
  TOTAL: 0
  loop N over {1..X} {
    DAMAGE: [attack Y]
    TOTAL: TOTAL + DAMAGE
  }
  result: TOTAL
}

function: boomingbladeplusregular ATTACK:d {
  result: 2d8 + ATTACK
}

function: boomingbladeplusregulariftargetmoves ATTACK:d {
  result: 2d8 + 3d8 + ATTACK
}



output [multiattack 3 1d8+4] named "3 attacks 1d8+4"
output [boomingbladeplusregular 1d8+4] named "1 booming blade attack + 1 bonus attack"
output [boomingbladeplusregulariftargetmoves 1d8+4] named "1 booming blade attack + 1 regular attack + target moves"

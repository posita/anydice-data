DIE: 1d20
DEV: 4
ARMOR: 0
BLUR: 0
CR: 0
BONUS: 0
LO: 10 - ARMOR + BLUR
HI: 10 + DEV + (2 * ARMOR)

function: result D:n bonus B:n crit C:n {
 DB: D + B
 DR: (DB >= LO) + (DB >= HI)
 if (DB < LO - 10) | D <= 1 {
  DR: DR - 1
 }
 if (DB >= HI + 10) | D >= 20 {
  DR: DR + 1
 }
 if DR < -1 {DR: -1}
 if DR > 3 {DR: 3}
 result: DR
}

loop N over {-4, -3, -2, 0, 1, 2, 4, 5, 6}{
 output [result DIE bonus N crit 3] named "Bonus [N]"
}


\
loop N over {0..3}{
 LO: 10 - N
 HI: 10 + DEV + (2 * N)
 output [result DIE bonus 0 crit 3] named "Armor [N]"
}
\

\
loop N over {0, 4, 10}{
 LO: 10 - 3 + N
 HI: 10 + DEV + 6
 output [result DIE bonus 0 crit 3] named "Blur [N]"
}
\

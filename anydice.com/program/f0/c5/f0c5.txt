function: black { R: d{0,0,0,1,1,2} 
result: (R > 0) + (R = 2) 
}
function: gray  { R: d{0,0,1,1,1,2} 
result: (R > 0) + (R = 2) 
}
function: white { R: d{0,1,1,1,1,2} 
result: (R > 0) + (R = 2) 
}

function: roll R black { 
if R > 1 { result: [black] + [roll (R-1) black]  }
result: [black]
}
function: roll R gray { 
if R > 1 { result: [gray] + [roll (R-1) gray]  }
result: [gray]
}
function: roll R white { 
if R > 1 { result: [white ] + [roll (R-1) white ]  }
result: [white ]
}

loop N over {1..3} {
  output [roll N black] named "roll [N] black"
}
loop N over {1..3} {
  output [roll N gray] named "roll [N] gray"
}
loop N over {1..3} {
  output [roll N white] named "roll [N] white"
}

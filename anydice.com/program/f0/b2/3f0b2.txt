function: success A:n {
 if A <= 6 { result: 0 }
 if A >= 7 & A <= 13 { result: 2d6 }
 if A >= 14 & A <= 18 { result: 4d6 }
 if A >= 19  { result: 6d6 }
 
}

output [success 2d12] named "straight"
output [success [highest 2 of 3d12]] named "advantage"
output [success [highest 2 of 4d12]] named "double advantage"

function: roll ROLL:n threshold T {
  if ROLL >=T {result: 1}
  result: 0
}

function: rollexploding ROLL:n threshold T{
  if ROLL =10 {result: 1+[rollexploding 1d10 threshold T]}
  result: [roll ROLL threshold T]
}

function: rerollfailures ROLL:n threshold T{
  if ROLL < T {result: [roll 1d10 threshold T]}
  result: [roll ROLL threshold T]
}

function: rerollthenexplode ROLL: n threshold T {
  if ROLL < T {result: [rollexploding 1d10 threshold T]}
  result: [rollexploding ROLL threshold T]
}


function: wrapper DICE:n threshold T explode E reroll R{
  RES:0
  loop N over {1..DICE} {
    if E & R {RES: RES+[rerollthenexplode 1d10 threshold T]}
    else {if E {RES:RES+[rollexploding 1d10 threshold T]}
      else {if R {RES:RES+[rerollfailures 1d10 threshold T]}
        else {RES:RES+[roll 1d10 threshold T]}
      }
    }
  }
  result:RES
}


function: b DICE:n{
  result:[wrapper DICE threshold 4 explode 0 reroll 0]
}

function: be DICE:n{
  result:[wrapper DICE threshold 8 explode 1 reroll 0]
}

function: gr DICE:n{
  result:[wrapper DICE threshold 3 explode 0 reroll 1]
}

function: dxpetviii DICE:n{
  result:[wrapper DICE threshold 8 explode 1 reroll 0]
}

function: dxpetvii DICE:n{
  result:[wrapper DICE threshold 7 explode 1 reroll 0]
}

function: dxpetvi DICE:n{
  result:[wrapper DICE threshold 6 explode 1 reroll 0]
}

\number of dice\ N: 6
\size of dice\   D: 12

function: allsuccess A:s { \function for any success\
HITS: [count {4..20} in A] \counts dice showing 4 or more\
ONES: [count 1 in A]       \counts 1s\
TEST: ONES < 2             \boolean check that there are less than two 1s\
result: HITS*TEST          \multiply the hits by the test\ 
}

function: fullsuccess B:s { \function for success without complications\
HITS: [count {4..20} in B]  \identical, but checking for a single 1 instead\
ONES: [count 1 in B]
TEST: ONES < 1
result: HITS*TEST
}

\loop A over {4..12} {
output [allsuccess NdA] named "[A]s"
output [fullsuccess Nd4] named "[A] mixed"
}\

loop B over {1..6} {
output [allsuccess BdD] named "[B]d[D]"
output [fullsuccess BdD] named "[B]d[D] mixed"
}

function: attack ROLL:n {
 WEAPON: 1d8
 SNEAK: 4d6
 if ROLL=20 {
  result: WEAPON+WEAPON+SNEAK+SNEAK+5
 }
 if ROLL>=10 {
  result: WEAPON+SNEAK+5
 }
 result: 0
}
output [attack 1d20] named "Normal"
output [attack [highest 1 of 2d20]] named "Advantage"
output [attack [highest 1 of 3d20]] named "Elven Accuracy"

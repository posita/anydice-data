loop A over {3..6} {
  loop S over {2..6} {
    ATT: 1d(A*2)
    if S = 2 {
      output ATT > 5 named "[ATT] Attribute, 0 Skill"
      output 1@2dATT > 5 named "[ATT] Attribute, 0 Skill. Pushed"
    } else {
      SKILL: 1d(S*2)
      output [highest of ATT and SKILL] > 5 named "[ATT] Attribute, [SKILL] Skill"
      output [highest of 1@2dATT and 1@2dSKILL] > 5 named "[ATT] Attribute, [SKILL] Skill. Pushed"
    }
  }
}

function: roll COMP:s under SKILL:n {
  SUCCESS: 0
  loop FACE over COMP {
    if FACE = 1 { SUCCESS: 1 }
    if FACE < SKILL { SUCCESS: SUCCESS + 1 }
  }
  result: SUCCESS
}

loop COMP over {1..5} {
  loop SKILL over {1..5} {
    SUCCESS : [roll COMP d6 under SKILL]
    output SUCCESS named "SUCCESS"
  }
}

function: classify R:n plus M:n {
  if R = 2  { result: -1 }
  if R = 20 { result: 2 }

  if R+M < 11 { result: -1 }
  if R+M < 17 { result: 0 }
  if R+M < 23 { result: 1 }
  result: 2
}

loop MOD over {-5..-1} {
  output [classify 2d10 plus MOD] named "2d10[MOD]"
}

loop MOD over {0..11} {
  output [classify 2d10 plus MOD] named "2d10+[MOD]"
}

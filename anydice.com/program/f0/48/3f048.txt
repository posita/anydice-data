
\-- Deflect Attacks: Reflect Attack --\

LEVEL: 9
GRAPPLE_SCORE: 20
GRAPPLE_BONUS: 1


GRAPPLE_MOD: GRAPPLE_SCORE/2 - 5
PROF_BONUS: (LEVEL - 1)/4  + 2


SAVE_DC: 8 + GRAPPLE_MOD + PROF_BONUS


function: saving throw ROLL:n{
  if ROLL < SAVE_DC {result: 1}
  result: 0
}

output SAVE_DC named "Save DC"

loop MOD over {SAVE_DC-21..SAVE_DC-1}{
  ROLL: d20 + MOD
  if MOD < 0 {output [saving throw ROLL] named  "[MOD] DEX Save"}
  else       {output [saving throw ROLL] named "+[MOD] DEX Save"}
}

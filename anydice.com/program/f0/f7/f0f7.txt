function: pick D:s {
 A: 1@D
 B: (#D/2+1)@D
 C: #D@D
 AB: B - A
 BC: C - B
 \if (BC = AB) {
  result: B
 }\
 if (BC > AB) {
  result: A
 }
 else {
  result: C
 }
}

output [pick 3d20] named "Middle one decides"
output [lowest 1 of 3d20] named "Lowest of 3d20"
output [highest 1 of 3d20] named "Highest of 3d20"
output 1d20 named "1d20"

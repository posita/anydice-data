function: attack ROLL:n USEPA:n {
 WEAPON: 1d8
 SNEAK: 4d6
 CRIT: 1d8+4d6
 DMGMOD: 5
 ATKMOD: 5+4+2
 if USEPA {
  ATKMOD: ATKMOD-5
  DMGMOD: DMGMOD+10
 }
 if ROLL = 20 {
  result: WEAPON+SNEAK+DMGMOD+CRIT
 }
 if ROLL = 1 {
  result: 0
 }
 if ROLL+ATKMOD >= 15 {
  result: WEAPON+SNEAK+DMGMOD
 }
 result: 0
}
output [attack 1d20 0] named "No Adv, No PA"
output [attack [highest 1 of 3d20] 0] named "Adv, No PA"
\output [attack 1d20 1] named "No Adv, PA"\
\output [attack [highest 1 of 3d20] 1] named "Adv, PA"\

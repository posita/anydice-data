\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}



ATTACKMOD: 4
DMGBONUS: 4
PROFICIENCY: 4
ACCBONUS: 0
CRITROLL: 19
DAMAGE: 2d6 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 4d6 + ATTACKMOD + DMGBONUS

loop ENEMYAC over {15..25}{
output [attackroll 1@3d20 vs ENEMYAC]+[attackroll d20 vs ENEMYAC] named "Fgtr1 Greatsword vs [ENEMYAC] AC"
}

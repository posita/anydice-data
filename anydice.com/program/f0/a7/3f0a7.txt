  result: [A B C helper]
 }else{
  if A>=B & C>=B{
   result: [A C B helper]
  }else{
   if B>=A & A>=C{
    result: [B A C helper]
   }else{
    if B>=A & C>=A{
     result: [B C A helper]
    }else{
     if C>=A & A>=B{
      result: [C A B helper]
     }else{
      result: [C B A helper]
     }
    }
   }    
  }
 }
}

 }else{

  \if B=C {result: 1000*C +A}\
  result: A
 }
}

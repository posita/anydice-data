function: roll NUMBER:n drop DROP:n mod MOD:n {
 ROLL: {NUMBER + [absolute MOD]}d6

 if MOD > 0 {
   MOD: 0
 }
 
 FROM: 1 + [absolute MOD]
 TO: FROM + NUMBER - DROP - 1
 result: {FROM..TO}@ROLL
}

output [roll 2 drop 0 mod 0] named "2d6"
output [roll 2 drop 0 mod 1] named "2d6 +1"
output [roll 4 drop 2 mod 0] named "4d6 drop 2 aka 2d6 +2"
output [roll 4 drop 2 mod -1] named "4d6 drop 2 -1"
output [roll 4 drop 2 mod -2] named "4d6 drop 2 -2"

function: ROLL:n vs AC:n 
   {
    if (ROLL + ATTACK >= AC) | (ROLL = 20) { result: DAMAGE }
    result: 0
   }

DR: d{1..8}

STRENGTH: 4

ATTACK: BAB + STRENGTH + 1 - 2
DAMAGE:  [highest of 1 and (1d8 + STRENGTH - DR)]

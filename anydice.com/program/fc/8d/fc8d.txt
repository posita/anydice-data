function: attack ROLL:n bonus ATTACK:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: hamdamage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: HAMCRITDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HAMDAMAGE
 }
 result: MISSDAMAGE
}

function: ebdamage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: EBCRITDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: EBDAMAGE
 }
 result: MISSDAMAGE
}

function: reroll D:d under N:n{
   result: [rerollcore D die D under N]
}

function: rerollcore R:n die D:d under N:n{
   if R < N { result: D } else {result: R}
}

function: burning hands diff D:n against W:n{
   SAVE: d20 + W
   if SAVE >= D{
      result: BHSAVEDAMAGE
   }
   result: BHDAMAGE
}

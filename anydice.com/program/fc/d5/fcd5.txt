\**** Damage calculator ****\

function: multiattack ATTACKS:n DICE:n {
 TOTAL: 0
 loop N over {1..ATTACKS}{
  ADVANTAGE: [highest 1 of DICEd20]
  TOTAL: TOTAL + [attackroll ADVANTAGE vs ENEMYAC]
 }
 result: TOTAL
}

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 12-20\
ENEMYAC: d{12..20}

ATTACKMOD: 7
DMGBONUS: 10
PROFICIENCY: 3
ACCBONUS: -3
CRITROLL: 20
DAMAGE: 1d6 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d6 + ATTACKMOD + DMGBONUS
output [multiattack 3 1] named "Sharpshooter Hand Crossbow +2 Fighter 6"

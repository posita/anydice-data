
function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: MISS
    }
    if ROLL >= CRIT_RANGE {
        result: CRIT
    }
    if ROLL + BONUS >= AC {
        result: HIT
    }
    result: MISS
}

output
2d[attack d20 plus 3 + 4 + 2 - 5 vs d{14..15}
    hit  1d6 + 4 + 10
    miss 0
    crit 2d6 + 4 + 10 on 20
]

AC: 15
ATTBONUS: 8
DAMAGE: 1d8+1d6+6
CRIT: 3d8+2d6+6
NUMATT: 2

function: resolve ATT:n {
  if ATT = 20 {
    result: CRIT
  }
  else if ATT = 1 {
    result: 0
  }
  else if (ATT + ATTBONUS) >= AC {
    result: DAMAGE
  }
  else {
    result: 0
  }
}

function: roll DIE:d {
  DPT: 0
  loop I over {1 .. NUMATT}{
     DPT: DPT + [resolve DIE]
  }
  result: DPT
}

output [roll 1@1d20] named "Straight"
output [roll 1@2d20] named "Advantage"
output [roll 2@2d20] named "Disadvantage"

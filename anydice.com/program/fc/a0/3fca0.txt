function: check POOL diff DIFF {
 SUM: 0
 loop N over {1..POOL}{
  SUM: SUM + [implode 1d12 diff DIFF]
 }
 result: SUM
}

function: implode ROLL:s diff DIFF{
 HITS: 0
 if ROLL = 1{
  HITS: HITS + [implode 1d12 diff DIFF]
 }
 if ROLL <= DIFF{
  HITS: HITS + 1
 }
 result: HITS
}

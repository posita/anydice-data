function: pool DICE:s {
  D: 1@DICE
  loop X over {2..#DICE} {
    if ((X-1)@DICE = X@DICE) { D: D + 1 }
  }
  result: D
}


function: compare PLAYER:s vs CHALLENGE:s {
   result: [pool PLAYER] - [pool CHALLENGE] 
}

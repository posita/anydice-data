function: astens A:n {
if A = 10 {
result: 0
}
result: A * 10
}

function: asones A:n {
if A = 10 {
result: 0
}
result: A
}

function: percentile X:n Y:n {
if [astens X] = 0 & [asones Y] = 0 {
result: 100
}
result: [astens X] + [asones Y]
}


function: damage A:n B:n SKILL {
if [percentile A B] <= SKILL {
result: A + B
\ "0" counts as 10 \
}
}

function: flipflop A:n B:n SKILL {
if [percentile A B] > SKILL {
result: [percentile B A]
}
result: [percentile A B]
}

RED: d10
BLUE: d10

SKILL: 40

loop S over {1..19} {
output [flipflop RED BLUE S*5] <= S*5 named "[S]"
}

loop S over {1..19} {
output [lowest 1 of 2d20] <= S named "x[S]"
}

\
 0: No, and... (fail, and trouble)
 1: No, but... (fail, but something good)
 2: Yes, but... (success, but some problem)
 3: Yes, and... (success, and something extra)
\

function: blades ROLL:s {
 if ROLL >= 21 {
  result: 3
 }
 else if ROLL >= 12 {
  result: 2
 }
 else if ROLL >= 7 {
  result: 1
 }
 else { 
  result: 0
 }
}

loop P over {-5..5} {
 output [blades 2d10+P] named "2d10+[P]"
}

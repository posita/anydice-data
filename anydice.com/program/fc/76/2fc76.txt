AC: 25
RESISTANCE: 1
CRIT_DOES_DAMAGE: 0
ADVANTAGE: 1
ATTACKS: 6
DAMAGE_DIE: 3d6
DAMAGE_MOD: 6
CRIT: 20
TO_HIT: 10

TOTAL: 0
loop N over {1..ATTACKS}{
  R: 0
  if (ADVANTAGE) {
    R: [highest 1 of 2d20]
  }
  else {
    R: 1d20
  }
  TOTAL: TOTAL + ((DAMAGE_DIE + DAMAGE_MOD) * ((R + TO_HIT >= AC) + (R >= CRIT) >= 1) + DAMAGE_DIE * (CRIT_DOES_DAMAGE + (R >= CRIT) >= 2)) / (2 ^ (RESISTANCE != 0))
}
output TOTAL

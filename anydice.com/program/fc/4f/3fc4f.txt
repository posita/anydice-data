function: rollcount MAX:n DIE:d {
  PREV: 0
  STOP: 0
  COUNT: 0
  loop I over {1..MAX} {
    if STOP = 0 {
      R: DIE
      COUNT: I
      if PREV = 0 {
        PREV: R
      } else {
        if R >= PREV { STOP: 1 } else { PREV: R }
      }
    }
  }
  result: COUNT
}

output [rollcount 20 d6] named "Rolls until stop (d6, max 20)"

function: get range from A:n and B:n with low weight L:n and high weight H:n
{
 if A > B 
 { 
  result:{B+L, A+H}
 }
 else 
 {
  result:{A+L, B+H}
 }
}

function: is Z:n in range S:s
{
 if Z >= 1@S & Z <= 2@S
 {
  result: 1
 }
 result: 0
}

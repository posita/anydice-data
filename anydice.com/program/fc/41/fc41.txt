AC: 18
DEX: 5
CHA: 5
PROF: 4
ADVANTAGE: 1
CRIT_RANGE: 20

function: A:n and B:n hit D:d crit E:d against C:n{
if (A >= CRIT_RANGE | (B >= CRIT_RANGE & ADVANTAGE)) {result: E}
if (A = 0 | (A = 0 & B = 0 & ADVANTAGE)) {result: 0}
if (A >= C | (B >= C & ADVANTAGE)) {result: D}
result: 0
}

D_GFB_ONE:[d20 and d20 hit (1d8+DEX+2d8+1) crit (2d4+DEX+4d8+1) against (AC-DEX-PROF-1)]
D_GFB_TWO:[d20 and d20 hit (1d8+DEX+2d8+CHA+2d8+1) crit (2d4+DEX+4d8+CHA+2d8+1) against (AC-DEX-PROF-1)]
D_BOOM_NO:[d20 and d20 hit (1d8+DEX+2d8+1) crit (2d4+DEX+4d8+1) against (AC-DEX-PROF-1)]
D_BOOM_MOVE:[d20 and d20 hit (1d8+DEX+5d8+1) crit (2d4+DEX+7d8+1) against (AC-DEX-PROF-1)]
FIREBOLT:[d20 and d20 hit (3d10) crit (6d10) against (AC-DEX-PROF)]


output D_GFB_ONE named "Dagger GFB 1"

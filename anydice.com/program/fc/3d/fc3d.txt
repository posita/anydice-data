function: attack ATTACK:d {
  result: ATTACK
}
function: multiattack X:n Y:d {
  TOTAL: 0
  loop N over {1..X} {
    DAMAGE: [attack Y]
    TOTAL: TOTAL + DAMAGE
  }
  result: TOTAL
}
MULTIATTACK: 4
ATTACK: [highest 3 of 3d8]

output [multiattack MULTIATTACK ATTACK] named "[MULTIATTACK] attacks"
output 3d8 + 3d8 + 3d8

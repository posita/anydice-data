\cWoD Roller\

\ POOL = dice pool, Difficulty = target number \

POOL: 3

DIFFICULTY: 7

\ ---- \


function: roll ROLL:s {
 if 1@ROLL < 1 & (#ROLL)@ROLL = -1 {
  result: -1
 }
 result: [highest of 0 and ROLL]
}

function: roll ROLL:n with willpower {
 result: [highest of 0 and ROLL + 1]
}

function: specialty N:n {
 if N = 10 { result: 2 }
 if N = 1 { result: -1 }
 result: N >= DIFFICULTY
}



NORMAL: d{-1:1, 0:(DIFFICULTY - 2), 1:(11 - DIFFICULTY)}

SPECIALTY: [specialty d10]



output [roll POOL d NORMAL] named "normal"
output [roll POOL d SPECIALTY] named "specialty"
output [roll POOL d NORMAL with willpower] named "normal willpower"
output [roll POOL d SPECIALTY with willpower] named "specialty willpower"
\ automatic success is automatic, nothing to show \

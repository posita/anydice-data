function: attackcomp  ATTACKDIE:n ATTACKMOD:n DAMAGE:n DEADLYDAMAGE:n FATALDAMAGE:n AC:n {
	ATKROLL: ATTACKDIE + ATTACKMOD
	CRITDAMAGE: DEADLYDAMAGE + DEADLYDAMAGE + FATALDAMAGE
	if ATKROLL < AC {
		result: 0
	}
	else {
		if ATTACKDIE = 20 {
			result: CRITDAMAGE
		}
		
		if ATKROLL >= AC + 10 {
			result: CRITDAMAGE
		}
	}
		
	result: DAMAGE
}

function: atk ATTACKMOD:n DAMAGE:n DEADLYDAMAGE:n FATALDAMAGE:n AC:n {
	result: [attackcomp 1d20 ATTACKMOD DAMAGE DEADLYDAMAGE FATALDAMAGE AC]
}

function: atks COUNT:n ATTACKMOD:n MAP:n DAMAGE:n DEADLYDAMAGE:n FATALDAMAGE:n AC:n FLURRY:n {
  TOTAL: [atk ATTACKMOD DAMAGE DEADLYDAMAGE FATALDAMAGE AC]

  loop N over {1..COUNT-1} {
    if FLURRY = 1 {
      TEMPATK: ATTACKMOD - (MAP*(N-1))
    }
    else {
      TEMPATK: ATTACKMOD - MAP*N
    }

    TOTAL: TOTAL + [atk TEMPATK DAMAGE DEADLYDAMAGE FATALDAMAGE AC]
  }

  result: TOTAL
}

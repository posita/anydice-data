function: dsix one PROLL:n and NROLL:n{
  if PROLL = 6 {PROLL: 1}
  if NROLL = 6 {NROLL: 1}

  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}

function: dsix two PROLL:n and NROLL:n{
  \if PROLL = 6 | NROLL = 6 {result: 0}\
  if PROLL = 6 | NROLL = 6 {
    if PROLL = 5 | NROLL = 5 {result: 0 }
  }
  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}

output 4d{-1..1} named "4dF roll"
output [dsix one d6 and d6] named "6s to 1s 2d6 fudge roll"
output [dsix two d6 and d6] named "+/- 5 = 0 2d6 fudge roll"

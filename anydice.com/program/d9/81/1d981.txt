function: A:s B:s C:n D:n crit{
  if (A = C & B = D)|(A = 6 & B = 6){
    result: 1
  }else {
    result: 0
  }
}
function: E:d F:d crittest{
result: [E F [maximum of E] [maximum of F] crit]
}

loop G over {4, 6, 8, 10, 12}{
loop H over {4, 6, 8, 10, 12}{
if G <= H {
output [1dG 1dH crittest] named "1d[G] 1d[H]"
}
}
}

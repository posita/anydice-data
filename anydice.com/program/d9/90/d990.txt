MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULTS:s {
 D: 0
 loop A over ATTACKRESULTS {
  if A = CRITICALHIT {
   D: D + CRITICALDAMAGE + [maximum of MARKDMG]
  }
  else if A = HIT {
   D: D + HITDAMAGE + MARKDMG
  }
  else {
   D: D + MISSDAMAGE
  }
 }
 result: D + [bonus damage for ATTACKRESULTS]
}

function: bonus damage for ATTACKRESULTS:s {
 if [ATTACKRESULTS contains CRITICALHIT] {
  result: [maximum of SNEAKDMG] + [maximum of COLOSSUS]
 }
 if [ATTACKRESULTS contains HIT] {
  result: SNEAKDMG + COLOSSUS
 }
 result: 0
}


function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

function: N:n attacks {
 result: [damage for Nd[attack d20 vs DEFENSE]] + HORDE*[damage for [attack d20 vs DEFENSE]]*(1d100 <= HORDECHANCE)
}

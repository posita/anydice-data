set "maximum function depth" to 3

function: attack ROLL:d plus ATTACK:n vs DEFENSE:n {
 MIN: 2
 if DEFENSE-ATTACK > 2 {
  MIN: DEFENSE-ATTACK
 }
 result: [count {MIN..6} in ROLL]
}

function: attackstarlord ROLL:d plus ATTACK:n vs DEFENSE:n extra EXTRA:n {
 COUNT: [attack ROLL plus ATTACK vs DEFENSE]
 if EXTRA = 6 {
  COUNT: COUNT + [attack ROLL plus ATTACK vs DEFENSE] 
 }

 result: COUNT
}

output [attack 6d6 plus 5 vs 8] named "AttackNormal"
output [attackstarlord 6d6 plus 5 vs 8 extra 1d6] named "AttackStarlord"

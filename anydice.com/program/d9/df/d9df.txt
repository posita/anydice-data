ATTACK: 15
PARRY: 10
CRITICALHIT: 2d6+10
HIT: 1d6+5

function: attack ROLL:n n REROLL:n{
 if ROLL = 20 { 
  if REROLL <= ATTACK { result: MISS }
  result: FUMBLE }
 if ROLL = 1 {
  if REROLL <= ATTACK { result: CRITICALHIT }
  result: [defense d20 n d20 PARRY/2] }
 if ROLL <= ATTACK { result: [defense d20 n d20 PARRY] }
 result: MISS
}

function: defense ROLL:n n REROLL:n PARR:n{
 if ROLL =20 { if REROLL > PARR { result: FUMBLE*2}}
 if ROLL <= PARR { result: MISS }
 result: HIT
 }
FUMBLE: -1
MISS: 0

output [attack d20 n d20]

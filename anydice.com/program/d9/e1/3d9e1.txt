\ Reaper's Touch Balance Calculator \

ND: 2d8                             \Necrotic Damage per turn on Hit\
T: 4                                \Targets attempted to hit\
D: 2                                \Estimated average number of turns per affected target\

SA: (ND*62/100)                     \Single target damage, accounting for accuracy\
TD: SA*T*D                          \Total damage, accounting for accuracy\
HR: TD/2
\HR: ((2*T)*62/100)*D\              \Total Healing, accounting for accuracy\

DMP: TD*4/5                         \MP cost for damage\
HMP: HR*5/4                         \MP cost for healing\
SMP: 5*T                            \MP cost for slow\
MPC: DMP+HMP+SMP                    \Balanced MP cost\

output SA named "Single Target damage, accounting for accuracy"
output TD named "Total damage, accounting for accuracy"
output HR named "Health restored"
output MPC named "Balanced Magic Point Cost"
output MPC*8/10 named "MP cost, Focus"
output MPC*6/10 named "MP cost, Concentration"

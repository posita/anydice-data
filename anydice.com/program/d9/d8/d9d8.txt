ATTACK: 10
CRITICALHIT: 2d6+10
HIT: 1d6+5

function: attack ROLL:n n REROLL:n{
 if ROLL = 20 { 
  if REROLL <= ATTACK { result: MISS }
  result: FUMBLE }
 if ROLL = 1 {
  if REROLL <= ATTACK { result: CRITICALHIT }
  result: HIT }
 if ROLL <= ATTACK { result: HIT }
 result: MISS
}
FUMBLE: -1
MISS: 0

output [attack d20 n d20]

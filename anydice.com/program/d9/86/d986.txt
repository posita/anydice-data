\ START EDIT THESE \

\ NUMBER OF ATTACKS \
ATK: 32
\ TO HIT \
H:{3}
\ HIT REROLLS \
HR:{1}
W:{4,5,6}
\ WOUND REROLLS \
WR:{0}
\ TO SAVE \
S:{5,6}
\ WOUND REROLLS \
SR:{0}
\ END EDIT THESE \


function: rollhit R:s {
DICE: R
if R = HR { DICE: 1d6 }
result: DICE
}

function: rollwound R:s {
DICE: R
if R = WR { DICE: 1d6 }
result: DICE
}

function: rollsave R:s {
DICE: R
if R = SR { DICE: 1d6 }
result: DICE
}

function: rollattack HIT:s WOUND:s SAVE:s {
if [count H in HIT] & [count W in WOUND] & [count S in SAVE]=0 { result: 1 }
result: 0
}


WOUNDS: 0

loop N over {1..ATK} {
WOUNDS: WOUNDS + [rollattack [rollhit 1d6] [rollwound 1d6] [rollsave 1d6]]}

output WOUNDS named "unsaved wounds"

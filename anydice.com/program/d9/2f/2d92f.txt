function: pbval of N:n 
{
   if N <= 8 { result: 0}
   if N <= 13 { result: N-8}
   if N = 14 {result: 7}
   if N = 15 {result: 9}
   if N = 16 {result: 12}
   if N = 17 {result: 16}
   if N = 18 {result: 20}
}

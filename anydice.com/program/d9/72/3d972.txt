function: fudge PRI:n SEC:n {
  PCANC: 1 - (PRI = 6)
  PRI: PRI * PCANC

  SCANC: 1 - (SEC = 6)
  SEC: SEC * SCANC

  MCANC: 1 - (PRI = SEC)

  PHIGH: PRI < SEC
  SHIGH: SEC < PRI

  result: ((PRI * PHIGH) - (SEC * SHIGH)) * MCANC
}

function: sixes fudge PRI:n SEC:n {
  PCANC: 1 - (PRI = 6)
  PRI: ((PRI - 1) * PCANC) + 1

  SCANC: 1 - (SEC = 6)
  SEC: ((SEC - 1) * SCANC) + 1

  MCANC: 1 - (PRI = SEC)

  PHIGH: PRI < SEC
  SHIGH: SEC < PRI

  result: ((PRI * PHIGH) - (SEC * SHIGH)) * MCANC
}

output [fudge 1d6 1d6] named "OG Fudge"

function: loe DICE:d MOD:n {
 RT:0 RS:0 RP:0 EP:0 ET:0

 loop N over {DICE+MOD}{
  if N<3                        {ET:ET+1}
  else if [{3..5}  contains N]  {EP:EP+1}
  else if [{6..8}  contains N]  {RP:RP+1}
  else if [{9..11} contains N]  {RS:RS+1}
  else                          {RT:RT+1} 
 }
 result:{-2:ET,-1:EP,0:RP,1:RS,2:RT}
}

function: loe MOD:n {result:[loe 1d12 MOD]}

output [loe +2] named "Land of Eem"

function: dsix PROLL:n and NROLL:n{
  if PROLL = 6 {PROLL: 1}
  if NROLL = 6 {NROLL: 1}

  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}

function: broken dsix PROLL:n and NROLL:n{
  if PROLL = 6 {PROLL: 0}
  if NROLL = 6 {NROLL: 0}

  if PROLL = NROLL {result: 0}
  if PROLL < NROLL {result: PROLL}
  result:-NROLL
}

output 4d{-1..1} named "Fudge"
output [dsix d6 and d6] named "2d6 fudge"
output [broken dsix d6 and d6] named "broken 2d6 fudge"

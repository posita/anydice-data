MISS: 0
HIT: 1
CRITICALHIT: 1

function: successes from ROLL:n plus ATTACK:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 6 {
  result: CRITICALHIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: attack ROLL:s plus ATTACK:n vs DEFENSE:n {
 COUNT: 0
 loop P over {1..#ROLL} {
  COUNT: COUNT + [successes from P@ROLL plus ATTACK vs DEFENSE]
 }
 result: COUNT
}

output [attack 6d6 plus 5 vs 8]

function: attack ROLL:n with BONUS:n {
 if ROLL = 1 {result:0 }
 if ROLL >=CRIT_RANGE {result:DAMAGE+CRIT_DAMAGE }
 if ROLL+BONUS+ ATTACK_BONUS >= TARGET_AC { result:DAMAGE }
 result:0
}

function: multiattack MULTIATTACK:n {
TOTAL:0
loop N over {1..MULTIATTACK} {
 TOTAL:TOTAL+[attack ROLL with ATTACK_BONUS_DIE]
 }
result:TOTAL
}

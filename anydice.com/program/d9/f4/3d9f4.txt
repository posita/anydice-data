function: pbta MOD:n {
 R:0
 RP:0
 E:0 
 loop N over {2d6+MOD}{
       if N<7 {E:E+1}
  else if [{7..9} contains N] {RP:RP+1}
  else                        {R:R+1} 
 }
 result:{-1:E,0:RP,1:R}
}
output [pbta -1] named "2d6-1"
output [pbta 0] named "2d6+0"
output [pbta 1] named "2d6+1"
output [pbta 2] named "2d6+2"
output [pbta 3] named "2d6+3"

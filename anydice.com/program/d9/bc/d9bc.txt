ATR: 2d6+1
SHRM: 2d6+9
T: 2d6+5
IS: 2d6+10
R: 2
function: miss A:s vs D:s and R:n {
  result: A <= D+R
}

function: stun A:s vs D:s and R:n {
  result: A > D+R & (A - (D+R)) <= R
}

function: immobilize A:s vs D:s and R {
  result: A > D+R & (A - (D+R)) > R & (A-(D+R)) <= R*2
}

function: kill A:s vs D:s and R{
  result: A > D+R & (A - (D+R)) > R*2
}

output [miss ATR vs T and R] named "ATR vs T-34 Miss"
output [stun ATR vs T and R] named "ATR vs T-34 Stun"
output [immobilize ATR vs T and R] named "ATR vs T-34 Immobilize"
output [kill ATR vs T and R] named "ATR vs T-34 kill"

output [miss SHRM vs T and R] named "M3 75mm vs T-34 Miss"
output [stun SHRM vs T and R] named "M3 75mm vs T-34 Stun"
output [immobilize SHRM vs T and R] named "M3 75mm vs T-34 Immobilize"
output [kill SHRM vs T and R] named "M3 75mm vs T-34 kill"

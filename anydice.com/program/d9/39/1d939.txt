loop P over {6,5,4,3,2} {
\function: disadv ROLL:s {
 if {2,3}@ROLL = 12 {
  result: 66
 }
 if {2,3}@ROLL = 2 {
  result: -11
 }
 if 2@ROLL >= P {
  result: 1 + (3@ROLL >= P)
 }
 if 2@ROLL < P {
  result: 0
 }
}
output [disadv 3d6] named "[P] or Higher! DISADVANTAGE"\

function: rpg ROLL:s {
 if {1,2}@ROLL = 12 {
  result: 66
 }
 if {1,2}@ROLL = 2 {
  result: -11
 }
 if 1@ROLL >= P {
  result: 1 + (2@ROLL >= P)
 }
 if 1@ROLL < P {
  result: 0
 }
}
output [rpg 2d6] named "[P] or Higher!"

\output [rpg 3d6] named "[P] or Higher! ADVANTAGE"\
}

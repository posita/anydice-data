function: count past VALUES:n in SEQUENCE:s {
 COUNT:(VALUES <= SEQUENCE)
 result: COUNT
}

function: count below VALUES:n in SEQUENCE:s {
 COUNT:(VALUES >= SEQUENCE)
 result: COUNT
}

function: pool DICESEQ:s threshold THRESH:n crit CRIT:n maxcrit MAX:n bonus BONUS:n anticrit ANTICRIT:n{
 COUNT: 0
 CRITCOUNT: [count past CRIT in DICESEQ]
 if CRITCOUNT = 0 | MAX = 0{ 
   COUNT: [count past THRESH in DICESEQ]
 }else if CRITCOUNT >= MAX{
   COUNT: [count past THRESH in DICESEQ]
   COUNT: COUNT + [count past 5 in MAX d6]
 }else{
   COUNT: [count past THRESH in DICESEQ]
   COUNT: COUNT + [count past 5 in CRITCOUNT d6]
 }
if (BONUS >= 0){
 result: (COUNT + [count past 4 in BONUS d6])
}else{result: (COUNT - [count past 5 in (BONUS*-1) d6])}

}

function: limit SEQUENCE:s to NUMBER:n {
 NEWSEQ : {}
 loop P over {1..#SEQUENCE}{
  if (P@SEQUENCE > NUMBER){
  NEWSEQ: {NEWSEQ, NUMBER}
 }else if (P@SEQUENCE <= -1){
  NEWSEQ : {NEWSEQ, 0}
 }else{
  NEWSEQ : {NEWSEQ, P@SEQUENCE}
 }
}
result: NEWSEQ
}


\output [pool 5d6 threshold 3 crit 5 maxcrit 3] named "crits, unlimited"\
\loop X over {0..4}{
 Y : 6-X
 output [limit [pool 5d6 threshold Y crit 6 maxcrit 1 bonus 0 anticrit 0] to 5] named "threshold [Y]"
}\

\loop X over {-2..2}{
 output [limit [pool 5d6 threshold 4 crit 6 maxcrit 1 bonus X anticrit 0] to 5] named "threshold 4, [X] bonus"
}\

 Y : 6-X
 loop Z over {-2..2}{
  output [limit [pool 5d6 threshold Y crit 6 maxcrit 1 bonus Z anticrit 0] to 5] named "threshold [Y], [Z] bonus"
}}

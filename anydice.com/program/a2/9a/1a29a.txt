MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n and ROLL:n vs CON:n {
if ROLL + CON <= 15{
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE + VENOM
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE +VENOM
 }
 result: MISSDAMAGE}
else{
if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE}
}

SKILL: {12,10,8,6,4}
ATTRS: {0..4}
KEEP: 2

\function: roll A:n by S:n {
  if S = -3 {
    result: [lowest 1 of KEEPd12]
  } else if S = -2 {
    result: 1d12
  } else if S = -1 {
    result: [highest 1 of KEEPd12]
  } else {
    SD: (S+1)@SKILL
    BONUS: 2*(12-SD)
    result: BONUS+[highest KEEP of (KEEP+A)dSD]
  }
}\

function: roll A:n by S:n {
  SD: (S+1)@SKILL
  result: [lowest KEEP of (KEEP+A)dSD]
}

\output [roll 0 by 0] < [roll 0 by 0]\
\output [roll 0 by 1]
output [roll 0 by 2]
output [roll 0 by 3]
output [roll 0 by 4]\

\output [roll 0 by 0]
output [roll 1 by 0]
output [roll 2 by 0]
output [roll 3 by 0]
output [roll 4 by 0]\

\output [roll 0 by 0] > [roll 1 by 0]
output [roll 1 by 0] > [roll 2 by 0]
output [roll 2 by 0] > [roll 3 by 0]
output [roll 3 by 0] > [roll 4 by 0]\

\output [roll 0 by 1] >= [roll 1 by 0] named "0/1 >= 1/0"
output [roll 0 by 2] >= [roll 2 by 0] named "0/2 >= 2/0"
output [roll 0 by 3] >= [roll 3 by 0] named "0/3 >= 3/0"
output [roll 0 by 4] >= [roll 4 by 0] named "0/4 >= 4/0"\

\output [roll 2 by 1] >= [roll 1 by 0] named "0/1 >= 1/0"\
\output [roll 1 by 2] >= [roll 2 by 1] named "1/2 >= 2/1"
output [roll 1 by 2] >= [roll 3 by 1] named "1/2 >= 3/1"
output [roll 1 by 3] >= [roll 4 by 1] named "1/3 >= 4/1"\

\output [roll 0 by -3] named "0/-3"
output [roll 0 by -2] named "0/-2"
output [roll 0 by -1] named "0/-1"
output [roll 0 by 0] named "0/0"
output [roll 1 by 0] named "1/0"\

\output [roll 0 by 1] 
output [roll 1 by 0]
output [roll 0 by 2] 
output [roll 2 by 0]\
\output [roll 0 by 3] named "0/3"\
\
output [roll 1 by -1] named "1/-1"
output [roll 2 by -1] named "2/-1"
output [roll 3 by -1] named "3/-1"
output [roll 3 by 0] named "3/0"\
\output [roll 0 by 4] named "0/4"
output [roll 4 by 0] named "4/0"\


loop A over ATTRS {
loop S over {0..4} {
output [roll A by S] named "[A]/[S]"
}
}

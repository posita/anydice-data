function: ironsworn ROLL:n TARGETA:n TARGETB:n {
  if TARGETA = TARGETB {
    if ROLL > TARGETA {
      result: 3
    }
    result: -1
  }
  ACC: 0
  if ROLL > TARGETA {
    ACC: ACC + 1
  }
  if ROLL > TARGETB {
    ACC: ACC + 1
  }
  result: ACC
}

output [ironsworn 1d6 1d10 1d10] named "Straight"
loop I over {1..5} {
  output [ironsworn 1d6+I 1d10 1d10] named "+[I]"
}

DIE: d{1..3,1..3,2,8}

function: explode N:n {
 if N = 8 { result: 3 + [explode DIE] }
 result: N
}

loop D over {1..6} {
 set "maximum function depth" to 2
 output Dd[explode DIE]+D named "[D]dTRI8+[D] expl."
}
loop P over {7..9} {
 set "maximum function depth" to 2
 output Pd[explode DIE]+6 named "[P]dTRI8+6 expl."
}

function: skew LEV for DIE {
  if LEV > 1 { result: [skew LEV/2 for DIE] + [skew LEV-(LEV/2) for DIE] }
  if LEV = 1 { result: [highest of DIE and DIE] }
}

function: level LEV for DIE {
  if LEV > 10 {
    result: [skew 9 for DIE] + (LEV-10)*[highlev DIE] + [maximum of DIE]
  } else {
    if LEV > 1 { result: [skew LEV-1 for DIE] + [maximum of DIE] }
    else { result: [maximum of DIE] }
  }
}

function: highlev DIE {
  HP: [maximum of DIE]
  if HP >= 12 { result: 5 } 
  else {
    if HP <= 6 { result: 3 }
    else { result: 4 }
  }
}

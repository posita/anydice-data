\ Calculate number of successes. Limit of 7. 
  @param SUM: the dice result
  @param STEP: the success step
  @return: # of successes \
function: succ SUM:n step STEP:n {
  if SUM > 35 {result: 35/STEP}
  if SUM > 2 {result: SUM/STEP}
  if SUM < 0 {result: -1}
  result: 0
}
\ Calculate number of successes. Limit of 7. Default step of 5. 
  @param SUM: the dice result
  @return: # of successes \
function: succ SUM:n {
  result: [succ SUM step 5]
}

function: roll ATTR:d with SKILL:d mod DIFFICULTY:n step STEP:n  {
  SUM: [explode ATTR] + [explode SKILL] - (5 + DIFFICULTY)
  result: [succ SUM step STEP]
}
function: roll ATTR:d with SKILL:d mod DIFFICULTY:n  {
  result: [roll ATTR with SKILL mod DIFFICULTY step 5]
}

function: aid A:n with B:n {
  H: [highest of A and B]
  if A > 0 & B > 0 { result: H + 1 }
  result: H
}
function: aid A:n with B:n with C:n {
  I: [highest of B and C]
  H: [highest of A and I]
  if (A = B & B = C){result: H + 2}
  if ((A > 0) + (B > 0) + (C > 0)) > 1 {
    result: H + 1
  }
  result: H
}
function: aid A:n with B:n with C:n with D:n {
  I: [highest of C and D]
  J: [highest of A and B]
  H: [highest of I and J]
  if ((A > 0) + (B > 0) + (C > 0) + (D > 0)) > 2 {
    result: H + 2
  }
  if ((A > 0) + (B > 0) + (C > 0) + (D > 0)) > 1 {
    result: H + 1
  }
  result: H
}

function: bound A:n {
  if A < 0 {result: 0}
  result: A
}

function: is succ A:n {
  if A > 0 {result: 1}
  result: 0
}

function: spread A:n {
  if A < 2 {result: A}
  X: 1
  loop N over {1..A}{
     X: X*2
  }
  if X > 64 {result: 64}
  result: X
}

function: add half N:n to ATTR:d with SKILL:d vs D:n {
  X: 0
  if N = 1 {X: 1}
  if N = 2 {X: 1}
  if N = 3 {X: 2}
  if N = 4 {X: 2}
  if N = 5 {X: 3}
  result: [roll ATTR with SKILL mod D] + X
}

function: attack A:n with D:n armor AR:n{
  if A > 0 {
    if D > AR {result: A + [succ D - AR]}
    if D < AR {result: A - [succ AR - D]}
    result: A
  }
  result: 0
}

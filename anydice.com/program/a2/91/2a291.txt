
\ attack die value \
X: 3d8+5
\ defense die value \
Y: 3d8
YM: 0

\ Armor scaling value \
ARMORSCORE: 6

\ Block function \

function: block A B {
 C: A / 2
 D: 0
 if(C > B) { D: A - B }
 D: A - C
 if(D < A) { result: A - D }
 result: 0
}

function: negate A B {
 if(B < A) { result: A - B }
 result: 0
}

\ Armor Rolltable Functions \

function: arm light leather A:n B:n {
 if(B >= 17) { result: [negate A (ARMORSCORE * 1000)] }
 result: A
}

function: arm medium chain A:n B:n {
 if(B >= 19) { result: [negate A (ARMORSCORE)] }
 if(B >= 13) { result: [block A (ARMORSCORE)] }
 result: A
}

function: arm heavy plate A:n B:n {
 if(B >= 15) { result: [block A (ARMORSCORE * 1000)] }
 if(B >= 11) { result: [block A (ARMORSCORE)] }
 result: A
}


\ Attack Function (set the target armor here) \

function: atk A {
 result: [arm heavy plate A Y]
}




\ Weapon Rolltable Functions \

function: wep dagger A:n {
 if A >= 18 { result: 20 + 1d8 }
 if A >= 9 { result: 2 + 1d4 }
 result: 0
}

function: wep greatsword A:n {
 if A >= 18 { result: 10 + 1d12 }
 if A >= 15 { result: 8 + 1d4 }
 if A >= 10 { result: 4 + 1d4 }
 result: 0
}



\ Weapon damage outputs \

\output [arm heavy plate [wep greatsword X] Y+(YM*2)] named "Greatsword vs Heavy Plate"
output [arm medium chain [wep greatsword X] Y+YM] named "Greatsword vs Medium Chain"
output [arm light leather [wep greatsword X] Y+YM] named "Greatsword vs Light Leather"

output [arm heavy plate [wep dagger X] Y] named "Dagger vs Heavy Plate"
output [arm medium chain [wep dagger X] Y+YM] named "Dagger vs Medium Chain"
output [arm light leather [wep dagger X] Y+YM] named "Dagger vs Light Leather"\

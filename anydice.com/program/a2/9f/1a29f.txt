\ Comparing highest result. \
 \ 1 is a successful roll  \
  \ 0 is a failed roll    \

DIEARRAY: {4,6,8,10,12,20}
Z: 12
loop K over DIEARRAY {
output 42 named "DIE: [K]"
loop I over {-3..3} {
    output 1dZ+I > 1dK named "1d[Z] + [I] > 1d[K]"
    \output 1dZ+I < 1dK named "1d[Z] + [I] < 1d[K]"\
}
}

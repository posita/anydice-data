\Global Constants\
MISS: 0
HIT: 1
CRIT: 2

\User Variable Input\
TOHIT: 5 \to hit modifier - should be a constant\
DMG: 1d10+5 \attack damage - should be dice + modifier\
MISSDMG: 0 \damage done on a miss - usually 0 but some abilities may deal damage on miss\
AC: 20 \enemy AC - should be constant\

function: attack ROLL:n vs AC:n {
 if ROLL = 1 {result: MISS}
 if ROLL = 20 {result: CRIT}
 if ROLL + TOHIT > AC {result: HIT}
 else {result: MISS}
}

function: damage for RESULT:n {
 if RESULT = CRIT {result: CRIT * DMG}
 if RESULT = HIT {result: DMG}
 if RESULT = MISS {result: MISSDMG}
}

function: attack {
 result: [damage for [attack 1d20 vs AC]]
}

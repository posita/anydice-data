\ A variant of exploding dice, where you replace the maximum result of one die with two new dice. \

DICE: 12

function: ob NUMBER:n {
  TOTALSUM: 0
  loop LOOPING over {1..NUMBER} {
    X: 1d6
    X: [checkmax X]
    TOTALSUM: TOTALSUM + X
  }
  result: TOTALSUM
}

function: checkmax N:n {
  if N = 6 {
    result: [ob 2]
  }
  result: N
}

set "maximum function depth" to 7    \ Change this to a lower number if you get a timeout. \

  output [ob DICE] named "OB[DICE]T6 "

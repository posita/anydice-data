MISS: 0
HIT: 1
CRIT: 2

function: attack ROLL:n vs DEFENSE:n{
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRIT}
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRIT {
  result: CRITDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: 0
}

function: attack {
 result: [damage for [attack d20 vs AC]]
}

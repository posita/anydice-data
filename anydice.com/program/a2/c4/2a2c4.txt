function: dice for rank N {
 if N > 0 {
  if N < 6 {
   result: 2 + (2 * N)
  }
 }
}

function: two highest of ATTR:s ABIL:s {
  TOPS: {1@ATTR, 2@ATTR, 1@ABIL, 2@ABIL}
  result: [two highest of tops TOPS]
}
function: two highest of ATTR:s ABIL:s RES:s {
  TOPS: {1@ATTR, 2@ATTR, 1@ABIL, 2@ABIL, 1@RES, 2@RES}
  result: [two highest of tops TOPS]
}
function: two highest of ATTR:s ABIL:s RES:s STA:s {
  TOPS: {1@ATTR, 2@ATTR, 1@ABIL, 2@ABIL, 1@RES, 2@RES, 1@STA, 2@STA}
  result: [two highest of tops TOPS]
}
function: two highest of tops TOPS:s {
  SORTED: [sort TOPS]
  result: {1,2}@SORTED
}

loop ATTR over {1..5}{
 loop ABIL over {1..5}{
  DATTR: [dice for rank ATTR]
  DABIL: [dice for rank ABIL]
  output [two highest of 1dDATTR 1dDABIL] named "d[DATTR], d[DABIL]"

  \
  DRES: [dice for rank 1]
  output [two highest of 1dDATTR 1dDABIL 1dDRES] named "d[DATTR], d[DABIL], d[DRES]"
  \
 }
}

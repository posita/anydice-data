function: count successes A:s B:s C:s D:s E:s F:s vs TN:n{
    GROUP:[sort{A,B,C,D,E,F}]
    if 1@GROUP< TN {result:-1}
    if 1@GROUP= TN {result: 0}
    result: GROUP>TN
}

loop N over {2,4,6,8,10,12}{
    output [count successes 0d2 0d4 1d6 1d8 1d10 0d12 vs 1@2dN] named "TN 2d[N] at DR12: d6,d8,d10"
    output [count successes 0d2 0d4 0d6 0d8 0d10 2d12 vs 1@2dN] named "TN 2d[N] at DR12: 2d12"
}

\ handwritten variations on DR12
    output [count successes 12d2 0d8 0d10 vs [highest 1 of 3d8]] named "TN d8 at DR12:6d4"
    output [count successes 6d4 0d8 0d10 vs [highest 1 of 3d8]] named "TN d8 at DR12:6d4"
    output [count successes 4d6 0d8 0d10 vs [highest 1 of 3d8]] named "TN d8 at DR12:4d6"
    output [count successes 0d4 3d8 0d10 vs [highest 1 of 3d8]] named "TN d8 at DR12:3d8"
    output [count successes 1d6 1d8 1d10 vs [highest 1 of 3d8]] named "TN d8 at DR12:d6,d8,d10"
    output [count successes 1d2 1d10 1d12 vs [highest 1 of 3d8]] named "TN d8 at DR12:d12,d10,d2"
    output [count successes 0d6 0d8 2d12 vs [highest 1 of 3d8]] named "TN d8 at DR12:2d12"
\

\ Action Point Nova Round 								\
\ Actions:										\
\	Minor - summon silt elemental 							\
\	Move - order silt elemental to move; deal 5 damage, slow target 		\
\	Standard - charge								\
\	Action Point- charge 								\
\---------------------------------------------------------------------------------------\
\ FUNCTIONS										\
\---------------------------------------------------------------------------------------\

\ a function for the first attack's damage calculation \
	function: X:n conditional {
		if X - MOD = 1 {  \natural 1\
			MISS:1
			result: 0
		}
		if X - MOD = 20 {result: 50+1d8} \critical hit\
		if X >= AC {result: 5d8+10}	 \hit\
		else { 
			\Anya missed the attack, so SA can apply to the second one\
			MISS:1
			result:0
		} 
	}

\ a function for the second attack's damage calculation \
	function: X:n testing {
		if MISS = 0 { \Anya didn't miss the first attack\
			if X - MOD = 1 { result: 0}  \natural 1\
			if X - MOD = 20 {result: 34+1d8} \critical hit\
			if X >= AC {result: 3d8+10} \hit\
			else {result: 0}
		} else { \we missed the first attack, so SA can apply here\
			if X - MOD = 1 { result: 0}  \natural 1\
			if X - MOD = 20 {result: 50+1d8} \critical hit\
			if X >= AC {result: 5d8+10} \hit\
			else {result: 0}
		}
	}

\---------------------------------------------------------------------------------------\
\ CALCULATIONS										\
\---------------------------------------------------------------------------------------\

\ variable declarations \
	MISS: 0	\ we have not missed our first attack \
	MOD: 16	\ attack bonus is +16 \
	AC: 20 \ first target AC is 20 \

\ Case: attacking level 6 controller, lurker, or skirmisher (AC = 20) \
	MISS: 0	\ we have not missed our first attack \
	AC: 20	\ target AC is 20 \
	output [d20 + MOD conditional] + [d20 + MOD testing] + 5
 		named "Anya nova round (controller, lurker, skirmisher)"

\ Case: attacking level 6 soldier (AC = 22) \

	AC: 22	\ target AC is 22 \
	output [d20 + MOD conditional] + [d20 + MOD testing] + 5
 		named "Anya nova round (soldier)"

\ Case: attacking level 6 brute or artillery (AC = 18) \

	AC: 18	\ target AC is 18 \
	output [d20 + MOD conditional] + [d20 + MOD testing] + 5
 		named "Anya nova round (brute, artillery)"

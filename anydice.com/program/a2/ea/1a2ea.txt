SIDES: 6
MAXD: 3
SKILL: 4

loop SKILL over {1..8}{
function: roll SKILL:n{
 if SKILL = 1{
  result: [explode 1dSIDES]}
 if SKILL <= MAXD{
  result: [explode 1dSIDES] + (SKILL-1)dSIDES}
 result: ((SKILL-MAXD)*((SIDES/2)+1)) + [explode 1dSIDES] + (MAXD-1)dSIDES
}

output [roll SKILL] named "Skill lvl [SKILL]"
}

function: attack ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS + ARCH >= AC { result: DAMAGE }
 result: 0
}

ENEMYAC: 18

ATTACKMOD: 5

PROFICIENCY: 4
CRITROLL: 20
EXTRAATTACK: 1
DMGBONUS: 0
ACCBONUS: 0
EXTRADMGDICE: 0

ARCH: 0

DAMAGE: 3d10 + ATTACKMOD + DMGBONUS + EXTRADMGDICE
CRITDAMAGE: 6d10 + ATTACKMOD + DMGBONUS + 2dEXTRADMGDICE

output EXTRAATTACKd[attack d20 vs ENEMYAC] named "Firebolt"

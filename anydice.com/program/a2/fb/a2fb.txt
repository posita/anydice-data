  if DIE > 1{result: 1}
  result: 0

 result: 0

function: check die step DIE:n for HITSIZE:n{
 if HITSIZE = SMALLHIT{
  TEMP: DIE-2
  if TEMP < 3 {TEMP: 3}
  result: TEMP
 }
 if HITSIZE = BIGHIT{
  if DIE = 12{
   result: 20
  }
  result: DIE+2
 }
 result: DIE
}

function: attack against DIE:n doing NUMWOUNDS:n DAMAGE:n rolling ROLL:n plus CHANCE:n to hit against AC:n{
 RESULT: 0
 HITRESULT: 0
 if ROLL + CHANCE > AC { HITRESULT: HIT}
 else{HITRESULT: MISS}
 loop N over {1..NUMWOUNDS}{
  if HITRESULT = HIT{
   RESULT: RESULT + [check wound 1d[check die step DIE for DAMAGE]]
  }
 }
 result: RESULT
}

function: magic attack against DIE:n doing NUMWOUNDS:n DAMAGE:n rolling ROLL:n plus CHANCE:n to hit against AC:n{
 RESULT: 0
 HITRESULT: 0
 if ROLL + CHANCE > AC { HITRESULT: HIT}
 else{HITRESULT: MAGICMISS}
 loop N over {1..NUMWOUNDS}{
  if HITRESULT = HIT{
   RESULT: RESULT + [check wound 1d[check die step DIE for DAMAGE]]
  }
  else{
   RESULT: RESULT + [check wound [lowest 1 of 2d[check die step DIE for DAMAGE]]]
  }
 }
 result: RESULT
}

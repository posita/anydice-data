function: check wound ROLL:n{
 if ROLL > 1 {result: 1}
 result: 0
}

function: damage for DIE:n rolling ROLL:n plus ATTACK:n vs DEFENSE:n {
 if ROLL + ATTACK >= DEFENSE { 
  if DIE > 1{result: 1}
  result: 0
 }
 result: 0
}

function: check die step DIE:n for HITSIZE:n{
 if HITSIZE = SMALLHIT{
  TEMP: DIE-2
  if TEMP < 3 {TEMP: 3}
  result: TEMP
 }
 if HITSIZE = BIGHIT{
  if DIE = 12{
   result: 20
  }
  result: DIE+2
 }
 result: DIE
}

function: attack against DIE:n doing NUMWOUNDS:n DAMAGE:n rolling ROLL:n plus CHANCE:n to hit against AC:n{
 RESULT: 0
 HITRESULT: 0
 if ROLL + CHANCE > AC { HITRESULT: HIT}
 else{HITRESULT: MISS}
 loop N over {1..NUMWOUNDS}{
  if HITRESULT = HIT{
   RESULT: RESULT + [check wound 1d[check die step DIE for DAMAGE]]
  }
 }
 result: RESULT
}

function: multiattack against DIE:n with NUM:n attacks doing NUMWOUNDS:n DAMAGE:n with CHANCE:n to hit against AC:n{
 RESULT: 0

 loop N over{1..NUM}{
  RESULT: RESULT + [attack against DIE doing NUMWOUNDS DAMAGE rolling 1d20 plus CHANCE to hit against AC]
 }
 result: RESULT
}

function: magic attack against DIE:n doing NUMWOUNDS:n DAMAGE:n rolling ROLL:n plus CHANCE:n to hit against AC:n{
 RESULT: 0
 HITRESULT: 0
 if ROLL + CHANCE > AC { HITRESULT: HIT}
 else{HITRESULT: MAGICMISS}
 loop N over {1..NUMWOUNDS}{
  if HITRESULT = HIT{
   RESULT: RESULT + [check wound 1d[check die step DIE for DAMAGE]]
  }
  else{
   RESULT: RESULT + [check wound [lowest 1 of 2d[check die step DIE for DAMAGE]]]
  }
 }
 result: RESULT
}

function: favorable magic attack against DIE:n doing NUMWOUNDS:n DAMAGE:n rolling ROLL:n plus CHANCE:n to hit against AC:n{
 RESULT: 0
 HITRESULT: 0
 if ROLL + CHANCE > AC { HITRESULT: HIT}
 else{HITRESULT: MAGICMISS}
 loop N over {1..NUMWOUNDS}{
  if HITRESULT = HIT{
   RESULT: RESULT + [check wound [highest 1 of 2d[check die step DIE for DAMAGE]]]
  }
  else{
   RESULT: RESULT + [check wound 1d[check die step DIE for DAMAGE]]
  }
 }
 result: RESULT
}

function: explosive attack against DIE:n doing NUMWOUNDS:n DAMAGE:n rolling ROLL:n plus CHANCE:n to hit against AC:n{
 RESULT: 0
 HITRESULT: 0
 if ROLL + CHANCE > AC { HITRESULT: HIT}
 else{HITRESULT: MAGICMISS}
 loop N over {1..NUMWOUNDS}{
  if HITRESULT = HIT{
   RESULT: RESULT + [check wound 1d[check die step DIE for DAMAGE]]
  }
  else{
   RESULT: RESULT + [check wound [lowest 1 of 2d[check die step DIE for DAMAGE]]]
  }
  DIE: DIE + 2
  if DIE > 12 {DIE: 12}
 }
 result: RESULT
}

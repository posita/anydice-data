DIE: {0,0,1,1,2,3}

function: roll ROLL:n {
  ROLLEDSUM: 0
  DICETOROLL: 5
  loop P over {1..5} {
    if P@ROLL = 3 {
      ROLLEDSUM: ROLLEDSUM+3
      DICETOROLL: DICETOROLL-1
    }
    \if P@ROLL = 2 {
      ROLLEDSUM: ROLLEDSUM+2
      DICETOROLL: DICETOROLL-1
    }\
    
  }
  result: ROLLEDSUM+DICETOROLLdDIE
  
}


output 5dDIE named "throw 5d DIE"
ONE: [roll 5dDIE]
output ONE

\TWO: [roll ONE]
output TWO

THREE: [roll TWO]
output THREE\

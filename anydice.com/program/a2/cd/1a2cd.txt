function: reroll doubles SEQ:s {
    BONUS: 0
    if [at least 2 0 in SEQ] {BONUS: -1}
    if [at least 2 5 in SEQ] {BONUS: 1}
    result: SEQ + BONUS
}
function: reroll edges a SEQ:s {
    BONUS: 0
    if [sum SEQ] <=  2 {BONUS: -[lowest 1 of 2d{0..5}]}
    if [sum SEQ] >= 13 {BONUS:  [lowest 1 of 2d{0..5}]}
    result: SEQ + BONUS
}

    result: SEQ + BONUS
}

}

}

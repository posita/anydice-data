

function: reroll ROLL:n {
    if ROLL = 4 {
      result: 4
    }
    if ROLL = 3 { 
      result: 1+d2
    }

    if ROLL = 2 {
      result: 2d2
    }
}

output 2d2
output [reroll 2d2]

function: rerollcoin ROLL:n to REPLACEMENT:d {
   if ROLL = 2 { result: 2*REPLACEMENT }
   if ROLL = 3 { result: 2+REPLACEMENT }
   if ROLL = 4 { result: 4 }
}

output 2d2
output [rerollcoin 2d2 to d2]

function: rerolleight ROLL:n to REPLACEMENT:d {
  if ROLL = 1 | ROLL = 2 { result: REPLACEMENT }
  result: ROLL

}

output 1d8
output [rerolleight d8 to d8]

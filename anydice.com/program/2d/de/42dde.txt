function: average of X die N {
 DICE: XdN
 SUM: 0
 TOTAL: X * N
 loop I over {1..N} {
  SUM: I * [count {I} in DICE] / TOTAL
 }
 result: SUM
}

output 2d3
output 1d3+1d2
output 1d5

output [highest 1 of 10d4] \avg 25\
output [average of 10 die 4]
output [highest 1 of 6d6]  \avg 21\
output [highest 1 of 4d8]  \avg 18\
output [highest 1 of 2d10] \avg 11\
output [highest 1 of d12]  \avg 6.5\

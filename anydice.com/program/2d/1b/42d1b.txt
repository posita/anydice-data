set "position order" to "highest first" \ the default behavior \

function: tally successes on VALUE:n {
 result: VALUE/4
}


loop SIDES over {4,6,8,10,12} {
 loop DICE over {1..5} {
  output [tally successes on 1@DICE d SIDES] named "[DICE]d[SIDES] HR"
 }
}


set "position order" to "lowest first" \ low at 1 \
loop SIDES over {4,6,8,10,12} {
 loop DICE over {1..5} {
  output 1@DICE d SIDES named "[DICE]d[SIDES] LR"
 }
}

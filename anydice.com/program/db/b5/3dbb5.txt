function: std HIT:n {
  result: (1d20 >= HIT)d8
}

function: adv HIT:n {
  result: ([highest 1 of 2d20] >= HIT)d8
}

function: shield HIT:n {
  ISHIT: 1d20 >= HIT
  DAM: [lowest 1 of 2d8]
  result: DAM * ISHIT
}

function: adv shield HIT:n {
  ISHIT: [highest 1 of 2d20] >= HIT
  DAM: [lowest 1 of 2d8]
  result: DAM * ISHIT
}


loop I over {9..16} {
  output [std I] named "Standard [I]"
  output [shield I] named "Shield [I]"
  \
  output [adv I] named "Adv To Hit [I]"
  output [adv shield I] named "Adv To Hit Shield [I]"
  \
}

BODY: {1,2,2,2,3,3}
MIND: {3,4,4,4,3,5}
SPIRIT: {5,6,6,6,{5,1},1}

function: evaluate ROLLA:s ROLLB:s TARGET:n {
  result: (ROLLA = TARGET) + (ROLLB = TARGET)
}

loop N over {1..6} {
  output [evaluate 4dBODY 2dMIND N] named "[N] for Skirmisher"
}

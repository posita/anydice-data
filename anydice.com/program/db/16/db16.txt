function: fumble with N dice {
    result: N@Nd20 = 1
}

function: error with N prof {
    result: 1d20+N <= 10
}

function: fumble with N dice and K prof {
    result: [fumble with N dice] & [error with K prof]
}

output [fumble with 2 dice] named "fumble 2dice"
output [fumble with 3 dice] named "fumble 3dice"
output [fumble with 4 dice] named "fumble 4dice"

output [fumble with 2 dice and 2 prof] named "fumble* 2dice 2prof"
output [fumble with 3 dice and 2 prof] named "fumble* 3dice 2prof"
output [fumble with 4 dice and 2 prof] named "fumble* 4dice 2prof"

output [fumble with 2 dice and 5 prof] named "fumble* 2dice 5prof"
output [fumble with 3 dice and 5 prof] named "fumble* 3dice 5prof"
output [fumble with 4 dice and 5 prof] named "fumble* 4dice 5prof"

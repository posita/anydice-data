   SUCCESS: SUCCESS - 1
  }
 }

  if C < D { \-Power Check-\
   SUCCESS: SUCCESS - 1
  }
 }

 \-Factor degree of success into damage-\
 if SUCCESS < 0 { \-If degree of success is -1, damage is always 0-\
  result: 0
 }
 else {
  if ACC_CHECK_FAIL { \-If acc check failed but success is at least 0, it means pwr check succeeded w/reduced die-\

    result: C/2

  }
 }
}

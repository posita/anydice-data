function: gwfdices X:n die Y:d{
  TOTAL: 0
  loop N over {1..X} { TOTAL: TOTAL + [gwf Y] }
  result: TOTAL
}

\ **** FUNCTIONS OK **** \
function: gwf D:d{ result: [rerollcore D die D] }
function: rerollcore R:n die D:d{ if R < 3 { result: D } else {result: R} }

\ **** OUTPUTs **** \
output [gwfdices 5 die d4]+[gwfdices 8 die d8]+[gwfdices 3 die d6]+3*(4+1) named "gwf full attacks"
output 5d4+8d8+3d6+3*(4+1) named "normal full attacks"

function: counter D:n s S:n f F:n 
dice DICE:d roll ROLL:n {
  D: D + 1
  if ROLL >= 10 { S: S + 1 }
  if ROLL <  10 { F: F + 1 }
  if ROLL =  20 { S: S + 1 }
  if ROLL =   1 { F: F + 1 }

  if S >= 3 { result: 1 }
  if F >= 3 { result: 0 }
  if D >  6 { result: -1 }

  result: -1
}

output [counter 0 s 0 f 0 dice d20 roll d20]

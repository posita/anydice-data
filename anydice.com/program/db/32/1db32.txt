Y: 1d6
X: 2d6
SKILL: 3
DIFF: 16
function: unique values in S:s {
  U: {}
  loop N over S { if !(N = U) { U: {U, N} } }
  result: U
}  
function: find PATTERN:s in ROLL:s {
  COUNTS: {}
  loop N over [unique values in ROLL] {
    COUNTS: {COUNTS, N = ROLL}
  }
  result: [sort PATTERN] = [sort COUNTS]
}
output X+Y+SKILL<=DIFF &  [find {3} in 3d6] named "paradox"
output X+Y+SKILL<=DIFF-4 & Y=1 named "dud"
output X+Y+SKILL<=DIFF-4 & Y>1 named "burnout"
output X+Y+SKILL>=DIFF-3 & X+Y+SKILL<DIFF named "success with concequence"
output X+Y+SKILL>=DIFF named "success"

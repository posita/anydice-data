function: attack ROLL:n AB AC DAM  {
  if (ROLL = 20) { result: DAM + DAM }
  if (ROLL = 1) | (ROLL + AB < AC) { result: 0 }
  result: DAM
}

function: attack AB AC DAM  {
  result: [attack 1d20 AB AC DAM]
}

function: multiattack N FUNC {
  NUM: 0
  loop I over {1..N} { NUM: NUM + FUNC }
  result: NUM
}

function: spell ROLL:n DC STB DAM {
  if (ROLL = 1) | (ROLL + STB < DC) { result: DAM }
  result: DAM / 2
}

function: spell DC STB DAM {
  result: [spell 1d20 DC STB DAM]
}

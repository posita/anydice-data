function: CORRUPCAO:n to count consequence {
  SUCESSO_CORRUPCAO: [count {8..10} in CORRUPCAO d{1:7,8..10}]*2
  result: SUCESSO_CORRUPCAO*2
}

function: NORMAL:n and CORRUPCAO:n to count consequence again {
  DIFERENCA: CORRUPCAO + NORMAL - 7
  SUCESSO_CORRUPCAO: [count {(8+DIFERENCA)..(10+DIFERENCA)} in CORRUPCAO d{1:7,8..10}]*2
  result: SUCESSO_CORRUPCAO*2
}

function: roll NORMAL:n and CORRUPCAO:n smaller {
  SUCESSO_NORMAL: [count {6..10} in NORMAL d{1:5,6..10}]
  SUCESSO_CORRUPCAO: [count {8..10} in CORRUPCAO d{1:7,8..10}]*2
  result: SUCESSO_NORMAL + SUCESSO_CORRUPCAO
}

function: roll NORMAL:n and CORRUPCAO:n bigger {
  DIFERENCA: CORRUPCAO + NORMAL - 7

  result: SUCESSO_NORMAL + SUCESSO_CORRUPCAO
}

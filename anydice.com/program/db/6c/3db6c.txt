set "maximum function depth" to 1
N: 2
D: 4

BASEDIE: dD

function: target SEQ:s reroll REROLL:n VALUE:n {
 loop TARGET over SEQ {
  if VALUE = REROLL {
    result: 1 + [target SEQ reroll REROLL BASEDIE]
  }
  if VALUE = TARGET {
    result: 1
  }
  result: 0
 }
}

CRIT: [target {1} reroll D BASEDIE]
WND: [target {2,3} reroll D BASEDIE]

output [count {1} in NdD] named "critical wound"
output NdCRIT named "critical exploding"

output [count {2,3} in NdD] named "wound"
output NdWND named "wound exploding"

output [count {1,2,3} in NdD] named "some wound"

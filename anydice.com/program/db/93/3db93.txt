function: if CONDITION:n then A else B{
  if CONDITION {result:A}
  else {result:B}
}

function: fatecheck ten ROLL:n{
  result: [if ROLL < 21 & ROLL > 17 then 2 
           else [
             if ROLL>10 then 1
             else [
               if ROLL > 0 & ROLL < 5 then -2
               else [
                 if ROLL < 11 then -1
                 else -2
               ]
             ]
           ]
          ]
}

function: fatecheck six mod ROLL:n at MOD:n and RED:n and BLACK:n{
  result: [if RED = 1 | RED = 10 then 
            [if BLACK + MOD > 5 then 2 else -2] 
             else [if ROLL+MOD > 10 then 1 else -1]
             ]
             
}

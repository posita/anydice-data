SIDES: 10
ROUNDUPAVG: (1+SIDES)/2 +1
ROUNDDOWNAVG: (1+SIDES)/2

MODIFIED_OUTCOMES: {}
loop N over {1..SIDES} {
  if N < ROUNDDOWNAVG {
    MODIFIED_OUTCOMES: {MODIFIED_OUTCOMES, ROUNDDOWNAVG}
  } else {
    MODIFIED_OUTCOMES: {MODIFIED_OUTCOMES, N}
  }
}
MODIFIED_DIE: d(MODIFIED_OUTCOMES)

output 1dSIDES named "Distribution for 1d[SIDES]"

output ROUNDUPAVG named "Rounded Up Average for 1d[SIDES]"

output ROUNDDOWNAVG named "Rounded Down Average for 1d[SIDES]"

output MODIFIED_DIE named "Modified Average (rolls < [ROUNDDOWNAVG] become [ROUNDDOWNAVG]) for 1d[SIDES]"

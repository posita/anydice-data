function: rerollcore R:n die D:d{ if R < 3 { result: D } else {result: R} }

function: bonusattack B:n{ result: [reroll d4]+B+1 }
function: fullround D:n{ result: [baseattack D]+[baseattack D]+[bonusattack D] }

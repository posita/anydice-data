\------------------------ [HELPERS] ------------------------\
function: minimum of DIE:d
{
  result: 1@{DIE}
}


\------------------ [GREAT WEAPON FIGHTING] ------------------\
function: great weapon fighting DIE:d
{
  NUM_DICE: [minimum of DIE]
  DIE_TYPE: [maximum of DIE] / NUM_DICE
  FINAL_VALUE: 0
  loop N over {1..NUM_DICE}
  {
    FINAL_VALUE: FINAL_VALUE + [get roll 1dDIE_TYPE]
  }

  result: FINAL_VALUE
}
function: get roll ROLL:n
{
  if ROLL > 2
  {
    result: ROLL
  }
  result: 1dDIE_TYPE
}


\--------------------- [CALCULATIONS] ---------------------\

output [great weapon fighting 2d6] named "2d6"
output [great weapon fighting 1d12] named "1d12"
output [great weapon fighting 5d6] named "5d6"
output [great weapon fighting 3d12] named "3d12"
output [great weapon fighting 6d6] named "6d6"
output [great weapon fighting 4d12] named "4d12"

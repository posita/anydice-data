PROF : 4
MOD : 5
MAGIC: 1
AC: 17

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n and SMITE:d {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG + 2dSMITE
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

R : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC+1d4 vs AC for 1d8 + MOD + MAGIC + 6 crit 1d8 on 19 and 4d8]
PR :[attack [highest 1 of 1d20] plus MOD+MAGIC+1d4 vs AC for 1d8 + MOD + MAGIC + 6 + 2*PROF crit 1d8 on 19 and 4d8]
EB :[attack [highest 1 of 1d20] plus MOD+PROF+MAGIC+1d4 vs AC for 1d10+MOD crit 1d10 on 19]

output 2dR named "10th level Relius with Rapier, Hexblade's curse"
output 2dPR named "10th level Relius with Rapier,Hexblade's curse and power attack"
output 3dEB named "10th level Relius with Eldritch blast"

\ This function returns +1 if A wins, -1 if B wins and 0 if it's a tie: \

function: ATTR_A:n versus ATTR_B:n 
{
  result: [1d20 attribute ATTR_A versus 1d20 attribute ATTR_B]  \ call helper function \
}

function: ROLL_A:s attribute ATTR_A:n versus ROLL_B:s attribute ATTR_B:n 
{
  SUCCESS_A: ROLL_A <= ATTR_A
  SUCCESS_B: ROLL_B <= ATTR_B

  \ the higher number of successes wins: \
  if SUCCESS_A > SUCCESS_B { result: +1 }
  if SUCCESS_A < SUCCESS_B { result: -1 }

  \ if both player's rolls are a success, the highest roll wins: \
  if SUCCESS_A = 1 & ROLL_A > ROLL_B { result: +1 }
  if SUCCESS_A = 1 & ROLL_A < ROLL_B { result: -1 }

  \ otherwise it's a tie: \
  result: 0
}

A: 6 \ attribute target value for player 1 \
B: 5 \ attribute target value for player 2 \

output [A versus B] named "[A] vs. [B]"

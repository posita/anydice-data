PROF : 3
MOD : 4
WIS_MOD : 3
AC: 15
ARCHERY: 2

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: save ROLL:n plus BONUS:n vs DC:n for DMG:d fail FAIL_DMG:d {
    if ROLL + BONUS < DC {
        result: DMG
    }
    result: FAIL_DMG
}

X : [attack 1d20 plus MOD vs AC for 1d6+MOD+2*PROF crit 1d6 on 20]
Y : [attack 1d20 plus PROF+MOD vs AC for 1d6+MOD crit 1d6 on 20]
Z : [attack 1d20 plus PROF+WIS_MOD vs AC for 1d8+2+PROF crit 1d8 on 20]

output 2dX+Y+1dZ named "5th level beast master ranger attacking with 2 Shortswords using heavy strike (also using two-weapon fighting style) and his bonus action to allow the beast to attack"


function: ROLL:n vs SCORE:n {
  TOTAL: 0
  if ROLL > (SCORE * 2) {TOTAL: TOTAL -1}
  if ROLL <= [highest of 1 and (1 + (SCORE-1)/20)] {TOTAL: TOTAL +1}
  if  ROLL <= SCORE/4 {TOTAL: TOTAL +1}
  if  ROLL <= SCORE/2 {TOTAL: TOTAL +1}
  if  ROLL <= SCORE {TOTAL: TOTAL +1}
  if  ROLL = 20 {TOTAL: 0}
  result: TOTAL
}

loop D over {1..10}{
  DEF: D*2 + 5
  loop A over {D..10} {
    ATT: A*2 +5
    output [1d20 vs ATT] - [1d20 vs DEF] named "[ATT] vs [DEF]"
  }
}

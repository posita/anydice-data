PROF : 4
MOD : 5
MAGIC: 1
AC: 17

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n and SMITE:d {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG + 2dSMITE
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

R : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d8 + MOD + MAGIC + 2 crit 1d8 on 20]
PR :[attack [highest 1 of 1d20] plus MOD+MAGIC vs AC for 1d8 + MOD + MAGIC + 2 + 2*PROF crit 1d8 on 20]

RH : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 1d8+1d6 + MOD + MAGIC + 2 crit 1d8+1d6 on 20]
PRH :[attack [highest 1 of 1d20] plus MOD+MAGIC vs AC for 1d8+1d6 + MOD + MAGIC + 2 + 2*PROF crit 1d8+1d6 on 20]

RSM : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC+1d4 vs AC for 1d8 + MOD + MAGIC + 6 crit 1d8 on 19 and 4d8]
PRSM :[attack [highest 1 of 1d20] plus MOD+MAGIC+1d4 vs AC for 1d8 + MOD + MAGIC + 6 + 2*PROF crit 1d8 on 19 and 4d8]

EB :[attack [highest 1 of 1d20] plus MOD+PROF vs AC for 1d10+MOD+1d6 crit 1d10+1d6 on 20]

RIS : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC vs AC for 2d8 + MOD + MAGIC + 2 crit 2d8 on 20]
PRIS :[attack [highest 1 of 1d20] plus MOD+MAGIC vs AC for 2d8 + MOD + MAGIC + 2 + 2*PROF crit 2d8 on 20]



output 2dR named "10th level Relius with Rapier"
output 2dPR named "10th level Relius with Rapier and power attack"
output 2dRIS named "12th level Relius with Rapier with improved smite"
output 2dPRIS named "12th level Relius with Rapier with improved smite and power attack"

output 2dRH named "10th level Relius with Rapier and Hex"
output 2dPRH named "10th level Relius with Rapier, Hex and power attack"

output 2dPRSM named "10th level Relius with Rapier, Bless, Smite on Crit, Hexblade's curse and power attack"

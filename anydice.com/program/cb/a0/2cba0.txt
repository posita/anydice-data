PROF : 3
MOD : 4
AC: 15

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

X : [attack 1d20 plus MOD vs AC for 1d6 + MOD  + 2*PROF crit 1d6 on 20]
Z : [attack 1d20 plus PROF+MOD vs AC for 1d6 + MOD           crit 1d6 on 20]

output 2dX+2dZ named "5th level monk attacking with unarmed strikes using power attack, and flurry of blows"
output 2dX+1dZ named "5th level monk attacking with unarmed strikes using power attack, and unarmed strikes as bonus action"

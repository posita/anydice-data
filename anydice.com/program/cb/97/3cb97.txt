\ Daggerheart pair multiply \
function: dhpm N:n M:n {
 if N = M {
  result: N * 3
 }
 result: N + M
}

\ Daggerheart pair extra \
function: dhpe N:n M:n O:n {
 if N = M {
  result: N * 2 + O
 }
 result: N + M
}

function: threshold damage N:n B:n T:s {
 if N < 1@T {result: -1}
 if N >= 3@T {
  result: 2 + B + (N-3@T)/5
 }
 if N >= 2@T {
  result: 1 + B
 }
 result: B
}

function: multihit threshold damage N:d B:n DT:s T:n FO:n {
 R: 0
 loop X over {0..T-1}{
  R: R+[threshold damage [dhpm N N]-(X*FO) B DT]
 }
 result: R
}

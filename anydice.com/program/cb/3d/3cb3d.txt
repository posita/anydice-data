function: phaseone ATT:s DEF:s {                    \Find the environment modifiers\
  VELOCITY_MODS: [compare 8@DEF with 7@ATT]         \compare the velocity for the dodge capacity\

  ATT_DICES: (1@ATT +2@ATT)                              \final number of hit-dices to roll\
  DEF_DICES: (4@DEF +5@DEF +VELOCITY_MODS)                \final number of save-dices to roll\

  result: [phasetwo ATT_DICESdQ DEF_DICESdQ ATT DEF]  \freeze the hits-dices & saves-dices sequences\
}

function: phasetwo ATT_ROLL:n DEF_ROLL:n  ATT:s DEF:s {       \Analysis of the rolls\
  SETS: [find damage bonus ATT_ROLL DEF_ROLL]                 \find the damage bonus from the sub-sets\

  HITS:  ATT_ROLL +0                                       \apply post-roll modifiers for Attack\
  SAVES: DEF_ROLL +0                                       \apply post-roll modifiers for Save\

  DAMAGES: HITS - SAVES
  result: [crop DAMAGES 0 4]
}

\***** Echo Knight Fighter Nova Comparison *****\

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {result: 0}
    if ROLL >= CRIT_RANGE {result: DMG + CRIT_DMG}
    if ROLL + BONUS >= AC {result: DMG}
    result: 0
}

ROLL: [highest 1 of 1d20]
AC: d{16..20}
LEVEL: 15
ATTACKS: 3
ATTACKMOD: (20-10)/2
ECHOTIMES: (18-10)/2
ACCBONUS: 1
DMGBONUS: 1

if LEVEL < 5 {PROFICIENCY: 2}
else if LEVEL > 5 & LEVEL < 9 {PROFICIENCY: 3}
else if LEVEL > 10 & LEVEL < 13 {PROFICIENCY: 4}
else if LEVEL > 13 & LEVEL < 17 {PROFICIENCY: 5}
else if LEVEL > 17 {PROFICIENCY: 6}



output 0 named "Maximum Potential"

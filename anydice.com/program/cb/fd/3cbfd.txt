 result: d6 >= DIFFICULTY
}

function: numberofsuccesses NUMBER_OF_DICES DIFFICULTY {
  X: 0
  loop N over {1..NUMBER_OF_DICES}{
    X: X + [success DIFFICULTY]
  }

 result: X
}


loop NUMBER_OF_DICES over {1..6} {
  loop DIFFICULTY over {2..6} {
    output [numberofsuccesses NUMBER_OF_DICES DIFFICULTY] named "[NUMBER_OF_DICES] Dices with a success on [DIFFICULTY] or more"
  }
}

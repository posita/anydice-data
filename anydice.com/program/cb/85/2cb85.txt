PROF : 3
MOD : 4
RAGE: 2
AC: 15

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

X : [attack [highest 1 of 2d20] plus MOD vs AC for 1d6 + MOD + RAGE + 2*PROF crit 1d6 on 20]
Y : [attack [highest 1 of 2d20] plus PROF+MOD vs AC for 1d6 + RAGE  crit 1d6 on 20]

Z : [attack [highest 1 of 2d20] plus MOD vs AC for 2d6 + MOD + RAGE + 2*PROF crit 2d6 on 20]
ZZ: [attack [highest 1 of 2d20] plus MOD vs AC for 1d12 + MOD + RAGE + 2*PROF crit 1d12 on 20]

output 2dX+Y named "5th level Barbarian with 2 Scimitars using power attack, reckless attack and rage"
output 2dZ named   "5th level Barbarian with greatsword using power attack, reckless attack and rage"
output 2dZZ named  "5th level Barbarian with maul using power attack, reckless attack and rage"

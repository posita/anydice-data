PROF : 5
MOD : 5
RAGE: 2
MAGIC: 1
DUELING: 2
AC: 18

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

PTW : [attack [highest 1 of 2d20] plus MOD+MAGIC vs AC for 1d6 + MOD + RAGE + 2*PROF+MAGIC crit 1d6 on 20]
TW : [attack [highest 1 of 2d20] plus MOD+PROF+MAGIC vs AC for 1d6+ MOD + RAGE + MAGIC crit 1d6 on 20]
GS : [attack [highest 1 of 2d20] plus MOD+PROF-5+MAGIC vs AC for 2d6 + RAGE + MAGIC + 10 + MOD crit 2d6 on 20]
SB : [attack [highest 1 of 2d20] plus MOD+PROF+MAGIC vs AC for 1d8 + RAGE + MAGIC + MOD+DUELING crit 1d8 on 20]
PSB : [attack [highest 1 of 2d20] plus MOD+MAGIC vs AC for 1d8 + RAGE + MAGIC + MOD+DUELING+2*PROF crit 1d8 on 20]

output 3dTW named "13th level Rawuni with 2 Scimitars using reckless attack and rage"
output 3dPTW named "13th level Rawuni with 2 Scimitars using reckless attack and rage and power attack"
output 2dGS named "13th level Rawuni with Greatsword using reckless attack and rage and Great Weapon Master"
output 2dSB named "13th level Rawuni with Battleaxe and Shield (AC 19) using reckless attack and rage (dueling FS)"
output 2dPSB named "13th level Rawuni with Battleaxe and Shield (AC 19) using reckless attack and rage (dueling FS) and power attack"

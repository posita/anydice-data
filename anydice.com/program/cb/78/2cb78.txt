PROF : 5
MOD : 5
MAGIC: 1
AC: 18
BLESS: 1d4

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n and SMITE:d {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG + 2dSMITE
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

SPEAR : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC+BLESS vs AC for 1d6+1d8 + MOD + MAGIC + 2 crit 1d6+1d8 on 20 and 2d8]
BUD   : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC+BLESS vs AC for 1d4+1d8 + MOD + MAGIC + 2 crit 1d6+1d8 on 20 and 2d8]

PA_SPEAR : [attack [highest 1 of 1d20] plus MOD+MAGIC+BLESS vs AC for 1d6+1d8 + MOD + MAGIC + 2+2*PROF crit 1d6+1d8 on 20 and 2d8]
PA_BUD   : [attack [highest 1 of 1d20] plus MOD+MAGIC+BLESS vs AC for 1d6+1d8 + MOD + MAGIC + 2+2*PROF crit 1d6+1d8 on 20 and 2d8] 
 
EB :[attack [highest 1 of 1d20] plus MOD+PROF+BLESS vs AC for 1d10+1d6 crit 1d10+1d6 on 20]



output 2dSPEAR+1dBUD        named "13th level Relius with +1 Spear PAM"
output 2dPA_SPEAR+1dPA_BUD  named "13th level Relius with +1 Spear PAM and power attack"
output 3dEB                 named "11th level Relius with Eldritch Blast + HEX"

PROF : 4
MOD : 4
WIS_MOD: 3
MAGIC: 0
AC: 17
ARCHERY: 2

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n bonusdmg BDMG:d and ATTACKS:n {
    if ATTACKS = 0 {
        result: 0
    }
    if ROLL = 1 {
        result: [attack 1d20 plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE bonusdmg BDMG and ATTACKS-1]
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG + 2dBDMG + [attack 1d20 plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE bonusdmg 0 and ATTACKS-1]
    }
    if ROLL + BONUS >= AC {
        result: DMG + BDMG+ [attack 1d20 plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE bonusdmg 0 and ATTACKS-1]
    }
    result: [attack 1d20 plus BONUS vs AC for DMG crit CRIT_DMG on CRIT_RANGE bonusdmg BDMG and ATTACKS-1]
}

LBSSH : [attack 1d20 plus MOD+PROF+MAGIC-5+ARCHERY vs AC for 1d8+1d6 + MOD + MAGIC+10 crit 1d8+1d6 on 20 bonusdmg 1d4 and 2]

LBH : [attack 1d20 plus MOD+PROF+MAGIC+ARCHERY vs AC for 1d8+1d6 + MOD + MAGIC crit 1d8+1d6 on 20 bonusdmg 1d4 and 2]

FEY : [attack 1d20 plus WIS_MOD+PROF vs AC for 2d6+6 crit 2d6 on 20]

LBSS : [attack 1d20 plus MOD+PROF+MAGIC-5+ARCHERY vs AC for 1d8 + MOD + MAGIC+10 crit 1d8 on 20 bonusdmg 1d4 and 2]

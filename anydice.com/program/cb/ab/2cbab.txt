A: 18 \Target AC\
B: 12 \Attack Bonus\
M: 12 \Spell Attack Bonus\
D: 17 \Save DC\
S: 2  \Save Bonus\

function: B vs A {
result: 1d20+B>A-1
}

function: B vs A one {
result: [B vs A]+[B vs A]
}

function: S saves D {
result: 1d20+S<D
}

function: A:s sneak {
if A=0 {result: 0}
if A=1 {result: 1d6+7+2d6}
if A=2 {result: 2d6+14+2d6}
}

output [B vs A]*(1d6+7)+[B vs A]*(1d6+7) named "Black Blade"
output [[B vs A one] sneak] named "Black Blade (SA)"
output [B vs A]*(1d6+7+2d8) named "Black Blade (Flame Blade)"
output [B vs A]*(1d6+7+2d8+2d6) named "Black Blade (Flame Blade + SA)"
output [S saves D]*8d6 named "Synaptic Static"
output 7d4+7 named "Magic Missile"
output [M vs A]*(1d10+4)+[M vs A]*(1d10+4)+[M vs A]*(1d10+4) named "Eldritch Blast"

function: high N:n of A:s B:s C:s {
    result: {1..N} @ [sort {A, B, C}]
}

function: low N:n of A:s B:s C:s {
    S: [sort {A, B, C}]
    result: {#S-N+1 .. #S} @ S
}

output [high 2 of 1d4 1d4 1d4] named "high 3d4"
output [high 2 of 1d4 1d4 1d6] named "high 2d4+d6"
output [high 2 of 1d8 1d8 1d8] named "high 3d8"
output [high 2 of 1d4 1d4 1d8] named "high 2d4+d8"
output [low 2 of 1d4 1d4 1d4] named "low 3d4"
output [low 2 of 1d4 1d4 1d6] named "low 2d4 + d6"
output [low 2 of 1d4 1d4 1d8] named "low 2d4 + d8"

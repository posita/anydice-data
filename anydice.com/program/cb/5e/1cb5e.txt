function: evaluate ROLL:s at SKILL:n {
  EIGHTS: (ROLL = 8)
  SUCCESSES: (ROLL <= SKILL) - (ROLL = 7)
  if SUCCESSES < 0 { SUCCESSES: 0 }
  result: (SUCCESSES + EIGHTS) > 0
}

loop ATTR over {1..5} {
  loop SKILL over {0..5} {
    output [evaluate ATTR d8 at SKILL] named "ATTR: [ATTR] @ SKILL: [SKILL]"
  }
}

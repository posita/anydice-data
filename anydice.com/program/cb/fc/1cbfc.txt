\ HIT POINTS CALCULATIONS \
function: average of HITDICE:d plusone {
 result: (#HITDICE + [maximum of HITDICE])/2 + 1
}

function: fixedhp LEVEL:n hitdice HITDICE:d conmod CON:n {
 result: ([maximum of HITDICE] + CON) + (([average of HITDICE plusone] + CON) * (LEVEL - 1))
}

function: rollhp LEVEL:n hitdice HITDICE:d conmod CON:n {
 HP: 0
 loop L over {2..LEVEL} {
   HP: HP + (HITDICE + CON)
 }
 result: HP + ([maximum of HITDICE] + CON)
}

\ HIT POINTS CALCULATIONS \

output [rollhp 4 hitdice d12 conmod 0]
output [fixedhp 4 hitdice d12 conmod 0]

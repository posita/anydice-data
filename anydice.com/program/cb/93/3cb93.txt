\ Daggerheart pair multiply \
function: dhpm N:n M:n {
 if N = M {
  result: N * 3
 }
 result: N + M
}

\ Daggerheart pair extra \
function: dhpe N:n M:n O:n {
 if N = M {
  result: N * 2 + O
 }
 result: N + M
}

function: light weapon N:n {
 if N >=5 {
  result: N-5
 }
 result: N
}

function: heavy weapon N:n {
 if N >=5 {
  result: N+5
 }
 result: N
}

function: damage N:n {
 if N < 0 {
  result: 0
 }
 result: 1+N/5
}

function: damage N:n B:n {
 if N < 0 {
  result: 0
 }
 result: B+N/5
}

function: threshold damage N:n B:n T:s {
 if N < 1@T {result: 0}
 if N > 3@T {
  result: 2 + B + (N-3@T)/5
 }
 if N > 2@T {
  result: 1 + B
 }
 result: B
}

function: M:n damage N:n {
 result: 1+N/5+M/5
}

function: dodge N:n {
 if N >= 10 {
  result: N
 }
 result: 0
}

function: brace N:n {
 result: N-5
}

function: multihit threshold damage N:d B:n DT:s T:n FO:n {
 R: 0
 loop X over {0..T-1}{
  R: R+[threshold damage [dhpm N N]-(X*FO) B DT]
 }
 result: R
}

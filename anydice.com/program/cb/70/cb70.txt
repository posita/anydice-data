function: attack ROLL:n BAB:n mod ATK:n damage DMG:n die DIE:n crit CRIT:n times MULT:n fail FAIL:n ac AC:n {
  if ROLL >= CRIT {
    result: (MULT)d(DIE) + MULT * DMG
  }
  if ROLL <= FAIL {
    result: 0
  }
  if ROLL + BAB + ATK > AC {
    result: 1d(DIE) + DMG
  }
  result: 0
}

function: manyshot ROLL:n BAB:n mod ATK:n damage DMG:n die DIE:n crit CRIT:n times MULT:n fail FAIL:n ac AC:n {
  if ROLL >= CRIT {
    result: (MULT)d(DIE) + MULT * DMG + 1d(DIE) + DMG
  }
  if ROLL <= FAIL {
    result: 0
  }
  if ROLL + BAB + ATK > AC {
    result: 1d(DIE) + DMG + 1d(DIE) + DMG
  }
  result: 0
}

function: full BAB:n mod ATK:n damage DMG:n die DIE:n crit CRIT:n times MULT:n fail FAIL:n ac AC:n {
  if BAB > 0 {
    result: [attack 1d20 BAB mod ATK damage DMG die DIE crit CRIT times MULT fail FAIL ac AC] + [full BAB-5 mod ATK damage DMG die DIE crit CRIT times MULT fail FAIL ac AC]
  }
  result: 0
}

function: rapid BAB:n mod ATK:n damage DMG:n die DIE:n crit CRIT:n times MULT:n fail FAIL:n ac AC:n {
  result: [attack 1d20 BAB mod ATK-2 damage DMG die DIE crit CRIT times MULT fail FAIL ac AC] + [full BAB mod ATK-2 damage DMG die DIE crit CRIT times MULT fail FAIL ac AC]
}
function: manyrapid BAB:n mod ATK:n damage DMG:n die DIE:n crit CRIT:n times MULT:n fail FAIL:n ac AC:n {
  result: [manyshot 1d20 BAB mod ATK-2 damage DMG die DIE crit CRIT times MULT fail FAIL ac AC] + [full BAB mod ATK-2 damage DMG die DIE crit CRIT times MULT fail FAIL ac AC]
}

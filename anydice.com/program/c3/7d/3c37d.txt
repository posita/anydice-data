function: b S:s {
  if [S contains 1] & [maximum of S] <= 6 {
    result: 1
  } else {
    result: 0
  }
}

function: s S:s {
  result: [count {7,8,9} in S] + [count 10 in S] * 2
}

function: r S:s {
  if [maximum of S] <= 6 & [S contains 1] {
    result: -1
  } else {
    result: [count {7,8,9} in S] + [count 10 in S] * 2
  }
}

loop X over {1,2,3,4,5,6} {
  output [b Xd10] named "[X]d10 - botch"
  output [s Xd10] named "[X]d10 - success"
  output [r Xd10] named "[X]d10 - overall"
}

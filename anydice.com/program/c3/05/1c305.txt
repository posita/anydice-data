PL_DMG: 1d4
PL_SNEAK: 4d6
PL_DEX: 4
PL_PRO: 3
EN_AC: 15

function: right RIGHT:n left LEFT:n {
	if RIGHT + PL_DEX + PL_PRO >= EN_AC & RIGHT = 20 {
		if LEFT + PL_DEX + PL_PRO >= EN_AC & LEFT = 20 {
			result: (PL_SNEAK + PL_DMG + PL_DEX)*2 + (PL_DMG + PL_DEX)*2
		} else if LEFT + PL_DEX + PL_PRO >= EN_AC {
			result: (PL_SNEAK + PL_DMG + PL_DEX)*2 + (PL_DMG + PL_DEX)
		} else if LEFT + PL_DEX + PL_PRO < EN_AC {
			result: (PL_SNEAK + PL_DMG + PL_DEX)*2
		}
	} else if RIGHT + PL_DEX + PL_PRO >= EN_AC {
		if LEFT + PL_DEX + PL_PRO >= EN_AC & LEFT = 20 {
			result: (PL_SNEAK + PL_DMG + PL_DEX) + (PL_DMG + PL_DEX)*2
		} else if LEFT + PL_DEX + PL_PRO >= EN_AC {
			result: (PL_SNEAK + PL_DMG + PL_DEX) + (PL_DMG + PL_DEX)
		} else if LEFT + PL_DEX + PL_PRO < EN_AC {
			result: (PL_SNEAK + PL_DMG + PL_DEX)
		}
	} else if RIGHT + PL_DEX + PL_PRO < EN_AC {
		if LEFT + PL_DEX + PL_PRO >= EN_AC & LEFT = 20 {
			result: (PL_SNEAK + PL_DMG + PL_DEX)*2
		} else if LEFT + PL_DEX + PL_PRO >= EN_AC {
			result: (PL_SNEAK + PL_DMG + PL_DEX)
		} else if LEFT + PL_DEX + PL_PRO < EN_AC {
			result: 0
		}
	}
}

output[right d20 left d20] named "D&D 5E: Rogue Sneak Attack + Dual Weapon Fighting on a Single Target"

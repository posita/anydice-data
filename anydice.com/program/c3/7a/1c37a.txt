function: N:n df DIE:n {
 result: Nd{{3..DIE}:(1+DIE/2),1,2}
}

function: ragnar HIT:n HITT:n with GWM:n {
 X: 10+[2df10]+[4df6]+2*GWM
 if HIT >=19 {
  X: X+[1df10]
 }
 if HITT >=19 {
  X: X+[1df10]
 }
 if (HIT>=19) | (HITT>=19) {
  X: X+5+[1df10]+[2df6]+GWM
 }
 result: X
}

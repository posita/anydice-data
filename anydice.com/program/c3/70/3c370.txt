function: armorexplode N:n on ARMOR:n {
 if N > ARMOR { result: N + 1d6 }
 result: N
}

function: armorimplode N:n on ARMOR:n {
 if N <= ARMOR { result: N + 1d6 }
 result: N
}

function: critchance N:n on CRITMIN:n{
 if N >= CRITMIN {
 result: 1
 }
 result: 0
}

function: avgdamagecrit N:n on CRITDICE:n CRITMIN:n {
 if ([critchance CRITDICE on CRITMIN] >= 1) { result: N + 5 }
 result: N
}

function: avgdamagecritpower N:n on CRITDICE:n CRITMIN:n{
 if ([critchance CRITDICE on CRITMIN] >= 1) { result: N + 10 }
 result: N
}

function: headshot DAM:n MULT:n {
 if MULT = 2 {result: DAM * 2}
else if MULT = 3{
result: DAM * 2
}
 result: 0
}

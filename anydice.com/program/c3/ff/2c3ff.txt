function: mod R:n {
  if (R = 3) { result: -3}
  if (R < 6) { result: -2}
  if (R < 9) { result: -1}
  if (R < 12) { result: 0}
  if (R < 15) { result: 1}
  if (R < 18) { result: 2}  
  result: 3
}

function: hopeless A:n B:n {
  if (A < 0) { result:B }
  result: A

}

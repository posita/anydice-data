MIN_DIE_COUNT: 5
MAX_DIE_COUNT: MIN_DIE_COUNT

ACTION_DICE: d{0, 0, 1, 1, 2, 10}
set "maximum function depth" to 3

function: explode N:n {
  if N > 2 {
   result: 1 + [explode ACTION_DICE]
  } else {
   result: N
  }
}

function: stamina N:n {
  
  TOTAL: [count {0} in N]
  if N > 2 {
   result: TOTAL + [stamina ACTION_DICE]
  } else {
   result: TOTAL
  }
}

loop N over {MIN_DIE_COUNT..MAX_DIE_COUNT} {
  output Nd[explode ACTION_DICE] named "Successes on [N]D"
  output Nd[stamina ACTION_DICE] named "Stamina on [N]D"
}

MINIMUM: 0
PENALTY: 0

loop STR over {-1..5} {
  loop DEX over {-1..5} {
    BONUS: [highest of (DEX - PENALTY) and MINIMUM]
    output BONUS named "STR:[STR] DEX:[DEX]"
    output 1d8+STR named "Longsword"
    output 1d8+BONUS+STR named "Quality Longsword"
    output 1d8+STR+3 named "Longsword +3"
    output 1d8+DEX named "Rapier"
    output 1d8+DEX+3 named "Rapier +3"
    output BONUS+STR > STR+3 & BONUS+STR > DEX+3 named "better than +3"
  }
}

ARCANA_MOD: EXPERTISE+INT_MOD
CRITFAIL_TOGGLE: 1

function: overflow A:n over B:n {
if A > B { result: B }
if B < 0 { result: 0 }
result: A}

function: cast A:n with B:n{
if A = 0 {result: B} 
if A <= B { result: 1 }
result: 0}

function: miz A:n B:n C:n {
if A = 1 & C = 1 { result: -1 }
result: [cast SPELL_LEVEL with [overflow ((A + B)/2) over 9]]
}

function: starmiz A:n B:n C:n{
if A = 1 & C = 1{ result: -1 }
if A < 10 { result: 10 }
result: [cast SPELL_LEVEL with [overflow ((A + B)/2) over 9]]
}

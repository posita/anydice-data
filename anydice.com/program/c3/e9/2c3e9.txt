N: 99
KILLS_PER_HOUR: 160
EXP_PER_KILL: 10
EXP_PER_FEATHER: 50
EXP_PER_RARE: 500
loop X over {1..N} {
RARE_SPAWNS: (KILLS_PER_HOUR)/(40-X/5)
UNRARE_SPAWNS: KILLS_PER_HOUR - RARE_SPAWNS
EXTRA_EXP_PER_FEATHER: 5 + X
A: X
B: 100 + A*2 
DIVISOR: 120
FROLL: (10 * UNRARE_SPAWNS/KILLS_PER_HOUR + 500 * RARE_SPAWNS/KILLS_PER_HOUR + A * 2 + 1dB)/DIVISOR
FHOUR: KILLS_PER_HOUR * FROLL
EHOUR: UNRARE_SPAWNS * EXP_PER_KILL + RARE_SPAWNS * EXP_PER_RARE + FHOUR * EXP_PER_FEATHER + EXTRA_EXP_PER_FEATHER * FHOUR

output EHOUR named "exp/h lv[A] hunter"
output FHOUR named "feathers/h lv[A] hunter"
}

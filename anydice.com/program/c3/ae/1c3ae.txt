
DICE:{4,6,8,10,12}
loop D over {1..(#DICE-1)}{
  loop SECOND over {0..5}{
    FIRST: 5-SECOND
    FIRSTD: D@DICE
    SECONDD: (D+1)@DICE
    output [count {5..12} in FIRSTdFIRSTD] + [count {5..12} in SECONDdSECONDD] named "Success [FIRST]d[FIRSTD] and [SECOND]d[SECONDD]"
    output [count 1 in FIRSTdFIRSTD] + [count 1 in SECONDdSECONDD] named "Complication [FIRST]d[FIRSTD] and [SECOND]d[SECONDD]"
  }
}

function: blades DICE:s {
  PART: 1@DICE >= 4
  FULL: 1@DICE = 6
  CRIT: 2@DICE = 6
  result: PART + FULL + CRIT
}

function: evens DICE:s {
  ACC: 0
  loop I over {1..#DICE} {
    EVEN: (I@DICE / 2) = ((I@DICE + 1) / 2)
    ACC: EVEN + ACC
  }
  result: ACC
}

function: even blades DICE:s {
  PART: 1@DICE >= 4
  FULL: 1@DICE = 6
  CRIT: 2@DICE = 6

  SUCCESS: PART + FULL + CRIT
  EVENS: [evens DICE]

  result: SUCCESS * 10 + EVENS
}

output [even blades 2d6]

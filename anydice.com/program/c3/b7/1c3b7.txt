function: D:s boon N:n {
  if 2@D <= N {
    result: 1@D + 1d10
  }
  else {
    result: D
  }
}

function: D:s double boon N:n {
  R: 0
  loop M over D {
    if M <= N {
      R: R + 1d10
    }
    else {
      R: R + M
    }
  }
  result: R
}

function: D:s bane N:n {
  if 1@D >= N {
    result: 2@D + 1d10
  }
  else {
    result: D
  }
}

function: D:s double bane N:n {
  R: 0
  loop M over D {
    if M >= N {
      R: R + 1d10
    }
    else {
      R: R + M
    }
  }
  result: R
}

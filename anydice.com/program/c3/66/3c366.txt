\Math\
function: attack M:s {
 ACCURACY: 4@M*(OACM-EEVM)@{280:7,315,360,420,504,630,840,1120,1400,1680,1960,2240,2520:7}/840
 result:(3@M\Power\ *1@M\Atk\ *42 /(2@M\Def\ *50)+2) *\RNG\1d{85..100}/100 *5@M\TYPE\/16 *1d{1:ACCURACY,0:100-ACCURACY}
}
function: minimum M:s{
 result:(\Power\3@M *\Atk\1@M *42 /(\Def\2@M *50)+2) *\RNG\85/100 *\TYPE\5@M/16
}
function: maximum M:s{
 result:(\Power\3@M *\Atk\1@M *42 /(\Def\2@M *50)+2) *\RNG\100/100 *\TYPE\5@M/16
}

function: round N move{
 if N = 1 {result:OMA}
 if N = 2 {result:OMB}
 if N = 3 {result:OMC}
 if N = 4 {result:OMD}
 if N = 5 {result:OME}
 if N = 6 {result:OMF}
 if N = 7 {result:OMG}
 if N = 8 {result:OMH}
}
function: round N counter{
 if N = 1 {result:EMA}
 if N = 2 {result:EMB}
 if N = 3 {result:EMC}
 if N = 4 {result:EMD}
 if N = 5 {result:EME}
 if N = 6 {result:EMF}
 if N = 7 {result:EMG}
 if N = 8 {result:EMH}
}
 EDAMAGE: {}
 ODAMAGE: {}
 STACK: {}
loop N over {1..[lowest of (EHP/[minimum OMA]) and (OHP/[minimum EMA])]+1}{
 OUR_MOVE: [round N move]
 ENEMY_MOVE: [round N counter]
 if ESP <= OSP {
  ODAMAGE: ODAMAGE + [attack ENEMY_MOVE]*(EDAMAGE < EHP)
  EDAMAGE: EDAMAGE + [attack OUR_MOVE]*(ODAMAGE < OHP)
  TURN: (EDAMAGE >= EHP)*(ODAMAGE < OHP)
 }
 if ESP > OSP {
  EDAMAGE: EDAMAGE + [attack OUR_MOVE]*(ODAMAGE < OHP)
  TURN: (EDAMAGE >= EHP)*(ODAMAGE < OHP)
  ODAMAGE: ODAMAGE + [attack ENEMY_MOVE]*(EDAMAGE < EHP)
 }
 STACK: STACK + TURN
 output 0 < STACK named "Win on Turn [N]"

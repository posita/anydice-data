function: FDICE:s vs EDICE:s with fight FFV:n vs EFV:n{
  FROLL: {1..1}@FDICE
  EROLL: {1..1}@EDICE
  WIN: 1
  if FROLL > EROLL {WIN: 1}
  if FROLL = EROLL {
    if FFV > EFV {WIN: 1} 
    else {if FFV < EFV {WIN: -1}
       else {WIN: d{-1, 1} * 1}}}
  if FROLL < EROLL {WIN: -1}

  result: WIN
}

function: wound on WOUNDROLL:n with ATTACKS:s attacks{
  WOUNDS: [count {WOUNDROLL..6} in ATTACKS]
  
  result: WOUNDS
}

function: get wound on WOUNDROLL:n with ATTACKS:s attacks{
  WOUNDS: [count {WOUNDROLL..6} in ATTACKS]
  
  result: WOUNDS
}

output [6d6 vs 5d6 with fight 6 vs 3] named "win fight"
output [wound on 5 with 8d6 attacks] named "wounds"
output [get wound on 6 with 4d6 attacks] named "get wound"

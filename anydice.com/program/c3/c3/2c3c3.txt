function: saveforhalf DAMAGE:n dc DC:n skill S:n {
 if S >= DC {result: DAMAGE/2}
if S < DC {result: DAMAGE}
}

function: attackroll TOHIT:n damage DAMAGE:n ac AC:n {
 if TOHIT >= AC { 
if TOHIT=20{ result: DAMAGE}
if TOHIT <20 {result: 2*DAMAGE}}
if TOHIT < AC {result: 0}
}

function: multiattack TOHIT:n damage DAMAGE:n ac AC:n attacks NUM:n{
if NUM=1{result: [attackroll TOHIT damage DAMAGE ac AC] }
if NUM=2{result: [attackroll TOHIT damage DAMAGE ac AC] + [attackroll TOHIT damage DAMAGE ac AC] }
if NUM=3{result: [attackroll TOHIT damage DAMAGE ac AC] + [attackroll TOHIT damage DAMAGE ac AC] + [attackroll TOHIT damage DAMAGE ac AC]}
if NUM=4{result: [attackroll TOHIT damage DAMAGE ac AC] + [attackroll TOHIT damage DAMAGE ac AC] + [attackroll TOHIT damage DAMAGE ac AC]+ [attackroll TOHIT damage DAMAGE ac AC]}
}

output [saveforhalf 9d8 dc 16 skill 1d20+5] named "Blight"

output [saveforhalf 8d6 dc 16 skill 1d20+2] named "Synaptic Static"

function: concentration D:n competence C:n chance L:n seuil S:n{
POOL:(D+L)d6
SUCCES:[count {S..6} in POOL]
DOUBLE:[count {6} in POOL]
ECHECS:[count {1} in POOL]
COMP:[lowest of ECHECS and C]
MISS:[count {S-1,S+1} in POOL]
COMPE:[lowest of C-COMP and MISS]
MISSED:[count {S-2} in POOL]
COMPL:[lowest of (C-COMP-COMPE)/2 and MISSED]

result:[lowest of SUCCES and D]+DOUBLE-ECHECS+COMP+COMPE+COMPL}

output [concentration 10 competence 0 chance 0 seuil 4]
output [concentration 10 competence 1 chance 0 seuil 4]
output [concentration 10 competence 2 chance 0 seuil 4]

output [concentration 10 competence 3 chance 0 seuil 4]
output [concentration 10 competence 4 chance 0 seuil 4]
output [concentration 10 competence 5 chance 0 seuil 4]
output [concentration 10 competence 6 chance 0 seuil 4]

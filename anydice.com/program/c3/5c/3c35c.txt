function: crit RA:n and RB:n {
 CRITX: 0
 CRITY: 0
 if RA = RB { CRITX: 6 }
 if RA = 12 { CRITY: CRITY + 6 }
 if RB = 12 { CRITY: CRITY + 6 }
 if RA = 1 { CRITY: CRITY - 6 }
 if RB = 1 { CRITY: CRITY - 6 }
 if CRITY < -6 { CRITY: -6 }
 if CRITY > 6 { CRITY: 6 }
 result: RA + RB + CRITX + CRITY
}

output [crit d12 and d12]

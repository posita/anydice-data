function: round N move{
 if N = 1 {result:OMA}
 if N = 2 {result:OMB}
 if N = 3 {result:OMC}
 if N = 4 {result:OMD}
 if N = 5 {result:OME}
 if N = 6 {result:OMF}
 if N = 7 {result:OMG}
 if N = 8 {result:OMH}
}
function: round N counter{
 if N = 1 {result:EMA}
 if N = 2 {result:EMB}
 if N = 3 {result:EMC}
 if N = 4 {result:EMD}
 if N = 5 {result:EME}
 if N = 6 {result:EMF}
 if N = 7 {result:EMG}
 if N = 8 {result:EMH}
}

  TURN: (EDAMAGE >= EHP)*(ODAMAGE < OHP)
  ODAMAGE: ODAMAGE + [enemy ENEMY_MOVE attack]

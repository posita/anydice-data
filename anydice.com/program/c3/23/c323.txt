function: ties in A:s vs B:s {
  LEN: [lowest of #A and #B]
  COUNT: 0
  loop N over {1..LEN} {
    if (N@A = N@B) {
      COUNT: COUNT+1
    }
  }
  result: COUNT
}

output [ties in {1, 2, 3} vs {3, 2, 7}] named "1 match"
output [ties in {1, 2, 3} vs {1, 2, 3}] named "3 matches"
output [ties in [sort 3d6] vs [sort 3d6]] named "two sorted collections of 3d6"

function: partial explode DICE:d depth DEPTH:n{
 OLD: d{0}
 ADD: d{1}
 TEMP: d{0}
 PERM: d{1}
 
 loop N over {1..DEPTH}{
  loop M over {1..N}{
   OLD: ADD
   ADD: OLD * DICE
  }
  TEMP: PERM
  PERM: TEMP + ADD
 }

result: PERM
}

function: transpose S:d {
 result: S
}
D:20
loop R over{1..D}{
 output [partial explode d{0,{1}:R} depth 9]
}

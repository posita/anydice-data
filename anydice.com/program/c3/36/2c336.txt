function: success N:n
{
 if (N < 7) {result:-1}
 if (N < 10) {result:0}
 if (N < 11) {result:1}
 if (N < 12) {result:2}
 if (N < 13) {result:3}
 result:-2
}

loop MOD over {-3,-2,-1,0}
{
 R:2d6 + MOD 
 output [success R] named "Modifier [MOD]"
}

loop MOD over {1,2,3}
{
 R:2d6 + MOD 
 output [success R] named "Modifier +[MOD]"
}

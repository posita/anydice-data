\
 Ironsworn 1d8+stat vs 2d12

 -2: Critical Miss
 -1: Miss.
 0: Success, but with undesired consequences.
 +1: Clean success.
 +2: Critical success!
\

function: roll A:s and B:s vs S:s {
 R: -1
 if A < S {
   R: R+1
 }
 if B < S {
   R: R+1
 }
 if A = S {
   if B <= S {
     R: 2
   }
 }
 if B = S {
   if A < S {
     R: 2
   }
 }
 result: R
}

loop P over {3..10} {
 output [roll 1d12 and 1d12 vs P] named "[P]"
}

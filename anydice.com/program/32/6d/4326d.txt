function: difficulty N:n in POOL:d {
  result: [count {N..10} in POOL] - [count 1 in POOL]
}

function: icosa DIFF:n in ROLL:d {
  result: (ROLL >= DIFF & ROLL < 20) + 2*(ROLL = 20) - (ROLL = 1)
}

function: one roll POOL:n {
  ROLL: POOLd10
  WIDTH: [count 5 in ROLL]
  if #{WIDTH} >= 2 {result: WIDTH}
  result: 0
}


loop N over {14..17} {
  loop M over {1..4} {
    output [icosa N in Md20] named "[M]d20 vs [N]"
  }
}

function: A:n max B:n vezes C:n{
 if A = X {result: A}
 MAX:B
 if A > B {MAX: A}
 if C = 0 {result: MAX}
 result: [SEQ max MAX vezes C-1]
}
function: A:n max B:n vezes C:n com falha{
 if A = X | A = 0 {result: A}
 MAX:B
 if A > B {MAX: A}
 if C = 0 {result: MAX}
 result: [SEQ max MAX vezes C-1]
}

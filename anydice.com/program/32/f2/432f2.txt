\ Set your dice pools here \
X: 4
Y: 4

function: pairs between A:s and B:s {
  SUCCESSES: 0
  
  \ Check each possible die face from 1 to 6 \
  loop FACE over {1d6} {
    COUNT_A: [count FACE in A]
    COUNT_B: [count FACE in B]
    
if COUNT_A = 0 | COUNT_B = 0 {} 
else {
    \ Add the lowest count of matches for this face to the total \
      if COUNT_A < COUNT_B {
        SUCCESSES: SUCCESSES + COUNT_A
      } else {
        SUCCESSES: SUCCESSES + COUNT_B
     }
    }
  }
  
  result: SUCCESSES
}

\ Output the result \
output [pairs between Xd6 and Yd6] named "Pairs between [X]d6 and [Y]d6"

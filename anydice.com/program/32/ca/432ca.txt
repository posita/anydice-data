\ Time to kill calculation \
function: ttk HP:n with DMG:d {
  if HP<=0 { result: 0 }
  result: 1+[ttk HP-DMG with DMG]
}

\ Rolling dice to determine degree of success: \
function: auxroll D:d rolled N:n vs TN:n {
  RES: 0
  if N>=(TN+10) { RES: 2}
  if N>=TN { RES: 1 }
  if N<=(TN-10) { RES: -1 }
  if N=[maximum of D] { RES: [lowest of RES+1 and 2] }
  if N=(1@{D}) { RES: [highest of RES-1 and -1] }
  result: RES
}
function: roll D:d vs TN:n {
  result: [auxroll D rolled D vs TN]
}
FORTUNE: 1@2d20
MISSFORTUNE: 2@2d20

\ Attacking for damage: \
function: attack SUCCESS:n for DMG:n on crit CRIT:n on miss MISS:n {
  if SUCCESS=2 { result: 2*DMG+CRIT }
  if SUCCESS=1 { result: DMG }
  if SUCCESS=0 { result: MISS }
  result: 0
}
function: attack SUCCESS:n for DMG:n on crit CRIT:n {
  result: [attack SUCCESS for DMG on crit CRIT on miss 0]
}
function: attack SUCCESS:n for DMG:n on miss MISS:n {
  result: [attack SUCCESS for DMG on crit 0 on miss MISS]
}
function: attack SUCCESS:n for DMG:n {
  result: [attack SUCCESS for DMG on crit 0 on miss 0]
}

\ Basic save against damaging effect \
function: save SUCCESS:n for DMG:n on crit CRIT:n {
  if SUCCESS=-1 { result: DMG*2+CRIT }
  if SUCCESS=0 { result: DMG }
  if SUCCESS=1 { result: DMG/2 }
  result: 0
}
function: save SUCCESS:n for DMG:n {
  result: [save SUCCESS for DMG on crit 0]
}

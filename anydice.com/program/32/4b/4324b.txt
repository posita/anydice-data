function: check SUCCESS:n FAILURE:n MOD:n {
  DIFF: (SUCCESS + MOD) - FAILURE
  if DIFF >= 7 { result: 3 }
  if DIFF >= 3 { result: 2 }
  if DIFF > 0 { result: 1 }
  if DIFF = 0 { result: 1 }
  if DIFF <= -7 { result: -3 }
  if DIFF <= -3 { result: -2 }
  result: -1
}

function: twist SUCCESS:n FAILURE:n {
  result: SUCCESS = FAILURE
}

function: total SUCCESS:n FAILURE:n MOD:n {
  BAND: [check SUCCESS FAILURE MOD]
  if BAND > 0 { result: 1 }
  result: -1
}

output [check 1d12 1d12 0] named "bands, modifier +0"
output [total 1d12 1d12 0] named "yes or no"
output [twist 1d12 1d12] named "twist rate"

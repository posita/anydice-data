function: typeconvert A:n and B:n {
 result: A & B
}

function: montroll N streak S {
 X: d4
 NS: X = 4
 NCONT: N > 0
 \Continue if streak is alive or more dice have to be rolled\
 if NS | NCONT {
  result: X + [montroll N - 1 streak NS & S]
 }
 else {
  result: X
 }
}

\Always set streak to 1 on initial roll\
output [montroll 3 streak 1]

MISS: 0
HIT: 1
CRIT: 2

ATKMOD: 12

function: attack ROLL:n vs AC:n {
 if ROLL = 1  { result: MISS }
 if ROLL = 20 { result: CRIT }
 if ROLL + ATKMOD >= AC { result: HIT }
 result: MISS
}

MISSDMG: 0
HITDMG:  1d8 + 6
CRITDMG: 2d8 + 6
function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRIT { result: CRITDMG }
 if ATTACKRESULT = HIT  { result: HITDMG }
 result: MISSDMG
}

AC: 22
output [damage for [attack d20 vs AC]] named "normal vs. AC [AC]"
output [damage for [attack [highest 1 of 3d20] vs AC]] named "Super Advantage vs. AC [AC]"
HITDMG:  1d8 + 1d6 + 6
CRITDMG: 2d8 + 2d6 + 6
output [damage for [attack [highest 1 of 2d20] vs AC]] named "Hunter's Mark vs. AC [AC]"

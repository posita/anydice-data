loop FIGHTER_LEVEL over {12..32}{
MONK_LEVEL: 50 - FIGHTER_LEVEL
UNARMED_ATTACKS: 2+(MONK_LEVEL-1)/3
WEAPON_ATTACKS: UNARMED_ATTACKS-1
KENSEI_BONUS: FIGHTER_LEVEL/3
POWER_ATTACK_BONUS: 5
FIGHTER_LEVEL_BONUS: FIGHTER_LEVEL / 5
WEAPON_SPECIALIZATION_BONUS: 0
if FIGHTER_LEVEL >= 4 { WEAPON_SPECIALIZATION_BONUS: 2 }
if FIGHTER_LEVEL >= 12 { WEAPON_SPECIALIZATION_BONUS: 4 }
STR_BONUS: 10
DAMAGE_REDUCTION: 30
TOTAL_BONUS: KENSEI_BONUS + POWER_ATTACK_BONUS + FIGHTER_LEVEL_BONUS + WEAPON_SPECIALIZATION_BONUS + STR_BONUS - DAMAGE_REDUCTION
if 0 {output WEAPON_ATTACKSd4 + WEAPON_ATTACKS*TOTAL_BONUS named "Fighter/Monk [FIGHTER_LEVEL]/[MONK_LEVEL]"}
}
output  130d4

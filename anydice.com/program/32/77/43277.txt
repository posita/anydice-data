function: panama ROLL:s {
    \ Straight \
    if [count 1 in ROLL] = 1
    & [count 2 in ROLL] = 1
    & [count 3 in ROLL] = 1
    & [count 4 in ROLL] = 1
    & [count 5 in ROLL] = 1
    & [count 6 in ROLL] = 1 {
        result: -10
    }

    SCORE: 0

    loop N over {1..6} {
        COUNT: [count N in ROLL]

        if COUNT = 3 {
            SCORE: SCORE + 0
        }
        if COUNT = 4 {
            SCORE: SCORE + N * 4
        }
        if COUNT = 5 {
            SCORE: SCORE + N * 5
        }
        if COUNT = 6 {
            SCORE: SCORE - N
        }
    }

    result: SCORE
}

output [panama 6d6] named "Panama Score"

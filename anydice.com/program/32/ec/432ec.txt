function: explode DIE:d {
 if [maximum of DIE] < 6 {
  DEC: [maximum of DIE] + 1
 } else {
  if [maximum of DIE] < 10 {
   DEC: [maximum of DIE]
  } else {
   if [maximum of DIE] < 20 {
    DEC: [maximum of DIE] - 1
   } else {
    DEC: [maximum of DIE] - 2
   }
  }
 }
 result: [explode DIE helper]
}

function: explode F:n helper {
 if F >= DEC { result: F + 1@DIE }
 result: F
}

loop D over {4,6,8,10,12,20}{
  loop N over {1..3}{
    output [explode [highest 1 of NdD]] named "d[D] adv [N] ex "
  }
}

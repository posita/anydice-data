function: check SUCCESS:n FAILURE:n MOD:n {
  if SUCCESS = FAILURE { result: 0 }
  DIFF: (SUCCESS + MOD) - FAILURE
  if DIFF = 0 {
    if MOD > 0 { result: 1 }
    result: -1
  }
  if DIFF >= 7 { result: 3 }
  if DIFF >= 3 { result: 2 }
  if DIFF >= 1 { result: 1 }
  if DIFF <= -7 { result: -3 }
  if DIFF <= -3 { result: -2 }
  result: -1
}

function: total SUCCESS:n FAILURE:n MOD:n {
  BAND: [check SUCCESS FAILURE MOD]
  if BAND = 0 { result: 0 }
  if BAND > 0 { result: 1 }
  result: -1
}

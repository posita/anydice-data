
loop POOL over {3,9} {
  loop SIDES over {6,8,10,12} {
    ONES: [ count 1 in POOLdSIDES ] 
    TOTE: 1@POOLdSIDES
    output [ lowest of 1d20 and TOTE + ONES ] named "[POOL]d[SIDES]"
  }
}

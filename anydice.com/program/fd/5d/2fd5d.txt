function: sign S:n {
\
   if S < 0 {result: -1}
   if S = 0 {result: 0}
\
   if S <= 0 {result: 0}
   if S > 0 {result: 1}
}

function: negative to zero S:n {
\
   if S < 0 {result: -1}
   if S = 0 {result: 0}
\
   if S <= 0 {result: 0}
   if S > 0 {result: S}
}



function: reroll if max D:d {
   result: [reroll D when [maximum of D]]
}

function: reroll D:n when T:n {
   if D >= T {result: [reroll 1dD when T] }
   else { result: D }
}


function: challenge P vs T {
   DIFF: [roll P] - [roll T]
\   result: [sign DIFF] \
   result: [negative to zero DIFF] 
\result: DIFF\
}

function: roll LEVEL {
  if LEVEL = 1 { result: d4/2-1 }
  if LEVEL = 2 { result: d4/2 }
  if LEVEL = 3 { result: [reroll if max d4] }

  if LEVEL = 4 { result: d4 }
  if LEVEL = 5 { result: [reroll if max d6]}

  if LEVEL = 6 { result: d6 }
  if LEVEL = 7 { result: [reroll if max d8]}

  if LEVEL = 8 { result: d8}
  if LEVEL = 9 { result: [reroll if max d10]}

  if LEVEL = 10 { result: d10}
  if LEVEL = 11 { result: [reroll if max d12]}

  if LEVEL = 12 { result: d12}
}

\loop P over { -6, 6, 7, -8, 8, 9, -10, 10 }\
loop T over { 4, 8 }
{
loop P over { 3..12 }
{
   output [ challenge P vs T] named "[P]vs[T]"
}
}

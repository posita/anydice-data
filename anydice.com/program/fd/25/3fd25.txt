PROF_ARRAY: {0, 2, 3, 4, 5, 6, 8, 10, 12}
loop PROF over PROF_ARRAY {
NORMAL_CHECK: 1d20 + 4 + PROF
ADV_CHECK: [highest 1 of 2d20] + 4 + PROF
DIS_CHECK: [lowest 1 of 2d20] + 4 + PROF
output NORMAL_CHECK/5 named "Number of Damage Dice with Proficiency [PROF]"
output (NORMAL_CHECK/5)d4 named "Expected Damage with Proficiency [PROF]"
output ADV_CHECK/5 named "Number of Damage Dice with Proficiency [PROF] at Advantage"
output (ADV_CHECK/5)d4 named "Expected Damage with Proficiency [PROF] at Advantage"
output DIS_CHECK/5 named "Number of Damage Dice with Proficiency [PROF] at Disadvantage"
output (DIS_CHECK/5)d4 named "Expected Damage with Proficiency [PROF] at Disadvantage"
}

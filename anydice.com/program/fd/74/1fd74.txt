function: select INDICES:s from SEQUENCE:s {
  RESULT: {}
  loop INDEX over INDICES {
    RESULT: {RESULT, INDEX@SEQUENCE}
  }
  result: RESULT
}

function: ROLL:s drop highest up to LIMIT:n {
  \ assumes ROLL is sorted in descending order! \
  OFFSET: ROLL > LIMIT 
  result: [select {1 .. OFFSET, OFFSET+2 .. #ROLL} from ROLL]
}

function: roll ROLL:s disadvantage DIS:s {
  \ first count the ones in both ROLL and DIS \
  ONES: (ROLL = 1) + (DIS = 1)

  \ then use remaining dice in DIS to knock out smaller dice \
  POOL: [sort ROLL]
  loop D over DIS {
    if D > 1 { POOL: [POOL drop highest up to D] }
  }

  \ calculate sum of remaining dice, excluding any ones \
  SUM: POOL - (POOL = 1)  

  \ finally combine SUM and ONES into a single number \
  result: 10*SUM + ONES
}

\ tests with fixed rolls \
output [roll {2,3} disadvantage 1] named "2,3 with extra 1 added, total of first 2 dice is 5"
output [roll {2,3} disadvantage 4] named "4 knocks out 3, leaving only 2"
output [roll {1,2} disadvantage 2] named "2 knocks out 2, leaving only 1"
output [roll {4,4} disadvantage 4] named "4 knocks out one 4, leaving 4"
output [roll {2,4} disadvantage 1] named "2,4 with extra 1 added, total of first 2 dice is 6"

\ actual distribution with two normal dice and one disadvantage die \
output [roll 2d6 disadvantage {}] named "2d6 with no disadvantage dice" 
output [roll 2d6 disadvantage 1d6] named "2d6 with one disadvantage die"

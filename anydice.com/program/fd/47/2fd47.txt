function: sign S:n {
\
   if S < 0 {result: -1}
   if S = 0 {result: 0}
\
   if S <= 0 {result: 0}
   if S > 0 {result: 1}
}

function: challenge P vs T {
   DIFF: [roll P] - [roll T]

   result: [sign DIFF]
}

function: roll LEVEL {
  if LEVEL = 2 { result: d4-2 }
  if LEVEL = 3 { result: d4-1 }

  if LEVEL = 4 { result: d4 }
  if LEVEL = 5 { result: d4+1 }

  if LEVEL = -6 { result: d6-1 }
  if LEVEL = 6 { result: d6 }
  if LEVEL = 7 { result: d6+1 }

  if LEVEL = -8 { result: d8-1 }
  if LEVEL = 8 { result: d8}
  if LEVEL = 9 { result: d8+1 }

  if LEVEL = -10 { result: d10-1 }
  if LEVEL = 10 { result: d10}
  if LEVEL = 11 { result: d10+1 }

  if LEVEL = -12 { result: d12-1 }
  if LEVEL = 12 { result: d12}
  if LEVEL = 13 { result: d12+1 }
  if LEVEL = 14 { result: d12+2}
  if LEVEL = 15 { result: d12+3 }


}

\loop P over { -6, 6, 7, -8, 8, 9, -10, 10 }\
loop T over { 12,8 }
{
loop P over { 6..10 }
{
   output [ challenge P vs T] named " [P] vs [T] "
}

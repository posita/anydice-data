function: DH:n hours of crime with DEX:n dex CHA:n cha and IN:n in at level N:n{
PAY: 100 * DH * DH * N / 16
X: 0
X: X + ( 1d20 + DEX >= DH / 2 )
X: X + ( 1d20 + CHA >= DH / 2 )
X: X + ( 1d20 + IN >= DH / 2 )
result: [help X PAY 100 * DH / 2]
}
function: help IN:n OUT FEE{
if IN=0 {result: 0 - FEE}
if IN=1 {result: OUT / 4 - FEE}
if IN=2 {result: OUT / 2 - FEE}
if IN=3 {result: OUT - FEE}
}

DIE: 10
FIRST: 2
SECOND: 5

function: roll A:n {
  result: A
}
function: aux A:n B:n C:n {
  if A <= B {
    A: 1dC
  }
  result: A
}
function: reroll A:n B:n C:n D:n E:n {
 X: 1
 
 if A <= B { 
  A: 1dC
  X: 0 
 }
 
 if X {
   D: [aux D E C]
 }

 result: A + D 
}

function: dicemagic A:n B:n C:n{
  X: [reroll 1dA B A 1dA C]
  result: X
}

output [dicemagic DIE FIRST SECOND]

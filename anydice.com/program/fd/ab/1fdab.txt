function: action AROLL:n AMOD:n CROLL:s{
  result: [lowest of (AROLL + AMOD) and 10] > CROLL
}

loop X over {0..7} {
  output [action 1d6 X 2d10] named "Action 1d6+[X]"
}

function: progress CROLL:s PROG:n {
  result: PROG > CROLL
}
output [progress 2d10 4] named "Progress 4"

function: highest N:n of A:s B:s {
    result: {1..N}@[sort {A, B}]
}

loop DIFFICULTY over {1..5} {
    loop RANK over {0..3} {
        output [highest 2 of (1+RANK)d8] - [highest 2 of DIFFICULTYd12] >= 0 named "[RANK] vs [DIFFICULTY]"
    }
    loop MASTERY over {1..3} {
        output [highest 2 of MASTERYd12 (4-MASTERY)d8] - [highest 2 of DIFFICULTYd12] >=0 named "M[MASTERY] vs [DIFFICULTY]"
    }
}

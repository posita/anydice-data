loop HM over {1..5} {
    loop P over {0..3} {
        output [highest 2 of (P+1)d8] - [highest 2 of HMd12] <=0 named "[P] vs [HM]"
    }
    loop MP over {1..3} {
        output [highest 2 of (MPd12 + (3-MP)d8)] - [highest 2 of HMd12] >=0 named "M[MP] vs [HM]"
    }
    output [highest 2 of HMd8] + 1d6 - [highest 2 of HMd12] >= 0 named "+d6@[HM]"
}

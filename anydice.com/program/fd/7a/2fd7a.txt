function: cap N:n {
  if N > 20 {
    N: [cap 7d3]
  }
  result: N
}

function: transform ROLL:n {
  ORIGINAL: ROLL
  if ROLL > 10 {
    ROLL: 21 - ROLL
  }

  ROLL: ROLL*2

  if ORIGINAL <= 10 {
    ROLL: ROLL-1
  }

result: ROLL
}

function: roll N:n {
  if N = 7 {
    M: 3
  } else {
    M: 20 / N
  }

  ROLL: [cap NdM]

  result: [transform ROLL]
}

output [roll 1]
output [roll 2]
output [roll 3]
output [roll 4]
output [roll 5]
output [roll 6]

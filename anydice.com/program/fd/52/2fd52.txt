function: abs A:n {
 if A > 0{result: A}
 result: A - (2* A)
}
function: emph of A:n and B:n{
  DA: [abs A - 10]
  DB: [abs B - 10]
  if DA > DB { result: A }
  if DB > DA { result: B }
  if DB = DA { result: A }
  result: 1d20
}
output 1d20 named "Flat Check"
output [highest 1 of 2d20] named "Advantage"
output [lowest 1 of 2d20] named "Disadvantage"
output [emph of 1d20 and 1d20] named "Emphasis"

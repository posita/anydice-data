function: sign S:n {
\
   if S < 0 {result: -1}
   if S = 0 {result: 0}
\
   if S <= 0 {result: 0}
   if S > 0 {result: 1}
}

function: pos S:n {
\
   if S < 0 {result: -1}
   if S = 0 {result: 0}
\
   if S <= 0 {result: 0}
   if S > 0 {result: S}
}


function: challenge P vs T {
   DIFF: [roll P] - [roll T]
\   result: [sign DIFF] \
   result: [pos DIFF]
}

function: roll LEVEL {
  if LEVEL = 1 { result: d4/2-1 }
  if LEVEL = 2 { result: d4/2 }
  if LEVEL = 3 { result: d4-1 }

  if LEVEL = 4 { result: d4 }
  if LEVEL = 5 { result: [lowest of 5 and d6] }

  if LEVEL = 6 { result: d6 }
  if LEVEL = 7 { result: [lowest of 7 and d8] }

  if LEVEL = 8 { result: d8}
  if LEVEL = 9 { result: [lowest of 9 and d10] }

  if LEVEL = 10 { result: d10}
  if LEVEL = 11 { result: [lowest of 11 and d12] }

  if LEVEL = 12 { result: d12}
  if LEVEL = 13 { result: [lowest of 13 and d20] }
  if LEVEL = 14 { result: [lowest of 14 and d20] }
  if LEVEL = 15 { result: [lowest of 15 and d20] }
  if LEVEL = 16 { result: [lowest of 16 and d20] }
  if LEVEL = 17 { result: [lowest of 17 and d20] }
  if LEVEL = 18 { result: [lowest of 18 and d20] }
  if LEVEL = 19 { result: [lowest of 19 and d20] }
  if LEVEL = 20 { result: d20 }
}

\loop P over { -6, 6, 7, -8, 8, 9, -10, 10 }\
loop T over { 8 }
{
loop P over { 1..12 }
{
   output [ challenge P vs T] named "[P]vs[T]"
}
}

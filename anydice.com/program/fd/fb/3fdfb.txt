function: pitch P:n {
 if P=1 {result: 2d6}
 if P=2 {result: 2d4}
 if P=3 {result: 1d6+1d4}
 result: [absolute 1d6-1d4]+2
} 
 
output [pitch 1] named "Fastball"
output [pitch 2] named "Curveball"
output [pitch 3] named "Changeup"
output [pitch 4] named "Slider"

function: zone ZONE:n {
 if ZONE>8 {result: 0}
 if ZONE<5 {result: 0}
 result: 1
}

output [zone [pitch 1]] named "Fastball Zone%"
output [zone [pitch 2]] named "Curveball Zone%"
output [zone [pitch 3]] named "Changeup Zone%"
output [zone [pitch 4]] named "Slider Zone%"

function: ROLL:n vs AC:n {
    if ROLL >= 20 {
        result: DAMAGE + DAMAGE + STRENGTH
    }
    if ROLL + ATTACK >= AC { result: DAMAGE + STRENGTH}
    result: 0
}

AC: d{14..20}
STRENGTH: 2
ATTACK: 2 + STRENGTH
DAMAGE: 1d6

output 16d[d20 vs AC] named "regular pugelist"
output 21d[d20 vs AC] named "Arena Royale"

STRENGTH: 3

output 16d[d20 vs AC] named "regular pugelist beef"
output 21d[d20 vs AC] named "Arena Royale beef"

function: cap N:n {
  if N > 20 {
    N: [cap 7d3]
  }
  result: N
}

function: transform ROLL:n {
  ORIGINAL: ROLL
  if ROLL > 10 {
    ROLL: 21-ROLL
  }

  ROLL: ROLL*2

  if ORIGINAL <= 10 {
    ROLL: ROLL-1
  }

  result: ROLL
}

function: roll ROLL:d {
  ROLL: [cap ROLL]

  result: [transform ROLL]
}

output [roll 1d20]
output [roll 2d10]
output [roll 4d5]
output [roll 5d4]
output [roll 7d3]

    if ((X-1)@DICE = X@DICE & (X-2)@DICE = (X-1)@DICE) {result: 2}
  }
  loop X over {2..#DICE} {
    if ((X-1)@DICE = X@DICE) {result: 1}
  }
  result: 0
}

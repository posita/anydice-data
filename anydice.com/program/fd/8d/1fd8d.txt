function: roll equal DICE:s {
  EQUAL: 0

  loop N over {2..#DICE} {
    if EQUAL = 0 {
      EQUAL: [is equal {N-1}@DICE N@DICE]
    }
  }

  result: EQUAL
}

function: is equal ONE:n TWO:n {
  result: (ONE = TWO) * ONE
}

DEFENSE: 14
LEVEL: 4
WEAPON_DICE: 2d6
WEAPON_NUMBER: 1
CRIT_RANGE: 20
PRECISION: 1d8
GWF: 3
NORMAL_HIT: d20
HIT_DIS: [lowest 1 of 2d20]
EXTRA_DAMAGE: 2

MOD: 4 + 1
PROF: (LEVEL+7)/4
GWMATTACK: -5
GWMDAMAGE: 10

MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n plus BONUS:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= CRIT_RANGE { result: CRITICALHIT }
 if ROLL + BONUS + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: reroll R:n under N:n {
   if R <= N {result: WEAPON_DICE} else {result: R}
}

function: attack {
 result: [damage for [attack NORMAL_HIT vs DEFENSE plus PRECISION]]
}

function: attackdis {
 result: [damage for [attack HIT_DIS vs DEFENSE plus PRECISION]]
}



ATTACK: MOD + PROF
HITDAMAGE: WEAPON_NUMBERd[reroll WEAPON_DICE under GWF] + MOD + EXTRA_DAMAGE
CRITICALDAMAGE: 2d(WEAPON_NUMBERd[reroll WEAPON_DICE under GWF]) + MOD + EXTRA_DAMAGE
MISSDAMAGE: 0
output [attack] named "Normal"

ATTACK: MOD + PROF
HITDAMAGE: WEAPON_NUMBERd[reroll WEAPON_DICE under GWF] + MOD + EXTRA_DAMAGE
CRITICALDAMAGE: 2d(WEAPON_NUMBERd[reroll WEAPON_DICE under GWF]) + MOD + EXTRA_DAMAGE
MISSDAMAGE: 0
output [attackdis] named "Disadvantage Normal"

ATTACK: MOD + PROF + GWMATTACK
HITDAMAGE: WEAPON_NUMBERd[reroll WEAPON_DICE under GWF] + MOD + GWMDAMAGE + EXTRA_DAMAGE
CRITICALDAMAGE: 2d(WEAPON_NUMBERd[reroll WEAPON_DICE under GWF]) + MOD + GWMDAMAGE + EXTRA_DAMAGE
MISSDAMAGE: 0
output [attack] named "GWM"

ATTACK: MOD + PROF + GWMATTACK
HITDAMAGE: WEAPON_NUMBERd[reroll WEAPON_DICE under GWF] + MOD + GWMDAMAGE + EXTRA_DAMAGE
CRITICALDAMAGE: 2d(WEAPON_NUMBERd[reroll WEAPON_DICE under GWF]) + MOD + GWMDAMAGE + EXTRA_DAMAGE
MISSDAMAGE: 0
output [attackdis] named "Disadvantage GWM"

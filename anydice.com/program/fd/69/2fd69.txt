function: sign S:n {
   if S < 0 {result: -1}
   if S = 0 {result: 0}
\   if S <= 0 {result: 0} \
   if S > 0 {result: 1}
}

function: negative to zero S:n {
\
   if S < 0 {result: -1}
   if S = 0 {result: 0}
\
   if S <= 0 {result: 0}
   if S > 0 {result: S}
}



function: reroll if max D:d {
   result: [reroll D if above ([maximum of D]-1)]
}

function: reroll D:n if above T:n {
   if D > T {result: 1dD }
   else { result: D }
}


function: challenge P vs T {
   DIFF: [roll P] - [roll T]

   result: [sign DIFF] 
\   result: [negative to zero DIFF]  \
\   result: DIFF \
}

function: roll LEVEL {
  if LEVEL = 1 { result: [reroll d4 if above LEVEL] }
  if LEVEL = 2 { result: [reroll d4 if above LEVEL] }
  if LEVEL = 3 { result: [reroll if max d4] }

  if LEVEL = 4 { result: d4 }
  if LEVEL = 5 { result: [reroll d6 if above LEVEL]}

  if LEVEL = 6 { result: d6 }
  if LEVEL = 7 { result: [reroll d8 if above LEVEL]}

  if LEVEL = 8 { result: d8}
  if LEVEL = 9 { result: [reroll d10 if above LEVEL]}

  if LEVEL = 10 { result: d10}
  if LEVEL = 11 { result: [reroll d12 if above LEVEL]}

  if LEVEL = 12 { result: d12}
  if LEVEL = 13 { result: [reroll d20 if above LEVEL] }
  if LEVEL = 14 { result: [reroll d20 if above LEVEL] }
  if LEVEL = 15 { result: [reroll d20 if above LEVEL] }
  if LEVEL = 16 { result: [reroll d20 if above LEVEL] }
  if LEVEL = 17 { result: [reroll d20 if above LEVEL] }
  if LEVEL = 18 { result: [reroll d20 if above LEVEL] }
  if LEVEL = 19 { result: [reroll d20 if above LEVEL] }
  if LEVEL = 20 { result: [reroll d20 if above LEVEL] }
}

\loop P over { -6, 6, 7, -8, 8, 9, -10, 10 }\
loop T over { 4, 8 }
{
loop P over { 1..12 }
{
   output [ challenge P vs T] named "[P]vs[T]"
}

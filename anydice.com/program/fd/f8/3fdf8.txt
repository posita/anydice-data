function: roll A:s damage D:d deadly DD:d against AC:n with bonus B:n {
  M: -1
  if (A + B) > AC - 10 {
    M: 0
  }
  if (A + B) >= AC {
    M: 1
  }
  if (A + B) >= AC + 10 {
    M: 2 \ Crit \
  }
  if (A = 1) {
    M: M - 1
  }
  if (A = 20) {
    M: M + 1
  }

  M: [lowest of 2 and M]
  M: [highest of 0 and M]
  if (M != 2) {
    DD: 0
  }

  result: D * M + DD
}

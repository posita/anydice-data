function: success MOD:n POOL:n {
  result: 1d20 + MOD + POOL >= 16
}

loop M over {1..8} {
  output [success M (-1d6)] named "-1d6, mod +[M]"
}
loop M over {1..8} {
  output [success M 0] named "0d6, mod +[M]"
}
loop M over {1..8} {
  output [success M 1d6] named "+1d6, mod +[M]"
}
loop M over {1..8} {
  output [success M 2d6] named "+2d6, mod +[M]"
}
loop M over {1..8} {
  output [success M 3d6] named "+3d6, mod +[M]"
}

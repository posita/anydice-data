\ CONFIGURATION \
TO_HIT: 5
AC: 13
CRIT_RANGE: 20
FLAT_MOD: 4

\ UNARMED STRIKES \
function: unarmed ROLL:n {
  if ROLL = 1 { result: 0 }
  if ROLL >= CRIT_RANGE { 
    result: 6 + 4 + 1d6 + 1d4 + FLAT_MOD 
  }
  if ROLL + TO_HIT >= AC { 
    result: 1d6 + 1d4 + FLAT_MOD 
  }
  result: 0
}

\ RAPIER \
function: rapier ROLL:n {
  if ROLL = 1 { result: 0 }
  if ROLL >= CRIT_RANGE { 
    result: 8 + 4 + 1d8 + 1d4 + FLAT_MOD + 2
  }
  if ROLL + TO_HIT >= AC { 
    result: 1d8 + 1d4 + FLAT_MOD + 2
  }
  result: 0
}

\ OUTPUTS \
output [unarmed 1d20] + [unarmed 1d20] named "Unarmed Strikes"
output [rapier 1d20] named "Rapier"

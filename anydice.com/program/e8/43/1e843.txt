function: recount R:n over N:n {
   if R > N { result: 1d20 } else {result: R}
}

SUM: 0
loop N over {1..1}{
   HIT: [count {1..15,1..3} in [recount 1d20 over 15]]
   SUM: SUM+HIT
}

output SUM + [count {1..15,1..3} in 1d20]
output SUM + [count {1..15,1..3} in 2d20]-1

SUM: 0
loop N over {1..1}{
   HIT: [count {1..17,1..1} in [recount 1d20 over 17]]
   SUM: SUM+HIT
}

output SUM + [count {1..17,1..1} in 1d20]
output SUM + [count {1..17,1..1} in 2d20]-1

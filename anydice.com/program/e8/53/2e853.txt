MISS: 0
CRITICALHIT: 2d10 + 10 + 5 + 4d6
HIT: 1d10 + 10 + 5 + 2d6
ATTACK: 3 + 5 - 5

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

DAMAGE: 0

loop N over {1..2}{
 DAMAGE: DAMAGE + [attack d20 vs 15]
}
output DAMAGE


function: check helper SCORE {
 if SCORE / 3 = 0 {
  result: [explode d4]
 } else if SCORE / 3 = 1 {
  result: [explode d6]
 } else if SCORE / 3 = 2 {
  result: [explode d8]
 } else if SCORE / 3 = 3 {
  result: [explode d10]
 } else if SCORE / 3 = 4 {
  result: [explode d12]
 } else {
  result: 0
 }
}

function: check SCORE {
 result: [check helper SCORE+2] + [check helper SCORE+1] + [check helper SCORE]
}

function: check SCORE against DIFF {
 result: [check SCORE] >= DIFF
}

DC: 15
SKILL: 5

\output [check 0]\


loop N over {0..12}{
 output [check N against DC] named "[N] against [DC]"
}

loop N over {3..20} {
 output [check SKILL against N]+2 named "[SKILL] against [N]"
}

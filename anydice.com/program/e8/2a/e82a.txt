\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n DAMAGE:n{
 if DAMAGE <0 {result: 0}
 if ROLL  = CRITROLL { result: CRITDAMAGE } 
 if ROLL  = TRUECRITROLL{ result: TRUECRITDAMAGE }
 if ROLL = 1 { result: 0 }
 if ROLL + ATTACKMOD + ACCBONUS >= ENEMYAC{ result: DAMAGE }
 result: 0}

TRUECRITDAMAGE: 15+ DMGBONUS
output ATTACKTIMESd[attackroll 1d20 vs ENEMYAC DAMAGE] named "Normal"

function: X:n attack AC:n ac M:n attackmod Y:n damage B:n critsrange C:n critmult D:n addcritconfirm{

if X= 1 {result: 0} \sad crit\

if X >=B { \crit chance\
    if X+M+D >=AC { result: C*Y} \crit confirmed\
    else if X=20 { result: Y}    \crit no confirm still hits on a 20\ 
}

if X+M >=AC {result: Y} \hit\

else { result: 0 } \miss\

}



loop N over {12..30}{ \to set the AC range\


KUK_PRI_ATTACK_ONE: [d20 attack N ac 14 attackmod d4+4 damage 15 critsrange 2 critmult 4 addcritconfirm]
KUK_SEC_ATTACK_ONE: [d20 attack N ac 14 attackmod d4+3 damage 15 critsrange 2 critmult 4 addcritconfirm]

KUK_PRI_ATTACK_TWO: [d20 attack N ac 9 attackmod d4+4 damage 15 critsrange 2 critmult 4 addcritconfirm]
KUK_SEC_ATTACK_TWO: [d20 attack N ac 9 attackmod d4+3 damage 15 critsrange 2 critmult 4 addcritconfirm]

output KUK_PRI_ATTACK_ONE+KUK_SEC_ATTACK_ONE+KUK_PRI_ATTACK_TWO+KUK_SEC_ATTACK_TWO named "Kukri on [N] AC"

}

set "maximum function depth" to 3

function: roll DICE:d against dr DIFFICULTY:n {
  result: [count {DIFFICULTY..10} in DICE]
}

function: roll DICE:d against dr DIFFICULTY:n with disadvantage of DIS:n {
  result: [roll DICE against dr DIFFICULTY with disadvantage of DIS]
}

function: get successes for ROLLEDVALUE:n DIE:n against dr DIFFICULTY:n {
  if ROLLEDVALUE = DIE { result: 1 + [get successes for dDIE DIE against dr DIFFICULTY] }
  if ROLLEDVALUE >= DIFFICULTY { result: 1 }
  result: 0
}

function: get successes for ROLLEDVALUE:n DIE:n against dr DIFFICULTY:n with disadvantage of DISADVANTAGE:n {
  if ROLLEDVALUE = DIE { result: 1 + [get successes for dDIE DIE against dr DIFFICULTY] }
  if ROLLEDVALUE >= DIFFICULTY+DISADVANTAGE { result: 1 }
  if ROLLEDVALUE <= DISADVANTAGE { result: -1 }
  result: 0
}


function: get failures only for ROLLEDVALUE:n DIE:n with disadvantage of DISADVANTAGE:n {
  if ROLLEDVALUE = DIE { result: 1 + [get successes for dDIE DIE against dr DIFFICULTY] }
  if ROLLEDVALUE <= DISADVANTAGE { result: -1 }
  result: 0
}

function: roll exploded COUNT:n de DIE:n against dr DIFFICULTY:n {
  RES: 0
  loop N over {1..COUNT} {
    RES: RES + [get successes for dDIE DIE against dr DIFFICULTY]
  }
  result: RES
}

function: roll exploded COUNT:n de DIE:n against dr DIFFICULTY:n with disadvantage of DIS:n {
  RES: 0
  loop N over {1..COUNT} {
    RES: RES + [get successes for dDIE DIE against dr DIFFICULTY with disadvantage of DIS]
  }
  if DIS > COUNT {
    DIFFERENCE: DIS - COUNT
    loop N over {1..DIFFERENCE} {
      RES: RES + [get failures only for dDIE DIE with disadvantage of DIS]
    }
  }
  result: RES
}

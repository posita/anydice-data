function: damage DMG:n with DIE:n by RATING:n {
 if DIE >= 6 {
  result: DMG * RATING
 }
 else {
  result: DMG
 }
}

loop ATTR over {0..4}{
 loop MOD over {0..3}{
  output [damage 1d6 + ATTR with 1d8 - MOD by 2] named "Light ([ATTR], [MOD])"
  output [damage 1d8 + ATTR with 1d6 - MOD by 3] named "Heavy ([ATTR], [MOD])"
 }
}

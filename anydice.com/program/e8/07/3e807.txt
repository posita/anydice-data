AT: ATT
CO: CRITOVER
CM: CRITMULT
T: N + ATT
CON: CHECK + ATT
WP: WEAPONDAMAGE
  if N >= CRITOVER {
    if CON >= TARGETAC {
      result: WP*CRITMULT
    }
    else {
      result: WP
    }
  }
  else {
    if T >= TARGETAC {
      result: WP
    }
    else {
      result: 0
    }
  }
}

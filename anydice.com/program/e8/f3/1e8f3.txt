\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ABILITYMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

ABILITYMOD: 3
DMGBONUS: 1
PROFICIENCY: 2
ACCBONUS: 2
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 1d8 + 1d6 + ABILITYMOD + DMGBONUS
CRITDAMAGE: 2d8 + 2d6 + ABILITYMOD + DMGBONUS

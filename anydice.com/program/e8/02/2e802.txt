set "maximum function depth" to 10

function: attack ATK:d doing DMG:d vs AC:n {
  result: (ATK > AC) * DMG
}

function: number of HIT:d to deal HP:n {
  if HP <= 0 {
    result: 0
  } else {
    result: [number of HIT to deal (HP - HIT)] + 1
  }
}

MONSTER_ATK: 1d20+5
MONSTER_DMG: 1d6+3

WIZARD_AC: 13
WIZARD_HP: 7 \+ 1d6\
output [number of [attack MONSTER_ATK doing MONSTER_DMG vs WIZARD_AC] to deal WIZARD_HP] - 1 named "Wizard Survivability"

FIGHTER_AC: 18
FIGHTER_HP: 12 \+ 1d10\
output [number of [attack MONSTER_ATK doing MONSTER_DMG vs FIGHTER_AC] to deal FIGHTER_HP] - 1 named "Fighter Survivability"

function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE - STRENGTH
 }
 result: DAMAGE 
}

AC: 12
STRENGTH: 4
ATTACK: 6

DAMAGE: 1d6+4
THREAT: 20
CRITICAL: 2
output [d20 vs AC] named "Gun"

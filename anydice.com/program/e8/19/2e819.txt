DICE: {6, 8, 10, 12, 20}
ATTR_RANGE: {1..5}

function: middle value of A:n B:n and C:n {
  SORTED: [sort {A, B, C}]
  result: 2@SORTED
}

FIRST: 0
loop GRIT_RAW over ATTR_RANGE {
  loop FOCUS_RAW over ATTR_RANGE {
    loop WILL_RAW over ATTR_RANGE {
      if (FIRST <= 0 | GRIT_RAW > 1 | FOCUS_RAW > 1 | WILL_RAW > 1 ) {
        FIRST: 1
        DICE_SIZE: [highest of [middle value of GRIT_RAW FOCUS_RAW and WILL_RAW] and 1 ]@DICE
        DICE_COUNT: [highest of (GRIT_RAW + FOCUS_RAW + WILL_RAW) and 1 ]
        output DICE_COUNTdDICE_SIZE named "[DICE_COUNT]d[DICE_SIZE] G:[GRIT_RAW] F:[FOCUS_RAW] W:[WILL_RAW]"
      }
    }
  }
}

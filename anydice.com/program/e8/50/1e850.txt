
function: precise reroll R:n aka RESULT:n under N:n {
 if R < N { result: RESULT } else { result: R }
}
function: one reroll R:n under N:n {
 if R < N { result: d6 } else { result: R }
}
function: two reroll R:n under N:n {
 if R < N { result: 2d6 } else { result: R }
}

function: save DIE:n SAVE rend REND {
 if DIE = 6 { result: 0 } 
 if DIE = 1 { result: 1 }
 result: (DIE - REND) < SAVE
}

function: unwieldy ATT attacks HIT hit WOU wound vs SAV rend REND {
 result: ATTd((d6 >= HIT) * (d6 >= WOU) * [save d6 SAV rend REND])
}
function: unwieldy stabbas ATT attacks HIT hit DHIT:n and WOU wound DWOU:n  vs SAV rend REND with LOONBOSS loonboss {
 result: ATTd((DHIT >= HIT) * (DWOU >= WOU) * [save d6 SAV rend REND]) + LOONBOSS * (ATTd((DHIT >= HIT) * (DWOU = 6)))
}
function: unwieldy HIT hit DHIT:n and WOU wound with EXPLODE exploding hits vs SAV rend REND {
 if DHIT = 6 { result: (1 + EXPLODE)d((d6 >= WOU) * [save d6 SAV rend REND]) }
 else { result: ((DHIT >= HIT) * (d6 >= WOU) * [save d6 SAV rend REND]) }
}

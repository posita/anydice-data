NUMBEROFDICE: 3  
BONUS: 0           
DIE: 1d6           

function: ob NUMBER:n plus BONUS {
  TOTALSUM: 0
  loop LOOPING over {1..NUMBER} {
    X: DIE
    X: [checkmax X]
    TOTALSUM: TOTALSUM + X 
  }
  result: TOTALSUM+ BONUS
}

function: checkmax N:n {
  if N = 6 {
    result: [ob 2 plus 0]
  }
  result: N
}

set "maximum function depth" to 6    

output [ob 4 plus 0]>[ob 5 plus 0]

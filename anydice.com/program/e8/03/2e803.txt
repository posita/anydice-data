function: ex evaluate ROLL:s diff DIFF:n {
 SUCCESSES: (ROLL >= 7) + (ROLL = 10)
 if SUCCESSES >= DIFF { result: 1 }
 if SUCCESSES > 0 { result: 0 }
 if [count 1 in ROLL] { result: -1 }
 result: 0
}

function: ex roll DICE:n diff DIFF:n {
  result: [ex evaluate DICE d10 diff DIFF]
}

loop DICE over {1..10} {
 output [ex roll DICE diff 4] named "[DICE]d"
}

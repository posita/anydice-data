  result: d4
 } else if SCORE / 3 = 1 {
  result: d6
 } else if SCORE / 3 = 2 {
  result: d8
 } else if SCORE / 3 = 3 {
  result: d10
 } else if SCORE / 3 = 4 {
  result: d12
 } else {
  result: 0
 }
}

}

}

function: differences of D:s {
  F: 1@D
  S: 2@D
  DX: [absolute F - S]
  if F >= 15 {
    if S >= 15 {
      result: DX + 1
    } else {
      result: 0
    }
  } else {
    if F <= 3 {
      if S <= 3 {
        result: -DX - 1
      } else {
        result: 0
      }
    } else {
      result: 0
    }
  }
}

function: crits of F:n and S:n {
  if F = S {
    if F >= 6 {
      result: 1
    } else {
      if F = 1 {
        result: -1
      } else {
        result: 0
      }
    }
  } else {
    result: 0
  }
}

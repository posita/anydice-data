BOOST:5 \* Maximum number of dice to be rolled *\
SETBACK:0 \* Maximum number of dice to be rolled *\


function: evaluate ROLL_ADV:s ROLL_DIS:s
{
     ADVANTAGE: ROLL_ADV >= 4
     DISADVANTAGE: ROLL_DIS >= 4

     if ADVANTAGE > DISADVANTAGE
     {
          SUCCESSES: ADVANTAGE - DISADVANTAGE
          result: SUCCESSES
     }
     else 
     {
          SUCCESSES: 0
          result: SUCCESSES
     }
}


output [evaluate BOOST d6 SETBACK d6] named "[BOOST]d / [SETBACK]d"

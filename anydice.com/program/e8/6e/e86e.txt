 if A-B > 0 {
  if A-B <= SIZE { result: WOUND }
}
 if A-B > SIZE {
  if A-B <= SIZE*2 { result: MAJOR }
}
 if A-B > SIZE*2 {
  if A-B <= SIZE*3 { result: CRITICAL }
}
 if A-B > SIZE*3 {
  if A-B <= SIZE*4 { result: DEATHSDOOR }
}
 if A-B > SIZE*4 {
  if A-B <= SIZE*5 { result: INSTANTDEATH }
}
 if A-B > -SIZE {
  if A-B <= 0 { result: LIGHT }
}
 if A-B > -SIZE*2 {
  if A-B <= -SIZE { result: FATIGUE }
}
 if A-B > -SIZE*3 {
  if A-B <= -SIZE*2 { result: NOTHING }
}
 if A-B <= -SIZE*3 {
  if A-B <= -SIZE*3 { result: NOTHING }
}
 if A-B >= SIZE*5 {
  if A-B >= SIZE*5 { result: INSTANTDEATH }
}
}

function: successes ADV:s THR:s {
  result: ([count {6,5,4} in ADV]) - ([count {6,5,4} in THR])
}

function: crits ADV:s THR:s {
  BOONS: (ADV = 6) - (THR = 6)  
  BANES: (THR = 6) - (ADV = 6)
    
    if (BOONS > BANES) { result: 100 }
    if (BANES > BOONS) { result: -100 }
    if (BANES = BOONS) { result: 0 }
}

function: successesalt ADV:s THR:s {
  result: ([count {6,5,4} in ADV]) - ([count {6,5} in THR])
}

\output [crits 5d6 1d6] named "2 Adv | 1 Thr"

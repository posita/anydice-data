set "explode depth" to 10

RANGE: 10
LEGEND: {0,0,1,1,1,1}
POISSON: {0,0,0,1,1,1,2,3}
XSIX: {0,0,0,1,2,3}
XLUCK: {0,0,0,1,1,2}

function: clamp N:n between MIN:n and MAX:n {
  if N > MAX { result: MAX }
  if N < MIN { result: MIN }
  result: N
}

function: xluck R:n {
  if R > 5 { result: 2 + [xluck d6] }
  if R > 3 { result: 1 }
  result: 0
}

function: lluck R:n {
  if R > 5 { result: 1 + [lluck d6] }
  if R > 3 { result: 1 }
  result: 0
}

function: mluck R:n and P:n {
  if R > 5 { result: 1 + [mluck d6 and 0] }
  R: R + P
  if R > 3 { result: 1 }
  result: 0
}

\ XSIX \
\
loop N over {0,1,4,5} {
  loop P over {0..1} {
    output [clamp NdXSIX+[xluck d6+P] between -RANGE and RANGE] named "[N]dX+[P] dL"
  }
}
\

\ Modified Legend \
\
loop N over {0..8} {
  loop P over {0..1} {
    output [clamp Nd{0..1}+[lluck d6+P] between -RANGE and RANGE] named "[N]dX+[P] dL"
  }
}
\

\ Modified Luck \
loop N over {0..8} {
  output [clamp Nd{0..1}+[mluck d6 and -1] between -RANGE and RANGE] named "[N]dX-1 dL"
  output [clamp Nd{0..1}+[mluck d6 and 0] between -RANGE and RANGE] named "[N]dX dL"
  output [clamp Nd{0..1}+[mluck d6 and 1] between -RANGE and RANGE] named "[N]dX+1 dL"
}

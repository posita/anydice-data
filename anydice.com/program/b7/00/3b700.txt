HIEBWAFFEN: 17

function: ochsenherde mod TPA{
 result: (3d6) + 4 + TPA
}

function: attack erschw ERSCHW{
 result: (1d20<=(HIEBWAFFEN-ERSCHW))
}

loop FIRST over {0..HIEBWAFFEN}{loop SECOND over {0..FIRST}{
 output [attack erschw FIRST]*[ochsenherde mod FIRST]+[attack erschw (SECOND+4)]*[ochsenherde mod SECOND] named "Wucht: [FIRST],[SECOND]"
}}

loop FIRST over {2..HIEBWAFFEN}{loop SECOND  over {2..FIRST}{
 output [attack erschw (FIRST-2-1)]*[ochsenherde mod (FIRST+1)]+[attack erschw(SECOND+2-2-1)]*[ochsenherde mod (SECOND+1)] named "Aggro: [FIRST],[SECOND]"
}}

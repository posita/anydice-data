function: ratingdbl ROLLA:n and ROLLB:n {
 if ROLLA >= 6 { result: 1 }
 else { result: ROLLB >= 6 }
}

function: ratingfcs ROLLA:n and ROLLB:n {
 if ROLLA >= 5 { result: 1 }
 else { result: ROLLB >= 5 }
}

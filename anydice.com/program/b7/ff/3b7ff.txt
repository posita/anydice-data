DIE: d{0,0,1,1,2,3}

function: reroll N:n at most T:n times {
  if N < 3 & T {
    result: [reroll DIE at most T - 1 times]
  }
  result: N
}

loop R over {0..3} {
  output 3d[reroll DIE at most R times] named "3d6 rerolls x[R]"
}

loop R over {0..3} {
  output 4d[reroll DIE at most R times] named "rerolls x[R]"
}

loop R over {0..3} {
  output 5d[reroll DIE at most R times] named "rerolls x[R]"
}

loop R over {0..5} {
  output Rd[reroll DIE at most 1 times] named "number of [R]dice"
}

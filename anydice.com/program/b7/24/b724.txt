function: ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD >= AC { result: DAMAGE }
 result: 0
}

ENEMYAC: d{15}

ATTACKMOD: 3
PROFICIENCY: 2
CRITROLL: 20
DAMAGE: 1d12 + ATTACKMOD
CRITDAMAGE: 2d12 + ATTACKMOD
output [d20 vs ENEMYAC] named "Greataxe"

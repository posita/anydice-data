function: ratingdbl ROLLA:n and ROLLB:n {
 if ROLLA >= 6 { result: 1 }
 else { result: ROLLB >= 6 }
}

\output [ratingdbl 1d6 and 1d8] named "1d6 & 1d8"\
output [ratingdbl 1d8 and 1d8] named "2d8"
output [ratingdbl 1d8+1 and 1d8] named "2d8 +1 on one"
output [ratingdbl 1d8+1 and 1d8+1] named "2d8+1"
output [ratingdbl 1d8 and 1d10] named "1d8 & 1d10"

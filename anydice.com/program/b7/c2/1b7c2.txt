function: jet DICE:s plus BONUS  {
  MAX: 0
  loop X over {1..#DICE} {
    if MAX=0 {if (X)@DICE=0 {MAX:99}}
    if MAX=0 {MAX:(X)@DICE+BONUS}
    if #DICE>1 {
        if ((X-1)@DICE = X@DICE) {
            if X@DICE+10>MAX {MAX: X@DICE+10}
            if #DICE>2 {
                if ((X-2)@DICE = X@DICE) {
                    if X@DICE+20>MAX {MAX: X@DICE+20}
                }
            }
        }
    }
  }
  if MAX=99 {MAX:0}
  result: MAX
}

output [jet 6d{0..9} plus 1] named "6d+1"
output [jet 6d{0..9} plus 3] named "6d+3"
output [jet 6d{0..9} plus 5] named "6d+5"

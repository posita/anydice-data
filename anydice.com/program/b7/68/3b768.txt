
BASICS: 6d({1,2,3}@4d6)
loop P over {1..6} {
output P @ BASICS named "Basic [P]"
}

DOOMS: 6d(12+(1d8/1d{-2,-1,2,1}))
loop P over {1..6} {
 output P @ DOOMS named "Doom [P]"
}

function: picker C:n A:n B:n{if C = 1{result: A} else {result: B}}
function: pick A:n B:n{result: [picker 1d2 A B]}
MIXED: 6d[pick {1,2,3}@4d6 (12+(1d8/1d{-2,-1,2,1}))]
loop P over {1..6} {
 output P @ MIXED named "Mixed [P]"
}

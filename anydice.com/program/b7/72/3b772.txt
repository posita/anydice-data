LOW: 9
MID: 14
HIGH: 19

function: evaluate ROLL:s {
 if (ROLL <= LOW) { result: 0 }
 else {
  if (ROLL <= MID) { result: 1 }
  else {
   if (ROLL <= HIGH) { result: 2 }
   else {
    result: 3
   }
  }
 }
}

\output [evaluate d6+d6] named "QUALITY"\
\output d6+d6 named "RAW VALUES"\

loop LEVEL over {1..5} {
 loop P over {4,6,8,10,12} {
  \output LEVELdP + d6 named "d[P]+d6"\
  output [evaluate 1@LEVELdP+d6] named "[LEVEL]d[P]"
  output [evaluate 1@LEVELdP+d6+d4] named "[LEVEL]d[P] blessed"
 }
}

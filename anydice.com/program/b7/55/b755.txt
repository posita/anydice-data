function: make TARGET:n from two of DICE:s {
  PAIRS: 0
  COUNT: #DICE
  loop I over {1..COUNT} {
    loop J over {(I+1)..COUNT} {
      if (I@DICE + J@DICE = TARGET) {
        PAIRS: PAIRS + 1
      }
    }
  }
  result: PAIRS > 0
}

output [make 8 from two of 5d6] named "Target = 8"
output [make 12 from two of 5d6] named "Target = 12"

output [highest 2 of 5d6] = 12 named "also Target = 12"

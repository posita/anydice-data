function: crithit A:s{
 result: [count 6 in A] >= 2
}

function: directhit A:s with B{
 result: ([count 6 in A] = 1) & (A + B >= 12)
}

function: hit A:s with B{
 result: ((A + B) >= 12) & ([count 6 in A] < 1)
}

function: glancehit A:s with B{
 result: ((A + B) < 12) & ([count 6 in A] >= 1)
}

function: miss A:s with B{
 result: ((A + B) < 12) & ([count 6 in A] < 1)
}

function: directhit A:s with B{

DICE : 3d6 
BRAVERY : 2

output (DICE + BRAVERY) >= 12        named "Chance to Succeed"
output (DICE + BRAVERY) < 12         named "Chance to Fail" 
output [crithit DICE]                named "Chance to get a Critical Hit"
output [directhit DICE with BRAVERY] named "Chance to get a Direct hit"
output [hit DICE with BRAVERY]       named "Chance to get a Hit"
output [glancehit DICE with BRAVERY] named "Chance to get a Glancing hit"
output [miss DICE with BRAVERY]      named "Chance to get a Miss"

function: blades POOL:s SUCCESS:n PARTIAL:n {
  if 1@POOL >= SUCCESS & 2@POOL >= SUCCESS {
    result: 3
  }
  if 1@POOL >= SUCCESS {
    result: 2
  }
  result: 1@POOL >= PARTIAL
}

loop N over {1..4} {
  output [blades Nd6 6 4] named "[N]d6"
}

loop N over {1..7} {
  output [blades Nd10 10 7] named "[N]d10"
}

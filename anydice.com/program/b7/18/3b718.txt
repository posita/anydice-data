function: specialbestof S:s max T:n {
  HIGHEST_VALID: 0
  HIGHEST_COUNT: 0
  BEST_RESULT: 0

  loop I over S {
    if I <= T {
      COUNT: [count I in S]
      MODIFIED_VALUE: I + (COUNT - 1)

      if MODIFIED_VALUE > BEST_RESULT {
        BEST_RESULT: MODIFIED_VALUE
      }
    }
  }

  result: BEST_RESULT
}

function: bestof S:s max T:n  {
  HIGHEST_VALID: 0
  loop I over S {
    if I <= T & I > HIGHEST_VALID { HIGHEST_VALID: I }
  }
  result: HIGHEST_VALID
}

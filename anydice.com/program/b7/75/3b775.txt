\Regular Functions\
function: crithit with A:s and B and C{
 result: ([count 6 in A] >= 2) & (A + B >= C)
}

function: directhit with A:s and B and C{
 result: ([count 6 in A] = 1) & (A + B >= C)
}

function: hit with A:s and B and C{
 result: ([count 6 in A] < 1) & (A + B >= C)
}

function: glancehit with A:s and B and C{
 result: ([count 6 in A] >= 1) & (A + B < C)
}

function: miss with A:s and B and C{
 result: ([count 6 in A] < 1) & (A + B < C)
}

\Advantage Functions\
function: crithita with A:s and B and C{
 result: ([count 6 in A] >= 1) & (A + B >= C)
}

function: directhita with A:s and B and C{
 result: ([count 6 in A] < 1) & (A + B >= C)
}

function: hita with A:s and B and C{
 result: 0
}

function: glancehita with A:s and B and C{
 result: (A + B < C)
}

function: missa with A:s and B and C{
 result: 0
}

\Disadvantage Functions\
function: crithitd with A:s and B and C{
 result: ([count 6 in A] >= 2) & (A + B >= C)
}

function: directhitd with A:s and B and C{
 result: ([count 6 in A] = 1) & (A + B >= C)
}

function: hitd with A:s and B and C{
 result: ([count 6 in A] < 1) & (A + B >= C)
}

function: glancehitd with A:s and B and C{
 result: ([count 6 in A] >= 1) & (A + B < C)
}

function: missd with A:s and B and C{
 result: ([count 6 in A] < 1) & (A + B < C)
}

\Variables\
DICE    : 5
BRAVERY : 0
DIFFICULTY: 20

DICER    : DICEd6
DICEA   : (DICE-1)d6
DICED   : (DICE-1)d6

DIFFICULTYA: DIFFICULTY - 6
DIFFICULTYD: DIFFICULTY - 1

\Regular Outputs\
output (DICER + BRAVERY) >= DIFFICULTY        named "Chance to Succeed"
output (DICER + BRAVERY) < DIFFICULTY         named "Chance to Fail" 

output [crithit with DICER and BRAVERY and DIFFICULTY]         named "Chance to get a Critical Hit"
output [directhit with DICER and BRAVERY and DIFFICULTY]       named "Chance to get a Direct hit"
output [hit with DICER and BRAVERY and DIFFICULTY]             named "Chance to get a Hit"
output [glancehit with DICER and BRAVERY and DIFFICULTY]       named "Chance to get a Glancing hit"
output [miss with DICER and BRAVERY and DIFFICULTY]            named "Chance to get a Miss"

\Advantage Outputs\
output (DICEA + BRAVERY) >= DIFFICULTYA        named "Chance to Succeed with Advantage"
output (DICEA + BRAVERY) < DIFFICULTYA         named "Chance to Fail with Advantage" 

output [crithita with DICEA and BRAVERY and DIFFICULTYA]         named "Chance to get a Critical Hit with Advantage"
output [directhita with DICEA and BRAVERY and DIFFICULTYA]       named "Chance to get a Direct hit with Advantage"
output [hita with DICEA and BRAVERY and DIFFICULTYA]             named "Chance to get a Hit with Advantage"
output [glancehita with DICEA and BRAVERY and DIFFICULTYA]       named "Chance to get a Glancing hit with Advantage"
output [missa with DICEA and BRAVERY and DIFFICULTYA]            named "Chance to get a Miss with Advantage"

\Disadvantage Outputs\
output (DICED + BRAVERY) >= DIFFICULTYD        named "Chance to Succeed with Disadvantage"
output (DICED + BRAVERY) < DIFFICULTYD         named "Chance to Fail with Disadvantage" 

output [crithitd with DICED and BRAVERY and DIFFICULTYD]         named "Chance to get a Critical Hit with Disadvantage"
output [directhitd with DICED and BRAVERY and DIFFICULTYD]       named "Chance to get a Direct hit with Disadvantage"

function: damage ACC:n MOD:n HIGH:n LOW:n {
 if ACC >= HIGH {
   result: [adjust 5d8+3 with MOD]
 }
 else if ACC < LOW {
   result: 0
 }
 else {
   result: [adjust 2d8+3 with MOD]
 }
}

function: adjust DMG:n with MOD:n {
 if MOD = 1 {
   result: DMG / 2
 }
 else if MOD >= 5 {
   result: 2 * DMG
 }
 else {
   result: DMG
 }
}

ACC: 1d20 + 1d12 - 1
MOD: 1d6
HIGH: 1d10 + 18
LOW: 1d10 + 8

output [damage ACC MOD HIGH LOW] named "Damage Output fickling"

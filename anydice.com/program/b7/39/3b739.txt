function: specialbestof S:s max T:n {
  HIGHEST_VALID: 0
  HIGHEST_COUNT: 0
  BEST_RESULT: 0

  loop I over S {
    if I <= T {
      COUNT: [count I in S]
      MODIFIED_VALUE: I + (COUNT - 1)

      if MODIFIED_VALUE > BEST_RESULT {
        BEST_RESULT: MODIFIED_VALUE
      }
    }
  }

  result: BEST_RESULT
}

function: bestof S:s max T:n  {
  HIGHEST_VALID: 0
  loop I over S {
    if I <= T & I > HIGHEST_VALID { HIGHEST_VALID: I }
  }
  result: HIGHEST_VALID
}


function: verybestof S:s max T:n {
  HIGHEST_VALID: 0
  HIGHEST_COUNT: 0
  BEST_RESULT: 0
  COUNTTEN: [count 10 in S]

  loop I over S {
    if I <= T {
      COUNT: [count I in S]
      MODIFIED_VALUE: I + (COUNT - 1)

      if MODIFIED_VALUE > BEST_RESULT {
        BEST_RESULT: MODIFIED_VALUE
      }
    }
  }

  FINAL_RESULT: BEST_RESULT - COUNTTEN
  if FINAL_RESULT < 0 { FINAL_RESULT: 0 }

  result: FINAL_RESULT
}




N: 1d10
M:  5

\ output [specialbestof N max M] named "normal 1"\

output [verybestof N max M] named "special 1"

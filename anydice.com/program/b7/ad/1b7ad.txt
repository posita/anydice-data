\
 HEART
 0: Critical Fail
 1: Fail
 2: Success at cost
 3: Clean success
 4: Critical success!
\

function: heart ROLL:s {
 if 1@ROLL = 10{
  result: 4} 
 if 1@ROLL >= 8{
  result: 3} 
 if 1@ROLL >= 6{
  result: 2} 
 if 1@ROLL >= 2{
  result: 1}
 result: 0
}

loop P over {1..4} {
 output [heart Pd10] named "[P]d"
}

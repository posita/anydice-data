TARGET: 18

loop N over {2..4} {
  ROLL: Nd20
  FIRST: 1@ROLL
  SECOND: 2@ROLL
  
  output FIRST <= TARGET & SECOND <= TARGET & (FIRST = SECOND) named "Critical Success ([N]d20)"
  output FIRST <= TARGET & SECOND <= TARGET & (FIRST != SECOND) named "Full Success ([N]d20)"
  output (FIRST <= TARGET & SECOND > TARGET) | (FIRST > TARGET & SECOND <= TARGET) named "Partial Success ([N]d20)"
  output FIRST > TARGET & SECOND > TARGET & (FIRST != SECOND) named "Failure ([N]d20)"
  output FIRST > TARGET & SECOND > TARGET & (FIRST = SECOND) named "Critical Failure ([N]d20)"
}

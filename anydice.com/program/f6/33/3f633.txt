function: bracketed range ROLL:n {
  PART: ROLL >= 6
  FULL: ROLL >= 9
  CRIT: ROLL >= 12
  result: PART + FULL + CRIT
}

function: target ability ROLL:n TARGET:n {
  ABILITY: ROLL >= TARGET - 1
  FULL: ROLL >= TARGET
  result: ABILITY + FULL
}

\
output [bracketed range [lowest 2 of 3d6]] named "Disadvantage"
output [bracketed range 2d6] named "Standard"
output [bracketed range [highest 2 of 3d6]] named "Advantage"

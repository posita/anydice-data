function: biggest set of SEQ:s {
  C: 1
  loop N over {1..6} {
    if ( (SEQ = C) <= (SEQ = N) ) { C: N }
  }
  R: C
  loop N over {1..(SEQ = C)} {
    R: (R * 10) + C
  }
  R: R - C
  R: R / 10
  result: R
}

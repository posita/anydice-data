function: dupes in DICE:s {
  D: 0
  if(1@DICE = 2@DICE){
 		D: D + 1
  }
  loop X over {3..#DICE} {
    if ((X-1)@DICE = X@DICE & (X-2)@DICE != (X-1)@DICE) { 
		D: D + 1
	}	
  }
  result: D
}
function: triples in DICE:s {
  D: 0
  loop X over {3..#DICE} {
    if ((X-1)@DICE = X@DICE & (X-2)@DICE = (X-1)@DICE) { 
		D: D + 1
	}	
  }
  result: D
} 
 
 
 
output [dupes in 8d6] named "doubles"
output [triples in 8d6] named "triples"

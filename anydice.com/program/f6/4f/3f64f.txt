\output 4d{0:LOSE, 1:WIN}\

function: target MIN with ROLLS:n {
  LOSE: MIN - 1
  WIN: 501 - MIN
  result: 1@((ROLLS)d{0:LOSE, 1:WIN})
}

\
output [target 320 with 4] named "yokukara"
output [target 350 with 4] named "unique kinesis"
output [target 450 with 4] named "double kinesis/magic"

output [target 500 with 4] named "500"

function: ROLLN:n ROLLD:d reroll N:n{
  if N != 0 {
    if ROLLN = 1 {
      result: 1
    }
    result: [ROLLD ROLLD reroll (N-1)]
  }
  result: 0
}

loop N over {0..4} {
  output [[target 450 with 4] [target 450 with 4] reroll N+1] named "[N] alt double kinesis/magic"
}

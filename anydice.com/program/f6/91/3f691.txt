function: explode ROLLEDVALUE:n {
 if ROLLEDVALUE = 6 { result: 1 + [explode d6] }
 else if ROLLEDVALUE > 2 { result: 1}
 else if ROLLEDVALUE = 2 { result: 0}
 result: -1 
}

output [highest of [explode d6] and 0] named "1M"
loop N over {1..11} {
  X: N+1
  W: [explode d6]
  output [highest of W and 0] + (Nd{0,0,1,1,1,1})*(W > -1) named "[X]M"
}

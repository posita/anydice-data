function: scroll ROLL:n {
 if ROLL <= 10 { result: 10 }
 if ROLL <= 20 { result: 14 }
 if ROLL <= 30 { result: 18 }
 if ROLL <= 50 { result: 24 }
 if ROLL <= 70 { result: 32 }
 if ROLL <= 84 { result: 40 }
 if ROLL <= 95 { result: 80 }
 result: 0
}

function: scrolls NUM:n {
 TOTAL: 0
 loop COUNT over {1..NUM} {TOTAL: TOTAL+[scroll 1d100]}
 result: TOTAL
}

function: dupes in DICE:s of RESULTS {
  D: 0 
  H: 0
  loop X over {2..#DICE} {
    if (((X-1)@DICE = X@DICE)&(X@DICE >= RESULTS)) { D: D + 1}
    else {if D>H {H:D D:0}}
  }
  result: H
}

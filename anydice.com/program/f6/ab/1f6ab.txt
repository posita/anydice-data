function: attack xbow ROLL:n plus BONUS:n for DAMAGE:d {
  if ROLL >= 19 {
    result: DAMAGE * 2
  } else if ROLL = 1 {
    result: 0
  }
}

function: attack longbow ROLL:n for DAMAGE:d {
  if ROLL = 1 {
    result: 0
  } else if ROLL = 20 {
    result: DAMAGE * 3
  } else {
    result: DAMAGE
  }
}

function: attack whip ROLL:n for DAMAGE:d {
  if ROLL = 1 {
    result: 0
  } else if ROLL = 20 {
    result: DAMAGE * 2
  } else {
    result: DAMAGE
  }
}

  named "Longbow, Composite"

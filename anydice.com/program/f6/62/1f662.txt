function: adv ROLL:s {
  if 1@ROLL <  2@ROLL { result: (1@ROLL - 1)*10 + (2@ROLL - 1) }
  if 1@ROLL >= 2@ROLL { result: (2@ROLL - 1)*10 + (1@ROLL - 1) }
}

function: disadv ROLL:s {
  if 1@ROLL >= 2@ROLL { result: (1@ROLL - 1)*10 + (2@ROLL - 1) }
  if 1@ROLL <  2@ROLL { result: (2@ROLL - 1)*10 + (1@ROLL - 1) }
}
output [adv 2d10] named "adv"
output d100-1 named "d100"
output [disadv 2d10] named "disadv"

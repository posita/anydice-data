NORMAL: 7
HARD: 11
EXTREME: 15

function: evaluate ROLL:s {
 if (ROLL < NORMAL) { result: 0 }
 else {
  if (ROLL < HARD) { result: 1 }
  else {
   if (ROLL < EXTREME) { result: 2 }
   else {
    result: 3
   }   
  }
 }
}

\OUTPUT LIST\
\
loop N over {1..5} {
 output [evaluate [highest 1 of Nd4]+d6] named "[N]d4"
 output [evaluate [highest 1 of Nd6]+d6] named "[N]d6"
 output [evaluate [highest 1 of Nd8]+d6] named "[N]d8"
 output [evaluate [highest 1 of Nd10]+d6] named "[N]d10"
 output [evaluate [highest 1 of Nd12]+d6] named "[N]d12"
}
\

loop N over {0..3} {
 output [evaluate d6+d6+N] named "d6+d6+[N]"
 output [evaluate d6+d8+N] named "d6+d8+[N]"
 output [evaluate d6+d10+N] named "d6+d10+[N]"
 output [evaluate d6+d12+N] named "d6+d12+[N]"
}

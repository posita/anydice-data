MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

PROF_BONUS: 2
ABILITY_BONUS: 3
ITEM_BONUS: 1
OTHER_BONUS: 0

ATTACK: PROF_BONUS + ABILITY_BONUS + ITEM_BONUS + OTHER_BONUS
loop AC over {4..30} {
 output [attack d20 vs AC] named "[AC]"
}

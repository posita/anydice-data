W: {0:3, 1:2, 2:1}
D: {0:4, -1, 1}

function: roll POOL:s with NIX:n {

  BLANK: (POOL = 0)
  MAJOR: (POOL = 2)
  MINOR: (POOL = 1) + NIX
  if MINOR < 0 { MINOR: 0 }
  OUTCOME: MINOR + MAJOR
  if OUTCOME < 0 { OUTCOME: 0 }

  if MINOR >= 2 { result: 2 }
  if OUTCOME >= 2 { result: 2 }
  result: OUTCOME
}

loop N over {1..3} {
  output [roll NdW with 1dD] named "ROLLING [N] DICE WITH 1 NIX"
}


\
KEY:
0  = Fail
1  = Minor Success
2  = Major Success
3+ = Critical

REMNANTS_SKILL_POINTS: 1
TARGET_NUMBER: 7

function: nwod R:n again N:n {
 if N >= R { result: 1 + [nwod R again d10] }
 
 if ((REMNANTS_SKILL_POINTS > 0) & (N < TARGET_NUMBER) &((N + REMNANTS_SKILL_POINTS) >= TARGET_NUMBER))
 {
   REMNANTS_SKILL_POINTS : REMNANTS_SKILL_POINTS - (TARGET_NUMBER - N)
   result: 1
 } else
 {
   result: N >= TARGET_NUMBER
 }
}

NWOD: [nwod 10 again d10]

loop N over {1..5} {
 output [lowest of 10 and NdNWOD] named "[N]d"
}

function: attack once ROLL:n DMG:d SNEAK:d BONUS:n {
  if ROLL <= 0 {
    result: 0
  }
  if ROLL >= 20 {
    result: 2 d (DMG + SNEAK) + BONUS
  }
  if ROLL + PROF + DEX >= AC {
    result: 1 d (DMG + SNEAK) + BONUS
  }
  result: 0
}

function: attack twice ROLLR:n ROLLL:n DMG:d SECONDBONUS:n {
  if ROLLR <= 0 | ROLLR + PROF + DEX < AC {
    result: [attack once ROLLL DMG SNEAK SECONDBONUS]
  }
  result: [attack once ROLLR DMG SNEAK DEX] + [attack once ROLLL DMG 0 SECONDBONUS]
}

function: prof LVL:n {
  result: ((LVL - 1) / 4) + 2
}
function: sneak LVL:n {
  result: ((LVL + 1) / 2)
}

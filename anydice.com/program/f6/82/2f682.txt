DIE: d6
TARGET: {5,6}
DEPTH: 3
set "maximum function depth" to DEPTH
MAXDICE: 10

function: nightares R:n again N:n {
 if N >= R { result: 1 + [nightares R again DIE] }
 result: N >= TARGET
}

NIGHTARES: [nightares 6 again DIE]

loop N over {1..MAXDICE} {
 output [lowest of DEPTH+N and NdNIGHTARES] named "[N]d"
}

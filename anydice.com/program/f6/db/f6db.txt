function: biggest set of SEQ:s dsize SIZE:n {
  C: 1
  loop N over {1..SIZE} {
    if ( (SEQ = C) <= (SEQ = N) ) { C: N }
  }
  R: C
  loop N over {1..(SEQ = C)} {
    R: (R * 100) + C
  }
  R: R - C
  R: R / 100
  result: R
}

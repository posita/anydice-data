function: reroll R:n under N:n this D {
   if R < N { result: D } else {result: R}
}

function: hit H:n mod M crit C dmg D bonus B extra E ac A {
   if H >= C {result: D+D+B+E}
   else if H+M >= A {result: D+B} 
   else {result: 0}
}

function: save S:n mod M dmg D bonus B dc DC {
   if S+M < DC {result: D+B}
   else {result: 0}
}

\Great Weapon d12 for greataxes\
GWA: [reroll 1d12 under 3 this d12] 

\Great Weapon d6 for swords & mauls\
GWS: [reroll 1d6 under 3 this d6]

\Adv & Dis\
DA: [highest of 1d20 and 1d20]
DD: [lowest of 1d20 and 1d20]

\hit = die to hit
mod = modifier to attack
crit = critical hit value, 20 for most
dmg = damage dice on a hit
bonus = modifier to damage
extra = extra damage on a crit, 0 for most
ac = target's armor class\


RD: [hit d20 mod 8 crit 20 dmg 2d6 bonus 5 extra 0 ac 18]
RC: [hit d20 mod 3 crit 20 dmg 2d6 bonus 13 extra 0 ac 18]
B: [hit DA mod 4 crit 20 dmg 2d6 bonus 16 extra 0 ac 18]

output RD+RD+RD named "Two-Weapon Fighting"
output RC+RC+RC 
output B+B+B

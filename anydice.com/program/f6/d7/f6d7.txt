function: biggest set of SEQ:s dsize SIZE:n {
  C: 1
  loop N over {1..SIZE} {
    if ( (SEQ = C) <= (SEQ = N) ) { C: N }
  }
  R: C
  loop N over {1..(SEQ = C)} {
    R: (R * 100) + C
  }
  R: R - C
  R: R / 100
  result: R
}


loop N over {101,202,303,404,505,606,10101,20202,30303,40404,50505,60606,1010101} {
   output [biggest set of 3d6 dsize 6] >= N named "[N]"
}

\ sort Foudre \

E_DEF: 15   \ ennemi def \
E_DEX: 2    \ ennemi dex \
E_RD: 0     \ ennemi RD \

AM: 11      \ attaque magique \
DEX: -1     \ dex lanceur \

DM: 4d6+4   \ DM du sort \

AM_S: 1d20

S_CRIT: AM_S >= 20  \ reussite critique \
E_CRIT: AM_S <= 1   \ echec critique \


ESQUIV: 1d20 + E_DEX >= (12 + DEX)   \ esquive reussie ? \
DM: DM * (1 + S_CRIT) * (1 - E_CRIT) \ dommages avec critique \
DM: DM / (1 + ESQUIV)                \ dommages avec esquive \
DM: [highest of (DM-E_RD) and 0]     \ reduction de dommages \

output ((AM + AM_S) >= E_DEF) * DM

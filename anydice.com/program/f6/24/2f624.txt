set "position order" to "lowest first"
loop SKILL over {3..18} {
  loop BONUS over {-5..5} {
    TARGET: SKILL + BONUS
    FIRST: 1@2d20
    SECOND: 1@2d20

    output FIRST <= TARGET & SECOND <= TARGET & (FIRST = SECOND) named "Success Critical (2d20 target [SKILL] + [BONUS] = [TARGET])"
    output FIRST <= TARGET & SECOND <= TARGET & (FIRST != SECOND) named "Success Full (2d20 target [SKILL] + [BONUS] = [TARGET])"
    output (FIRST <= TARGET & SECOND > TARGET) | (FIRST > TARGET & SECOND <= TARGET) named "Success Partial (2d20 target [SKILL] + [BONUS] = [TARGET])"
    output FIRST > TARGET & SECOND > TARGET & (FIRST != SECOND) named "Failure (2d20 target [SKILL] + [BONUS] = [TARGET])"
    output FIRST > TARGET & SECOND > TARGET & (FIRST = SECOND) named "Failure Critical (2d20 target [SKILL] + [BONUS] = [TARGET])"
  }
}

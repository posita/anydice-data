LONGBOW: 1
WHIP: 2
HAND_CROSSBOW: 3

BAB: 3
AC: 10

function: attack with WEAPON:n roll ROLL:n plus BONUS:n for DAMAGE:d {
  TOTAL: ROLL + BONUS

  if ROLL = 1 | TOTAL < AC {
    result: 0
  } else if WEAPON = 1 & ROLL = 20 {
    result: DAMAGE * 3
  } else if WEAPON = 2 & ROLL = 20 {
    result: DAMAGE * 2
  } else if WEAPON = 3 & ROLL >= 19 {
    result: DAMAGE * 2
  } else {
    result: DAMAGE
  }
}

\ Modifiers: Weapon Focus (Longbow), Point-Blank Shot, Precise Shot, Rapid Shot, Sacred Weapon (Longbow) \
output
  [attack with LONGBOW roll d20 plus BAB+4+1-2 for d10+1+1] + [attack with LONGBOW roll d20 plus BAB+4+1-2 for d10+1+1]
  named "Longbow, Composite"

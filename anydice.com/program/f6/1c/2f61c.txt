loop TARGET over {3..18} {
  N: 2
  FIRST: 1@2d20
  SECOND: 2@2d20
  
  output FIRST <= TARGET & SECOND <= TARGET & (FIRST = SECOND) named "Critical Success ([N]d20 [FIRST] target [TARGET])"
  output FIRST <= TARGET & SECOND <= TARGET & (FIRST != SECOND) named "Full Success ([N]d20 target [TARGET])"
  output (FIRST <= TARGET & SECOND > TARGET) | (FIRST > TARGET & SECOND <= TARGET) named "Partial Success ([N]d20 target [TARGET])"
  output FIRST > TARGET & SECOND > TARGET & (FIRST != SECOND) named "Failure ([N]d20 target [TARGET])"
  output FIRST > TARGET & SECOND > TARGET & (FIRST = SECOND) named "Critical Failure ([N]d20 target [TARGET])"
}

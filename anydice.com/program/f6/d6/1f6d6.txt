function: attack with WEAPON:n roll ROLL:n plus BONUS:n for DAMAGE:d {
  TOTAL: ROLL + BONUS

  if ROLL = 1 | TOTAL < AC {
    result: 0
  } else if WEAPON = 1 & ROLL = 20 {
    result: DAMAGE * 3
  } else if WEAPON = 2 & ROLL = 20 {
    result: DAMAGE * 2
  } else if WEAPON = 3 & ROLL >= 19 {
    result: DAMAGE * 2
  } else {
    result: DAMAGE
  }
}

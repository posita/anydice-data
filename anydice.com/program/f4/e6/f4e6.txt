\d10 Dice Pool\

FOC:10  \ <---- set your focus\
TN:6   \ <---- set you target number for die success\

function: D:s target T:n focus F:n {
   if (D>=F) {result:3}
   else if (D>=T){result: 1}
   else {result:0}
}

loop S over {1..5} {
  output Sd[d10 target TN focus FOC] named "[S]d10 Pool Target [TN] Focus [FOC]"
}

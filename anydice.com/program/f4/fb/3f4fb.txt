function: roll D:s under P {

if P = 18 {
S: 12
C: 5
}

if P = 16 | P = 17
{
S: 10
C: 3
}

if P > 12 & P < 16 {
S: 7
C: 2
}

if P > 8 & P < 13 {
S: 5
C: 2
}

if P < 9 & P > 5 {
S: 3
C: 1
}

result: (D <= P) + (D <= S) + (D <= C)
}

if P < 6 {
S: 2
C: 1
}

loop P over {3..18} {
output [roll 1d20 under P] named "[P]"
}

function: first{result: 134}
function: second {result: 134}

function: rf X:n {
T: X
if X < 2{
T: 1d4
}
result: T
}

function: rs X:n {
T: X
if X < 2{
T: 1d6
}
result: T
}

function: X:n divide by Y:n {
T: X/Y
if X > T*Y {
T: T + 1
}
result: T
}

function: filter X:n {
FUP:0
SUP:0
if X >= [first] {FUP: 1}
if X >= [second] {SUP: 2}
result: FUP+SUP
}

function: die size from X:n {
if X > 16 {result: 12}
if X > 12 {result: 10}
if X > 8 {result: 8}
if X > 4 {result: 6}
result: 4
}

function: type one {
F: [first]
S: [second]
NUMBER: [F divide by 4]
SIZE: [die size from [S divide by 5]]
result: [filter NUMBERdSIZE]
}

function: type two {
F: [first]
S: [second]
SUM: F+S
BONUS: [SUM divide by 4]
result: 1d100+BONUS\[filter 1d100+BONUS]\
}

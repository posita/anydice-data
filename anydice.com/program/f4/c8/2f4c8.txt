
function: check DC and MOD and ADV:n {
  if ADV = 1 {
    result: ( [highest 1 of 2d20] + MOD) > DC
  } else {
  result: (1d20 + MOD) > DC}
}

loop N over {1..11}{
 output [check 15 and N and 0] named "DC 15 mod+[N]"
}

loop N over {1..11}{
 output [check 15 and N and 1] named "DC 15 mod+[N] with advantage"
}

function: bleem A:n and B:n {
  AD: [absolute A - 10]
  BD: [absolute B - 10]
  if AD > BD {
    result: A
  } else if BD > AD {
    result: B
  }
  result: d{}
}

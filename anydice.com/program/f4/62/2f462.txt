\**** DnD Modifiable Attack Calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: sixgwfight ROLLX:n {
 if ROLLX <= 2 { result:d6 }
 result: ROLLX
}

function: twelvegwfight ROLLX:n {
 if ROLLX <= 2 { result:d12 }
 result: ROLLX
}

\loop through enemy AC from 11 to 16\
ENEMYAC: d{11..16}

ATTACKMOD: 3
DMGBONUS: 0
PROFICIENCY: 2
ACCBONUS: 0
CRITROLL: 20
DAMAGE: 1d8 + 2d6 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d8 + 4d6 + ATTACKMOD + DMGBONUS

output [attackroll [highest 1 of 2d20] vs ENEMYAC] named "Rogue 3 Assassin"

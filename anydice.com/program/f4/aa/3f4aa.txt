function: halfonsave ROLL:n DC:n DAMAGE {
  if (ROLL>= DC) { result: DAMAGE/2 }
  result: DAMAGE
}

function: dtwenty ROLL:n {
  if ROLL=20 { result: 99 }
  if ROLL=1 { result: -99 }
  result: ROLL
}

function: attackroll ROLL:n ATTBON AC DMGDIE DMGBON {
  if ROLL=20 { result: 2dDMGDIE+DMGBON }
  if ROLL=1 { result: 0 }
  if ROLL+ATTBON>=AC { result: DMGDIE + DMGBON }
  result: 0
}

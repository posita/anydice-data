
ATK:  1d20  \ attack roll: use 1@2d20 for advantage or 2@2d20 for disadvantage \
AC:     15  \ target AC: attack hits if ATK + HIT >= AC \

function: attack ROLL:n with HIT:n vs AC:n {
  \ return the number of times damage is applied on an attack roll of ROLL \
  if ROLL <= 1 { result: 0 }     \ natural 1 is always a miss \
  if ROLL >= CRIT { result: 2 }  \ critical hit deals damage damage twice \
  result: ROLL + HIT >= AC
}

function: multiattack N:n times with HIT:n vs AC:n for DMG:d plus BONUS:d {
  result: [roll [attack ATK with HIT vs AC] to multiattack N times for DMG plus BONUS]
}
function: roll FIRSTHIT:n to multiattack N:n times for DMG:d plus BONUS:d {
  if N <= 0 {
    result: 0
  }
  if FIRSTHIT > 0 {
    DAMAGE: FIRSTHIT d (DMG + BONUS)
    loop I over {2..N} {
      DAMAGE: DAMAGE + [attack ATK with HIT vs AC] d DMG
    }
    result: DAMAGE
  } else {
    result: [roll [attack ATK with HIT vs AC] to multiattack N-1 times for DMG plus BONUS]
  }
}

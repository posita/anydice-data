PLAYERS: 2
ROUNDS_G: 5
ROUNDS_E: 3
DICE: 8

TOTAL_ONES: 0
CURRENT_ROUND: 0
ROUNDS: ROUNDS_G + ROUNDS_E + 1

loop ONES_THIS_ROUND over {1..ROUNDS} {
 ONES_THIS_ROUND: 0
 if CURRENT_ROUND = 0 {
  ONES_THIS_ROUND: 4 - (1d6/2)
  TOTAL_ONES: TOTAL_ONES + ONES_THIS_ROUND
  }
 else {
  loop PLAYERS over {1:PLAYERS} {
   ONES_THIS_ROUND: ONES_THIS_ROUND + [count {1} in DICEd6]
  }
  if CURRENT_ROUND <= ROUNDS_G {
   TOTAL_ONES: TOTAL_ONES + ONES_THIS_ROUND + 1
   }
  else {
   TOTAL_ONES: TOTAL_ONES + ONES_THIS_ROUND + 2
   } 
 }
 output ONES_THIS_ROUND named "Ones for Round [CURRENT_ROUND]"
 output TOTAL_ONES named "Round [CURRENT_ROUND] Total"
 CURRENT_ROUND: CURRENT_ROUND + 1
}

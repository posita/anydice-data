function: highest N:n of A:s B:s {
    result: {1..N}@[sort {A, B, C}]
}

function: best FIRST:d or SECOND:d {
  if (FIRST > SECOND) {
    result: FIRST
  } else {
    result: SECOND
  }
}

output 1d20
output [highest 1 of 2d20]
output [best 1d20 or 1d20]
output 3d6
output [highest 3 of 4d6]
output [best 3d6 or 3d6]

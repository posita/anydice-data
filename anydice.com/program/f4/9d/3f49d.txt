function: A:n mod B:n {
 result: A-(A/B)*B
}

function: successes from pickle A:n {
 R: [A mod 100]
 S: A / 100
 if R > 50 {result: S+1}
 if R < -50 {result: S-1}
 result: S

}

function: advantages from pickle A:n {
  R: [A mod 100]
  if R > 50 {result: R-100}
  if R < -50 {result: R+100}
  result: R
}

function: link damage A:n {
 S: [successes from pickle A]
 ESOAK: [highest of 0 and SOAK-LAZAXPIERCE]
 R: [highest of [advantages from pickle A] and 0]
 if S < 1 {result: 0}
 if S < ESOAK - LAZAXDAMAGE + 1 {result:0}
 C: [lowest of R and 6]
 result: (S+LAZAXDAMAGE-ESOAK)*(1+C/2)
}


function: plasma damage A:n {
 S: [successes from pickle A]
 R: [advantages from pickle A]
 if S < 1 {result:0}
 if S < SOAK - 9 {result:0}
 if R > 1 {result: 20 + S - SOAK - SOAK}
 result: 10 + S - SOAK
}

function: pistol damage R:n {
 S: [successes from pickle R]
 A: [advantages from pickle R]
 BASEDAMAGE: 6
 PIERCE: 0
 ESOAK: [highest of SOAK - PIERCE and 0]
 if S<1 {result:0}
 DAMAGE: BASEDAMAGE + S - ESOAK
 result: [highest of DAMAGE and 0]
}

function: dual pistol damage R:n {
 S: [successes from pickle R]
 A: [advantages from pickle R]
 BASEDAMAGE: 6
 PIERCE: 0
 ESOAK: [highest of SOAK - PIERCE and 0]
 if S<1 {result:0}
 DAMAGE: BASEDAMAGE + S - ESOAK
 if A > 1 {result: [highest of DAMAGE * 2 and 0]}
 result: [highest of DAMAGE and 0]
}

function: roll ROLL:n at plus ATTACKBONUS:n for CRITABLEDAM:d plus HITDAM:d {
    if ROLL >= CRITRANGE {
    result: CRITABLEDAM + CRITABLEDAM + HITDAM
    }
        if ROLL = 1 {
        result: 0
        }
            if ROLL >= AC - ATTACKBONUS {
            result: CRITABLEDAM + HITDAM
            }
                else {
                result: 0
                }
}


     \Adjust AC and CRITRANGE here\
AC: 16
CRITRANGE: 20


function: explodefive N:n {
 if N > 4 { result: N + [explode d6] }
 result: N
}
set "explode depth" to 2


FB: [roll 1d20 at plus 9 for 2d10 plus 0]
output FB named "Fire Bolt"

SB: [roll 1d20 at plus 9 for 2d[explodefive 1d6] plus 0]
output SB named "Sorcerous Burst Explode 5 or 6"

SBB: [roll 1d20 at plus 9 for 2d[explode 1d6] plus 0]
output SBB named "Sorcerous Burst Explode 6"

EB: [roll 1d20 at plus 9 for 1d10 plus 0]

PLAYERS: 2
SCENES_G: 5
SCENES_E: 3
ROUNDS:SCENES_G + SCENES_E
TOTAL_ONES: 1
CURRENT_ROUND: 0

loop ONES_THIS_ROUND over {1..ROUNDS} {
 CURRENT_ROUND: CURRENT_ROUND+1
 if CURRENT_ROUND > SCENES_G {
 ONES_THIS_ROUND: [count {1} in 8d6]+[count {1} in 8d6]+2
 TOTAL_ONES: TOTAL_ONES + ONES_THIS_ROUND
 output TOTAL_ONES
 output CURRENT_ROUND named "Current round is [CURRENT_ROUND]"
}
 else {
 ONES_THIS_ROUND: [count {1} in 8d6]+[count {1} in 8d6]+1
 TOTAL_ONES: TOTAL_ONES + ONES_THIS_ROUND
 output TOTAL_ONES
 output CURRENT_ROUND named "Current round is [CURRENT_ROUND]"
}
}

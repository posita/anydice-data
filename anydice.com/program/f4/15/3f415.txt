function: stepup HOPE:n and FEAR:n add TRAIT:n {
 TARGET: 6 - TRAIT
 HIGH: HOPE
 if FEAR > HOPE { HIGH: FEAR }
 HOPEHIGH: HOPE >= FEAR
 FEARHIGH: FEAR > HOPE
 PASS: HIGH >= TARGET
 if PASS & HOPEHIGH { result: 2 }
 if PASS & FEARHIGH { result: 1 }
 if !PASS & HOPEHIGH { result: 0 }
 if !PASS & FEARHIGH { result: -1 }
 result: -5
}

loop N over {0..3}{
 output [stepup 1d6 and 1d6 add N] named "+[N]"
}

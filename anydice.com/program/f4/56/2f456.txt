\**** Great Weapon Fighting modifiable Calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: sixgwfight ROLLX:n {
 if ROLLX <= 2 { result:d6 }
 result: ROLLX
}

function: twelvegwfight ROLLX:n {
 if ROLLX <= 2 { result:d12 }
 result: ROLLX
}

ENEMYAC: 13

ATTACKMOD: 3
DMGBONUS: 3
PROFICIENCY: 2
ACCBONUS: 0
CRITROLL: 19
DAMAGE: [sixgwfight d6] + [sixgwfight d6] + ATTACKMOD + DMGBONUS
CRITDAMAGE: [sixgwfight d6] + [sixgwfight d6] + [sixgwfight d6] + [sixgwfight d6] + ATTACKMOD + DMGBONUS

output [attackroll 1d20 vs ENEMYAC] named "Fighter 3 GWF Champion"

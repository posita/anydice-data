function: WOD ROLL:s {
  SUCCESSES: [count {8..10} in ROLL]    \  /* compte 8,9,10 comme succÃ¨s */
  TENS: [count 10 in ROLL]              \  /* compte combien de 10 */
  ONE: [count 1 in ROLL]                \  /* compte combien de 1 */
  result: SUCCESSES + TENS - ONE        \  /* chaque 10 ajoute +1 succÃ¨s (donc 2 au total) et chaque 1 retire 1 succÃ¨s */
}

loop DICE over {1..20} {
  output [WOD DICE d10] named "[DICE]d10"
}

function: wackpool A:s
{
  NEW: {}
  O: [count {1} in A]

  if O > 0
  { 
   loop N over {O+1..#A}
   {
    if N@A>1
    {
    NEW: {NEW, N@A}
    }
   }
   result: {1,2} @NEW
  }
  else
  {
   result: {1,2} @A
  }
}

loop N over {4,6,8}
{
 output [wackpool 5dN] named "Wackpool 5d[N]"
 output [highest 2 of 5dN] named "Regular 5d[N]"
}

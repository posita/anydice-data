ATTDMG: 4d12
CRTDMG: 8d12
CRTRNG: 18
AC: 20
TOHIT: 5+6

function: hitit RLL:n {
if RLL + TOHIT < AC { result: 0 }
else if RLL >= CRTRNG { result: ATTDMG+CRTDMG }
else { result: ATTDMG }
}

output [hitit d20] named "Normal"
output [hitit [highest 1 of 2d20]] named "Advantage"
output [hitit [lowest 1 of 2d20]] named "Disadvantage"

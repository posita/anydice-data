function: count_matches NUM:n DICE:d TARGETS:s {
  result: [count {1,2,3} in DICE]
}

function: is_success COUNT:n {
  if COUNT >= 2 { result: 3 }
  else if COUNT == 1 { result: 2 }
  else { result: 1 }
}

function: is_critical COUNT:n DICE:d {
  if COUNT >= 2 {
    if [count 2 in [count {1,2,3} in DICE]] >= 2 { result: 4 }
    else { result: 3 }
  }
  else { result: is_success COUNT }
}

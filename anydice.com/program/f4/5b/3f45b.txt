set "explode depth" to 2

X: {{1, 1}, {1, 2}, {1, 3}, {1, 4}, {1, 5}, {2, 2}, {2, 3}, {2, 4}, {2, 5}, {3, 3}, {3, 4}, {3, 5}, {4, 4}, {4, 5}, {5, 5}}
Y: {1, 2, 3, 4, 5, 6}

function: roll levels L:s {
 loop N over L {
 if L = 1 {
  result: [explode 1d4]
 } else if L = 2 {
  result: [explode 1d6 + 1]
 } else if L = 3 {
  result: [explode 1d8 + 2]
 } else if L = 4 {
  result: [explode 1d10 + 3]
 } else if L = 5 {
  result: [explode 1d12 + 4]
 }
 result: 0
}
}

loop D over Y {
 loop R over X {
  loop S over X {
   output [roll level R] + [roll level S] - Dd4 named "level [D] with skills [R] and [S]"
  }
 }
}

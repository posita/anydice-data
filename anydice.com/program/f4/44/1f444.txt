AC: {10..19}
PROFICIENCY: 3

function: gwm DMG:d crit CRIT_DMG:d strength STR:n roll ROLL:n{
    DMG_WITH_BONUS: DMG + STR + 10
	
    if ROLL = 1 {
        result: 0
    }
    if ROLL = 20 {
        result: DMG_WITH_BONUS + CRIT_DMG
    }
    if ROLL + STR + PROFICIENCY - 5 >= AC {
        result: DMG_WITH_BONUS
    }
    result: 0
}

function: normal DMG:d crit CRIT_DMG:d strength STR:n roll ROLL:n{
    DMG_WITH_BONUS: DMG + STR
	
    if ROLL = 1 {
        result: 0
    }
    if ROLL = 20 {
        result: DMG_WITH_BONUS + CRIT_DMG
    }
    if ROLL + STR + PROFICIENCY >= AC {
        result: DMG_WITH_BONUS
    }
    result: 0
}


function: check DC and MOD and A {
  result: DC - (A + MOD) > 0
}
output [check 10 and 1 and d20] named "DC 10 mod 1 no advantage"

function: bleem A:n and B:n {
  AD: [absolute A - 10]
  BD: [absolute B - 10]
  if AD > BD {
    result: A
  } else if BD > AD {
    result: B
  }
  result: d{}
}

output [bleem d20 and d20] named "Rolling With Emphasis"

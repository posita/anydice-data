function: remove N:n from S:s {
  X:{}
  loop M over {1..#S}{
    if M!=N {X:{X,M@S}}
  }
  result: X
}

function: pickthree X:s w A:n w B:n w C:n{
  Y: [remove A from X]
  Z: [remove B from Y]
  result: {A@X,B@Y,C@Z}
}

function: pickth X:s{
  result: [pickthree X w d(#X) w d(#X-1) w d(#X-2)]
}

function: tokensum S:s{
  if ((1@S)*(2@S)*(3@S))=0 {
    result: 1@(S)
  }
  if S=9  {result:3}
  if S=12 {result:4}
  result:5
}

output [tokensum [pickth {1,1,1,3,3,3,4,4,4}]]

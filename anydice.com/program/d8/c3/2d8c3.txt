function: analyze DIE_A:n DIE_B:n {
  if (DIE_A = 1 & DIE_B = 1) {
    result: -2
  }
  if (DIE_A = 6 & DIE_B = 6) {
    result: +2
  }
  if (DIE_A <= 3 & DIE_B <= 3) {
    result: -1
  }
  if (DIE_A >= 4 & DIE_B >= 4) {
    result: 1
  }
  result: 0
}

function: roll BOON {
  D: 6
  if (BOON < 0) {
    DN: 2 - BOON
    DIE_A: DN@DNdD
    DIE_B: (DN-1)@DNdD
  } else {
    DN: 2 + BOON
    DIE_A: 1@DNdD
    DIE_B: 2@DNdD
  }
  result: DIE_A + DIE_B
}

output [roll -3] named "3 banes"
output [roll -2] named "2 banes"
output [roll -1] named "1 bane"
output [roll 0] named "normal"
output [roll 1] named "1 boon"
output [roll 2] named "2 boons"
output [roll 3] named "3 boons"

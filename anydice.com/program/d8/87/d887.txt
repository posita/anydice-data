SMITE: 0
ATTACK: 0

PSMITE:0
PATTACK:0

function: X:n pal {
  PATTACK: PATTACK + 1
  if X = 20 {
    PSMITE: PSMITE + 1
    result: 4d6 + 6d8 + 4
  }
  if PATTACK = 2 & !PSMITE{
    PSMITE: PSMITE + 1
    result: 2d6 + 3d8 + 4
  }
  result: 2d6 + 4
}

function: X:n palb {
  if X = 20 {
    result: 4d6 + 6d8 + 4
  }
  result: 2d6 + 3d8 + 4
}


function: X:n war {
  ATTACK: ATTACK+1
  if X >= 19 & !SMITE{
    SMITE:1
    result: 4d6  + 2d6 + 8d8 + 3 + 4 + 1
  }else{
    if X>= 19 {       
      result: 4d6  + 2d6 + 3 + 4 + 1
    }
  }
  if ATTACK=2 & !SMITE{
    SMITE:1
    result: 2d6 + 1d6 + 4d8 + 3 + 4 + 1
  }
  result: 2d6  + 1d6 + 3 + 4 + 1
}

output [d20 pal]  + [d20 pal] named "Paladin"
output [d20 war]  + [d20 war] named "Warlock"
output [d20 palb] + [d20 palb] named "Paladin Always Smite"

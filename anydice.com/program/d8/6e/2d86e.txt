function: highest number of matches in ROLL:s {
  MAX : 0
  loop I over ROLL {
    COUNT: ROLL = I
    if MAX < COUNT { MAX: COUNT }
   }
   result: MAX
}

function: customcount VALUES:s in SEQUENCE:s {
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
 }
 result: COUNT
}

output [highest number of matches in 5d5]
output [customcount {5} in 2d5]

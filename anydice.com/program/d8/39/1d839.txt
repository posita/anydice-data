function: repeat DICE:d NUMBER:n {
 SUM: 0
 loop N over {1..NUMBER} {
  SUM: DICE + SUM
 }
 result: SUM
}

function: points NUMBER:n {
 POINTS: NUMBER - 8
 if NUMBER > 13 { POINTS: POINTS + NUMBER - 13 }
 if NUMBER > 15 { POINTS: POINTS + NUMBER - 15 }
 if NUMBER > 17 { POINTS: POINTS + NUMBER - 17 }
 result: POINTS
}

function: sumpoints ROLL:s {
 SUM: 0
 loop N over {1..#ROLL} {
  SUM: SUM + [points N@ROLL]
 }
 result: SUM
}

function: sumhighest NUMBER:n of DICE:d plus OFFSET:n {
 SUM: 0
 loop N over {1..NUMBER} {
  SUM: SUM + [points N@DICE + OFFSET]
 }
 result: SUM
}

output 6d[highest 3 of 4d6] named "sum scores default roll"
output 6d(6d3) named "sum scores 6*6d3"
output 6d[highest 6 of 7d3] named "sum scores 6*(7d3k6)"
output [highest 6 of 7d(6d3)] named "sum scores 7*(6d3)k6"
output [highest 6 of 8d8] + 42 named "sum scores 8*(d8+7) keep 6"

output 6d[sumpoints {1..3}@4d6] named "point buy equivalent default roll"
output 6d[sumpoints {1..6}@6d3] named "point buy equivalent 6*6d3"
output 6d[sumpoints {1..6}@7d3] named "point buy equivalent 6*(7d3k6)"
output [sumhighest 6 of 7d(6d3) plus 0] named "point buy equivalent 7*(6d3)k6"

output [sumhighest 6 of 8d8 plus 7] named "point buy equivalent 8*(d8+7) keep 6"

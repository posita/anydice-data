function: remove N:n from S:s {
  X:{}
  loop M over {1..#S}{
    if M!=N {X:{X,M@S}}
  }
  result: X
}

function: pickthree X:s w A:n w B:n w C:n{
  Y: [remove A from X]
  Z: [remove B from Y]
  result: {A@X,B@Y,C@Z}
}

function: pickth X:s{
  result: [pickthree X w d(#X) w d(#X-1) w d(#X-2)]
}

function: tokensum S:s{
  if S=0 {result:0}
  if S=3|S=4 {result:1}
  if S=6|S=7|S=8 {result:2}
  if S=9  {result:3}
  if S=12 {result:4}
  result:5
}

function: toke X:s{
  result:[tokensum [pickth X]]
}

output [toke {0,0,0,3,3,3,4,4,4}]

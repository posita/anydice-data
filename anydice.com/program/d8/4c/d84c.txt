function: explode N:n {
 if N = 9 { result: N + [explode d{0..9}] }
 result: N
}

loop DICE over {1..3} {
set "maximum function depth" to DICE 
output [explode [highest 1 of DICE d{0..9}]]  named "d10 explodiert [DICE]Vorteil"
}


function: evaluate ROLL:s {
 SUCCESSES: ROLL >= 9
 if SUCCESSES >= 2 { result: SUCCESSES + (ROLL = 10) }
 if [count 1 in ROLL] { result: -1 }
 result: 0
}

MISS: 0
HIT: 1
CRITICALHIT: 2
function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}
function: attack {
 result: [damage for [attack 1d20 vs DEFENSE]]
}
function: advantage {
 result: [damage for [attack 1@2d20 vs DEFENSE]]
}
function: bless {
 result: [damage for [attack 1d20+1d4 vs DEFENSE]]
}
function: advbless {
 result: [damage for [attack 1@2d20+1d4 vs DEFENSE]]
}

MAXAC: 20
DEFENSE: d{MINAC..MAXAC}

ATTACK: PROF + STAT + POWER_HIT
MISSDAMAGE: 0 

HITDAMAGE: 2d6 + STAT + POWER_DMG
CRITICALDAMAGE: 4d6 + STAT + POWER_DMG

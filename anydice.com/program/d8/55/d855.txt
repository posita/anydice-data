\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 14-20\
ENEMYAC: d{14..20}

ATTACKMOD: 0
DMGBONUS: 0
PROFICIENCY: 4
ACCBONUS: 4
CRITROLL: 20
ATTACKTIMES: 6
CHARISMAMOD: 4
DAMAGE: [highest 2 of 6d6] + ATTACKMOD + DMGBONUS
CRITDAMAGE: [highest 4 of 8d6] + ATTACKMOD + DMGBONUS
output ATTACKTIMESd[attackroll 1d20 vs ENEMYAC] named " Empowered ScorchingRay Spell-slot 5"

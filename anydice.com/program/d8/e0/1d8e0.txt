 result: Ad[hit 1d6 H W R S D]
}

function: hit X:n H:n W:n R:n S:n D:n{
 if X>=H {result: [wound 1d6 W R S D]}
 result: 0
}

function: wound X:n W:n R:n S:n D:n{
 if X>=W {result: [save 1d6 R S D]}
 result: 0
}

function: save X:n R:n S:n D:n{
 if X-R<=S {result: D}
 result: 0
}

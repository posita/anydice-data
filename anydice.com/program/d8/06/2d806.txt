function: explode N:n on SIZE:n {
 if N >= SIZE {
  result: N + [explode d(N)]
 }
 result: N
}

function: reduce N:n from B:n below M:n by AMOUNT:n {
  if B<=M {
    result: N - AMOUNT
  }
  result: N
}

function: diminish N:n on SIZE:n {
 if N = SIZE & SIZE > 4 {
  result: N + [diminish d(SIZE - 2) on SIZE - 2]
 }
 result: N
}

function: reroll lowest of ROLL:s as REROLL:d if less than MIN:n {
  LOWEST: (#ROLL)@ROLL       \ the lowest die is sorted last \
  if LOWEST >= MIN {
    result: ROLL + 0
  } else {
    REST: {1..#ROLL-1}@ROLL  \ all but the lowest die \
    result: REST + REROLL
  }
}

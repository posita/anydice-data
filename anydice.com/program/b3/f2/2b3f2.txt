function: countmodif VALUES:s in SEQUENCE:s where RESULT:n {
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
 }
 if COUNT >= RESULT {
   result: COUNT-ESQUIVE+1
   }
 else{
   result: -1
 }
}

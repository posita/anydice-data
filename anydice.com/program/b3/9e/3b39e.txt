function: lowest N:n of A:s B:s C:s {
    SEQ: [sort {A, B, C}]
    L:#(SEQ)
    result: {L-N+1..L}@SEQ
}
function: highest N:n of A:s B:s {
    result: {1..N}@[sort {A, B}]
}
function: highest N:n of A:s B:s C:s {
    result: {1..N}@[sort {A, B, C}]
}
function: highest N:n of A:s B:s C:s D:s {
    result: {1..N}@[sort {A, B, C, D}]
}
function: lowest N:n of A:s B:s C:s D:s {
    SEQ: [sort {A, B, C, D}]
    L:#(SEQ)
    result: {L-N+1..L}@SEQ
}

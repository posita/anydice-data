\ This function returns +1 if A (attacker) wins, -1 if D (defender) wins: \
function: ATTR_A:n versus ATTR_D:n {
  \ make relabeled d10s where anything above the target becomes zero: \
  DA: 2d{1..ATTR_A, 0:(12-ATTR_A)}
  DD: 2d{1..ATTR_D, 0:(12-ATTR_D)}

  \ call a helper function to freeze the rolls: \
  result: [roll DA versus DD]
}

function: roll ROLL_A:s versus ROLL_D:s {
  SUCC_A: ROLL_A > 0
  SUCC_D: ROLL_D > 0

  \ higher number of successes wins: \
  if SUCC_A > SUCC_D { result: +1 }
  if SUCC_A < SUCC_D { result: -1 }
  if SUCC_D = 0 & SUCC_A = 0 { result: -1 }
  if ROLL_A = ROLL_D { result: -1 }

  \ otherwise do a lexicographic comparison: \
  result: (ROLL_A > ROLL_D) - (ROLL_A < ROLL_D)
}

function: reroll lowest of ROLL:s as REROLL:d if less than or equal to MIN:n {
  LOWEST: (#ROLL)@ROLL       \ the lowest die is sorted last \
  if LOWEST > MIN {
    result: ROLL + 0
  } else {
    REST: {1..#ROLL-1}@ROLL  \ all but the lowest die \
    result: REST + REROLL
  }
}

output 4d{-1,0,1} named "Standard Roll"
output [reroll lowest of 4d{-1,0,1} as d{-1,0,1} if less than or equal to -1] named "Fate Favor"
output 4d{-1,0,1} + 2 named "Standard Roll with Invoke"

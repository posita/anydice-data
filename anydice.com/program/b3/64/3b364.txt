}

result: (T>13) + T/2*10 + (10+[points T])*1000
}

if (T - T/10*10 <= 0)*((T - T/1000*1000)/10-30 <= 0) { result: -3 }
if (T - T/1000*1000)/10-30 <= 0 { result: -2 }
if T - T/10*10 <= 0 { result: -1 }
result: T/1000-60
}
function: failures of T:n {
if T>0 { result: 1 }
result: T
}
function: successes of T:n {
if T>0 { result: T }
}

function: evaluate ROLL:s {
 SUCCESSES: ROLL >= 6
 BOTCH: [count 1 in ROLL]
 if SUCCESSES > 0 { result: SUCCESSES \+ (ROLL = 10)\ } \counts a natural 10 as two successes\
 if BOTCH > 0 { SUCCESSES: SUCCESSES - BOTCH } \Botches subtract from successes\
 if SUCCESSES < 0 { result: SUCCESSES } \'-1' gives the percentile chance of no successes with remaining botches\
 result: 0
}

output [evaluate 10d10 ]-4 named "# of Successes" \Subtract from the evaluated dice to represent "Thresholds"\

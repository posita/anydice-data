\**** Damage calculator normal Vs advantage ****\

function: attackroll ROLL:n vs AC:n {
if ROLL >= CRITROLL { result: CRITDAMAGE }
if ROLL = 1 { result: 0 }
if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
result: 0
}

\looped against enemy AC from 16-18\
ENEMYAC: d{14..18}




ATTACKMOD: 4
PROFICIENCY: 4
ACCBONUS: 0
CRITROLL: 20
DMGBONUS: 13
ATTACKTIMES: 2
DAMAGE: 2*(3d8 + 8d6 + 1d8 + 3*(ATTACKMOD + DMGBONUS))
CRITDAMAGE: 2*(3d8 + 8d6 + 1d8 + 3*(ATTACKMOD + DMGBONUS))
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "BOW SHARPSHOOTER NORMAL"

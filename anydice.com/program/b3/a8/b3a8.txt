DAMAGE:0
HITS:0

function: if A:n HITS DC then damage{

	EXTRA:3

	if HITS=1{A:A+1}

	if HITS=2{A:A+1
	EXTRA:EXTRA+1}

	if HITS=3{A:A+1
	EXTRA:EXTRA+1}

	if HITS=4{A:A+1
	EXTRA:EXTRA+2}
	
	if HITS=5{result:4d4+4+EXTRA}
	
	if A < DC {result: 0}
	
	result: 2d4+EXTRA
	
}



function: roll to hit{

	result:d20+5

}



function:check STACKS on AC with ATTACK:n{

	STACKS:STACKS+1
			
	if STACKS-1=0{
	
		if AC>ATTACK{STACKS:0}
	
	}
	
	
	if STACKS-1>0{
				
		if AC>ATTACK+1{STACKS:0}
	}
	if STACKS-1=5{
		STACKS:0
	}

result:STACKS

}



function: swing at AC{

THISATTACK:0

	loop N over {1..3}{
		
		THISATTACK:[roll to hit]
		DAMAGE:DAMAGE+[if THISATTACK HITS AC then damage]
		HITS:[check HITS on AC with THISATTACK]
		
		
		
	}

	result:DAMAGE

}



output[swing at 10] named "AC:10"

\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 4
DMGBONUS: 0
PROFICIENCY: 3
ACCBONUS: 1
CRITROLL: 20
DAMAGE: 1d12 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d12 + ATTACKMOD + DMGBONUS
output [attackroll d20 vs ENEMYAC] named "1"

ATTACKMOD: 4
DMGBONUS: 1
PROFICIENCY: 3
ACCBONUS: 1
CRITROLL: 19
DAMAGE: 1d6 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d6 + ATTACKMOD + DMGBONUS
output [attackroll d20 vs ENEMYAC] named "2"

set "explode depth" to 50
set "maximum function depth" to 10

function: swaderoll D:d {
  result: [highest of [explode d6] and [explode D]]
}

function: usagedie D:d keep on TN:n {
 SIZE: [maximum of D]
 if SIZE <= 4 {
  result: 1 + [swaderoll D>=TN]
 } else {
  result: 1 + [swaderoll D>=TN] + [usagedie d(SIZE-2) keep on TN]
 }
}

output [lowest of [usagedie d4 keep on 8] and 50] named "d4"
output [lowest of [usagedie d6 keep on 8] and 50] named "d6"
output [lowest of [usagedie d8 keep on 8] and 50] named "d8"
output [lowest of [usagedie d10 keep on 8] and 50] named "d10"
output [lowest of [usagedie d12 keep on 8] and 50] named "d12"

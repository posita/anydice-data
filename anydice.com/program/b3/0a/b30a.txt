function: reroll D:d under N:n{
   result: [rerollcore D die D min N]
}

function: rerollcore R:n die D:d min N:n{
   if R < N { result: D } else {result: R}
}

function: reroll D:d on S:s{
   result: [rerollcore D die D on S]
}

function: rerollcore R:n die D:d on S:s{
   if S = R { result: D } else {result: R}
}

function: r D:n{
result: [reroll dD on {1,2}]
}

function: pbta D:n {
if D < 7 {result: 0}
if D > 9 {result: 2}
result: 1
}

function: reroll A:n plus B:n over T:n depth D:n {
  if (A <= T & B <= T) | (D = 0) { result: A+B }
  if A < B & B > T { result: [reroll A plus 1d6 over T depth D-1] }
  if B < A & A > T { result: [reroll 1d6 plus B over T depth D-1] }
  result: [reroll A plus 1d6 over T depth D-1]
}

function: reroll A:n plus B:n under T:n depth D:n {
  if (A > T & B > T) | (D = 0) { result: A+B }
  if A > B & B <= T { result: [reroll A plus 1d6 under T depth D-1] }
  if B > A & A <= T { result: [reroll 1d6 plus B under T depth D-1] }
  result: [reroll A plus 1d6 under T depth D-1]
}

function: unique values in S:s {
  U: {}
  loop N over S { if !(N = U) { U: {U, N} } }
  result: U
}

function: find PATTERN:s in ROLL:s {
  COUNTS: {}
  loop N over [unique values in ROLL] {
    COUNTS: {COUNTS, N = ROLL}
  }
  result: [sort PATTERN] = [sort COUNTS]
}

output [pbta 2d6+2] named "+2"
output 1d6 > 1 named "1d6 > 1"

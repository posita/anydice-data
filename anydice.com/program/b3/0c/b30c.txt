function: X:n better then Y:n {
  if X > Y {result: 1} 
  if X = Y {result: 0}
  if X < Y {result: -1}
}
function: X:n sbetter then Y:n {
  if X >= Y {result: 1} else {result: -1}
}
output [4d{-1..1} better then 4d{-1..1}] named "+0"
output [4d{-1..1} sbetter then 4d{-1..1}] named "+1/2"
output [4d{-1..1}+1 better then 4d{-1..1}] named "+1"

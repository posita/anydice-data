FUDGE: {-1, 0, +1}

function: ROLL:s reroll up to SKILL:n {
  N: [lowest of SKILL and [count 0 in ROLL]]
  result: NdFUDGE + {1 .. #ROLL-N}@ROLL
}

loop SKILL over {1} {
  output [4dFUDGE reroll up to SKILL] named "skill [SKILL]"
}

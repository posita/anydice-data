DAMAGE:0
HITS:0
THISATTACK:0

function: if A:n HITS DC then damage{
	
	EXTRA:3

	if A=25{

		if HITS=1{A:A+1}
	
		if HITS=2{A:A+1
		EXTRA:EXTRA+1}
	
		if HITS=3{A:A+1
		EXTRA:EXTRA+1}
	
		if HITS=4{A:A+1
		EXTRA:EXTRA+2}

		if HITS=5{result: 8d4+8+EXTRA}
		result:4d4+4+EXTRA
	
	}

	if HITS=1{A:A+1}

	if HITS=2{A:A+1
	EXTRA:EXTRA+1}

	if HITS=3{A:A+1
	EXTRA:EXTRA+1}

	if HITS=4{A:A+1
	EXTRA:EXTRA+2}
	
	if HITS=5{HITS:0
	result:4d4+4+EXTRA}
	
	if A < DC {HITS:0
	result: 0}
	
	HITS:HITS+1
	result: 2d4+EXTRA
	
}



function: roll to hit{

	result:d20+5

}





function: swing at AC{



	loop N over {1..5}{
		
		THISATTACK:[roll to hit]
		DAMAGE:DAMAGE+[if THISATTACK HITS AC then damage]
				
	}

	result:DAMAGE

}

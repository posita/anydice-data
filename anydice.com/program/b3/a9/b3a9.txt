DAMAGE:0
HITS:0

function: if A:n hits DC then damage{

	EXTRA:3
	
	if A < DC {result: 0}
	
	result: 2d4+EXTRA
	
}



function: roll to hit{

	result:d20+5

}



function: swing at AC{

THISATTACK:0

	loop N over {1..3}{
		
		HITS:HITS+1
		THISATTACK:[roll to hit]
		if HITS<5{DAMAGE:DAMAGE+[if THISATTACK hits AC then damage]}
		if HITS=5{DAMAGE:DAMAGE+[if THISATTACK hits AC then damage]+[if THISATTACK hits AC then damage]}
		
	}

	result:DAMAGE

}

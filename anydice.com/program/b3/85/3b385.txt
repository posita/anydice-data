DIE: 6
POOL: 10  \* Adjust the total dice pool size here *\
START: 0

\* Define custom dice for white and black outcomes *\
WHITE_DIE: d{0,0,0,0,1,1}  \* Success on 5-6 *\
BLACK_FAILURE: d{1,0,0,0,0,0}  \* Failure on 1 *\
BLACK_SINGLE: d{0,0,0,0,1,0}   \* 1 Success on 5 *\
BLACK_DOUBLE: d{0,0,0,0,0,1}   \* 2 Successes on 6 *\

loop B over {START..POOL} {
  W: POOL - B
  if B = 0 {
    \* Handle case with no black dice *\
    result: W d WHITE_DIE
  } else {
    \* Calculate black dice components *\
    F: B d BLACK_FAILURE      \* Number of failures *\
    S: B d BLACK_SINGLE       \* Single successes *\
    D: B d BLACK_DOUBLE       \* Double successes *\
    BLACK_SUCCESSES: S + 2*D  \* Total black successes *\
    
    \* Calculate white successes *\
    WHITE_SUCCESSES: W d WHITE_DIE
    
    \* Determine if failures exceed white successes *\
    FAILURE_CONDITION: (F > WHITE_SUCCESSES)
    
    \* Combine results: -1 on failure, else total successes *\
    result: FAILURE_CONDITION ? -1 : (WHITE_SUCCESSES + BLACK_SUCCESSES)
  }
  output result named "[POOL] dice: [B] Black + [W] White"
}

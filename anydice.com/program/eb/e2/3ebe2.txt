function: success ATTR:n SKILL:n vs DC:n {
  if ATTR >= DC { result: 2 }    \ Critical hit from attribute die \
  if SKILL >= DC { result: 2 }   \ Critical hit from skill die \
  if ATTR + SKILL >= DC { result: 1 } \ Normal success \
  result: 0  \ Miss \
}

\ Test d6 attribute, d8 skill vs DC 12 \
output [success 1d6 1d8 vs 12] named "d6+d8 vs DC12"

\ Compare different attribute dice sizes \
output [success 1d4 1d6 vs 10] named "d4+d6 vs DC10"
output [success 1d6 1d6 vs 10] named "d6+d6 vs DC10"  
output [success 1d8 1d6 vs 10] named "d8+d6 vs DC10"

\ Test DC reduction from stress \
output [success 1d6 1d6 vs 12] named "Fresh (DC12)"
output [success 1d6 1d6 vs 10] named "2 Stress (DC10)"
output [success 1d6 1d6 vs 8] named "4 Stress (DC8)"

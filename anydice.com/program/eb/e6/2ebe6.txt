function: check dc DC:n mod MOD:n roll ROLL:n {
  if (ROLL + MOD) >= (DC + 10) {
    S: 2
  } else if (ROLL + MOD) >= DC {
    S: 1
  } else if (ROLL + MOD) >= (DC - 10) {
    S: 0
  } else {
    S: -1
  }
  if ROLL = 20 {
    S: S + 1
  } else if ROLL = 1 {
    S: S - 1
  }
  result: [highest of [lowest of S and 2] and -1]
}

function: medicine mod MOD:n dc DC:n bonus BONUS:n roll ROLL:n{
  SUCCESS: [check dc DC mod MOD roll ROLL]
  if SUCCESS = 2 {
     result: 4d8 + BONUS
  } else if SUCCESS = 1 {
     result: 2d8 + BONUS
  } else if SUCCESS = 0 {
     result: 0
  } else {
     result: -1d8
  }
}

function: trained mod MOD:n {
  result: [medicine mod MOD dc 15 bonus 0 roll 1d20]
}

function: expert mod MOD:n {
  result: [medicine mod MOD dc 20 bonus 10 roll 1d20]
}

function: master mod MOD:n {
  result: [medicine mod MOD dc 30 bonus 30 roll 1d20]
}

function: legendary mod MOD:n {
  result: [medicine mod MOD dc 40 bonus 50 roll 1d20]
}

loop MOD over {-5..40} {
  output [trained mod MOD] named "trained mod [MOD]"
  output [expert mod MOD] named "expert mod [MOD]"
  output [master mod MOD] named "master mod [MOD]"
  output [legendary mod MOD] named "legendary  mod [MOD]"
}

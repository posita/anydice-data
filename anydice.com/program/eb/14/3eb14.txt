function: bodylotto MOD:n {
result: [bodylottohelp 1d10 MOD]
}

function: bodylottohelp ROLL:s MOD:n{
if ROLL=10 {result: [bodylottohelp 1d10 MOD]+10}
else if ROLL+MOD<=1 {result: [bodylow 1d[absolute MOD-1] [absolute MOD-1]]}
else {result: ROLL+MOD}
}

function: bodylow ROLL:s MOD:n{
if ROLL=MOD {result: 1}
else {result: 0}
}

output [bodylotto -3] named "Rebuilding Urban Center"
output [bodylotto -1] named "Overpacked Suburbs"
output [bodylotto +3] named "Combat Zones"

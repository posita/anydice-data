}

AC: 30
ATKBON: 16
DAM: 6d8+12
ADVANTAGE: 1

if ADVANTAGE = 1 {
 ROLLA: [highest 1 of 2d20]
 ROLLB: [highest 1 of 2d20]
} else {
 ROLLA: d20
 ROLLB: d20
}

CRA: 12d8
CRB: 12d8

ATA: [if ROLLA+ATKBON > AC then DAM else 0]
ATB: [if ROLLB+ATKBON > AC then DAM else 0]

  C:ADR-W if C<0 {C:-C}  D:ADR-X if D<0 {D:-D}
  E:ADR-Y if E<0 {E:-E}

  if (C<=A & C<=B & C<=D & C<=E) {RES:W}
  if (D<=A & D<=B & D<=C & D<=E) {RES:X}
  if (E<=A & E<=B & E<=C & E<=D) {RES:Y}
  result:RES
}

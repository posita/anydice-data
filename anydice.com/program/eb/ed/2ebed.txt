INIT: 4
MAXDEPTH: 10

set "maximum function depth" to MAXDEPTH+1

function: continues N:n S:n C:n {
  if N = 1 & C = 0 {
    result: S + 1
  }
  if N = 1 {
    S: S + 1
    C: C - 1
    N: 1d{0, 1}
    result: [continues N S C]
  }
  result: S
}

output [continues 1d{0, 1} INIT MAXDEPTH-1] named "Number of Rounds"

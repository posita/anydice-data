\CRIT: 20  Minimum roll for a crit. Default 20\
ATTACKS: 1 \Number of attacks per round\

function: getcrit A:n B:n { 
 if A >= B { 
  result:2 
 }
 else { 
  result:1 
 } 
}

\MUL:[getcrit 1d20 CRIT]\
REGCRIT: [getcrit 1d20 20]
IMPCRIT: [getcrit 1d20 19]
SUPCRIT: [getcrit 1d20 18]

\output 5d12\
\output 7d6\
output ATTACKS *(REGCRIT * 2d{{1..2}, {3..6}:4}) named "Greatsword + GWF Style + Crits"
output ATTACKS *(REGCRIT * d{{1..2}, {3..12}:7}) named "Greataxe + GWF Style + Crits"

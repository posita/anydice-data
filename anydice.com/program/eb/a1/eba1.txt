\loop DICE over {1..5} {
 output [evaluate DICE d10] named "[DICE]d"
}\

function: evaluate ROLL:s {
 SUCCESSES: ROLL >= DIFF
 if SUCCESSES > 0 { result: SUCCESSES \+ (ROLL = 10)\ }
 if [count 1 in ROLL] { result: -1 } 
 result: 0
 }


function: roll DICE:n dice with difficulty DIFF:n {
EVAL: [evaluate DICE d10]
result: EVAL
}

output [roll 3 dice with difficulty 6] named "Percentile Difficulty"

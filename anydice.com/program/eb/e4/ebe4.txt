MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: CRITICALHIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

BONUS: 5
ATTACK: BONUS + 6
HITDAMAGE: d6 + BONUS
CRITICALDAMAGE: 2d6 + BONUS
MISSDAMAGE: 0
 output [damage for [attack d20 vs 14]]+[damage for [attack d20 vs 14]]

function: A:n bettor B:n {
  if A {
    result: A
  }
  result: B
}
SIEGE_RATE: 1
FORT_BASE: 20
CITY_BASE: 15 
TOWN_BASE: 10
loop BASE_THRESHOLD over {TOWN_BASE, CITY_BASE, FORT_BASE} {
  THRESHOLD: BASE_THRESHOLD - SIEGE_RATE
  RESULT: 2d6 > THRESHOLD
  loop WEEKS over {2..BASE_THRESHOLD} {
    THRESHOLD: THRESHOLD - SIEGE_RATE
    WEEK_RESULT: (2d6 > THRESHOLD) * WEEKS
    RESULT : [RESULT bettor WEEK_RESULT]
  }
  output RESULT named "[BASE_THRESHOLD]"

}

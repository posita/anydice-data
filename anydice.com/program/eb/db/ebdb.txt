function: X:n axe {
if X < 19 {result: 3d8 + FLAT}
else {result: 9d8 + FLAT*3}
}

function: X:n falcata {
if X<17 {result: 1d8+FLAT}
else {result: 3d8 + FLAT*3}
}

function: DIE:d hits to kill HP:n {
    if HP <= 0 { result: 0 }
    result: 1 + [DIE hits to kill HP - DIE]
}

FLAT: 40
HP:100

output [[d20 axe] hits to kill HP] named "AXE"
output [[d20 falcata] hits to kill HP] named "SHITTY WEAPON"

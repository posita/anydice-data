
\-- Death Save --\

DMG_DICE: 3d8
SAVE_DC: 10


function: live or die S:s{
  if [count 100 in S] >= 1 {result: 1}
  if [count 1 in S] >= 3 {result: 1}
  if [count -1 in S] + 2 *[count -2 in S] >= 3 {result: -1}  
  result: 0
}

function: death save N:n {
  if N = 20 {result: 100}
  if N >= SAVE_DC {result: 1}
  if N = 1 {result: -2}
  result: -1
}


output [live or die [death save d20]]

output {2d20}

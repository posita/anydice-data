function: dice N:n tn TN:n {


 if N > TN { result: 0 }         
                                   
 if N <= TN { result: 1 }        
                                   
 result: 1 + [dice d6 tn TN]
}

DICE: [dice d6 tn 5]

loop N over {1..10} {
 output [highest of [lowest of NdDICE and 10] and 0] named "[N]d"
}

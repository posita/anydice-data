function: attack AC attackroll ATK nb NB weapon DICE{
    DAMMAGE: 0
    loop I over {1..NB}{
        if ATK >= AC {
            DAMMAGE: DAMMAGE + DICE
        }
        else{
            DAMMAGE: DAMMAGE + 0
        }
    }
    result: DAMMAGE
}

output [attack 15 attackroll [highest 1 of 2d20] nb 4 weapon 2d6]

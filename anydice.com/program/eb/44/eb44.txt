MISS: 0
HIT: 1
CRITICALHIT: 2

function: luck ROLL:n {
 if ROLL < REROLL+1 {result: d20}
 result: ROLL
} 

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 { result: HIT }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

REROLL:1
PROF:2
ABILITY:4
ATTACK: PROF + ABILITY
loop AC over {10..20} {
 output [attack [luck 1@2d20] vs AC] named "[AC]"
}

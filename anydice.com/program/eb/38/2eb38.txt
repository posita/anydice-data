SHOTS: 20
BS: 3
WOUND_ON: 4
SAVE: 2

function: roll MIN:n DICE:n {
 result: [count {MIN..6} in DICEd6]
}

function: save MAX:n DICE:n {
 result: [count {1..MAX} in DICEd6]
}

N_HITS: [roll BS SHOTS]
N_WOUNDS: [roll WOUND_ON N_HITS]
SAVE_FAIL: SAVE - 1
N_SUCCESS: [save SAVE_FAIL N_WOUNDS]

output N_HITS named "Hits"
output N_WOUNDS named "Wounds"
output N_SUCCESS named "Failed Saves"

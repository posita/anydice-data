\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

\looped against enemy AC from 10-20\
ENEMYAC: d{4..24}

ATTACKMOD: 0
DMGBONUS: 0
PROFICIENCY: 7
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAMAGE: 1
CRITDAMAGE: 1
output ATTACKTIMESd[attackroll d20 vs ENEMYAC] named "Shove Chance"

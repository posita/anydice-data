function: pbta outcome IN:n {
  if IN < 7 {
    result: 0
  }
  if IN < 10 {
    result: 1
  }
  else {
    result: 2
  }
}


output 2d6 named "Standard Roll"


function: pbta adv IN:s {
  result: 1@IN + 2@IN
}

function: pbta disadv IN:s {
  result: 2@IN + 3@IN
}

output [pbta adv 3d6] named "Advantage"

output [pbta disadv 3d6] named "Disadvantage"


function: pbta confi IN:s {
  TOTAL: 0
  loop N over {1..#IN} {
    if N@IN = 1 {
      TOTAL: TOTAL + 6
    }
    else {
      TOTAL: TOTAL + N@IN
    }
  }
  result: TOTAL
}

function: pbta desp IN:s {
  TOTAL: 0
  loop N over {1..#IN} {
    if N@IN = 6 {
      TOTAL: TOTAL + 1
    }
    else {
      TOTAL: TOTAL + N@IN
    }
  }
  result: TOTAL
}

output [pbta confi 2d6] named "Confident"

output [pbta desp 2d6] named "Desperate"

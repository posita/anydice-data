function: roll ROLL:s TN {

 if (1@ROLL >= 10) {
  A: 1@ROLL-9
 } else {
  A: 1@ROLL
 }
 
 if (1@ROLL >= 10) {
  B: 2@ROLL-9
 } else {
  B: 2@ROLL
 }
 
 if (1@ROLL >= 10) {
  C: 3@ROLL-9
 } else {
  C: 3@ROLL
 }
 
 R: ROLL <= TN
 
 result: R+(A=B)+(A=C)+(B=C)
}

loop D over {1..3}{
 loop N over {1..20} {
  output [roll Dd20 N] named "[D]d20 TN [N]"
 }
}

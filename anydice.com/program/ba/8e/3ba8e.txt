\Step Die PbtA\
\1 Step Die for Attributes D6 to D12\
\1 Step Die for Modifiers D6 to D20\
\Modifers: First you step up/down your attributes\
\If you reach a limit step up/down the modifier die\
\5-: failure, 6+: mixed result, 10+ (or double 6+): Full success\

function: stepdiepbta A:s M:s {
RA: [count {6..9} in A]+[count {10..20} in A]*2
RM: [count {6..9} in M]+[count {10..20} in M]*2
SUM : RA+RM
if SUM>2 {SUM:2}
result: SUM
}

output [stepdiepbta 1d6 1d6] named "d6&d6(dM-2)"
output [stepdiepbta 1d6 1d8] named "d6&d8(dM-1)"
output [stepdiepbta 1d6 1d10] named "d6&d10"
output [stepdiepbta 1d8 1d10] named "d8&d10"
output [stepdiepbta 1d10 1d10] named "d10&d10"
output [stepdiepbta 1d12 1d10] named "d12&d10"
output [stepdiepbta 1d12 1d12] named "d12&d12(dM+1)"
output [stepdiepbta 1d12 1d20] named "d12+d20(dM+2)"

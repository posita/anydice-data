function: maxsetl T:s {
  X: {[count 1 in T],[count 2 in T],[count 3 in T],[count 4 in T],[count 5 in T],[count 6 in T]}
  result: 1@([sort X])
}

function: setcount T:s {
  X: {[count 1 in T],[count 2 in T],[count 3 in T],[count 4 in T],[count 5 in T],[count 6 in T]}
  result: X>1
}

function: setdicecount T:s {
  X: {[count 1 in T],[count 2 in T],[count 3 in T],[count 4 in T],[count 5 in T],[count 6 in T]}
  result: (X - (X>0))
}

function: paircount T:s {
  X: {[count 1 in T]/2,[count 2 in T]/2,[count 3 in T]/2,[count 4 in T]/2,[count 5 in T]/2,[count 6 in T]/2}
  result: X
}

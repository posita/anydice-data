\This program calculates the probabilities for 
 each difference in successes between two dice
 pools of 1 to 8 d6\

function: nwod R:n again N:n {
 if N >= R { result: 1 + [nwod R again d6] }
 result: N >= 5
}

NWOD: [nwod 6 again d6]

loop N over {1..10} {
 loop M over {1..10} {
  output [lowest of 10 and NdNWOD] - [lowest of 10 and MdNWOD] named "[N]d vs [M]d"
 }
}

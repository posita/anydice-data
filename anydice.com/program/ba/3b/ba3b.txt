loop CATEGORY over {-2..-1} {
  loop ATTRIBUTE over {-2..-1} {
    SUM: 0
    loop N over {1..(1+[highest of CATEGORY and 0])} {
      SUM: SUM + [explode 1d4]
    }
    output [lowest of 120 and (
      SUM
      + [highest (3+ATTRIBUTE) of (5+CATEGORY+[lowest of 0 and ATTRIBUTE])d12]
      + [highest ([highest of 1 and ATTRIBUTE]) of ([highest of 0 and ATTRIBUTE+1])d8]
    )] named "[CATEGORY],[ATTRIBUTE]"
  }
}

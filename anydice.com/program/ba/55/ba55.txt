function: roll stat STAT with skill SKILL {
  SYN: [lowest of STAT and SKILL]      \ Skill Synergy \
  result: [roll STAT+SKILL small dice with 0 big dice and SYN synergy]
}

function: roll NUM_SM_DICE small dice with NUM_BIG_DICE big dice and SYN synergy {
  DIE: {0,0,1,1,2,SYN}                 \ Normal Die Roll \
  BIG_DIE: {0,1+SYN,2,3+SYN,4,5}

  result: {NUM_SM_DICE}d{DIE} + {NUM_BIG_DICE}d{BIG_DIE}
}

function: roll stat STAT with skill SKILL and NUM_BIG_DICE big dice {
  SYN: [lowest of STAT and SKILL]      \ Skill Synergy \
  NUM_BIG_DICE: [lowest of NUM_BIG_DICE and (STAT+SKILL)/3]
  NUM_SM_DICE: (STAT+SKILL) - NUM_BIG_DICE*3

  result: [roll NUM_SM_DICE small dice with NUM_BIG_DICE big dice and SYN synergy]
}

output [roll stat 4 with skill 0 and 1 big dice] named "X1"
output [roll stat 3 with skill 1 and 0 big dice] named "X2"

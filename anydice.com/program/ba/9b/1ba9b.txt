
function: check ROLL:n vs DC:n then TRUE:d else FALSE:d {
    if ROLL >= DC {
        result: TRUE
    }
    result: FALSE
}

DISADVANTAGE: [lowest 1 of 2d20]

function: sickening radiance CON:n vs DC:n {

    A: 0
    B: 1
    C: 2
    D: 3
    E: 4
    F: 5

    loop N over {1 .. 60} {
        A: [check d20 + CON vs DC then A else B]
        B: [check d20 + CON vs DC then B else C]
        C: [check d20 + CON vs DC then C else D]
        D: [check DISADVANTAGE + CON vs DC then D else E]
        E: [check DISADVANTAGE + CON vs DC then E else F]
        F: [check DISADVANTAGE + CON vs DC then F else 6]
    }

    result: A
}

loop CON over {11 .. 17} {

    output
    [sickening radiance CON vs 18] named "+[CON] CON vs DC 18"

}

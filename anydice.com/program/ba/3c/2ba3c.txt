
function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

\output [reroll {1,2} on 3d4] named "reroll 1s and 2s on 3d4"\
\output [reroll {1,2} on 2d6] named "reroll 2s and 2s on 2d6"\

output [reroll {1,2} on 2d6] + [reroll {1,2} on 2d10] named "Uzon's Dagri"

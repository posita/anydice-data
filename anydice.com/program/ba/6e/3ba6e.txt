DIE: d6               \ Die type \
MISS: 0		       \ Value of missed result\
C_HIT : [maximum of DIE] \ Crit threshold (d6, 6)\

\ Parameters \
N: 4			 \ Number of dice \
DMG: 2	            \ Value of hit damage\ 
CRT_DMG: 2	   	\ Value of crit damage\
HIT : 5		 	\ Hit threshold (d6, 4+)\ 

function: attack ROLL:n vs HIT :n {
 if ROLL >= HIT {
 if ROLL = C_HIT {result: CRT_DMG}
  result: DMG
 }
 result: MISS
}

\ Reroll and sum hits ( requires generic damage calculator) \
function: attackrerollhits ROLL:n vs HIT :n {
 if ROLL >= HIT {
  REROLL: [attack DIE vs HIT] 
  if ROLL = C_HIT {result: CRT_DMG + REROLL}
  result: DMG + REROLL
 }
 result: MISS
}

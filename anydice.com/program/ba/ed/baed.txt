function: ob NUMBER:n DICE:n {
  TOTALSUM: 0
  loop LOOPING over {1..NUMBER} {
    X: 1dDICE
    X: [checkmax X DICE]
    TOTALSUM: TOTALSUM + X
  }
  result: TOTALSUM
}

function: checkmax N:n D:n {
  if N = D {
    result: [ob 2 D]
  }
  result: N
}

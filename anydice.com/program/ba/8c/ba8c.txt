function: convert SUM:n {
 TENSROLLED: 0
 NINESROLLED: 0
 if SUM >= 10000 {
   TENSROLLED: SUM / 10000
 }
 TEMPA: SUM - TENSROLLED * 9990
 if TEMPA >= 1000 {
  NINESROLLED: TEMPA / 1000
 }
 TEMPB: TEMPA - NINESROLLED * 991
 result: TEMPB + TENSROLLED d [explode d10] + NINESROLLED d [explode d10]
}

function: ROLLSIZE:n k KEEPSIZE:n {
 result: [convert [highest KEEPSIZE of ROLLSIZE d{1..8, 1000, 1000}]]
}

loop POOL over {6..6} {
 loop KEEP over {4} {
  output [POOL k KEEP] named "[POOL]k[KEEP]"
 }
}

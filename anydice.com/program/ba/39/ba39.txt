loop CATEGORY over {-2..-1} {
  loop ATTRIBUTE over {-2..-1} {
    output [lowest of 100 and (
      [explode (1+[highest of CATEGORY and 0])d4]
      + [highest (3+ATTRIBUTE) of (5+CATEGORY+[lowest of 0 and ATTRIBUTE])d8]
      + [highest ([highest of 1 and ATTRIBUTE]) of ([highest of 0 and ATTRIBUTE+1])d12]
    )] named "[CATEGORY],[ATTRIBUTE]"
  }
}

loop CATEGORY over {0..3} {
  loop ATTRIBUTE over {-2..CATEGORY} {
    output [lowest of 100 and (
      [explode (1+[highest of CATEGORY and 0])d4]
      + [highest (3+ATTRIBUTE) of (5+CATEGORY+[lowest of 0 and ATTRIBUTE])d12]
      + [highest ([highest of 1 and ATTRIBUTE]) of ([highest of 0 and ATTRIBUTE+1])d8]
    )] named "[CATEGORY],[ATTRIBUTE]"
  }
}

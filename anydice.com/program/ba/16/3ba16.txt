function: N:n reroll LOW:n {
  if N <= LOW { result: [N reroll LOW] }
  else { result: N }
}
function: reroll N:d {
  result: N
}
function: N:n reroll LOW:n once {
  if N <= LOW { result: [reroll d12] }
  else { result: N }
}
function: attack ROLL:n plus MODIFIER:n plus BONUS:n vs AC:n for DMG:d crit on CRIT_RANGE:n {
    if ROLL = 1 {result: 0}
    if ROLL >= CRIT_RANGE            {result: DMG + BONUS + [maximum of DMG]}
    if ROLL + MODIFIER + BONUS >= AC {result: DMG + BONUS}
    else {result: 0}
}

function: roll SET:s le SUCC:n with REROLLS:n rerolls {
  SUCCESSES: 0
  loop N over SET {
    DIEROLL: dN
    RE: [reroller DIEROLL if over SUCC up to REROLLS times]
    SUCCESSES: SUCCESSES + 1d (dN <= SUCC) 
  }
  result: SUCCESSES 
}

function: helper N:n le LE:n {
  if N <= LE { result: 1 }
  result: 0
}


\##########################\

function: reroller D:d if over N:n up to T:n times {
   if T < 1 {result: D }
   if T = 1 { result: [rerollerone D of D if over N] }
   if T = 2 { result: [rerollertwo D of D if over N] }
   if T = 3 { result: [rerollerthree D of D if over N] }
   if T = 4 { result: [rerollerfour D of D if over N] }
   if T = 5 { result: [rerollerfive D of D if over N] }
}


function: rerollerone R:n of D:d if over N:n {
   if R > N { result: D } else {result: R}
}

function: rerollertwo R:n of D:d if over N:n {
   if R > N { 
      result: [rerollerone D of D if over N] 
   } else {result: R}
}

function: rerollerthree R:n of D:d if over N:n {
   if R > N { 
      result: [rerollertwo D of D if over N] 
   } else {result: R}
}

function: rerollerfour R:n of D:d if over N:n {
   if R > N { 
      result: [rerollerthree D of D if over N] 
   } else {result: R}
}

function: rerollerfive R:n of D:d if over N:n {
   if R > N { 
      result: [rerollerfour D of D if over N] 
   } else {result: R}
}

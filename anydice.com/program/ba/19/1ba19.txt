function: attackroll ROLL:n with ATKMOD:n vs AC:n for DMG:n or CRIT_DMG:n {
    if ROLL >= CRITROLL { result: CRIT_DMG } 
    if ROLL = 1 { result: 0 }
    if ROLL + ATKMOD >= AC { result: DMG }
    result: 0
}

CRITROLL: 20
PROFICIENCY: 3
FIGHTING_STYLE: 2
STAT_MOD: 5

SS_ATK: -5
SS_DMG: 10

WEAPON_DICE: 1d8
SNEAK_ATTACK_DICE: 1d6
HM_DICE: 1d6

STARTAC: 16
ENDAC: 18

loop AC over {STARTAC..ENDAC}{
    ATTACK_MOD: PROFICIENCY + STAT_MOD + FIGHTING_STYLE + SS_ATK
    DMG_DICE: WEAPON_DICE + SNEAK_ATTACK_DICE + HM_DICE
    DMG_MOD: STAT_MOD + SS_DMG
    DAMAGE: DMG_DICE + DMG_MOD
    CRIT_DAMAGE: 2 * DMG_DICE + DMG_MOD

    ATTACK_ONE: 1d[attackroll 1d20 with ATTACK_MOD vs AC for DAMAGE or CRIT_DAMAGE]

    ATTACK_MOD: PROFICIENCY + STAT_MOD + FIGHTING_STYLE + SS_ATK
    DMG_DICE: WEAPON_DICE + HM_DICE
    DMG_MOD: STAT_MOD + SS_DMG
    DAMAGE: DMG_DICE + DMG_MOD
    CRIT_DAMAGE: 2 * DMG_DICE + DMG_MOD

    ATTACK_TWO: 1d[ attackroll 1d20 with ATTACK_MOD vs AC for DAMAGE or CRIT_DAMAGE]
    
    output ATTACK_ONE + ATTACK_TWO  named "SS, SA, HM vs. AC [AC]"
}


loop AC over {STARTAC..ENDAC}{
    ATTACK_MOD: PROFICIENCY + STAT_MOD + FIGHTING_STYLE
    DMG_DICE: WEAPON_DICE + SNEAK_ATTACK_DICE + HM_DICE

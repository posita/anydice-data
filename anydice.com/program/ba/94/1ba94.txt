function: calc ROLL:n TOHIT:n DIE:n DMG:n {
if ROLL = 20 {result: 2dDIE + 2d8 + 6d8 + DMG}
if ROLL + TOHIT < 17 {result: 0} 
result: 1dDIE + 1d8 + DMG
}
output 2d[calc 1d20 4+5-5 10 5+10] named "GWM + ASI"
output 2d[calc 1d20 4+4-5 10 4+10] + [calc 1d20 4+4-5 4 4+10] named "GWM + PAM"
output 2d[calc 1d20 4+5 10 5] + [calc 1d20 4+5 4 5] named "ASI + PAM"

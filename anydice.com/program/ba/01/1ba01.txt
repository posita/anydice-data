function: rollorig SET:s le SUCC:n {
  EXPLODED: 0
  SUCCESSES: 0
  loop N over SET {
    DIEROLL: dN
    SUCCESSES: SUCCESSES + 1d (dN <= SUCC) 
  }
  result: SUCCESSES 
}

function: roll SET:s le SUCC:n with N:n rerolls {
  EXPLODED: 0
  SUCCESSES: 0
  loop N over SET {
    DIEROLL: dN
    SUCCESSES: SUCCESSES + 1d (dN <= SUCC) 
  }
  result: SUCCESSES 
}
\
function: helper N:n le LE:n {
  if N <= LE { result: 1 }
  result: 0
}
\

function: checkroll R:n lt N:n {
   if R < N { result: 1 } else { result: 0 }
}

function: reroller R:n of D:d if over N:n {
   if R > N { result: D } else {result: R}
}

function: rerollhelper D:d of ROLL:n if over N:n up to T:n times {
  loop L over {1..T} {
    ROLL: [reroller D of D if over N]
    COMP: ROLL < N
    if COMP { result: ROLL }
  }
  result: ROLL
}

function: reroll D:d if over N:n up to T:n times {
  result: [rerollhelper D of D if over N up to T times]
}



function: reroll D:d if over N:n {
   result: [rerollhelper D of D if over N]
}

function: rerollhelper R:n of D:d if over N:n {
   if R > N { result: D } else {result: R}
}

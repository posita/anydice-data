function: roll A:n plus B:n plus C:n vs D:n {
 if A+B=0 | A+B+C+5 < D { result: 2 }
 if A+B=18 | A+B+C >= D+10 { result: 3 }
 if A + B <= 2 | A+B+C < D { result: 2 }
 if A+B+C >= D+5 { result: 3 }
 result: 3
}


loop N over {-3..3}{
loop X over {-2,0,2,4,8}{
output [roll 1d{0..9} plus 1d{0..9} plus N vs 2d{0..9} + X] named "+[N] vs [X]"
}}

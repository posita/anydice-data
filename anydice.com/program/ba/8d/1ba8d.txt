function: roll SET:s le SUCC:n {
  EXPLODED: 0
  SUCCESSES: 0
  loop N over SET {
    DIEROLL: dN
    SUCCESSES: SUCCESSES + 1d (dN <= SUCC) 
  }
  result: SUCCESSES 
}

function: helper N:n le LE:n {
  if N <= LE { result: 1 }
  result: 0
}

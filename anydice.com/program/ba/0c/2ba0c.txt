function: roll A:n plus B:n plus C:n vs D:n {
if A + B = 18 | A + B + C >= D + 10  { result: 3 }
if A + B = 0 | A + B + C + 10 <= D { result: 2 }
if A < 2 { if A >= B { result: 2 }}
if B < 2 { if B >= A { result: 2 }}
if A + B + C >= D + 5 { result: 3 }
if A + B + C >= D { result: 3 }
result: 2
}


loop Y over {-2..2} {
output [roll d{0..9} plus d{0..9} plus Y vs 2d{0..9}-3] named "Stat ([Y]) vs Easy -2"
output [roll d{0..9} plus d{0..9} plus Y vs 2d{0..9}] named "Stat ([Y]) vs Average 0"
output [roll d{0..9} plus d{0..9} plus Y vs 2d{0..9}+3] named "Stat ([Y]) vs Hard 2"
output [roll d{0..9} plus d{0..9} plus Y vs 2d{0..9}+4] named "Stat ([Y]) vs Insane 4"
output [roll d{0..9} plus d{0..9} plus Y vs 2d{0..9}+8] named "Stat ([Y]) vs Impossible 8"
}

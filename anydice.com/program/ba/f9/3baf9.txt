function: ROLL:n reroll orless MIN:n as REROLL:d 
{
  if ROLL <= MIN { 
        result: REROLL   \If the first roll is below or equal to the set minimum value, it rolls again. \
         }
  else { result: ROLL }
}

function: hit TO_HIT:s 
{
    result: TO_HIT >= BS \Basically return how many in TO_HIT which is a sequence (s) are above or equal to BS\
}


function: sustained TO_HIT:n {
    if TO_HIT = 6 {result: 2}
    else if TO_HIT >= BS {result: 1}
    else {result: 0}
}

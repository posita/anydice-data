set "maximum function depth" to 6

function: explode DIE:n count successes at TARGET with a fate point worth FATE_POINT {
  if DIE >= 12 { 
    result: 1 + [explode d12 count successes at TARGET with a fate point worth FATE_POINT]
  }
  result: DIE+FATE_POINT >= TARGET
}

TARGET: 5
FATE_POINT_BONUS: 0

output 4d[explode d12 count successes at 5 with a fate point worth FATE_POINT_BONUS] named "YKMR"
output 4d[explode d12 count successes at 4 with a fate point worth FATE_POINT_BONUS] named "OR/ATR"
output 4d[explode d12 count successes at 5 with a fate point worth FATE_POINT_BONUS] + 4d[explode d12 count successes at 4 with a fate point worth FATE_POINT_BONUS] named "TOTAL HEAL"

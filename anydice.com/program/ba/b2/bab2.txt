function: rerolla DUAL:s vs SHIELD:n {
R: 1@DUAL
if R >= SHIELD {
R: [highest of d6+1 and 2@DUAL]
}
result: R - SHIELD
}

function: rerollb DUAL:s vs SHIELD:n {
R: 1@DUAL
if R >= SHIELD {
R: [highest of d{1,1,2,3,4,5} and 2@DUAL]
}
result: R - SHIELD
}

function: rerollc DUAL:s vs SHIELD:n {
R: 1@DUAL
if R >= SHIELD {
R: [highest of d6+2 and 2@DUAL]
}
result: R - SHIELD
}



output 1@2d6 > d6+2 named "D > p r1"
output 1@2d6 = d6+2 named "d = p r1"
output 1@2d6 < d6+2 named "d < P r1"

output 1@2d6 = d{1,1,2,3,4,5} named "d = p r2"
output 1@2d6 < d{1,1,2,3,4,5} named "d < P r2"

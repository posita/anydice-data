DEX: 3
STR: 1

function: notfail ROLL:n {
  if (ROLL>1) {
    result: ROLL+DEX
  }
  result: 0
}

loop DIE over {4, 6, 8, 10, 12} {
  if (STR=0) {
    output [notfail dDIE] named "DIE=d[DIE], STR=[STR]" \No explosions for STR 0\
  } else { 
    set "explode depth" to STR
    output [notfail [explode dDIE]] named "DIE=d[DIE], STR=[STR]"
  }
}

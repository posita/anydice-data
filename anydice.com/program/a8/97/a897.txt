function: A:s vs B:s {
 loop P over {1..[lowest of #A and #B]} {
  if P@A > P@B { result: [A beats P@B at P] }
  if P@B > P@A { result: -[B beats P@A at P] }
 }
 if #A > #B {result: [A emptied other at #B+1 so roll 1d10]}
 if #B > #A {result: -[B emptied other at #A+1 so roll 1d10]}
 result: d{1,-1}
}

function: A:s emptied other at W:n so roll R:s {
 if W@A > R {result: [A beats R at W]}
 if W@A < R {result: -1}
 if #A = W {result: d{1,-1}}
 result: [A emptied other at W+1 so roll 1d10]
}

function: A:s beats T:n at W:n {
 V: 1
 loop P over {W + 1 .. #A} {
  if P@A > T { V: V + 1 }
  else { result: V }
 }
 result: V
}

output [4d10 vs 3d10]

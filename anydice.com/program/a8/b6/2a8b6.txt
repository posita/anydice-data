function: test S:n and L:n and LL:n and H:n vs D:n {
  T: H+D
  if L = 6 { LL: LL + L } 
  else { LL: L } \only use extra luck when exploding \
  R: 0
  if S >= T { R: 1 }
  if LL >= T { R: R + 10 }
  if S = H & L = H { R: R + 100 }
  result: R
}

SKILL: d6
LUCK: d6
LL: [explode d6]
HAZARD: d6

loop DIFFICULTY over {-5..10} {
  output [test SKILL and LUCK and LL and HAZARD vs DIFFICULTY] named "[DIFFICULTY]"
}

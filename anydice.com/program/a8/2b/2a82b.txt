\
 Fates in the Dark
 0: Things go bad.
 1: Success, but with undesired consequences.
 2: Clean success.
 3: Critical success!
\

function: fates ROLL:s {
 if 1@ROLL = 1 {
  result: 2 + (2@ROLL = 1)
 }
 result: 1@ROLL >= 0
}

output [fates 2@2d6] named "0d"
loop P over {1..7} {
 output [fates Pd{-1,0,1}] named "[P]d"
}

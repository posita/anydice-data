function: attack ACC:n DAM:n DOD:n ARM:n {
  result: [damage [result ACC DAM DOD] ARM]
} 

function: hit ACC:n DOD:n  {
  HIT: ACCd{0,0,1,1,1,2} >= DOD
  result: HIT
}

function: strike DAM:n {
 STRIKE: DAMd{0,0,1,1,1,2}
 result: STRIKE 
}

function: damage STRIKE:n ARM:n {
if STRIKE < ARM {
 result: 0
}
result: STRIKE - ARM 
} 

function: health DAM:n ARM:n {
result: [damage [strike DAM] ARM]
}
output [health 6 1]

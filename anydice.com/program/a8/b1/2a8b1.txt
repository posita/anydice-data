function: test S:n and L:n and H:n vs D:n {
  T: H+D
  R: 0
  if S >= T { R: 1 }
  if L >= T { R: R + 10 }
  if S = L { R: R + 100 }
  result: R
}

SKILL: d6
LUCK: [explode d6]
HAZARD: d6

loop DIFFICULTY over {-5..10} {
  output [test SKILL and LUCK and HAZARD vs DIFFICULTY] named "[DIFFICULTY]"
}

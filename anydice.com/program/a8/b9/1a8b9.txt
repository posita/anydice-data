 function: hit ACC:n DOD:n  {
  HIT: ACCd{0,0,1,1,1,2} >= DOD
  result: HIT
}

function: strike DAM:n {
 STRIKE: DAMd{0,0,1,1,1,2}
 result: STRIKE 
}

function: result STRIKE:n ARM:n {
if STRIKE < ARM {
 result: 0
}
result: STRIKE - ARM 
} 

function: damage DAM:n ARM:n {
result: [result [strike DAM] ARM]
}
output [damage 5 3]

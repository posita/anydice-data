function: normie ROLL:n vs AV:n{
 if ROLL = 20 {result: CRIT}
 if ROLL = 1 {result: MISS}
 if ROLL + ATTACK >= AC {result: HIT}
 result:MISS
}
function: power ROLL:n vs AC:n{
 if ROLL = 20 {result:PBC}
 if ROLL = 1 {result:MISS}
 if ROLL+ATTACK-2*N  >= AC {result: PA}
 result:MISS
}

function: attack ROLL:n and R:n and RR:n vs AC:n {
 if ATTACK >= AC {
  if ROLL = 20 {result: BC}
  if ROLL+ATTACK >=20 {result:CRIT}
   result: HIT}
 if ROLL = 20 {result: BC}
 if ROLL = 1 {
  if RR < R {
   if RR+ATTACK >=AC {result:HIT}
   result: MISS}
  if RR >= R {
   if R+ATTACK >=AC {result:HIT}
   result: MISS}
  result:MISS }
 if ROLL + ATTACK >= AC {result: HIT}
 if ROLL + ATTACK < AC {
   if RR < R {
   if RR+ATTACK >=AC {result:HIT}
   result: MISS}
  if RR >= R {
   if R+ATTACK>=AC {result:HIT}
   result: MISS}
  result:MISS }
}

function: damage for ATTACKRESULT:n{
 if ATTACKRESULT = MISS {result:0} 
 if ATTACKRESULT = HIT {result: DMG}
 if ATTACKRESULT = CRIT{ result: CDMG}
 if ATTACKRESULT = BC {result: BCDMG}
 if ATTACKRESULT = PA{ result: PADMG}
 if ATTACKRESULT = PBC {result: PBCDMG}
}

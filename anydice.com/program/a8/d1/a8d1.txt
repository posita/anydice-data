set "explode depth" to 5
set "maximum function depth" to 5

function: ccn CCN:n{
if CCN = 0 { result: -1 + [ccn d{0,1}] }
else       { result: 0 }
}

function: ccp CCP:n{
if CCP = 1 { result: +1 + [ccp d{0,1}] }
else       { result: 0 }
}

function: cc CC:n{
if CC = 1 { result: +1 + [ccp d{0,1}] }
else      { result: -1 + [ccn d{0,1}] }
}

loop N over {1..10}{
output (Nd{0,1} + [cc d{0,1}]) named "[N] Coins + The Crazy Coin"
}

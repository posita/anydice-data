\ determines if a roll beats the given DC \
function: check A:n is above B:n {
 
 if A >= B & A != 28{ result: 1 }
 if A = 28 { result: 2 }
 result: 0
}
function: check C:n is below D:n {
 if C <= D { result: 4 }
 result: 0
}

\ DAMAGE \
function: winnings from A:n {
 if A = 0 { result: 0 }
 if A = 1 { result: 3d10 }  \change this damage for attack\
 if A = 2 { result: 3d10*2 } \CRIT DAMAGE\
 if A = 3 { result: 0 }      \if half damage on save, replace 0 with half damage\
 if A = 4 { result: 2d12 }   \Change this damage to save damage\
 
 result: 0

}

\ 0 is a miss, 1 is a weak hit, 2 is a strong hit \

function: progress PROG:n {
  result: 2d(PROG > d10)
}

function: action AROLL:n AMOD:n MOM:n {
  if MOM < 0 & AROLL = -MOM {AROLL: 0}
  result: 2d([highest of [lowest of AROLL + AMOD and 10] and MOM] > d10)
}
function: actionb AROLL:n AMOD:n MOM:n {
  if MOM < 0 & AROLL = -MOM {AROLL: 0}
  result: 2d([highest of [lowest of AROLL + AMOD and 12] and MOM] > d12)
}
function: actionc AROLL:n AMOD:n MOM:n {
  if MOM < 0 & AROLL = -MOM {AROLL: 0}
  result: 2d([highest of [lowest of AROLL + AMOD and 8] and MOM] > d8)
}
function: actiond AROLL:n AMOD:n MOM:n {
  if MOM < 0 & AROLL = -MOM {AROLL: 0}
  result: 2d([highest of [lowest of AROLL + AMOD and 20] and MOM] > d20)
}

output [actionc 1d6 4 0] named "Action8 1d6+2"
output [action 1d6 4 0] named "Action10 1d6+2"
output [actionb 1d6 4 0] named "Action12 1d6+2"
output [actiond 1d6 4 0] named "Action20 1d6+2"

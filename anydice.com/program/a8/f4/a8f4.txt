function: roll N:n DAMAGE:n {
 if N > 0 {
  N: N+DAMAGE-1
  result: NdTOWOUND
 }
 if N = 0 {
  result: 0
 }
 if N < 0 {
  result: -1
 }
}

function: owod N:n TN:n {
 if N = 1 { result: -1 }
 if N < TN { result: 0 }
 else { result: 1 }
}

TOHIT: [owod d10 6]
TOWOUND: [owod d10 6]

loop N over {1..20} {
 output [highest of [lowest of [roll NdTOHIT 13] and 15] and -4] named "[N]d"
}

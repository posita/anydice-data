function: attack ACC:n DAM:n DOD:n ARM:n {
  result: [damage [result ACC DAM DOD] ARM]
} 

function: hit ACC:n DOD:n  {
  HIT: ACCd{0,0,1,1,1,2} >= DOD
  result: HIT
}

function: strike DAM:n HIT:n {
 STRIKE: 0
 if HIT > 0 {
 STRIKE: DAMd{0,0,1,1,1,2}
}
 result: STRIKE 
}

function: damage STRIKE:n ARM:n {
if STRIKE < ARM {
 result: 0
}
result: STRIKE - ARM 
} 

function: result ACC:n DAM:n DOD:n {
result: [strike DAM [hit ACC DOD]]
}
output [attack 6 4 3 3]

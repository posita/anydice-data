function: attack ATTACK:n AMOD:n AC:n DMG:d CRIT:d {
  if ATTACK = 20 { result: DMG + CRIT }
  if ATTACK + AMOD >= AC { result: DMG }
  result: 0
}

output [attack 1@3d20 10 15 1d6+15 1d6] named "Glaive normal"
output [attack 1@3d20 5 15 1d6+25 1d6] named "Glaive GWF"
output [attack 1@3d20 5 15 1d6+25 1d6] - [attack 1@3d20 10 15 1d6+15 1d6] named "difference"

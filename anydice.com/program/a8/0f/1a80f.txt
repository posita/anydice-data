  if X >= Y { result: 0 }
  result: Z
}

function: horse X:n and Y:n {
  if X >= 18 { result: 3 }
  if X >= 11 { result: 2 }
  if X < 11 & Y >= 11 {
    result: 1
  }
  result: 0
}
ISACSTR: 1d20 + 1 + [check ISAC_WILL vs 15 debuff -2]
ISACCON: 1d20 + 3 +[check ISAC_WILL vs 15 debuff -2]
output [horse ISACSTR and ISACCON] named "Issac Horse Success"

function: horsecha X:n and Y:n {
  if X >= 23 { result: 3 }
  if X >= 14 { result: 2 }
  if X < 14 & Y >= 11 {
    result: 1
  }
  result: 0
}

LSTR: 1d20 - 1  + [check LUCY_WILL vs 15 debuff -2]
LCON: 1d20 + 5  + [check LUCY_WILL vs 15 debuff -2]
LCHA: 1d20 + 10 + [check LUCY_WILL vs 15 debuff -2]
output [horse LSTR and LCON] named "Lucy STR Horse"
output [horsecha LCHA and LCON] named "Lucy CHA Horse"

output [highest of [horsecha LCHA and LCON] and [horse ISACSTR and ISACCON]] named "Combo Horse"

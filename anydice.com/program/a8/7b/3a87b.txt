function: foo DIE:n TO_HIT:n BASE_DAM:n TARGET:n {
  if DIE = 20 { result: 2d8 + BASE_DAM  }
  if DIE + TO_HIT > TARGET { result: 1d8 + BASE_DAM }
  result: 0
}

function: gethigh A:n B:n {
  if A > B {result: A}
  result: B
}

MY_TO_HIT: 11
TARGET_AC: 18


function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}
output 2d[reroll {6} on d6] named "raw, reroll 6s"

output 2d6 named "average"

output 2d[reroll {1} on d6] named "superior, reroll 1s"

output 2d[reroll {1,2} on d6] named "elite, reroll 1s and 2s"

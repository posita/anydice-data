\
 Fates in the Dark
 0: Things go bad.
 1: Success, but with undesired consequences.
 2: Clean success.
 3: Critical success!
\

function: fates A:n
{
 if A < 0
{result : 0}
 if A = 0
{result : 1}
 if A > 0 & A < 3
{result : 2}
 if A > 3
{result : 3}
}

loop P over {0..5} {
 output [fates 4d{-1,0,1}+P] named "Fates + [P]"
}

function: deplodetwo N:n on SIZE:n {
 if N = 1 | N = 2 {
  result: N + [deplode d(SIZE) on SIZE] 
  }
 result: N
}

function: deplode N:n on SIZE:n {
 if N = 1 {
  result: 1 + [deplode d(SIZE) on SIZE] 
  }
 result: N
}

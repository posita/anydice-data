function: goodluck R:n{
if R = 1 {result: +1 + [goodluck d{0,1}]}
if R = 0 {result:  0}
}

function: badluck R:n{
if R = 1 {result:  0}
if R = 0 {result: -1 + [badluck  d{0,1}]}
}

function: luckycoin R:n{
if R = 1 {result: +1 + [goodluck d{0,1}]}
if R = 0 {result: -1 + [badluck  d{0,1}]}
}

loop N over {1..10}{
output [highest of (Nd{0,1} + [luckycoin d{0,1}]) and 0] named "[N] Coins + The Lucky Coin"
}

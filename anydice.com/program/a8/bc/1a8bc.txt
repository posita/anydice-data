function: attack ARC:n RAT:n DMG:n DEF:n ARM:n {
  result: [damage [result RAT ARC DEF DMG] ARM]
} 

function: hit RAT:n ARC:n DEF:n  {
  SUCCESS: RATd{0,0,0,1,1,2} + ARCd{0,1,1,1,1,2} - DEFd{0,0,0,1,1,2}
  result: SUCCESS
}

function: strike DMG:n SUCCESS:n  {
  STRIKE: 0
  if SUCCESS >= 0 {
    STRIKE: SUCCESSd{0,1,1,1,1,2}+DMGd{0,0,0,1,1,2}
  } 
  result: STRIKE
}

function: result RAT:n ARC:n DEF:n DMG:n {
  result: [strike DMG [hit RAT ARC DEF]]
}

function: damage STRIKE:n ARM:n {
  if STRIKE < ARM {
    result: 0
  }
  if STRIKE < 2*ARM {
    result: 1
  }
  if STRIKE < 3*ARM {
    result: 2
  }
  result: 3
}

output [attack 0 4 4 3 3]

output [attack 0 4 2 3 2]

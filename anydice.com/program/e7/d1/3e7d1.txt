function: sum SEQUENCE:s{
 R: 0
 loop P over {1..#SEQUENCE}{
  R: {P@SEQUENCE} + R
 }
 result: R
}

function: attack with ROLL:s precision P:n max M:n {
 if [count 1 in ROLL] { result: -1 } \ failure \
 if [sum ROLL] >= (M - P) { result: 1 } \ precise \
 result: 0 \ normal hit \
}

PRECISION: 0

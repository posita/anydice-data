function: profit ROLL:n {
  if ROLL <= 10  { result: -45}
  if ROLL <= 20  { result: -30}
  if ROLL <= 30  { result: -15}
  if ROLL <= 40  { result: 0}
  if ROLL <= 60  { result: 1d6  * 5 }
  if ROLL <= 80  { result: 2d8  * 5 }
  if ROLL <= 90  { result: 3d10 * 5 }
  if ROLL >= 91  { result: 4d12 * 5 }
}

output [profit 1d100] named "No Marketing"
loop X over {1..10} {
  output [profit 1d100+(10*X)] - 10*X named "[X]0gp Marketing"
}


output [profit 1d100+1d20+6] named "No Marketing + Performance"
loop X over {1..10} {
  output [profit 1d100+(10*X)+1d20+6] - 10*X named "[X]0gp Marketing + Performance"
}


output [profit 1d100+2d20+10] named "No Marketing + Performance + Arcana"
loop X over {1..10} {
  output [profit 1d100+(10*X)+2d20+10] - 10*X named "[X]0gp Marketing + Performance + Arcana"
}

MISS: 0
HIT: 1
GREATHIT: 2
CRITHIT: 3


function: attack ROLL:n vs DEFENSE:n vs CONFIRM:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE & CONFIRM >= 15  { result: CRITHIT }
  if ROLL + ATTACK >= DEFENSE  { result: GREATHIT }

  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

ATTACK: 0
output [attack d20 vs 20 vs d20]

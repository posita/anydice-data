\Description -----------------------------------------------------------------------------------\
\This function returns the re-rolled value of the die if it is lower or equal to the lower limit\
\Variables -------------------------------------------------------------------------------------\
\A = Initial input value. I.E. 1 to 6 for d6                                                    \
\B = Lower reroll limit. I.E. 2 for Great Weapon Fighter                                        \
\C = Dice value. I.E. 6 for d6                                                                  \
\-----------------------------------------------------------------------------------------------\

function: reroll A:n B:n C:n{ 
 if A <= B { 
  A: 1dC 
 }  
 result: A 
}

\Description ---------------------------------------------\
\This function simulates the Great Weapon fighting style  \
\Variables -----------------------------------------------\
\A = Dice value I.E. A=6 for 1d6                          \
\B = Number of damage dice I.E. B = 2 for Greatsword (2d6)\
\C = Lower limit for reroll. Default 2                    \
\---------------------------------------------------------\

function: greatweaponmaster A:n B:n C:n{
 X:0
 loop N over {1..B} {
  X: X + [reroll 1dA C A]
 }
 result: X
}

\Description ---------------------------------------------\
\This function simulates great sword damage rolls         \
\Variables -----------------------------------------------\
\A = Attack Roll to determine critical hits               \
\---------------------------------------------------------\

function: gsdmg A:n {
 if A >= 19 {
  result: 2d[greatweaponmaster DIE NUMDIE MIN] + 1d6 \savage attacker racial trait\ + STRMOD + DMGMOD
 }
 else {
  result: [greatweaponmaster DIE NUMDIE MIN] + STRMOD + DMGMOD
 }
}

\Description ---------------------------------------------\
\This function simulates an attack with GWM Feat -5/+10   \
\Variables -----------------------------------------------\
\A = AC of target                                         \
\B = Attack Roll                                          \
\---------------------------------------------------------\

function: gwmatk A:n B:n {
 if (B + PROF + STRMOD - 5) >= A {
  result: [gsdmg B] +10 \if hit, result is great sword damage utilizing the attack roll to calculate any crits\
 }
 else {
  result: 0 \if miss, result is 0 damage\
 }
}

\Description ---------------------------------------------\
\This function simulates an attack with a great sword     \
\Variables -----------------------------------------------\
\A = AC of target                                         \
\B = Attack Roll                                          \
\---------------------------------------------------------\

function: gsatk A:n B:n {
 if (B+PROF+STRMOD) >= A {
  result: [gsdmg B]
 }
 else {
  result: 0

}

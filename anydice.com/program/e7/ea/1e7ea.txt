AC:               19  \ target AC: attack hits if ATK + all mods >= AC \
LEVEL:            20  \ character's level, can be used for some formulas \
ATK:            1d20  \ attack roll: use 1@2d20 for advantage or 2@2d20 for disadvantage \
ATKABILITY:        5  \ bonus to attack and damage contributed by ability score modifier \
PROFBONUS:         6  \ proficiency bonus added to attack roll and some damage rolls \
ATKMOD:            0  \ modifier, either positive or negative, to attack rolls for base and bonus attacks \
MAGICBONUS:        0  \ bonus from magical weapons, usually +1-3 \
CRIT:             20  \ minimum attack roll to crit \
CRITMULTI:         2  \ number of times to roll damage dice on a crit \
DMG:               0  \ weapon damage die rolled on successful hit for base attacks \
BONUSDMG:          0  \ weapon damage die rolled on successful hit for bonus attacks \
DMGMODS:           0  \ integer modifiers added to damage rolls for base attacks \
BONUSDMGMODS:      0  \ integer modifiers added to damage rolls for bonus attacks \
DMGADDLDICE:       0  \ extra dice added on top of DMG to each damage roll \
BONUSDMGADDLDICE:  0  \ extra dice added on top of DMG to each damage roll for bonus attacks \
DMGOPTDICE:        0  \ extra dice added on top of total damage once per turn \
DMGOPTMODS:        0  \ integer modifiers added to total damage once per turn \
CRITDMGDICE:       0  \ bonus damage dice added when you trigger a critical hit (damage is doubled) \
CRITDMGMODS:       0  \ integer modifiers added to critical hits \
CRITOPTDICE:       0  \ once per turn, bonus damage dice added when you trigger a critical hit (damage is doubled) \
CRITOPTMODS:       0  \ once per turn, integer modifiers added to critical hits \
RAGEDMGMOD:        4  \ bonus damage from rage for barbarians \
CHAMPIONCRIT:     17  \ champions score a critical hit on a roll of this or higher \
KENSEIACCURACY:    0  \ rerolls the first missed attack if set to 1 \

MADMG:             0  \ the damage dice that are used when making monk unarmed strikes \
MADMGMODS:         0  \ the integer modifiers that are added to monk unarmed strikes, unaffected by critical hits \

SMITEDMG:        5d8  \ the extra damage added by paladin's divine smite \
SCAGTRIPDMG:     3d8  \ the extra damage dealt by using a SCAGtrip (Green-Flame Blade, Booming Blade) \
SNEAKDICE:      10d6  \ the extra dice rolled for rogue's sneak attack \
LIFEDRINKINVO:     1  \ for warlock - 1 if lifedrinker is on, 0 if lifedrinker is off \
BRUTALMULTI:       3  \ number of barbarian's brutal critical dice \
BRUTALDIE:         0  \ die that is rolled for barbarian's brutal critical for base attacks \
BONUSBRUTALDIE:    0  \ die that is rolled for barbarian's brutal critical for bonus attacks \
GWMASTER:          0  \ grants a bonus attack on a base attack crit which overrides other bonus attacks \

set "maximum function depth" to 20

function: attackroll ROLL:n {
	\ return the number of times damage is applied on an attack roll of ROLL \
	if ROLL <= 1 { result: 0 }     \ natural 1 is always a miss \
	if ROLL >= CRIT { result: CRITMULTI }  \ critical hit multiplies damage dice \
	result: ROLL + ATKABILITY + PROFBONUS + ATKMOD + MAGICBONUS >= AC
}

ATKRESULT: [attackroll ATK]  \ optimization: precalculate the miss/hit/crit probabilities \
BONUSATKRESULT: [attackroll ATK] \ allows for different attack mod for TWF attacks \
MAATKRESULT: [attackroll ATK] \ allows for different attack mod for monk martial arts attacks \

function: baseattack N:n times and bonusattack M:n times {
	result: [roll ATKRESULT to baseattack N times and bonusattack M times]
}

function: roll BASEDMGMULTI:n to baseattack N:n times and bonusattack M:n times {
	ATKS: N
	BATKS: M
	TOTALDMG: 0
	GWMATK: 0
	if ATKS > 0 {
		if BASEDMGMULTI > 0 {
			if BASEDMGMULTI > 1 {
				if GWMASTER > 0 & GWMATK < 1 { 
					GWMATK: 1
                                        ATKS: ATKS + 1
				}
				TOTALDMG: TOTALDMG + BASEDMGMULTI d (CRITOPTDICE) + [lowest of 1 and BASEDMGMULTI] * (CRITOPTMODS)
			}
			TOTALDMG: TOTALDMG + BASEDMGMULTI d (DMGOPTDICE) + [lowest of 1 and BASEDMGMULTI] * (DMGOPTMODS)
		} else if KENSEIACCURACY > 0 {
                        ATKS: ATKS + 1
                }
		loop I over {1..ATKS} {
			TOTALDMG: TOTALDMG + [roll ATKRESULT to baseattack]
		}
	}
	if BATKS > 0 {
		if GWMATK < 1 {
			loop I over {1..BATKS} {
				TOTALDMG: TOTALDMG + [roll BONUSATKRESULT to bonusattack]
			}
		}
	}
        if MASTRIKES > 0 {
                loop I over {1..MASTRIKES} {
                        TOTALDMG: TOTALDMG + [roll MAATKRESULT to maattack]
                }
        }
	result: TOTALDMG
}

function: roll ATKDMGMULTI:n to baseattack {
	ATKTOTALDMG: 0
	if ATKDMGMULTI > 0 {
		if ATKDMGMULTI > 1 {
			ATKTOTALDMG: ATKTOTALDMG + ATKDMGMULTI d (CRITDMGDICE) + [lowest of 1 and ATKDMGMULTI] * (CRITDMGMODS) + MAGICBONUS + BRUTALMULTI d BRUTALDIE
		}
		ATKTOTALDMG: ATKTOTALDMG + ATKDMGMULTI d (DMG + DMGADDLDICE) + [lowest of 1 and ATKDMGMULTI] * (DMGMODS)
	}
	result: ATKTOTALDMG
}

function: roll BONUSDMGMULTI:n to bonusattack {
	BONUSTOTALDMG: 0
        if BONUSDMGMULTI > 0 {
	        if BONUSDMGMULTI > 1 {
		        BONUSTOTALDMG: BONUSTOTALDMG + BONUSDMGMULTI d CRITDMGDICE + [lowest of 1 and BONUSDMGMULTI] * CRITDMGMODS + MAGICBONUS + BRUTALMULTI d BONUSBRUTALDIE
	        }
	        BONUSTOTALDMG: BONUSTOTALDMG + BONUSDMGMULTI d (BONUSDMG + BONUSDMGADDLDICE) + [lowest of 1 and BONUSDMGMULTI] * (BONUSDMGMODS)
        }
	result: BONUSTOTALDMG
}

function: roll MADMGMULTI:n to maattack {
	MATOTALDMG: 0
        if MADMGMULTI > 0 {
	        if MADMGMULTI > 1 {
		        MATOTALDMG: MATOTALDMG + MADMGMULTI d CRITDMGDICE + [lowest of 1 and MADMGMULTI] * CRITDMGMODS
	        }
	        MATOTALDMG: MATOTALDMG + MADMGMULTI d (MADMG) + [lowest of 1 and MADMGMULTI] * (MADMGMODS)
        }
	result: MATOTALDMG
}

function: reroll NEWDIE:d for ORIGDIE:n { \ Great Weapon Fighting damage die rerolls on 1 or 2 \
	if ORIGDIE <= 2 { result: NEWDIE }
	result: ORIGDIE
}

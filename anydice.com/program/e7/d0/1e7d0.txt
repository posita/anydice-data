
function: investigate ROLL:n {
  if ROLL < 6 {
    result : 5
  }
  else if ROLL < 11 {
    result : 10
  }
  else if ROLL < 16 {
    result : 25
  }
  else if ROLL < 21 {
    result : 50
  }
  else if ROLL < 26 {
    result : 75
  }
  result: 100
}

function: performance ROLL:n TOTAL:n {
  if ROLL < 6 {
    result : TOTAL/2
  }
  else if ROLL < 11 {
    result : (TOTAL * 3) / 4
  }
  else if ROLL < 16 {
    result : TOTAL
  }
  else if ROLL < 21 {
    result : (TOTAL * 5) / 4
  }
  else if ROLL < 26 {
    result : (TOTAL * 3) / 2
  }
  result: TOTAL * 2
}

function: persuasion ROLL:n TOTAL:n {
  if ROLL < 6 {
    result : TOTAL/2
  }
  else if ROLL < 11 {
    result : (TOTAL * 3) / 4
  }
  else if ROLL < 16 {
    result : TOTAL
  }
  else if ROLL < 21 {
    result : (TOTAL * 5) / 4
  }
  else if ROLL < 26 {
    result : (TOTAL * 3) / 2
  }
  result: TOTAL * 2
}

\ output [investigate ([highest 1 of 2d20]+8)] \
\ output [performance 1d20+6 [investigate ([highest 1 of 2d20]+8)]] \
output [persuasion 1d20+6 [performance 1d20+6 [investigate ([highest 1 of 2d20]+8)]]] named "Reporting"

function: check ROLL:n vs DC:n {
  if ROLL >= DC {
    result: 50
  }
  result: 0
}

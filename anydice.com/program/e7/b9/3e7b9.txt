HIT: 4
function: rollhits VALUE:n {
    TOTAL_HITS: 0
    
    if VALUE >= HIT {
        TOTAL_HITS: TOTAL_HITS + 1
    }
    if VALUE >= 6 {
            TOTAL_HITS: TOTAL_HITS + 1d3
    }
    result: TOTAL_HITS
}
function: rolllllhits {
    result:[rollhits d6] + [rollhits d6]
}

output [rolllllhits]

function: handType of LIST {
  FREQS: []
  loop N over LIST {
    FREQS: FREQS + [COUNT of N in LIST]
  }

  PAIRS: 0
  THREE: 0

  loop N over FREQS {
    if N = 2 {
      PAIRS: PAIRS + 1
    }
    if N = 3 {
      THREE: THREE + 1
    }
  }

  if THREE = 1 {
    if PAIRS >= 1 {
      result: 5  \ Full house
    } else {
      result: 4  \ Three of a kind
    }
  } else {
    if PAIRS >= 2 {
      result: 3  \ Two pairs
    } else {
      if PAIRS = 1 {
        result: 2  \ One pair
      } else {
        result: 1  \ All different
      }
    }
  }
}

output [handType of [d6, d8, d10, d12]] named "Hand Type"

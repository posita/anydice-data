function: attack ATTACK:n AC:n DAMAGE:n {
  if ATTACK >= AC + 10{
    result: DAMAGE * 2
  }
  if ATTACK >= AC {
    result: DAMAGE
  }

  result: 0
}

function: save DC:n SAVE:n DAMAGE:n {
  if(SAVE < DC - 10){
    result: DAMAGE * 2
  }
  if(SAVE < DC) {
    result: DAMAGE
  }
  if(SAVE < DC + 10) {
    result: 0
  }

  result: DAMAGE / 2
}

AC: 17
SAVEMOD: 4

output [attack 1d20+7 AC (1d6+4+1d6+2)] named "Longsword Gouging Claw"

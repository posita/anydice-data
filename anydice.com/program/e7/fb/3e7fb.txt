function: ascend DIE:n on ORDER:n {
if DIE = 4 { result: 3 }
else if DIE = ORDER { result: DIE - 1 + [ascend d(DIE - 2) on (ORDER - 2)] }
else { result: DIE }
}

function: roll A:n and B:n {
result: [ascend dA on A] + [ascend dB on B]
}

output [roll 4 and 4] named "d4+d4"
output [roll 6 and 6] named "d6+d6"
output [roll 8 and 8] named "d8+d8"
output [roll 10 and 10] named "d10+d10"
output [roll 12 and 12] named "d12+d12"

function: damage DIE:n MOD:n CRIT:n CHANCE:n {
 if(CRIT <= CHANCE) {
  result: 2dDIE + MOD
 }
 else {
  result: 1dDIE + MOD
 }
}

BASECRIT: 10

PUNCH: [damage 4 0 1d100 BASECRIT]
SWORD: [damage 6 0-1 1d100 10+BASECRIT]
MACE: [damage 8 0-2 1d100 15+BASECRIT]

output PUNCH named "Punch Damage"
output SWORD named "Sword Damage"
output MACE named "Mace Damage"

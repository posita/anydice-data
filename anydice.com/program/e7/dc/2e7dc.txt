function: attack ATTACK:n ATTACKMOD:n AC:n DAMAGE:n {
  if ATTACK = 20 {
    result: DAMAGE * 2
  } 
  if ATTACK + ATTACKMOD >= AC + 10{
    result: DAMAGE * 2
  }
  if ATTACK + ATTACKMOD >= AC {
    result: DAMAGE
  }

  result: 0
}

function: save DC:n SAVE:n DAMAGE:n {
  if(SAVE < DC - 10){
    result: DAMAGE * 2
  }
  if(SAVE < DC) {
    result: DAMAGE
  }
  if(SAVE < DC + 10) {
    result: 0
  }

  result: DAMAGE / 2
}

AC: 17
SAVEMOD: 4

output [attack 1d20 6 AC (1d4+3+1d6+1)] named "Unarmed Gouging Claw"
output [attack 1d20 6 AC 1d4+3] + [save 14 1d20+SAVEMOD 1d4+1] named "Unarmed Electric Arc"

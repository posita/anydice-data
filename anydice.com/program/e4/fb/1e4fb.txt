
function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

output [reroll {1,2,3,4,5} on d10] named "6+"
output [reroll {1,2,3,4,5,6} on d10] named "7+"
output [reroll {1,2,3,4,5,6,7} on d10] named "8+"
output [reroll {1,2,3,4,5,6,7,8} on d10] named "9+"
output [reroll {1,2,3,4,5,6,7,8,9} on d10] named "10"

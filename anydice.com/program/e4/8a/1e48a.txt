set "explode depth" to 10

function: murk HEAT:d over THRESHDIE:d {
    TMAX: [maximum of THRESHDIE]
    EXPLODED: [explode HEAT >= THRESHDIE]
    result: [flood HEAT]
}

function: flood H:n {
    if H < 4 { result: 0 }
    result: 1 + EXPLODED
}

output 1d12
output 1d8+1d4
output 2d6

output [murk 1d6 over 1d20] named "Very low heat"
output [murk 1d8 over 1d20] named "Low heat"
output [murk 1d10 over 1d20] named "Moderate heat"
output [murk 1d12 over 1d20] named "High heat"
output [murk 1d16 over 1d20] named "Very high heat"
output [murk 1d20 over 1d20] named "Extremely high heat"

function: attack ROLL:n vs AC:n {
 if ROLL = 1 { result: 0 }
 if ROLL = 20 { result: 3 }
 if ROLL + ATTACK >= AC { result: 2 }
 if ROLL + ATTACK < AC { result: 1 }
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = 3 {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = 2 {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

PROF: 6
CHR: 5
ATTACK: PROF + CHR
HITDAMAGE: d10 + CHR + PROF
CRITICALDAMAGE: 1d10 + 10 + CHR + PROF
MISSDAMAGE: 0
output [attack d20 vs d{5..25}] named "Attack Roll [0=CritFail, 1=Miss, 2=Hit, 3=Crit]"
output 4d[damage for [attack d20 vs d{5..25}]] named "Damage Roll"

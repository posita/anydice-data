function: inspire BREAK:s on ROLL:n to INSPIRE:d {
    if [BREAK contains ROLL] {
        result: [highest of ROLL and INSPIRE]
    } else {
        result: ROLL
    }
}
function: inspire BREAK:s on DICE:d {
    result: [inspire BREAK on DICE to DICE]
}

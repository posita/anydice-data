function: highestsum A:n on AX:n B:n on BX:n C:n on CX:n D:n on DX:n {
  POOL: {A, B, C, D}
  POOLX: {AX, BX, CX, DX}
  loop N over {1..#POOL} {
    if N @ POOL = N @ POOLX & N @ POOLX < 20 {
      if N @ POOLX < 12 {
        XX: N @ POOLX + 2
        X: 1dXX
        POOL: {POOL, X}
        POOLX: {POOLX, XX}
      }
      else {
        XX: 20
        X: 1d20
        POOL: {POOL, X}
        POOLX: {POOLX, XX}
      }
    }
  }
  result: {1,2} @ [sort POOL]
}

output {1,2} @ [highestsum 1d10 on 10 0 on 8 0 on 8 0 on 8]


function: attack ROLL:n vs DEFENSE:n {
  if ROLL=20 { result: HITDAMAGE+HITDAMAGE-DEX+20 }
  else if ROLL+ATTACK >= DEFENSE { result: HITDAMAGE }
  else { result: 0 }
}


function: magicattack ROLL:n vs DEFENSE:n {
  if ROLL>=19 { result: HITDAMAGE+HITDAMAGE-DEX+20 }
  else if ROLL+ATTACK >= DEFENSE { result: HITDAMAGE }
  else { result: 0 }
}

loop AC over {10, 15, 20} {
  DEX: 3
  ATTACK: DEX+2+2-5
  HITDAMAGE: 1d6 + 1d10 + DEX
  output [attack d20 vs AC] + [attack d20 vs AC] named "CBow Normal+2 vs [AC] AC"

  DEX: 3
  ATTACK: DEX+2+2-5
  HITDAMAGE: 1d6 + 1d10 + DEX+10
  output [attack d20 vs AC] + [attack d20 vs AC] named "CBow Sharpshooter+2 vs [AC] AC"

  DEX: 3
  ATTACK: DEX+2-5
  HITDAMAGE: 1d6 + 1d10 + DEX
  output [attack d20 vs AC] + [attack d20 vs AC] named "CBow Normal vs [AC] AC"

  DEX: 3
  ATTACK: DEX+2-5
  HITDAMAGE: 1d6 + 1d10 + DEX+10
  output [attack d20 vs AC] + [attack d20 vs AC] named "CBow Sharpshooter vs [AC] AC"
}

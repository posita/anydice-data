loop N over {6} {
  loop ATR over {0} {
    output ATR + [lowest 1 of 5dN]  named "Menor de 5D[N], Dif -4"
    output ATR + [lowest 1 of 4dN]  named "Menor de 4D[N], Dif -3"
    output ATR + [lowest 1 of 3dN]  named "Menor de 3D[N], Dif -2"
    output ATR + [lowest 1 of 2dN]  named "Menor de 2D[N], Dif -1"

    output ATR + [middle 1 of 3dN]  named "Middle of 3D[N], Dif"
    output dN  named "D[N], Dif"

    output ATR + [highest 1 of 2dN]  named "Mejor de 2D[N], Dif +1"
    output ATR + [highest 1 of 3dN]  named "Mejor de 3D[N], Dif +2"
    output ATR + [highest 1 of 4dN]  named "Mejor de 4D[N], Dif +3"
    output ATR + [highest 1 of 5dN]  named "Mejor de 5D[N], Dif +4"
  }
}

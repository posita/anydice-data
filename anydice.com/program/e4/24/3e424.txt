function: ironsworn check A:s B:s C:s {
  COUNT: 0
  if A >= B { COUNT: COUNT + 1 }
  if A >= C { COUNT: COUNT + 1 }
  result: COUNT
}

function: spider check A:s B:s C:s {
  COUNT: 0
  if A < B { COUNT: COUNT + 1 }
  if A < C { COUNT: COUNT + 1 }
  result: COUNT
}

function: highestexplode NUMBER:n of ROLLS:n size SIZE:n {

function: explode DIE:d {
 MAX: [maximum of DIE]
 result: [explode DIE helper]
}

function: explode N:n helper {
 if N = MAX { result: N + [explode DIE helper] }
 result: N
}

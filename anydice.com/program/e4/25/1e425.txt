
function: regular ROLL:n vs AC:n {
	DAMAGE: WEAPON + STRENGTH + STYLE
	CRITICAL: WEAPON + WEAPON + STRENGTH + STYLE

	if ROLL = 20 { result: CRITICAL }
	if ROLL + ATTACK >= AC { result: DAMAGE }
	result: 0
}

function: regular attacks vs AC:n {
	result: ([regular d20 vs AC] + [regular d20 vs AC] + [regular d20 vs AC])
}

function: true strike attacks vs AC:n {
	result: ([regular [highest 1 of 2d20] vs AC] + [regular [highest 1 of 2d20] vs AC])
}

AC: 20
STRENGTH: 3
PROFICIENCY: 2
STYLE: 2
ATTACK: PROFICIENCY + STRENGTH

WEAPON: 1d8 output [regular attacks vs AC]		named "Regular 1d8 vs AC 20 - 3 Attacks"
WEAPON: 1d8 output [true strike attacks vs AC]		named "True strike 1d8 vs AC 20 - 2 Attacks"

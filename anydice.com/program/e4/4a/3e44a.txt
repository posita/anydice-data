function: attack ROLLS:s plus MOD:n against AC:n crit CRIT:d hit HIT:d add ADD:n {
    HIGHEST: (1 @ ROLLS)
    ATK: HIGHEST + MOD
    if HIGHEST >= 19{
        result: CRIT + ADD
    }
    else {
        if ATK >= AC {
            result: HIT + ADD
        }
        else {
            result: 0
        }
    }
}

function: reroll lowest of ROLL:s as REROLL:d if less than MIN:n {
  if ((#ROLL)@ROLL) >= MIN {
     result: ROLL + 0
  } else {
    result: (({1..#ROLL-1}@ROLL) + REROLL)
  }
}

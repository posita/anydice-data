DC:17 
MOD:5
LVL:3

function: save SAVE:n damage DMG:n plus ADMG:n above SLVL:n {
 TDMD: DMG + (ADMG * (LVL - SLVL))
 if SAVE < DC { result: TDMD}
 result: (TDMD)/2
}

output [save d20+MOD damage 8d6 plus 1d6 above 3] named "Fireball"
output [save d20+MOD damage 3d8 plus 1d8 above 2] named "Shatter"

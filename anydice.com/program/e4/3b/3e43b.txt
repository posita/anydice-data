function: roll DICE:n dice at tn TN:n critting at CRIT:n {
 TARGET: {1..TN}
 CRITRANGE: {1..CRIT}
 DICEROLLS: {1..DICE}
 COUNT: 0
 loop P over DICEROLLS {
  COUNT: COUNT + [count {TARGET, CRITRANGE} in d20]
 }
 result: COUNT
}

function: roll DICE:n dice with mishap tn at MISHAP:n {
 MISHAPRANGE: {MISHAP..20}
 if MISHAP = 20 { 
  MISHAPRANGE: 20
 }
 DICEROLLS: {1..DICE}
 COUNT: 0
 loop P over DICEROLLS {
  COUNT: COUNT + [count {MISHAPRANGE} in d20]
 }
 result: COUNT
}

output [roll 5 dice at tn 12 critting at 4] named "5d20 TN 13 Crit 4 Successes"
output [roll 2 dice at tn 10 critting at 1] named "4d20 TN 12 Crit 3 Successes"
output [roll 2 dice at tn 10 critting at 1] named "3d20 TN 11 Crit 2 Successes"

output [roll 5 dice with mishap tn at 17] named "5d20 M17 Mishaps"
output [roll 4 dice with mishap tn at 18] named "4d20 M18 Mishaps"

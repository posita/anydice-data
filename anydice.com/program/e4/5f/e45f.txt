
function: attack ROLL:n vs DEFENSE:n {
  if ROLL=20 { result: HITDAMAGE+HITDAMAGE-DEX }
  else if ROLL+ATTACK >= DEFENSE { result: HITDAMAGE }
  else { result: 0 }
}

loop AC over {10, 15, 20} {
  DEX: 3
  ATTACK: DEX+1
  HITDAMAGE: 1d8+ 1d4 + DEX
  output [attack d20 vs AC] named "Sharpshooter longbow vs [AC] AC"

  DEX: 3
  ATTACK: DEX
  HITDAMAGE: 1d6 + 1d4 + DEX
  output [attack d20 vs AC] + [attack d20 vs AC] named "Crossbow expert vs [AC] AC"
}

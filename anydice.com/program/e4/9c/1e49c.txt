function: min A B { if A<B { result: A } result: B }
function: datarun POWER ROLL:s {
 if 1@ROLL = 6 { result: POWER + [count {6} in ROLL] }
 if 1@ROLL >= 4 { result: [min POWER [count {1@ROLL} in ROLL]] }
}

loop A over {1..4} {
 output [datarun A 2@2d6] named "[A]p@0d"
 loop S over {1..7} {
  output [datarun A Sd6] named "[A]p@[S]d"
 }
}

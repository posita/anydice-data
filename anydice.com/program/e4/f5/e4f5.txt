ATK: 1d20  \ attack roll plus to-hit bonuses (if any) \
AC:  8     \ target AC: attack hits if HIT >= AC \
DMG: 1d4   \ damage rolled on successful hit \

function: attack N:n times DMGBONUS:n on first hit {
  result: [roll ATK >= AC to attack N times DMGBONUS on first hit]
}
function: roll SUCCESS:n to attack N:n times DMGBONUS:n on first hit {
  if N <= 0 {
    result: 0
  }
  if SUCCESS {
    result: DMG + DMGBONUS + [roll ATK >= AC to attack N-1 times +0 on first hit]
  } else {
    result: [roll ATK >= AC to attack N-1 times DMGBONUS on first hit]
  }
}

output [attack 3 times +2 on first hit]

function: degree of success A:n plus BA:n versus D:n plus BD:n {
  if A + BA >= 21 \| A = 20\ { RP: 2 } else {
    if A + BA < 11 \| A = 2\ { RP: 0 } else { RP: 1 }
  }
  if D + BD >= 21 \| D = 20\ { RO: 2} else {
    if D + BD < 11 \| D = 2\ { RO: 0 } else { RO: 1 }
  }
  R: RP - RO
  if R < 0 { result: 0 }
  if R = 0 { result: 2 }
  if R = 1 { result: 1d10+2 }
  if R = 2 { result: 12 }
\  if R < 0 { result: 0 }
  if R = 0 { result: 3 }
  if R = 1 { result: 1d10 + 2 }
  if R = 2 { result: 12 }\
}

loop M over { 0..10 } {
  output [degree of success 1d20 plus M versus 1d20 plus 0] named "d20+[M]"
\  output [degree of success 2d10 plus M versus 2d10 plus 0] named "2d10+[M]"\
}

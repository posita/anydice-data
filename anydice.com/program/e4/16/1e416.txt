function: dupes in DICE:s {
  D: 0
  loop X over {2..#DICE} {
    if ((X-1)@DICE = X@DICE) { D: D + 1}
  }
  result: D
}

function: triples in DICE:s {
  D: 0
  loop X over {3..#DICE} {
    if ((X-2)@DICE = X@DICE) {
      if ((X-1)@DICE = X@DICE) { D: D + 1}
    }
  }
  result: D
}

output [triples in 5d6]

DEXTERITY: 5
STRENGTH: 5

function: thw {
   result: 2d6 + STRENGTH
}

function: thw fs {
   DICEONE: 1d6
   DICETWO: 1d6

   if (DICEONE = 1) {
      DICEONE: 1d6
   }

   if (DICETWO = 1) {
      DICETWO: 1d6
   }


   result: DICEONE + DICETWO + STRENGTH
}


function: twf {
 result: 1d6 + DEXTERITY
}

function: twf offhand {
 result: 1d6
}

output [thw] named "Two Handed Weapon"
output [thw fs] + [thw fs] named "Two Handed Weapon Fighting Style"

output [thw] + [thw] named "Two Handed Weapon (2x)"
output [thw] + [thw] named "Two Handed Weapon Fighting Style (2x)"

output [twf] + [twf offhand] named "Two Handed Weapon"
output [twf] + [twf] + [twf offhand] named "Two Handed Weapon (2x)"

function: convert SUM:n {
  if SUM >= 1000{
    TENSROLLED: SUM / 1000
    result: SUM - TENSROLLED * 990 + TENSROLLED d [explode d10]
  }
  result: SUM
}

function: ROLL:n k KEEP:n {
  result: [convert [highest KEEP of ROLL d {1..9, 1000}]]
}

loop POOL over {1..10} {
  output [POOL k1] named "[POOL]k1"
}

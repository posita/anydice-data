function: gionata DICE:s
{
   if (1@DICE = 3 & 2@DICE = 3)
   |  (1@DICE = 4 & 2@DICE = 3)
   |  (1@DICE = 4 & 2@DICE = 4)
      {result: 1}
   else 
      {result: 1@DICE + 2@DICE}
}

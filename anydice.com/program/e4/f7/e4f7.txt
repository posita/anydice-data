ATK: 1d20  \ attack roll plus to-hit bonuses (if any) \
AC:  8     \ target AC: attack hits if HIT >= AC \
DMG: 1d4   \ damage rolled on successful hit \

function: attack N:n times DMGBONUS:n on first hit {
  result: [actually attack N - [miss ATK >= AC max N] times DMGBONUS on first hit]
}
function: miss SUCCESS:n max N:n {
  if SUCCESS | (N <= 0) { result: 0 }
  result: 1 + [miss ATK >= AC max N-1]
}
function: actually attack N:n times DMGBONUS:n on first hit {
  if N <= 0 { result: 0 }
  TOTAL: DMGBONUS + DMG  \ first roll always hits \
  loop I over {2..N} {
    TOTAL: TOTAL + DMG * (ATK >= AC)
  }
  result: TOTAL
}

output [attack 3 times +2 on first hit]


DICE_RANGE: {2, 4, 6, 8, 10, 12, 20}
function: adjust DC:n with v N:n {
  result: DC - (dN - (N/2))
}


\ Loop through M modifiers to the skill check \
DC: 10
loop M over {1..DC} { 
  loop VAR over DICE_RANGE {
    \output [adjust DC with v VAR] named "Adjusted DC[DC] with Var[VAR]"\
    output (d20 + M) >= [adjust DC with v VAR] named "Roll with +[M] vs DC10 Var[VAR]"
  }
}

\Description -----------------------------------------------------------------------------------\
\This function returns the re-rolled value of the die if it is lower or equal to the lower limit\
\Variables -------------------------------------------------------------------------------------\
\A = Initial input value. I.E. 1 to 6 for d6                                                    \
\B = Lower reroll limit. I.E. 2 for Great Weapon Fighter                                        \
\C = Dice value. I.E. 6 for d6                                                                  \
\-----------------------------------------------------------------------------------------------\

function: reroll A:n B:n C:n{ 
 if A <= B { 
  A: 1dC 
 }  
 result: A 
}

\Description ---------------------------------------------\
\This function simulates the Great Weapon Master feat     \
\Variables -----------------------------------------------\
\A = Dice value I.E. A=6 for 1d6                          \
\B = Number of damage dice I.E. B = 2 for Greatsword (2d6)\
\C = Lower limit for reroll. Default 2                    \
\---------------------------------------------------------\

function: greatweaponmaster A:n B:n C:n{
 X:0
 loop N over {1..B} {
  X: X + [reroll 1dA C A]
 }
 result: X
}

\Description -----------------------------------------------------------------\
\This function returns 2 if the user crits 1 For a nomral hit and 0 for a miss\
\Variables -------------------------------------------------------------------\
\A = Dice roll. Default 1 to 20 (1d20)           					          \
\B = Minimum critical roll. Default 20                                        \  
\C = to Hit Modofier                                 					      \
\D = Armour Class of Enemy                                   			      \                 
\-----------------------------------------------------------------------------\

function: getcritorhit A:n B:n C:n D:n { 
	if A >= 20 { 
		result:2 
	}else {
		if A+C >= D{
			result:1
		}else {
			
			result:0 
		} 
	}
}

MUL:[getcritorhit 1d20 CRIT TOHITMOD ARMOURCLASS]
\Description -----------------------------------------------------------------\
\This function returns 1 if the user hits the enemy 0 For a miss			  \
\Variables -------------------------------------------------------------------\
\A = the multiplier that is checked to be greater than 0			          \               
\-----------------------------------------------------------------------------\

function: modhit A:n{
	if A >0{
		result: 1
	}else{
		result: 0
	}
}

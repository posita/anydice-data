function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

output  4d[reroll {1,2} on 1d6] + 10d6 + 12d8                     + 5 + 4 + 4 named "Critical: Reroll Weapon Dice Only"
output 14d[reroll {1,2} on 1d6]        + 12d[reroll {1,2} on 1d8] + 5 + 4 + 4 named "Critical: Reroll All Dice"

output  2d[reroll {1,2} on 1d6] +  5d6 +  6d8                     + 5 + 4 + 4 named "No Critical: Reroll Weapon Dice Only"
output  7d[reroll {1,2} on 1d6]        +  6d[reroll {1,2} on 1d8] + 5 + 4 + 4 named "No Critical: Reroll All Dice"

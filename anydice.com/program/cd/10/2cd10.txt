PROF: 4
MOD: 4
MAGIC: 1
AC: 17

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

TG    : [attack [highest 1 of 2d20] plus MOD+PROF+MAGIC vs AC for 1d8 + MOD + MAGIC          crit 1d8 on 20]
TG_PA : [attack [highest 1 of 2d20] plus MOD+     MAGIC vs AC for 1d8 + MOD + MAGIC + 2*PROF crit 1d8 on 20]
CA    : [attack [highest 1 of 2d20] plus MOD+PROF+MAGIC vs AC for 1d6 + MOD                  crit 1d6 on 20]

output 2dTG_PA named "10th level Tazwunax with Thunder Gauntlets and rage (reckless attack)"

function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: reroll REROLL:s on DICE:d {
    result: [reroll REROLL on DICE to DICE]
}

output  4d[reroll {1,2} on 1d6] + 10d6 + 12d8                     + 5 + 4 + 4 named "Reroll Weapon Dice Only"
output 14d[reroll {1,2} on 1d6]        + 12d[reroll {1,2} on 1d8] + 5 + 4 + 4 named "Reroll All Dice"

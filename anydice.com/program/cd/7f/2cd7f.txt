PROF : 4
MOD : 4
MAGIC: 1
BLESS: 1d4
DUELING:2
AC: 17

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n and SMITE:d {
    if ROLL = 1 {
        result: 0
    }
    if ROLL >= CRIT_RANGE {
        result: DMG + CRIT_DMG + 2dSMITE
    }
    if ROLL + BONUS >= AC {
        result: DMG
    }
    result: 0
}

LS : [attack [highest 1 of 1d20] plus MOD+PROF+MAGIC+BLESS vs AC for 1d8 + MOD + MAGIC + DUELING crit 1d8 on 20 and 2d8]
LS_PA   : [attack [highest 1 of 1d20] plus MOD+MAGIC+BLESS vs AC for 1d8 + MOD + MAGIC + DUELING+ 2*PROF crit 1d8 on 20 and 2d8]
 
EB :[attack [highest 1 of 1d20] plus MOD+PROF+BLESS vs AC for 1d10+4 crit 1d10 on 20]



output 2dLS                 named "10th level Relius with Longsword + Bless"
output 2dLS_PA              named "10th level Relius with Longsword + Bless+power attack"
output 2dEB                 named "10th level Relius with Eldritch Blast + Curse"
output 3dEB                 named "11th level Relius with Eldritch Blast + Curse"

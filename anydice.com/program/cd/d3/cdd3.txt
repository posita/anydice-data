function: openlegendhelper A:n S:s D:n {
 R: 0
 if A < 0 {
  START: 1-A
  END: #S
 }
 else {
  START: 1
  END: #S-A
 }
 loop I over {START..END} {
  R: R + I@S
  if I@S = D { R: R + [explode 1dD] }
 }
 result: R
}

function: openlegend L:n A:n {
 if A < 0 { N: 1-A }
 else { N: 1+A }
 if L = 0 {
   R: 0
   D: 20
 }
 else {
  R: [explode 1d20]
  if L > 4 {
   if L > 7 { 
    N: N+2
    D: (L*2)-8
   }
   else { 
    N: N+1
    D: (L*2)-4
   }
  }
  else { D: (L*2)+2 }
 }
 result: R + [openlegendhelper A NdD D]
}

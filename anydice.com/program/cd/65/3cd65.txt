set "maximum function depth" to 5

X: d{-2, -1, 0, 1, 2, 3}

function: rollp Y:n {
  if (Y > 2) {
    result: [rollp X] + [rollp X]
  }
  else {
    result: Y
  }
}

function: rollm Y:n {
  if (Y > 2) {
    result: [rollm X] * 2
  }
  else {
    result: Y
  }
}



output 1d6 - 1d6 named "1d-1d"
output [rollp (X)] + [rollp (X)] named "2x p"
output [rollm (X)] + [rollm (X)] named "2x m"

output 3d6-10 named "3d-10"
output [rollp (X)] + [rollp (X)] + [rollp (X)] named "3x p"
output [rollm (X)] + [rollm (X)] + [rollm (X)] named "3x m"

output 2d6-2d6 named "2d-2d"
output [rollp (X)] + [rollp (X)] + [rollp (X)] + [rollp (X)] named "4x p"
output [rollm (X)] + [rollm (X)] + [rollm (X)] + [rollm (X)] named "4x m"

function: attackroll ROLL:n vs AC:n with DMG:n and CRITDMG:n {
 if ROLL >= 20 { result: CRITDMG} 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD >= AC { result: DMG}
 result: 0
}

function: advattackroll vs AC:n with DMG:n and CRITDMG:n{
 result: [attackroll [highest 1 of 2d20] vs AC with DMG and CRITDMG]
}

function: dmgdice per hit for cme cast at LVL:n {
 result: 2d8 + (LVL-4)d8
}

function: dmgdice per hit for sb cast at LVL:n {
 if (LVL=2) {result: 2d8}
 if (LVL=3 | LVL=4) {result: 3d8}
 if (LVL=5 | LVL=6) {result: 4d8}
 if (LVL >=7) {result: 5d8}
 result: 0
}

function: get std bb dmg with WPN_ATK at TIER:n {
 if (TIER > 4) {result: WPN_ATK + 3d8 + 4dHALFEIGHT}
 result: WPN_ATK + (TIER-1)d8 + TIERdHALFEIGHT
}

function: get crit bb dmg with CRIT_WPN_ATK at TIER:n {
 if (TIER > 4) {result: WPN_ATK + 6d8 + 4dHALFEIGHT}
 result: CRIT_WPN_ATK + 2*(TIER-1)d8 +TIERdHALFEIGHT
}

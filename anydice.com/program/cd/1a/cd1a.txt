function: dndability N {
 result: [highest 3 of Nd6]
}
function: dndabilitydice S {
 R: 0
 loop N over S {
  R: R + [dndability N]
 }
 result: R
}

output [dndabilitydice {4, 4, 4, 4, 4, 4}] named "Standard DnD 5E"
output [dndabilitydice {3, 3, 4, 4, 5, 5}] named "Weighted DnD 5E"
output [dndabilitydice {1, 1, 1, 1, 1, 19}] named "EXTREME DnD 5E"

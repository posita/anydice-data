MAJORSETBACK: -1
MINORSETBACK: 0
PROGRESS: 1


function: check ROLL:n plus TRAIT:n plus SKILL:n vs SR:n and TN:n{
 BONUS: TRAIT
 if SKILL > TRAIT { BONUS: SKILL}
 if ROLL + BONUS >= TN { result: PROGRESS}
 if ROLL = 1 { result: MAJORSETBACK }
 if ROLL + BONUS < TN {
  if ROLL + BONUS <= SR { result: MAJORSETBACK }
  result: MINORSETBACK
 }
}

TD: 4  \number of trait dice\
SD: 0  \number of skill dice\
AD: 1  \number of action dice\
loop SCALE over {1..10} {
 output [check [highest 1 of ADd20] plus [highest 1 of TDd6] plus [highest 1 of SDd10] vs SCALE and SCALE + 10] named "[SCALE]"
}

function: hit TO_HIT:s {
  result: TO_HIT >= BS
}

function: wound TO_WOUND:s {
  if S > T * 2 {
    result: TO_WOUND >= 2
  }
  if S > T {
    result: TO_WOUND >= 3
  }
  if S = T {
    result: TO_WOUND >= 4
  }
  if S < T / 2 {
    result: TO_WOUND >= 6
  }
  if S < T {
    result: TO_WOUND >= 5
  }
}

function: save TO_SAVE:s {
  result: (TO_SAVE) >= [lowest of (AS - AP) and INVS]
}

function: hitme NN:n {
  result: [hit NNd6]
}

function: woundme HH:n {
  result: [wound HHd6]
}

function: savethem SS:n {
  result: SS - [save SSd6]
}

SHOTS: 2
BS: 1
S: 4
T: 4
D: 1
AS: 3
INVS: 7
AP: -4

output [savethem [woundme [hitme SHOTS]]] * D

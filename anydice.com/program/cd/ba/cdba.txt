function: olahelper A:n S:s D:n {
 X: 0
 if A < 0 {
  START: 1-A
  loop I over {START..#S} {
   X: X + I@S
   if I@S = D { X: X + [explode 1dD] }
  }
 }
 else {
  N: #S-A
  loop I over {1..N} {
   X: X + I@S
   if I@S = D { X: X + [explode 1dD] }
  }
 }
 result: X
}

function: openlegend L:n A:n {
 if A < 0 { N: 1-A }
 else { N: 1+A }
 if L = 0 {
   R: 0
   D: 20
 }
 else {
  R: [explode 1d20]
  if L > 4 {
   if L > 7 { 
    N: N+2
    D: (L*2)-8
   }
   else { 
    N: N+1
    D: (L*2)-4
   }
  }
  else { D: (L*2)+2 }
 }
 result: R + [olahelper A NdD D]
}

output [openlegend 0 {-2}] named "Untrained Disadvantage 2"
output [openlegend 0 {-1}] named "Untrained Disadvantage 1"
output [openlegend 0 0] named "Untrained"
output [openlegend 0 1] named "Untrained Advantage 1"
output [openlegend 0 2] named "Untrained Advantage 2"

output [openlegend 9 {-1}] named "Level 9 Disadvantage 1"
output [openlegend 9 0] named "Level 9"
output [openlegend 9 1] named "Level 9 Advantage 1"
output [openlegend 9 2] named "Level 9 Advantage 2"

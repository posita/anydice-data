function: highest N:n of D:d reroll ones eightteen twenty{
MAX: [maximum of D]/#(D)
  RESULT: [highest N of #(D)d(MAX-1)]+N
  result: [helper RESULT]
}

function: helper IN:n {
  if IN = 18 {result: 20}
  else {result: IN}
}

function: highest N:n of D:d reroll ones{
MAX: [maximum of D]/#(D)
  RESULT: [highest N of #(D)d(MAX-1)]+N
  result: RESULT
}

function: calculate ability mod of ABILITY:n {
  if ABILITY < 10 {result: (ABILITY-11)/2}
  else {result: (ABILITY-10)/2}
}

function: one whole stat array made from the highest N:n of D:d{
  TOTAL: 0
  loop I over {1..6} {
    TOTAL: TOTAL + [calculate ability mod of [highest N of D reroll ones]]
  }
  result: TOTAL
}

function: best net modifier of X:n arrays made from the highest N:n of D:d {
  result: [highest 1 of Xd[one whole stat array made from the highest N of D]]
}

function: one whole stat array made from the highest N:n of D:d eightteen twenty{
  TOTAL: 0
  loop I over {1..6} {
    TOTAL: TOTAL + [calculate ability mod of [highest N of D reroll ones eightteen twenty]]
  }
  result: TOTAL
}

function: best net modifier of X:n arrays made from the highest N:n of D:d eightteen twenty{
  result: [highest 1 of Xd[one whole stat array made from the highest N of D eightteen twenty]]
}

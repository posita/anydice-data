function: roll N:n{
 EX: N-10
 result: [lowest of 13 and [highest of 0 and EX]]
}
function: rank RANK:n{
D: 1d20
if RANK = 1 {result: D       }
if RANK = 2 {result: D + 1d4 }
if RANK = 3 {result: D + 1d6 }
if RANK = 4 {result: D + 1d8 }
if RANK = 5 {result: D + 1d10}
if RANK = 6 {result: D + 1d12}
else        {result: 2@2d20}
}

loop R over {0..5}{ 
output [roll [rank R]] named "Rank [R]"
}

function: bonus C:n ECHECS:n MISS:n MISSED:n{
if C=0{
A:-ECHECS}
if C>0 & C<=(ECHECS+MISS){
A:C-ECHECS}
if C> ECHECS+MISS & C<=ECHECS+MISS+MISSED+MISSED{
BONUS:ECHECS+MISS
A:BONUS+C/2-ECHECS}
if C>ECHECS+MISS+MISSED+MISSED{
A:MISS+MISSED}
result:A}


function: concentration D:n competence C:n chance L:n seuil S:n{
POOL:(D+L)d6
SUCCES:[count {S..6} in POOL]
DOUBLE:[count {6} in POOL]
ECHECS:[count {1} in POOL]
MISS:[count {S-1,S+1} in POOL]
MISSED:[count {S-2} in POOL]

result:[lowest of SUCCES+DOUBLE+[bonus C ECHECS MISS MISSED] and D+D+L+L]
}

output [concentration 12 competence 0 chance 0 seuil 4]
output [concentration 11 competence 1 chance 0 seuil 4]
output [concentration 9 competence 2 chance 0 seuil 4]
output [concentration 8 competence 3 chance 0 seuil 4]
output [concentration 7 competence 4 chance 0 seuil 4]
output [concentration 6 competence 5 chance 0 seuil 4]
output [concentration 5 competence 6 chance 0 seuil 4]

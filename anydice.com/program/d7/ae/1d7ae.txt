function: apoc A:n {
   if A >= 10 { result: 10 }
   if A >= 7 { result: 9 }
   result: 6
}

function: apocabs A:n {
   if A >= 7 { result: 10 }
   result: 6
}

function: altdie A:n {
   if A = -1 { result: 1d4 }
   if A = 0 { result: 1d6 }
   if A = 1 { result: 1d8 }
   if A = 2 { result: 1d10 }
   result: 1d12
}

loop MOD over {-1..3}{
   ALTDIE: [altdie MOD]


   output [apoc 2d6+MOD] named "Vanilla [MOD]"
   output [apoc 1d6+ALTDIE] named "Alt [MOD] ([ALTDIE])"
   
}

function: explodeunder DIEROLL:n POINTS:n DIETYPE:n {
 if POINTS > 0 {
   if DIEROLL <= POINTS { result: DIEROLL + 1dDIETYPE }
 }
 result: DIEROLL
}

function: check CHECK:n SUCCESSTHRESHHOLD:n {
 if CHECK >= SUCCESSTHRESHHOLD { result: HIT }
 result: MISS
}

MISS: 0
HIT: 1


loop N over {0..4}{
output [check [explodeunder 1d20 N 20] 12] named "D20 with [N] points"
}

loop N over {0..4}{
output [check [explodeunder 1d12 N 12] 8] named "D12 with [N] points"
}

loop N over {0..4}{
output [check [explodeunder 1d10 N 10] 7] named "D10 with [N] points"
}

loop N over {0..4}{
output [check [explodeunder 1d8 N 8] 6] named "D8 with [N] points"
}

loop N over {0..4}{
output [check [explodeunder 1d6 N 6] 5] named "D6 with [N] points"
}

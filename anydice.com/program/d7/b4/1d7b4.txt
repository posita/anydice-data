function: apoc A:n {
   if A >= 10 { result: 10 }
   if A >= 7 { result: 9 }
   result: 6
}

function: apocabs A:n {
   if A >= 7 { result: 10 }
   result: 6
}

function: altdie A:n {
   if A = -1 { result: 1d4 }
   if A = 0 { result: 1d6 }
   if A = 1 { result: 1d8 }
   if A = 2 { result: 1d10 }
   result: 1d12
}

function: altmain A:n {
   if A = -1 { result: 1d4 }
   if A = 1 { result: 1d8 }
   result: 1d6
}


loop MOD over {-1..3}{
   ALTDIE: [altdie MOD]

loop FWD over {-1..1} {
   MAINDIE: [altmain FWD]

   output [apoc 2d6+MOD+FWD] named "Vanilla [MOD] ([FWD] Forward)"

}

loop FWD over {-1..1} {
   MAINDIE: [altmain FWD]
   output [apoc MAINDIE+ALTDIE] named "Alt [MAINDIE]+[ALTDIE] ([FWD] Forward)"
   
}

}

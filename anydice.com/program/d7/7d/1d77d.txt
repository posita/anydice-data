function: mine ROLL:n {
 if ROLL = 6 {
  result: 1d3
 }
 if ROLL = 1 {
  result: 0
 }
 result: 1
}
function: wounds ATTACKS:n {
 HIT: 0
 loop N over {1..ATTACKS}{
  HIT: HIT + [mine 1d6]
 }
 result: HIT
}
function: meh ROLL:n {
 if ROLL >= 4 {
  result: [wounds 1]
 }
 else {
  result: 0
 }
}
function: hits MEH:n {
 WOUNDS: 0
 loop N over {1..MEH}{
  WOUNDS: WOUNDS + [meh 1d6]
 }
 result: WOUNDS
}
loop N over {1..6} {
 output [hits 2*N] named "[N] biovore"
}

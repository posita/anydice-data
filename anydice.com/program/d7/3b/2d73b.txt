ATK: d20
ATB: d4
DEF: d20
DEB: d4

function: ROLL:s roll {
 result: ROLL
}

ATTACK: [ATK roll]
DEFENSE: [DEF roll]

function: difference {
 result: ATTACK - DEFENSE
}

ACC: [difference]

output ACC named "Attack Accuracy"

function: first hit mATK mDEF hATTACK NUMBER:n {
 if NUMBER > 0 {
  result: (NUMBER - ATB) - (NUMBER - DEB)
  }
}

function: first hit pATK mDEF hATTACK NUMBER:n {
 if NUMBER > 0 {
  result: (NUMBER + ATB) - (NUMBER - DEB)
  }
}

function: first hit pATK pDEF hATTACK NUMBER:n {

  result: (NUMBER + ATB) - (NUMBER + DEB)
  }
}

function: first hit mATK pDEF hATTACK NUMBER:n {
 if NUMBER > 0 {
  result: (NUMBER - ATB) - (NUMBER + DEB)
  }
}

output [first hit mATK mDEF hATTACK ACC] named "First Hit hAttack (-ATK -DEF)"

output [first hit pATK pDEF hATTACK ACC] named "First Hit hAttack (+ATK +DEF)"

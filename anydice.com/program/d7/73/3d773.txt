DICE: {4,6,8,10,12,20}

function: fumble A:n at L:n {
  if A <= L { result: 0 }
  result: A
}

function: check A:d S:n {
  result: [fumble [explode A] at S]
}

function: cost J:n {
  COSTDIE: 0
  loop DIE over DICE {
    COSTDIE: COSTDIE + DIE ^ 2
    if J = DIE { result: COSTDIE - 4^2}
  }
  result: COSTDIE -4^2
}

function: cost N:n by J:n {
  COST: 0
  loop I over {1..N} {
    COST: COST + I * 4
  }
  result: COST + [cost J]
}

function: explodef N:n {
     if N > 3 {
      result: N + [explodef d6]
     }
     result: N
    }
    set "maximum function depth" to 5

function: explodee N:n {
     if N > 4 {
      result: N + [explodee d6]
     }
     result: N
    }
    set "maximum function depth" to 5

set "maximum function depth" to 6
loop Y over {1..6} {
 output (Yd[explodef d6]-Yd6)>=0 named "Y explodef"
}

set "maximum function depth" to 6
loop Y over {1..6} {
 output (Yd[explodee d6]-Yd6)>=0 named "Y explodee"
}

set "maximum function depth" to 6
loop Y over {1..6} {
 output (Yd[explode d6]-Yd6)>=0 named "Y explode"
}

set "maximum function depth" to 6
loop Y over {1..6} {
 output (Yd6-Yd6)>=0 named "Yd6"
}

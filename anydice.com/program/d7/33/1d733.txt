function: A:n on B:n {
if A >= B {result: 1}
result: 0
}

function: A:n on B:n reroll C:n {
if A >= B {result: 1}
if A <= C {result: [d6 on B]}
result: 0
}

function: extra wound A:n strength B {
 if B > 10 & A > 2 { result: d6+d3}
 if B > 10 { result: d3}
 if B > 8 & A >= 2 {result: d6}
 if B = 8 & A >= 3 {result: d6}
 if B > 4 & A >= 4 {result: d6}
 if A >= 5 {result: d6}
 result: 0
}

function: wound A:n strength B {
 if B > 10 { result: d3}
 if B > 8 & A >= 3 {result: d6}
 if B = 8 & A >= 4 {result: d6}
 if B > 4 & A >= 5 {result: d6}
 if A >= 6 {result: d6}
 result: 0
}

function: shot A:n with dakka {
 if A >= 6 {result: 1+[d6 on 6]}
 result: 0
}


function: shot A:n with extra dakka {
 if A >= 5 {result: 1+[d6 on 5 reroll 1]}
 result: 0
}

function: shot A:n with extra dakka and reroll {
 if A >= 5 {result: 1+[d6 on 5 reroll 1]}
 if A = 1 {result: [shot d6 with extra dakka]}
 result: 0
}


function: soupedup shoot A:n strength B:n {
 result: Ad[shot d6 with extra dakka and reroll]d[extra wound d6 strength B]
}

function: shoot A:n strength B:n {
 result: Ad[shot d6 with dakka]d[wound d6 strength B]
}

output 1d[shoot 1d6 strength 2d6]
output 2d[soupedup shoot 2d6 strength 2d6]

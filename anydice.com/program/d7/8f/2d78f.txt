function: LEFT:s covers RIGHT:s
{
  if (#LEFT < #RIGHT) {  \ RIGHT is longer than LEFT \
    result: 0
  } else {
    A: [sort LEFT]
    B: [sort RIGHT]
    FAILED: 0
    loop N over {1..#LEFT} {
      if (N@A < N@B) {
        FAILED: 1
      }
    }
    result: 1 - FAILED
  }
}

output [4d6 covers {4, 3, 2}] named "4d6 to get 4+ 3+ 2+"
output [4d6 covers {6, 4, 2}] named "4d6 to get 6+ 4+ 2+"
output [4d6 covers {4, 4, 2, 2}] named "4d6 to get 4+ 4+ 2+ 2+"

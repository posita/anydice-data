function: standard N:n {
if N>4 {result: 1}
else {result: 0}
}

function: edge N:n {
if N>5 {result: 1 + [edge 1d6]}
else if N>4 {result: 1}
else {result: 0}
}

function: edgereroll N:n {
if N>4 {result: 1}
else {result: [edge 1d6]}
}

output 1d[standard 1d6]
output 1d[edge 1d6]
output 1d[edgereroll 1d6]



MONK_LEVEL: 5

function: fall damage from H:n {
  REDUCTION: 0
  if MONK_LEVEL >= 4 {
    REDUCTION: 5 * MONK_LEVEL
  }
  DAMAGE: [highest of 0 and (([lowest of H and 200]/10)d6 - REDUCTION)]
  result: DAMAGE
}


loop HEIGHT over {(1d20 - 1) * 10} {
  output [fall damage from HEIGHT] named "[HEIGHT] feet"
}

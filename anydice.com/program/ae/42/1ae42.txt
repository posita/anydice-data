\ output [ATTACKS vs AC crit THREAT MULT CRITEXTRA CRITDIE dmg DAMAGE EXTRA] named "My Weapon"
This function calculates how much damage you do on an attack
ATTACKS is your attack bonus (e.g. 10), this can include iterative attacks (e.g. {6,1})
AC is the opponent's armor class (e.g. 17)
THREAT is the lower number of the critical threat range (e.g. 19 if your threat is 19-20)
MULT is your critical multiplier (e.g. 2)
CRITEXTRA is extra damage dice only on a critical hit (e.g. 1d10 for a flaming burst sword) otherwise leave blank
CRITDIE is the alternate die to use for crits, e.g. Fatal d10
DAMAGE is the damage you deal with your weapon not including extra dice (e.g. 1d8+2)
EXTRA is extra damage dice (e.g. 1d6 for a sneak attack) otherwise leave blank \

function: ATTACKS:s vs AC:n crit THREAT:n MULT:n CRITEXTRA:d CRITDIE:d dmg DAMAGE:d EXTRA:d {
  SUM: {0}
  loop ATTACK over ATTACKS {
    SUM: SUM + [1d20 plus ATTACK ac AC crit THREAT MULT CRITEXTRA CRITDIE dmg DAMAGE EXTRA]
  }
  result: SUM
}
\standard attack\
function: ATTACKS:s vs AC:n crit THREAT:n MULT:n dmg DAMAGE:d {
  result: [ATTACKS vs AC crit THREAT MULT 0 DAMAGE dmg DAMAGE 0]
}
\standard attack with extra crit effect\
function: ATTACKS:s vs AC:n crit THREAT:n MULT:n CRITEXTRA:d dmg DAMAGE:d {
  result: [ATTACKS vs AC crit THREAT MULT CRITEXTRA DAMAGE dmg DAMAGE 0]
}
\attack with extra non crit multiplied damage\
function: ATTACKS:s vs AC:n crit THREAT:n MULT:n dmg DAMAGE:d EXTRA:d{
  result: [ATTACKS vs AC crit THREAT MULT 0 DAMAGE dmg DAMAGE EXTRA]
}

\result of a roll, 0-3: crit fail, fail, success, crit success\
function: result ROLL:n MOD:n DC:n {
  if (ROLL + MOD < DC-10) {
     RES: 0
  } else if(ROLL + MOD < DC) {
     RES: 1
  } else if(ROLL + MOD < DC + 10) {
     RES: 2
  } else {
     RES: 3
  }
  if (ROLL = 1) { RES: RES-1 }
  if (ROLL = 20) { RES: RES+1 }
  if (RES < 0) { result: 0 }
  else if (RES > 3) { result: 3 }
  else { result: RES }
}
 
\individual roll calculation\
function: ROLL:n plus ATTACK:n ac AC:n crit THREAT:n MULT:n CRITEXTRA:d CRITDIE:d dmg DAMAGE:d EXTRA:d {
  RES: [result ROLL ATTACK AC]
  if (RES = 3) { result: [crit MULT CRITDIE] + EXTRA + CRITEXTRA } \crit damage\
  else if (RES = 2) {result: (ROLL + ATTACK >= AC) * (DAMAGE + EXTRA)} \regular damage\
  else { result: 0 }
}
function: crit MULT:n DAMAGE:d {
  result: (DAMAGE * MULT)
}

\Damage from a saving roll spell
SPELL: The spell's DC
SAVE: Saving throw modifier of target
DAMAGE: Damage on a fail
Does half damage on success, double on critical fail
\
function: SPELL:n save SAVE:n dmg DAMAGE:d {
  result: [1d20 spell SPELL save SAVE DAMAGE]
}

function: ROLL:n spell SPELL:n save SAVE:n DAMAGE:d {
  RES: [result ROLL SAVE SPELL]
  if (RES = 3) { result: DAMAGE * 2 }
  else if (RES = 2) { result: DAMAGE }
  else if (RES = 1) { result: DAMAGE / 2 }
  else { result: 0}
}

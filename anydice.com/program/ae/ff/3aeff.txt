

FAILURE: -1
TIE: 0
SUCCESS: 1
SUCCESSWITHSTYLE: 2

function: fate ROLL:n against TARGET {
  if ROLL < TARGET { result: FAILURE }
  if ROLL = TARGET { result: TIE }
  if ROLL < TARGET+3 { result: SUCCESS }
  result: SUCCESSWITHSTYLE 
}

loop TN over {0..6} {
 output [ fate 4d{-1,0,1} against TN ] named "4dF > [TN]"
}

DIE: d100
SCALE: 20
ATTRIBUTE: 5 * d{0..10}
SKILL: 10 * d{0..5}

function: check ROLL:n against A:n with S:n {
 TN: A + S
 if TN < ROLL {
  result: -1
 } else {
  result: (ROLL - 1@{DIE}) / SCALE
 }
}

function: roll A:n with S:n {
 result: [check DIE against A with S]
}

output [roll ATTRIBUTE with SKILL] named "Degrees of Success"

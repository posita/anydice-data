set "position order" to "lowest first"

function: SEQ:s against ATT:n and SKILL:n {
  TARGET: [sort {ATT, SKILL}]
  DIE: 1@SEQ
  if DIE <= 1@TARGET {
    result: 2
  }
  if DIE <= 2@TARGET {
    result: 1
  }
  result: 0
}

loop ATT over {5..14} {
  loop N over {1..3} {
    output [Nd20 against ATT and SKILL] named "[N]d against [ATT] and [SKILL]"
  }
}

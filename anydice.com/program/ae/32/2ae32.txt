function: atkroll SKILL:n damage DAM:n armor RES:n {
  if (SKILL-8) < 0 { result: 0}
  X: DAM+(SKILL-8)-RES
  if X < 0 {result: 0}
  result: X
}


function: auto AUT:n atkroll SKILL:d damage DAM:d armor RES:n {
  ACCUM: 0
  loop I over {1..AUT} {
    ACCUM: ACCUM + [atkroll SKILL damage DAM armor RES]
  }
  result: ACCUM
}


output [auto 4 atkroll (2d6+4) damage (5d6+3) armor 20] named "5D vs 20 armor"
output [auto 3 atkroll (2d6+4) damage (8d6+3) armor 30] named "8D A3 vs 30 armor"

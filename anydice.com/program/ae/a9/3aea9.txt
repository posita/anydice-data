WHITE: {-1,0,0,0,1,1}
BLACK: {-1,0,0,0,2,2}
POOLSIZE: 6
START: 1


function: normalroll W:n B:n WHITE:s BLACK:s {
  if W > 0 {
    if WHITE <= -1 {
       if BLACK <= -1 {result: -1}
       else {result: 0}
    }
    else if WHITE > -1 {
       if BLACK <= -1 {result: WHITE}
       if BLACK > -1 {result: WHITE + BLACK}
    }
  }
  else if BLACK <= -1 {result: -1}
  else if BLACK > -1 {result: BLACK}
}


loop B over {START..POOLSIZE-1} {
  W: POOLSIZE - B
  output [normalroll W B WdWHITE BdBLACK] named "[B] Black + [W] White / [POOLSIZE]P"
}

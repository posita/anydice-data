HIT: 11
ATTACK: 1d20 + HIT
DAMAGE: 1d10 + 1d6 + 11
AC: 20
CRIT: 19

function: blast damage ROLL {
  result: (DAMAGE * (ROLL > AC)) + ((DAMAGE - 11) * ((ROLL - 11) >= CRIT))
}

function: total damage {
  result: [blast damage ATTACK] + [blast damage ATTACK] + [blast damage ATTACK]
}

output [total damage] named "Eldritch blast against AC"

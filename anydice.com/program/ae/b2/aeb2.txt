\ L is your level \
\ K is the kindling level \
function: estus L K
{
    result: ((L+3)/4)d6 + (K*2)
}

output 2d[estus  2 1] \ level 2,  +1, no shards \
output 4d[estus  6 2] \ level 6,  +2,  2 shards \
output 6d[estus 10 3] \ level 10, +3,  4 shards \
output 8d[estus 15 4] \ level 15, +4,  6 shards (max) \
output 8d[estus 20 5] \ level 20, +5,  6 shards (max) \

MISS: 0
HIT: 1
CRITICALHIT: 2

function: strbruteattack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + STRBRUTEATTACK >= 0 { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + STRBRUTEATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: dexattack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + DEXATTACK >= 0 { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + DEXATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: champattack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL >= 18 {
  if ROLL + STRCHAMPATTACK >= 0 { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + STRCHAMPATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: dexattack {
 result: [damage for [dexattack [highest 1 of 3d20] vs DEFENSE]]
}

function: strchampattack {
 result: [damage for [champattack d20 vs DEFENSE]]
}

function: strbruteattack {
 result: [damage for [strbruteattack d20 vs DEFENSE]]
}

output [strbruteattack] named "Str Brute"

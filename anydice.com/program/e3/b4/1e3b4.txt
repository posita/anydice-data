function: get loss for X:s vs Y:s {
 if (#X = 1) | (#Y = 1) {
  if (1@X > 1@Y) {
   result: 0
  }
  else {
   result: 1
  }
 }
 else if (1@X > 1@Y) & (2@X > 2@Y) {
  result: 0
 }
 else if (1@X > 1@Y) | (2@X > 2@Y) {
  result: 1
 }
 else if (1@X <= 1@Y) & (2@X <= 2@Y) {
  result: 2
 }
}

\output [get loss for 3d6 vs 2d6]\


function: battle N_ATT:n vs N_DEF:n {
 \Note: N_ATT includes the 1 army who must be left behind\
 NEW_N_ATT: N_ATT
 NEW_N_DEF: N_DEF

 if (NEW_N_ATT <= 1) | (NEW_N_DEF <= 0) {
  result: NEW_N_ATT

  N_ATT_DICE: 3

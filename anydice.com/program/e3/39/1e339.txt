function: ROLL:n reroll BAD:s as REROLL:d {
  if ROLL = BAD { result: REROLL }
  result: ROLL
}

function: HIT:n tohit CHANCE {
  if CHANCE > 100 {CHANCE: 100}
  if HIT >= (100 - CHANCE) {result: 1}
  result: 0
} 

X: [d6 reroll {1,2} as d6]
CHANCE: 65

output ([d100 tohit CHANCE]*(2d6+5))     + ([d100 tohit CHANCE]*(2d6+5))     + ([d100 tohit CHANCE]*(2d6+5)) named "Claws"
output ([d100 tohit CHANCE]*(2dX+1d6+5)) + ([d100 tohit CHANCE]*(2dX+1d6+5))                                 named "GS"
output 6d6+15                                                                                                named "Claws all hit"
output 4dX+2d6+10                                                                                            named "GS    all hit"

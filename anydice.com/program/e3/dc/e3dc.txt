\ Attacker Diepool \   A: 18
\ Edge? \  EDGE:5
\Weapon Limit\        AL: 8

function: explode N:n {
 if N = 6 { result: 1 + [explode d6] }
 result: N = 5
}

function: limit HITS:n by LIMIT:n{
 if HITS > LIMIT {result: LIMIT}
 result: HITS
}

function: hitslost HITS:n by LIMIT:n{
 if HITS > LIMIT {result: (HITS-LIMIT)}
 result: 0
}

\Attacker's Roll Result\ 

ARX:0
ARX: Ad[explode d6] + EDGEd[explode d6]
ARLX:[limit ARX by 25]

ARY:0
ARY: Ad(d6 > 4)
\Attacker's roll with Limit\
AOL: A - ARY
ARXX: AOLd(d6 > 4)
ARLY: [limit ARY + ARXX by AL]


output ARLX named "Push The Limit"
output ARLY named "Second Chance"

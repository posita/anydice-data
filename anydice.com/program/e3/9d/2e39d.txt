function: models MODELS:n wounds WOUNDS:n dmg DMG:n {
    HP: MODELS*WOUNDS
    COUNT: 0
    loop H over {1..HP} {
        if H + HP - DMG > H {
            HP: HP - DMG
            COUNT: COUNT + 1
        }
    }
    result: COUNT
}

function: models MODELS:n wounds WOUNDS:n damage DMG:n {
    HP: MODELS*WOUNDS
    loop H over {1..HP} {
        result: HdDMG >= HP
    }
}

HP: 2
DMG: 3
output HPdDMG
output 2*3

output [models 2 wounds 6 damage d6]

function: is OLD:n better than NEW:n {
  result: OLD > NEW
}

function: hit points at LVL:n level {
  NEW_HP: d6
  if LVL > 1 {
    OLD_HP: [hit points at LVL-1 level]    
    if [is OLD_HP better than NEW_HP] { result: 1 }
  }
  result: NEW_HP
}

function: comboj of A:n and B:n and C:n {
if A=B=C { result: 2}
if A=B { result: 1 }
  if B=C { result: 1 }
  if A=C { result: 1 }
if A+B=7 { result: 1 }
  if B+C=7 { result: 1 }
  if A+C=7  { result: 1 }
  result: 0
}

  output [comboj of 1d6 and 1d6 and 1d6]

function: compare A:n to B:n and C:n with M:n {
 WINS: 0
 if A+M > B { WINS: WINS + 1 }
 if A+M > C { WINS: WINS + 1 }
 result: WINS
}

function: elegy A:n and B:n with M:n {
 WINS: 0
 if M > A { WINS: WINS + 1 }
 if M > B { WINS: WINS + 1 }
 result: WINS
}

loop N over {1..3}{
 output [compare 1d6 to 1d10 and 1d10 with N] named "Ironsworn [N]"
}

loop N over {4..6}{
 output [elegy 1d10 and 1d10 with N] named "Elegy [N]"
}

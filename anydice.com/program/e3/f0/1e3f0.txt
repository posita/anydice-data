function: regular ROLL:n vs AC:n {
	DAMAGE: WEAPON + STRENGTH + STYLE
	CRITICAL: WEAPON + WEAPON + STRENGTH + STYLE

	if ROLL = 20 { result: CRITICAL }
	if ROLL + ATTACK >= AC { result: DAMAGE }
	result: 0
}

function: regular attacks vs AC:n {
	result: ([regular d20 vs AC] + [regular d20 vs AC])
}

function: nocrit ROLL:n vs AC:n {
	DAMAGE: WEAPON + STRENGTH + STYLE

	if ROLL + ATTACK >= AC { result: DAMAGE }
	result: 0
}

function: nocrit attacks vs AC:n {
	result: ([nocrit d20 vs AC] + [nocrit d20 vs AC]+ [nocrit d20 vs AC])

}

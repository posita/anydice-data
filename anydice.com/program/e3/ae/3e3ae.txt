function: target TARGET:n reroll REROLL:n VALUE:n {
  if VALUE = REROLL { result: 2 + [target TARGET reroll REROLL BASEDIE]  }
  if VALUE > TARGET { result: 1 }
  if VALUE = 1 { result: -1 }
  result: 0
}

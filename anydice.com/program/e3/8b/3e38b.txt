function: compare A:n to B:n and C:n with M:n {
 WINS: 0
 if A+M > B { WINS: WINS + 1 }
 if A+M > C { WINS: WINS + 1 }
 result: WINS
}

function: compare BASE:n and A:n to B:n {
 WINS: 0
 if (BASE + A) > B+5 { WINS: WINS + 2 }
 else if (BASE + A) > B { WINS: WINS + 1 }
 result: WINS
}

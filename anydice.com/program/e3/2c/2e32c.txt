function: attack ROLL:n with ATTACK:n vs FULL:n and HALF:n for HIT:n or CRIT:n on THREAT:n {
 if ROLL = 1 { result: 0 }
 if ROLL >= THREAT & ROLL + ATTACK >= FULL { result: [highest of CRIT and 1] }
 if ROLL >= THREAT { result: [highest of CRIT/2 and 1] }
 if ROLL + ATTACK >= FULL { result: [highest of HIT and 1] }
 if ROLL + ATTACK >= HALF { result: [highest of HIT/2 and 1]}
 result: 0
}

\NEW SYSTEM\
C:3        \Number of dice\
DIE:{1..6} \Used die as SET!\

I: 1       \Add. dice on +1\
II: I+1    \Add. dice on +2\
III: II+1  \Add. dice on +2\
NI: I      \Add. dice on -1\
NII: II    \Add. dice on -2\
NIII: III  \Add. dice on -3\
MX: 0      \Flat bonus to result\

AVG:(((1@DIE+#DIE@DIE)*C)/2)
MI: 4 \None, Zeroed, Fitted for 1d20, Fitted for Curves\
M: MI@{0,-AVG,-AVG+10,0} 
MDD: MI@{0,-10,0,AVG-10}

KRIT:  d{3,6,9,12,15,18}
LIN:   d{1@DIE*C..#DIE@DIE*C}
\STD:   ((MDD+1d20))\
WORST: ((M+{1+NIII..C+NIII}@(C+NIII)dDIE))
WORSE: ((M+{1+NII..C+NII}@(C+NII)dDIE))
BAD:   ((M+{1+NI..C+NI}@(C+NI)dDIE))
NORM:  ((M+CdDIE))
GOOD:  ((M+{1..C}@(C+I)dDIE))
BETTER:((M+{1..C}@(C+II)dDIE))
BEST:  ((M+{1..C}@(C+III)dDIE))

output MX+KRIT named "Kritische Punkte"
output MX+LIN named "linear"
\output MX+STD named "standard"\
output MX+WORST named "worst"
output MX+WORSE named "worse" 
output MX+BAD named "bad"
output MX+NORM named "normal"
output MX+GOOD named "good"
output MX+BETTER named "better"
output MX+BEST named "best"

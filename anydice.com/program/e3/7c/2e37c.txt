function: explode N:n {
 if N = 1 {
  result: N - d10
 }
 if N = 10 {
  result: N + d10
 }
 result: N
}

function: damage N:n {
 result: N
}

function: crit N:n {
 if N = 6 {
 result: 1
 }
 result: 0
}

V: 5d([damage d6])
W: 5d([crit d6])

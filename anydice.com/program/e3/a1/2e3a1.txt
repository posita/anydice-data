function: cascade N:n size SIZE:n {
 if N >= SIZE {
  if SIZE > 4 {
   result: N + [cascade d(SIZE - 2) size SIZE - 2] + [cascade d(SIZE - 2) size SIZE - 2]
  }
  else {
   result: N + [cascade dSIZE size SIZE - 2] + [cascade dSIZE size SIZE - 2]
  }
 }

 result: N
}

set "maximum function depth" to 5

output [cascade d8 size 8]

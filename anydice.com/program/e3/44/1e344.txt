MISS: 0
HIT: 1

function: attack TO_HIT:n below STAT:n against defense DEFENSE:n {
 if TO_HIT > STAT + DEFENSE { result: MISS }
 result: HIT
}

function: damage for ATTACK_RESULT:n {
  if ATTACK_RESULT = HIT {
    result: DAMAGE
  }
  result: 0
}

function: DAMAGE_RESULT:n against armor {
  result: [highest of DAMAGE_RESULT - ARMOR and 0]
}

ATTACK_STAT: {13..15}
ABILITY_BONUS: 1d4
DAMAGE: d{d6,d8,d10,d12}+0

\ armor impeding is factored into ENEMY DEFENSE \
ARMOR_IMPEDING: {0,-2,-3,-4}
ENEMY_DEFENSE: {-5..7}
ARMOR: d{d4,d6,d8,d10}

output [[damage for [attack d20 below ATTACK_STAT against defense ENEMY_DEFENSE]] against armor]

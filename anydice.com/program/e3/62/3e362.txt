function: interpret ROLL:s {
 COUNT: [count 6 in ROLL]
 if COUNT > 0 { result: 6 + (COUNT-1) }
 if [count 5 in ROLL] { result: 5 }
 if [count 4 in ROLL] { result: 4 }
 if [count 3 in ROLL] { result: 3 }
 if [count 2 in ROLL] { result: 2 }
 result: 1
}

loop DICE over {1..10} {
 output [interpret DICE d6] named "[DICE]d6"
}

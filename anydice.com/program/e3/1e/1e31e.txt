set "maximum function depth" to 3

CRIT:0
Y:3
B:Y
D:12
X:Yd(D)
Z:{1..Y}@X
NIGWARD:{1..Y}@X
TRIAL:{1..(Y-B)}@X
function: call N:n {
 if N = {8,9} { result: [call d12] } else
 if N = 12 { result: 2 + [call d12] } else
 if N = {10, 11} { result: 1 } else
 {result: 0}
 if CRIT = 1{result: N*3} else
 {result: N}
 result:N
}

function: check T:s for crits{
C:0
M:0
loop N over T{
if N = M {
C: C + 1
if C = 3{CRIT:1}
} else{
C:1
M:N}
}
result: T
}

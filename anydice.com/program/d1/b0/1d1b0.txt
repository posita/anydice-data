function: fame X:n Y:n {
if X=0 {result:Y}
if Y=0 {result:X}
else {result:X+Y}
}

function: ffyou X:n Y:n {
M : ( ( 2d6 + X - 1 ) / ( 2d6 + Y ) )
result: [lowest of M and 2]
}
function: ffthem X:n Y:n {
T : -1 * ( ( 2d6 + Y - 1 ) / ( 2d6 + X ) )
result: [highest of T and -2]
}

output [ffyou 0 0]
output [ffthem 0 0]
output [fame [ffyou 0 0] [ffthem 0 0]]

function: combatround ROLLONE:n ROLLTWO:n {
 if ROLLONE = 1 {
  result: 0
 }
 if ROLLTWO = 1 {
  result: [ROLLONE vs AC at ATTACK]
 }
 result: [ROLLONE vs AC at ATTACK]*2 + [ROLLTWO vs AC at ATTACK-5]

}
function: ROLL:n vs AC:n at BONUS:n{
 if ROLL >= THREAT & ROLL + BONUS >= AC {
  result: [threat 1d20 AC BONUS]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: (DAMDICE + DAMBON)*DAMMULT/DAMDIV }
 result: 0
}
function: threat ROLL:n AC:n BONUS:n {
 if ROLL+BONUS >= AC {
  result: ((1d DAMDICE)*2 + DAMBON)*DAMMULT/DAMDIV
 }
 result: (DAMDICE + DAMBON)*DAMMULT/DAMDIV
}


AC: 18
DEX: 4 + 2
CON: 9 + 1
BAB: 6
ATTACK: 1 + DEX + BAB + 3

DAMDICE: (5d6)
DAMBON: (CON/5+8)
DAMMULT: 1
DAMDIV: 1
THREAT: 20
CRITICAL: 2
output [combatround 1d20 1] named "Kinetic Blade"

\

function: factorial N {
RES: 1
loop X over {1..N} {
RES: RES * X
}
result: RES
}
function: N choose K:n {
result: [factorial N]/([factorial K]*[factorial N-K])
}

\ Pool of 3d6, want any sum=5 \
function: multi or N of COND {
  RES: 0
  loop X over {1..N} {
    RES: RES | COND
  }
  result: RES
}

function: megamulti COUNT eq VALUE:n {
RES: 0
loop N over {1..COUNT} {
RES: [multi or [COUNT choose N] of Nd6 = VALUE]
}
result: RES
}

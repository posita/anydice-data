DIE: d6
LOWESTAMOUNT: 1
HIGHESTAMOUNT: 16
LOWESTTN: 2
HIGHESTTN: 17

function: check SINGLEDIE:n TN:n {
  result: SINGLEDIE >= TN
}

function: roll POOL:n TN:n {
  COUNT: 0
  loop N over {1..POOL} {
    COUNT: COUNT + [check [explode DIE] TN]
  }
  result: COUNT
}

loop TN over {LOWESTTN..HIGHESTTN} {
  loop POOL over {LOWESTAMOUNT..HIGHESTAMOUNT} {
    output [roll POOL TN] named "[POOL]d6 vs TN [TN]"
  }
}

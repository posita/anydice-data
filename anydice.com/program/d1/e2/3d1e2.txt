function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 {X: ROLL + MODIFIER - 10}
 else if ROLL >= CRITRANGE {X: ROLL + MODIFIER + 10}
 else {X: ROLL + MODIFIER}

 CRIT: 1+(X-DEFENSE)/10
 HIT: 1
 MISS: 0

 if X >= DEFENSE  + 10 {result: CRIT}
 else if X >= DEFENSE {result: HIT}
 else {result: MISS}
}

loop AC over {5..40} {
 MODIFIER: 25
 CRITRANGE: 19
 output [attack [highest 1 of 2d20] vs AC] named "PRECISION [AC]"
}

function: attack ROLL:n CRIT_RANGE:n CONFIRM:n TO_HIT:n {
 if ROLL >= CRIT_RANGE {
  if CONFIRM >= CRIT_RANGE {
   result: 2
  }
  result: 1
 }
 if ROLL >= TO_HIT {
  result: 1
 }
 result: 0
}

loop TO_HIT over {3..6} {
 output [attack [highest 1 of 2d6] 6 1d6 TO_HIT] named "Sword and Board ([TO_HIT])"
 output [attack [highest 1 of 2d6] 5 1d6 TO_HIT] named "Brutal Strikes ([TO_HIT])"
 output [attack [highest 1 of 3d6] 6 1d6 TO_HIT] named "Dual Weapons ([TO_HIT])"
}

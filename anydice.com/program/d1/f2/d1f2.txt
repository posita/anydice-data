function: change N:n {
if N = 0 {result:-2}
if N = 1 {result:1}
if N = 2 {result:3}
result:6
}

DC: 5+2d10
DEC: 13
INS: 5
INT: 7

CHA: [highest 1 of 1d20]
WIS: [highest 1 of 1d20]
CHAA: [highest 1 of 2d20]
WISA: [highest 1 of 2d20]

output [change (CHA+DEC >= DC)+(WIS+INS >= DC)+(CHA+INT >= DC)] named "Basic"
output [change (CHA+DEC+1d4 >= DC)+(WIS+INS+1d4 >= DC)+(CHA+INT+1d4 >= DC)] named "Guidance"
output [change (CHAA+DEC+1d4 >= DC)+(WIS+INS+1d4 >= DC)+(CHAA+INT+1d4 >= DC)] named "Guidance + Advantage CHA"
output [change (CHA+DEC+1d4 >= DC)+(WISA+INS+1d4 >= DC)+(CHA+INT+1d4 >= DC)] named "Guidance + Advantage WIS"
output [change (CHAA+DEC >= DC)+(WIS+INS >= DC)+(CHAA+INT >= DC)] named "Advantage CHA"
output [change (CHA+DEC >= DC)+(WISA+INS >= DC)+(CHA+INT >= DC)] named "Advantage WIS"
output [change (CHAA+DEC >= DC)+(WISA+INS >= DC)+(CHAA+INT >= DC)] named "Advantage Both"

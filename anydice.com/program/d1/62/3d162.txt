function: stat {
  result: [highest 3 of 4d6]
}

function: topstat {
X: 0
loop N over {1..6}{
 X: [highest of X and [stat]]
}
result: X
}

function: topset {
X: 0
loop N over {1..3}{
 X: [highest of X and [topstat]]
}
result: X
}

output [stat] named "Individual stat"
output [topstat] named "Highest stat in a set"
output [topset] named "Highest stat in 3 sets"

function: attack ROLL:n plus ATTACK:n vs DEFENSE:n dealing DAMAGE:n{
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: DAMAGE}
  result: DAMAGE
 }
 if ROLL + ATTACK >= DEFENSE { result: DAMAGE}
 result: MISS
}

function: attack ROLL:n and ADVANTAGE:n plus ATTACK:n vs DEFENSE:n dealing DAMAGE:n{
 if ROLL = 1 & ADVANTAGE = 1 { result: MISS }
 if ROLL = 20 | ADVANTAGE = 20 {
  if (ROLL + ATTACK >= DEFENSE) | (ADVANTAGE + ATTACK >= DEFENSE) { result: DAMAGE}
  result: DAMAGE
 }
 if (ROLL + ATTACK >= DEFENSE) | (ADVANTAGE + ATTACK >= DEFENSE) { result: DAMAGE}
 result: MISS
}

function: whisperingaura STHROW:n saving throw dc DC {
 if STHROW < DC { result: 5 }
 result: 0
}

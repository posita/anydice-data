function: interpret ROLL:s {
 if [count 1 in ROLL] { result: -1 } \ failure \
 if [count 3 in ROLL] { result: 1 }  \ success \
 if [count 6 in ROLL] { result: 1 }  \ success + explode\
 result: 0 \ guided event \
}

set "explode depth" to 9

loop DICE over {1..10} {
 output [explode [interpret DICE d6]] named "[DICE]d"
}

function: R:s has a match {
   result: 1@R = 2@R | 1@R = 3@R | 2@R = 3@R
}
function: R:s under N:n successes {
  if 1@R = 1 | 1@R <= N { result: 3 }
  if 2@R = 1 | 2@R <= N { result: 2 }
  if 3@R = 1 | 3@R <= N { result: 1 }
  if 3@R > N { result: 0 }
}

loop N over {1..11} {
   output [3d12 under N successes] named "3d12 <= [N]"
}

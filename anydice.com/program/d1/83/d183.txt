loop N over {1..10} {
  \output [count {6} in Nd6] named "Erfolge mit [N]w6"\
  X: [count {6} in Nd6]
  Y: N-X
  output X > 0 named "X"
  \output Y > 0 named "Y"\
  output [count {6} in Yd6] + X > 0 named "count [Y]W6"
}

output {1,1,1,2,3} named "output {1,1,1,2,3}"
output {d4,d6} named "output {d4,d6}"

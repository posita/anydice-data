PROF: d{3..6}
ABILITY: d{3..6}
AC: d{12..24}

function: attack X:n {
  result: ((1d20 + PROF + ABILITY) >= AC) * X
}

output [attack (1d8 + ABILITY)] + [attack 1d8] named "Dual Wielder"

output [attack (1d8 + ABILITY)] + [attack 1d8] + [attack 1d8] named "Lightning Striker"

output [attack (1d8 + ABILITY)] + [attack 1d8] + [attack ( [attack (2*PROF)])] named "Precise Striker"

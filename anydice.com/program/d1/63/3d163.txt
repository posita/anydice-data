function: multi or N of COND {
  RES: 0
  loop X over {1..N} {
    RES: RES | COND
  }
  result: RES
}

function: megamulti COUNT eq VALUE {
\ needs factorial \
RES: 0
loop N over {1..COUNT} {
RES: [multi or (COUNT-N) of Nd6 = VALUE]
}
}

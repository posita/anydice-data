function: normal ROLL:s {
 SUCCESSES: ROLL >= 1
 result: SUCCESSES
}
function: risky ROLL:s {
 RISK: ROLL = 1
 SUCCESSES: ROLL >= 2 
 TOTAL : (RISK + SUCCESSES)
 if RISK > SUCCESSES { result: -TOTAL}
 result: TOTAL
}
function: competent ROLL:s {
 CRIT: ROLL = 10
 SUCCESSES: ROLL >=1
 result: (CRIT + SUCCESSES)
}

function: full ROLL:s {
 RISK: ROLL = 1
 CRIT: ROLL = 10
 SUCESSES: ROLL = 2
 TOTAL : (RISK + SUCESSES+ 2*CRIT)
 if RISK > [highest of SUCESSES and CRIT] { result: -TOTAL}
 result: TOTAL
}

loop N over {1..4}{
output [normal N d{0,0,1,2,2,10}] named "[N] compass dice"
output [risky N d{0,0,1,2,2,10}] named "[N] risky dice"
output [competent N d{0,0,1,2,2,10}] named "[N] Competent dice"
output [full N d{0,0,1,2,2,10}] named "[N] Full dice"
}

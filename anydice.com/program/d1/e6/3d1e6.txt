function: sign of NUMBER:n {
  if NUMBER > 0 { result: +1 }
  if NUMBER < 0 { result: -1 }
  result: 0
}

BATTLEMASTER: 2d6 + 2
ENEMY: 2d6 + 0

WINNER: [sign of BATTLEMASTER- ENEMY] 
output WINNER named "single round: +1 = Battlemaster win, -1 = Enemy win, 0 = tie"

function: restrict ROLL:n to RANGE:s {
  if ROLL = RANGE { result: ROLL }
  else { result: d{} }
}

ROUND: [restrict WINNER to {+1,-1}] = +1
output ROUND named "single round with rerolls: 1 = Battlemaster win, 0 = Enemy win"

output 5dROUND named "number of Battlemaster wins over 5 rounds"

output 5dROUND >= 3 named "probability of Battlemaster winning at least 3 rounds out of 5"

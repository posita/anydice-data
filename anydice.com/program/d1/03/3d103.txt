Q: {0..3}

function: halve SEQUENCE:n {                  \ divide per 2, rounded up \
 DIV: SEQUENCE/2
 TEMP: DIV*2
 if (DIV=TEMP){result: DIV}
 result: (SEQUENCE+1)/2
}
function: sets in DICES:s {
  ZEROS:  [count 0 in DICES]
  ONES:   [count 1 in DICES]
  TWOS:   [count 2 in DICES]
  THREES: [count 3 in DICES]
  SETS: {ZEROS,ONES,TWOS,THREES}
  result: SETS
}
function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  if(VAL_X < VAL_Y) { result: -1}
  result: 0
}

\***********************************************************************************\

function: phaseone ATT:s DEF:s {
  SIZE_MODS: [compare 15@ATT with 15@DEF]                 \compare the nbr of bodies\
  ATT_ROLL: (1@ATT +SIZE_MODS) dQ + 6@ATT                      \freeze the hits-dices sequence\

  VELOCITY_MODS: [compare 14@DEF with 13@ATT]             \compare the velocity for the dodge capacity\
  DEF_ROLL: (7@DEF +VELOCITY_MODS) dQ + 12@DEF

  result: [phasetwo ATT_ROLL DEF_ROLL ATT DEF]
}

function: phasetwo ATT_ROLL:n DEF_ROLL:n  ATT:s DEF:s {
  HITS: [attack curve ATT from ATT_ROLL]
  SAVES: [save curve DEF from DEF_ROLL]
  HIT_SETS: [sets in ATT_ROLL]
  SAVE_SETS: [sets in DEF_ROLL]

  DAMAGES: HITS - SAVES
  \DAMAGES: [halve HITS] - [halve SAVES]\
  result: DAMAGES
}

function: attack curve THRESHOLDS:s from POOL:n{
  if(POOL >= 5@THRESHOLDS) { result: 4}
  if(POOL >= 4@THRESHOLDS) { result: 3}
  if(POOL >= 3@THRESHOLDS) { result: 2}
  if(POOL >= 2@THRESHOLDS) { result: 1}
  result: 0
}
function: save curve THRESHOLDS:s from POOL:n{
  if(POOL >= 11@THRESHOLDS) { result: 4}
  if(POOL >= 10@THRESHOLDS) { result: 3}
  if(POOL >= 9@THRESHOLDS) { result: 2}
  if(POOL >= 8@THRESHOLDS) { result: 1}
  result: 0
}

function: variable thresholds SUM_ROLL:n THRESHOLDS:s MODS:s{
 if (SUM_ROLL >= 4@THRESHOLDS+4@MODS){result: 4}
 if (SUM_ROLL >= 3@THRESHOLDS+3@MODS){result: 3}
 if (SUM_ROLL >= 2@THRESHOLDS+2@MODS){result: 2}
 if (SUM_ROLL >= 1@THRESHOLDS+1@MODS){result: 1}
 if (SUM_ROLL < 1@THRESHOLDS+1@MODS){result: 0}
 result: 0
}

function: attack DATAS:s with TYPE:n {           \TYPE = 1 for thresholds & 0 for basic\
  ATT_ROLL: (1@DATAS) dQ + 6@DATAS                 \freeze the hits-dices sequence\
  if (TYPE=1){result: [attack curve DATAS from ATT_ROLL]}
  result: ATT_ROLL
}
function: defense DATAS:s with TYPE:n {
  DEF_ROLL: (7@DATAS) dQ + 12@DATAS
  if (TYPE=1){result: [save curve DATAS from DEF_ROLL]}
  result: DEF_ROLL
}

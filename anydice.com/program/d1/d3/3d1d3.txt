function: attack ROLL:n plus BONUS:n vs AC:n hit HIT:d miss MISS:d crit CRIT:d on CRIT_RANGE:n {
if ROLL = 1 {
result: MISS
}
if ROLL >= CRIT_RANGE {
result: CRIT
}
if ROLL + BONUS >= AC {
result: HIT
}
result: MISS
}

function: attack ROLL:n plus BONUS:n vs AC:n for DMG:d crit CRIT_DMG:d on CRIT_RANGE:n {
result: [attack ROLL plus BONUS vs AC hit DMG miss 0 crit DMG + CRIT_DMG on CRIT_RANGE]
}

loop AC over {11..20} {
output[ 
attack [highest 1 of 2d20] plus 9 vs AC
hit 1d12 + 5 + [attack [highest 1 of 2d20] plus 9 vs AC for 1d12 + 5       crit 4d12 on 20]
miss [attack [highest 1 of 2d20] plus 9 vs AC for 1d12 + 5 crit 4d12 on 20]
crit 5d12 + 5 + [attack [highest 1 of 2d20] plus 9 vs AC for 1d12 + 5       crit 4d12 on 20] on 20
] 

named "Axe Damage vs AC [AC]"
}

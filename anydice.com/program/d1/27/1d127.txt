\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}

function: gloomattackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: GLOOMCRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: GLOOMDAMAGE }
 result: 0
}

function: gloomattackrollautocrit ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: GLOOMCRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: GLOOMCRITDAMAGE }
 result: 0
}

function: attackrollautocrit ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: CRITDAMAGE }
 result: 0
}

\looped against enemy AC from 15-20\
ENEMYAC: d{15..20}

ATTACKMOD: 5
DMGBONUS: 2

ACCBONUS: 2
CRITROLL: 20
DAMAGE: 2d12 + ATTACKMOD + DMGBONUS
CRITDAMAGE: 4d12 + ATTACKMOD + DMGBONUS
GLOOMDAMAGE: 2d12 + ATTACKMOD + DMGBONUS + 1d8
GLOOMCRITDAMAGE: 4d12 + ATTACKMOD + DMGBONUS + 2d8

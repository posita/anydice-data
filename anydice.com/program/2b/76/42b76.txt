function: hit P S H {
 R: 0

 if P > 0 {
  loop N over {1..P}{
   R: R+(1d4>3)
  }
 }
 if S > 0 {
  loop N over {1..S}{
   R: R+(1d6>3)
  }
 }
 if H > 0 {
  loop N over {1..H}{
   R: R+(1d8>3)
  }
 }

 result: R
}

function: crit P S H {
R: 0

 if P > 0 {
  loop N over {1..P}{
   R: R+(1d4=4)
  }
 }
 if S > 0 {
  loop N over {1..S}{
   R: R+(1d6=6)
  }
 }
 if H > 0 {
  loop N over {1..H}{
   R: R+(1d8=8)
  }
 }

 result: R
}

\output [hit 3 0 0]
output [hit 0 3 0]
output [hit 0 0 3]\


\output [crit 3 0 0]
output [crit 0 3 0]
output [crit 0 0 3]\

output [hit 1 2 0]
output [crit 1 2 0]

output [hit 0 3 0]
output [crit 0 3 0]

output [hit 0 0 3]
output [crit 0 0 3]

\ loop N over {1..9}{
 output [hit N]
} \

 \output [crit 3]\

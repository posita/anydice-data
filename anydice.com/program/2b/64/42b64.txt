\ two digit values for all stats \
function: mix base MAX:n {
 result: (MAX*2)+1
}
function: mix base {
 MAX: 9
 result: (MAX*2)+1
}
function: parse VALUE:n x BASE:n {
 if VALUE < 0 {
  result: -(-VALUE+(BASE/2))/BASE
 }
 result: (VALUE+(BASE/2))/BASE
}

function: get ability size by DISADVANTAGE:n {
 if DISADVANTAGE < 1 {
  result: 12
 }
 if DISADVANTAGE = 1 {
  result: 8
 }
 if DISADVANTAGE > 1 {
  result: 6
 }
}
function: get challenge by DISTANCE:n {
 if DISTANCE <= 5 {
  result: 3
 }
 if DISTANCE <= 45 {
  result: 1
 }
 if DISTANCE <= 120 {
  result: 2
 }
 result: 99
}
function: ability distribution AWED:s {
 ATTACKER:     1@AWED  \ {ABILITY_MOD, ADVANTAGE, DISADVANTAGE} \
 WEAPON:       2@AWED  \ {SIZE, BRUTAL, PRECISE, VERSATILE, CONTROLLED, GRAZE, DUAL_WIELD, RANGED} \
 ENVIRONMENT:  3@AWED  \ {DISTANCE, DARKNESS} \
 DEFENDER:     4@AWED  \ {EVASION, ABSORPTION} \

 \ extract attacker stats \
 ATTACKER_MOD: [get ability mod of ATTACKER]
 ADVANTAGE:    [get advantage of ATTACKER]+1
 DISADVANTAGE: [get disadvantage of ATTACKER]

 \ extract weapon effects \
 DUAL_WIELD:   [get dual wield of WEAPON]    \ adjusts ABILITY_SIZE \
 VERSATILE:    [get versatile of WEAPON]     \ adjusts ABILITY_DICE \
 RANGED:       [get ranged of WEAPON]        \ adjusts ABILITY_MOD  \

 \ extract environment effects \
 DARKNESS:     [get darkness of ENVIRONMENT] \ adjusts ABILITY_SIZE \
 DISTANCE:     [get distance of ENVIRONMENT] \ adjusts ABILITY_MOD  \

 \ extract defender effects \
 EVASION:      [get evasion of DEFENDER]     \ adjusts ABILITY_MOD  \

 \ apply effects \
 DISADVANTAGE: [lowest of 1 and DISADVANTAGE+DARKNESS]+DUAL_WIELD
 ABILITY_SIZE: [get ability size by DISADVANTAGE]
 ABILITY_DICE: 1d{1d(ABILITY_SIZE), 1d(ABILITY_SIZE)+VERSATILE}
 ABILITY_MOD:  ATTACKER_MOD - (RANGED*[get challenge by DISTANCE]) - EVASION

 \ create dice \
 ABILITY_DICE: [highest 1 of ADVANTAGE d ABILITY_DICE] + ABILITY_MOD

 result:       [highest of 0 and ABILITY_DICE]
}
ABILITY_DICE: [ability distribution {ATTACKER_TEST, WEAPON_TEST, ENVIRONMENT_TEST, DEFENDER_TEST}]
\ output ABILITY_DICE named "ABILITY_DICE" \

 BRUTAL:       [get brutal of WEAPON]

 DUAL_WIELD:   [get dual wield of WEAPON]

 \ apply effects \
 MIGHT_SIZE:   SIZE
 MIGHT_DICE:   1d(MIGHT_SIZE)

function: compare MIGHT:n with ABILITY:n using CONTROLLED:n {
 DAMAGE:  MIGHT
 CONTROL: ABILITY-MIGHT
 loop N over {1..CONTROLLED}{
  if CONTROL < 0 {
   CONTROL: ABILITY-(MIGHT-N)
   DAMAGE:  MIGHT-N
  }
 }
 if CONTROL < 0 {
  result: 0
 }
 result: DAMAGE
}
function: might roll AWED:s {
 ATTACKER:     1@AWED  \ {ABILITY_MOD, ADVANTAGE, DISADVANTAGE} \
 WEAPON:       2@AWED  \ {SIZE, BRUTAL, PRECISE, VERSATILE, CONTROLLED, GRAZE, DUAL_WIELD, RANGED} \
 ENVIRONMENT:  3@AWED  \ {DISTANCE, DARKNESS} \
 DEFENDER:     4@AWED  \ {EVASION, ABSORPTION} \

 \ get dice \
 ABILITY_DICE: [ability distribution AWED]
 MIGHT_DICE:   [might distribution AWED]

 \ extract weapon effects \
 COUNT:        [highest of 1 and [get count of WEAPON]] \ default: 1 \
 CONTROLLED:   [get controlled of WEAPON]
 DUAL_WIELD:   [get dual wield of WEAPON]+1             \ options: 1,2 \

 \ apply effects \
 MIGHT_COUNT:  COUNT*DUAL_WIELD
 MIGHT_ROLL: (MIGHT_COUNT)d[compare MIGHT_DICE with ABILITY_DICE using CONTROLLED]

 result:      MIGHT_ROLL
}

MIGHT_ROLL: [might roll {ATTACKER_TEST, WEAPON_TEST, ENVIRONMENT_TEST, DEFENDER_TEST}]
\ output MIGHT_ROLL named "MIGHT_ROLL" \

function: apply GRAZE:n graze to DMG:n dmg {
 if DMG <= 0 {
  result: GRAZE
 }
 result: DMG
}
function: dmg roll AWED:s {
 ATTACKER:     1@AWED  \ {ABILITY_MOD, ADVANTAGE, DISADVANTAGE} \
 WEAPON:       2@AWED  \ {SIZE, BRUTAL, PRECISE, VERSATILE, CONTROLLED, GRAZE, DUAL_WIELD, RANGED} \
 ENVIRONMENT:  3@AWED  \ {DISTANCE, DARKNESS} \
 DEFENDER:     4@AWED  \ {EVASION, ABSORPTION} \

 \ get might roll \
 MIGHT_ROLL: [might roll AWED]

 \ extract weapon effects \
 PRECISE:      [get precise of WEAPON]       \ adjusts ABSORPTION \
 GRAZE:        [get graze of WEAPON]         \ adjusts DMG_ROLL \
 DUAL_WIELD:   [get dual wield of WEAPON]+1  \ options: 1,2 \

 \ extract defender effects \
 ABSORPTION:   [get absorption of DEFENDER]  \ adjusts ABILITY_MOD  \

 \ apply effects \
 ABSORPTION: [highest of 0 and (ABSORPTION-PRECISE)]
 GRAZE_DMG: [highest of 0 and (GRAZE*DUAL_WIELD)-ABSORPTION]
 DMG_ROLL: [apply GRAZE_DMG graze to (MIGHT_ROLL-ABSORPTION) dmg]

 result: DMG_ROLL
}

 DEFENDER:     4@AWED  \ {EVASION, ABSORPTION} \

 \ get might roll \
 MIGHT_ROLL: [might roll AWED]

 result: [lowest of 1 and MIGHT_ROLL]
}

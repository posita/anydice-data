function: myexploder ADIE:d into BDIE:d
{
MAXA: [maximum of ADIE]
MAXB: [maximum of BDIE]
result: [abhelper ADIE BDIE]
}

function: bhelper A:n B:n
{
if B=MAXB {result: B+ [bhelper A B]  }
result :B
}

function: abhelper A:n B:n
{
if A=MAXA {result: A+ [bhelper A B]  }
result :A
}

 result: N
}

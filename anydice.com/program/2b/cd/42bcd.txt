function: consecutive DECK:s{
  result: [consecutive DECK DECK helper]
}

function: cons N:n S:s helper{
  if N = 5 {result: [consecutive [sub 5 from S]]}
}
function: sub N:n from S:s {
  D: 0
  R: {}
  loop P over S {
    if P@S = N {
      if D < 1 {
        D: 1
      }
      else { R:{R,P@S}}
    }
    else { R:{R,P@S}}
  }
}
output [consecutive {0:6,5:2,-10:2,10:6,15,20,100,-100}]

MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICALHIT }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}
function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

DEFENSE: d{12..25}
STRENGTHMOD: 7

ATTACK: 6 + STRENGTHMOD
HITDAMAGE: 1d8 + STRENGTHMOD
CRITICALDAMAGE: 3d8 + STRENGTHMOD
MISSDAMAGE: 0
output [attack] named "Punching Fire Giant"

ATTACK: 8 + STRENGTHMOD
HITDAMAGE: 3d6 + STRENGTHMOD
CRITICALDAMAGE: 8d6 + STRENGTHMOD
output [attack] named "Greatsword Fire Giant"

STRENGTHMOD: 5
ATTACK: 6 + STRENGTHMOD

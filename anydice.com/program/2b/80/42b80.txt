WILLOWISPS: 0
WERENUTRIA: 0
GRINDYLOWS: 1d4*200 + 1d4*50
FIVE: 0
SIX: 0
SEVEN: 0
EIGHT: 0

function: total ROLL:n {
  if (ROLL = 2) {result: WILLOWISPS}
  if (ROLL = 3) {result: WERENUTRIA}
  if (ROLL = 4) {result: GRINDYLOWS}
  if (ROLL = 5) {result: FIVE}
  if (ROLL = 6) {result: SIX}
  if (ROLL = 7) {result: SEVEN}
  if (ROLL = 8) {result: EIGHT}
}

function: total ROLL:n with activity ACTIVITY:n under LEVEL:n {
  if (ACTIVITY <= LEVEL) {result: [total ROLL]}
  else {result: 0}
}

output WILLOWISPS named "Will-o-Wisps"
output WERENUTRIA named "Werenutria"
output GRINDYLOWS named "Grindylows"

output [total 2d6] named "Encounter"
output [total 2d6 with activity 1d6 under 2] named "Activity"

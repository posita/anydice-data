\ two digit values for all stats \
function: mix base {
 MAX: 99
 result: (MAX*2)+1
}
function: parse VALUE:n x BASE:n {
 if VALUE < 0 {
  result: -(-VALUE+(BASE/2))/BASE
 }
 result: (VALUE+(BASE/2))/BASE
}

 result: ABILITY_MOD*[ability mod]
}

}
function: disadvantage DISADVANTAGE:n {
 result: DISADVANTAGE*[disadvantage]
}
function: get ability mod of ATTACKER:n {
 MOD: ATTACKER-[attacker]
 result: [parse MOD x [ability mod]]
}
function: get advantage of ATTACKER:n {
 MOD: [get ability mod of ATTACKER]
 ADV: ATTACKER-[attacker]-MOD*[ability mod]
 result: [parse ADV x [advantage]]
}
function: get disadvantage of ATTACKER:n {
 MOD: [get ability mod of ATTACKER]
 ADV: [get advantage of ATTACKER]
 DIS: ATTACKER-[attacker]-MOD*[ability mod]-ADV*[advantage]
 result: [parse DIS x [disadvantage]]
}

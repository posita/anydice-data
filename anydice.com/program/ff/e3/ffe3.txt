function: weirdsplosion SEQ:s EXP:n BONUS:n{
  if (SEQ = EXP) >= 1 { result: (SEQ >= 4) + BONUS }
  result: (SEQ >= 4)
}

function: normalexplosion SEQ:s BONUS:n{
  if (SEQ = 6) >= 1 { result: (SEQ >= 4) + BONUS }
  result: (SEQ >= 4)
}

loop N over {1..8} {
  output [weirdsplosion Nd6 1d8 1d{0,1}] named "1d8 & [N]d6"
}
loop N over {1..8} {
  output [weirdsplosion Nd6 1d6 1d{0,1}] named "1d6 & [N]d6"
}
loop N over {1..8} { 
  output [normalexplosion Nd6 1d{0,1}] named "Static [N]d6"
}

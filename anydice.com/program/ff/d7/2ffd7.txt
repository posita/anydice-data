function: cs ROLL {
 result: ROLL >= 18 & ROLL <= 20
}
function: cf ROLL {
 result: ROLL >= 2 & ROLL <= 4
}
function: csp ROLL for L{
 result: 100 *([cs ROLL]) / (ROLL >= L)
}
function: cfp ROLL for L{
 result: 100 * ([cf ROLL]) / (ROLL >= L)
}

loop L over {11..15}{
 loop B over {-3..6}{
  output 2d10 + B >= L named "Bonus [B] Limit [L] -> success"
  output [cs 2d10+B] named "CS"
  output ![cf 2d10+B] named "CF"
 }
}
loop B over {-3..6}{
 output [cs 2d10 + B] named "crit success if bonus [B]"
}
loop B over {-3..6}{
 output [cf 2d10 + B] named "crit fail if bonus [B]" 
}

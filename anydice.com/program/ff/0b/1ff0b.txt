MISS: 0
HIT: 1
CRITICALHIT: 2

function: attack ROLL:n plus ATTACKBONUS:n with crit at CRITTHRESH:n vs DEFENSE:n
{
 if ROLL = 1 { result: MISS }
 if ROLL >= CRITTHRESH { result: CRITICALHIT }
 if ROLL + ATTACKBONUS >= DEFENSE { result: HIT }
 result: MISS
}

function: damage ROLL:d plus DAMAGEBONUS:n for ATTACKRESULT:n 
{
 if ATTACKRESULT = CRITICALHIT { result: ROLL + ROLL + DAMAGEBONUS }
 if ATTACKRESULT = HIT { result: ROLL + DAMAGEBONUS }
 result: 0
}

function: damage PROFICIENCY:n STRENGTH:n MAGIC:n
{
 BONUS: PROFICIENCY + STRENGTH + MAGIC
 result: [damage d8 plus STRENGTH for [attack d20 plus BONUS with crit at 19 vs AC]]
}

function: use action surge A:n B:n
{
 if (A = 0) { result: B }
 result: A
}

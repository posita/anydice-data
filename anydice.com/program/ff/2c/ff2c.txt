function: ROLL:n vs AC:n{
 if ROLL >= THREAT | ROLL + ATTACK >= AC {
  result: [threat ROLL]
 }
 result: 0
}

function: threat ROLL:n {
 if SNEAKCOUNT = 0 {
  SNEAKCOUNT: SNEAKCOUNT + 1
  if ROLL >= THREAT {
   result: (CRITICAL d (DAMAGE + SNEAK)) + STRENGTH
  }
  result: DAMAGE + STRENGTH + SNEAK
 }
 if ROLL >= THREAT {
  result: (CRITICAL d DAMAGE) + STRENGTH
 }
 result: DAMAGE + STRENGTH
}

function: sneak {
 if LVL < 6 {
  result: 0
 }
 result: (LVL - 6)/2
}

function: atkcount  {
 if LVL < 5 {
  result: 1
 }
 if LVL < 7 {
  result: 2
 }
 result: 3
}

function: range {
 if LVL < 7 {
  result: 20
 }
 result: 19
}

function: prof {
 result: (LVL - 1)/4
}

function: luck {
 if LVL < 6 {
  result: 2
 }
 result: 3
}

function: clevel {
 if LVL < 3 {
  result: 0
 }
 if LVL > 7 {
  result: 5
 }
 result: LVL - 2
}

function: blevel {
 if LVL > 2 {
  result: 2
 }
 result: LVL
}

function: rlevel {
 if LVL < 8 {
  result: 0
 }
 result: LVL - 7
}

function: twf {
 if LVL < 3 {
  result: 0
 }
 result: STRENGTH
}

function: boost {
 if LVL < 17 {
  result: 0
 }
 if LVL < 19 {
  result: 1
 }
 result: 2
}

function: post LEVEL:n {
 SNEAKCOUNT: 0
 STRENGTH: 2 + [boost]
 ATTACK: [prof] + STRENGTH
 ATKCOUNT: [atkcount]
 DAMAGE: 1d6
 SNEAK: [sneak]d6
 THREAT: [range]
 CRITICAL: 2
 MAINHAND: ATKCOUNTd[[highest 1 of [luck]d20] vs AC] 
 DAMAGE: 1d6
 ATTACK: [prof] + STRENGTH
 STRENGTH: [twf]
 OFFHAND: [[highest 1 of [luck]d20] vs AC]
 result: MAINHAND+OFFHAND
}

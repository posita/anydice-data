function: dagger ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: DAGGERCRIT } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAGGER }
 result: 0
}

function: gsword ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: GSWORDCRIT } 
 if ROLL = 1 { result: 0 }
 if ROLL + ATTACKMOD + ACCBONUS >= AC { result: GSWORD }
 result: 0
}

AC:{12}

ATTACKMOD: 0
DMGBONUS: 0
PROFICIENCY: 2
ACCBONUS: 0
CRITROLL: 20
ATTACKTIMES: 1
DAGGER: 1d4 + ATTACKMOD + DMGBONUS
DAGGERCRIT: 2d4 + ATTACKMOD + DMGBONUS
GSWORD: 2d6 + ATTACKMOD + DMGBONUS
GSWORDCRIT: 4d6 + ATTACKMOD + DMGBONUS


output ATTACKTIMESd[dagger d20 vs AC] named "dagger"
output ATTACKTIMESd[gsword d20 vs AC] named "gsword"

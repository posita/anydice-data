function: hpistol ROLL:n vs DEFENSE:n {
 if ROLL >= DEFENSE { result: d8 * 3 }
 result: d8
}

function: lpistol ROLL:n vs DEFENSE:n {
 if ROLL >= DEFENSE { result: d6 * 2 }
 result: d6
}

loop TTN over {6..8} {
output [hpistol d6 vs TTN] named "H. Pistol Target Number [TTN]"
output [lpistol d8 vs TTN] named "L. Pistol Target Number [TTN]"
}

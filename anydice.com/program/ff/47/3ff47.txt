function: pitch P:n {
 if P=1 {result: 2d6} \fastball\
 if P=2 {result: 2d4} \sinker\
 if P=3 {result: 1d6+1d4} \cutter\
 if P=4 {result: [absolute 1d6-1d4]+1d6} \curveball\
 if P=5 {result: [absolute 1d6-1d4+1d6]} \slider\
 if P=6 {result: [absolute 1d6-1d4]+1d4} \changeup\
 if P=7 {result: [absolute 1d6-1d4+1d4]} \splitter\
  result: 0
} 

function: swing S:n {
 if S=1 {result: 2@4d6+2} \regular swing\
 if S=2 {result: 2@3d6+2} \two-strike swing\
  result: 0 \take\
}

function: zone Z:n {
 if Z<5 | Z>8 {result: 0}
  result: 1
}

function: ball or strike P:n SW:n Z:n R:n {
 if P > SW-R & P < SW+R {result: 1} \ball in play, reset count\
 if SW = 0 & Z = 0 {
  if B < 3 {result: 1} \ball\
  if B = 3 {result: 4} \walk, reset count\
 }
 if S < 2 {
  if SW = 0 & Z = 1 {result: 2} \called strike\
  if SW !=0 {
   if P < SW-R | P > SW+R {result: 2} \swing & miss\
   if P = SW-R | P = SW+R {result: 2} \foul (strike), less than 2 strikes\
  }
 }
 if S = 2 {
  if SW = 0 & Z = 1 {result: 3} \strikeout looking\
  if P < SW-R | P > SW+R {result: 3} \strikeout swinging\
  if P = SW-R | P = SW+R {result: 5} \foul\
  }
}

B: 1
S: 1
O: 0

X: 1 \pitch selection\
Y: d{0,1} \swing selection\
R: 3 \swing contact range\
PITCH: [pitch X]
ZONE: [zone PITCH]
SWING: [swing Y]

output PITCH named "Pitch"
output SWING named "Swing"
output [ball or strike PITCH ZONE SWING R] named "Ball or Strike"

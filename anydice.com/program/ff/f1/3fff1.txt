

function: range on DIE:n between LOW:n and HIGH:n {
  if DIE >= LOW & DIE <= HIGH { result: 1 }
  result: 0
}

function: print pbta DICE:n {
  if DICE <= 6  { result: 0 }
  if DICE >= 10 { result: 2 }
  result: 1
}



loop N over {-2..3}{
 output [print pbta ([lowest 2 of 3d6]+N)] named "[N] 3d6 disadvantage"
 output [print pbta (2d6+N)] named "[N] 2d6 normal"
 output [print pbta ([highest 2 of 3d6]+N)] named "[N] 3d6 advantage"
}

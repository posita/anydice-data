function: roll ROLL:n ATTACK_BONUS:n DAMAGE:d CRITICAL:d AC:n {
  if ROLL = 20 {
    result: CRITCAL
  }
  if ROLL + ATTACK_BONUS >= AC { 
    result: DAMAGE
  }
  result: 0
}


function: regularattack {
  ATTACK_ROLL: d20
  ATTACK: PROF + DEX + ARCHERY
  DAMAGE: 1d8 + DEX
  CRITCAL: 2d8 + DEX
  result: [roll ATTACK_ROLL ATTACK DAMAGE CRITCAL AC]
}

function: sharpshooter {
  ATTACK_ROLL: d20
  ATTACK: PROF + DEX + ARCHERY - 5
  DAMAGE: 1d8 + DEX + 10
  CRITCAL: 2d8 + DEX + 10
  result: [roll ATTACK_ROLL ATTACK DAMAGE CRITCAL AC]
}

function: sharpshooteradvantage {
  ATTACK_ROLL: [highest of d20 and d20]

  DAMAGE: 1d8 + DEX + 10
  CRITCAL: 2d8 + DEX + 10
  result: [roll ATTACK_ROLL ATTACK DAMAGE CRITCAL AC]
}

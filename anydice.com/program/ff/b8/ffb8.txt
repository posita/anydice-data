function: crit D:n {
 if D = 6 { result: D + [crit 1d6] }
 result: D
}

function: roll P:n N:n {
 POS: P
 NEG: N
 if POS = 6 { POS: POS + [crit 1d6] }
 if NEG = 6 { NEG: NEG + [crit 1d6] }
 result: POS - NEG
}

set "maximum function depth" to 3
output [roll 1d6 1d6] named "feng shui"

function: tcount VALUES:s in SEQUENCE:s {
 COUNT: 0
 if SEQUENCE > 1 {
  if SEQUENCE = 20 {
   COUNT: COUNT + 3
  }
  else {
   loop P over {1..#VALUES} {
    COUNT: COUNT + (P@VALUES = SEQUENCE)
    COUNT: COUNT + (P@VALUES = SEQUENCE + 5)
    COUNT: COUNT + (P@VALUES = SEQUENCE - 5)
   }
  }
 }
 result: COUNT
}
output [tcount {18..25} in 1d20]
output [count {18..20} in 1d20] + [count {18..20} in 1d20] + [count {18..20} in 1d20]
output [count {18..20} in [lowest 1 of 2d20]] + [count {18..20} in 1d20] + [count {18..20} in [highest 1 of 2d20]]

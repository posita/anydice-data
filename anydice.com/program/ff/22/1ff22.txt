function: clip A:n from B:n to C:n {
  result: [lowest of [highest of A and B] and C]
}

function: get score {
  result: [regulate {1..3}@4d6]
}
function: regulate X:n {
  if (X > 16) {result: 16}
  if (X < 7) {result: 7}
  result: X
}
SD: [get score]
output SD named "Reroll both"

function: get points X:n {
  if (X > 15) {result: 3*X-35}
  if (X > 13) {result: 2*X-20}
  result: X-7
}
PD: [get points SD]
output PD named "Points"

TD: PD+PD+PD+PD+PD+PD
output TD named "Totals"

CD: [clip TD from 30 to 36]
output CD named "Clipped"

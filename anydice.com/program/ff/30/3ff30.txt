function: compare R:n to X:n and Y:n {
 H: -1
 if R >= X {
  H: H+1
 }
 if R >= Y {
  H: H+1
 }
 result : H
}

function: highest NUMBER:n of DICE:d plus PLUS:n {
 result: ({1..NUMBER}@DICE)+PLUS
}

function: lowest NUMBER:n of DICE:d plus PLUS:n {
 result: {(#DICE - NUMBER + 1)..#DICE}@DICE+PLUS
}

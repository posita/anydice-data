function: tcount VALUES:s in SEQUENCE:s {
 COUNT: 0
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE)
 }
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE + 5)
 }
 loop P over {1..#VALUES} {
  COUNT: COUNT + (P@VALUES = SEQUENCE - 5)
 }
 result: COUNT
}
output [tcount {20..25} in 1d20]
output [count {20..25} in 1d20+5] + [count {20..25} in 1d20] + [count {20..25} in 1d20-5]
output [count {20..20} in 1d20] + [count {20..20} in 1d20] + [count {20..20} in 1d20]
output [count {20..20} in [lowest 1 of 2d20]] + [count {20..20} in 1d20] + [count {20..20} in [highest 1 of 2d20]]

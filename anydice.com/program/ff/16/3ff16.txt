function: adv ROLL:s v TN {
 A: 1@ROLL <= TN 
 B: 2@ROLL <= TN
 R: A + B
 if (1@ROLL = 2@ROLL) {
  if(R) {
   R:R+1 
  } else {
   R:-1
  }
 }
 result:R
}

function: dis ROLL:s v TN {
 X: #ROLL
 Y: #ROLL-1
 A: X@ROLL <= TN 
 B: Y@ROLL <= TN 
 R: A + B
 if (X@ROLL = Y@ROLL) {
  if(R) {
   R:R+1 
  } else {
   R:-1
  }
 }
 result:R
}


loop N over {1..9} {
 output [adv 4d10 v N]
}

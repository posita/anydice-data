function: pool result DICE { 
    PLUS: [count 6 in DICE]
    SUB: [count 1 in DICE]
    result: PLUS - SUB
}
function: pbta result DICE { 
    DOUBLES: 1@DICE = 2@DICE
    LOW: DICE = 2
    HIGH: DICE = 12
    if LOW {
        result: -4
    }
    if HIGH {
        result: +4
    }
    if DICE = 7 | DOUBLES {
        result: 0
    }
    if DICE > 2 & DICE < 7 {
        result: -2
    }
    if DICE > 7 & DICE < 12 {
        result: +2
    }
}
ROLL: 2d6
output [pbta result 2d6]
output [pool result 8d6]
output [count {4} in ROLL]
output [count {1,6} in ROLL]
output ROLL < 5 & ROLL > 2
output ROLL = 7

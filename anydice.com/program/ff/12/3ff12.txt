function: pitch P:n {
 if P=1 {result: 2d6}
 if P=2 {result: 2d4}
 if P=3 {result: 1d6+1d4}
 if P=4 {result: [absolute 1d6-1d4]+1d6}
 if P=5 {result: [absolute 1d6-1d4+1d6]}
 if P=6 {result: [absolute 1d6-1d4]+1d4}
 if P=7 {result: [absolute 1d6-1d4+1d4]}
  result: 0
} 

function: zone Z:n {
 if Z<5 | Z>8 {result: 0}
 result: 1
}

function: swing SWING:n {
 if SWING=1 {result: 2@4d6+2}
 if SWING=2 {result: 2@3d6+2}
  result: 0
}

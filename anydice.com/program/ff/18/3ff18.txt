function: pitch P:n {
 if P=1 {result: 2d6} \fastball\
 if P=2 {result: 2d4} \sinker\
 if P=3 {result: 1d6+1d4} \cutter\
 if P=4 {result: [absolute 1d6-1d4]+1d6} \curveball\
 if P=5 {result: [absolute 1d6-1d4+1d6]} \slider\
 if P=6 {result: [absolute 1d6-1d4]+1d4} \changeup\
 if P=7 {result: [absolute 1d6-1d4+1d4]} \splitter\
  result: 0
} 

function: zone Z:n {
 if Z<5 | Z>8 {result: 0}
 result: 1
}

function: swing SWING:n {
 if SWING=1 {result: 2@4d6+2}
 if SWING=2 {result: 2@3d6+2}
  result: 0
}

B: 0
S: 0
O: 0
function: ball Z:n SW:n {
 if Z=0 & SW=0 {result: B+1}
 result: B
}

X: 1 \pitch selection\
Y: 1 \swing selection\
PITCH: [pitch X]
ZONE: [zone PITCH]
SWING: [swing Y]

output {PITCH, ZONE, SWING}
output [ball ZONE SWING]

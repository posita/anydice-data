function: attack ROLL:n plus BONUS:n vs AC:n for DMG_BASE:n plus DMG_EX:d crit CRIT_DMG:n on CRIT_RANGE:n conf CRIT_CONFIRM:n {
    if ROLL = 1 | (ROLL != 20 & ROLL + BONUS < AC ) {
        result: 0
    } else {
        if (DMG_BASE < 1) {
            DMG_BASE: 1
        }
        if (CRIT_DMG < 1) {
            CRIT_DMG: 1
        }
        if (ROLL >= CRIT_RANGE) & (CRIT_CONFIRM + BONUS >= AC)  {
            result: DMG_BASE + DMG_EX + CRIT_DMG
        } else {
            result: DMG_BASE + DMG_EX
        }
    }
    result: -5
}

output
[attack 1d20 plus 14 vs 20 for 1d10 - 100 plus 3d6 crit 1d10 - 100  on 18 conf 1d20]

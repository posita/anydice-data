\**** Damage calculator ****\

function: attackroll ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE }
 result: 0
}
\looped against enemy AC from 13-20\
ENEMYAC: d{13..20}


ATTACKMOD: 5
DMGBONUS: 2
PROFICIENCY: 4
ACCBONUS: 2
CRITROLL: 20
ATTACKTIMES: 2
DAMAGE: 1d10+1d8+ ATTACKMOD + DMGBONUS
CRITDAMAGE: 2d10+2d8+ ATTACKMOD + DMGBONUS
output [attackroll [highest 1 of 2d20] vs ENEMYAC] named "pally lv 11 PaM and ADV"

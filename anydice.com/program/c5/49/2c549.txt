WEAPON_HITMOD: 9
AC: 15
DMG: 1d12+5

function: advantage attack at HITMOD {
  result: [attack at HITMOD] + [attack at HITMOD] != 0
}

function: attack at HITMOD {
  result: d20+WEAPON_HITMOD + HITMOD >= AC
}

output  DMG       * [attack at 0]               named "Basic attack"
output  DMG       * [advantage attack at 0]     named "Advantaged Basic attack"

output (DMG+10)   * [attack at -5]              named "GWM attack"
output (DMG+10)   * [advantage attack at -5]    named "Feint/Reckless GWM attack"
output (DMG+10)/2 * [advantage attack at -5]    named "Avged True Strike + GWM attack"

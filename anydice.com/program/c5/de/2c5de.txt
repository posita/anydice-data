Fighting styles
* Dueling
When you are wielding a melee weapon in one hand and no other weapons, you gain a +2 bonus to damage rolls with that weapon.
* Great Weapon Fighting
When you roll a 1 or 2 on a damage die for an Attack you make with a melee weapon that you are wielding with two hands, you can re-roll the die and must use the new roll, even if the new roll is a 1 or a 2.

HEX_CRIT_THRESHOLD: 19

EA_ROLL: [highest 1 of 3d20] \ELVEN_ADVANTAGE\

SPELL_SAVE: 14
SPELL_DMG: 1d6
function: R:n halfsavespellattack{
    if R < SPELL_SAVE {result: SPELL_DMG}
    if R > SPELL_SAVE {result: SPELL_DMG / 2}
}


function: R:n savespellattack{
    
    if R < SPELL_SAVE {result: SPELL_DMG}
    if R > SPELL_SAVE {result: 0}
}








function: R:n basicattack{
    \if R <= 1 {result: 0}\
    if R >= CRIT_THRESHOLD {result: 2 d DMG_DIE + DMG_MOD}
    if (R + HIT_MOD) >= TARGET_AC {result: 1 d DMG_DIE + DMG_MOD}
    if (R + HIT_MOD) < TARGET_AC {result: 0}
}


function: R:n secondattack{

    \if R <= 1 {result: 0}\
    if R >= CRIT_THRESHOLD {result: 2 d SCND_DMG_DIE + SCND_DMG_MOD}
    if (R + SCND_HIT_MOD) >= TARGET_AC {result: 1 d SCND_DMG_DIE + SCND_DMG_MOD}
    if (R + SCND_HIT_MOD) < TARGET_AC {result: 0}
}


function: R:n sneaksecondattack{

    \if R <= 1 {result: 0}\
    if R >= CRIT_THRESHOLD {result: 2 d SNEAK_ATTACK_DMG_DIE}
    if (R + SCND_HIT_MOD) >= TARGET_AC {result: 1 d SNEAK_ATTACK_DMG_DIE}
    if (R + SCND_HIT_MOD) < TARGET_AC {result: 0}
}


function: R:n gwfattack{
if R <= 1 {result: 0}
    if R >= CRIT_THRESHOLD {result: 4d{{1..2}:2, {3..6}:8} + DMG_MOD}
    if (R + HIT_MOD) >= TARGET_AC {result: 2d{{1..2}:2, {3..6}:8} + DMG_MOD}
    if (R + HIT_MOD) < TARGET_AC {result: 0}
}

function: R:n hexgwfattack{
if R <= 1 {result: 0}
    if R >= HEX_CRIT_THRESHOLD {result: 4d{{1..2}:2, {3..6}:8} + DMG_MOD}
    if (R + HIT_MOD) >= TARGET_AC {result: 2d{{1..2}:2, {3..6}:8} + DMG_MOD}
    if (R + HIT_MOD) < TARGET_AC {result: 0}
}

REGULAR_ROLL: 1d20  \ attack roll: use 1@2d20 for advantage or 2@2d20 for disadvantage \
DMG_MOD: 3 \damage modifier added on each successful hit\
DMG_DIE:  1d6  \ damage rolled on successful hit \
SNEAK_ATTACK_DMG_DIE: 0 \ sneak attack damage on first hit \




function: attack ROLL:n {
  \ return the number of times damage is applied on an attack roll of ROLL \
  \if ROLL <= 1 { result: 0 }\  \natural 1 is always a miss \
  if ROLL >= CRIT_THRESHOLD { result: 2 }  \ critical hit deals damage damage twice \
  if (ROLL + HIT_MOD) >= TARGET_AC { result: 1 }
  if (ROLL + HIT_MOD) < TARGET_AC { result: 0 }
}



function: roll FIRSTHIT:n to multiattack N:n times {
  if N <= 0 {
    result: 0
  }
}


function: roll FIRSTHIT:n to smitemulti N:n times {
  if N <= 0 {
    result: 0
  }
}


function: multiattack N:n times {
  ATK_ROLL: ROLL
  result: [roll [attack ATK_ROLL] to multiattack N times]
}



function: smitemulti N:n times {
  ATK_ROLL: ROLL
  result: [roll [attack ATK_ROLL] to smitemulti N times]
}

  ATK_ROLL: EA_ROLL
  result: [roll [attack ATK_ROLL] to smitemulti N times]
}

  ATK_ROLL: ROLL
  result: [roll [attack ATK_ROLL] to smitegwfmulti N times]
}

  ATK_ROLL: EA_ROLL
  result: [roll [attack ATK_ROLL] to smitegwfmulti N times]
}







function: multiattack with advantage N:n times {
  ATK_ROLL: ADVANTAGE_ROLL
  result: [roll [attack ATK_ROLL] to multiattack N times]
}



function: roll FIRSTHIT:n to multiattack N:n times{
  if N <= 0 {
    result: 0
  }
  if N = 1 {
  result: [ATK_ROLL secondattack] + [ATK_ROLL sneaksecondattack]
  }


  if FIRSTHIT > 0 {
    DAMAGE: (FIRSTHIT d (DMG_DIE + SNEAK_ATTACK_DMG_DIE)) + DMG_MOD
    loop I over {2..N} {
      if I=N {DAMAGE: DAMAGE + [ATK_ROLL secondattack] } \ one bonus off hand damage\
      else {DAMAGE: DAMAGE + [ATK_ROLL basicattack] +13}
 
    }
    result: DAMAGE 
  } else {
    result: [roll [attack ATK_ROLL] to multiattack N-1 times]
  }
}



function: roll FIRSTHIT:n to smitemulti N:n times{
  if N <= 0 {
    result: 0
  }
  if FIRSTHIT = 2  {       \ only choosing to smite if a critical \
    DAMAGE: (FIRSTHIT d (DMG_DIE + SMITE_ATTACK_DMG_DIE)) + DMG_MOD
    loop I over {2..N} {
      DAMAGE: DAMAGE + [ATK_ROLL basicattack]
    }
    result: DAMAGE 
  }

   else if FIRSTHIT = 1 {
      DAMAGE: (FIRSTHIT d (DMG_DIE)) + DMG_MOD
    result: DAMAGE + [roll [attack ATK_ROLL] to smitemulti N-1 times]
  } 
  else {
    result: [roll [attack ATK_ROLL] to smitemulti N-1 times]
  }
}


GWF_DMG_DIE: 2d{{1..2}:2, {3..6}:8}

  } 
  else {
    result: [roll [attack ATK_ROLL] to smitegwfmulti N-1 times]
  }
}

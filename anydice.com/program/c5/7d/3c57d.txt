function: counttargets D:s {
  FIRSTTARGET: [count {5} in D] 
  SECONDTARGET: [count {6} in D] 
  REROLLS: [count {5..6} in (SECONDTARGET d6)] 
  RREROLLS: [count {5..6} in (REROLLS d6)] 
  TOTAL: FIRSTTARGET + SECONDTARGET + REROLLS + RREROLLS
  result: [lowest of 2 and TOTAL] 
}



output [counttargets 2d6]

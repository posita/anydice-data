function: counttargets D:s {
  FIRSTTARGET: [count {6,7} in D] 
  SECONDTARGET: [count {8} in D] 
  REROLLS: [count {6..8} in (SECONDTARGET d8)] 
  RREROLLS: [count {6..8} in (REROLLS d8)] 
  TOTAL: FIRSTTARGET + SECONDTARGET + REROLLS + RREROLLS
  result: [lowest of 2 and TOTAL] 
}



output [counttargets 2d8]

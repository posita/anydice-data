function: ironsworn ROLL:n over CHALLENGEA:s andover CHALLENGEB:s{
  X: ROLL
  if X > CHALLENGEA & X > CHALLENGEB { result: 0 }
  if X <= CHALLENGEA & X <= CHALLENGEB { result: 2 }
  result: 1
}
loop N over {4,6,8,10,12} {
output [ironsworn d10 over d4 andover dN] named "Pool d4 & d[N] vs d10"
output [ironsworn d10 over d6 andover dN] named "Pool d6 & d[N] vs d10"
output [ironsworn d10 over d8 andover dN] named "Pool d8 & d[N] vs d10"
output [ironsworn d10 over d10 andover dN] named "Pool d10 & d[N] vs d10"
output [ironsworn d10 over d12 andover dN] named "Pool d12 & d[N] vs d10"
}

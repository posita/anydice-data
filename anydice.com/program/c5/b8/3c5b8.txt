function: counttargets D:s U:n{
  FIRSTTARGET: [count {5,6} in D] 
  SECONDTARGET: [count 6 in D]
  REROLLTARGET: [count {4,6} in D]
  REROLLS: [count {6} in (REROLLTARGET d6)] 
  RREROLLS: [count {6} in (REROLLS d6)] 
  TOTAL: FIRSTTARGET + REROLLS + RREROLLS
  result: [lowest of 2 and TOTAL] 
}

output [counttargets 1d6 1] named "1d6"
output [counttargets 2d6 0] named "2d6"
output [counttargets 7d6 0] named "7d6"

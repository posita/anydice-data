function: our M:s attack{
 ATTACK_STAT: (2@M)@{OPA,OSA,OSA}
 DEFENSE_STAT: (2@M)@{EPD,ESD,EPD}
 ACCURACY: 4@M*(6@OSM-7@ESM+14)@{280:7,315,360,420,504,630,840,1120,1400,1680,1960,2240,2520:7}/840
 TYPE: (5@M)@{ENEMY_TYPE_A}*(5@M)@{ENEMY_TYPE_B}*((5@M=1@{OUR_TYPE_A})+(5@M=1@{OUR_TYPE_B})+2)
 if 2@M > 0 {
  result:(3@M\Power\ *ATTACK_STAT *42 /(DEFENSE_STAT *50)+2) *\RNG\1d{85..100}/100 *TYPE/8 *1d{1:ACCURACY,0:100-ACCURACY}
 }
}
function: enemy M:s attack{
 ATTACK_STAT: (2@M)@{EPA,ESA,ESA}
 DEFENSE_STAT: (2@M)@{OPD,OSD,OPD}
 ACCURACY: 4@M*(6@ESM-7@OSM+14)@{280:7,315,360,420,504,630,840,1120,1400,1680,1960,2240,2520:7}/840
 TYPE: (5@M)@{OUR_TYPE_A}*(5@M)@{OUR_TYPE_B}*((5@M=1@{ENEMY_TYPE_A})+(5@M=1@{ENEMY_TYPE_B})+2)
 if 2@M > 0 {
  result:(3@M\Power\ *ATTACK_STAT *42 /(DEFENSE_STAT *50)+2) *\RNG\1d{85..100}/100 *TYPE/8 *1d{1:ACCURACY,0:100-ACCURACY}
 }
}

function: round N move{
 if N = 1 {result:OMA}
 if N = 2 {result:OMB}
 if N = 3 {result:OMC}
 if N = 4 {result:OMD}
 if N = 5 {result:OME}
 if N = 6 {result:OMF}
 if N = 7 {result:OMG}
 if N>= 8 {result:OMH}
}
function: round N counter{
 if N = 1 {result:EMA}
 if N = 2 {result:EMB}
 if N = 3 {result:EMC}
 if N = 4 {result:EMD}
 if N = 5 {result:EME}
 if N = 6 {result:EMF}
 if N = 7 {result:EMG}
 if N>= 8 {result:EMH}
}


loop N over {1..8}{
 OUR_MOVE: [round N move]
 ENEMY_MOVE: [round N counter]
 ORDER: [highest of (1@{ENEMY_MOVE}>1@{OUR_MOVE})|(ESP>OSP & 1@{ENEMY_MOVE}=1@{OUR_MOVE}) and (ESP=OSP)*2]
 OCHP: OCHP - [enemy ENEMY_MOVE attack]*([maximum of ECHP]>=0)*([maximum of OCHP]>=0)*(ORDER=1)
 ECHP: ECHP - [our OUR_MOVE attack]*([maximum of OCHP]>=0)*([maximum of ECHP]>=0)
 WIN: WIN + (ECHP <=0)*(OCHP >0)
 OCHP: OCHP - [enemy ENEMY_MOVE attack]*([maximum of ECHP]>=0)*([maximum of OCHP]>=0)*(ORDER=0)
 if [maximum of ECHP]<=0 & 1@{OCHP}>0 & END=0 {
  output OCHP>0 named "Guaranteed Turn [N] Enemy KO, Final Win Chance(1)"
  END:1
 }
 if [maximum of OCHP]<=0 & END=0 {
  output WIN>0 named "Guaranteed Turn [N] Faint, Final Win Chance(1)"
  END:1
 }
 if 1@{ECHP}<=0 & [maximum of ECHP]>0 & END=0 {
  output 0 < WIN named "Possible Turn [N] Win Chance(1)"
 }
\Diagnostic outputs\\

 output [highest of ECHP and 0] named "Enemy HP turn [N] out of [EHP]"
 output [highest of OCHP and 0] named "Our HP turn [N] out of [OHP]"\
}

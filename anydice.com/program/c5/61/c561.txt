function: ROLL:n vs AC:n {
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK >= AC | ROLL = 20 {
  result: CRITICAL d ALLOW_DAMAGE + DENY_DAMAGE
 }
 result: DAMAGE
}

AC: 5
STRENGTH: 7
ATTACK: 18 + STRENGTH + 1 + 1

ALLOW_DAMAGE: (24d8 + 20 + 1)*2
DENY_DAMAGE: 2d6
DAMAGE: ALLOW_DAMAGE + DENY_DAMAGE
THREAT: 20
CRITICAL: 2
output [d20 vs AC] named "NARUL"

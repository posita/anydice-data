function: gwfb DIE:d {
  result: [reroll DIE as DIE if under 3]
}
function: reroll ROLL:n as DIE:d if under LIMIT:n {
  if ROLL < LIMIT { result: DIE }
  else { result: ROLL }
}

GWF: [gwfb d6]
GWFTEN: [gwfb d10]

function: attack ROLL:n vs DEFENSE:n{
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  result: CRITICALHIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

MISS: 0
HIT: 1
CRITICALHIT: 2

function: damage for ATTACKRESULT:n {
 if ATTACKRESULT = CRITICALHIT {
  result: CRITICALDAMAGE
 }
 if ATTACKRESULT = HIT {
  result: HITDAMAGE
 }
 result: MISSDAMAGE
}

AC: 20
MISSDAMAGE: 0

CHA: 5
PROF: 4

ATTACK: CHA + PROF + 1
HITDAMAGE: 2dGWF + CHA + 1
CRITICALDAMAGE: 4dGWF + CHA + 1
MISSDAMAGE: 0
output 2d[damage for 1d[attack d20 vs AC]] named "[AC] Double Maul"

ATTACK: CHA + PROF + 2
HITDAMAGE: 1dGWFTEN + CHA + 2
CRITICALDAMAGE: 2dGWFTEN + CHA + 2
MISSDAMAGE: 0
output 2d[damage for 1d[attack d20 vs AC]] named "[AC] Double Sunblade"

ATTACK: CHA + PROF + 1
HITDAMAGE: 1d10 + CHA
CRITICALDAMAGE: 2d10 + CHA
MISSDAMAGE: 0
output 3d[damage for 1d[attack d20 vs AC]] named "[AC] Double Blast"

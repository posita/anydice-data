\Eric Spanner damage calculation\
MODIFIER:3
THUNDERMONGER:0
\Info box end\


MOD:MODIFIER
TM:THUNDERMONGER

function: crit DIE:n SIZE:n {
result: (DIE*2)d(SIZE)
}

function: thundercannon CR:n {
if CR=1 {
result:[crit 2 6]
}else {result: 2d6}
}

function: thundermonger CR:n {
if CR=1 {
result:[crit TM 6]
}else {result: TMd6}
}

function: dagger CR:n {
if CR=1 {
result:[crit 1 4]
}else {result: 1d4}
}

function: crossbow CR:n {
if CR=1 {
result:[crit 1 8]
}else {result: 1d8}
}


\outputs\
output [thundercannon 0]+MOD+[thundermonger 0] named "Thunder Cannon + Thunder Monger"
output [thundercannon 0]+MOD+[thundermonger 0]+10 named "Thunder Cannon + Thunder Monger (sharpshooter)"
output [thundercannon 1]+MOD+[thundermonger 1] named "Thunder Cannon + Thunder Monger (crit)"
output [thundercannon 1]+MOD+[thundermonger 1]+10 named "Thunder Cannon + Thunder Monger (crit,sharpshooter)"
output [thundercannon 0]+MOD named "Thunder Cannon"
output [thundercannon 0]+MOD+10 named "Thunder Cannon (sharpshooter)"
output [thundercannon 1]+MOD named "Thunder Cannon (crit)"
output [thundercannon 1]+MOD+10 named "Thunder Cannon (crit,sharpshooter)"
output [thundermonger 0] named "Thunder Monger"
output [thundermonger 1] named "Thunder Monger (crit)"

output [dagger 0]+MOD named "Dagger"
output [dagger 0]+MOD+10 named "Dagger (sharpshooter)"
output [dagger 1]+MOD named "Dagger (crit)"
output [dagger 1]+MOD+10 named "Dagger (crit,sharpshooter)"

output [crossbow 0]+MOD named "Crossbow"
output [crossbow 0]+MOD+10 named "Crossbow (sharpshooter)"
output [crossbow 1]+MOD named "Crossbow (crit)"
output [crossbow 1]+MOD+10 named "Crossbow (crit,sharpshooter)"

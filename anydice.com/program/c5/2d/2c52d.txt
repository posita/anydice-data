function: strS tT svSAVE apAP rapid RAPID shock SHOCK shred SHRED crit CRIT:n saveroll TO_SAVE:s rapidroll RAPID_ROLL:n
{
if RAPID
 {
 if RAPID_ROLL <= 3 {TO_WOUND:d6}
 if RAPID_ROLL > 3 & RAPID < 6 {TO_WOUND:2d6}
 if RAPID_ROLL = 6 {TO_WOUND:3d6}
 }
else {TO_WOUND:d6}

if SHRED>0 & (CRIT=6) {AP:AP-1}
if TO_SAVE >= (SAVE-AP) {result: 0}
if SHOCK>0 & (CRIT=6) {result: [count {1..6} in TO_WOUND]}

if S >= T * 2 {result: [count {2..6} in TO_WOUND]}
if S > T {result: [count {3..6} in TO_WOUND]}
if S = T {result: [count {4..6} in TO_WOUND]}
if S <= T / 2 {result: [count {6} in TO_WOUND]}
if S < T {result: [count {5..6} in TO_WOUND]}
}

output [str3 t3 sv4 ap0 rapid 1 shock 0 shred 0 crit d6 saveroll d6 rapidroll d6] named "+RAPID+ STR3 T3 SV4+ AP-1"

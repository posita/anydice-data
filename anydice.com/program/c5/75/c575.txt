AC: 30
DR: 0
FIRERESIST: 0


DRAKKENTOHIT: 6
DRAKKENDMGBONUS: 3
DAMAGEDEALT:0
loop N over {1..3}{
  DRAKKENATKROLL: 1d20
  DRAKKENCRITCONFIRM: 1d20
  DRAKKENCRIT: DRAKKENATKROLL > 19
  DRAKKENHIT: [lowest of ((DRAKKENATKROLL + DRAKKENTOHIT) >= AC) and 1 - DRAKKENCRIT]
  DRAKKENDMG: [highest of 1d6+DRAKKENDMGBONUS-DR  and 0] + [highest of 2d6-FIRERESIST and 0]
  DRAKKENCRITDMG: DRAKKENDMG
  if [highest of (DRAKKENCRITCONFIRM+DRAKKENTOHIT)>=AC and DRAKKENCRITCONFIRM>19] {
    DRAKKENCRITDMG: [highest of 2*(1d6+DRAKKENDMGBONUS)-DR  and 0] + [highest of 2d6-FIRERESIST and 0]
  }
  DAMAGEDEALT: DAMAGEDEALT + DRAKKENDMG*DRAKKENHIT + DRAKKENCRITDMG*DRAKKENCRIT

}


PALYTOHIT: 9
PALYDMGBONUS: 1

loop N over {1..3}{
  PALYATKROLL: 1d20
  PALYCRITCONFIRM: 1d20
  PALYCRIT: PALYATKROLL > 19
  PALYHIT: [lowest of ((PALYATKROLL + PALYTOHIT) >= AC) and 1 - PALYCRIT]
  PALYDMG: [highest of 1d6+PALYDMGBONUS-DR and 0] + [highest of 2d6-FIRERESIST and 0]
  PALYCRITDMG: PALYDMG
  if [highest of (PALYCRITCONFIRM+PALYTOHIT)>=AC and PALYCRITCONFIRM>19] {
    PALYCRITDMG: [highest of 2*(1d6+PALYDMGBONUS)-DR  and 0] + [highest of 2d6-FIRERESIST and 0]
  }

  DAMAGEDEALT: DAMAGEDEALT + PALYDMG*PALYHIT + PALYCRIT*PALYCRITDMG
}

output DAMAGEDEALT

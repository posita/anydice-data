function: nwod R:n again N:n {
 if N >= R { result: 1 + [nwod R again d10] }
 result: N >= 8
}

NWOD: [nwod 10 again d10]

function: TARGET:n ritual DICE:n NUM:n POOL:d{
 if DICE >= TARGET{result: 1}
 else
 {
 if NUM = 1{result: 0}
 result: [(TARGET-DICE) ritual POOL NUM-1 POOL]
 }

}

output [15 ritual 6dNWOD 6 6dNWOD]

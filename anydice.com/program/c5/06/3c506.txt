function: crop SEQUENCE:n MIN:n MAX:n {                  \ limit the display range between [min;max] \
 if (SEQUENCE < MIN){result: MIN}
 if (SEQUENCE > MAX){result: MAX}
 result: SEQUENCE
}
function: halve SEQUENCE:n {                  \ divide per 2, rounded up \
 DIV: SEQUENCE/2
 TEMP: DIV*2
 if (DIV=TEMP){result: DIV}
 result: (SEQUENCE+1)/2
}

function: phaseone ATT:s DEF:s {
  SIZE_MODS: [compare 11@ATT with 11@DEF]               \compare the nbr of bodies\
  VELOCITY_MODS: [compare 10@DEF with 9@ATT]            \compare the velocity for the dodge capacity\

\freeze the hits-dices & saves-dices sequences\
  ATT_ROLL: (1@ATT +SIZE_MODS) dQ                      \final number of hits-dices to roll\
  DEF_ROLL: (5@DEF +VELOCITY_MODS) dQ                  \final number of saves-dices to roll\

  result: [phasetwo ATT_ROLL DEF_ROLL ATT DEF]
}

function: phasetwo ATT_ROLL:s DEF_ROLL:s  ATT:s DEF:s {
  HITS: [crop (ATT_ROLL+4@ATT) 2@ATT 3@ATT]
  SAVES: [crop (DEF_ROLL+8@DEF) 6@DEF 7@DEF]
  SETS: [find damage bonus ATT_ROLL DEF_ROLL]

  \DAMAGES: HITS - SAVES\
  DAMAGES: [halve HITS] - [halve SAVES]
  result: [crop DAMAGES+SETS 0 4]
}

function: phasetest ATT:s DEF:s {
  SIZE_MODS: [compare 11@ATT with 11@DEF]               \compare the nbr of bodies\
  VELOCITY_MODS: [compare 10@DEF with 9@ATT]            \compare the velocity for the dodge capacity\

  ATT_DICES: (1@ATT +SIZE_MODS)                      \final number of hits-dices to roll\
  DEF_DICES: (5@DEF +VELOCITY_MODS)                  \final number of saves-dices to roll\

  result: [phasetestbis ATT_DICES DEF_DICES ATT DEF]
}

function: phasetestbis ATT_DICES:n DEF_DICES:n ATT:s DEF:s{
  ATT_ROLL: ATT_DICES dQ                           \freeze the hits-dices sequence\
  DEF_ROLL: DEF_DICES dQ                           \freeze the saves-dices sequence\

  result: [phasetesttris ATT_ROLL DEF_ROLL ATT DEF]
}

function: phasetesttris ATT_ROLL:s DEF_ROLL:s ATT:s DEF:s{
  ATT_SET: [find best set ATT_ROLL]
  DEF_SET: [find best set DEF_ROLL]

  SETS: [compare sets ATT_SET DEF_SET]

  \if (ATT_SET>DEF_SET) { SETS: ATT_SET-DEF_SET }
  else{SETS: 0}\

  HITS: [crop (ATT_ROLL+4@ATT) 2@ATT 3@ATT]
  SAVES: [crop (DEF_ROLL+8@DEF) 6@DEF 7@DEF]

  DAMAGES: [halve HITS] - [halve SAVES]
  result: [crop DAMAGES+SETS 0 4]
}

\********Functions to compare Sets of Dices (Optimized at max) **********\

function: find damage bonus ATT:s DEF:s {
 ATT_SET: [find best set ATT]
 DEF_SET: [find best set DEF]
 result: [compare sets ATT_SET DEF_SET]
}

function: find best set ROLL:s {
  \separate the sets in a roll\
  THREES: ROLL=3                     \test the value on each element of the Sequence parameter : optimized\
  TWOS:   ROLL=2
  ONES:   ROLL=1

  \compare the sets to find the more numerous one, choosing the higher if equal\
  if (THREES>0) & (THREES>=TWOS) & (THREES>=ONES) { result: 3}
  if (TWOS>0) & (TWOS>=ONES) { result: 2}
  if (ONES>0) { result: 1}
  result: 0
}

function: compare sets ATT_SET:n DEF_SET:n {
\if Attacker has a higher set's face than Defender, add Damage Bonus\
 if (ATT_SET>DEF_SET) { result: ATT_SET-DEF_SET }
 result: 0
}

function: compare VAL_X:n with VAL_Y:n {
  if(VAL_X > VAL_Y) { result: 1}
  if(VAL_X = VAL_Y) { result: 0}
  result: -1       \VAL_X < VAL_Y\
}

  TWOS:   [count 2 in DICES]
  THREES: [count 3 in DICES]
  SETS: {ZEROS,ONES,TWOS,THREES}
  result: [save DICES with SETS]
}

function: save HITS:n with SETS:s {
  DICES: RD dQ                              \freeze the saves-dices sequence\
  result: [damages DICES with SETS]
}

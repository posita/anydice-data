function: explode specific N:n {
if N = X {
result: N + [explode specific d{0..9}]
}
result: N
}

function: explode doubles ROLL:s {
X: 1@ROLL
if X > 0 & X = 2@ROLL {
result: ROLL + [explode specific d{0..9}]
}
result: ROLL
}

output 2d{0..9} named "normal"
output 2d[explode d{0..9}] named "exploded regular"
output [explode doubles 2d{0..9}] named "exploded doubles"

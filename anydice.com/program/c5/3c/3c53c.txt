\ 0 is a miss, 1 is a weak hit, 2 is a strong hit \

function: action AROLL:n BROLL:n CROLL:s {
  result: (AROLL > CROLL) + (BROLL > CROLL)
}

function: hiaction AROLL:n BROLL:n CROLL:n DROLL:n {
  result: (AROLL > [highest of CROLL and DROLL]) + (BROLL > [highest of CROLL and DROLL])
}


output [action 1d4 1d4 1d6] named "L0/4: 2d4 vs d6"
output [action 1d6 1d6 1d6] named "L1/6: 2d6 vs d6"
output [action 1d6 1d6 1d8] named "L1/6: 2d6 vs d8"
output [action 1d6 1d6 1d10] named "L1/6: 2d6 vs d10"
output [action 1d8 1d8 1d6] named "L2/8: d8 vs d6"
output [action 1d10 1d10 1d6] named "L3/10: 2d10 vs d6"
output [action 1d12 1d12 1d6] named "L4:12 2d12 vs d6"

output [hiaction 1d6 1d6 1d6 1d6] named "L1/6: 2d6 vs highest of 2d6"
output [hiaction 1d8 1d8 1d6 1d6] named "L2/8: 2d8 vs highest of 2d6"

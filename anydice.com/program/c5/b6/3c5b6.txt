function: counttargets D:s {
  FIRSTTARGET: [count {6,7} in D] 
  SECONDTARGET: [count 8 in D] 
  REROLLTARGET: [count 8 in D]
  REROLLS: [count {6,7,8} in (REROLLTARGET d8)]
  TOTAL: FIRSTTARGET + SECONDTARGET + REROLLS 
  result: [lowest of 2 and TOTAL] 
}

output [counttargets 1d8] named "1d8"
output [counttargets 2d8] named "2d8"
output [counttargets 8d8] named "5d8"

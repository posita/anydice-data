function: check success D:n A:n R:n {
  if D+A+R >= 23 { result: 3 }
  if D+A+R >= 17 { result: 2 }
  if D+A+R >= 11 { result: 1 }
  result: 0
}

loop RANK over { -4, -2, 1, 2, 3, 4, 5 } {
  loop ATTRIBUTE over { +3 } {
    output [check success 2d10 ATTRIBUTE RANK] named "2d10+[RANK]+[ATTRIBUTE]"
  }
}

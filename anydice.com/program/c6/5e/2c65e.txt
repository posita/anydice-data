function: plusone ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: CRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + ACCBONUS >= AC { result: DAMAGE  }
 result: 0
}

function: onedfour ROLL:n vs AC:n {
 if ROLL >= CRITROLL { result: ACRITDAMAGE } 
 if ROLL = 1 { result: 0 }
 if ROLL + PROFICIENCY + ATTACKMOD + 1 >= AC { result: ADAMAGE  }
 result: 0
}

ENEMYAC: d{14..25}

ATTACKMOD: 5
PROFICIENCY: 4
CRITROLL: 20
ATTACKTIMES: 2
DAMAGEBONUS: 2
ACCBONUS: 2

DAMAGE: 2d6 + ATTACKMOD + DAMAGEBONUS
CRITDAMAGE: 2d6 + ATTACKMOD + DAMAGEBONUS

ADAMAGE: 2d6 + ATTACKMOD + 1d4 + 1
ACRITDAMAGE: 4d6 + ATTACKMOD + 2d4 + 1

output ATTACKTIMESdDAMAGE named "+2 greatsword on hit"
output ATTACKTIMESd[plusone d20 vs ENEMYAC] named "+2 v 14-25 AC"
output ATTACKTIMESdADAMAGE named "Redcap Greatsword on hit"
output ATTACKTIMESd[onedfour d20 vs ENEMYAC] named "Redcap v 14-25 AC"

DICE: 4
SIDES: 6

function: reroll {
    result: [rerollhelper 0]
}

\ Helper function processes individual NUMBERS (not dice) \
function: rerollhelper REROLLS:n {
    ROLL: d(SIDES)
    \ Pass the numeric result of ROLL to next layer \
    result: [rerollhelperinner REROLLS ROLL]
}

\ Core logic with number parameters \
function: rerollhelperinner REROLLS:n ROLL:n {
    if ROLL = 1 {
        \ Recurse with new reroll count \
        result: [rerollhelper REROLLS + 1]
    } else {
        \ Final result \
        result: ROLL + REROLLS
    }
}

output DICEd[reroll] named "overkill"
output (DICE+1)d[reroll] named "overpower overkill"
output (DICE+1)d(SIDES) named "Overpower"

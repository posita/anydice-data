function: organized chaos ROLL:n DFOUR:n BESTOFTWODFOUR:n BESTOFTWODSIX:n {
  OUT: ROLL
  
  if ROLL=20 { OUT: OUT-DFOUR}
  if ROLL=8 & ROLL=9 { OUT: OUT+BESTOFTWODFOUR }
  if ROLL>=5 & ROLL<=7 { OUT: OUT+BESTOFTWODSIX }
  if ROLL<=4 { OUT: 8 }

  result: OUT
}

function: ten six four ROLL:n DFOUR:n{
  OUT: ROLL
  NEWDFOUR: 1d4
  if ROLL<10 {
    OUT: OUT+DFOUR
    result: [ten six four OUT NEWDFOUR]
  }
  
  result: OUT
}

function: reroll R:n under N:n {
   if R < N { result: 1d12+N } else {result: R}
}

function: range on DIE:n between LOW:n and HIGH:n {

if DIE >= LOW & DIE <= HIGH { result: 1 }

result: 0

}

output [reroll 1d12 under 2] < 6
output [range on [reroll 1d12 under 2] between 6 and 8]
output [range on [reroll 1d12 under 2] between 9 and 12]
output [reroll 1d12 under 2] > 13

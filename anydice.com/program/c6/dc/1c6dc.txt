function: dupes in A:s {
  D: 0
  loop X over {2..#A} {
    if ((X-1)@A = X@A) { result: X@A}
  }
  result: D
}
function: dupes in A:s and B:s {
  D: 0
  C: [sort {A,B}]
  loop X over {2..#C} {
    if ((X-1)@C = X@C) { result: X@C}
  }
  result: D
}
function: dupes in A:s and B:s resist R:n {
  D: 0
  C: [sort {A,B}]
  loop X over {2+R..#C} {
    if ((X-1)@C = X@C) { 
      result: X@C
    }
  }
  result: D
}

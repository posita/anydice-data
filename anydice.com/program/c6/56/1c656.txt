function: check mod M:n skill S:n adv A:n focus F:n {
 \ Permanent advantage \
 if M = 4 {
  M: 3
  A: A + 1
 }
 if M = 5 {
  M: 3
  A: A + 2
 }

 R: [highest (F+2) of (A+F+2)d6]

 \ No crits with focus \
 if F = 0 {
  result: R + M * [crit 2d6 mod M] + S
 } else {
  result: R + M + S
 }
}


function: crit D:s mod M:n {
 if 1@D = 2@D {
  \ Negative mods go to 0 with crit \
  if M <= 0 { result: 0 }
  if 1@D = 6 { result: 3 }
  else { result: 2 }
 }
 else { result: 1 }
}

output [check mod 3 skill 2 adv 0 focus 0]

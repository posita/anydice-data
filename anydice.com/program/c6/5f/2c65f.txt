function: reroll REROLL:s on ROLL:n to REPLACEMENT:d {
    if [REROLL contains ROLL] {
        result: REPLACEMENT
    }
    result: ROLL
}

function: DICE:d reroll REROLL:s {
    result: [reroll REROLL on DICE to DICE]
}

function: FATE:n twist DIE:d {
    if FATE = 1 {
        result: #DIE
    }
    if FATE = 2 {
        result: [lowest 1 of 2dDIE]
    }
    if FATE = 3 {
        result: [highest 1 of 2dDIE]
    }
    if FATE = 4 {
        result: [maximum of DIE]
    }
}

function: twist DIE:d {
    result: [1d4 twist DIE]
}

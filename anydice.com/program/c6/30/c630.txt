function: max A:n B:n {
 if A > B { result: A }
 result: B
}

function: replicate A:n B:d {
 if A > 0 { result: B + [replicate (A-1) B] }
}

function: ar COUNT:n ATTACKVAL:d DMGVAL:d {
  HITS: [replicate COUNT (ATTACKVAL >= 0)]
  DMG: [highest of DMGVAL and 0]
  DMGADJ: (DMG > 0) * (-1d3) + DMG
  result: HITS * DMGADJ
}

function: a COUNT:n ATTACKVAL:d DMGVAL:d {
  result: [replicate COUNT (ATTACKVAL >= 0)] * [highest of DMGVAL and 0]
}

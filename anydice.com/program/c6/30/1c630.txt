PL_DMG: 1d4
PL_SNEAK: 4d6
PL_DEX: 4
PL_PRO: 3
EN_AC: 15

function: hey HIT:n {
	if HIT + PL_DEX + PL_PRO >= EN_AC & HIT = 20 {
		result: (PL_SNEAK + PL_DMG)*2 + PL_DEX
	} else if HIT + PL_DEX + PL_PRO >= EN_AC {
		result: (PL_SNEAK + PL_DMG + PL_DEX)
	} else if HIT + PL_DEX + PL_PRO < EN_AC {
		result: 0
	}
}

output[hey d20] named "D&D 5E: Rogue Sneak Attack"

function: reroll {
    result: [rerollhelper 0]  \ Start with 0 rerolls \
}

\ Helper function with recursion counter \
function: rerollhelper REROLLS {
    ROLL: d6
    if ROLL = 1 {
        \ Reroll and add 1 to the total \
        result: [rerollhelper REROLLS + 1]
    } else {
        \ Return final value + accumulated rerolls \
        result: ROLL + REROLLS
    }
}

output 4d[reroll] named "Total (4d6 reroll 1s)"

function: nwod R:n again N:n{
 if N >= R { result: 1 + [nwod R again d10] }
 result: N >= 8
}

function: nwod rote R:n again N:n 
{
 if N < 8 { result: [nwod R again d10] }
 if N >= R { result: 1 + [nwod R again d10] }
 result: N >= 8
}

function: nwod extended E:n rolls with R:n again N:n dice
{
SUM: 0
loop N over {1..E}{
 SUM: SUM + [lowest of 10 and Nd[nwod R again d10]]
}
result: SUM
}

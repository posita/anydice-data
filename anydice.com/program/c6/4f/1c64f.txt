function: ROLL:n vs AC:n {
 if ROLL = 1 {result: 0}
 if ROLL >= THREAT & ROLL + ATTACK >= AC {
  result: [threat d20]
 }
 if ROLL + ATTACK >= AC | ROLL = 20 { result: DAMAGE }
 result: 0
}

function: threat ROLL:n {
 if ROLL + ATTACK + CONFIRM >= AC | ROLL = 20 {
  result: CRITICAL d DAMAGE
 }
 result: DAMAGE
}

AC: 25
STRENGTH: 5
TWOHANDED: STRENGTH + STRENGTH/2
OFFHAND: STRENGTH/2
DEXTERITY: 0
MODIFIER: 2
ATTACK: 9 + STRENGTH + MODIFIER

DAMAGE: 1d10 + STRENGTH + 11
THREAT: 20
CRITICAL: 3
CONFIRM: 0
output [d20 vs AC] named "Attack vs AC 25"


\-- Deflect Attacks --\

DMG_DICE: 2d8 + 4
SAVE_DC: 14

function: save M{result: d20 + M}

function: save for half ROLL:n{
  if ROLL < SAVE_DC {result: DMG_DICE}
  result: DMG_DICE/2
}

function: save against ROLL:n{
  if ROLL < SAVE_DC {result: DMG_DICE}
  result: 0
}

loop MOD over {-5..SAVE_DC}{
  if MOD < 0 {output [save for half [save MOD]] named "Save [MOD]"}
  else {output [save against [save MOD]] named "Save +[MOD]"}
}

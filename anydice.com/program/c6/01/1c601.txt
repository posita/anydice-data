
function: longbow first attack super advantage multiply crit TOHIT:n CRIT_THRESHOLD:n {
  OUT: -1
  
  DMG_ROLL: 1d8+1d4+6d6
  DMG_FLAT: 5

  if TOHIT >= CRIT_THRESHOLD {
    OUT: DMG_ROLL*2
  } else {
    OUT: DMG_ROLL
  }

  OUT: OUT + DMG_FLAT

  result: OUT
}
function: longbow second attack super advantage multiply crit TOHIT:n CRIT_THRESHOLD:n {
  OUT: -1
  
  DMG_ROLL: 1d8+1d4
  DMG_FLAT: 5

  if TOHIT >= CRIT_THRESHOLD {
    OUT: DMG_ROLL*2
  } else {
    OUT: DMG_ROLL
  }

  OUT: OUT + DMG_FLAT

  result: OUT
}

function: longbow TOHIT:n CRIT_THRESHOLD:n {

  ATK_ONE: [longbow first attack super advantage multiply crit TOHIT CRIT_THRESHOLD]
  ATK_TWO: [longbow second attack super advantage multiply crit TOHIT CRIT_THRESHOLD]

  result: ATK_ONE + ATK_TWO
}

TOHIT: 1@3d20
output [longbow TOHIT 19]

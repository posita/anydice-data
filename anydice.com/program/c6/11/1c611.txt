function: longbow sneak attack TOHIT:n CRIT_THRESHOLD:n {
  OUT: -1
  
  DMG_ROLL: 1d8+1d4+6d6+1d6
  DMG_FLAT: 5

  if TOHIT >= CRIT_THRESHOLD {
    OUT: DMG_ROLL*2
  } else {
    OUT: DMG_ROLL
  }

  OUT: OUT + DMG_FLAT

  result: OUT
}
function: longbow regular attack TOHIT:n CRIT_THRESHOLD:n {
  OUT: -1
  
  DMG_ROLL: 1d8+1d4+1d6
  DMG_FLAT: 5

  if TOHIT >= CRIT_THRESHOLD {
    OUT: DMG_ROLL*2
  } else {
    OUT: DMG_ROLL
  }

  OUT: OUT + DMG_FLAT

  result: OUT
}

function: longbow TOHIT:n CRIT_THRESHOLD:n {

  ATK_ONE: [longbow sneak attack TOHIT CRIT_THRESHOLD]
  ATK_TWO: [longbow regular attack TOHIT CRIT_THRESHOLD]

  result: ATK_ONE + ATK_TWO
}

loop MOD over {0..4} {

\MOD: 4\
DIE: [highest of 1d6+MOD and 1]

A: DIE
B: [highest of A+1 and 2d{DIE}]
C: [highest of B+1 and 3d{DIE}]
D: [highest of C+1 and 4d{DIE}]
E: [highest of D+1 and 5d{DIE}]
\
output A named "1 die/level 1 HP, mod [MOD]"
output 2d{DIE} named "2 dice, mod [MOD]"
output B named "level 2 HP, mod [MOD]"
output 3d{DIE} named "3 dice, mod [MOD]"
output C named "level 3 HP, mod [MOD]"\
output 4d{DIE} named "4 dice, mod [MOD]"
output D named "level 4 HP, mod [MOD]"\
output 5d{DIE} named "5 dice, mod [MOD]"
output E named "level 5 HP, mod [MOD]"\


A: DIE
B: [highest of A+1+[highest of MOD and 0] and 2d{DIE}]
C: [highest of B+1+[highest of MOD and 0] and 3d{DIE}]
D: [highest of C+1+[highest of MOD and 0] and 4d{DIE}]
E: [highest of D+1+[highest of MOD and 0] and 5d{DIE}]
\
output B named "variant level 2 HP, mod [MOD]"
output C named "variant level 3 HP, mod [MOD]"\
output D named "variant level 4 HP, mod [MOD]"\
output E named "variant level 5 HP, mod [MOD]"\
}

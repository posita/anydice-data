MISS: 0
HIT: 1
CRITICAL: 2
MISSDAMAGE: 0

function: attack ROLL:n vs DEFENSE:n {
 if ROLL = 1 { result: MISS }
 if ROLL = 20 {
  if ROLL + ATTACK >= DEFENSE { result: CRITICAL }
  result: HIT
 }
 if ROLL + ATTACK >= DEFENSE { result: HIT }
 result: MISS
}

function: damage for ATTACKRESULTS:s {
 D: 0
 loop A over ATTACKRESULTS {
  if A = CRITICAL {
   D: D + CRITICALDAMAGE
  }
  else if A = HIT {
   D: D + HITDAMAGE
  }
  else {
   D: D + MISSDAMAGE
  }
 }
 result: D + [extra damage for ATTACKRESULTS]
}

function: extra damage for ATTACKRESULTS:s {
 if [ATTACKRESULTS contains CRITICAL] {
  result: [maximum of BONUS]
 }
 if [ATTACKRESULTS contains HIT] {
  result: BONUS
 }
 result: 0
}

function: attack {
 result: [damage for [attack d20 vs DEFENSE]]
}

function: N:n attacks {
 result: [damage for Nd[attack d20 vs DEFENSE]]
}

DEFENSE: d{13..17}

ATTACK: 8
HITDAMAGE: d12
CRITICALDAMAGE: 12
BONUS: d6

output [2 attacks] named "Twin Strike with Quarry"

BONUS: 0

output [2 attacks] named "Twin Strike without Quarry"

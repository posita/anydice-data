BONUS: 1
TOHIT: 9
TOHITT: 4
DAMAGE: 1d6
SNEAK: 4d6

function: hits AC with ADV and TOHIT {
   if ADV { result: (d20+TOHIT >= AC) | (d20+TOHIT >= AC)}
   result: 1d20+TOHIT >= AC
}

function: damage AC with ADV {
  HIT: [hits AC with ADV and TOHIT]
  HITT: [hits AC with ADV and TOHITT]
  result: HIT*HITT*(DAMAGE) + ((HIT+HITT)>=1)*(DAMAGE + SNEAK + 3 + BONUS)
}

loop AC over {12..25} {
    output [damage AC with 0] named "hits [AC]"
}

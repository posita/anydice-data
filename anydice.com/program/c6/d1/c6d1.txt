function: explode X:n {
  Y: 1dX
  result: [exploding X Y]
}

function: exploding X:n Y:n {
  if X = Y {
    result: Y + [explode X]
  }
  else {
    result: Y
  }
}



output [explode 4] named "exploding d4"
output [explode 6] named "exploding d6"
output [explode 8] named "exploding d8"
output [explode 10] named "exploding d10"

ATTACK_MOD: 10
DAMAGE_DICE: d8
DAMAGE_MOD: 5
ATTACK_NUM: 4
CRIT_MOD: 0
AC: 18

function: attack A:n if above B:n {
if A >= B { result: DAMAGE_DICE + DAMAGE_MOD }
 result: 0
}

function: crit A:n {
if A >= 20 { result: DAMAGE_DICE }
 result: 0
}

SUM: 0

loop N over {1..ATTACK_NUM}{
SUM: SUM + [attack d20 + ATTACK_MOD if above AC]
SUM: SUM + [crit d20 + CRIT_MOD]
}

output SUM

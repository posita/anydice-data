function: attack bonus BONUS:n adv ADV:d {
  result: d20 + BONUS + [highest 1 of ADV]
}

function: reroll ROLL:n bonus BONUS:n adv ADV:d {
  if ROLL < 20 {
    result: [attack bonus BONUS adv ADV]
  }
  result: ROLL
}

function: crits bonus BONUS:n adv ADV:d {
  result: [reroll [attack bonus BONUS adv ADV] bonus BONUS adv ADV] >= 20
}

output [crits bonus 2 adv 1d6] named "low LL2"
output [crits bonus 2 adv 2d6] named "med LL2"
output [crits bonus 2 adv 3d6] named "high LL2"
output [crits bonus 3 adv 2d6] named "med LL3-4"
output [crits bonus 4 adv 2d6] named "med LL5-6"
